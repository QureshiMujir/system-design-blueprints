pub struct NodeId(pub NonZeroU64);       // stable for the database's lifetime
pub struct RelId(pub NonZeroU64);        // stable for the database's lifetime
pub struct PropertyId(pub NonZeroU64);   // index into the property store
pub struct LabelId(pub u32);             // index into the label dictionary
pub struct RelTypeId(pub u32);           // index into the rel-type dictionary
pub struct TxnId(pub u64);               // monotonic; Ord for xmin/xmax comparison (I4)
pub struct Lsn(pub u64);                 // monotonic log sequence number (I3, I8)
pub const PAGE_SIZE: usize = 8 * 1024;
pub const PAGE_HEADER_SIZE: usize = 32;

pub struct PageHeader {
    pub magic: u32, pub page_type: u8, pub slot_count: u16,
    pub page_id: u64, pub lsn: Lsn, pub checksum: u64,   // xxhash3-64 over body (I2)
}
pub struct Page<'a> { pub header: &'a PageHeader, pub body: &'a [u8] }
pub struct Snapshot { pub xmin: TxnId, pub xmax: TxnId, pub active: SmallVec<[TxnId; 32]> }

pub fn is_visible(rec_xmin: TxnId, rec_xmax: TxnId, snap: &Snapshot) -> bool;
pub enum Statement { Query(Query), Schema(SchemaCommand) }
pub struct Query { pub clauses: Vec<Clause> }
pub enum Clause {
    Match(MatchClause), OptionalMatch(MatchClause), Where(Expression),
    Create(Vec<PatternPart>), Merge(MergePattern),
    Delete { detach: bool, exprs: Vec<Expression> }, Set(Vec<SetItem>),
    With { items: Vec<ProjectionItem>, distinct: bool, where_: Option<Expression> },
    Return { items: Vec<ProjectionItem>, distinct: bool, order_by: Vec<OrderItem>,
             skip: Option<Expression>, limit: Option<Expression> },
    Unwind { expr: Expression, alias: Identifier }, Call(CallClause), Union { all: bool },
}
pub struct PatternPart { pub start: NodePattern, pub chain: Vec<(RelPattern, NodePattern)> }
pub enum LogicalOp {
    NodeByLabelScan { label: LabelId, alias: VarSlot }, AllNodesScan { alias: VarSlot },
    Expand { input: Box<LogicalOp>, alias: VarSlot, dir: Direction,
             types: SmallVec<[RelTypeId; 4]>, target: VarSlot },
    Filter { input: Box<LogicalOp>, predicate: Expression },
    Project { input: Box<LogicalOp>, items: Vec<ProjectionItem> },
    Sort { input: Box<LogicalOp>, keys: Vec<OrderItem> }, Limit { input: Box<LogicalOp>, count: u64 },
    Apply { left: Box<LogicalOp>, right: Box<LogicalOp> },
    Aggregate { input: Box<LogicalOp>, groups: Vec<Expression>, aggs: Vec<AggExpr> },
    Distinct { input: Box<LogicalOp> }, Union { left: Box<LogicalOp>, right: Box<LogicalOp>, all: bool },
    Create { items: Vec<CreateItem> },
    Merge { pattern: MergePattern, on_create: Vec<SetItem>, on_match: Vec<SetItem> },
    Delete { detach: bool, exprs: Vec<Expression> },
}
pub enum PhysicalOp {
    NodeByLabelScan(NodeByLabelScanOp), AllNodesScan(AllNodesScanOp), ExpandAll(ExpandAllOp),
    ExpandInto(ExpandIntoOp), NodeByIdSeek(NodeByIdSeekOp), Filter(FilterOp),
    Projection(ProjectionOp), Sort(SortOp), Top(TopOp), Limit(LimitOp), Distinct(DistinctOp),
    Apply(ApplyOp), OptionalExpand(OptionalExpandOp), Union(UnionOp),
    EagerAggregation(EagerAggOp), Create(CreateOp), Merge(MergeOp), Delete(DeleteOp),
}
pub trait Operator {
    fn init(&mut self, cx: &mut ExecCx) -> Result<()>;
    fn next(&mut self, cx: &mut ExecCx) -> Result<Option<Row>>;
    fn close(&mut self, cx: &mut ExecCx) -> Result<()>;
    fn estimated_rows(&self) -> u64;
}
pub trait PregelProgram {
    type State: Clone + Default;
    type Message: Clone;
    fn init(&self, vertex: NodeId) -> Self::State;
    fn compute(&self, vertex: NodeId, state: &mut Self::State,
               inbox: &[Self::Message]) -> Vec<(NodeId, Self::Message)>;
    fn combine(&self, target: &mut Self::Message, incoming: Self::Message);
    fn halted(&self, state: &Self::State) -> bool;
}
pub struct BoltSession { pub state: SessionState, pub current_txn: Option<TxnHandle>,
                          pub auth: AuthContext, pub bookmarks: Vec<Bookmark> }
pub enum SessionState {
    Disconnected, Connected, Ready, Streaming, TxReady, TxStreaming, Failed, Interrupted,
}
pub trait Store {
    fn read_node(&self, id: NodeId) -> Option<NodeRecord>;
    fn read_rel(&self, id: RelId) -> Option<RelRecord>;
    fn read_property(&self, id: PropertyId) -> Option<PropertyRecord>;
    fn neighbours_out(&self, node: NodeId) -> NeighbourIter<'_>;   // I1: O(degree) pointer chase
    fn neighbours_in(&self, node: NodeId) -> NeighbourIter<'_>;
    fn create_node(&mut self, txn: TxnHandle, labels: &[LabelId],
                   props: &[(PropertyKeyId, PropertyValue)]) -> Result<NodeId>;
    fn create_rel(&mut self, txn: TxnHandle, kind: RelTypeId, start: NodeId, end: NodeId,
                  props: &[(PropertyKeyId, PropertyValue)]) -> Result<RelId>;
    fn delete_node(&mut self, txn: TxnHandle, id: NodeId, detach: bool) -> Result<()>;
    fn delete_rel(&mut self, txn: TxnHandle, id: RelId) -> Result<()>;
    fn set_property(&mut self, txn: TxnHandle, owner: PropertyOwner,
                     key: PropertyKeyId, value: PropertyValue) -> Result<()>;
    fn snapshot(&self) -> Snapshot;
}
pub trait Index {
    fn label(&self) -> LabelId;
    fn property_key(&self) -> Option<PropertyKeyId>;
    fn lookup_eq(&self, snap: &Snapshot, value: &PropertyValue) -> NodeIdIter<'_>;
    fn lookup_range(&self, snap: &Snapshot, lo: &PropertyValue, hi: &PropertyValue) -> NodeIdIter<'_>;
    fn count(&self, snap: &Snapshot) -> u64;    // the P8 cardinality source
    fn maintain(&mut self, mutation: &IndexMutation) -> Result<()>;
}
pub fn bfs(store: &impl Store, source: NodeId, max_depth: u32) -> BfsIter<'_>;
pub fn dfs(store: &impl Store, source: NodeId, max_depth: u32) -> DfsIter<'_>;
pub fn dijkstra(store: &impl Store, source: NodeId, weight_key: PropertyKeyId) -> DistanceMap;
pub fn a_star(store: &impl Store, source: NodeId, target: NodeId,
              heuristic: impl Fn(NodeId) -> f64) -> Option<Path>;
pub fn page_rank(store: &impl Store, damping: f64, max_iters: u32, tolerance: f64) -> RankVector;
pub fn weakly_connected_components(store: &impl Store) -> ComponentMap;
pub fn louvain(store: &impl Store) -> CommunityAssignment;
pub fn k_shortest_paths(store: &impl Store, source: NodeId, target: NodeId, k: u32) -> Vec<Path>;
pub fn betweenness(store: &impl Store) -> CentralityMap;
pub fn closeness(store: &impl Store) -> CentralityMap;
