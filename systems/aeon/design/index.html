Core schema & types (§11)

/// Ownership: vaultd-owned, immutable after creation.
/// Mutation: none (new VaultId = new vault).
/// Concurrency: Copy, Send, Sync.
/// Persistence: stored in vault manifest header.
/// Serialization: CBOR, 16-byte form.
/// Compatibility: format-version-independent (I14 covers payload, not this).
pub struct VaultId([u8; 16]);

/// Ownership: objstore-owned; derived, never chosen by caller.
/// Mutation: none — content-addressed (I2, P3).
/// Concurrency: Copy, Send, Sync.
/// Persistence: primary key in objstore's redb table (ADR-04).
/// Serialization: CBOR, 32-byte BLAKE3 digest of ciphertext.
/// Compatibility: stable across format versions (hash domain fixed at v1).
pub struct ObjectRef([u8; 32]);

/// Ownership: objstore-owned on disk; keyring-owned in memory pre-persist.
/// Mutation: none post-write (I2).
/// Concurrency: Send, not Sync (owns ciphertext buffer).
/// Persistence: objstore record, byte layout in §8.
/// Serialization: raw bytes per §8 layout table, not CBOR-wrapped.
/// Compatibility: version + algo_id fields gate decode path (P12, I14).
pub struct EncryptedObject { header: ObjectHeader, nonce: [u8; 24], aad: Vec<u8>, ciphertext: Vec<u8>, mac: [u8; 16] }

/// Ownership: keyring-owned, never crosses the RPC boundary (I1).
/// Mutation: replaced wholesale on rotation (I11), never edited in place.
/// Concurrency: !Send, zeroized on drop.
/// Persistence: never persisted raw; only wrapped form persists.
/// Serialization: not serializable (compile-time enforced).
/// Compatibility: tied to DerivationPath's algorithm tag.
pub struct KeyHandle { raw: SecretBytes32, algo: AeadAlgo }

/// Ownership: keyring-owned, deterministic from vault topology.
/// Mutation: append-only (new levels append, existing levels immutable).
/// Concurrency: Copy, Send, Sync.
/// Persistence: derivable, not persisted (recomputed from HKDF info string).
/// Serialization: CBOR, UTF-8 info-string per §8 table.
/// Compatibility: "v1" literal embedded; a "v2" path coexists under P12.
pub struct DerivationPath(String);
/// Ownership: keyring-owned during issuance; user-custodied after export.
/// Mutation: none post-issuance.
/// Concurrency: Send, Sync; zeroized on drop.
/// Persistence: not persisted by keyring — exported to user-chosen custody.
/// Serialization: CBOR + a human-transcribable mnemonic encoding.
/// Compatibility: threshold/index fields versioned per Shamir scheme id.
pub struct RecoveryShare { index: u8, threshold: u8, scheme: u8, payload: SecretBytes }

/// Ownership: capd-owned (issuance), holder-owned (bearer) after issuance.
/// Mutation: none — attenuation creates a new Capability referencing a parent.
/// Concurrency: Send, Sync (no secret material beyond a signature).
/// Persistence: capability log adjacent to objstore; revocation set in capd.
/// Serialization: CBOR, UCAN-compatible wire form (ADR-12).
/// Compatibility: caveats schema versioned independently of vault format.
pub struct Capability { issuer: DidKey, audience: DidKey, exp: u64, nbf: u64, caveats: Vec<Caveat>, proof: Option<Box<Capability>>, sig: Signature }

/// Ownership: embedded in Capability, immutable.
/// Mutation: none.
/// Concurrency: Send, Sync.
/// Persistence: inline in Capability's CBOR form.
/// Serialization: CBOR enum.
/// Compatibility: unknown caveat variants MUST fail verification closed (I13).
pub enum Caveat { ResourceScope(ObjectRef), CollectionScope(String), RateLimit(u32), ExpiresBefore(u64) }

/// Ownership: connectors-owned, authored by the connector developer.
/// Mutation: versioned replacement, never in-place edit of a shipped spec.
/// Concurrency: Send, Sync (plain data).
/// Persistence: TOML manifest, `templates/connector_spec.toml` shape.
/// Serialization: TOML on disk, CBOR over RPC.
/// Compatibility: spec_version field gates the sandbox ABI (ADR-17).
pub struct ConnectorSpec { source_id: String, spec_version: u16, capabilities_required: Vec<Caveat>, rate_limit: u32 }

/// Ownership: connectors-owned, one per (VaultId, source_id) pair.
/// Mutation: monotonic advance only (I8).
/// Concurrency: Send, Sync; single-writer per source.
/// Persistence: redb table keyed by source_id.
/// Serialization: CBOR, opaque cursor token + monotonic sequence number.
/// Compatibility: cursor token format is connector-defined, opaque to host.
pub struct IngestCursor { source_id: String, seq: u64, token: Vec<u8> }
/// Ownership: syncd-owned, one per CRDT/pointer mutation.
/// Mutation: none post-creation — ops are immutable facts (P8).
/// Concurrency: Send, Sync.
/// Persistence: op log, redb table, replicated via relay.
/// Serialization: CBOR, Automerge binary op encoding where applicable.
/// Compatibility: op_kind discriminates CRDT-op vs. pointer-register-op.
pub enum SyncOp { CrdtOp(Vec<u8>), PointerUpdate { name: String, target: ObjectRef, clock: VectorClock } }

/// Ownership: syncd-owned, ephemeral (recomputed per merge).
/// Mutation: n/a — result value.
/// Concurrency: Send, Sync.
/// Persistence: not persisted; drives subsequent objstore/indexer writes.
/// Serialization: not wire-serialized (internal to syncd).
/// Compatibility: n/a.
pub struct MergeResult { applied: u32, conflicts_resolved: u32, tombstones_created: u32 }

/// Ownership: twinadapt-owned, one per inference invocation.
/// Mutation: none post-creation.
/// Concurrency: Send, Sync.
/// Persistence: not persisted beyond the audit entry it generates.
/// Serialization: CBOR over the vaultd–twinadapt boundary.
/// Compatibility: scope must resolve against a live ConsentGrant.
pub struct TwinRequest { scope: Vec<Caveat>, prompt_ref: ObjectRef, backend: InferenceBackend }

/// Ownership: user-issued via capd, structurally a Capability (P6).
/// Mutation: none — revoked, not edited.
/// Concurrency: Send, Sync.
/// Persistence: capability log, alongside other Capability records.
/// Serialization: CBOR, Capability wire form with a twin caveat marker.
/// Compatibility: same versioning as Capability.
pub struct ConsentGrant(Capability);

/// Ownership: auditlog-owned, append-only.
/// Mutation: none — hash-chained (I10).
/// Concurrency: Send, Sync.
/// Persistence: redb append-only table.
/// Serialization: CBOR, includes prev_hash for chain verification.
/// Compatibility: entry_version field, additive fields only.
pub struct AuditEntry { seq: u64, prev_hash: [u8; 32], actor: DidKey, action: AuditAction, object: Option<ObjectRef>, ts: u64 }

/// Error enums, one per module boundary; all #[non_exhaustive] (P12-friendly).
pub enum VaultError { HashMismatch, FormatVersionUnsupported, NonceReuseDetected, Io(IoErrorKind) }
pub enum CapabilityError { SignatureInvalid, Expired, NotYetValid, Revoked, CaveatUnenforceable }
pub enum ConnectorError { CursorRegression, RateLimited, SandboxTrap, SourceUnavailable }
pub enum SyncError { MergeConflictUnresolved, TombstoneRetentionExceeded, TransportUnavailable }
