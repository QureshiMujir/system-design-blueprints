Core schema & types (§11)

// crates/kee-core/src/clock.rs
struct MonotonicNs(u64);   // owned copy-type; produced only by ClockSource; never persisted
struct WallNs(i128);       // owned copy-type; wire/WAL-persisted as i128 ns-since-epoch
struct Ns(i64);            // signed duration; arithmetic result type, not independently constructed
trait ClockSource: Send + Sync {
    fn monotonic_now(&self) -> MonotonicNs;           // non-decreasing (I4)
    fn wall_now(&self) -> WallNs;                     // may step/jitter
    fn wall_to_monotonic(&self, w: WallNs) -> MonotonicNs;  // boundary-only conversion (P1)
}
// No `From<WallNs> for MonotonicNs` — conversion requires an explicit ClockSource call.
// crates/kee-core/src/deadline.rs
type KeyId = u64;                 // hash of user-visible key bytes; stable across process life
type Generation = u32;            // monotonic per-key counter; wraps at 2^32 (documented, P12)
struct Deadline { key: KeyId, generation: Generation, deadline: MonotonicNs }  // Copy, wheel/heap-resident
struct Handle { key: KeyId, generation: Generation }  // Copy; caller-held for cancel/CAS
// crates/kee-core/src/wheel.rs
const SLOTS_PER_LEVEL: usize = 64; const NUM_LEVELS: usize = 6; const TICK_NS: u64 = 1_000_000;
struct Slot { head: Option<ArenaIdx> }                 // intrusive list head, arena-pooled (I11)
struct Level { slots: [Slot; SLOTS_PER_LEVEL], cursor: u32, tick_shift: u32 }
struct HierarchicalWheel { levels: [Level; NUM_LEVELS], epoch: MonotonicNs, now: MonotonicNs, arena: NodeArena }
// insert/cancel/advance_to: signatures only — see §8.1/§16 for the algorithm, not code.
trait DeadlineQueue: Send {
    fn insert(&mut self, d: Deadline) -> Result<(), InsertError>;
    fn cancel(&mut self, h: Handle) -> bool;
    fn advance_to(&mut self, t: MonotonicNs) -> Vec<Deadline>;  // fired, deadline-ascending order
    fn outstanding(&self) -> usize;
}
// crates/kee-core/src/heap.rs
struct HeapEntry { deadline: MonotonicNs, key: KeyId, generation: Generation }  // Ord by deadline, tie-break (key, generation)
struct HeapTop { heap: BinaryHeap<Reverse<HeapEntry>>, top_horizon: Ns, capacity: usize }  // capacity default 256 (ADR-04)
// crates/kee-core/src/engine.rs
enum DeadlineSpec { Relative(Ns), AbsoluteWall(WallNs), AbsoluteMonotonic(MonotonicNs) }
enum ExistenceFlag { None, Nx, Xx, Gt, Lt }             // mirrors Redis flags (P9)
enum Durability { BestEffort, DurableLog }
struct ExpireOptions { deadline: DeadlineSpec, if_: ExistenceFlag, durability: Durability, generation: Option<Generation> }
struct KeyState { generation: Generation, deadline: MonotonicNs }   // per-shard map value
struct Engine {
    wheel: HierarchicalWheel, heap_top: HeapTop, clock: Arc<dyn ClockSource>,
    sinks: Vec<Box<dyn Sink>>, wal: Option<Wal>, policy: ActiveExpirePolicy,
    shards: [DashMap<KeyId, KeyState>; SHARD_COUNT],   // SHARD_COUNT = 256 (P11)
}
// submit/ttl/persist/force_expire_pass/run_forever: signatures only — see §19 for control flow narrative.
// crates/kee-core/src/notify.rs
enum NotificationKind { Expired, Persisted, Cancelled }
struct Notification { key: KeyId, generation: Generation, fired_at: MonotonicNs, kind: NotificationKind }
enum SinkOutcome { Delivered, Enqueued, Dropped, Failed(SinkError) }
trait Sink: Send + Sync { fn deliver(&self, n: &Notification) -> SinkOutcome; fn name(&self) -> &str; }
// Conformant iff deliver() returns within MAX_SINK_DELIVER_DURATION (I13).
// crates/kee-wal/src/log.rs, record.rs
enum WalRecord {
    Insert   { key: KeyId, generation: Generation, deadline_wall_ns: i128 },
    Persist  { key: KeyId, generation: Generation },
    Tombstone{ key: KeyId, generation: Generation, fired_at_wall_ns: i128 },
    Snapshot { lsn: u64 },
}
enum FsyncPolicy { Always, EveryNRecords(u32), EveryDuration(Duration), Never }
struct Wal { inner: Mutex<WalInner>, fsync: FsyncPolicy }  // append persists before submit() acks (I3)
// crates/kee-replication/src/hlc.rs
struct Hlc { wall_ms: u64, logical: u32, node_id: u32 }   // Ord, Copy; strictly increasing per node (I9)
// now()/merge(): signatures only — algorithm is Kulkarni et al. 2014, applied verbatim (§23.2).
