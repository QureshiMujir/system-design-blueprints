Core schema & types (§11)

// nf-core — packet & pipeline (signatures only, no bodies)
pub struct Packet<'a> { pub raw: &'a mut [u8], pub l2_offset: usize, pub l3_offset: usize,
    pub l4_offset: usize, pub l5_offset: usize, pub direction: Direction,
    pub iface_in: Option<IfaceId>, pub iface_out: Option<IfaceId>, pub mark: u32, pub tos: u8 }
pub enum Direction { Original, Reply }
#[derive(Clone, Copy, PartialEq, Eq, Hash)]
pub struct FiveTuple { pub src_ip: IpAddr, pub dst_ip: IpAddr, pub src_port: u16,
    pub dst_port: u16, pub proto: u8 }
impl FiveTuple { pub fn reverse(&self) -> Self; }
pub enum Verdict { Continue, Accept, Drop, Reject(RejectReason), Stolen, Queue(u16) }
pub struct PacketPath { pub prerouting: Vec<RuleSet>, pub input: Vec<RuleSet>,
    pub forward: Vec<RuleSet>, pub output: Vec<RuleSet>, pub postrouting: Vec<RuleSet> }
impl PacketPath { pub fn process(&self, pkt: Packet<'_>) -> Verdict; }
// nf-conntrack — flow state (I1, I4, I5, I9, I14)
pub enum ConntrackState { New, Established, Related, Invalid }
pub enum TcpState { SynSent, SynReceived, Established, FinWait, CloseWait, LastAck, TimeWait, Close }
pub enum ProtoState { Tcp(TcpState), Udp, Icmp(IcmpType), Other }
pub struct ConntrackEntry { pub original: FiveTuple, pub reply: FiveTuple,
    pub state: ConntrackState, pub proto_state: ProtoState, pub last_seen: Instant,
    pub timeout: Duration, pub mark: u32, pub zone: u16, pub helper: Option<Arc<dyn Helper>>,
    pub refs: AtomicU32 }
pub struct Slot { pub next: AtomicPtr<Slot>, pub key: FiveTuple, pub entry: Box<ConntrackEntry> }
pub struct Bucket { pub head: AtomicPtr<Slot>, pub lock: SpinLock }
pub struct Expectation { pub partial_tuple: PartialTuple, pub parent: FiveTuple,
    pub helper: Arc<dyn Helper>, pub timeout: Duration }
pub struct PartialTuple { pub src_ip: Option<IpAddr>, pub dst_ip: IpAddr,
    pub src_port: Option<u16>, pub dst_port: u16, pub proto: u8 }
pub trait ExpectationTable { fn insert(&self, e: Expectation) -> Result<(), CapacityError>;
    fn match_and_consume(&self, pkt: &Packet<'_>) -> Option<Expectation>; }
// nf-rules — policy (P3, P4, I8, I10)
pub struct Rule { pub matches: Vec<Match>, pub action: Action, pub counter: AtomicU64 }
pub enum Match { SrcIp(IpRange), DstIp(IpRange), SrcPort(PortRange), DstPort(PortRange),
    Proto(u8), Mark(u32), ConntrackState(ConntrackState), DpiProtocol(DpiProto),
    Set { name: String, key: SetKey } }
pub enum Action { Accept, Drop, Reject(RejectReason), Snat { source: IpAddr, ports: Option<PortRange> },
    Dnat { dest: IpAddr, port: Option<u16> }, Masquerade, Mark(u32), Jump(ChainName),
    Goto(ChainName), Return, SynProxy, QueueToUserspace(u16) }
pub struct RuleSet { pub generation: u64, pub chains: Vec<Chain> }
// nf-nat — address/port translation (I6, ADR-14, ADR-15)
pub enum AllocationStrategy { RandomWithRetry { max_retries: u8 }, PortBlock { block_size: u16 } }
pub struct NatPool { pub addrs: Vec<IpAddr>, pub ports: PortRange, pub strategy: AllocationStrategy }
pub struct NatMapping { pub internal: FiveTuple, pub external: FiveTuple, pub pool_id: PoolId,
    pub created_at: Instant, pub refs: AtomicU32 }
pub trait PortAllocator { fn allocate(&self, internal: &FiveTuple) -> Result<u16, NatPoolExhausted>;
    fn release(&self, port: u16); }
// nf-algs / nf-ddos — extensions behind stable traits (P7, P14, I11, I12)
pub trait Helper { fn protocol_name(&self) -> &'static str;
    fn on_packet(&self, pkt: &Packet<'_>, entry: &ConntrackEntry) -> HelperAction; }
pub enum HelperAction { NoOp, Rewrite { checksum_valid: bool }, CreateExpectation(Expectation) }
pub struct SynCookie { pub secret_generation: u32, pub mss_index: u8, pub time_bucket: u32 }
pub trait CookieCodec { fn encode(&self, tuple: &FiveTuple, mss: u16) -> u32;
    fn verify(&self, tuple: &FiveTuple, cookie: u32) -> Option<u16>; }
