Core schema & types (§11)

interface Manifest {
  "@context": "https://lineage.dev/v1";
  "@type": "LineageManifest";
  lineage_version: string;          // semver 1.x.y; immutable once issued (I11)
  issuer: string;                   // DID; owner of the signing key
  issued_at: string;                // RFC3339 UTC; validated against I6
  nonce: string;                    // 128-bit base32; unique per issuer (I5)
  ttl_seconds?: number;             // staleness window; consumer policy
  model: ModelDescriptor;
  input_context: InputContext;
  output: OutputDescriptor;
  influence: InfluenceVector[];     // one per OutputSpan; append-only at build time
  training_shards: TrainingShardRef[];
  retrieval: RetrievalRef[];
  system_prompt_fp: string;         // "sha256:" prefix; never the raw prompt (I10)
  kind: "full" | "sampled" | "fallback";  // constrains rendering (I7)
  coverage_score: number;           // 0..1, whole-manifest (I3)
  influence_method: InfluenceMethod;
  tracer_budget: TracerBudget;
  signature: SignedAttestation;     // set only after canonicalisation (P13)
  extensions?: Record<string, unknown>;   // namespaced, e.g. "lab:foo"
}
// Ownership: producer-constructed, immutable post-signature. Mutation: none after signing —
// a changed field invalidates the signature by construction. Concurrency: build is
// single-threaded per output unit. Persistence: not persisted by the verifier by default (§28).
// Serialization: JSON-LD, URDNA2015-canonicalised before signing. Compatibility: I11.
interface ModelDescriptor {
  family: string; version: string; parameter_count: number;
  weight_fingerprint: string;       // BLAKE3 of canonical weights (ADR-11)
  adapter_chain?: AdapterRef[];     // LoRA/PEFT chain, apply order
  inference_config: { temperature: number; top_p: number; seed?: number | null;
                       quantization?: string | null };
}
// Ownership: producer. Mutation: immutable. Persistence: embedded only, never stored keyed
// separately. Serialization: nested JSON-LD object. Compatibility: additive fields only (I11).
interface InputContext {
  prompt_fp: string; prompt_token_count: number; conversation_id?: string;
  retrieval_fps?: string[]; upstream_manifest?: string;  // chain to a prior manifest
}
interface OutputDescriptor {
  output_fp: string; output_token_count: number;
  span_granularity: "token" | "sentence" | "paragraph" | "document";  // PhaseDeferred("§14")
  spans: OutputSpan[];
}
interface OutputSpan { index: number; start_codepoint: number; end_codepoint: number;
                        token_start: number; token_end: number; }
interface InfluenceVector {
  span_index: number; method: InfluenceMethod; top_k: ShardInfluence[];
  coverage_score: number; estimator_variance?: number; zk_proofs?: ZkInfluenceProof[];
}
interface ShardInfluence { shard_id: ShardId; score: number; rank: number;
  attribution: "training" | "retrieval" | "system_prompt" | "model_prior"; }
type InfluenceMethod = "trak" | "datamodels" | "if_lissa" | "if_arnoldi" | "if_ekfac"
                      | "if_grad_dot" | `vendor:${string}`;
// Ownership: Tracer (§14). Mutation: none post-build. Concurrency: one InfluenceVector
// computed per span, spans computed independently (embarrassingly parallel). Persistence:
// embedded in manifest only. Compatibility: InfluenceMethod is an open enum (I11).
interface TrainingShardRef {
  shard_id: ShardId; shard_set: string; byte_range?: [number, number];
  licence: LicenceTag; rights_holder?: string; publisher?: string;
  visibility: "public" | "restricted" | "private"; resolver?: string;
  excerpt_policy: "none" | "snippet" | "full"; snippet?: string;
}
type LicenceTag = "cc-by-4.0" | "cc-by-sa-4.0" | "cc0-1.0" | "public-domain"
                 | "commercial-licensed" | "fair-use-claimed" | "unknown" | `spdx:${string}`;
// Ownership: Tracer emits, Shard Resolver (§18) dereferences. Mutation: `visibility` may
// transition via ShardRevocation (§8.7) without altering the shard_id. Persistence: registry
// row in Postgres (ADR-06), keyed by shard_id. Compatibility: I12 governs resolution authZ.
interface SignedAttestation {
  algorithm: "ed25519" | "ml-dsa-65" | "ml-dsa-87";  // signed material itself, T11
  key_id: string; canonicalisation: "urdna2015"; signature: string;
  cert_chain?: string[]; countersignatures?: SignedAttestation[];
}
interface RetrievalRef { doc_fp: string; source_url?: string; retrieval_score: number;
                          used_in_spans: number[]; licence: LicenceTag; }
interface TracerBudget { wallclock_ms: number; flops?: number; cohort_size?: number;
                          hardware_class?: string; }
// Ownership: Signer (§13). Mutation: none — a `SignedAttestation` mismatch anywhere
// invalidates I2. Concurrency: single signer key per issuer; key rotation is out-of-band
// (§26 T4 mitigation). Compatibility: `algorithm` enum growth is additive only (I11).
class LineageTracer(Protocol):
    """Model-side instrumentation producing InfluenceVectors. Signature only — see §14."""
    lineage_version: str
    def trace(self, *, model_state: ModelStateRef, input_context: InputContext,
              output_span: OutputSpan, budget: TracerBudget) -> InfluenceTrace: ...
    def fingerprint_shard(self, shard: TrainingShard) -> ShardId: ...
    def attest(self, manifest: Manifest, signing_key: Ed25519PrivateKey) -> SignedAttestation: ...
    def supports(self, capability: TracerCapability) -> bool: ...
