Core schema & types (§11)

//! crates/spe-core/src/time.rs — signatures only
pub struct EventTime(pub i64);                      // ms since Unix epoch, allows pre-1970
pub struct Watermark { pub timestamp: EventTime }
pub enum WatermarkStatus { Active, Idle }
//! crates/spe-core/src/element.rs
pub struct Barrier { pub ckpt_id: CheckpointId }

pub enum StreamElement<T> {
    Record { ts: Option<EventTime>, payload: T },
    Watermark(Watermark),
    WatermarkStatus(WatermarkStatus),
    Barrier(Barrier),
    LatencyMarker { source_id: u64, emitted_at: i64 },
}
//! crates/spe-window/src/window.rs
pub enum Window {
    Tumbling { start: EventTime, end: EventTime },
    Sliding  { start: EventTime, end: EventTime },
    Session  { start: EventTime, end: EventTime, key_hash: u64 },
    Global,
}
// fn end(&self) -> EventTime;              // Tumbling/Sliding/Session -> `end`; Global -> MAX
// fn max_timestamp(&self) -> EventTime;    // end() - 1, the window's last valid ts
//! crates/spe-window/src/trigger/mod.rs
pub enum TriggerResult { Continue, Fire, Purge, FireAndPurge }

pub trait Trigger<K, IN>: Send + Sync + 'static {
    fn on_element(&self, element: &IN, ts: EventTime, window: &Window,
                  ctx: &mut TriggerContext<K>) -> TriggerResult;
    fn on_event_time(&self, time: EventTime, window: &Window,
                      ctx: &mut TriggerContext<K>) -> TriggerResult;
    fn on_processing_time(&self, time: i64, window: &Window,
                           ctx: &mut TriggerContext<K>) -> TriggerResult;
    fn on_merge(&self, windows: &[Window], ctx: &mut TriggerContext<K>);
    fn clear(&self, window: &Window, ctx: &mut TriggerContext<K>);
}
//! crates/spe-state/src/keyed/mod.rs
pub trait ValueState<V>: Send {
    fn get(&self) -> Option<V>;
    fn update(&mut self, value: V);
    fn clear(&mut self);
}
pub trait ListState<V>: Send {
    fn get(&self) -> Vec<V>;
    fn add(&mut self, value: V);
    fn clear(&mut self);
}
pub trait MapState<K2, V>: Send {
    fn get(&self, k: &K2) -> Option<V>;
    fn put(&mut self, k: K2, v: V);
    fn remove(&mut self, k: &K2);
    fn iter(&self) -> Box<dyn Iterator<Item = (K2, V)> + '_>;
}
//! crates/spe-core/src/operator/mod.rs
pub trait Operator<IN, OUT>: Send + Sync + 'static {
    fn open(&mut self, ctx: &mut OperatorContext) -> Result<(), Error>;
    fn process_element(&mut self, element: StreamElement<IN>,
                        out: &mut dyn Collector<OUT>) -> Result<(), Error>;
    fn on_watermark(&mut self, wm: Watermark, out: &mut dyn Collector<OUT>) -> Result<(), Error>;
    fn snapshot_state(&mut self, ckpt_id: CheckpointId, out: &mut dyn StateWriter) -> Result<(), Error>;
    fn restore_state(&mut self, ckpt_id: CheckpointId, src: &mut dyn StateReader) -> Result<(), Error>;
    fn close(&mut self) -> Result<(), Error>;
}

pub trait Source<OUT>: Send + Sync + 'static {
    fn open(&mut self, ctx: &mut SourceContext) -> Result<(), Error>;
    fn emit_records_and_watermark(&mut self, out: &mut dyn Collector<OUT>) -> Result<(), Error>;
    fn snapshot_state(&mut self, ckpt_id: CheckpointId, out: &mut dyn StateWriter) -> Result<(), Error>;
    fn restore_state(&mut self, ckpt_id: CheckpointId, src: &mut dyn StateReader) -> Result<(), Error>;
    fn notify_checkpoint_complete(&mut self, ckpt_id: CheckpointId) -> Result<(), Error>;
}

pub trait TwoPhaseCommitSink<IN>: Send + Sync + 'static {
    type Txn: Send;
    fn begin_transaction(&mut self) -> Result<Self::Txn, Error>;
    fn invoke(&mut self, txn: &mut Self::Txn, value: IN) -> Result<(), Error>;
    fn pre_commit(&mut self, txn: Self::Txn) -> Result<PreCommittedTxn<Self::Txn>, Error>;
    fn commit(&mut self, pre: PreCommittedTxn<Self::Txn>) -> Result<(), Error>;
    fn abort(&mut self, txn: Self::Txn) -> Result<(), Error>;
}
//! crates/spe-core/src/graph/job_graph.rs
pub struct JobGraph { pub vertices: Vec<JobVertex>, pub edges: Vec<JobEdge> }
pub struct JobVertex { pub id: VertexId, pub parallelism: u32, pub operator: Box<dyn AnyOperator> }
pub enum Partition {
    Forward,
    Hash { key_fn: Arc<dyn Fn(&[u8]) -> u64 + Send + Sync> },
    Broadcast,
    Custom(Arc<dyn Partitioner + Send + Sync>),
}
//! crates/spe-checkpoint/src/coordinator.rs — fields only, methods elided
pub struct CheckpointCoordinator {
    pending: HashMap<CheckpointId, PendingCheckpoint>,
    completed: BTreeMap<CheckpointId, CompletedCheckpoint>,
    config: CheckpointConfig,
    store: Arc<dyn CheckpointStore>,
}
pub struct CheckpointConfig {
    pub interval: Duration,          // default 60 s (§4)
    pub timeout: Duration,           // default 10 min (§4)
    pub min_pause_between: Duration, // default 5 s
    pub max_concurrent: u32,         // default 1
    pub retention: u32,              // default keep last 3
}
