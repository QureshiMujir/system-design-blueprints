Core schema & types (§11)

// crates/psm-core/src/proto (types module)
pub struct ChannelName(Arc<str>);
impl ChannelName {
    pub fn new<S: Into<Arc<str>>>(s: S) -> Self;
    pub fn as_str(&self) -> &str;
}

pub struct PatternName(Arc<str>);   // Redis-glob only (I5, ADR-16)

#[derive(Copy, Clone, Eq, PartialEq, Hash)]
pub struct ClientId(pub u64);       // nanoid(21)-derived, ADR-19

pub struct NodeId(pub [u8; 16]);    // cluster peer identity
pub enum Frame {
    SimpleString(Bytes), Error(Bytes), Integer(i64), BulkString(Option<Bytes>),
    Array(Vec<Frame>), Push(Vec<Frame>), Null, Double(f64), Boolean(bool),
    Map(Vec<(Frame, Frame)>), Set(Vec<Frame>), BigNumber(Bytes),
    Verbatim { encoding: [u8; 3], data: Bytes },
}

pub enum Command {
    Subscribe(Vec<ChannelName>), Unsubscribe(Vec<ChannelName>),
    PSubscribe(Vec<PatternName>), PUnsubscribe(Vec<PatternName>),
    Publish { channel: ChannelName, payload: Bytes },
    PubsubChannels(Option<PatternName>), PubsubNumsub(Vec<ChannelName>), PubsubNumpat,
    XAdd { stream: Bytes, id: StreamId, fields: Vec<(Bytes, Bytes)>, maxlen: Option<u64> },
    XRead { count: Option<u64>, block_ms: Option<u64>, streams: Vec<(Bytes, StreamId)> },
    XReadGroup { group: Bytes, consumer: Bytes, count: Option<u64>, block_ms: Option<u64>,
                 streams: Vec<(Bytes, StreamId)> },
    XAck { stream: Bytes, group: Bytes, ids: Vec<StreamId> },
    ClusterShards, ClusterSlots, ClusterNodes,
    AclWhoami, AclGetUser(Bytes), Ping(Option<Bytes>), Quit,
    Unknown { name: Bytes, args: Vec<Bytes> },
}

pub enum Reply {
    Ok, Pong(Option<Bytes>), Integer(i64), Bulk(Option<Bytes>),
    Array(Vec<Reply>), Push(Vec<Reply>), Error { kind: &'static str, message: String },
}
pub struct SubscriptionSet {
    channels: HashSet<ChannelName>,
    patterns: HashSet<PatternName>,
}
impl SubscriptionSet {
    pub fn new() -> Self;
    pub fn subscribe(&mut self, c: ChannelName) -> bool;
    pub fn unsubscribe(&mut self, c: &ChannelName) -> bool;
    pub fn psubscribe(&mut self, p: PatternName) -> bool;
    pub fn punsubscribe(&mut self, p: &PatternName) -> bool;
    pub fn total(&self) -> usize;
}
pub struct Subscriber { pub client_id: ClientId, pub buffer: OutputBufferHandle }

pub struct ChannelStore { shards: [RwLock<HashMap<ChannelName, Vec<Subscriber>>>; 256] }
pub struct PatternIndex { inner: RwLock<HashMap<PatternName, Vec<Subscriber>>> }
pub struct FanoutEngine { channel_store: Arc<ChannelStore>, pattern_index: Arc<PatternIndex>,
                           config: FanoutConfig, metrics: Arc<FanoutMetrics> }
pub struct FanoutConfig { hard_limit_bytes: usize, soft_limit_bytes: usize,
                          soft_seconds: u32, batch_size: usize }
pub struct OutputBufferHandle;   // opaque handle over an unbounded mpsc + byte counter (ADR-24)
pub enum DispatchOutcome { Delivered { subscribers: u32 },
                           SlowConsumerDisconnected { client_id: ClientId, bytes_pending: usize },
                           NoSubscribers }
#[derive(Copy, Clone, Eq, PartialEq, Ord, PartialOrd)]
pub struct StreamId { pub ms: u64, pub seq: u64 }          // I7
pub struct StreamEntry { pub id: StreamId, pub fields: Vec<(Bytes, Bytes)> }
pub struct Stream { entries: BTreeMap<StreamId, StreamEntry>, last_id: StreamId,
                     length: u64, max_len: Option<u64>, groups: HashMap<Bytes, ConsumerGroup> }
pub struct ConsumerGroup { pub name: Bytes, pub last_delivered: StreamId,
                           pub consumers: HashMap<Bytes, Consumer>, pub pel: PendingEntryList }
pub struct Consumer { pub name: Bytes, pub seen_time: u64, pub active_time: u64,
                       pub pel: HashMap<StreamId, PendingEntry> }
pub struct PendingEntry { pub id: StreamId, pub consumer: Bytes,
                          pub delivery_time: u64, pub delivery_count: u64 }
pub struct PendingEntryList { inner: BTreeMap<StreamId, PendingEntry> }
pub const SLOTS: u16 = 16384;
pub struct SlotMap { owners: [NodeId; SLOTS as usize] }   // I8
pub struct BusPacket { pub header: BusHeader, pub body: BusBody }
pub struct BusHeader { pub magic: u32, pub version: u16, pub kind: BusKind,
                        pub sender: NodeId, pub epoch: u64, pub flags: u16, pub body_len: u32 }
pub enum BusKind { Ping, Pong, FanoutPublish, FanoutPattern, StreamsAppend, StreamsAck,
                   GossipDelta, SlotMigrationStart, SlotMigrationEnd }
pub enum BusBody { Ping { seq: u64 }, Pong { seq: u64 },
                   FanoutPublish { channel: Bytes, payload: Bytes } /* one variant per BusKind */ }
pub enum BrokerError { Io(std::io::Error), Wire(String), Acl(String),
                        Cluster(String), Streams(String), PhaseDeferred(&'static str) }
