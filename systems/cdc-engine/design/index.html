Core schema & types (§11)

enum Offset {                                   // non-Default, non-Copy (P4 enforcement)
    PostgresLsn { lsn: u64, flushed: u64 },     // owned; immutable once constructed
    MysqlGtidSet(String),                       // owned; monotonic under I9
}
fn Offset::to_bytes(&self) -> Vec<u8>;          // CBOR; used by every offset-store backend
fn Offset::from_bytes(&[u8]) -> Result<Offset, OffsetError>;
struct ChangeEvent {
    source: SourceInfo,       // owned; identifies connector+table
    schema: Arc<Schema>,      // shared, immutable (P2); refcounted across batch
    key: Value,                // owned; the partition key (P9)
    op: Op,                    // owned; the mutation variant
    ts_ms: i64,                 // wall-clock at source commit (ADR-28)
    tx_id: Option<TxnId>,       // owned; None only for Heartbeat
    offset: Offset,              // owned; moved, never cloned implicitly
}
struct SourceInfo { connector: &'static str, db: String, schema: String, table: String, server_id: String }
struct TxnId(pub u64);
enum Op {
    Insert { after: Value },
    Update { before: Option<Value>, after: Value },   // before: Some only if pre-image available
    Delete { before: Option<Value> },
    Truncate,
    SchemaChange { ddl: String, new_schema: Arc<Schema> },  // routed to schema-history only (I7)
    Heartbeat,
}
struct Schema { name: String, fields: Vec<Field>, primary_key: Vec<String>, version: SchemaVersion }
struct Field { name: String, ty: FieldType, nullable: bool, default: Option<Value>, aliases: Vec<String> }
enum FieldType { Bool, Int8, Int16, Int32, Int64, Float32, Float64, String, Bytes,
                 Decimal{precision:u8,scale:u8}, Date, Time, Timestamp{tz:bool}, Json, Uuid,
                 Array(Box<FieldType>), Map(Box<FieldType>) }
enum Value { Null, Bool(bool), I64(i64), F64(f64), Str(String), Bytes(Vec<u8>),
             Record(BTreeMap<String,Value>), Array(Vec<Value>) }
struct SchemaVersion(pub u32);   // monotonically increasing per subject (I6)
trait SourceConnector: Send + Sync + 'static {
    fn start(&self, start: Option<Offset>) -> Result<EventStream, SourceError>;
    fn ack(&self, offset: &Offset) -> Result<(), SourceError>;
    fn connector_kind(&self) -> &'static str;
}
type EventStream = Pin<Box<dyn Stream<Item = Result<ChangeEvent, SourceError>> + Send>>;
trait Sink: Send + Sync + 'static {
    fn commit_batch(&self, batch: Vec<ChangeEvent>) -> Result<SinkAck, SinkError>;  // MUST NOT
                                                                                     // return Ok
                                                                                     // before durable
    fn delivery_guarantee(&self) -> DeliveryGuarantee;
}
struct SinkAck { committed_through: Offset, durable_at: Instant }
enum DeliveryGuarantee { AtLeastOnce, ExactlyOnceIdempotent, ExactlyOnceTransactional }
trait SchemaRegistry: Send + Sync + 'static {
    fn register(&self, subject: &str, schema: &Schema) -> Result<SchemaId, RegistryError>;  // I6
    fn get_by_id(&self, id: SchemaId) -> Result<Schema, RegistryError>;
    fn get_latest(&self, subject: &str) -> Result<(SchemaId, Schema), RegistryError>;
}
struct SchemaId(pub u32);
trait Codec: Send + Sync { fn encode(&self, e: &ChangeEvent, id: SchemaId) -> Vec<u8>;
                            fn decode(&self, buf: &[u8]) -> Result<(ChangeEvent, SchemaId), CodecError>; }
trait TransformStage: Send + Sync { fn apply(&self, e: ChangeEvent) -> Result<Option<ChangeEvent>, TransformError>; }
struct TransformChain(Vec<Box<dyn TransformStage>>);   // ordered, pure (P13)
