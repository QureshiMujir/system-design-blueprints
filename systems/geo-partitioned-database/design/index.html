  key_range: KeyRange

  leaseholder: NodeId
  epoch: u64

  txn_id: TxnId
  begin_ts: HybridLogicalTimestamp
  commit_ts: Option<HybridLogicalTimestamp>
  state: TxnState
  write_set: Vec<PartitionKey>
  coordinator_id: NodeId

  term: u64

  payload: Bytes

  snapshot_id: SnapshotId
  applied_index: u64
  sstable_refs: Vec<SstableRef>
  checksum: Crc32c

  geographic_region: RegionId

  hash_suffix: u32

  key: Bytes
  version_ts: HybridLogicalTimestamp
  txn_id: TxnId
  tombstone: bool
  payload: Bytes
trait StorageEngine:
  fn put(key: Bytes, value: Bytes, ts: HybridLogicalTimestamp) -> Result<(), StorageError>
  fn get(key: Bytes, ts: HybridLogicalTimestamp) -> Result<Option<Bytes>, StorageError>

  fn flush_memtable() -> Result<SnapshotManifest, StorageError>
  fn create_snapshot() -> Result<SnapshotManifest, StorageError>
