Core schema & types (§11)

/// Container for one `{ ... }` body: attributes + nested blocks. Owns its children.
pub struct Body { pub attributes: Vec<Attribute>, pub blocks: Vec<Block>, pub span: Span }

/// A typed, labeled block: `resource "aws_instance" "web" { ... }`.
/// `ty` = block keyword; `labels` = quoted/ident labels; `body` = recursive child.
pub struct Block { pub ty: String, pub labels: Vec<String>, pub body: Body, pub span: Span }

/// `name = <expr>`. Owns the unevaluated expression.
pub struct Attribute { pub name: String, pub expr: Expression, pub span: Span }

/// Unevaluated expression tree. Box on recursive arms keeps the enum small (≤ 48 bytes target).
/// Number is `Decimal` (arbitrary precision, base-10) — no f64, to keep plans byte-deterministic.
pub enum Expression {
    Null, Bool(bool), Number(Decimal), Str(String),
    Tuple(Vec<Expression>), Object(Vec<(Expression, Expression)>),
    Template(Vec<TemplatePart>),                       // "pre-${x}-post" / here-docs
    Variable(String),                                  // bare leaf name
    ScopeTraversal(Vec<Traversal>),                    // a.b[0].c
    FunctionCall { name: String, args: Vec<Expression>, expand_final: bool }, // expand_final = `f(args...)`
    BinaryOp(BinOp, Box<Expression>, Box<Expression>),
    UnaryOp(UnOp, Box<Expression>),
    Conditional { cond: Box<Expression>, t: Box<Expression>, f: Box<Expression> },
    ForExpr(Box<ForExpr>),                             // [for k,v in coll : expr if cond]
    Index { coll: Box<Expression>, key: Box<Expression> },
    Splat { source: Box<Expression>, each: Box<Expression> }, // attr.*.id
}

/// One step of a reference path. LegacyIndex = `a.0` form; Index = `a[expr]` (expr eval'd late).
pub enum Traversal { Attr(String), Index(Expression), LegacyIndex(u64) }

/// One static segment or one `${...}` interpolation hole.
pub enum TemplatePart { Literal(String), Interp(Expression) }

/// Source location: byte range into a named file; drives Diagnostic underlining.
pub struct Span { pub file: FileId, pub start: usize, pub end: usize }

pub struct Diagnostic { pub severity: Severity, pub summary: String, pub detail: String, pub span: Span }
pub enum Severity { Error, Warning, Note }
pub type Diagnostics = Vec<Diagnostic>;   // accumulator; parser/eval never panic, they push
/// The cty-like dynamic value lattice. Total Ord/Eq over canonical encoding (Unknown excepted).
pub enum Value {
    Null, Bool(bool), Number(Decimal), Str(String),
    List(Vec<Value>),                    // ordered, homogeneous-by-unify
    Set(BTreeSet<Value>),                // dedup + sorted → canonical
    Map(BTreeMap<String, Value>),        // string-keyed, sorted
    Object(BTreeMap<String, Value>),     // fixed attr set, sorted keys
    Tuple(Vec<Value>),                   // ordered, heterogeneous
    Unknown(Type),                       // value computed by a provider; carries its known Type
    Sensitive(Box<Value>),              // privacy mark; wraps any of the above; propagates
}
/// Static type, possibly Dynamic (top). Object/Tuple carry exact shape.
pub enum Type { Dynamic, Bool, Number, String,
    List(Box<Type>), Set(Box<Type>), Map(Box<Type>),
    Object(BTreeMap<String, Type>), Tuple(Vec<Type>) }

/// Read-only evaluation scope. Lifetime-borrowed; cheap to derive child scopes (for/each).
pub struct EvalContext<'a> {
    // variables, locals, outputs by name; current each/count scope; function registry; module path.
}

/// A pure, deterministic engine function (string/numeric/collection/encoding/hash/fs).
pub trait Function {
    fn name(&self) -> &str;
    fn call(&self, args: &[Value]) -> Result<Value, EvalError>;     // pure: no IO except declared fs fns
    fn param_types(&self) -> &[Type];                                // for arity/type check + variadic tail
    fn return_type(&self, args: &[Type]) -> Type;                    // drives Unknown result typing
}

pub fn evaluate(expr: &Expression, ctx: &EvalContext) -> Result<Value, EvalError>;
pub fn convert(v: Value, to: &Type) -> Result<Value, EvalError>;     // cty convert: widen/narrow/coerce
pub fn unify(types: &[Type]) -> Option<Type>;                        // least common supertype, None if incompat
pub struct NodeId(u32);                              // dense index into Graph.nodes; Copy
pub struct Edge { pub from: NodeId, pub to: NodeId } // semantic: dependency(from) -> dependent(to)

/// Generic DAG over node payload N. Edges = adjacency, dependency -> dependent.
pub struct Graph<N> { /* nodes: Vec<N>, edges: adjacency lists both directions */ }

impl<N> Graph<N> {
    pub fn add_node(&mut self, n: N) -> NodeId;
    pub fn add_edge(&mut self, from: NodeId, to: NodeId);
    pub fn detect_cycles(&self) -> Result<(), Vec<Vec<NodeId>>>;   // Tarjan SCC; Err = nontrivial SCCs
    pub fn transitive_reduction(&mut self);                         // drop redundant edges; preserves reachability
    /// Reverse-topological (leaves→roots) walk; ≤ parallelism nodes concurrent; errors defer dependents.
    pub fn walk<F>(&self, parallelism: usize, visit: F) -> WalkResult
        where F: Fn(NodeId, &N) -> Result<(), Diagnostics> + Sync;
}

pub struct WalkResult { pub diagnostics: Vec<Diagnostic>, pub visited: usize }
/// The single durable artifact. version = document schema; serial = monotonic write counter;
/// lineage = identity of this state's history (detects "wrong state file").
pub struct State {
    pub version: u32,
    pub serial: u64,
    pub lineage: Uuid,
    pub resources: Vec<ResourceStateBucket>,
    pub outputs: BTreeMap<String, OutputValue>,
}

/// All instances of one resource address (count/for_each expand into many instances).
pub struct ResourceStateBucket {
    pub address: ResourceAddress,
    pub provider: ProviderAddress,
    pub instances: Vec<ResourceInstance>,
}

pub struct ResourceInstance {
    pub index_key: Option<IndexKey>,         // None=single; Some(Int|Str) for count/for_each
    pub schema_version: u64,                 // provider schema this row was written under → migration
    pub attributes: Value,                   // last-known attrs (canonical encoding)
    pub dependencies: Vec<ResourceAddress>,  // recorded for correct destroy order
    pub status: InstanceStatus,
}

pub enum InstanceStatus { Ready, Tainted, Deposed(DeposedKey) }   // Tainted→force replace; Deposed→old copy pending destroy

pub struct ResourceAddress {
    pub module: ModulePath,
    pub mode: ResourceMode,                  // Managed | Data
    pub ty: String,                          // e.g. "aws_instance"
    pub name: String,
}
pub enum ResourceMode { Managed, Data }
pub enum IndexKey { Int(u64), Str(String) }

pub trait StateBackend {
    fn lock(&self, info: &LockInfo) -> Result<LockId, BackendError>;
    fn unlock(&self, id: LockId) -> Result<(), BackendError>;
    fn read(&self) -> Result<Option<State>, BackendError>;        // None = no state yet
    fn write(&self, s: &State) -> Result<(), BackendError>;       // ATOMIC: temp + fsync + rename, prior→.backup
}

pub struct StateLock { /* holder LockInfo, LockId, heartbeat clock */ }
pub struct LockInfo { pub who: String, pub operation: String, pub created: SystemTime }
pub struct LockId(String);
/// Out-of-process provider, driven over RPC (a child process). Methods map to plugin RPCs.
pub trait Provider {
    fn schema(&self) -> &ProviderSchema;
    fn configure(&mut self, config: Value) -> Result<(), Diagnostics>;
    fn validate_resource(&self, ty: &str, config: &Value) -> Diagnostics;
    fn read_resource(&self, ty: &str, prior: &Value) -> Result<Value, Diagnostics>;          // REFRESH
    fn plan_resource(&self, ty: &str, prior: &Value, proposed: &Value)
        -> Result<PlannedChange, Diagnostics>;                                                // diff + computed
    fn apply_resource(&self, ty: &str, prior: &Value, planned: &Value)
        -> Result<Value, Diagnostics>;                                                        // do mutation
    fn import_resource(&self, ty: &str, id: &str) -> Result<Vec<ImportedResource>, Diagnostics>;
}

pub struct ProviderSchema {
    pub provider: Schema,                          // the `provider "x" {}` config block
    pub resources: BTreeMap<String, Schema>,       // managed resource types
    pub data_sources: BTreeMap<String, Schema>,
}

pub struct Schema { pub attributes: BTreeMap<String, Attribute>, pub blocks: BTreeMap<String, NestedBlock> }

pub struct Attribute {
    pub ty: Type,
    pub required: bool, pub optional: bool, pub computed: bool, // computed ⇒ Unknown until apply
    pub sensitive: bool,                                         // ⇒ Value::Sensitive on read
    pub force_new: bool,                                         // changed ⇒ Replace (requires-replace)
}

pub struct NestedBlock { pub schema: Schema, pub nesting: Nesting, pub min: u32, pub max: u32 }
pub enum Nesting { Single, List, Set, Map }

/// Provider's planned next value: Unknown for attrs it will compute; paths forcing Replace.
pub struct PlannedChange { pub planned: Value, pub requires_replace: Vec<Path> }
pub struct Path(Vec<PathStep>);                    // attribute path into a Value
pub struct ImportedResource { pub ty: String, pub id: String, pub state: Value }
/// The minimal mutation for one resource instance. Replace = delete+create, order per create_before_destroy.
pub enum Action {
    NoOp,
    Read,                                              // data source / refresh-only
    Create, Update, Delete,
    Replace { create_before_destroy: bool },
}

/// One line of the plan. before = prior+refreshed; after = desired (Unknown for computed).
pub struct ResourceChange {
    pub address: ResourceAddress,
    pub action: Action,
    pub before: Value,
    pub after: Value,                                  // may contain Unknown
    pub requires_replace: Vec<Path>,
    pub deposed: Option<DeposedKey>,
}

/// The full, saved, byte-deterministic plan. Applyable = no errors + no required re-plan.
pub struct Plan {
    pub changes: Vec<ResourceChange>,
    pub output_changes: Vec<OutputChange>,
    pub prior_state: State,                            // pinned snapshot the plan was built against
    pub variables: BTreeMap<String, Value>,           // pinned inputs → reproducibility
    pub applyable: bool,
}

/// Implemented by every graph node (resource / data / module-call / local / output / provider).
pub trait GraphNode {
    fn address(&self) -> &Addr;
    fn references(&self) -> Vec<Addr>;                 // → edges (dependency extraction, §16)
    fn execute(&self, op: &Op, ctx: &mut WalkCtx) -> Result<(), Diagnostics>;  // op = Refresh|Plan|Apply|Destroy
}
