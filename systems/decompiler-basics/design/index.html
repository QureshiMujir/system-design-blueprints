Core schema & types (§11)

pub enum Type { Unknown, Int{bits:u16, signed:bool}, Ptr(Box<Type>), Float{bits:u16},
                Struct(Vec<(String,Type)>), Func{ret:Box<Type>, args:Vec<Type>}, Void }

pub enum Value { Reg{name:&'static str, ver:u32}, Tmp(u32), StackSlot(i64), Flag{which:FlagId, ver:u32} }
pub enum IrExpr { Val(Value), Const(i128,u16), Load{addr:Box<IrExpr>, size:u16},
                  Bin{op:BinOp, l:Box<IrExpr>, r:Box<IrExpr>}, Un{op:UnOp, e:Box<IrExpr>},
                  Cmp{op:RelOp, l:Box<IrExpr>, r:Box<IrExpr>}, Cast{kind:CastKind, e:Box<IrExpr>, to:Type} }
pub enum IrStmt { Assign(Value, IrExpr), Store{addr:IrExpr, val:IrExpr, size:u16},
                  Branch{cond:IrExpr, t:Label, f:Label}, Jump(Label),
                  Call{target:IrExpr, args:Vec<IrExpr>, ret:Option<Value>}, Ret(Option<IrExpr>),
                  Phi{dst:Value, srcs:Vec<(Value,Label)>}, Nop, Unknown(&'static str) }   // I9

pub struct IrBlock { pub label: Label, pub stmts: Vec<IrStmt>, pub term: IrStmt }
pub struct IrCfg { pub blocks: Vec<IrBlock>, pub entry: Label, pub edges: Vec<(Label,Label)> }

pub fn lift(func:&[Instruction], cfg:&Cfg) -> IrCfg;                 // #334 → IR (I1)
pub struct DomTree { /* idom + dominance frontier */ }
pub fn dominators(cfg:&IrCfg) -> DomTree;                            // Cooper-Harvey-Kennedy
pub fn to_ssa(cfg:&mut IrCfg, dom:&DomTree);                         // φ + rename (I2/I3)
pub trait Analysis { type Fact: Lattice; fn transfer(&self, s:&IrStmt, f:&Self::Fact) -> Self::Fact; }
pub fn solve<A:Analysis>(a:&A, cfg:&IrCfg) -> Facts<A::Fact>;        // worklist fixed-point (I5)
pub fn propagate_and_dce(cfg:&mut IrCfg, facts:&Facts<_>);           // I11 semantics-preserving (I4)
pub fn recover_types(cfg:&IrCfg) -> TypeMap;                         // constraints+solve (I10)
pub enum Ast { Seq(Vec<Ast>), If{c:IrExpr, t:Box<Ast>, e:Option<Box<Ast>>},
               While{c:IrExpr, body:Box<Ast>}, DoWhile{body:Box<Ast>, c:IrExpr},
               Switch{v:IrExpr, cases:Vec<(i128,Ast)>, default:Option<Box<Ast>>},
               Stmt(IrStmt), Goto(Label), Label(Label), Return(Option<IrExpr>) }
pub fn structure(cfg:&IrCfg, dom:&DomTree) -> Ast;                   // Cifuentes (I7/I8)
pub fn emit_c(ast:&Ast, types:&TypeMap, names:&NameMap) -> String;   // deterministic (I6)
