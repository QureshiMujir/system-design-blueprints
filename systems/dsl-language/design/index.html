Core schema & types (§11)

// crates/lume-syntax/src/token.rs
pub struct Span { pub file: FileId, pub start: u32, pub end: u32 }   // byte offsets
pub struct Token { pub kind: TokenKind, pub span: Span, pub leading_trivia: Vec<Trivia>, pub trailing_trivia: Vec<Trivia> }
pub enum TokenKind {
    Int(i128), Float(f64), Str(String), Bool(bool), Null, Ident(String),
    KwLet, KwIn, KwIf, KwThen, KwElse, KwMatch, KwWith, KwImport, KwType, KwAssert, KwForall,
    LBrace, RBrace, LBracket, RBracket, LParen, RParen, Comma, Semi, Colon, Equals, Arrow, FatArrow, Pipe, Dot, At,
    Plus, Minus, Star, Slash, Percent, EqEq, BangEq, Lt, Gt, Le, Ge, AmpAmp, PipePipe, Bang,
    Whitespace, Comment, DocComment, Newline, Eof,
}
// crates/lume-syntax/src/ast.rs
pub struct Module<'a> { pub items: bumpalo::collections::Vec<'a, Item<'a>>, pub span: Span }
pub enum Item<'a> { Let(LetItem<'a>), TypeAlias(TypeAlias<'a>), Import(Import<'a>) }
pub struct LetItem<'a> { pub name: Ident, pub ty: Option<&'a TypeExpr<'a>>, pub value: &'a Expr<'a>, pub span: Span }
pub enum Expr<'a> {
    Lit(Lit, Span), Ident(Ident, Span),
    BinOp(BinOp, &'a Expr<'a>, &'a Expr<'a>, Span), UnOp(UnOp, &'a Expr<'a>, Span),
    If(&'a Expr<'a>, &'a Expr<'a>, &'a Expr<'a>, Span),
    Let(Ident, Option<&'a TypeExpr<'a>>, &'a Expr<'a>, &'a Expr<'a>, Span),
    Lambda(&'a [Param<'a>], &'a Expr<'a>, Span), App(&'a Expr<'a>, &'a [Expr<'a>], Span),
    Record(&'a [Field<'a>], Span), Field(&'a Expr<'a>, Ident, Span), Update(&'a Expr<'a>, &'a [Field<'a>], Span),
    List(&'a [Expr<'a>], Span), Match(&'a Expr<'a>, &'a [Arm<'a>], Span), Assert(&'a Expr<'a>, &'a Expr<'a>, Span),
}
// crates/lume-syntax/src/ast.rs
pub enum TypeExpr<'a> {
    Var(Ident, Span), Con(Ident, &'a [TypeExpr<'a>], Span), Arrow(&'a TypeExpr<'a>, &'a TypeExpr<'a>, Span),
    Record(&'a [TypeField<'a>], Option<Ident>, Span),   // {a: Int, b: Str | r}
    Forall(&'a [Ident], &'a TypeExpr<'a>, Span),
}
pub enum Pattern { Tag(String, Option<Ident>, Span), Wild(Span) }
// crates/lume-types/src/ty.rs
pub enum Ty { Var(TyVarId), Con(Symbol, Vec<Ty>), Arrow(Box<Ty>, Box<Ty>), Record(Row), Forall(Vec<TyVarId>, Box<Ty>) }
pub struct Row { pub fields: BTreeMap<Symbol, Ty>, pub tail: Option<TyVarId> }   // I8
pub struct TyVar { pub id: TyVarId, pub level: u32, pub link: Option<Ty>, pub origin: Span }   // I7
// crates/lume-eval/src/value.rs
pub enum Value {
    Null, Bool(bool), Int(BigInt), Float(f64), Str(Rc<str>), List(Rc<[Value]>),
    Record(Rc<BTreeMap<Symbol, Value>>), Closure(Rc<Closure>), Builtin(BuiltinFn),
    Capability(Capability), Thunk(Rc<RefCell<Thunk>>),
}
// crates/lume-diag/src/diagnostic.rs
pub struct Diagnostic { pub code: Code, pub severity: Severity, pub message: String, pub primary: Label, pub secondary: Vec<Label>, pub notes: Vec<String>, pub helps: Vec<Help> }
pub struct Label { pub span: Span, pub message: String }
pub struct Help { pub message: String, pub replacement: Option<Replacement> }
pub struct Replacement { pub span: Span, pub text: String }
// crates/lume-eval/src/effects.rs
pub enum Capability { FileRead(PathScope), FileWrite(PathScope), NetFetch(HostAllowlist), ProcExec(ArgAllowlist) }
pub struct PathScope { pub root: PathBuf }          // a capability is scoped, never global (I14)
pub struct HostAllowlist { pub hosts: Vec<String> }
pub struct ArgAllowlist { pub programs: Vec<String> }
