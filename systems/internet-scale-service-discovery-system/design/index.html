Core schema & types (§11)

pub struct ServiceName { service: String, namespace: String, region: Option<RegionId> }  // canonical, I1
pub struct RegionId(String);                 // e.g. "us-east"
pub struct InstanceId(Uuid);                 // unique within a service, I2
pub struct Locality { region: RegionId, zone: Option<String>, rack: Option<String> }      // explicit, ADR-21
impl ServiceName { fn parse(fqdn: &str) -> Result<Self, SdError>; fn to_dns(&self) -> String; }  // I1
pub struct Instance {
    id: InstanceId, service: ServiceName, addr: SocketAddr,
    locality: Locality, tags: Vec<String>, meta: BTreeMap<String,String>, weight: u32,
}
pub struct Lease { instance: InstanceId, ttl: Duration, last_renew: Instant, expiry: Instant }   // I3
pub enum HealthState { Passing, Warning, Critical }
pub struct Health { instance: InstanceId, state: HealthState, incarnation: u64, logical_ts: u64, output: String }  // I4/I7
pub struct Version { incarnation: u64, logical_ts: u64 }                 // I7/I11 total order
impl Version { fn dominates(&self, other: &Version) -> bool; }           // (inc desc, ts desc)
pub struct Versioned<T> { value: T, version: Version, key: Key }         // an LWW register entry, §8.12
pub enum MemberStatus { Alive, Suspect, Dead }
pub struct NodeMember { node: NodeId, addr: SocketAddr, status: MemberStatus, incarnation: u64 }   // I6/I11
pub trait Membership {
    fn probe_cycle(&self);                                   // one SWIM round (direct + indirect)
    fn members(&self) -> &MemberTable;
    fn on_status_change(&self, cb: impl Fn(NodeId, MemberStatus));   // → gossip + health
}
pub struct GossipEvent { key: Key, value: StateValue, version: Version, hops: u8 }   // I7/I8
pub enum StateValue { Health(Health), Member(NodeMember), Catalog(CatalogDelta) }
pub trait Gossip {
    fn enqueue(&self, ev: GossipEvent);                      // disseminate, fanout k (I8)
    fn push_pull_round(&self, peer: NodeId);                 // §8.4
    fn apply(&self, ev: GossipEvent) -> ApplyResult;         // I7 monotonic apply
}
pub enum ApplyResult { Applied, StaleDropped, Conflict { winner: Version } }   // I11
pub struct MerkleDigest { root: Hash, depth: u8 }
pub trait AntiEntropy {
    fn digest(&self) -> MerkleDigest;                        // #71, leaves = hash(key,version)
    fn reconcile(&self, peer: NodeId) -> ReconcileStats;     // I9, exchange only the diff
}
pub struct ReconcileStats { compared: usize, diff_keys: usize, bytes: usize, repaired: usize }   // §4.4
pub struct RegionLink { region: RegionId, wan_peers: Vec<SocketAddr>, status: LinkStatus }
pub enum LinkStatus { Connected, Degraded { stale_age: Duration }, Partitioned }   // I13
pub struct Projection { origin: RegionId, instances: Vec<Versioned<Instance>>, age: Duration }   // I10
pub trait Federation {
    fn wan_summary_round(&self);                             // sparse WAN gossip (ADR-10)
    fn project(&self, name: &ServiceName) -> Option<Projection>;          // selective replication
    fn subscribe(&self, name: &ServiceName, region: RegionId);            // demand-driven (§8.13)
}
pub struct ResolveQuery {
    name: ServiceName, tags: Vec<String>, health: HealthPolicy,
    locality: Locality, allow_remote: bool, max_results: usize, weight_mode: WeightMode,
    last_seen_version: Option<u64>,        // blocking query (I5)
    timeout: Duration,
}
pub enum HealthPolicy { PassingOnly, PassingOrWarning, Any }     // I4
pub struct Resolution { candidates: Vec<Candidate>, version: u64, stale: bool, stale_age: Duration }   // I10
pub struct Candidate { instance: Instance, health: HealthState, locality_rank: u8, weight: u32 }
pub trait Resolver {
    fn resolve(&self, q: &ResolveQuery) -> Result<Resolution, SdError>;
    fn watch(&self, q: &ResolveQuery) -> Result<Resolution, SdError>;   // blocking, returns on change/timeout (I5)
}
pub enum Rr { A(Ipv4Addr), Aaaa(Ipv6Addr), Srv { prio:u16, weight:u16, port:u16, target:String },
              Txt(Vec<String>), Ptr(String) }                 // RFC 1035 / 2782 / 6763
pub trait DnsServer {
    fn answer(&self, q: &DnsQuery) -> DnsResponse;            // A/AAAA/SRV/TXT + DNS-SD; TTL from cache (I16)
}
pub struct CacheEntry<T> { value: T, inserted: Instant, ttl: Duration, negative: bool }    // I10/I17
pub trait Cache<K, V> {
    fn get_or_fetch(&self, k: K, fetch: impl Fn() -> Result<V, SdError>) -> Result<Cached<V>, SdError>;  // single-flight (ADR-16)
}
pub struct Cached<V> { value: V, fresh: bool, age: Duration }   // fresh=false → stale-if-error (principle 10)
pub struct NodeConfig {
    node_id: NodeId, region: RegionId, locality: Locality,
    join: Vec<SocketAddr>, wan_join: Vec<SocketAddr>,
    probe_interval: Duration, gossip_interval: Duration, ae_interval: Duration,
    default_lease_ttl: Duration, gossip_fanout: u8, gossip_key: SecretKey,
    catalog_consistency: CatalogMode, self_preservation: bool, serve_stale: StaleMode,
}
pub enum CatalogMode { Eventual, RegionRaft }                  // ADR-12 / I14
pub enum StaleMode { ServeStale, FailClosed }                  // partition behavior, P9 / I13
fn bootstrap(cfg: NodeConfig) -> Result<Node, SdError>;        // join region gossip + one anti-entropy
pub struct PropagationTrace { event_key:String, origin_node:u32, gossip_rounds:u32,
                              propagate_ms:f64, ae_repaired:bool, cross_region_ms:Option<f64> }
pub struct ResolveTrace { name:String, tier:&'static str /*agent|regional|authoritative|remote*/,
                          latency_us:u64, candidates:usize, stale:bool, stale_age_ms:f64, locality_rank:u8 }
pub struct MembershipTrace { node:u32, members:usize, suspect:usize, dead:usize,
                             detect_ms:Option<f64>, false_positive:bool, incarnation:u64 }
pub struct GossipTrace { events_out:u64, events_in:u64, bytes:u64, fanout:u8,
                         ae_diff_keys:usize, ae_bytes:u64, stale_dropped:u64 }
pub enum SdError { NameInvalid(String), InstanceNotFound(InstanceId), ServiceNotFound(ServiceName),
                   LeaseExpired(InstanceId), HealthPolicyEmpty, ProtoVersion{ours:u16,theirs:u16},
                   GossipAuth, WanPartition(RegionId), Corruption{seq:u64}, BadArg(String),
                   CatalogUnavailable, Timeout }
