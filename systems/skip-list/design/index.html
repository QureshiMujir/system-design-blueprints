// ordered_map.rs
use core::ops::Bound;

/// Canonical ordered-map interface; all four list variants implement this.
pub trait OrderedMap<K: Ord, V> {
    fn get(&self, key: &K) -> Option<&V>;
    fn insert(&mut self, key: K, value: V) -> Option<V>; // returns evicted value
    fn remove(&mut self, key: &K) -> Option<V>;
    fn range(&self, lo: Bound<&K>, hi: Bound<&K>) -> RangeIter<'_, K, V>;
}

/// Lazy forward iterator produced by range(); yields (key, value) in ascending order.
pub struct RangeIter<'a, K, V> {
    current: *const Node<K, V>,   // raw pointer into list; valid while 'a lives
    hi:      Bound<K>,
    _marker: core::marker::PhantomData<&'a (K, V)>,
}
impl<'a, K: Ord, V> Iterator for RangeIter<'a, K, V> {
    type Item = (&'a K, &'a V);
    fn next(&mut self) -> Option<Self::Item> { /* advance current at level 0 */ }
}
// level_gen.rs

/// Geometric level generator: P(height >= i) = p^i, p = 1/p_inv.
/// max_level = ceil(log_{p_inv}(N_max)); typical: p_inv=2, max_level=32.
pub struct LevelGenerator {
    pub p_inv:     u32,    // reciprocal of p; 2 → p=1/2; 4 → p=1/4
    pub max_level: usize,  // hard cap; E[height] = p_inv / (p_inv - 1)
    rng:           Rng,
}

impl LevelGenerator {
    /// Constructs generator. Panics if p_inv < 2 or max_level == 0.
    pub fn new(p_inv: u32, max_level: usize) -> Self;

    /// Returns a random level in [1, max_level].
    /// Geometric height: starts at 1 and rises one level per successful Bernoulli(1/p_inv)
    /// trial, capped at max_level. E[random_level()] = p_inv/(p_inv-1).
    pub fn random_level(&mut self) -> usize;
}
// rng.rs

/// Fast non-cryptographic per-thread RNG (xorshift64 / wyrand variant).
/// Seed via thread-local or explicit value; never shared across threads.
pub struct Rng(u64);

impl Rng {
    pub fn new(seed: u64) -> Self;
    pub fn next_u64(&mut self) -> u64;  // xorshift: x ^= x<<13; x ^= x>>7; x ^= x<<17
    pub fn next_u32(&mut self) -> u32 { self.next_u64() as u32 }
}

/// A skip-list node. Owned by the enclosing SkipList via raw pointer.
/// forward[i] is the next node at level i (None sentinel = null ptr).
/// INVARIANT: forward.len() == height of this node's tower.
pub struct Node<K, V> {

    pub value:   V,
    pub forward: Vec<*mut Node<K, V>>,  // forward[0] = level-0 successor
}

    // Allocates via Box::into_raw; height forward slots initialised to null.

// list.rs

/// Single-threaded probabilistic ordered map.
/// Ownership: SkipList owns all Node allocations; Drop walks level-0 freeing each.
/// NOT Send/Sync — single-thread only; use ConcurrentSkipList for multi-thread.
pub struct SkipList<K: Ord, V> {
    head:      *mut Node<K, V>, // sentinel; key uninitialised; forward[0..max_level]
    level:     usize,           // current highest non-empty level (1-indexed)
    max_level: usize,           // hard cap from LevelGenerator
    len:       usize,           // number of user-inserted nodes
    gen:       LevelGenerator,
}

impl<K: Ord, V> SkipList<K, V> {
    pub fn new(p_inv: u32, max_level: usize) -> Self;
    pub fn len(&self)     -> usize;
    pub fn is_empty(&self)-> bool;
}

impl<K: Ord, V> Drop for SkipList<K, V> {
    fn drop(&mut self) { /* walk level 0, Box::from_raw each non-head node, then head */ }
}
// search.rs

impl<K: Ord, V> SkipList<K, V> {
    /// Core search: fills update[0..self.level] with predecessor pointers at each level.
    /// Returns pointer to the level-0 predecessor of key (or head if key < all keys).
    /// Complexity: O(log n) expected; starts at self.level-1, drops down on key overshoot.
    pub(crate) fn search(&self, key: &K, update: &mut [*mut Node<K, V>]) -> *mut Node<K, V>;

    /// Public point lookup. Returns &V if found, None otherwise.
    pub fn get(&self, key: &K) -> Option<&V>;

    /// Insert key/value. Returns Some(old_value) on duplicate, else None.
    /// Algorithm: search → fill update[]; pick random_level; splice into forward[0..lvl];
    ///   grow self.level if new tower is taller.
    pub fn insert(&mut self, key: K, value: V) -> Option<V>;

    /// Remove key. Returns Some(value) if present.
    /// Algorithm: search → fill update[]; verify node is key match;
    ///   unlink from each level; shrink self.level while top level is empty.
    pub fn remove(&mut self, key: &K) -> Option<V>;
}

impl<K: Ord, V> OrderedMap<K, V> for SkipList<K, V> { /* delegates */ }

/// Full ordered iteration at level 0.
pub struct Iter<'a, K, V> {
    current: *const Node<K, V>,
    _marker: core::marker::PhantomData<&'a SkipList<K, V>>,
}
impl<'a, K: Ord, V> Iterator for Iter<'a, K, V> {
    type Item = (&'a K, &'a V);
    fn next(&mut self) -> Option<Self::Item> { /* advance via forward[0] */ }
}
impl<K: Ord, V> SkipList<K, V> {
    pub fn iter(&self) -> Iter<'_, K, V>;
}
// deterministic.rs

/// Deterministic skip list (1-2-3 variant).
/// INVARIANT (gap invariant): between any two consecutive level-(i+1) nodes,
///   there are exactly 1, 2, or 3 level-i nodes. No randomness; height O(log n) worst-case.
/// Isomorphic to a B-tree with branching factor 2–4.
pub struct DetSkipList<K: Ord, V> {
    head:      *mut Node<K, V>,
    level:     usize,
    max_level: usize,
    len:       usize,
}

impl<K: Ord, V> DetSkipList<K, V> {
    pub fn new(max_level: usize) -> Self;
    /// Insert: search as in SkipList, then raise/lower levels to restore gap invariant.
    pub fn insert(&mut self, key: K, value: V) -> Option<V>;
    /// Remove: unlink, then rebalance to restore gap invariant.
    pub fn remove(&mut self, key: &K) -> Option<V>;
    pub fn get(&self, key: &K) -> Option<&V>;
}

impl<K: Ord, V> OrderedMap<K, V> for DetSkipList<K, V> { /* delegates */ }
// indexable.rs

/// Skip list augmented with per-forward-pointer span counts.
/// INVARIANT: forward[i].span = number of level-0 nodes skipped by that pointer.
/// rank(key):   sum spans along the search path until key found → O(log n).
/// select(i):   follow pointers consuming span budget i → O(log n).
pub struct IndexableSkipList<K: Ord, V> {
    head:      *mut INode<K, V>,  // INode extends Node with span per forward ptr
    level:     usize,
    max_level: usize,
    len:       usize,
    gen:       LevelGenerator,
}

/// Node variant carrying per-level span alongside the forward pointer.
pub struct INode<K, V> {
    pub key:     K,
    pub value:   V,
    pub forward: Vec<(*mut INode<K, V>, usize)>,  // (ptr, span)
}

impl<K: Ord, V> IndexableSkipList<K, V> {
    pub fn new(p_inv: u32, max_level: usize) -> Self;
    pub fn insert(&mut self, key: K, value: V) -> Option<V>;
    pub fn remove(&mut self, key: &K) -> Option<V>;
    pub fn get(&self, key: &K) -> Option<&V>;

    /// Number of keys strictly less than `key`. O(log n) via span accumulation.
    pub fn rank(&self, key: &K) -> usize;
    /// The `i`-th smallest key (0-indexed). O(log n) via span budget consumption.
    pub fn select(&self, i: usize) -> Option<&K>;
}

impl<K: Ord, V> OrderedMap<K, V> for IndexableSkipList<K, V> { /* delegates */ }
// marked_ptr.rs

/// Tagged atomic pointer: low bit encodes logical-deletion mark.
/// INVARIANT: all Node allocations are ≥2-byte aligned (low bit is always 0 in valid ptrs).
/// Memory ordering: load uses Acquire; CAS uses AcqRel/Acquire pair; store uses Release.
pub struct MarkedPtr<K, V>(AtomicPtr<CNode<K, V>>);

impl<K, V> MarkedPtr<K, V> {
    pub fn new(ptr: *mut CNode<K, V>) -> Self;
    pub fn load(&self, order: Ordering) -> (*mut CNode<K, V>, bool);  // (ptr, is_marked)
    pub fn is_marked(&self, order: Ordering) -> bool;
    /// CAS: expects (expected_ptr, expected_mark); sets to (new_ptr, new_mark).
    /// Returns Ok(()) on success, Err(actual) on failure.
    pub fn compare_exchange(
        &self,
        expected: (*mut CNode<K, V>, bool),
        new:      (*mut CNode<K, V>, bool),
        success:  Ordering,
        failure:  Ordering,
    ) -> Result<(), (*mut CNode<K, V>, bool)>;
    /// Mark this pointer (logical deletion); returns false if already marked.
    pub fn mark(&self, current_ptr: *mut CNode<K, V>) -> bool;
}
// list.rs  (skl-concurrent)

/// Lock-free concurrent skip list node.
/// top_level: the height of this node's tower (immutable after creation).
/// next[i]: MarkedPtr; marked bit = node is logically deleted at level i.
pub struct CNode<K, V> {
    pub key:       K,
    pub value:     V,
    pub top_level: usize,
    pub next:      Vec<MarkedPtr<K, V>>,  // next.len() == top_level + 1
}

/// Lock-free concurrent ordered map.
/// Send + Sync: all mutation via CAS on MarkedPtr; no locks.
/// Ownership: ConcurrentSkipList owns CNode allocations; reclamation via Ebr.
pub struct ConcurrentSkipList<K: Ord, V> {
    head:      *const CNode<K, V>,  // permanent sentinel; never retired
    max_level: usize,
    gen:       LevelGenerator,      // interior-mutable via Mutex or per-thread Rng
    ebr:       Ebr,
}

// SAFETY: raw ptr to head sentinel is valid for the lifetime of self.
unsafe impl<K: Ord + Send, V: Send> Send for ConcurrentSkipList<K, V> {}
unsafe impl<K: Ord + Send + Sync, V: Send + Sync> Sync for ConcurrentSkipList<K, V> {}

impl<K: Ord, V> ConcurrentSkipList<K, V> {
    pub fn new(p_inv: u32, max_level: usize) -> Self;

    /// Lock-free point lookup. Ignores logically-deleted nodes.
    pub fn get(&self, key: &K) -> Option<&V>;

    /// Lock-free insert. Linearization point: level-0 CAS succeeding.
    /// Pre-condition: find() returns preds/succs; loop retries on CAS failure.
    pub fn insert(&self, key: K, value: V) -> Option<V>;

    /// Lock-free remove. Two phases:
    ///   1. Logical delete: mark next[top_level] down to next[0] via CAS.
    ///   2. Physical unlink: call find(), which snips marked nodes as a side effect (HELPING).
    pub fn remove(&self, key: &K) -> Option<V>;
}
// find.rs

impl<K: Ord, V> ConcurrentSkipList<K, V> {
    /// Traverses list top-down, filling preds[i] and succs[i] for each level i.
    /// HELPING: on encountering a marked node at level i, CASes it out of preds[i]'s next[i]
    ///   before continuing — physically unlinks without a lock.
    /// Returns true if key is found (and not marked) at level 0.
    /// Correctness: if find() returns true, succs[0] is the live node with key == target.
    /// Linearization: get() reads succs[0].value after find() returns true (Acquire load).
    pub(crate) fn find(
        &self,
        key:   &K,
        preds: &mut [*const CNode<K, V>; 32],
        succs: &mut [*const CNode<K, V>; 32],
    ) -> bool;
}
// ebr.rs

/// Epoch-based memory reclamation.
/// Three epochs (0, 1, 2) rotate globally; per-thread state: current_epoch + retire list.
/// Protocol: pin() reads global epoch; unpin() may advance global epoch if all threads
///   are in the same epoch; retire(ptr) adds to thread-local list; freed when epoch safe.
pub struct Ebr {
    // global epoch counter (mod 3), per-thread slots
}

impl Ebr {
    pub fn new() -> Self;
    /// Pin current thread; returns a Guard that unpins on drop.
    pub fn pin(&self) -> Guard<'_>;
    /// Retire a raw pointer; freed after all threads advance past current epoch.
    /// SAFETY: ptr must not be accessed after retire(); no live references may exist.
    pub unsafe fn retire<K, V>(&self, ptr: *mut CNode<K, V>);
}

/// RAII guard: holds a pin for the duration of an operation.
pub struct Guard<'a> { ebr: &'a Ebr, epoch: u64 }
impl Drop for Guard<'_> { fn drop(&mut self) { /* unpin; try advance global epoch */ } }
