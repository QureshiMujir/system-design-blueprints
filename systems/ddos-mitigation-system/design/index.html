Core schema & types (§11)

PacketMeta { ts: u64 (monotonic, fast-path), len: u16, ingress_if: u32, attacker-controlled: full wire bytes; not persisted }
FiveTuple { src_ip: IpAddr, dst_ip: IpAddr, src_port: u16, dst_port: u16, proto: u8 }
  // ownership: copy-type, no allocation; attacker-controlled: all fields
FlowKey(FiveTuple)  // hashable, fast-path lookup key
FlowState { key: FlowKey, syn_ts: u64, pkt_count: u64, byte_count: u64, tcp_state: TcpState, last_seen: u64 }
  // mutation: shard-local lock-free CAS updates only; persistence: none, memory-only; not serialized off-node
FlowTable { shards: Vec<Shard<FlowKey, FlowState>>, capacity: usize, evict: EvictPolicy }
  // concurrency: per-shard, no cross-shard lock; bounded per I4
Verdict { Pass, Drop, Challenge, RateLimit }
  // fast-path return type; attacker input never directly selects Drop vs Pass — always derived
XdpAction(Verdict)  // maps to XDP_PASS/XDP_DROP/XDP_TX at the eBPF ABI boundary
SynCookie { mac_tag: u32, t_coarse: u5, mss_enc: u3 }
  // constructed fast-path; validated fast-path; no persistence; key rotation is control-plane owned
RateLimiter { buckets: Map<LimitKey, TokenBucket> }
LimitKey { Source(IpAddr) | Prefix(IpNet) | Asn(u32) }
TokenBucket { capacity: f64, tokens: f64, refill_rate: f64, last_refill: u64 }
  // mutation: per-key atomic; attacker-controlled: which key is touched (via src IP), not the bucket math itself
Baseline { window: RingBuffer<f64>, mean: f64, stddev: f64, k: f64 }
EntropyEstimate { counts: Map<IpAddr, u64>, total: u64 }  // bounded via periodic reset, not per-packet growth
  // computed near-fast-path (batched), not true per-packet fast path
CountMinSketch { width: usize, depth: usize, table: Vec<Vec<u32>>, hashers: [Hasher; depth] }
  // fixed memory at construction (I7); attacker-controlled: which cells increment, never the memory footprint
HeavyHitter { key: FlowKey, estimate: u64 }
Detector { baseline: Baseline, entropy: EntropyEstimate, sketch: CountMinSketch }
AttackSignature { class: AttackClass, evidence: Vec<EvidenceItem> }
AttackClass { Volumetric, Protocol, Reflection, Application }
MitigationRule { id: RuleId, match_key: LimitKey, action: Verdict, expiry: RuleExpiry }
  // I5: expiry is mandatory, not optional — no MitigationRule constructor omits it
RuleExpiry { created_at: u64, ttl: Duration }
Reputation { key: LimitKey, score: f32, last_update: u64 }  // decays toward 0 monotonically (I9)
AllowList(Set<LimitKey>) / BlockList(Set<LimitKey>)
Challenge { JsChallenge { nonce: [u8;16] } | ProofOfWork { difficulty: u8, seed: [u8;32] } }
ProofOfWork { difficulty: u8, seed: [u8;32] }  // difficulty attacker-visible, seed server-chosen
L7Request { method: HttpMethod, path: String, headers: Map<String,String>, fingerprint: Fingerprint }
  // fully attacker-controlled; parsed near-fast-path, never on the eBPF fast path
Fingerprint { header_order_hash: u64, ua_class: UaClass, tls_ja3: Option<[u8;16]> }
Diversion { active: bool, activated_at: Option<u64>, target_prefix: IpNet }
BgpAnnouncement { prefix: IpNet, community_tags: Vec<u32>, state: AnnounceState }
ScrubberTunnel { gre_endpoint: IpAddr, tunnel_id: u32, mtu: u16 }
Telemetry { event_id: Uuid, verdict: Verdict, evidence: Vec<EvidenceItem>, sampled: bool }
  // async-emitted; never gates fast-path return (I8)
MitigationState { Baseline, Detecting, Mitigating, DeEscalating, Clear }
ErrorTaxonomy { ParseError, VerifierRejected, CapacityExceeded, PolicyInvalid, ChallengeExpired, DiversionFailed }
