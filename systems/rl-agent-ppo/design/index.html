Core schema & types (§11)

// space types — immutable, hashable, constructed once at agent-init (I13)
DiscreteSpaceSpec      { n: int }
BoxSpaceSpec           { low: ndarray, high: ndarray, shape: tuple[int, ...] }
MultiDiscreteSpaceSpec { nvec: tuple[int, ...] }

// EnvSpec — owner: envs.registry; immutable; picklable (Async-vec prerequisite, deferred);
// no persistence (reconstructed from config each run); compat: pinned to a Gymnasium env_id string
EnvSpec:
  env_id: str
  seed: int
  idx: int
  wrappers: tuple[WrapperCtor, ...]
  def make() -> GymEnv

// StepResult — owner: envs.vector_env; mutation: none (frozen); concurrency: produced by the
// single-writer VecEnv step; no persistence; serialization: numpy arrays only, no pickled callables
StepResult:
  obs: ndarray            // (N, *obs_shape)
  reward: ndarray         // (N,)
  terminated: ndarray     // (N,) bool
  truncated: ndarray      // (N,) bool
  info: dict[str, Any]
// VecEnvProtocol — owner: envs.vector_env; structural protocol (P1); no implementation coupling
VecEnvProtocol(Protocol):
  num_envs: int
  observation_space: Space
  action_space: Space
  def reset(seed: Sequence[int] | None) -> tuple[ndarray, dict]
  def step(actions: ndarray) -> StepResult
  def close() -> None

// RunningMeanStd — owner: utils.running_stats; mutation: update() only during COLLECT (I2, P5);
// concurrency: single-writer per normalizer instance; persistence: checkpointed (I10)
RunningMeanStd:
  mean: ndarray
  var: ndarray
  count: float
  def update(batch: ndarray) -> None
  def normalize(x: ndarray, clip: float) -> ndarray
// RolloutBuffer — owner: buffer.rollout_buffer; mutation: add()/finalize()/reset() only (I3, I5);
// concurrency: single writer (env_runner), single reader (ppo_update) per generation; no cross-run
// persistence; serialization: flat numpy arrays only
RolloutBuffer:
  T: int
  N: int
  obs: ndarray            // (T,N,*obs_shape)
  actions: ndarray         // (T,N,*action_shape)
  log_probs: ndarray       // (T,N)
  values: ndarray          // (T,N)
  rewards: ndarray         // (T,N)
  dones: ndarray           // (T,N)
  advantages: ndarray      // (T,N)
  returns: ndarray         // (T,N)
  def add(transition: Transition) -> None
  def finalize(bootstrap_value: ndarray, bootstrap_done: ndarray) -> None
  def as_flat_tensors() -> dict[str, Tensor]
  def reset() -> None

// Trajectory — owner: runner.eval_harness; immutable; produced per eval episode; not persisted
// beyond the EvalReport aggregate
Trajectory:
  obs: ndarray
  actions: ndarray
  rewards: ndarray
  episode_return: float
  episode_length: int
// PolicyNetwork / ValueNetwork — owner: agent.actor_critic; pure forward functions, no rollout
// state; mutation: parameters only, via optimizer.step() (sole writer: PpoUpdater)
PolicyNetwork(Protocol):
  def forward(obs: Tensor) -> Distribution

ValueNetwork(Protocol):
  def forward(obs: Tensor) -> Tensor        // (B,)

// PPOConfig — owner: training.config; frozen dataclass (I9, ADR-18); the complete reproducibility
// contract; compat: versioned alongside checkpoint format (§27)
PPOConfig:
  env_id: str
  seed: int
  num_envs: int
  num_steps_per_rollout: int
  learning_rate: float
  num_epochs_per_update: int
  num_minibatches: int
  clip_ratio: float
  gamma: float
  gae_lambda: float
  target_kl: float | None
  total_env_steps: int
  def derived() -> dict[str, int]
// UpdateStats — owner: agent.ppo_update; produced once per UPDATE state transition; logged,
// never persisted beyond telemetry (ADR-21)
UpdateStats:
  policy_loss: float
  value_loss: float
  entropy: float
  approx_kl: float
  clip_fraction: float
  explained_variance: float
  grad_norm: float
  kl_early_stop_epoch: int        // -1 if never fired (I14)

// EvalReport — owner: runner.eval_harness; immutable; written to disk per eval cadence
EvalReport:
  mean_return: float
  std_return: float
  mean_length: float
  num_episodes: int
  policy_checkpoint_id: str

// CheckpointRecord — owner: training.checkpoint; versioned (ADR-17, §27); bundles I10's five fields
CheckpointRecord:
  format_version: int
  weights_path: str                // safetensors
  optimizer_state: dict[str, Tensor]
  rng_state: dict[str, Any]
  normalizer_stats: dict[str, RunningMeanStd]
  config: PPOConfig

// Error types — owner: shared across envs/buffer/training; raised, never silently swallowed
EnvContractError(Exception)         // space/step-shape violations (§8 malformed-input examples)
BufferStateError(Exception)         // I1/I3/I5 violations
CheckpointCompatibilityError(Exception)  // format_version mismatch (§27)
