Core schema & types (§11)

pub struct DocId(pub u32);
pub struct RepoId(pub u32);
pub struct ShardId(pub u32);
pub struct Trigram(pub u32);  // upper 8 bits unused; bits 23-0 = b0<<16 | b1<<8 | b2

impl Trigram {
    pub fn pack(b: [u8; 3]) -> Self;     // bijection: 3 bytes -> 24-bit value (I8)
    pub fn unpack(self) -> [u8; 3];      // inverse of pack
    pub fn from_bytes(s: &[u8]) -> Option<Self>;  // None if len < 3
}
pub struct Posting(pub DocId);   // a single entry in a trigram's posting list

pub struct PostingList<'a> {
    data: &'a [u8],              // mmap-backed delta+varint encoded bytes
    len: u32,                    // number of DocId entries
}

impl<'a> PostingList<'a> {
    pub fn iter(&self) -> PostingIter<'a>;   // ascending DocId iterator; I2
    pub fn len(&self) -> u32;
    pub fn is_empty(&self) -> bool;
}

pub struct PostingIter<'a> { /* opaque; decodes delta+varint lazily */ }
impl<'a> Iterator for PostingIter<'a> { type Item = DocId; }
pub struct PostingWriter {
    buf: Vec<u8>,
    last: u32,   // last DocId written, for delta computation
}

impl PostingWriter {
    pub fn push(&mut self, id: DocId);   // panics if id <= last (order violation)
    pub fn finish(self) -> Vec<u8>;      // returns encoded bytes
}
pub struct DocMeta {
    pub path:       String,      // repo-relative file path; UTF-8
    pub repo:       RepoId,
    pub lang:       String,      // MIME-like lang tag, e.g. "rust", "go", ""
    pub size:       u32,         // file byte length at index time
    pub sha256:     [u8; 32],    // content address (I5)
    pub line_count: u32,
}
pub enum QueryAst {
    Literal(String),
    Regex(Hir),                          // csearch-regex Hir
    And(Vec<QueryAst>),
    Or(Vec<QueryAst>),
    Not(Box<QueryAst>),
    Field(Filter),
}

pub enum Filter {
    File(Hir),                           // file: <regex> — match on DocMeta.path
    Lang(String),                        // lang: <tag>
    Repo(String),                        // repo: <name>
    Case(bool),                          // case:yes / case:no
    Sym(String),                         // sym: <name> — symbol search
}
pub enum TrigramQuery {
    All,                              // matches all documents (wildcard / .* lowering)
    Any,                              // matches no documents (impossible; prune branch)
    Trigram(Trigram),                 // single trigram must be present
    And(Vec<TrigramQuery>),           // all children must be satisfied
    Or(Vec<TrigramQuery>),            // at least one child must be satisfied
}
// csearch-regex::nfa
pub struct Nfa {
    pub states: Vec<NfaState>,
    pub start:  u32,
}

pub enum NfaState {
    Match,
    Byte(u8, u32),               // on byte b, go to state next
    Split(u32, u32),             // epsilon split to two states
    Capture(u8, u32),            // capture group open/close + next
}

// csearch-regex::pike_vm
pub struct PikeVm<'a> {
    nfa: &'a Nfa,
}

impl<'a> PikeVm<'a> {
    pub fn new(nfa: &'a Nfa) -> Self;
    pub fn find_iter<'t>(&self, text: &'t [u8]) -> PikeMatchIter<'t, 'a>;
    // guarantee: O(|text| * |NFA states|); no backtracking
}
pub struct MatchSpan {
    pub doc:       (ShardId, DocId),
    pub line:      u32,              // 1-based line number
    pub col:       u32,              // 0-based byte column within line
    pub len:       u32,              // match byte length
    pub line_text: String,           // full text of the matched line (for display)
}

pub struct SearchResult {
    pub doc_meta:  DocMeta,
    pub spans:     Vec<MatchSpan>,   // all matches within this doc, ascending line order
    pub score:     RankScore,
}

pub struct RankScore(pub f32);       // higher = more relevant; NaN forbidden
pub struct SymbolEntry {
    pub name: String,
    pub kind: SymbolKind,
    pub doc:  (ShardId, DocId),
    pub line: u32,                   // 1-based definition line
}

pub enum SymbolKind {
    Function,
    Struct,
    Enum,
    Trait,
    Class,
    Const,
    TypeAlias,
    Module,
    Other(String),
}
pub struct ShardHeader {
    pub magic:              [u8; 8],    // b"CSEARCH1"
    pub version:            u16,        // format version; currently 1
    pub shard_id:           ShardId,
    pub n_docs:             u32,
    pub n_trigrams:         u32,
    pub trigram_table_off:  u64,        // byte offset into shard file
    pub posting_lists_off:  u64,
    pub doc_table_off:      u64,
    pub symbol_index_off:   u64,        // 0 if no symbol index in this shard
    pub crc32:              u32,        // CRC of all bytes following the header
}
// csearch-index::shard_writer
pub struct PostingWriter {
    buf:  Vec<u8>,
    last: u32,
}
impl PostingWriter {
    pub fn push(&mut self, id: DocId);     // contract: id > last, else panic
    pub fn finish(self) -> Vec<u8>;        // encoded posting list bytes
}

// csearch-shard::posting_reader
pub struct PostingReader<'a> {
    data:  &'a [u8],                       // mmap slice for the entire posting lists region
}
impl<'a> PostingReader<'a> {
    pub fn list(&self, offset: u64, len: u32) -> PostingList<'a>;
    // offset + (u64 * len) must be within data; else panic in debug, UB-safe in release
}
// All identity/value types must implement:
trait CoreValue: Copy + Clone + Eq + Ord + Hash + Debug + Send + Sync
    + serde::Serialize + for<'de> serde::Deserialize<'de> {}

// Implemented by: DocId, RepoId, ShardId, Trigram, RankScore (except Copy constraint relaxed for RankScore)

// Shard-reading abstraction (csearch-shard)
pub trait ShardRead: Send + Sync {
    fn posting_list(&self, t: Trigram) -> Option<PostingList<'_>>;
    fn doc_meta(&self, id: DocId) -> Option<&DocMeta>;
    fn n_docs(&self) -> u32;
    fn shard_id(&self) -> ShardId;
}
