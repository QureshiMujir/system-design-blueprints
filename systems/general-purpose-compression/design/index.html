// bit_io.rs — LSB-first bit I/O; no heap allocation on BitReader

/// Accumulates bits into a byte stream; LSB of each byte first.
pub struct BitWriter {
    buf: Vec<u8>,       // output byte buffer
    bits: u64,          // pending bit accumulator
    bit_count: u32,     // valid bits in `bits` (0..=63)
}
impl BitWriter {
    pub fn write_bits(&mut self, value: u64, n: u32);   // n <= 56
    pub fn flush(&mut self);                             // zero-pad to byte boundary
    pub fn finish(self) -> Vec<u8>;
}

/// Reads bits from a shared byte slice; bounds-checked on every access.
/// Returns Err(DecompressError::Truncated) on over-read — never panics.
pub struct BitReader<'a> {
    src: &'a [u8],
    byte_pos: usize,
    bits: u64,
    bit_count: u32,
}
impl<'a> BitReader<'a> {
    pub fn read_bits(&mut self, n: u32) -> Result<u64, DecompressError>;
    pub fn peek_bits(&self, n: u32) -> Result<u64, DecompressError>;
    pub fn skip_bits(&mut self, n: u32) -> Result<(), DecompressError>;
    pub fn bytes_remaining(&self) -> usize;
}
// checksum.rs
/// xxhash64 streaming state; initialized with seed 0.
pub struct Checksum { state: u64, /* internal xxhash64 accumulators */ }
impl Checksum {
    pub fn update(&mut self, data: &[u8]);
    pub fn finish(&self) -> u64;
}
// params.rs
pub enum CompressionLevel { Fast, Default, Max, Custom(CompressParams) }

pub struct CompressParams {
    pub window_log: u8,      // window size = 1 << window_log; valid 10..=27
    pub hash_log: u8,        // hash table size = 1 << hash_log; valid 6..=23
    pub chain_log: u8,       // chain/tree depth log; valid 6..=29
    pub search_depth: u32,   // max chain walk length per position
    pub target_length: u32,  // optimal parse: stop searching beyond this match length
    pub strategy: Strategy,
}

pub enum Strategy { Greedy, Lazy, Optimal }
// error.rs
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CompressError { InputTooLarge, WindowTooSmall, LevelMismatch }

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DecompressError {
    BadMagic,           // frame magic mismatch
    Truncated,          // input ended before frame complete
    Overflow,           // declared size exceeds cap
    BadBlockType,       // unknown block type nibble
    BadHuffmanWeights,  // Kraft inequality violated
    BadFseTable,        // normalized counts don't sum to 2^log
    ChecksumMismatch,   // xxhash64 content checksum failure
    WindowTooLarge,     // window_log > decoder cap
    OutputLimitExceeded,// decoded output > caller-supplied limit
    SequenceError,      // match copy out of window bounds
    DictMissing(u32),   // dict_id required but not supplied
}
// sequence.rs
/// Fundamental LZ output atom: a run of `literal_length` literal bytes
/// followed by one match of `match_length` bytes at `offset` back.
/// literal_length > 0 && match_length == 0 => literals-only tail sequence.
/// offset ∈ 1..=3 => repeat-offset code (resolved via RepeatOffsets).
/// offset > 3 => real distance into Window.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Sequence {
    pub literal_length: u32,
    pub match_length:   u32,   // 0 for literals-only tail
    pub offset:         u32,   // raw distance; 0 only in literals-only tail
}

pub struct Match { pub length: u32, pub offset: u32 }
// window.rs
/// Immutable view into the sliding history buffer.
/// Lifetime 'a ties to the input + history allocation.
pub struct Window<'a> {
    base: *const u8,    // start of allocated window (size = 1 << window_log)
    size: usize,        // 1 << window_log
    cur: usize,         // current write position (mod size for circular access)
    _marker: core::marker::PhantomData<&'a u8>,
}
impl<'a> Window<'a> {
    /// Copy `len` bytes from `offset` positions back; bounds-checked.
    pub fn copy_match(&self, offset: u32, len: u32, dst: &mut Vec<u8>) -> Result<(), DecompressError>;
    pub fn byte_at_offset(&self, offset: usize) -> Option<u8>;
}
// hashchain.rs
/// Hash-chain match finder: head[hash(3 bytes)] -> most-recent pos;
/// prev[pos & mask] chains older occurrences. Walk bounded by search_depth + window.
pub struct HashChainFinder {
    head: Vec<u32>,     // len = 1 << hash_log; u32::MAX = empty
    prev: Vec<u32>,     // len = 1 << window_log; circular
    hash_log: u8,
    window_log: u8,
    min_match: u32,     // always 3
}

pub trait MatchFinder {
    /// Register current position's bytes into the index.
    fn insert(&mut self, pos: usize);
    /// Find the longest match at `pos`; returns None if < min_match.
    fn find(&mut self, pos: usize, max_len: usize) -> Option<Match>;
}
impl MatchFinder for HashChainFinder { /* hash 3 bytes; walk prev chain up to search_depth */ }
// bt_finder.rs
/// Binary-tree match finder for OptimalParser: maintains a binary search tree
/// over the sliding window; provides all match lengths up to target_length in
/// one pass — O(search_depth) comparisons vs O(search_depth * avg_chain) for chains.
pub struct BtFinder {
    greater: Vec<u32>,  // right child per node (position)
    less:    Vec<u32>,  // left child per node
    hash_log: u8,
    window_log: u8,
}
impl MatchFinder for BtFinder { /* insert + find update tree; split on mismatch byte */ }
// repeat.rs
/// LRU cache of the 3 most-recently used match offsets.
/// Rep-code 1 = most recent, 2 = second, 3 = third (zstd convention).
pub struct RepeatOffsets { pub recent: [u32; 3] }
impl RepeatOffsets {
    pub fn update(&mut self, used_offset: u32);
    /// Attempt to encode `offset` as a repeat-offset code (1..=3); returns None if not in cache.
    pub fn encode_rep(&self, offset: u32) -> Option<u32>;
    pub fn decode_rep(&self, code: u32) -> u32; // code ∈ 1..=3 -> real offset
}
// parse.rs
/// Single entry point: selects parser by Strategy; returns Sequences covering all of `input`.
pub fn parse_block(
    input: &[u8],
    finder: &mut dyn MatchFinder,
    strat: Strategy,
    reps: &mut RepeatOffsets,
) -> Vec<Sequence>;

// Internal parser types (not pub-exported):
struct GreedyParser;   // take first match >= min_match; no lookahead
struct LazyParser;     // check pos+1; if longer match, emit literal at pos, take later match
struct OptimalParser;  // price-based shortest-path (Dijkstra over block positions); uses BtFinder
// Price model: cost(literal) = -log2(p(lit)) bits; cost(match) = OF_bits + ML_bits + LL_bits
// histogram.rs
pub struct Histogram {
    pub counts: [u32; 256],
    pub total:  u32,
}
impl Histogram {
    pub fn from_bytes(data: &[u8]) -> Self;
    pub fn merge(&mut self, other: &Self);
    pub fn symbol_count(&self) -> usize; // number of symbols with count > 0
}
// huffman.rs
/// Canonical code table: code_lengths[sym] = 0 means symbol absent.
/// Codes derived from lengths by canonical assignment (sort by length, then symbol).
pub struct HuffmanTable { pub code_lengths: Box<[u8]> /* len=256 */ }

pub struct HuffmanEncoder {
    table: [(u32, u8); 256],  // (code, len) per symbol; len=0 => absent
}
pub struct HuffmanDecoder {
    fast_table: Vec<(u8, u8)>, // indexed by next `max_len` bits: (symbol, consumed_bits)
    max_len: u8,               // capped, e.g. 11
}

/// Length-limited via package-merge algorithm; max_len typically 11 (literals) or 6 (weights).
/// Panics if hist.total == 0; returns table satisfying Kraft inequality.
pub fn build_huffman(hist: &Histogram, max_len: u8) -> HuffmanTable;

impl HuffmanEncoder {
    pub fn from_table(t: &HuffmanTable) -> Self;
    pub fn encode_sym(&self, sym: u8, w: &mut BitWriter);
}
impl HuffmanDecoder {
    pub fn from_table(t: &HuffmanTable) -> Self;
    /// Decode one symbol; advances BitReader by consumed_bits.
    pub fn decode_sym(&self, r: &mut BitReader) -> Result<u8, DecompressError>;
}
// range.rs
/// [low, low+range) interval arithmetic coder; byte-oriented output.
/// Renormalize when range < 2^24: emit one byte, shift left by 8.
/// Carry propagation: deferred-byte count handles carry from low overflow.
pub struct RangeEncoder {
    low: u64,
    range: u32,
    cache: u8,          // deferred carry byte
    cache_size: u32,    // number of deferred 0xFF bytes
    out: BitWriter,
}
pub struct RangeDecoder<'a> {
    code: u32,
    range: u32,
    src: &'a mut BitReader<'a>,
}
impl RangeEncoder {
    pub fn encode(&mut self, cum_freq: u32, freq: u32, total: u32);
    pub fn flush(&mut self); // emit termination bytes
}
impl<'a> RangeDecoder<'a> {
    pub fn decode(&mut self, total: u32) -> u32; // returns cumulative freq slot
    pub fn normalize(&mut self) -> Result<(), DecompressError>;
}
// normalize.rs
/// Map raw counts to normalized counts summing to exactly 2^table_log.
/// Symbol with count > 0 but rounded to 0 gets count = 1 (marked as -1 in i16 table).
/// table_log ∈ [5, 12] for FSE; asserts hist.total > 0.
pub fn normalize_counts(hist: &Histogram, table_log: u8) -> Box<[i16]>; // len=256; -1 = low-prob
// fse.rs
/// tANS state machine table; log = table_log; table size = 2^log.
pub struct FseTable {
    pub log: u8,
    // Decode entries: one per state slot
    decode: Box<[FseDecodeEntry]>,  // len = 1 << log
    // Encode map: symbol -> (start_state, nb_bits, next_state_base) transform
    encode: Box<[FseEncodeEntry; 256]>,
}
#[derive(Clone, Copy)]
struct FseDecodeEntry { pub symbol: u8, pub nb_bits: u8, pub baseline: u32 }
#[derive(Clone, Copy)]
struct FseEncodeEntry { pub delta_nb_bits: u32, pub delta_state: i32 }

/// Spread symbols across [0, 2^log) by step = (5/8)*size+3; assign decode entries.
pub fn build_fse_decode(norm: &[i16], table_log: u8) -> FseTable;

pub struct FseEncoder { table: FseTable, state: u32 }
pub struct FseDecoder { table: FseTable, state: u32 }
impl FseEncoder {
    pub fn encode_sym(&mut self, sym: u8, w: &mut BitWriter);
    pub fn flush(&mut self, w: &mut BitWriter); // emit final state bits
}
impl FseDecoder {
    pub fn init(&mut self, r: &mut BitReader) -> Result<(), DecompressError>;
    pub fn decode_sym(&mut self, r: &mut BitReader) -> Result<u8, DecompressError>;
}
// frame.rs
/// Parsed frame header; validated fields only (post-parse invariants hold).
pub struct FrameHeader {
    pub magic: u32,                 // must equal FRAME_MAGIC = 0xFD2FB528 (zstd-style) or own magic
    pub window_log: u8,             // decoded window size = 1 << window_log; max enforced at decode
    pub content_size: Option<u64>,  // None if unknown; bounds-check output allocation
    pub dict_id: Option<u32>,       // None if no dict; error if dict required but not supplied
    pub has_checksum: bool,         // if true: 8-byte xxhash64 appended after last block
}
// block.rs
pub struct BlockHeader {
    pub last: bool,
    pub block_type: BlockType,
    pub size: u32,   // payload bytes following header; max 128 KiB for Compressed
}
pub enum BlockType { Raw, Rle, Compressed }
// lib.rs (comp-format)
/// Compress `input` to a complete frame byte vector.
/// Selects block type per block (Raw/RLE/Compressed — whichever is smallest).
/// Applies dict prefix content to MatchFinder if dict.is_some().
pub fn encode_frame(
    input: &[u8],
    params: &CompressParams,
    dict: Option<&Dictionary>,
) -> Result<Vec<u8>, CompressError>;

/// Decode a complete frame from `input`; verifies checksum if present.
/// Enforces window_log ≤ DECODER_WINDOW_LOG_MAX (27) and output ≤ OUTPUT_SIZE_MAX.
/// Returns Err on any malformed field — never panics on adversarial input.
pub fn decode_frame(
    input: &[u8],
    dict: Option<&Dictionary>,
) -> Result<Vec<u8>, DecompressError>;
// dict.rs
pub struct EntropyTables {
    pub huf_table: HuffmanTable,    // for literals; pre-built from dict samples
    pub ll_table:  FseTable,        // literal-length FSE table
    pub ml_table:  FseTable,        // match-length FSE table
    pub of_table:  FseTable,        // offset FSE table
}

/// Wire format: [DICT_MAGIC:4][id:4][entropy_tables:var][content:var]
/// entropy_tables serialized in same layout as CompressedBlock header tables.
pub struct Dictionary {
    pub id:      u32,
    pub entropy: EntropyTables,
    pub content: Box<[u8]>,   // prefix bytes; fed into Window before first block
}
impl Dictionary {
    pub fn serialize(&self) -> Vec<u8>;
    pub fn deserialize(data: &[u8]) -> Result<Self, DecompressError>;
}
// Dictionary is Send + Sync: immutable after construction; shared across threads.

// train.rs
/// Select representative byte segments from `samples` using COVER-lite / suffix heuristic.
/// Enforces dict_size ≤ 112 KiB (zstd convention); builds entropy tables from sample histograms.
pub fn train(samples: &[&[u8]], dict_size: usize) -> Result<Dictionary, CompressError>;
BitWriter / BitReader  ←→  HuffmanEncoder/Decoder, FseEncoder/Decoder, RangeEncoder/Decoder
Histogram              →   build_huffman, normalize_counts → HuffmanTable, FseTable
Vec<Sequence>          →   codes.rs (Sequence→LL/ML/OF streams + extra bits)
LL/ML/OF streams       →   FseEncoder (literals: HuffmanEncoder) → CompressedBlock bytes
CompressedBlock bytes  →   FrameHeader + BlockHeader → frame bytes
frame bytes            →   Checksum.update() → appended xxhash64 (if has_checksum)
Dictionary.content     →   Window pre-fill; Dictionary.entropy → prime encoder tables
