Core schema & types (§11)

// crates/mlc-span/src/lib.rs
pub struct FileId(pub u32);                 // Copy, Eq, Hash
pub struct Span { pub file: FileId, pub lo: u32, pub hi: u32 }  // byte offsets
pub struct SourceFile { pub id: FileId, pub path: PathBuf, pub source: Arc<str>, pub line_starts: Vec<u32> }
pub struct FileMap { /* files: Vec<SourceFile>, ownership: FileMap-owned */ }

// crates/mlc-id/src/lib.rs
pub struct NodeId(pub u32);                 // Copy, Eq, Hash
pub struct DefId(pub u32);                  // Copy, Eq, Hash
pub struct TyVar(pub u32);                  // Copy, Eq, Hash
pub struct IdGen { /* next: AtomicU32 */ }
impl IdGen {
    pub fn fresh_node(&self) -> NodeId;
    pub fn fresh_def(&self) -> DefId;
    pub fn fresh_ty(&self) -> TyVar;
}
// crates/mlc-lex/src/lib.rs
pub enum TokKind {
    Ident, Constructor, TyVar, Int, Float, String, Char,
    Let, Rec, In, Fun, If, Then, Else, Match, With, Module, Import, Type, And, As, Where, Mutable, Open,
    Plus, Minus, Star, Slash, Mod, Eq, EqEq, Neq, Lt, Le, Gt, Ge, AmpAmp, PipePipe,
    Pipe, Arrow, FatArrow, ColonColon, Bar, LParen, RParen, LBrace, RBrace, LBracket, RBracket,
    Comma, Colon, Semi, PipeGt, PlusDot, MinusDot, StarDot, SlashDot, GtDot,
    Newline, Eof, Error,
}
pub struct Token { pub kind: TokKind, pub span: Span, pub text: Arc<str> }  // text empty for keywords
// crates/mlc-ast/src/lib.rs
pub struct Module<'a> { pub id: NodeId, pub name: Option<Symbol>, pub items: &'a [Item<'a>], pub span: Span }
pub enum Item<'a> {
    Import { path: ModPath, span: Span, id: NodeId },
    TypeDef { name: Symbol, params: &'a [Symbol], def: TypeDef<'a>, span: Span, id: NodeId },
    LetBinding(LetBinding<'a>),
    Open { path: ModPath, span: Span, id: NodeId },
}
pub struct LetBinding<'a> {
    pub id: NodeId, pub rec: bool, pub name: Symbol,
    pub params: &'a [Pat<'a>], pub ascription: Option<&'a Type<'a>>,
    pub body: &'a Expr<'a>, pub span: Span,
}
pub enum Expr<'a> {
    Lit { lit: Lit, span: Span, id: NodeId },
    Path { path: Path, span: Span, id: NodeId },
    Lam { params: &'a [Pat<'a>], body: &'a Expr<'a>, span: Span, id: NodeId },
    App { fun: &'a Expr<'a>, arg: &'a Expr<'a>, span: Span, id: NodeId },
    Let { rec: bool, bindings: &'a [LetBinding<'a>], body: &'a Expr<'a>, span: Span, id: NodeId },
    If { cond: &'a Expr<'a>, then_: &'a Expr<'a>, else_: &'a Expr<'a>, span: Span, id: NodeId },
    Match { scrutinee: &'a Expr<'a>, arms: &'a [Arm<'a>], span: Span, id: NodeId },
    BinOp { op: BinOp, lhs: &'a Expr<'a>, rhs: &'a Expr<'a>, span: Span, id: NodeId },
    Error(Span, NodeId),
}
pub enum Pat<'a> {
    Wild { span: Span, id: NodeId },
    Var { name: Symbol, span: Span, id: NodeId },
    Constructor { name: Path, args: &'a [Pat<'a>], span: Span, id: NodeId },
    Or { lhs: &'a Pat<'a>, rhs: &'a Pat<'a>, span: Span, id: NodeId },
}
pub enum Type<'a> {
    Var { name: Symbol, span: Span, id: NodeId },
    Con { name: Path, args: &'a [Type<'a>], span: Span, id: NodeId },
    Arrow { from: &'a Type<'a>, to: &'a Type<'a>, span: Span, id: NodeId },
}
// crates/mlc-resolve/src/lib.rs
pub struct ResolvedModule<'a> { pub name: ModName, pub items: &'a [ResolvedItem<'a>], pub def_table: DefTable<'a> }
pub struct DefTable<'a> { /* defs: HashMap<DefId, DefRecord<'a>> */ }
pub struct DefRecord<'a> { pub id: DefId, pub name: Symbol, pub kind: DefKind, pub span: Span, pub home_module: ModName }
pub enum DefKind { Value, Type, Constructor, Module, TypeVar }
// crates/mlc-types/src/lib.rs
pub enum Ty {
    Int, Float, Bool, Char, String, Unit,
    Var(TyVar), Con(TyConId, Vec<Ty>), Arrow(Box<Ty>, Box<Ty>), Tuple(Vec<Ty>),
    Record(RecordRow), Forall(Vec<TyVar>, Box<Ty>), Err,
}
pub struct RecordRow { pub fields: Vec<(Symbol, Ty)>, pub tail: RowTail }
pub enum RowTail { Closed, Open(TyVar) }
pub struct UnificationCtx { /* table: UnionFind<TyVar, Ty>, level: u32 */ }
// crates/mlc-ir/src/lib.rs
pub struct Program { pub fns: Vec<FunctionId>, pub fn_table: HashMap<FunctionId, Function>, pub ty_table: HashMap<TypeId, IrType> }
pub struct Function { pub id: FunctionId, pub sig: Signature, pub blocks: Vec<BasicBlock>, pub entry: BlockId, pub locals: Vec<Local> }
pub struct BasicBlock { pub id: BlockId, pub params: Vec<Local>, pub instructions: Vec<Inst>, pub terminator: Term, pub preds: Vec<BlockId> }
pub enum Inst {
    Const(LocalId, Constant), BinOp(LocalId, BinOp, LocalId, LocalId),
    Call(LocalId, FunctionId, Vec<LocalId>), Alloc(LocalId, IrType),
    Load(LocalId, LocalId, FieldOffset), Store(LocalId, LocalId, FieldOffset),
    Closure(LocalId, FunctionId, Vec<LocalId>), Tag(LocalId, LocalId, Tag),
}
pub enum Term {
    Br(BlockId, Vec<LocalId>), CondBr(LocalId, BlockId, Vec<LocalId>, BlockId, Vec<LocalId>),
    Switch(LocalId, Vec<(Discriminant, BlockId, Vec<LocalId>)>, BlockId),
    Return(Option<LocalId>), TailCall(FunctionId, Vec<LocalId>), Unreachable,
}
// crates/mlc-diag/src/lib.rs
pub struct Diagnostic { pub code: DiagCode, pub severity: Severity, pub message: String, pub primary: Label, pub secondaries: Vec<Label>, pub help: Option<String> }
pub struct Label { pub span: Span, pub message: String }
pub struct DiagCode(pub u16);               // E0001..E0999
pub enum Severity { Error, Warning, Note, Help, Bug }
pub trait IntoDiagnostic { fn into_diagnostic(self) -> Diagnostic; }
