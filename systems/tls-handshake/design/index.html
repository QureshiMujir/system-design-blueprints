Core schema & types (§11)

enum ContentType { Invalid, ChangeCipherSpec, Alert, Handshake, ApplicationData }
// [PEER]-derived after AEAD-open; not persisted; not serialized beyond the record

struct ProtocolVersion(u16);
// immutable, Copy, no secret material, compatibility: legacy 0x0303 frozen by RFC 8446

struct TlsPlaintext { content_type: ContentType, legacy_record_version: ProtocolVersion,
    fragment: Vec<u8> /* [PEER] pre-decrypt bytes */ }
// mutation: constructed once per record; no persistence; not directly serialized (wrapped by TlsCiphertext on send)

struct TlsInnerPlaintext { content: Vec<u8> /* [PEER] post-decrypt */, real_type: ContentType, padding_len: usize }
// exists only transiently inside the AEAD boundary; zeroized on drop if content_type indicates handshake secret material

struct TlsCiphertext { opaque_type: ContentType /* always ApplicationData post-handshake */, legacy_record_version: ProtocolVersion,
    encrypted_record: Vec<u8> /* [PEER] on receive */ }
// serialization: wire format per §8.3; no compatibility concern beyond frozen outer version
struct RecordLayer { send_keys: Option<TrafficKeys>, recv_keys: Option<TrafficKeys>,
    send_seq: SequenceNumber, recv_seq: SequenceNumber }
// owns SECRET material; single-threaded per direction; not persisted; not Clone

struct SequenceNumber(u64);
// mutation: increment-only, checked-add with rekey trigger before overflow (I2); Copy

struct Nonce([u8; 12]);
// derived per-record from static_iv XOR seq (§8.3); zeroized immediately after AEAD call; not logged

struct AeadKey(Vec<u8> /* zeroize on drop */);
// SECRET; constant-time comparison only; never Debug/Display; scope = one traffic-secret epoch

enum AeadAlgorithm { Aes128Gcm, Aes256Gcm, ChaCha20Poly1305 }
// immutable negotiated value; Copy; no secret material
enum HandshakeType { ClientHello, ServerHello, EncryptedExtensions, Certificate,
    CertificateVerify, Finished, NewSessionTicket, KeyUpdate, HelloRetryRequest,
    CertificateRequest, EndOfEarlyData }
// [PEER]-observed on decode; immutable once parsed

struct HandshakeMessage { msg_type: HandshakeType, body: Vec<u8> /* [PEER] raw, pre-typed-parse */ }
// transient; consumed into a typed struct below; feeds TranscriptHash before being dropped

struct ClientHello { legacy_version: ProtocolVersion, random: [u8; 32] /* [PEER] on recv, local RNG on send */,
    legacy_session_id: Vec<u8>, cipher_suites: Vec<u16>, extensions: Vec<Extension> }
// [PEER] entirely on the receive path; compatibility: legacy_session_id echoed, not interpreted (RFC 8446 section 4.1.2)

struct ServerHello { legacy_version: ProtocolVersion, random: [u8; 32] /* may carry downgrade/HRR sentinel */,
    legacy_session_id_echo: Vec<u8>, cipher_suite: u16, extensions: Vec<Extension> }
// [PEER] on client receive path; sentinel check is mandatory, not optional (I7)
struct Extension { extension_type: ExtensionType, data: Vec<u8> /* [PEER] opaque, parsed per type */ }
enum ExtensionType { SupportedVersions, SupportedGroups, KeyShare, SignatureAlgorithms,
    PreSharedKey, PskKeyExchangeModes, ServerNameIndication, ApplicationLayerProtocolNegotiation }
struct KeyShareEntry { group: NamedGroup, key_exchange: Vec<u8> /* [PEER] public value only, never secret */ }
enum NamedGroup { X25519, Secp256r1, X448, Secp384r1 }
enum SignatureScheme { RsaPssRsaeSha256, RsaPssRsaeSha384, EcdsaSecp256r1Sha256, Ed25519 }
struct PskIdentity { identity: Vec<u8> /* [PEER] opaque ticket label */, obfuscated_ticket_age: u32 }
// all [PEER] on receive; NamedGroup/SignatureScheme are Copy enums with no secret material
enum HandshakeState { /* shared base before role split */ }
enum ClientState { Start, WaitServerHello, WaitEncryptedExtensions, WaitCertOrCertRequest,
    WaitCertificate, WaitCertVerify, WaitFinished, SendFinished, Connected, Closed }
enum ServerState { Start, ChooseParameters, SendHelloRetryRequest, SendServerHello,
    SendEncryptedExtensions, SendCertRequestOrCert, SendCertificate, SendCertVerify,
    SendFinished, WaitClientFinished, Connected, Closed }
// each variant carries only fields legal in that state (ADR-08); not Clone (owns TrafficKeys
// transitively); mutation: transition fn consumes self, returns new state (typestate pattern)
struct TranscriptHash { hasher: /* SHA-256 or SHA-384 running state */ HashAlg, retry_reset_called: bool }
// mutation: append-only via update(); retry_reset() callable exactly once (I13); not persisted; not Clone

struct KeySchedule { stage: KeyScheduleStage, early_secret: Option<Secret>, handshake_secret: Option<Secret>,
    master_secret: Option<Secret> }
enum KeyScheduleStage { Early, Handshake, Application }
// owns SECRET material end to end; every field zeroize-on-drop; not Serialize; not Clone

struct Secret(Vec<u8> /* zeroize on drop, ADR-21 */);
// SECRET; constant-time-compared only; scope = single derivation epoch; never logged (P12, I5)

struct TrafficKeys { key: AeadKey, iv: [u8; 12] /* zeroize on drop */ }
// SECRET; derived once per epoch; feeds RecordLayer; not persisted across process restart

struct EarlyData { plaintext: Vec<u8> /* [PEER] on recv path */, replayable: bool /* always true, P9 */ }
// mutation: single-use per ticket (I10); not Clone (prevents accidental replay-by-copy)
struct Certificate { der: Vec<u8> /* [PEER], raw */, parsed: Option<X509Fields> }
struct CertChain { certs: Vec<Certificate> /* [PEER], leaf-first order */ }
struct TrustAnchor { subject: Vec<u8>, public_key: Vec<u8> /* locally configured, NOT [PEER] */ }
struct PathValidator { anchors: Vec<TrustAnchor>, max_chain_len: usize /* DoS guard, T10 */ }
// PathValidator holds no secret material; CertChain is [PEER] until validate() returns Ok (I4, P3)

struct SignatureVerifier { scheme: SignatureScheme }
// stateless per call; verifies over §8.9's exact signed_content construction

struct SessionTicket { ticket_bytes: Vec<u8> /* [PEER] on client recv, sent opaque on issue */,
    lifetime_secs: u32, age_add: u32 }
struct ResumptionSecret(Vec<u8> /* zeroize on drop */);
// SessionTicket label is [PEER]-opaque to the client; ResumptionSecret is SECRET, server-local

enum Alert { Warning(AlertDescription), Fatal(AlertDescription) }
enum AlertLevel { Warning, Fatal }
enum AlertDescription { CloseNotify, UnexpectedMessage, BadRecordMac, RecordOverflow,
    HandshakeFailure, BadCertificate, CertificateExpired, IllegalParameter, UnknownCa,
    CertificateRevoked, DecodeError, DecryptError, ProtocolVersion, MissingExtension }
enum TlsError { Alert(Alert), Io(/* transport error, opaque */), Parse(DecodeError) }
// TlsError maps 1:1 to AlertDescription per ADR-20; no secret material in any variant
