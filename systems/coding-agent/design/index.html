Core schema & types (§11)

TaskId: NewType[str]                    // opaque UUIDv4; trusted; immutable; JSON string

TaskSpec:                                // trusted container, UNTRUSTED instruction field
  id: TaskId                              // trusted, set at intake, immutable
  repo_ref: RepoSnapshot                   // trusted
  instruction: str                          // UNTRUSTED (may originate from an issue body)
  constraints: BudgetAllocation             // trusted
  created_at: Timestamp                      // trusted
  // ownership: created by task-spec parser; process-local then persisted to trajectory
  // header; JSON-serializable; schema-versioned (ADR-32)

RepoSnapshot:                             // trusted metadata about an UNTRUSTED content tree
  commit_sha: str
  root_path: Path
  file_count: int
  loc_estimate: int
  // ownership: created by repository indexer; immutable per task; not concurrently mutated
  // during a task (sandbox overlay isolates writes, ADR-12)

FileRef:
  path: Path                               // relative to repo root; canonicalized (I8)
  content_hash: ContentHash                 // trusted digest of UNTRUSTED file content

ContentHash: NewType[str]                 // SHA-256 hex digest; used by I2

SymbolIndex:                               // trusted structural index over UNTRUSTED source
  symbols: dict[str, list[FileRef]]
  built_at: Timestamp
  // ownership: repository indexer; rebuilt incrementally on mtime change; persisted to
  // SQLite (ADR-08); not mutated mid-task

RetrievalQuery:
  text: str                                 // may embed UNTRUSTED task instruction fragments
  mode: Literal["lexical", "structural", "embedding"]
  max_results: int

RetrievedChunk:                             // UNTRUSTED content, trusted provenance metadata
  file: FileRef
  span: tuple[int, int]
  content: str                               // UNTRUSTED
  trust_tag: Literal["untrusted_repo_content"]  // P3
TokenBudget:
  window: int                               // section 4 constant, 200_000
  reserved_output: int                        // 8_000
  system_reserved: int                        // 3_000
  task_reserved: int                          // 1_000

BudgetAllocation:                            // section 8.3 arithmetic, enforced pre-flight
  system: int
  task: int
  retrieved: int
  history: int
  reserved_output: int
  // invariant: sum(fields) <= TokenBudget.window (I13); constructed by budgeter only

PromptAssembly:
  system_segment: str                        // trusted
  task_segment: str                           // UNTRUSTED instruction embedded, tagged
  retrieved_segments: list[RetrievedChunk]     // UNTRUSTED, tagged
  history_segment: str                         // mixed trust, compacted (section 8.3)
  // ownership: prompt assembler; single-use per turn; not persisted beyond trajectory ref

ModelClient(Protocol):                       // section 5 RFC: provider tool-calling API
  def send(request: ModelRequest) -> ModelResponse: ...
  // implementations: live provider client, replay_transport (P10); versioned per ADR-02

ModelRequest:
  prompt: PromptAssembly
  tools: list[ToolSpec]
  temperature: float                          // ADR-20
  max_output_tokens: int

ModelResponse:
  text: str | None
  tool_calls: list[ToolCall]
  stop_reason: StopReason
  usage: tuple[int, int]                       // (input_tokens, output_tokens)

StopReason: Literal["tool_use", "end_turn", "max_tokens", "error"]
ToolSpec:                                    // trusted, JSON-Schema-bearing (I3, ADR-04)
  name: str
  json_schema: dict                             // JSON Schema spec (section 5 RFC)
  effect_class: EffectClass
  idempotent: bool
  timeout_sec: int

ToolCall:                                     // model-authored, validated before use
  tool_name: str
  arguments: dict                                // validated against ToolSpec.json_schema
  call_id: str

ToolResult:
  call_id: str
  ok: bool
  payload: str                                    // may be UNTRUSTED (e.g., file content, shell stdout)
  trust_tag: Literal["untrusted_tool_output"]        // P3

EffectClass: Literal["read_only", "mutating", "executing"]   // P12
EditOp:                                       // section 8.4, ADR-05
  path: FileRef
  anchor: str                                    // must match exactly once (I2 precondition input)
  replacement: str
  precondition: EditPrecondition

EditPrecondition:
  expected_hash: ContentHash                      // I2
  anchor_occurrence_count: int                     // must equal 1 or reject (ambiguous)

PatchSet:                                     // section 21, size-limited (section 4)
  task_id: TaskId
  edits: list[EditOp]
  diff_text: str
  changed_files: int
  changed_lines: int

ConflictReport:
  path: FileRef
  reason: Literal["stale_hash", "ambiguous_anchor", "anchor_absent", "path_outside_root"]
SandboxSpec:                                  // ADR-11..14
  image_ref: str
  mounts: list[tuple[Path, Path, Literal["ro", "rw"]]]
  network_policy: Literal["deny_all", "allowlist"]
  cpu_limit: float
  mem_limit_mb: int
  timeout_sec: int

SandboxHandle:
  sandbox_id: str
  pid: int
  spawned_at: Timestamp
  // ownership: sandbox pool; released back to pool on task completion (ADR-34)

CommandResult:
  exit_code: int
  stdout: str                                    // UNTRUSTED
  stderr: str                                    // UNTRUSTED
  duration_ms: int
  truncated: bool

VerificationRun:                               // section 8.6, section 19
  task_id: TaskId
  command: str
  outcome: TestOutcome
  quorum_runs: int                                // ADR-17
  duration_ms: int

TestOutcome: Literal["pass", "fail", "timeout", "flaky_suspect"]
AgentState: Literal["Planning", "Retrieving", "Acting", "Observing", "Verifying",
                     "Reflecting", "Done", "Failed", "Aborted"]            // section 8.1

Turn:                                          // one row of a Trajectory
  index: int
  state_from: AgentState
  state_to: AgentState
  model_response_ref: str
  tool_calls: list[ToolCall]
  tool_results: list[ToolResult]
  token_usage: tuple[int, int]
  cost_delta: float
  wall_clock_ms: int

Trajectory:                                    // I7, ADR-21, append-only JSONL
  task_id: TaskId
  turns: list[Turn]
  final_state: AgentState
  schema_version: str                            // section 27

CostLedger:                                    // I10, P13
  task_id: TaskId
  entries: list[tuple[int, float]]                // (turn_index, cost_delta)
  running_total: float
  soft_ceiling: float
  hard_ceiling: float

EvalCase:                                      // section 23
  case_id: str
  task_spec: TaskSpec
  gold_patch_ref: str
  hidden_tests_ref: str

EvalReport:
  case_id: str
  outcome: Literal["resolved", "unresolved", "error"]
  turns_used: int
  cost: float
  wall_clock_ms: int
// Error taxonomy (all inherit from AgentError; trust-relevant errors are marked)
AgentError: Protocol
  SchemaValidationError(AgentError)             // I3
  StalePreconditionError(AgentError)             // I2
  AmbiguousAnchorError(AgentError)                // section 8.4
  BudgetExceededError(AgentError)                  // I4
  SandboxTimeoutError(AgentError)                   // section 8.1
  SandboxEscapeSuspectedError(AgentError)            // I1, T2 — highest severity
  VerificationHangError(AgentError)
  InjectionDetectedError(AgentError)                 // UNTRUSTED-content-triggered, section 22
  CostCeilingExceededError(AgentError)                 // I10/T7
