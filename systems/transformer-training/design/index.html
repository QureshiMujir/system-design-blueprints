Core schema & types (§11)

@dataclass(frozen=True)
class TrainConfig:
    run_id: str; seed: int; output_dir: Path
    shard_glob: str; eval_shard_glob: str
    sequence_length: int; micro_batch_size: int; grad_accum_steps: int
    pack_sequences: bool; eos_token_id: int          # ADR-17
    model: "ModelConfig"                              # §11.1
    optimizer: str; learning_rate: float; weight_decay: float
    betas: tuple[float, float]; eps: float; grad_clip: float   # ADR-12
    warmup_steps: int; max_steps: int; min_lr_ratio: float     # ADR-08
    precision: str; autocast_dtype: str                        # ADR-09
    log_every_steps: int; eval_every_steps: int
    ckpt_every_steps: int; ckpt_retain_last: int; ckpt_retain_every: int
    dp_world_size: int; sharding: str                 # "none"|"ddp"|"fsdp_zero2", ADR-24
    wandb_project: Optional[str] = None
    jsonl_path: Optional[Path] = None
    deterministic_cudnn: bool = True
    torch_compile: bool = False; torch_compile_mode: str = "reduce-overhead"  # ADR-20
    git_commit: Optional[str] = None; config_hash: Optional[str] = None       # I8

@dataclass(frozen=True)
class ModelConfig:
    vocab_size: int; n_layer: int; n_head: int; n_kv_head: Optional[int]
    d_model: int; d_ff: Optional[int]; sequence_length: int
    norm: str; norm_eps: float                        # ADR-04
    positions: str; rope_base: float                  # ADR-05
    mlp_activation: str                                # ADR-06
    tie_word_embeddings: bool                          # ADR-14, I10
    init_scheme: str; init_std: float
    def derive(self) -> "ModelConfig": ...             # fills d_ff (I11-adjacent)
@dataclass(frozen=True)
class DataShard:
    path: Path; n_tokens: int; sha256: str; epoch_seen: int = 0

@dataclass(frozen=True)
class ShardCatalog:
    version: int; name: str; vocab_size: int
    eos_token_id: int; bos_token_id: Optional[int]
    shards: list[DataShard]; total_tokens: int
    @classmethod
    def load(cls, path: Path) -> "ShardCatalog": ...
    def integrity_check(self) -> None: ...            # ADR-15, I2-adjacent
@dataclass
class Batch:
    tokens: "torch.Tensor"           # (B,T) int64, cuda
    targets: "torch.Tensor"          # (B,T) int64, shifted
    pos_ids: "torch.Tensor"          # (B,T) int64
    attn_mask: Optional["torch.Tensor"] = None   # (B,T) bool, ADR-17
    step: int = 0; micro: int = 0

class DataPipeline:
    """Stateful iterator over Batch; bit-exact across resumes (I1, I2)."""
    def __init__(self, cfg: TrainConfig, catalog: ShardCatalog): ...
    def __iter__(self) -> "Iterator[Batch]": ...
    def state_dict(self) -> dict: ...       # (epoch, shard_idx, offset_idx)
    def load_state_dict(self, sd: dict) -> None: ...
@dataclass
class OptimizerState:
    step: int
    exp_avg: dict[str, "torch.Tensor"]       # fp32, I4
    exp_avg_sq: dict[str, "torch.Tensor"]    # fp32, I4
    master_params: Optional[dict[str, "torch.Tensor"]] = None

@dataclass(frozen=True)
class CheckpointMeta:
    version: int; run_id: str; step: int; epoch: int
    dp_rank: int; dp_world_size: int
    model_config: ModelConfig; train_config_hash: str
    rng_state: bytes                          # I12, RngSnapshot pickle
    created_at: str; git_commit: str
    sha256_model: str; sha256_opt: str        # I7 integrity
@dataclass(frozen=True)
class LossPoint:
    step: int; tokens_seen: int; loss: float
    grad_norm: float; grad_norm_clipped: float; lr: float
    step_time_ms: float; tokens_per_s: float; gpu_mem_mb: float
    spike: bool                                # I9

@dataclass(frozen=True)
class EvalPoint:
    step: int; eval_loss: float; eval_ppl: float
    bench_scores: dict[str, float]

@dataclass
class RngSnapshot:
    torch_cpu_state: bytes; torch_cuda_states: list[bytes]
    numpy_state: bytes; python_state: bytes
    data_pipeline_state: bytes                 # I12
def capture_rng() -> RngSnapshot: ...
def restore_rng(snap: RngSnapshot) -> None: ...
class Trainer:
    def __init__(self, cfg: TrainConfig, model: "torch.nn.Module",
                 opt: "Optimizer", sched: "Scheduler", data: DataPipeline,
                 evalh: "EvalHarness", ckpter: "Checkpointer",
                 logger: "Logger"): ...
    def train(self) -> None: ...
    def step(self, batch: Batch) -> LossPoint: ...
    def resume(self, path: Path) -> None: ...

class Checkpointer:
    def save(self, step: int, model, opt, sched, data, rng: RngSnapshot) -> Path: ...
    def load(self, path: Path) -> CheckpointMeta: ...
    def list(self) -> list[CheckpointMeta]: ...

class EvalHarness:
    def evaluate(self, step: int, model: "torch.nn.Module") -> EvalPoint: ...
