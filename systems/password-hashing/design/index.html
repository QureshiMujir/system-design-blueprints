Core schema & types (§11)

//! crates/password-hash-core/src/algorithm.rs
#[derive(Clone, Copy, Debug, Eq, PartialEq, Hash)]
pub enum Algorithm {
    Pbkdf2HmacSha256,
    Pbkdf2HmacSha512,
    Bcrypt,
    Scrypt,
    Argon2i,
    Argon2d,
    Argon2id,
}
// Ownership: Copy, no allocation. Concurrency: Send + Sync (stateless).
// Persistence: encoded as the PHC algorithm-id token (§8.1) or bcrypt's `$2*$` (§8.3).
// Compatibility: adding a variant is additive/non-breaking; removing one is a SemVer major (§25).
//! crates/password-hash-core/src/params.rs
#[derive(Clone, Copy, Debug)]
pub enum Params {
    Pbkdf2 { iterations: u32 },
    Bcrypt { cost: u8 },                                 // 4..=31
    Scrypt { ln: u8, r: u32, p: u32 },                   // N = 2^ln
    Argon2 { time_cost: u32, memory_kib: u32, parallelism: u32, output_len: u32 },
}

pub const PARAMS_2026: Params = Params::Argon2 {
    time_cost: 3, memory_kib: 65_536, parallelism: 4, output_len: 32,
};

pub fn enforce_floor(algo: Algorithm, params: &Params) -> Result<(), Error>;
// Ownership: Params is Copy. Mutation: none post-construction (immutable value type).
// Concurrency: no shared state — pure function. Persistence: encoded inline in the PHC param-list.
// Serialization: `to_param_list()` / `from_param_list()` companions in §18. I3, I13 bind here.
//! crates/password-hash-core/src/salt.rs
#[derive(Clone, ZeroizeOnDrop)]
pub struct Salt(Vec<u8>);
impl Salt {
    pub fn generate(len: usize) -> Result<Self, Error>;   // len >= 16 enforced (I5)
    #[cfg(any(test, feature = "kat"))]
    pub fn from_fixed(bytes: Vec<u8>) -> Self;             // ADR-13 test-only escape hatch
}

//! crates/password-hash-core/src/pepper.rs
#[derive(Clone, ZeroizeOnDrop)]
pub struct Pepper(Vec<u8>);
// Ownership: caller-constructed, never persisted by this crate (P12).
// Mutation: immutable after construction. Concurrency: Send, not Sync (single-use recommended).
//! crates/password-hash-core/src/encoded.rs
#[derive(Clone, Debug, Eq, PartialEq)]
pub struct EncodedHash(String);
impl EncodedHash {
    pub fn as_str(&self) -> &str;
    pub fn parse(s: &str) -> Result<ParsedHash, ParseError>;   // §8.1–8.3, §18
}

#[derive(Clone, Debug)]
pub struct ParsedHash {
    pub algorithm: Algorithm,
    pub params: Params,
    pub salt: Vec<u8>,
    pub hash: Vec<u8>,
    pub legacy: bool,                 // true forces needs_rehash = true (I6)
}
// Persistence: EncodedHash is the sole caller-facing storage type (opaque String, I1 round-trip).
// Serialization: Display for EncodedHash == PHC-or-legacy string exactly.
//! crates/password-hash-core/src/verify.rs
pub enum VerifyOutcome {
    Authenticated { needs_rehash: bool },
}
pub enum AuthError {
    WrongPassword,
    Malformed,
    Alloc,
    WeakParameters,
}
pub fn verify(password: &Password, encoded: &str) -> Result<VerifyOutcome, AuthError>;
pub fn verify_with_pepper(password: &Password, pepper: &Pepper, encoded: &str)
    -> Result<VerifyOutcome, AuthError>;
// Concurrency: Send + Sync, reentrant, no shared mutable state (§3 concurrency boundary).
// I2 (constant-time), I11 (pepper symmetry), I14 (needs_rehash freshness) all bind here.
//! crates/password-hash-core/src/lib.rs
#[derive(ZeroizeOnDrop)]
pub struct Password(Zeroizing<Vec<u8>>);
impl Password {
    pub fn new(bytes: Vec<u8>) -> Self;
}
impl core::fmt::Debug for Password {
    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result;  // prints "Password(REDACTED)"
}
// Ownership: exclusive; not Clone (avoids incidental copies of secret material).
// P14, I-adjacent: this type is the only legal way plaintext enters the API.
//! crates/argon2-impl/src/memory.rs
pub struct Memory {
    lanes: u32,
    lane_length: u32,
    blocks: Box<[[u8; 1024]]>,        // lanes * lane_length entries, §8.4
}
impl Memory {
    pub fn allocate(memory_kib: u32, parallelism: u32) -> Result<Self, Error>;   // I4
    pub fn block(&self, lane: u32, index: u32) -> &[u8; 1024];
    pub fn block_mut(&mut self, lane: u32, index: u32) -> &mut [u8; 1024];
}
// Ownership: exclusive per hash() call (§3 concurrency boundary). Mutation: fill-phase only,
// write-once per block. Persistence: none — scoped to one call's stack/heap lifetime.
