Core schema & types (§11)

struct LogRecord {                        // meridian-model::record
    timestamp_unix_nano: i64,             // owned; immutable post-construction; serialized always
    observed_timestamp_unix_nano: i64,    // owned; set at ingest (I7)
    severity_number: u32,                 // 1..=24 (OTel range); validated at construction
    severity_text: Option<String>,        // owned
    body: Value,                          // owned; may be redacted in place (mutable pre-ship)
    attributes: Vec<(String, Value)>,     // owned; insertion order preserved dev-mode, canonicalized prod
    resource: Resource,                   // shared (Arc) across records from one producer
    scope: InstrumentationScope,          // shared (Arc)
    trace_id: Option<[u8; 16]>,           // owned; injected, never overwritten downstream (I11)
    span_id: Option<[u8; 8]>,             // owned
    trace_flags: Option<u8>,              // owned
    dropped_attributes_count: u32,        // owned; incremented by cardinality gate (I9)
    schema_version: u16,                  // owned; compatibility dispatch (I13)
}

enum Value {                              // meridian-model::value — tagged union, OTel-shaped
    Bool(bool), Int(i64), Double(f64), String(String),
    Bytes(Vec<u8>), Array(Vec<Value>), KvList(Vec<(String, Value)>),
}                                         // owned; recursive; serialization is per-format (§16)

enum Severity { Trace, Debug, Info, Warn, Error, Fatal }
                                          // owned; Copy; maps 1:1 to severity_number ranges

struct Resource { attributes: Vec<(String, Value)> }
                                          // shared (Arc); immutable after LoggerProvider construction

struct InstrumentationScope { name: String, version: Option<String> }
                                          // shared (Arc); immutable

struct Logger { /* opaque */ }            // meridian-sdk::logger — value-type (P12); Clone is cheap (Arc inner)
// methods (signatures only): log(level, msg, attrs) -> (); with_attrs(attrs) -> Logger

trait LogProcessor {                      // meridian-sdk — mutation: internal queue; concurrency: Send+Sync
    fn on_emit(&self, record: LogRecord);
    fn force_flush(&self, timeout: Duration) -> Result<(), FlushError>;
    fn shutdown(&self);
}                                         // impls: SimpleLogProcessor (sync), BatchLogProcessor (async, default)

trait LogExporter: Send + Sync {          // meridian-sink::trait_def
    fn export(&self, batch: &[LogRecord]) -> Result<(), ExportError>;
    fn shutdown(&self);
}                                         // impls: StdoutExporter, FileExporter, OtlpGrpcExporter,
                                          //   OtlpHttpExporter, SyslogExporter, TcpExporter, InMemoryExporter
struct WalSegment {                       // meridian-buffer::wal — persistence: on-disk file
    path: PathBuf, offset: u64, sealed: bool,
}                                         // mutation: append-only until sealed (I10)

struct CheckpointManifest {               // meridian-buffer::wal::checkpoint — persistence: on-disk
    segments: Vec<WalSegment>, last_shipped_offset: u64,
}                                         // serialization: versioned (I13); rewritten atomically (I10)

struct Pipeline { stages: Vec<PipelineStage> }
                                          // meridian-aggregator::pipeline — owned; order is fixed (§8, I5)

enum PipelineStage {
    Parse, Enrich(EnrichRule), Transform(TransformRule), Sample(SamplerKind),
    RateLimit(RateLimitRule), Redact(RedactPolicy), Route(RouteRule),
}

struct RedactPolicy { rules: Vec<RedactRule>, salt: TenantSalt }
                                          // persistence: config-loaded; salt is per-tenant secret (§20)

struct DlqEntry {                         // meridian-aggregator::exporter::dlq — persistence: on-disk
    sink_id: String, batch: Vec<LogRecord>, first_failure_unix_nano: i64, attempt_count: u32,
}                                         // replayable byte-for-byte (I12)

struct Tenant { id: String, quotas: TenantQuotas }
                                          // meridian-tenant — owned; quotas enforced at ingest+query (ADR-11)

struct Query { predicate: PredicateAst, group_by: Vec<String>, time_range: (i64, i64) }
                                          // meridian-query::ast — owned; parsed once, executed per-chunk
