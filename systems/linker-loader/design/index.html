Core schema & types (§11)

pub struct SymbolNameId(u32);      // interned index; Copy, Eq, Hash — owned by SymbolStringTable
pub struct InputFileId(u32);       // Copy, Eq, Hash — owned by the driver's input list
pub struct SectionIndex(u32);      // Copy, Eq, Hash — file-relative, never cross-file
pub struct FileOffset(u64);        // Copy, Eq — byte offset within one file
pub struct VirtualAddr(u64);       // Copy, Eq — process address space
pub struct FileRange { pub start: FileOffset, pub end: FileOffset }  // half-open; len/contains: signatures only, no arithmetic body shown here
pub struct ElfFile<'mmap> {
    pub id: InputFileId,
    pub path: PathBuf,
    pub mmap: &'mmap [u8],
    pub header: ElfHeader,
    pub sections: Vec<SectionHeader>,
    pub symbols: Vec<Symbol>,
    pub relocs: Vec<Relocation>,
    pub kind: ElfKind,               // Reloc, Exec, Dyn, Core
}
pub enum ElfKind { Reloc, Exec, Dyn, Core }
pub struct Section<'mmap> {              // an input section (formerly "InputSection")
    pub file: InputFileId,
    pub index: SectionIndex,
    pub name: SymbolNameId,
    pub kind: SectionKind,
    pub flags: SectionFlags,             // SHF_ALLOC | SHF_WRITE | SHF_EXECINSTR | ...
    pub addralign: u64,
    pub data: &'mmap [u8],
    pub relocs: Vec<Relocation>,
    pub group: Option<SectionGroupId>,
    pub keep: KeepDecision,              // Keep, Discard, Tentative — set by §17
}
pub struct OutputSection {
    pub name: SymbolNameId,
    pub kind: SectionKind,
    pub flags: SectionFlags,
    pub members: Vec<InputSectionRef>,   // populated §16, mutated §17, frozen §16 writer
    pub vaddr: VirtualAddr,
    pub file_off: FileOffset,
    pub size: u64,
}
pub struct OutputSegment {               // a PT_LOAD/PT_* program-header-backed group
    pub kind: SegmentKind,               // Load, Dynamic, Tls, GnuRelro, GnuEhFrame, GnuStack, Interp, Note
    pub flags: SegmentFlags,             // R/W/X
    pub sections: Vec<OutputSectionRef>,
    pub vaddr: VirtualAddr,
    pub file_off: FileOffset,
    pub filesz: u64,
    pub memsz: u64,
    pub align: u64,
}
pub struct Symbol {
    pub name: SymbolNameId,
    pub bind: SymbolBind,               // Local, Global, Weak, GnuUnique
    pub kind: SymbolKind,               // NoType, Object, Function, Section, File, Common, Tls, Ifunc
    pub vis: SymbolVisibility,          // Default, Internal, Hidden, Protected
    pub section: Option<SectionIndex>,  // None for absolute/undefined
    pub value: u64,
    pub size: u64,
    pub version: Option<SymbolVersion>,
}
pub struct GlobalSymbol {
    pub name: SymbolNameId,
    pub winning: WinningDef,
    pub references: Vec<RefSite>,
}
pub enum WinningDef {
    Strong { file: InputFileId, section: SectionIndex, value: u64 },
    Weak   { file: InputFileId, section: SectionIndex, value: u64 },
    Common { size: u64, alignment: u64 },
    Undefined,
    Plt    { plt_index: u32 },
    Ifunc  { resolver_file: InputFileId, resolver_section: SectionIndex, resolver_offset: u64 },
}
pub struct RefSite { pub file: InputFileId, pub section: SectionIndex, pub offset_in_section: u64, pub reloc_kind: RelocKind }
pub struct Relocation {
    pub offset_in_section: u64,
    pub symbol: SymbolNameId,
    pub addend: i64,
    pub kind: RelocKind,
}
#[non_exhaustive]
pub enum RelocKind {
    None, Abs64, Abs32, Pc32, Plt32, Got32, GotPcRel, GotPcRelX,
    GotTpOff, TpOff, DtpMod64, DtpOff64,
    Copy, GlobDat, JumpSlot, Relative, Irelative,
    // arch crates extend via to_generic/from_generic; see §11.6
}
pub trait ArchBackend: Send + Sync + 'static {
    const TARGET: TargetTriple;
    type Reloc: Copy + Eq + std::fmt::Debug;
    fn parse_reloc_table(&self, raw: &[u8]) -> Result<Vec<Relocation>, ElfError>;
    fn apply_reloc(&self, reloc: &Relocation, section_data: &mut [u8], section_vaddr: VirtualAddr,
        symbol_value: VirtualAddr, got_index_for: impl FnMut(SymbolNameId) -> Option<u32>,
        plt_index_for: impl FnMut(SymbolNameId) -> Option<u32>) -> Result<(), RelocError>;
    fn plt_stub_size(&self) -> usize;
    fn emit_plt_stub(&self, dst: &mut [u8], got_offset: i64) -> Result<(), RelocError>;
    fn tls_dtv_layout(&self) -> TlsDtvLayout;
    fn tls_le_offset(&self, sym_offset: i64) -> i64;
}
pub struct GotEntry { pub index: u32, pub symbol: SymbolNameId, pub kind: GotEntryKind }
pub enum GotEntryKind { Data, PltIndirect, TlsGd, TlsIe, TlsLd }
pub struct PltStub { pub index: u32, pub symbol: SymbolNameId, pub got_slot: u32, pub lazy: bool }
pub struct DynamicEntry { pub tag: DynTag, pub value: DynValue }
pub enum DynValue { Addr(VirtualAddr), Offset(FileOffset), Num(u64), Flag(bool) }
pub struct Diagnostic {
    pub severity: Severity,             // Note, Warning, Error, Fatal
    pub code: DiagCode,                 // e.g. lk0023
    pub message: String,
    pub primary: SourceLoc,
    pub notes: Vec<SourceLoc>,
}
pub struct SourceLoc { pub file: Option<InputFileId>, pub section: Option<SectionIndex>, pub byte_range: Option<FileRange>, pub label: String }
pub struct LoadedObject {
    pub path: PathBuf,
    pub base: VirtualAddr,
    pub program_headers: Vec<ProgramHeader>,
    pub dynamic: Vec<DynamicEntry>,
    pub got_entries: Vec<GotEntry>,
    pub plt_stubs: Vec<PltStub>,
    pub tls: Option<TlsBlock>,
    pub state: ObjectState,
    pub deps: Vec<LoadedObjectId>,
}
pub struct LoadedObjectId(u32);
#[derive(Copy, Clone, Eq, PartialEq, Debug)]
pub enum ObjectState { Mapped, RelocsApplied, InitArrayCalled, Running, Detached }   // I10, monotonic
pub struct TlsBlock { pub modid: u64, pub tdata: &'static [u8], pub tbss_size: u64, pub align: u64, pub initial_exec_offset: Option<i64> }
