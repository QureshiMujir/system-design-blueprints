Core schema & types (§11)

struct AnalysisSession {
    session_id: Uuid,           // globally unique; immutable after creation
    replay_epoch: u64,          // I11: monotonically increasing
    shard_count: u32,
    config_hash: Sha256Digest,  // ties session to a rule-set version (I7)
}
// Ownership: immutable after orchestration seal; Arc-shared across workers.
enum IrNode {
    Assign(AssignNode),
    Branch(BranchNode),
    Phi(PhiNode),
    Call(CallNode),
    Return(ReturnNode),
    HeapLoad(HeapLoadNode),
    HeapStore(HeapStoreNode),
    Literal(LiteralNode),
}
struct AssignNode { dst: NodeId, src: NodeId, span: SourceSpan }
struct PhiNode { dst: NodeId, operands: Vec<(BlockId, NodeId)> }
struct CallNode { callee: CallTarget, args: Vec<NodeId>, site: CallSiteId }
// Concurrency: append-only per function; sealed at CFG-construction completion (I3).
struct AccessPath {
    base: NodeId,
    fields: Vec<FieldToken>,    // bounded to depth k by I8 / ADR-19
}
enum TaintKind { UserInput, FileContent, NetworkResponse, EnvironmentVar }
enum TaintState {
    Untainted,
    Tainted(TaintKind),
    Sanitized(TaintKind, SanitizerId),
    Unknown,                     // fail-closed per P11
}
struct TaintFact {
    fact_id: u64,
    state: TaintState,
    path: AccessPath,
    source_node: NodeId,
    propagation_depth: u32,
}
// Concurrency: append-only propagation; lock-free MPSC queue insertion (Mara Bos §5).
enum EdgeKind { Assignment, Call, Return, HeapStore, HeapLoad }
struct PathEdge { proc_entry: NodeId, d1: TaintFact, node: NodeId, d2: TaintFact }
struct SummaryEdge { proc_entry: NodeId, d1: TaintFact, proc_exit: NodeId, d2: TaintFact }
// Persistence: protobuf-encoded (ADR-03); WAL version-tagged; forward-compatible field
// reservation per I7.
trait PropagationRule {
    fn flow_function(&self, fact: &TaintFact, edge: &EdgeKind) -> PropagationResult;
}
trait Sanitizer {
    fn applies(&self, state: &TaintState) -> bool;
    fn transform(&self, state: &TaintState) -> TaintState;
}
// Lifecycle: rules immutable after registration; hot reload requires epoch transition (I7).
struct PointsToSet { targets: Vec<HeapRegionId> }     // ADR-17: engine-tagged
enum PointsToEngine { Andersen, Steensgaard }
struct CallGraphEdge { caller: FunctionId, callee: FunctionId, site: CallSiteId, confidence: DispatchConfidence }
enum DispatchConfidence { Exact, Devirtualized, Unknown }  // Unknown per P11
// Persistence: immutable after seal (I10); serialized alongside snapshot manifests.
struct Finding {
    finding_id: StructuralHash,   // P12: keyed on structural path, not line number
    source: TaintKind,
    sink: SinkId,
    path_trace: Vec<AccessPath>,
    severity: Severity,
}
struct SarifRun { tool_name: &'static str, results: Vec<Finding>, schema_version: &'static str }
// Serialization: SARIF 2.1.0 (ADR-22); wire-stable per §25.
