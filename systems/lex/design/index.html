Core schema & types (§11)

// crate: lex-core::identity  (owned by Issuer; immutable once issued; Postgres-persisted)
struct AgentPassport {
    subject_did: Did,                    // immutable; primary key
    controller_did: Did,                 // mutable only via re-issuance
    capabilities: Vec<Capability>,       // append-only via delegation, I6-bounded
    jurisdiction: IsoCountryOrSeat,
    acceptable_constitutions: Vec<ConstitutionRef>,
    key_material: PublicKeyMaterial,     // never private; I12
    delegation_chain: Vec<DelegationLink>, // len <= 5, I6
    issuance_date: Rfc3339Utc,
    expiration_date: Rfc3339Utc,
    revocation_endpoint: Url,
}
// serialization: W3C VC JSON-LD; compatibility: additive fields only post-v1.0 (P8)

enum RevocationStatus { Active, Revoked { at: Rfc3339Utc, reason: RevocationReason } }
// concurrency: single-writer (Issuer), many-reader (Aggregator cache, I11 SLO)
// crate: lex-core::actionlog  (owned by agent SDK; append-only; hash-chained, I1-I3)
struct ActionEnvelope {
    agent_did: Did,
    sequence: u64,                        // strictly increasing, I2
    prior_hash: Blake3Hash,               // I1
    timestamp: Rfc3339Utc,
    constitution_ref: Option<ConstitutionRef>,
    microcontract_ref: Option<MicroContractRef>,
    action: ActionPayload,
    outcome: ActionOutcome,
    self_hash: Blake3Hash,                // computed over all preceding fields
    signature: Signature,                 // Ed25519 default, ECDSA-P256 FIPS variant
}
// persistence: Aggregator append-only store + object-store archive (ADR-04)
// concurrency: single-writer per agent_did (local SDK), many-reader (Aggregator, judges)
// crate: lex-core::constitution  (content-addressed, I5; registry-persisted, immutable)
struct Constitution {
    id: ContentAddress,                   // = Blake3(Jcs(canonical_form)), I5
    parties: Vec<PartyRef>,
    jurisdiction: IsoCountryOrSeat,
    effective_from: Rfc3339Utc,
    expires_at: Rfc3339Utc,
    clauses: Vec<Clause>,
    adjudication_policy: AdjudicationPolicy, // quorum in [3,21], I7
    severability: SeverabilityMode,
    dispute_window: IsoDuration,           // I9
    signatures: Vec<Signature>,
}
enum Clause { Prohibition(LclPredicate), Obligation(LclPredicate),
              Permission(LclPredicate), Escalation(LclPredicate) }
// serialization: JSON-LD + JCS; compatibility: inherits by reference, cycles forbidden (I5-adjacent)
// crate: lex-core::microcontract  (content-addressed; broker- or peer-negotiated)
struct MicroContract {
    id: ContentAddress,
    constitution_ref: ConstitutionRef,
    parties: [Did; 2],
    subject: SubjectTerms,
    performance_clauses_referenced: Vec<ClauseId>,
    signed_at: Rfc3339Utc,
    signatures: [Signature; 2],
}
// concurrency: negotiation is a two-party handshake (offer/counter/accept), Phase 5 (§17)
// crate: lex-core::dispute  (Mesh Coordinator-owned; state machine, §8 DisputeCase State Transitions)
struct DisputeCase {
    id: ContentAddress,
    filed_by: Did,
    filed_at: Rfc3339Utc,                 // must satisfy I9 vs. constitution.dispute_window
    microcontract_ref: MicroContractRef,
    constitution_ref: ConstitutionRef,
    claim: DisputeClaim,
    respondent: Did,
    requested_remedy: Vec<RemedyRequest>,
    signature: Signature,
}
enum DisputeState { Filed, QuorumSelecting, QuorumFilled, OpinionsCollecting,
                     OutcomeTallying, Inconclusive, Breached, Upheld, PartitionStalled, Closed }
// persistence: Mesh Coordinator durable store; state transitions logged for forensics (FM-4,FM-5)

struct AdjudicationOutcome {
    id: ContentAddress,
    dispute_ref: DisputeCaseRef,
    quorum_size: u8,                      // I7
    valid_opinions: u8,                   // <= quorum_size, I10-filtered
    outcome_per_clause: Vec<ClauseVerdict>,
    remedy_awarded: Vec<RemedyRequest>,
    reasoning_digest_uri: Uri,            // P7 transparency
    zk_proofs: Vec<ZkProof>,              // one per valid opinion, I10
    anchor: Option<AnchorRef>,            // P4: only for disputed outcomes
    coordinator_signature: Signature,
}
// compatibility: outcome schema additive-only post lex.outcome/v1 (P8)
