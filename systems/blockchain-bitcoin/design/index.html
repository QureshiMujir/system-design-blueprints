Core schema & types (§11)

struct Hash256([u8; 32]);                 // SHA256d output; display byte-order reversed
struct BlockHeader {
    version: i32,
    prev_hash: Hash256,
    merkle_root: Hash256,                  // I5
    timestamp: u32,
    bits: u32,                              // compact PoW target (§8)
    nonce: u32,
}
struct Block { header: BlockHeader, transactions: Vec<Transaction> }
// Ownership: immutable after storage-layer commit; Arc-shared across validation workers.
struct TxInput { prev_txid: Hash256, prev_index: u32, script_sig: Script, sequence: u32 }
struct TxOutput { value: i64, script_pubkey: Script }        // satoshis, ADR-03
struct Transaction {
    version: i32,
    inputs: Vec<TxInput>,
    outputs: Vec<TxOutput>,
    locktime: u32,
}
// Serialization: consensus encoding (ADR-03); txid = SHA256d(serialize(self)) (I6).
struct Script(Vec<u8>);                    // raw opcode/data bytes, bounded by I9
enum OpCode { OpDup, OpHash160, OpEqualVerify, OpCheckSig, OpCheckMultiSig, OpEqual, OpReturn }
// Concurrency: immutable byte buffer; parsed lazily by chain-script (Phase 2, §14).
struct UtxoEntry {
    txid: Hash256,
    index: u32,
    value: i64,
    script_pubkey: Script,
    height: u32,
    is_coinbase: bool,                     // gates I4 maturity check
}
// Persistence: RocksDB key (txid, index) -> UtxoEntry (ADR-14); append/delete only, no mutation.
struct MempoolEntry {
    txid: Hash256,
    tx: Transaction,
    fee_rate_sat_vb: u64,                  // ADR-12 ordering key
    received_at: u64,
    ancestor_count: u32,
}
// Concurrency: lock-free priority-ordered insertion (Mara Bos §5); eventually consistent (P9/I11).
struct PeerState {
    peer_id: PeerId,
    height: u64,
    latency_ms: u64,
    services: u64,
    misbehavior_score: u32,                // ADR-23 ban-score model
}
enum HandshakeState { AwaitingVersion, AwaitingVerack, Ready }   // I14
// Lifecycle: PeerState sealed Ready only after full handshake completion.
struct ChainTip { hash: Hash256, height: u32, cumulative_work: u128 }   // I3 ordering key
trait UtxoSet {
    fn get(&self, txid: &Hash256, index: u32) -> Option<UtxoEntry>;
    fn apply_block(&mut self, block: &Block) -> Result<(), ConsensusError>;
    fn undo_block(&mut self, block: &Block) -> Result<(), ConsensusError>;
}
enum ConsensusError { InvalidPow, MerkleMismatch, DoubleSpend, ScriptFailure, SupplyOverrun }
// Persistence: apply_block/undo_block are WAL-logged (I12) before UTXO-DB commit.
