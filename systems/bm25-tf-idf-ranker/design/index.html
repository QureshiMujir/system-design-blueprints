Core schema & types (§11)

// Newtype wrappers prevent silent DocId/TermId mixing in scoring algebra
pub struct DocId(pub u32);          // local within segment
pub struct GlobalDocId(pub u64);    // (segment_index:u32)<<32 | (local:u32)
pub struct TermId(pub u32);         // monotonically assigned; stable within segment
pub struct TermFreq(pub u32);       // raw occurrence count; clamped at 65535
pub struct ByteOffset(pub u32);     // byte position in source text for highlighting
pub struct SegmentId(pub uuid::Uuid); // UUID v4; stable across restarts
pub struct Token {
    pub text: String,          // normalized, stemmed form
    pub start: ByteOffset,     // into original source (not normalized text)
    pub end: ByteOffset,       // one past end; start < end invariant (I-none)
    pub position: u32,         // monotonic; not renumbered after stop-word removal
}

pub trait Tokenizer: Send + Sync {
    fn tokenize<'a>(&'a self, text: &'a str) -> Box<dyn Iterator<Item=Token> + 'a>;
}

pub trait TokenFilter: Send + Sync {
    fn filter<'a>(&'a self, stream: Box<dyn Iterator<Item=Token> + 'a>)
        -> Box<dyn Iterator<Item=Token> + 'a>;
}

pub struct AnalysisPipeline {
    tokenizer: Arc<dyn Tokenizer>,
    filters: Vec<Arc<dyn TokenFilter>>,
}
impl AnalysisPipeline {
    pub fn analyze<'a>(&'a self, text: &'a str) -> impl Iterator<Item=Token> + 'a;
}
pub struct Posting { pub doc_id: DocId, pub term_freq: TermFreq }

pub struct PositionalPosting {
    pub doc_id: DocId, pub term_freq: TermFreq,
    pub positions: Vec<u32>,  // sorted ascending; delta-encoded on disk
}

pub trait PostingsIterator: Send {
    fn next(&mut self) -> Option<Posting>;
    fn seek(&mut self, target: DocId) -> Option<Posting>;  // seek(t).doc_id >= t
    fn doc_freq(&self) -> u32;
    fn max_score(&self) -> f32;  // upper bound for WAND (I7)
}
pub struct CorpusStats {
    pub num_docs: u64,         // non-deleted document count
    pub total_tokens: u64,     // Σ doc field lengths
    pub avgdl: f64,            // total_tokens / num_docs (I3)
    pub doc_freq: HashMap<TermId, u32>,
}
impl CorpusStats {
    pub fn merge(stats: &[CorpusStats]) -> CorpusStats;  // sum totals, recompute avgdl
}

pub struct Segment {
    pub id: SegmentId,
    pub path: PathBuf,
    pub term_dict: Arc<dyn TermDictionary>,
    pub postings: Arc<dyn PostingsReader>,
    pub doc_store: Arc<dyn DocStore>,
    pub deleted: RoaringBitmap,    // delete overlay (I6)
    pub stats: CorpusStats,        // snapshot-consistent with segment (P7)
}
pub trait Scorer: Send + Sync {
    fn score(&self, tf: TermFreq, doc_len: u32, df: u32, num_docs: u64, avgdl: f64) -> f32;
    fn max_term_score(&self, df: u32, num_docs: u64) -> f32; // must be >= all score() values (I7)
}

pub struct Bm25Params { pub k1: f32, pub b: f32, pub delta: f32 }
// defaults: k1=1.2, b=0.75, delta=0.0
// k1 range: [0.0, ∞); b range: [0.0, 1.0]; delta range: [0.0, ∞)
impl Bm25Params { pub fn validate(&self) -> Result<(), RankerError>; }

pub struct FieldParams { pub name: FieldName, pub weight: f32, pub b: f32 }
pub struct Bm25fScorer { pub k1: f32, pub field_params: Vec<FieldParams> }
impl Scorer for Bm25fScorer { ... }
pub enum QueryNode {
    Term(TermQuery),
    Phrase(PhraseQuery),
    And(Vec<QueryNode>),
    Or(Vec<QueryNode>),
    Not { must_not: Box<QueryNode>, filter: Box<QueryNode> },
}
pub struct TermQuery { pub term: String, pub field: Option<FieldName>, pub boost: f32 }
pub struct PhraseQuery { pub terms: Vec<String>, pub field: Option<FieldName>,
                         pub boost: f32, pub slop: u32 }
pub type FieldName = String;

pub enum QueryPlan {
    TermScan { term_id: TermId, df: u32, max_score: f32 },
    AndMerge(Vec<QueryPlan>),     // children sorted by ascending df (P13)
    OrMerge(Vec<QueryPlan>),
    NotFilter { required: Box<QueryPlan>, excluded: Box<QueryPlan> },
    PhraseScan { term_ids: Vec<TermId>, slop: u32 },
    Empty,
}
pub enum IndexCommand {
    InsertDocument { doc_id: DocId, fields: Vec<(FieldName, String)> },
    DeleteDocument { doc_id: DocId },
    Commit { segment_id: SegmentId },
}

#[derive(thiserror::Error)]
pub enum RankerError {
    Io(#[from] std::io::Error),
    SegmentFormat(String),
    QueryParse { pos: usize, msg: String },
    Codec(String),
    InvalidParam(String),
    NotFound(String),
    ConcurrentModification(String),
}
