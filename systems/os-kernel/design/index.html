Core schema & types (§11)

#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
#[repr(transparent)]
pub struct PhysAddr(pub u64);

#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
#[repr(transparent)]
pub struct VirtAddr(pub u64);

impl VirtAddr {
    pub const fn new(addr: u64) -> Self;      // panics if non-canonical (I2)
    pub const fn p4_index(self) -> u16;
    pub const fn p3_index(self) -> u16;
    pub const fn p2_index(self) -> u16;
    pub const fn p1_index(self) -> u16;
}

#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
pub struct Frame { pub start: PhysAddr }        // owned exclusively by mm::frame

impl Frame {
    pub const SIZE: u64 = 4096;
    pub fn containing(addr: PhysAddr) -> Self;
}

#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
pub struct Page { pub start: VirtAddr }

impl Page {
    pub const SIZE: u64 = 4096;
    pub fn containing(addr: VirtAddr) -> Self;
}
bitflags::bitflags! {
    pub struct PteFlags: u64 {
        const PRESENT = 1 << 0;  const WRITABLE = 1 << 1;  const USER = 1 << 2;
        const WRITETHROUGH = 1 << 3;  const NOCACHE = 1 << 4;  const ACCESSED = 1 << 5;
        const DIRTY = 1 << 6;    const HUGE = 1 << 7;       const GLOBAL = 1 << 8;
        const NX = 1 << 63;
    }
}

#[repr(transparent)]
pub struct PageTableEntry(u64);

impl PageTableEntry {
    pub fn frame(&self) -> Option<Frame>;                      // None if !PRESENT (I2)
    pub fn set(&mut self, frame: Frame, flags: PteFlags);
}
pub trait Mapper {
    fn map(&mut self, page: Page, frame: Frame, flags: PteFlags) -> Result<(), MapErr>;
    fn unmap(&mut self, page: Page) -> Result<Frame, MapErr>;   // flushes TLB (I12)
    fn translate(&self, addr: VirtAddr) -> Option<PhysAddr>;
    fn flush(&mut self, page: Page);
}

pub trait FrameAllocator {
    fn allocate(&mut self) -> Option<Frame>;
    fn allocate_contiguous(&mut self, count: usize) -> Option<Frame>;
    fn free(&mut self, frame: Frame);                           // must not double-free (I9)
}
pub enum ThreadState { New, Runnable, Running, Blocked(WaitReason), Zombie }

pub struct Thread {
    pub id: ThreadId,
    pub state: ThreadState,
    pub kernel_stack: KernelStack,
    pub user_stack: Option<UserStack>,
    pub context: Context,
    pub address_space: Arc<AddressSpace>,
    pub priority: i8,
    pub time_slice_ns: u64,
    pub runtime_ns: u64,
}

#[repr(C)]
pub struct Context {
    pub rsp: u64, pub rbp: u64, pub rbx: u64,
    pub r12: u64, pub r13: u64, pub r14: u64, pub r15: u64,
    pub rip: u64, pub cr3: u64,
}
pub struct Process {
    pub pid: Pid,
    pub address_space: Arc<AddressSpace>,
    pub threads: Vec<Arc<Thread>>,
    pub fdt: FileDescriptorTable,       // refcounted entries (I6)
    pub cwd: Arc<DirEntry>,
    pub parent: Option<Pid>,
    pub exit_code: Option<i32>,
    pub signals: SignalQueue,
}

pub struct AddressSpace {
    pub p4_frame: Frame,
    pub regions: VmarTree,              // §14.4
}
pub trait FileSystem: Send + Sync {
    fn root(&self) -> Arc<dyn Inode>;
    fn name(&self) -> &str;
}

pub trait Inode: Send + Sync {
    fn metadata(&self) -> Metadata;
    fn lookup(&self, name: &str) -> Result<Arc<dyn Inode>, FsError>;
    fn read_at(&self, offset: u64, buf: &mut [u8]) -> Result<usize, FsError>;
    fn write_at(&self, offset: u64, buf: &[u8]) -> Result<usize, FsError>;
    fn readdir(&self) -> Result<Vec<DirEntryRef>, FsError>;
}

pub struct File { pub inode: Arc<dyn Inode>, pub offset: AtomicU64, pub flags: OpenFlags }

    Read = 0, Write = 1, Open = 2, Close = 3, Mmap = 4, Munmap = 5, Brk = 6,
    Fork = 7, Execve = 8, Exit = 9, Wait = 10, Pipe = 11, Dup = 12, Dup2 = 13,
    Getpid = 14, Kill = 15, Sigaction = 16, Sigreturn = 17, Nanosleep = 18,
    Yield = 19, Stat = 20, Lseek = 21, Chdir = 22, Getcwd = 23, Mount = 24,
    Umount = 25, Reboot = 26, Uname = 27, Socket = 28, Bind = 29, Connect = 30,
}

#[repr(C)]
pub struct SyscallResult { pub value: i64 }   // negative => -errno
pub struct Spinlock<T> { /* IRQ-safe variant saves/restores rflags.IF */ }
pub struct Mutex<T>    { /* sleeping; parks the thread on contention, Phase 5+ */ }
pub trait Driver: Send + Sync {
    fn name(&self) -> &'static str;
    fn probe(&mut self, ctx: &mut DriverContext) -> Result<(), DriverError>;
    fn shutdown(&mut self);
}

pub trait IrqHandler { fn handle(&self, vector: u8) -> IrqResult; }

pub trait BlockDevice {
    fn read_block(&self, lba: u64, buf: &mut [u8]) -> Result<(), BlockError>;
    fn write_block(&self, lba: u64, buf: &[u8]) -> Result<(), BlockError>;
    fn block_size(&self) -> usize;
    fn capacity(&self) -> u64;
}

pub trait CharDevice {
    fn read(&self, buf: &mut [u8]) -> Result<usize, IoError>;
    fn write(&self, buf: &[u8]) -> Result<usize, IoError>;
    fn poll(&self) -> PollMask;
}
