Core schema & types (§11)

pub struct Token<'a> { pub kind: TokenKind<'a>, pub span: std::ops::Range<u32>,
                       pub line: u32, pub col: u32 }
pub enum TokenKind<'a> {
    Word(WordParts<'a>), AssignmentWord { name: &'a str, value: WordParts<'a> },
    Reserved(ReservedWord), IoNumber(i32), Operator(Operator), Newline, Eof,
}
pub enum ReservedWord { If, Then, Else, Elif, Fi, Do, Done, Case, Esac, In,
                         For, While, Until, LBrace, RBrace, Bang, Function }
pub enum Operator { Pipe, PipeAmp, AndIf, OrIf, Semi, DSemi, SemiAmp, LParen,
                     RParen, Amp, Less, LessLess, LessLessDash, LessAmp, LessGt,
                     Great, GreatGreat, GreatAmp, GreatGt, AmpGt, AmpGtGt,
                     LessLessLess, DLParen, DRParen, DLBracket, DRBracket }
pub struct WordParts<'a> { pub parts: bumpalo::collections::Vec<'a, WordPart<'a>> }
pub enum WordPart<'a> {
    Literal(&'a str), SingleQuoted(&'a str), DoubleQuoted(bumpalo::collections::Vec<'a, WordPart<'a>>),
    DollarVar(&'a str), DollarBrace(ParamExpansion<'a>), CommandSub(CommandSubKind<'a>),
    ArithSub(&'a str), Tilde(TildePrefix<'a>), Backslash(u8),
}
pub struct ParamExpansion<'a> { pub name: ParamName<'a>, pub op: Option<ParamOp<'a>>,
                                 pub indirection: bool, pub length: bool }
pub enum ParamName<'a> { Var(&'a str), Positional(u32), Special(SpecialParam),
                          ArrayElement { name: &'a str, index: ArrayIndex<'a> } }
pub enum SpecialParam { Argc, ArgsAtSign, ArgsStar, LastStatus, Pid, LastBgPid,
                         OptionFlags, ScriptName }
pub enum ParamOp<'a> {
    UseDefault { word: WordParts<'a>, colon: bool },
    AssignDefault { word: WordParts<'a>, colon: bool },
    ErrorIfUnset { word: WordParts<'a>, colon: bool },
    Alternate { word: WordParts<'a>, colon: bool },
    StripPrefix { pattern: WordParts<'a>, longest: bool },
    StripSuffix { pattern: WordParts<'a>, longest: bool },
    Substitute { pattern: WordParts<'a>, replacement: WordParts<'a>, all: bool },
    Substring { offset: WordParts<'a>, length: Option<WordParts<'a>> },
    Transform { op: TransformOp },
}
pub struct Program<'a> { pub list: CompoundList<'a> }
pub struct CompoundList<'a> { pub items: bumpalo::collections::Vec<'a, AndOrList<'a>> }
pub struct AndOrList<'a> { pub head: Pipeline<'a>,
    pub tail: bumpalo::collections::Vec<'a, (LogicalOp, Pipeline<'a>)>,
    pub terminator: Terminator }
pub enum LogicalOp { AndIf, OrIf }
pub enum Terminator { Sync, Async, None }
pub struct Pipeline<'a> { pub bang: bool, pub commands: bumpalo::collections::Vec<'a, Command<'a>> }
pub enum Command<'a> {
    Simple(SimpleCommand<'a>), BraceGroup(CompoundList<'a>), Subshell(CompoundList<'a>),
    If(IfClause<'a>), For(ForClause<'a>), Case(CaseClause<'a>), While(WhileClause<'a>),
    Until(UntilClause<'a>), FunctionDef(FunctionDef<'a>), Arithmetic(ArithCommand<'a>),
    Conditional(CondCommand<'a>),
}
pub struct SimpleCommand<'a> { pub assigns: bumpalo::collections::Vec<'a, Assignment<'a>>,
    pub words: bumpalo::collections::Vec<'a, WordParts<'a>>,
    pub redirs: bumpalo::collections::Vec<'a, Redirect<'a>> }
pub struct Redirect<'a> { pub fd: Option<i32>, pub op: RedirOp, pub word: WordParts<'a>,
    pub here_doc_body: Option<HereDocBody<'a>> }
pub enum RedirOp { Read, Write, Append, ReadDup, WriteDup, ReadWrite, Clobber,
                    Heredoc, HeredocStrip, Herestring }
pub struct Variable { pub name: std::sync::Arc<str>, pub value: VarValue, pub flags: VarFlags }
bitflags::bitflags! { pub struct VarFlags: u32 {
    const EXPORTED = 1<<0; const READONLY = 1<<1; const LOCAL = 1<<2;
    const ARRAY = 1<<3; const MAP = 1<<4; const INTEGER = 1<<5; const NAMEREF = 1<<6; } }
pub enum VarValue { Scalar(std::sync::Arc<str>), Indexed(Vec<std::sync::Arc<str>>),
                     Map(std::collections::HashMap<std::sync::Arc<str>, std::sync::Arc<str>>) }
pub struct Scope { frames: Vec<Frame> }
pub struct Frame { bindings: std::collections::HashMap<std::sync::Arc<str>, Variable> }
impl Scope {
    pub fn get(&self, name: &str) -> Option<&Variable>;
    pub fn set(&mut self, var: Variable);
    pub fn unset(&mut self, name: &str) -> Result<(), UnsetError>;
    pub fn export(&mut self, name: &str) -> Result<(), ExportError>;
    pub fn enter_function(&mut self);
    pub fn leave_function(&mut self);
    pub fn local(&mut self, var: Variable) -> Result<(), LocalError>;
}
pub struct JobId(pub u32);
pub struct Pgid(pub libc::pid_t);
pub struct Job { pub id: JobId, pub pgid: Pgid, pub processes: Vec<Process>,
    pub command_text: String, pub state: JobState, pub last_status: Option<i32>,
    pub started_at: std::time::SystemTime, pub current: bool, pub previous: bool,
    pub foreground: bool }
pub struct Process { pub pid: libc::pid_t, pub state: ProcessState, pub status: Option<i32> }
pub enum JobState { Running, Stopped, Done, Killed }
pub enum ProcessState { Running, Stopped, Exited(i32), Signaled(i32) }
pub struct Shell { pub mode: ShellMode, pub flags: ShellFlags, pub env: crate::env::Env,
    pub jobs: crate::jobs::Table, pub aliases: crate::lexer::AliasTable,
    pub functions: crate::env::FunctionTable, pub history: crate::history::Store,
    pub editor: Option<crate::editor::Editor>, pub last_status: i32,
    pub exit_code: Option<i32> }
pub enum ShellMode { Posix, Bash, Enhanced }
bitflags::bitflags! { pub struct ShellFlags: u32 {
    const ERREXIT=1<<0; const NOUNSET=1<<1; const PIPEFAIL=1<<2; const XTRACE=1<<3;
    const VERBOSE=1<<4; const NOCLOBBER=1<<5; const NOEXEC=1<<6; const NOTIFY=1<<7;
    const ALLEXPORT=1<<8; const HASHALL=1<<9; const MONITOR=1<<10; const RESTRICTED=1<<11;
    const INTERACTIVE=1<<12; const LOGIN=1<<13; const POSIXLY=1<<14; } }
pub trait Builtin: Send + Sync + 'static {
    fn name(&self) -> &'static str;
    fn is_special(&self) -> bool;
    fn run(&self, shell: &mut Shell, args: &[std::ffi::OsStr], io: &mut Io) -> i32;
    fn help(&self) -> &'static str;
    fn completion(&self) -> &'static [CompletionRule];   // default: empty slice
}
