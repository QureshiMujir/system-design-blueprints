Core schema & types (§11)

// hasher.rs — uniform 64-bit non-crypto hash; deterministic, avalanche (chi-square p>0.01)
pub trait Hasher64: Clone + Send + Sync { fn hash_one(&self, bytes: &[u8]) -> u64; }

// XxHash64: default; seed (default 0); >=10 GB/s on 64-B keys; token gen + hot-path locate
#[derive(Clone)] pub struct XxHash64 { seed: u64 }
impl XxHash64 { pub const fn new(seed: u64) -> Self; pub const fn default_seed() -> Self; }
impl Hasher64 for XxHash64 { /* xxh64 */ }

// SipHasher64: keyed SipHash-1-3; adversarial keys; ~1 GB/s; k0/k1 secret + stable for ring life
#[derive(Clone)] pub struct SipHasher64 { k0: u64, k1: u64 }
impl SipHasher64 { pub fn new(k0: u64, k1: u64) -> Self; }
impl Hasher64 for SipHasher64 { /* sip-1-3 */ }
// ids.rs
// NodeId: opaque, from hashed UTF-8 name; eq/ord by u64; one name -> one NodeId for ring life
#[derive(Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug)] pub struct NodeId(pub u64);
impl NodeId { pub fn from_name<H: Hasher64>(name: &str, hasher: &H) -> Self; }

// Key: zero-copy byte slice; any length; hashed to u64 for positioning
#[derive(Copy, Clone, Debug)] pub struct Key<'a>(pub &'a [u8]);

// Weight: relative capacity; scales vnodes (ring) + score (WeightedRendezvous); default 1; 0 -> BadWeight
#[derive(Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Debug)] pub struct Weight(pub u32);
impl Weight { pub const ONE: Self = Self(1); }

// ZoneId: fault-domain id for zone-aware preference lists
#[derive(Copy, Clone, Eq, PartialEq, Hash, Debug)] pub struct ZoneId(pub u64);
// placement.rs — uniform API for all strategies; &mut self mutates, &self reads (arc-swap-safe).
// Invariants: locate=None iff no nodes (INV-1); locate_n=min(n,distinct) distinct, primary first
// (INV-3); add+remove of same node restores prior locate; single add/remove moves <=ceil(K/N) (INV-2).
pub trait Placement {
    fn add_node(&mut self, node: NodeId, weight: Weight);
    fn remove_node(&mut self, node: NodeId);
    fn locate(&self, key: &[u8]) -> Option<NodeId>;
    fn locate_n(&self, key: &[u8], n: usize) -> Vec<NodeId>;
}
// error.rs — all fallible placement ops return PlacementError
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PlacementError { NoNodes, BadWeight, CapacityExceeded }
impl core::fmt::Display for PlacementError { /* one-line per variant */ }
#[cfg(feature = "std")] impl std::error::Error for PlacementError {}
// vnode.rs — one virtual node; token = hash(name ++ replica_idx); BTreeMap key is the token
#[derive(Copy, Clone, Debug, Eq, PartialEq)]
pub struct VNode { pub token: u64, pub node: NodeId, pub replica_idx: u32 }
// ring.rs — consistent hashing ring; BTreeMap<u64,NodeId> O(log VN) successor via
// range(token..).next().or_else(first_key_value). add inserts V*weight tokens, remove deletes
// all of a node's tokens (both O(V log VN)). Not internally synced (wrap RwLock/arc-swap).
// Pure fn of membership+V+seed → serialize nodes/weights/zones, re-derive ring on load.
pub struct HashRing<H: Hasher64> {
    ring: BTreeMap<u64, NodeId>,          // token -> physical node
    vnodes_per_unit_weight: u32,          // V; typical 100–200
    hasher: H,
    nodes: HashMap<NodeId, Weight>,       // source of truth
    zones: HashMap<NodeId, ZoneId>,       // optional per-node zone
}
impl<H: Hasher64> HashRing<H> {
    pub fn new(hasher: H, vnodes_per_unit_weight: u32) -> Self;
    pub fn add_node_with_zone(&mut self, node: NodeId, weight: Weight, zone: Option<ZoneId>); // no-op if present
    pub fn remove_node(&mut self, node: NodeId);                 // no-op if absent
    pub fn locate(&self, key: &[u8]) -> Option<NodeId>;          // clockwise successor of hash(key)
    pub fn locate_n(&self, key: &[u8], n: usize) -> Vec<NodeId>; // N distinct, zone-skip if zones set
    pub fn tokens_for(&self, node: NodeId) -> Vec<u64>;          // sorted; for migration planner
}
impl<H: Hasher64> Placement for HashRing<H> { /* delegates */ }
// preference.rs — ordered distinct physical owners; primary = replicas[0]
#[derive(Debug, Clone)]
pub struct PreferenceList { pub primary: NodeId, pub replicas: Vec<NodeId> } // len = min(n, distinct)
// walk state: skip token if NodeId in seen_nodes OR zone in seen_zones
struct PreferenceWalk<'a> { seen_nodes: HashSet<NodeId>, seen_zones: HashSet<ZoneId>, ring: &'a HashRing<impl Hasher64> }
// bounded.rs — Consistent Hashing with Bounded Loads (Mirrokni et al.). max load <=
// ceil(c * total/num_nodes); forward clockwise to next non-full node on overflow. Load counters
// incremented on locate, resettable, not persisted; worst-case forward chain O(N), typical O(1).
pub struct BoundedRing<H: Hasher64> { ring: HashRing<H>, capacity_factor: f64, load: HashMap<NodeId, u64>, total_load: u64 }
impl<H: Hasher64> BoundedRing<H> {
    pub fn new(ring: HashRing<H>, capacity_factor: f64) -> Result<Self, PlacementError>; // c > 1.0; default 1.25
    pub fn capacity_ceiling(&self) -> u64;
    pub fn reset_counters(&mut self);                           // e.g. epoch boundary; no membership change
}
impl<H: Hasher64> Placement for BoundedRing<H> { /* locate forwards on overflow */ }
// hrw.rs — Rendezvous (HRW); O(n) locate; perfectly minimal disruption. owner = argmax
// score(node,key)=hash(node_bytes ++ key_bytes); ties by NodeId ascending. locate_n = top-N, same pass.
pub struct Rendezvous<H: Hasher64> { nodes: Vec<(NodeId, Weight)>, hasher: H } // weights unused here (uniform)
impl<H: Hasher64> Rendezvous<H> { pub fn new(hasher: H) -> Self; pub fn score(&self, node: NodeId, key: &[u8]) -> u64; }
impl<H: Hasher64> Placement for Rendezvous<H> { /* argmax; top-N */ }
// weighted.rs — heterogeneous capacity: score = -weight / ln(h01), h01 = hash/u64::MAX in (0,1].
// Higher Weight wins proportionally; expected fraction at node i = weight_i / Σ weights.
pub struct WeightedRendezvous<H: Hasher64> { nodes: Vec<(NodeId, f64)>, hasher: H } // (id, weight_f64)
impl<H: Hasher64> Placement for WeightedRendezvous<H> { /* -w/ln(h01) argmax */ }
// skeleton.rs — hierarchical O(log_f n) HRW for large n; fanout-f virtual tree, HRW per level,
// leaf returns winner. Slight disruption increase vs flat HRW; still beats ring on removal.
pub struct SkeletonHrw<H: Hasher64> { fanout: usize, nodes: Vec<NodeId>, hasher: H } // f recommended 16–64
impl<H: Hasher64> Placement for SkeletonHrw<H> { /* tree descent; O(log_f n) */ }
// jump.rs — Lamping-Veach jump consistent hash; O(ln n) time, O(1) memory. Ordinal buckets
// 0..num_buckets-1; add/remove only at the tail (no arbitrary removal/zone/non-ordinal ids).
pub fn jump_hash(key: u64, num_buckets: u32) -> u32; // PRNG recurrence; return last valid b
// maglev.rs — lookup-table hashing (Eisenbud et al., NSDI 2016). lookup[hash%M] = backend idx;
// build: offset_i=hash(name,0)%M, skip_i=hash(name,1)%(M-1)+1, perm(j)=(offset+j*skip)%M, round-robin fill.
// Near-perfect balance (~1/n), O(1) lookup, ~1/M table change per backend change. M prime, M >> n (>=100n).
pub struct Maglev<H: Hasher64> { lookup: Vec<i32>, table_size: usize, nodes: Vec<NodeId>, hasher: H } // -1 = empty pre-fill
impl<H: Hasher64> Maglev<H> {
    pub fn new(hasher: H, table_size: usize) -> Result<Self, PlacementError>;
    pub fn rebuild(&mut self);                                  // after membership change; O(M)
}
impl<H: Hasher64> Placement for Maglev<H> { /* lookup[hash % M] */ }
// modulo.rs — naive baseline node = hash(key) % n; ~(n-1)/n remapped on any change. DisruptionReport only.
pub struct ModuloHash<H: Hasher64> { n: u32, nodes: Vec<NodeId>, hasher: H }
impl<H: Hasher64> Placement for ModuloHash<H> { /* nodes[hash % n] */ }
// distribution.rs — load over nodes after placing keys_total keys. peak_to_mean=max/mean;
// gini 0..1; std_dev of per-node loads. Ring V=150 target: peak_to_mean<1.10, std_dev/mean<0.05.
#[derive(Debug, Clone)]
pub struct DistributionReport { pub per_node: HashMap<NodeId, u64>, pub peak_to_mean: f64, pub gini: f64, pub std_dev: f64, pub max_load: u64, pub min_load: u64 }
impl DistributionReport { pub fn compute(placer: &dyn Placement, keys: &[Vec<u8>]) -> Self; }

// disruption.rs — fraction remapped on one membership change. theoretical_minimum = 1/N (remove)
// or 1/(N+1) (add) per Karger; correct impls stay within 2× of it.
#[derive(Debug, Clone)]
pub struct DisruptionReport { pub keys_total: u64, pub keys_moved: u64, pub fraction_moved: f64, pub theoretical_minimum: f64 }
impl DisruptionReport { pub fn measure(before: &dyn Placement, after: &dyn Placement, keys: &[Vec<u8>]) -> Self; } // diff locate per key
