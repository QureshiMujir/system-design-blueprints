Core schema & types (§11)

/// FINAL_BLUEPRINT §11.1
#[derive(Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug)]
pub struct SnapshotId(pub [u8; 32]);     // BLAKE3 of (source-root, timestamp, salt)
#[derive(Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug)]
pub struct ManifestId(pub [u8; 32]);     // BLAKE3 of manifest body
#[derive(Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug)]
pub struct ChunkRef(pub [u8; 32]);       // BLAKE3 of chunk plaintext
#[derive(Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug)]
pub struct TreeNodeId(pub [u8; 32]);     // BLAKE3 of tree-node body
#[derive(Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug)]
pub struct KeyVersion(pub u32);          // monotonic; rotates on rekey
#[derive(Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug)]
pub struct RepoId(pub [u8; 16]);         // random per-repository identifier
#[derive(Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug)]
pub struct BackupId(pub [u8; 16]);       // per-backup run identifier (journal key)
/// FINAL_BLUEPRINT §11.2
pub struct Manifest {
    pub id:          ManifestId,
    pub parent:      Option<ManifestId>,
    pub repo_id:     RepoId,
    pub source_root: std::path::PathBuf,
    pub snapshot_id: SnapshotId,
    pub created_at:  chrono::DateTime<chrono::Utc>,
    pub key_version: KeyVersion,
    pub aead:        AeadAlgo,
    pub root:        TreeNodeId,
    pub stats:       ManifestStats,
    pub signature:   Ed25519Signature,
}

pub enum AeadAlgo { XChaCha20Poly1305, Aes256GcmSiv, None /* Phase 0 only; rejected by Phase-3+ validator */ }

pub struct ManifestStats {
    pub source_bytes: u64, pub chunk_count: u64, pub unique_chunks: u64,
    pub stored_bytes: u64, pub file_count: u64, pub directory_count: u64, pub elapsed_ms: u64,
}

pub struct ChunkEntry { pub chunk: ChunkRef, pub offset: u64, pub length: u32 }

pub enum TreeNode {
    File      { path: std::path::PathBuf, size: u64, mode: u32, uid: u32, gid: u32,
                mtime_ns: i64, ctime_ns: i64, xattrs: std::collections::BTreeMap<String, Vec<u8>>,
                chunks: Vec<ChunkEntry> },
    Directory { path: std::path::PathBuf, mode: u32, uid: u32, gid: u32, mtime_ns: i64,
                entries: Vec<TreeNodeId> },
    Symlink   { path: std::path::PathBuf, target: std::path::PathBuf, mode: u32, uid: u32, gid: u32, mtime_ns: i64 },
    Other     { path: std::path::PathBuf, kind: SpecialKind, mode: u32, uid: u32, gid: u32, mtime_ns: i64, dev: u64 },
}

pub enum SpecialKind { CharDevice, BlockDevice, Fifo, Socket }
/// FINAL_BLUEPRINT §11.3
pub enum RetentionPolicy {
    KeepLast  { n: u32 },
    KeepGfs   { last: u32, daily: u32, weekly: u32, monthly: u32, yearly: u32 },
    KeepWithin{ days: u32 },
    KeepAll,
    Custom(std::sync::Arc<dyn Fn(&[Manifest]) -> Vec<ManifestId> + Send + Sync>),
}
pub const DEFAULT_POLICY: &str = "keep-last=24, daily=7, weekly=4, monthly=12, yearly=2";
/// FINAL_BLUEPRINT §11.4 — master via Argon2id; purpose keys via HKDF-SHA256 (Appendix C)
pub struct MasterKey([u8; 32]);
pub struct ChunkKey([u8; 32]);          // HKDF(master, "bks.chunk.key/v1"   || chunk_ref || epoch)
pub struct ChunkNonce([u8; 24]);        // HKDF(master, "bks.chunk.nonce/v1" || chunk_ref || epoch)
pub struct ManifestKey([u8; 32]);       // HKDF(master, "bks.manifest.key/v1" || key_version)
pub struct ManifestSignKey([u8; 32]);   // Ed25519 seed; "bks.manifest.sign/v1" || key_version

pub struct KeyHierarchy { /* master, epoch:u64, version:KeyVersion */ }

impl KeyHierarchy {
    pub fn chunk_key(&self, chunk: ChunkRef) -> ChunkKey;
    pub fn chunk_nonce(&self, chunk: ChunkRef) -> ChunkNonce;
    pub fn manifest_key(&self) -> ManifestKey;
    pub fn sign_key(&self) -> ManifestSignKey;
}
/// FINAL_BLUEPRINT §11.5
#[async_trait::async_trait]
pub trait Repository: Send + Sync {
    async fn put         (&self, key: &Key, bytes: bytes::Bytes) -> Result<Etag>;
    async fn put_if_match(&self, key: &Key, bytes: bytes::Bytes, expected: Etag) -> Result<Etag>;
    async fn get         (&self, key: &Key) -> Result<bytes::Bytes>;
    async fn head        (&self, key: &Key) -> Result<ObjectHead>;
    async fn list        (&self, prefix: &str) -> Result<Vec<Key>>;
    async fn delete      (&self, key: &Key) -> Result<()>;
    fn repo_id(&self) -> RepoId;
}
pub struct Key(pub String);   // "chunks/aa/bbcc...", "manifests/<id>", "keys/<version>"
pub struct Etag(pub [u8; 32]);
pub struct ObjectHead { pub size: u64, pub etag: Etag, pub last_modified: chrono::DateTime<chrono::Utc> }
/// FINAL_BLUEPRINT §11.6
#[async_trait::async_trait]
pub trait SnapshotDriver: Send + Sync {
    async fn snapshot(&self, source_root: &std::path::Path) -> Result<SnapshotHandle>;
    async fn release (&self, handle: SnapshotHandle) -> Result<()>;
}
pub struct SnapshotHandle { pub id: SnapshotId, pub mount: std::path::PathBuf,
                            pub driver: SnapshotDriverKind, pub created_at: chrono::DateTime<chrono::Utc> }
pub enum SnapshotDriverKind { Btrfs, Zfs, Lvm, Fsfreeze, LiveWalk }

#[async_trait::async_trait]
pub trait Backup: Send + Sync {
    async fn backup(&self, source_root: &std::path::Path,
                    repo: std::sync::Arc<dyn Repository>, keys: std::sync::Arc<KeyHierarchy>) -> Result<Manifest>;
}
#[async_trait::async_trait]
pub trait Restore: Send + Sync {
    async fn restore(&self, manifest: &Manifest, repo: std::sync::Arc<dyn Repository>,
                     keys: std::sync::Arc<KeyHierarchy>, target: &std::path::Path) -> Result<RestoreReport>;
}
#[async_trait::async_trait]
pub trait Verify: Send + Sync {
    async fn verify(&self, manifest: &Manifest, repo: std::sync::Arc<dyn Repository>,
                    keys: std::sync::Arc<KeyHierarchy>) -> Result<VerifyReport>;
}
#[async_trait::async_trait]
pub trait Gc: Send + Sync {
    async fn run(&self, repo: std::sync::Arc<dyn Repository>, retained: &[Manifest]) -> Result<GcReport>;
}
/// FINAL_BLUEPRINT §11.7 — fixed error taxonomy; routed on error.kind in logs
pub enum BksError {
    RepoLockHeld, RepoNoSpace, RepoUnavailable, IfMatchFailed,
    ChunkCorrupt { chunk: ChunkRef, manifest: ManifestId },
    ChunkHashMismatch { chunk: ChunkRef, manifest: ManifestId },
    ChunkTruncated { chunk: ChunkRef }, ChunkMissing { chunk: ChunkRef },
    ManifestInvalid, ManifestIdMismatch, ManifestSignatureInvalid,
    AeadUnsupported, LayoutTooNew { found: u32, supported: u32 },
    CrossFsBoundary, SnapshotDriverError,
    PhaseDeferred(&'static str),   // anchor: PhaseDeferred("Deferred Features Register")
}
pub type Result<T> = std::result::Result<T, BksError>;
