Core schema & types (§11)

struct SpanContext { trace_id: [u8; 16], span_id: [u8; 8], trace_flags: u8,
                      trace_state: TraceState, is_remote: bool }
// owned by value; Copy; serializes to/from traceparent+tracestate; I1/I2/I4

struct Tracer { name: String, version: String, provider: Arc<TracerProviderInner> }
// Clone (Arc-backed); no interior mutation of identity; P15

impl Tracer {
    fn start_span(&self, name: &str, attrs: AttrMap) -> Span;
    fn start_active_span(&self, name: &str) -> SpanGuard;
}

struct Span { context: SpanContext, inner: SpanInner }
// owned; not Clone (single end() consumer); interior state mutable pre-end; not Send
// across an await without the async-local Context carrying it explicitly

impl Span {
    fn set_attribute(&mut self, key: &str, value: AttrValue);
    fn record_event(&mut self, name: &str, attrs: AttrMap);
    fn set_status(&mut self, code: StatusCode, message: &str);
    fn end(self);  // consumes; enqueues to SpanProcessor::on_end
}

struct SpanRecord { /* full field list: §8.1 */ }
// Send + Sync; serializes losslessly to/from OTLP (I11); the unit shipped to exporters
trait Sampler: Send + Sync {
    fn should_sample(&self, parent: &Context, trace_id: [u8;16], name: &str,
                      kind: SpanKind, attrs: &AttrMap, links: &[Link]) -> SamplingResult;
}
enum SamplingResult { Drop, RecordOnly, RecordAndSample }
// AlwaysOn | AlwaysOff | ParentBased | TraceIdRatioBased | ConsistentProbability

trait SpanProcessor: Send + Sync {
    fn on_start(&self, span: &Span, parent: &Context);
    fn on_end(&self, record: SpanRecord);
    fn force_flush(&self, timeout: Duration) -> Result<(), FlushError>;
    fn shutdown(&self) -> Result<(), FlushError>;
}
// SimpleSpanProcessor (sync) | BatchSpanProcessor (bounded MPSC + background task, I6)

trait Exporter: Send + Sync {
    fn export(&self, batch: Vec<SpanRecord>) -> Result<(), ExportError>;
    fn shutdown(&self) -> Result<(), ExportError>;
}
// OtlpGrpcExporter | OtlpHttpExporter | StdoutExporter | InMemoryExporter

struct Resource { attrs: AttrMap }
// required key service.name (enforced at TracerProvider construction, §13.9)

trait Propagator: Send + Sync {
    fn inject(&self, ctx: &Context, carrier: &mut dyn Carrier);
    fn extract(&self, carrier: &dyn Carrier) -> Option<Context>;
}
// TraceContextPropagator | BaggagePropagator | B3Propagator | JaegerPropagator

struct Context { span_context: Option<SpanContext>, baggage: Baggage }
// stored in tokio::task_local! (async) or std::thread_local! (sync); P15

trait Carrier { fn get(&self, key: &str) -> Option<&str>; fn set(&mut self, key: &str, v: String); }
// HashMapCarrier | HeaderMapCarrier | MetadataCarrier

struct Collector { pipelines: Vec<Pipeline>, tenant_registry: Arc<TenantRegistry> }
// constructed via CollectorBuilder; owns no application state, only routing config

trait Storage: Send + Sync {
    fn write(&self, batch: Vec<SpanRecord>) -> Result<(), StoreError>;
    fn lookup_trace(&self, trace_id: [u8;16]) -> Result<Trace, StoreError>;
    fn search(&self, query: Query) -> Result<SpanStream, StoreError>;
}

struct Query { ast: PredicateNode, time_range: (u64, u64), tenant: TenantId }
// parsed once (§19.7); PredicateNode: Match | And | Or | Not | Time (signatures only)

struct TenantId(u64);  // opaque identifier; quotas and isolation keyed on it, I8
