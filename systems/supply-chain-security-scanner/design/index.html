Core schema & types (§11)

// Canonical Package URL — RFC-compliant, normalized on construction.
struct Purl(String);
// Invariants:
//   - lowercase scheme "pkg"; percent-encoded namespace/name; version URL-encoded
//   - construction via Purl::parse() validates spec; invalid -> ScsError::InvalidPurl
//   - equality: byte-exact on normalized form; no case-folding at comparison time
//   - serialization: Display = inner String; Serde transparent
//   - ownership: owned String; Clone = O(n), cheap for typical lengths (~40 bytes)
//   - concurrency: Send + Sync; immutable after construction
//   - persistence: stored as VARCHAR; deserialization re-validates

enum Ecosystem { Npm, Cargo, PyPI, Go, Maven, RubyGems }

struct PackageId {
    ecosystem: Ecosystem,   // discriminant for version parsing + range semantics
    namespace: Option<String>, // e.g. npm scope "@scope", maven groupId
    name: String,           // bare name, lowercased for Npm/PyPI
}
// Invariants:
//   - (ecosystem, namespace, name) triple is canonical key in graph + index
//   - Eq/Hash by all three fields; namespace=None != namespace=Some("")
//   - From<Purl>: parses purl type + subpath -> PackageId; lossy (drops version/quals)
//   - no mutation after construction; Clone = O(n)
enum Version {
    Semver(semver::Version),          // npm, Cargo, Go (semver-coerced)
    Pep440(pep440::Version),          // PyPI
    Maven(maven_version::Version),    // Maven (composite numeric segments)
    Opaque(String),                   // Ruby Gems, fallback; compared lexicographically
}
// Invariants:
//   - constructed from (Ecosystem, &str); wrong variant -> ScsError::VersionParse
//   - Ord defined within same variant; cross-variant comparison -> panic (programmer error)
//   - serialization: String representation; round-trips through Display -> parse
//   - concurrency: Send + Sync; immutable

enum VersionReq {
    Semver(semver::VersionReq),       // >=1.2.3, ^1.2.3, ~1.2.3, *
    Pep440Req(pep440::Requirement),   // ~=1.4, >=2.0,<3.0
    MavenRange(maven_range::Range),   // [1.0,2.0), (,1.0]
    Opaque(String),
}
// Invariants:
//   - matches(version: &Version) -> bool; panics on ecosystem mismatch
//   - AffectedRange.satisfies(v) -> bool is the canonical half-open check
//   - no interior mutation; Send + Sync
enum HashAlgorithm { Sha256, Sha512, Blake3 }

struct Hash {
    algorithm: HashAlgorithm,
    digest: [u8; 64],   // zero-padded; SHA-256 uses first 32 bytes
}

enum Scope { Prod, Dev, Optional, Unknown }

enum Source {
    Lockfile { path: PathBuf },
    Manifest { path: PathBuf },
    Sbom     { path: PathBuf, format: SbomFormat },
}

struct Component {
    purl:     Purl,
    version:  Version,
    hashes:   Vec<Hash>,    // may be empty for manifest-only resolution
    licenses: Vec<License>, // SPDX expressions; empty if unknown
    source:   Source,
    direct:   bool,         // true iff declared in root manifest (not transitive)
    scope:    Scope,
}
// Invariants:
//   - purl.version segment == version.to_string() (enforced at construction)
//   - hashes: at most one entry per HashAlgorithm; deduped on insert
//   - direct=true implies depth==0 in the DependencyGraph from any root
//   - serialization: Serde with purl as primary key; hashes as hex strings
//   - ownership: fully owned; Clone = O(n) proportional to hashes+licenses count
//   - concurrency: Send + Sync; immutable after graph construction
struct Edge {
    from:        Purl,
    to:          Purl,
    req:         VersionReq,   // constraint declared by `from`
    scope:       Scope,
    optional:    bool,
}

struct DependencyGraph {
    roots:     Vec<Purl>,                           // direct roots (workspace members)
    nodes:     IndexMap<Purl, Component>,           // canonical node set; O(1) lookup
    edges:     Vec<Edge>,                           // directed from → to (dependsOn)
    adjacency: HashMap<Purl, Vec<usize>>,           // edge indices by source purl
}
// Invariants:
//   - DAG (no directed cycles); cycle detection run at finalize() -> ScsError::CyclicDependency
//   - every edge.to purl exists in nodes; referential integrity enforced at insert
//   - roots ⊆ nodes; roots are the only nodes with in-degree 0 (after pruning)
//   - deterministic iteration order: IndexMap preserves insertion order; insertion is
//     topological post-sort so output is stable across runs given same inputs
//   - path(purl) -> Vec<Vec<Purl>>: returns all simple paths from purl to any root
//     (used by Finding.path); max path length capped at 64 to prevent pathological O(n!)
//   - serialization: nodes as array, edges as array; adjacency rebuilt on load
//   - concurrency: read-only after finalize(); &DependencyGraph is Send + Sync
//   - mutation only via GraphBuilder (separate type); finalize() -> DependencyGraph is consuming
enum CvssVersion { V2, V31, V40 }

struct Cvss {
    version:  CvssVersion,
    vector:   String,       // canonical vector string, e.g. "CVSS:3.1/AV:N/..."
    score:    f32,          // 0.0–10.0; base score only
    severity: Severity,     // derived from score per spec thresholds
}

struct Epss {
    probability: f32,       // 0.0–1.0
    percentile:  f32,       // 0.0–1.0
    model_date:  NaiveDate,
}

enum Severity { Critical, High, Medium, Low, None, Unknown }

struct AffectedRange {
    ecosystem:     Ecosystem,
    introduced:    Option<Version>,  // None = beginning of time
    fixed:         Option<Version>,  // None = no fix; half-open [introduced, fixed)
    last_affected: Option<Version>,  // advisory-specified inclusive last; normalized to fixed = last_affected.next()
}
// Invariants:
//   - satisfies(v: &Version) -> bool: introduced <= v && (fixed.is_none() || v < fixed)
//   - last_affected and fixed are mutually exclusive in source; normalization picks fixed
//   - introduced=None matches all versions below fixed
//   - ecosystem must match queried version's variant; mismatch -> always false (never panic)

struct Advisory {
    id:            String,              // canonical form: "CVE-...", "GHSA-...", "OSVA-..."
    aliases:       Vec<String>,         // cross-source aliases (e.g. CVE + GHSA for same vuln)
    summary:       String,
    affected:      Vec<AffectedRange>,  // union of ranges; component affected if any range matches
    severity:      Option<Cvss>,
    epss:          Option<Epss>,
    fixed_versions: Vec<(Ecosystem, Version)>,
    withdrawn:     Option<DateTime<Utc>>,  // None = active; Some = retracted advisory
    source:        AdvisorySource,      // OSV | NVD | GHSA
    snapshot_id:   SnapshotId,          // feed snapshot this advisory was ingested from
}
// Invariants:
//   - withdrawn.is_some() -> advisory surfaced as Finding.vex_status = Withdrawn; NOT suppressed silently
//   - aliases deduplicated; canonical id is the most-specific available (CVE > GHSA > OSV)
//   - affected: empty vec means no versions listed (treated as affects-all unless withdrawn)
//   - serialization: full round-trip via serde; snapshot_id required (feed auditability invariant)
enum VexStatus {
    NotAffected,        // vendor-confirmed not affected (via VEX)
    AffectedConfirmed,  // vendor-confirmed affected
    Fixed,              // fixed version available
    UnderInvestigation,
    Withdrawn,          // advisory was retracted
    None,               // no VEX data
}

struct Finding {
    component:  Component,
    advisory:   Advisory,
    path:       Vec<Purl>,          // [affected_purl, ..., root_purl]; len >= 1
    severity:   Severity,           // may be overridden by policy after CVSS adjustment
    fixed:      Option<Version>,    // earliest fixing version for component.purl.ecosystem
    vex_status: VexStatus,
    suppressed: bool,               // true if waiver/ignore rule matched
}
// Invariants:
//   - path[0] == component.purl; path[last] ∈ DependencyGraph.roots (path completeness invariant)
//   - path length capped at 64; longer chains truncated + annotated with truncated=true flag
//   - suppressed findings still emitted in full report (never silently dropped); gating ignores suppressed
//   - severity = advisory.severity.map(|c| c.severity).unwrap_or(Severity::Unknown) before policy override
//   - Finding is value type; Clone used for multi-path duplicates; Eq by (component.purl, advisory.id)
//   - serialization: advisory.id + component.purl as composite key
enum SbomFormat { Spdx23, Spdx30, CycloneDx15, CycloneDx16 }

enum RelationshipKind {
    DependsOn,
    Contains,
    DescribedBy,
    DynamicLink,
    StaticLink,
    Other(String),
}

struct SbomRelationship {
    from:  String,          // element SPDXID or CycloneDX bom-ref
    to:    String,
    kind:  RelationshipKind,
}

struct Sbom {
    format:        SbomFormat,
    spec_version:  String,          // e.g. "SPDX-2.3", "1.5"
    serial_number: Option<String>,  // CycloneDX urn:uuid:...
    name:          String,          // document name
    namespace:     Option<String>,  // SPDX namespace URI
    created:       DateTime<Utc>,
    components:    Vec<Component>,
    relationships: Vec<SbomRelationship>,
    metadata:      HashMap<String, String>,  // tool name, version, authors
}
// Invariants:
//   - round-trip invariant: emit(parse(bytes)) produces structurally equivalent Sbom
//     (component set, relationship set, hashes preserved; timestamp may normalize to UTC)
//   - component.purl is the canonical cross-format identifier; bom-ref/SPDXID are internal
//   - duplicate components (same purl) merged with union of hashes; conflict -> ScsError::SbomConflict
//   - relationships: referential integrity (from + to must reference component in components set)
//   - serialization: lossless for SPDX JSON + CycloneDX JSON; XML supported read-only for CycloneDX
enum SlsaLevel { L0, L1, L2, L3 }

struct RekorInclusion {
    log_id:         String,
    log_index:      u64,
    integrated_time: DateTime<Utc>,
    inclusion_proof: Vec<u8>,   // raw proof bytes; verified separately
}

struct Provenance {
    purl:             Purl,
    attestation_type: String,              // e.g. "https://slsa.dev/provenance/v1"
    attestation_raw:  Option<Vec<u8>>,     // base64-decoded envelope bytes
    slsa_level:       Option<SlsaLevel>,
    rekor_inclusion:  Option<RekorInclusion>,
    verified:         bool,                // ONLY true after all checks pass (fail-closed)
    verification_err: Option<String>,      // populated when verified=false; required for auditability
}
// Invariants:
//   - FAIL-CLOSED: verified=false is the default; must be explicitly set to true by verifier
//   - verified=true requires: sig check passed AND cert chain valid AND Rekor inclusion valid (if present)
//   - verification_err must be Some when verified=false; empty string is not acceptable
//   - attestation_raw stored even on failure (for manual forensics)
//   - serialization: attestation_raw as base64; inclusion_proof as hex
struct License {
    expression: String,     // SPDX expression, e.g. "MIT OR Apache-2.0", "(GPL-2.0-only)"
    source:     LicenseSource,
}

enum LicenseSource {
    Manifest,           // declared in package manifest
    Sbom,               // from ingested SBOM
    InferredFromFile,   // heuristic from LICENSE file scan
    Unknown,
}

// Normative API on License:
//   parse_spdx_expression(s: &str) -> Result<SpdxNode, ScsError>
//   is_copyleft(expr: &SpdxNode) -> bool   // GPL-*, LGPL-*, AGPL-*, MPL-2.0
//   is_permissive(expr: &SpdxNode) -> bool // MIT, Apache-2.0, BSD-*
//   satisfies_policy(expr: &SpdxNode, policy: &LicensePolicy) -> PolicyResult
// Invariants:
//   - SPDX expression parser is strict; unknown license IDs -> ScsError::UnknownSpdxId (not silently accepted)
//   - LicenseOrLaterSuffix (+) resolved to OR with unknown future versions; treated as copyleft for safety
//   - LicenseSource::Unknown does not block scanning; surfaced as informational Finding
struct Waiver {
    id:          String,            // waiver identifier for audit log
    purl:        Option<Purl>,      // None = applies to all packages
    advisory_id: Option<String>,    // None = applies to all advisories
    reason:      String,            // mandatory human-readable justification
    expires:     Option<DateTime<Utc>>,
    created_by:  String,
}

struct Policy {
    min_severity_gate:  Severity,           // findings at or above this block CI
    license_deny:       Vec<SpdxNode>,      // expressions that trigger deny
    license_allow:      Vec<SpdxNode>,      // allowlist; deny takes precedence
    waivers:            Vec<Waiver>,
    vex_consume:        bool,               // whether VEX statuses suppress findings
    fail_on_unknown_license: bool,
    max_age_days_unfixed: Option<u32>,      // escalate if no fix for N days
}

enum PolicyResult {
    Pass,
    Fail { reasons: Vec<String> },
    Warn { reasons: Vec<String> },
}
// Invariants:
//   - expired waivers are ignored; waiver application logged in report regardless
//   - deny overrides allow for same expression
//   - vex_consume=true: VexStatus::NotAffected -> suppressed=true but Finding still emitted
//   - gating decision: PolicyResult::Fail -> process exit code 1 (CI gate blocks)
//   - deterministic: same Policy + Findings -> same PolicyResult (no randomness in rule eval)
enum ScsError {
    // Parse errors
    InvalidPurl     { input: String, reason: String },
    VersionParse    { input: String, ecosystem: Ecosystem, reason: String },
    ManifestParse   { path: PathBuf, ecosystem: Ecosystem, source: Box<dyn Error + Send + Sync> },
    SbomParse       { path: PathBuf, format: SbomFormat, source: Box<dyn Error + Send + Sync> },
    UnknownSpdxId   { id: String },

    // Graph errors
    CyclicDependency { cycle: Vec<Purl> },
    UnresolvedDep    { from: Purl, requirement: String },

    // Advisory / matching errors
    AdvisoryFeedSync { source_url: String, source: Box<dyn Error + Send + Sync> },
    StaleSnapshot    { snapshot_id: SnapshotId, age: Duration },

    // Provenance errors
    SignatureVerification  { purl: Purl, reason: String },
    RekorVerification      { purl: Purl, reason: String },
    AttestationParse       { purl: Purl, source: Box<dyn Error + Send + Sync> },

    // SBOM errors
    SbomConflict    { purl: Purl, field: String },
    RoundTripFail   { format: SbomFormat, diff: String },

    // Policy / report errors
    PolicyLoad      { path: PathBuf, source: Box<dyn Error + Send + Sync> },
    ReportWrite     { path: PathBuf, source: Box<dyn Error + Send + Sync> },

    // I/O
    Io              { path: PathBuf, source: std::io::Error },

    // Internal invariant violations (programmer errors)
    InvariantViolation { message: String },
}
// Invariants:
//   - all variants are Send + Sync (required for use in async contexts and ? in threads)
//   - InvariantViolation is unrecoverable; triggers panic in debug, structured log + exit(2) in release
//   - SignatureVerification and RekorVerification are NOT mapped to warnings; fail-closed invariant
//   - ScsError implements Display with structured context; never exposes raw OS error strings alone
//   - thiserror derive used; no anyhow in library crates (only in scs-cli for context chaining)
struct SnapshotId {
    source:    AdvisorySource,   // OSV | NVD | GHSA
    timestamp: DateTime<Utc>,   // fetch time (UTC, second precision)
    etag:      Option<String>,  // HTTP ETag for incremental sync
    sha256:    [u8; 32],        // hash of raw feed bytes (auditability)
}

enum AdvisorySource { Osv, Nvd, Ghsa }

struct FeedSnapshot {
    id:         SnapshotId,
    advisories: Vec<Advisory>,  // normalized, deduped
    indexed_at: DateTime<Utc>,
}
// Invariants:
//   - SnapshotId included in every Finding and every report (feed freshness invariant)
//   - sha256 computed over raw bytes before parsing; validated on load from disk cache
//   - two snapshots with same source + timestamp + sha256 are identical (safe to skip re-index)
//   - FeedSnapshot not persisted as single blob; advisories stored in per-ecosystem inverted index
//     with SnapshotId recorded separately in metadata table
// Manifest parsing trait — implemented by each ecosystem parser in scs-manifest
trait ManifestParser {
    fn ecosystem(&self) -> Ecosystem;
    fn can_parse(&self, path: &Path) -> bool;              // filename/extension check
    fn parse(&self, path: &Path) -> Result<Vec<Component>, ScsError>;
    fn parse_lockfile(&self, path: &Path) -> Result<Vec<Component>, ScsError>;
}
// Invariant: parse() never installs/executes; reads file bytes only

// Version satisfiability — implemented per AffectedRange
trait VersionSatisfies {
    fn satisfies(&self, version: &Version) -> bool;
}
// Invariant: half-open semantics [introduced, fixed); panics on ecosystem mismatch

// Policy evaluation trait
trait PolicyEvaluator {
    fn evaluate(&self, findings: &[Finding], policy: &Policy) -> PolicyResult;
}
// Invariant: deterministic; same inputs -> same output; no side effects

// Report emitter trait
trait ReportEmitter {
    fn emit(&self, findings: &[Finding], policy_result: &PolicyResult, writer: &mut dyn Write)
        -> Result<(), ScsError>;
}

// Provenance verifier trait
trait ProvenanceVerifier {
    fn verify(&self, purl: &Purl, hints: &Provenance) -> Result<Provenance, ScsError>;
}
// Invariant: fail-closed — error return or verified=false, never silently verified=true
// Scan lifecycle state machine (scs-cli / runner.rs):
//   Init -> Parsed -> Resolved -> VulnMatched -> ProvenanceChecked -> PolicyApplied -> Reported

// Concurrency model:
//   - Parsing (scs-manifest): embarrassingly parallel per-file; rayon par_iter over discovered files
//   - Graph construction (scs-resolve): single-threaded builder; finalize() produces immutable graph
//   - Advisory matching (scs-match): read-only graph + read-only index; rayon par_iter over nodes
//   - Provenance (scs-provenance): async (tokio); network I/O to Rekor/Fulcio; bounded concurrency (semaphore)
//   - Threat heuristics (scs-threat): CPU-bound; rayon par_iter; no shared mutable state
//   - Policy (scs-policy): single-threaded; O(findings) deterministic pass
//   - Reporting (scs-report): single-threaded writer; output is ordered by (severity desc, purl asc)

// Shared state rules:
//   - DependencyGraph: immutable after finalize(); Arc<DependencyGraph> shared across threads
//   - FeedSnapshot / advisory index: Arc<RwLock<Index>>; writers hold lock only during sync window
//   - No global mutable state; all configuration passed via function arguments
//   - Cancellation: tokio CancellationToken propagated to all async tasks; drop cleans up
