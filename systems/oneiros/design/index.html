Core schema & types (§11)

type World = {
  id: WorldId;                          // ULID, stable forever
  ownerUserId: UserId;
  createdAt: ISO8601;
  schemaVersion: SemVer;                 // current "1.0.0"
  title: string;
  toneEnvelope: {
    arousalRange: [number, number];      // hard rails, Composer cannot exceed (I4)
    valenceRange: [number, number];
    contentProfileId: ContentProfileId;
  };
  cosmology: string;                     // Composer-authored prose
  characters: CharacterId[];
  threads: ThreadId[];
  locations: LocationId[];
  vibeHistory: VibeRecord[];             // last N=200
  lastResumedAt: ISO8601;
  totalSessionHours: number;
  snapshotPointer: SnapshotId;
};
// Ownership: PWS. Mutation: Composer only (I1, P1). Concurrency: single-writer
// per world, session-orchestrator-serialized. Persistence: Postgres row +
// object-storage cosmology blob. Serialization: JSON over PWS client API.
// Compatibility: additive-only (P8); schemaVersion non-decreasing (I6).

type PersistentCharacter = {
  id: CharacterId;
  worldId: WorldId;
  schemaVersion: SemVer;
  name: string;
  archetype: string;
  appearance: { descriptionProse: string; referenceAssetIds: AssetId[]; seed: string };
  traits: string[];
  memories: MemoryRecord[];              // condensed past-scene summaries
  lastAppearedSceneId: SceneId | null;
};
// Ownership: PWS, keyed by worldId. Mutation: Composer only, identity fields
// change only via a logged world-drift event (I2). Concurrency: same lock as
// World. Persistence: Postgres row. Serialization: JSON. Compatibility:
// additive-only.
type Scene = {
  id: SceneId;
  worldId: WorldId;
  index: number;                          // monotonic, no gaps (I5)
  plannedAt: ISO8601;
  completedAt?: ISO8601;
  vibe: Vibe;
  mood: Mood;
  modalitySlots: ModalityKind[];           // e.g. ["text","image","music"]
  charactersPresent: CharacterId[];
  assetIds: AssetId[];
  safetyAudit: { classifiedAt: ISO8601; decision: "pass"|"soft-warn"|"block"; sleepModeChecked: boolean };
  costUsd: number;
};
// Ownership: PWS. Mutation: append-only (write-once at commit). Concurrency:
// CAS on (worldId, index) enforces I5. Persistence: Postgres row. Serialization:
// JSON. Compatibility: additive-only.

type ModalityAsset = {
  id: AssetId;
  sceneId: SceneId;
  kind: ModalityKind;
  adapterId: string;
  sizeBytes: number;
  edgeCacheKey: string;
  provenance: { promptHash: string; seed: string; modelVersion: string };  // I8
  createdAt: ISO8601;
  evictedAt?: ISO8601;
};
// Ownership: PWS metadata row + object-storage/CDN payload. Mutation: write-once,
// evictedAt is the only later mutation. Concurrency: none (immutable after
// creation). Persistence: Postgres row + S3-compatible blob. Serialization:
// JSON metadata, binary payload. Compatibility: additive-only; provenance
// fields are mandatory (I8).
type Stream = {
  id: StreamId;
  userId: UserId;
  worldId: WorldId;
  startedAt: ISO8601;
  endedAt?: ISO8601;
  mode: "free" | "sleep" | "meditate" | "bedtime-story-children";
  scenes: SceneId[];
  totalCostUsd: number;
  exitReason: "viewer-ended" | "sleep-detected" | "timeout" | "safety-halt" | null;
};
// Ownership: PWS, scoped to userId (I12). Mutation: session orchestrator
// appends scenes/updates totalCostUsd. Concurrency: single active Stream per
// (userId, worldId) at a time. Persistence: Postgres row. Compatibility:
// additive-only.

type Seed = {                             // type defined now; feature deferred
  id: SeedId;                              // 256-char base62
  worldId: WorldId;
  authorUserId: UserId;
  payload: { cosmology: string; characters: unknown[]; locations: unknown[]; initialVibe: Vibe };
  signature: string;                        // signed, tamper-evident
};
// Ownership: PWS. Mutation: write-once at creation. Concurrency: none.
// Persistence: Postgres row. Compatibility: additive-only. Feature build
// deferred — see Deferred Features Register.
type ModalityKind = "text" | "image" | "music" | "video" | "voice";

type Mood = {
  arousal: number;                         // -1 calming .. +1 intense
  valence: number;                         // -1 dark .. +1 joyful
  descriptors: string[];
  biometrics?: { heartRateBpm?: number; hrvMs?: number; sleepStage?: "awake"|"light"|"deep"|"rem" };
};

type AdapterCapabilities = {
  readonly id: string;                       // e.g. "image-sdxl-successor-v1"
  readonly kind: ModalityKind;
  readonly version: SemVer;
  readonly latency: { p50Ms: number; p95Ms: number; p99Ms: number };
  readonly costPerUnit: number;               // USD
  readonly supportsMood: boolean;
  readonly sleepModeApproved: boolean;
};
// Ownership: adapter process, published at registration. Mutation: immutable
// per adapter version. Concurrency: read-only after registration. Persistence:
// in-memory capability registry, snapshotted to Postgres for audit.
// Serialization: JSON. Compatibility: adapter versions coexist (ADR-02-adjacent).
interface ModalityAdapter {
  readonly capabilities: AdapterCapabilities;
  health(): Promise<HealthReport>;
  render(req: AdapterRequest): Promise<AdapterResponse>;
  preWarm?(hint: PreWarmHint): Promise<void>;
  cancel(requestId: string): Promise<void>;   // I11 contract
}

interface Composer {
  planScenes(req: PlanRequest): Promise<ScenePlan[]>;
  ingestSceneCompletion(req: SceneCompletion): Promise<WorldDelta>;
  resumeWorld(worldId: WorldId, vibe?: Vibe): Promise<ScenePlan[]>;
}

interface SceneScheduler {
  schedule(plan: ScenePlan, budget: SceneBudget): Promise<ScheduledScene>;
  events(streamId: StreamId): AsyncIterable<StreamEvent>;
}
// Ownership: one process per adapter kind implements ModalityAdapter; one
// Composer instance per active session (I1); one Scheduler per session.
// Mutation: n/a (behavioral interfaces). Concurrency: Composer single-threaded
// per session; ModalityAdapter instances are called concurrently across
// scenes/sessions, serialized internally per requestId. Persistence: n/a
// (interfaces); implementations persist via PWS. Compatibility: interface is
// versioned as a whole (ADR-25); adding an optional method is non-breaking,
// removing/changing a signature requires a major bump.
type Vibe = { raw: string; mode: "free"|"sleep"|"meditate"|"bedtime-story-children"; submittedAt: ISO8601 };

type VibeRecord = { vibe: Vibe; sceneIndexAtSubmission: number };  // one entry per World.vibeHistory item

type MemoryRecord = { sceneIdRange: [SceneId, SceneId]; condensedSummary: string; createdAt: ISO8601 };

type ContentProfileId = string;
type ContentProfile = {
  id: ContentProfileId;
  matureThemes: "off" | "mild" | "moderate" | "mature";
  violence: "none" | "stylized" | "realistic";
  horror: "off" | "mild" | "moderate" | "intense";
  isChildrensPreset: boolean;                 // immutable once true, PIN-gated to disable
  lastChangedAt: ISO8601;                      // 30-min cooldown enforced against this
};
type AdapterRequest = { requestId: string; sceneId: SceneId; mood: Mood; scenePlanFragment: unknown; budget: { deadlineMs: number; maxCostUsd: number } };
type AdapterResponse = { requestId: string; assetDraft: Omit<ModalityAsset, "edgeCacheKey"|"evictedAt">; realizedCostUsd: number; realizedLatencyMs: number };
type HealthReport = { healthy: boolean; lastCheckedAt: ISO8601; errorRateWindow: number };
type PreWarmHint = { likelyModality: ModalityKind; likelyMood: Mood };

type PlanRequest = { worldId: WorldId; vibe: Vibe; sceneCount: number };
type ScenePlan = { sceneIndex: number; mood: Mood; charactersPresent: CharacterId[]; modalitySlots: ModalityKind[] };
type SceneCompletion = { sceneId: SceneId; assetIds: AssetId[]; safetyAudit: Scene["safetyAudit"] };
type WorldDelta = { worldId: WorldId; characterUpdates: Partial<PersistentCharacter>[]; newThreads: ThreadId[] };

type SceneBudget = { deadlineMs: number; maxCostUsd: number; maxModalitySlots: number };
type ScheduledScene = { sceneId: SceneId; committedAt: ISO8601; assetIds: AssetId[] };
type StreamEvent =
  | { kind: "scene-start"; sceneId: SceneId }
  | { kind: "asset-ready"; sceneId: SceneId; assetId: AssetId }
  | { kind: "scene-end"; sceneId: SceneId }
  | { kind: "world-delta"; delta: WorldDelta };
type Brand<K, T extends string> = K & { readonly __brand: T };

type WorldId = Brand<string, "WorldId">;             // ULID
type UserId = Brand<string, "UserId">;
type CharacterId = Brand<string, "CharacterId">;
type ThreadId = Brand<string, "ThreadId">;
type LocationId = Brand<string, "LocationId">;
type SceneId = Brand<string, "SceneId">;
type AssetId = Brand<string, "AssetId">;
type StreamId = Brand<string, "StreamId">;
type SeedId = Brand<string, "SeedId">;               // 256-char base62
type SnapshotId = Brand<string, "SnapshotId">;
type ISO8601 = Brand<string, "ISO8601">;
type SemVer = Brand<string, "SemVer">;
