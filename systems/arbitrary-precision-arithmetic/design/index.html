Core schema & types (§11)

type Limb           = u64;
type DoubleLimb     = u128;   // limb*limb + carry; never stored
type SignedDoubleLimb = i128; // borrow capture for subtraction; never stored
enum Sign { Minus, NoSign, Plus }
enum ParseError { EmptyString, InvalidDigit { pos: usize, ch: char }, InvalidBase(u32) }
struct DivByZero;
struct BigUint { limbs: Vec<Limb> }
struct Limbs<'a>    { inner: &'a [Limb] }
struct LimbsMut<'a> { inner: &'a mut [Limb] }
enum MulAlgo { Schoolbook, Karatsuba, Toom3, Ntt }
struct DivRem { quotient: BigUint, remainder: BigUint }
struct Montgomery { modulus: BigUint, n_prime: BigUint, r2_mod_n: BigUint }
struct Barrett { modulus: BigUint, mu: BigUint, k: usize }
struct NttPlan { primes: Vec<u64>, roots: Vec<u64>, len: usize }
struct BigInt { sign: Sign, mag: BigUint }
enum Base { Base10, Base16, BaseN(u32) }
struct BigRational { sign: Sign, num: BigUint, den: BigUint }
// bn-uint
impl BigUint {
    fn new(limbs: Vec<Limb>) -> Self;              // normalizes on construction
    fn normalize(&mut self);                       // strip trailing zero limbs; enforce I1
    fn bit_length(&self) -> usize;                 // floor(log2(self)) + 1; 0 for zero
    fn from_u64(v: u64) -> Self;
    fn from_u128(v: u128) -> Self;
    fn to_u64(&self) -> Option<u64>;               // None if > u64::MAX
}

// bn-add
fn add_limbs(dst: LimbsMut<'_>, a: Limbs<'_>, b: Limbs<'_>) -> Limb; // returns carry
fn sub_limbs(dst: LimbsMut<'_>, a: Limbs<'_>, b: Limbs<'_>) -> Limb; // requires a>=b; returns borrow
fn add_mul_limb(acc: LimbsMut<'_>, b: Limbs<'_>, c: Limb) -> Limb;   // acc += b*c; returns carry

// bn-mul
fn mul(a: &BigUint, b: &BigUint) -> BigUint;                          // dispatches via MulAlgo
fn select_mul_algo(n_limbs: usize) -> MulAlgo;
fn schoolbook_mul(a: Limbs<'_>, b: Limbs<'_>, out: LimbsMut<'_>);
fn karatsuba_mul(a: &BigUint, b: &BigUint) -> BigUint;
fn toom3_mul(a: &BigUint, b: &BigUint) -> BigUint;
// bn-fftmul
fn ntt_mul(a: &BigUint, b: &BigUint, plan: &NttPlan) -> BigUint;

// bn-div
fn divrem(a: &BigUint, b: &BigUint) -> Result<DivRem, DivByZero>;
fn divrem_single(a: &BigUint, b: Limb) -> Result<(BigUint, Limb), DivByZero>;
fn newton_reciprocal(d: &BigUint, k: usize) -> BigUint;               // floor(b^k / d)

// bn-mod
fn barrett_reduce(x: &BigUint, ctx: &Barrett) -> BigUint;
fn montgomery_mul(a: &BigUint, b: &BigUint, ctx: &Montgomery) -> BigUint;
fn modpow(base: &BigUint, exp: &BigUint, ctx: &Montgomery) -> BigUint;
fn modpow_ct(base: &BigUint, exp: &BigUint, ctx: &Montgomery) -> BigUint; // constant-time (P9)

// bn-gcd
fn gcd(a: &BigUint, b: &BigUint) -> BigUint;
fn lehmer_gcd(a: &BigUint, b: &BigUint) -> BigUint;
fn ext_gcd(a: &BigInt, b: &BigInt) -> (BigUint, BigInt, BigInt);       // (gcd, s, t): a*s + b*t = gcd
fn mod_inverse(a: &BigUint, m: &BigUint) -> Option<BigUint>;           // None if not invertible

// bn-convert
fn format(x: &BigUint, base: Base) -> String;
fn parse(s: &str, base: Base) -> Result<BigUint, ParseError>;

// bn-rational
impl BigRational {
    fn new(num: BigInt, den: BigInt) -> Result<Self, DivByZero>;
    fn reduce(&mut self);                          // enforce I10
    fn to_decimal(&self, scale: usize) -> String;  // scale decimal places, truncating
}
