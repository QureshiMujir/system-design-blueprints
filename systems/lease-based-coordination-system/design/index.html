Core schema & types (§11)

LeaseId: NewType[str]                       # opaque, globally unique per resource generation
    # ownership: assigned by Lease Manager at grant time; immutable thereafter
    # persistence: durable via ConsensusStore; serialized as opaque string

Duration: NewType[float]                    # seconds, always non-negative
GuardInterval: NewType[float]               # seconds; derived, never client-supplied (§8.2)
ClockSkew: NewType[float]                   # milliseconds; sampled, not configured directly
Deadline: NewType[float]                    # monotonic-clock-relative absolute expiry instant

FencingToken: NewType[int]                  # strictly monotonic per resource (I2)
    # concurrency: single-writer (Fencing-Token Issuer) per resource key
    # persistence: high-water mark durable via ConsensusStore
Term: NewType[int]                          # alias domain for election/consensus epoch numbers
class LeaseState(Enum):
    FREE = "free"
    GRANTED = "granted"
    RENEWING = "renewing"
    EXPIRED = "expired"
    REVOKED = "revoked"
    # timing sensitivity: EXPIRED transition is clock-driven (Timer Wheel), never client-driven

@dataclass(frozen=True)
class Lease:
    lease_id: LeaseId
    resource_key: str
    owner_id: str
    state: LeaseState
    fencing_token: FencingToken
    granted_at: Deadline                    # monotonic-clock reading at grant commit
    expires_at: Deadline                    # grantor-side deadline, guard-adjusted (§8.2)
    generation: int                         # bumped on every grant, never on renewal
    # ownership: mutation only via Lease Manager; frozen dataclass forces copy-on-transition
    # concurrency: single-writer per resource_key (serialized through ConsensusStore commit order)
    # persistence: full struct durable; serialization: versioned protobuf/msgpack (ADR-23 compat)
    # compatibility: new fields additive-only (§27)
@dataclass(frozen=True)
class Grant:
    resource_key: str
    owner_id: str
    ttl: Duration
    requested_at: Deadline
    # timing sensitivity: requested_at used only for latency measurement, never for the
    # expiry decision itself (grantor's own clock reading at commit time is authoritative)

@dataclass(frozen=True)
class Renew:
    lease_id: LeaseId
    owner_id: str
    current_token: FencingToken
    # concurrency: rejected if lease.generation has advanced past current_token's generation (I5)

@dataclass(frozen=True)
class Revoke:
    lease_id: LeaseId
    reason: str                             # "admin" | "failsafe_trip" | "session_expiry"

@dataclass(frozen=True)
class Expiry:
    lease_id: LeaseId
    detected_at: Deadline
    margin_ms: float                        # I4 self-check value, must be >= 0
SessionId: NewType[str]

@dataclass(frozen=True)
class Session:
    session_id: SessionId
    owner_id: str
    heartbeat_deadline: Deadline
    attached_leases: FrozenSet[LeaseId]
    # ownership: mutation only via Session Tracker; attached_leases updated via copy-on-write
    # persistence: durable via ConsensusStore (mirrors Lease durability, I6/I7)
class LeaseManager(Protocol):
    def grant(self, req: Grant) -> Union[Lease, "RuntimeError"]: ...
    def renew(self, req: Renew) -> Union[Lease, "RuntimeError"]: ...
    def revoke(self, req: Revoke) -> Union[None, "RuntimeError"]: ...
    def on_expiry_sweep(self, tick: Deadline) -> Sequence[Expiry]: ...
    # concurrency: single logical owner per resource_key; internally may shard by resource_key
    # timing sensitivity: every method call is clock-boundary-crossing (§3); must route through
    #   ClockMonitor before committing any grant/renew decision

class LeaderElection(Protocol):
    def campaign(self, election_name: str, candidate_id: str) -> "Leadership": ...
    def resign(self, leadership: "Leadership") -> None: ...
    def observe(self, election_name: str) -> Optional["Leadership"]: ...

@dataclass(frozen=True)
class Leadership:
    election_name: str
    leader_id: str
    fencing_token: FencingToken
    lease: Lease
@dataclass(frozen=True)
class LockRequest:
    resource_key: str
    requester_id: str
    ttl: Duration
    # ordering: FIFO position assigned at enqueue time (ADR-07); bounded queue length (§4)

class Lock(Protocol):
    def acquire(self, req: LockRequest) -> Union[Lease, "RuntimeError"]: ...   # may block on queue
    def release(self, lease_id: LeaseId, owner_id: str) -> None: ...
class ClockMonitor(Protocol):
    def sample_skew(self) -> ClockSkew: ...
    def is_within_bound(self) -> bool: ...
    def trip_failsafe(self, reason: str) -> None: ...                        # I9
    # timing sensitivity: this IS the timing-sensitivity boundary; every other component
    #   treats ClockMonitor's output as ground truth for "is the clock model still valid"

class TimerWheel(Protocol):
    def schedule(self, lease_id: LeaseId, deadline: Deadline) -> None: ...
    def cancel(self, lease_id: LeaseId) -> None: ...
    def tick(self, now: Deadline) -> Sequence[LeaseId]: ...                  # ADR-15, I11
class ConsensusStore(Protocol):
    """Delegated durability boundary (§1, ADR-02). Any correct implementation — single-authority
    stub or Raft/Paxos-backed — must satisfy: commit is durable-before-ack, commits are totally
    ordered, and a crashed-then-restarted or failed-over instance replays exactly the committed
    prefix (I6)."""
    def propose(self, command: bytes) -> "CommitResult": ...
    def read_committed(self, key: str) -> Optional[bytes]: ...
    def watch(self, key_prefix: str) -> Iterator[bytes]: ...

@dataclass(frozen=True)
class CommitResult:
    committed: bool
    commit_index: int
class ResourceGuard(Protocol):                 # a.k.a. TokenChecker (I3)
    def check(self, token: FencingToken) -> "Verdict": ...
    def highest_seen(self) -> FencingToken: ...

class Verdict(Enum):
    ACCEPT = "accept"
    REJECT_STALE_TOKEN = "reject_stale_token"
class RuntimeError(Exception):
    """Base of the error taxonomy; every variant below is a distinct, matchable exception type."""

class LeaseConflict(RuntimeError): ...          # I1/I5 violation attempt
class FencingViolation(RuntimeError): ...       # I2/I3 violation attempt
class ClockSkewExceeded(RuntimeError): ...      # I9 trip
class QuorumUnavailable(RuntimeError): ...      # ConsensusStore cannot commit
class SessionExpired(RuntimeError): ...         # I7 cascade in progress
class SerializationFailure(RuntimeError): ...   # wire-format decode error (ADR-21 fuzz target)
class RenewalRejected(RuntimeError): ...        # I5/I8 rejection
