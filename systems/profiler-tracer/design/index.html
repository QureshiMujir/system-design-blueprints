Core schema & types (§11)

pub struct Process { pid: libc::pid_t, modules: Vec<Module>,
    build_id_index: HashMap<BuildId, ModuleId>, name: SmolStr }
// owner: agent lifetime; mutation: refresh_modules() only; concurrency: &mut behind one
// worker-thread owner; persistence: none (rebuilt from /proc on attach); no wire form.

pub struct Module { pub id: ModuleId, pub path: PathBuf, pub build_id: Option<BuildId>,
    pub base_address: u64, pub text_range: Range<u64>, pub is_executable: bool,
    pub is_anonymous: bool }
// immutable after construction (Clone); serialized only as (build_id, offset) pairs (I4, P3).

pub struct BuildId(pub [u8; 20]);        // Copy, Eq, Hash — GNU build-id, wire-stable (I4)
pub struct ModuleId(pub u32);            // Copy, Eq, Hash — process-local only, never on wire

impl Process {
    pub fn attach(pid: libc::pid_t) -> Result<Self, ProcError>;
    pub fn refresh_modules(&mut self) -> Result<(), ProcError>;
    pub fn module_for_address(&self, ip: u64) -> Option<&Module>;
    pub fn name(&self) -> &str;
}
pub struct Thread { pub tid: libc::pid_t, pub name: SmolStr, pub stack_top: u64,
    pub stack_bottom: u64 }
// owner: sampler registry; mutation: none post-construction; no persistence, no wire form.

impl Thread {
    pub fn current() -> Self;
    pub fn list_for(pid: libc::pid_t) -> Result<Vec<Thread>, ProcError>;
    pub fn set_name(name: &str) -> Result<(), ProcError>;
}
pub const MAX_STACK_DEPTH: usize = 127;                    // I6

#[derive(Copy, Clone)]
pub struct Sample { pub timestamp_ns: u64, pub tid: libc::pid_t, pub cpu: u32,
    pub kind: SampleKind, pub stack_len: u8, pub stack: [u64; MAX_STACK_DEPTH],
    pub value: u64 }
// owner: per-thread ring slot until drained (P10 window arena); mutation: producer writes
// once, consumer reads once; concurrency: SPSC (I1); persistence: none, ephemeral per window;
// wire form: none directly — aggregated into pprof/OTLP before export.

#[derive(Copy, Clone)]
pub enum SampleKind { OnCpu, OffCpu, Allocation, Free, LockWait, LockAcquire }

#[derive(Clone)]
pub struct Frame { pub ip: u64, pub function: SymbolId, pub file: SymbolId, pub line: u32,
    pub is_inlined: bool }
// owner: symbolizer call result, consumed immediately by aggregator; not retained past a window.
pub struct SymbolId(pub u32);            // Copy, Eq, Hash — interned handle (P9)

pub struct SymbolTable { arena: Bump, intern: HashMap<&'static str, SymbolId>,
    by_id: Vec<&'static str> }
// owner: interner arena (P10), freed at agent shutdown; mutation: intern() only, append-only;
// concurrency: single-writer (worker thread), read-many via Arc snapshot; no wire form —
// strings resolved only in the analysis tier (P3) or embedded in pprof's own string_table (I10).

impl SymbolTable {
    pub fn intern(&mut self, s: &str) -> SymbolId;
    pub fn resolve(&self, id: SymbolId) -> &str;
    pub fn len(&self) -> usize;
}
pub struct StackId(pub u32);             // Copy, Eq, Hash — parent-pointer tree leaf (§18.1)

pub struct StackTable { table: hashbrown::HashTable<StackId>, leaves: Vec<StackNode> }
struct StackNode { parent: Option<StackId>, function: SymbolId }
// owner: window arena (P10); mutation: intern() append-only within a window, cleared at
// window rollover; concurrency: single-writer worker thread; serialization: unfolds directly
// into pprof Location/Function tables (§8.5) or folded-stack lines (§18.3).

impl StackTable {
    pub fn intern(&mut self, frames: &[Frame]) -> StackId;
    pub fn unfold(&self, id: StackId, out: &mut Vec<SymbolId>);
}
pub trait Tracer: Send + Sync + 'static {
    fn start(&self, name: &'static str, attrs: &[KeyValue]) -> Span;
    fn current() -> Context;
    fn with_context<R>(ctx: Context, f: impl FnOnce() -> R) -> R;
}
// P11: the only public surface application code depends on. Implementations: GlobalTracer
// (production), InMemoryTracer (tests). Concurrency: Send+Sync, safe from any thread.

pub struct Span { inner: Arc<SpanInner> }
// owner: application call site until finish(); mutation: set_attribute/record_event/
// set_status append to inner (Arc<Mutex-free: single-writer-per-span discipline));
// on finish(), converts to FinishedSpan and moves into the batcher (§8.4 state machine).

impl Span {
    pub fn set_attribute(&self, key: &'static str, value: AttributeValue);
    pub fn record_event(&self, name: &'static str, attrs: &[KeyValue]);
    pub fn set_status(&self, status: SpanStatus);
    pub fn finish(self);
}

#[derive(Copy, Clone)]
pub struct Context { pub trace_id: TraceId, pub span_id: SpanId, pub trace_flags: u8,
    pub remote: bool }
// wire form: W3C traceparent (§8.2); immutable, Copy — passed by value across await points.

pub struct TraceId(pub [u8; 16]);        // Copy, Eq, Hash — OTLP wire: raw bytes (§8.6)
pub struct SpanId(pub [u8; 8]);          // Copy, Eq, Hash — OTLP wire: raw bytes (§8.6)
pub trait SpanExporter: Send + Sync + 'static {
    fn export(&self, batch: &[FinishedSpan]) -> ExportFuture;
    fn shutdown(&self) -> ExportFuture;
}
// implementations: OtlpHttpExporter, OtlpGrpcExporter (feature="grpc"), StdoutExporter,
// InMemoryExporter (tests). Compatibility: every impl validates against I12 (§26.4).
pub trait ProfileBuilder: Send + 'static {
    fn add_sample(&mut self, stack: StackId, value: u64);
    fn finish(self: Box<Self>) -> Result<Vec<u8>, BuildError>;   // gzip-compressed pprof (I10,I11)
}
// owner: window lifetime; mutation: add_sample append-only, finish() consumes; persistence:
// none in-process — output bytes go to §21.5 object store or stdout (prof record).
pub trait Sampler: Send + 'static {
    fn start(&mut self, target: SampleTarget) -> Result<(), SamplerError>;
    fn drain(&mut self, into: &mut Vec<Sample>) -> Result<usize, SamplerError>;
    fn stop(&mut self) -> Result<(), SamplerError>;
}
pub enum SampleTarget { Thread { tid: libc::pid_t }, Process { pid: libc::pid_t },
    Cgroup { fd: RawFd }, System }
// implementations: SigprofSampler, PerfSampler, EbpfCpuSampler, OffCpuSampler (§14–§15) —
// all backend-agnostic to callers (P13, I5).
pub trait Unwinder: Send + Sync {
    fn walk(&self, regs: &Registers, memory: &dyn StackMemory,
        out: &mut [u64; MAX_STACK_DEPTH]) -> Result<usize, UnwindError>;
}
pub struct Registers { pub rip: u64, pub rsp: u64, pub rbp: u64 } // aarch64: pc, sp, fp, lr
pub trait StackMemory { fn read(&self, addr: u64, len: usize, into: &mut [u8])
    -> Result<(), MemoryError>; }
// implementations: FpUnwinder, DwarfUnwinder (§16); dispatch is per-Module (I5, P5).
pub trait Symbolizer: Send + Sync {
    fn resolve(&self, module: &Module, offset: u64) -> Result<Vec<Frame>, SymbolError>;
}
// returns >1 Frame when offset is inlined (§17.4); deepest frame has is_inlined == false.
pub struct Agent { sampler: Box<dyn Sampler>, aggregator: Aggregator,
    exporter: Arc<dyn ProfileExporter>, config: AgentConfig }
#[derive(Clone)]
pub struct AgentConfig { pub frequency_hz: u32, pub window_seconds: u32, pub on_cpu: bool,
    pub off_cpu: bool, pub heap: bool, pub upload_url: String, pub api_key: Option<String> }
// owner: process lifetime, supervised (§21.3); mutation: config immutable post-load;
// persistence: config from env/file at startup only; no dynamic reload in Phase 0.
