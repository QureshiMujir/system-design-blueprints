Core schema & types (§11)

// physics/math/vec.py — value types, immutable by convention, no persistence,
// no determinism sensitivity beyond IEEE 754 arithmetic order.
class Vec2:
    x: float; y: float

class Vec3:
    x: float; y: float; z: float
    def dot(self, other: "Vec3") -> float: ...
    def cross(self, other: "Vec3") -> "Vec3": ...
    def normalize(self) -> "Vec3": ...          // determinism-sensitive: order of ops

// physics/math/quat.py — unit quaternion; mutated only via World-owned integration
// step; not user-mutated mid-step; determinism-sensitive (I7).
class Quat:
    x: float; y: float; z: float; w: float
    def normalize(self) -> "Quat": ...
    def to_mat3(self) -> "Mat3": ...

// physics/math/mat.py — 3x3 matrix, row-major; no persistence; recomputed, not stored
// across saves except as InertiaTensor.inv_world (a per-step cache, not canonical state).
class Mat3:
    m: "tuple[float, ...]"                       // 9 elements, row-major
// physics/math/transform.py — composite of position + orientation; canonical
// per-body state; serialized in world snapshots (§27); determinism-sensitive.
class Transform:
    position: Vec3
    orientation: Quat

// physics/body/rigid_body.py — BodyId is an opaque handle, not reused within a
// World's lifetime (I11 quarantine relies on stable identity). RigidBody is
// owned exclusively by World; mutated only inside step() or explicit setters
// that route through the wake-check (P8). Serialized in world snapshots.
BodyId = int                                      // opaque handle, monotonic alloc

class RigidBody:
    id: BodyId
    body_type: "Literal['Static', 'Kinematic', 'Dynamic']"
    transform: Transform                          // canonical pose, determinism-critical
    linear_velocity: Vec3                          // determinism-critical
    angular_velocity: Vec3                         // world frame, determinism-critical
    mass_data: "MassData"
    shapes: "list[Shape]"
    restitution: float
    friction: float
    sleep_state: "SleepState"
    quarantined: bool                              // I6, I11
// physics/body/mass_data.py — derived from shapes at construction; recomputed
// only on shape mutation (rare, not a per-step operation). Not user-serialized
// directly; reconstructible from shapes + density, so persisted as a cache only.
class MassData:
    mass: float
    inv_mass: float                                 // 0.0 for Static/Kinematic

class InertiaTensor:
    local: Mat3                                     // body-local, load-time constant
    inv_local: Mat3
    inv_world: Mat3                                  // per-step cache, NOT persisted
// physics/shapes/shape.py — Shape variants are immutable value data owned by a
// RigidBody; shared (by reference) across compound bodies is explicitly NOT
// supported in the core phases to keep ownership single-parent (simplicity over
// memory reuse). Determinism-neutral (geometry doesn't change mid-run in core
// phases). Serialized verbatim in scene files and snapshots.
class Shape:                                         // tagged union, see subtypes
    kind: "Literal['Sphere','Box','Capsule','ConvexPolytope','Plane']"

class SupportFunction:                               // protocol, not a class hierarchy
    def support(self, direction: Vec3) -> Vec3: ...  // P6 contract; pure, no side effects
// physics/broadphase/base.py — BroadPhase is a protocol implemented by
// SpatialHash and DynamicBvh (ADR-07); owns proxy bookkeeping only, never body
// state. ProxyId is stable across a body's lifetime; PairKey is a canonical,
// order-independent pair identity used for I4 determinism.
ProxyId = int
class Aabb:
    lo: Vec3; hi: Vec3                               // padded per ADR-07/P5

class PairKey:                                        // (min(a,b), max(a,b)) canonical
    a: BodyId; b: BodyId

class BroadPhase:                                      // protocol
    def update(self, proxies: "list[tuple[ProxyId, Aabb]]") -> None: ...
    def query_pairs(self) -> "list[PairKey]": ...      // order NOT guaranteed here;
                                                        // canonical sort happens later (ADR-19)
// physics/solver/manifold.py — ContactPoint/ContactManifold are per-step
// derived data, rebuilt every narrow-phase pass but carrying forward warm-start
// fields via ManifoldKey + feature_id matching (I12). Mutated only within the
// solver stage; NOT serialized in snapshots (recomputed from shapes+transforms;
// only accumulated impulses matter for warm start continuity across a save/load
// boundary, and a fresh cold-start after load is an accepted, documented cost).
class ManifoldKey:
    pair: PairKey
    shape_pair_kind: str

class ContactPoint:
    local_a: Vec3; local_b: Vec3                     // body-local, determinism-critical
    feature_id: int                                   // P4 persistence key
    depth: float
    normal_impulse: float                              // warm-start state, I8-bounded
    tangent_impulse_1: float; tangent_impulse_2: float // I9-bounded

class ContactManifold:
    key: ManifoldKey
    normal: Vec3                                        // body-A-local (P4 stability)
    points: "list[ContactPoint]"                        // <= 4, P6/§8 clipping output
// physics/solver/constraint.py — VelocityConstraint/PositionConstraint are
// ephemeral per-step solver working data, never persisted, rebuilt each step
// from ContactManifold/Joint state; concurrency: single-threaded per island in
// core phases (ADR-20), no cross-island mutation.
class ConstraintJacobian:
    linear_a: Vec3; angular_a: Vec3
    linear_b: Vec3; angular_b: Vec3

class VelocityConstraint:
    jacobian: ConstraintJacobian
    effective_mass: float
    bias: float                                          // restitution or joint-drift term
    lower: float; upper: float                            // I8/I9 clamp bounds

class PositionConstraint:
    reference: "ContactManifold | Joint"
    max_correction: float                                  // bounds deep-overlap recovery

class SolverConfig:
    velocity_iterations: int                                // default 8, P1
    position_iterations: int                                // default 3, P1
    penetration_slop: float                                  // default 0.002 m, I2
    restitution_velocity_threshold: float                    // default 1.0 m/s, ADR-15
// physics/islands/island_manager.py — Island is per-step derived data, rebuilt
// every step (I13); not persisted. physics/body/sleep_state.py — SleepState is
// canonical per-body state, serialized (a sleeping body should reload sleeping).
class Island:
    bodies: "list[BodyId]"
    manifolds: "list[ManifoldKey]"
    joints: "list[int]"                                      // indices into World.joints

class SleepState:
    state: "Literal['AllAwake','Settling','Asleep']"           // island-level (P8)
    timer: float
// physics/joints/joint.py — Joint variants are canonical, user-authored state;
// serialized in scene files and snapshots; mutated only via explicit joint APIs
// (limits, motor target), never by the solver's ephemeral working data.
class Joint:                                                  // tagged union
    kind: "Literal['Distance','Revolute','Prismatic','Ball','Weld','Motor']"
    body_a: BodyId; body_b: BodyId
    anchor_a: Vec3; anchor_b: Vec3

class DistanceJoint(Joint):
    rest_length: float

class RevoluteJoint(Joint):
    axis: Vec3; lower_angle: float; upper_angle: float

class PrismaticJoint(Joint):
    axis: Vec3; lower: float; upper: float

class MotorJoint(Joint):
    target_speed: float; max_torque: float
// physics/ccd/conservative_advancement.py, physics/query — ToiResult and
// RaycastHit are ephemeral query-result value types, never persisted, safe to
// share across threads read-only (no mutation).
class ToiResult:
    hit: bool
    t: float                                                  // fraction of dt, [0,1]
    normal: Vec3

class RaycastHit:
    hit: bool
    body: BodyId
    point: Vec3; normal: Vec3; fraction: float

// physics/stats.py — StepStats is per-step telemetry, append-only history kept
// by the host, not by World; determinism-neutral (read-only reporting).
class StepStats:
    contacts: int; velocity_iterations_used: int
    max_penetration: float; total_kinetic_energy: float
    awake_islands: int; nan_quarantine_count: int

// physics/errors.py — error types; raising one never leaves World partially
// mutated mid-step (steps are atomic: fully applied or fully rejected at I10).
class PhysicsError(Exception): ...
class NaNQuarantineError(PhysicsError): ...
class DegenerateShapeError(PhysicsError): ...
