Core schema & types (§11)

// resource-model
struct ResourceVector { cpu_mc:u64, mem_b:u64, disk_b:u64, net_bps:u64, gpu:u32, ports:u16 }
   // own: Copy value type; mut: none (immutable); conc: Send+Sync; ser: protobuf; compat: additive dims only
struct NodeCapacity { total:ResourceVector, allocated:ResourceVector, reserved:ResourceVector, available:ResourceVector }
   // invariant I10 checked on every mutation
enum ResourceDim { Cpu, Mem, Disk, Net, Gpu, Ports }   // ser: proto enum, never renumbered (P14)

// metadata-store
struct LogIndex(u64)                                    // monotonic (I3)
struct Epoch(u64)                                       // monotonic cluster generation (I2)
enum WalEntry { AdmitWorkload, EnqueueWorkload, CommitLease, ReleaseLease,
                FenceNode, PreemptWorkload, RegisterNode, ExpireOffer, TimeTick }
   // dur: appended to Raft WAL; conc: single-writer apply; ser: length-prefixed protobuf
struct Snapshot { at:LogIndex, state_blob:Bytes, checksum:Blake3 }   // I12
trait StateMachine { fn apply(&mut self, idx:LogIndex, e:&WalEntry); fn snapshot(&self)->Snapshot; }

// cluster state
struct NodeId(Uuid); struct WorkloadId(Uuid); struct LeaseId(Uuid); struct OfferId(Uuid)
struct TenantId(String); struct QueueId(String); struct Attempt(u32)
enum NodeState { Registering, Running, Suspect, Lost, Draining }   // I5 transitions
struct ClusterNode { id:NodeId, rack:String, zone:String, labels:Map<String,String>,
                     cap:NodeCapacity, last_hb:Hlc, epoch:Epoch, state:NodeState }
   // own: manager-authoritative; agent reports validated before apply
// leases & workloads
struct AllocationLease { id:LeaseId, workload:WorkloadId, node:NodeId, attempt:Attempt,
                         resources:ResourceVector, issued_epoch:Epoch, expires:Hlc }
   // own: lease-engine; mut: immutable post-commit; dur: WAL (I7); invariant I1
struct WorkloadDescriptor { id:WorkloadId, tenant:TenantId, queue:QueueId, priority:i16,
                            request:ResourceVector, gang:Option<GangId>, constraints:Vec<Constraint>,
                            runtime:RuntimeSpec, retry:RetryPolicy, state:WorkloadState }
enum WorkloadState { Submitted, Admitted, Queued, Placed, Launching, Running,
                     Completed, Failed, Lost, Preempted, Rejected, Terminating }   // §8.2
struct Constraint { kind:ConstraintKind, key:String, op:MatchOp, val:String, hard:bool }

// queues & quota
struct QueueDefinition { id:QueueId, parent:Option<QueueId>, guaranteed:ResourceVector,
                         max:ResourceVector, weight:u32, borrow:BorrowPolicy }
struct DominantShare(f64)                              // DRF metric (I8)
struct QuotaLedger { tenant:TenantId, used:ResourceVector, borrowed:ResourceVector }   // I4

// offers & scheduling
struct ResourceOffer { id:OfferId, node:NodeId, vector:ResourceVector, epoch:Epoch, ttl:Hlc }  // P8/I6
struct PlacementScore(f64)                             // Tetris alignment + soft constraints
struct GangId(Uuid)                                     // I9 atomic set key

// time & errors
struct Hlc { wall_ms:u64, logical:u32 }                // hybrid logical clock (ADR-20)
enum AllocError { OverQuota, Infeasible, StaleEpoch, OfferExpired, GangPartial,
                  NotLeader, WalUnavailable, Fenced }   // ser: proto; compat: additive variants
