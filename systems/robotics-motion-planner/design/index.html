Core schema & types (§11)

// mp-space
pub enum State {
    R2 { x: f64, y: f64 },
    Se2 { x: f64, y: f64, theta: f64 },     // theta in [-PI, PI)
    Se3 { pos: [f64; 3], rot: UnitQuaternion },
    Joint(SmallVec<[f64; 8]>),               // n joint values
}

pub trait StateSpace {
    type S;                                   // associated state type
    fn dim(&self) -> usize;
    fn sample(&self, rng: &mut ChaChaRng) -> Self::S;          // uniform in bounds
    fn interpolate(&self, a: &Self::S, b: &Self::S, t: f64) -> Self::S; // t in [0,1]
    fn distance(&self, a: &Self::S, b: &Self::S) -> f64;       // metric (#7.9)
    fn validate(&self, s: &Self::S) -> bool;                   // within bounds
    fn enforce_bounds(&self, s: &mut Self::S);                 // clamp/wrap
}
pub trait Metric { fn distance(&self, a: &State, b: &State) -> f64; }   // may differ from space default
pub trait Sampler {
    fn sample(&mut self, rng: &mut ChaChaRng) -> State;
    fn sample_near(&mut self, rng: &mut ChaChaRng, center: &State, std: f64) -> State;
}
// Implementors: Uniform, Gaussian, Bridge, Halton, InformedEllipsoid { c_best, c_min, focii }
pub struct Motion { pub from: State, pub to: State, pub cost: f64, pub kind: SegmentKind }
pub enum SegmentKind { Straight, Dubins(DubinsWord), ReedsShepp(RsWord), Lattice(u16) }

pub struct Tree { nodes: Vec<TreeNode>, kd: KdTree }
pub struct TreeNode { state: State, parent: Option<u32>, cost: f64 }   // cumulative cost from root

pub struct Blueprint { nodes: Vec<State>, adj: Vec<Vec<(u32, f64)>>, kd: KdTree } // PRM graph
// mp-collision
pub trait CollisionChecker {
    fn in_collision(&self, robot: &RobotModel, q: &State) -> bool;        // config check
    fn motion_valid(&self, robot: &RobotModel, a: &State, b: &State, eps: f64) -> bool; // edge check
    fn clearance(&self, robot: &RobotModel, q: &State) -> f64;            // distance to nearest obstacle
}

// mp-env
pub trait Environment {
    fn occupied(&self, p: &[f64]) -> bool;
    fn cost(&self, p: &[f64]) -> u8;          // 0..=254, 255 lethal
}
pub struct OccupancyGrid { w: u32, h: u32, res: f64, origin: [f64;2], cells: Vec<u8> }
pub struct Costmap { base: OccupancyGrid, layers: Vec<CostLayer>, inflation_radius: f64 }
pub struct Esdf { dims: [u32;3], res: f64, dist: Vec<f32>, grad: Vec<[f32;3]> }
// mp-steer
pub trait SteeringFn {
    fn steer(&self, from: &State, to: &State) -> Option<Path>;            // exact connection
    fn steer_bounded(&self, from: &State, to: &State, max_len: f64) -> Path; // RRT step
    fn distance(&self, from: &State, to: &State) -> f64;                  // path length (pseudo-metric)
}
// Implementors: StraightLine, Dubins { radius }, ReedsShepp { radius }, StateLattice { primitives }
// mp-planner
pub trait Planner {
    fn solve(&mut self, req: &PlanRequest, deadline: Deadline) -> PlanResult;
    fn stats(&self) -> PlannerStats;
}

pub struct PlanRequest {
    pub robot: Arc<RobotModel>,
    pub env: Arc<dyn Environment>,
    pub space: Arc<dyn StateSpace<S = State>>,
    pub checker: Arc<dyn CollisionChecker>,
    pub start: State,
    pub goal: GoalRegion,           // point or region predicate + tolerance
    pub cfg: PlannerCfg,            // planner kind, caps, sampler, optimize flag
    pub seed: u64,                  // determinism (#7.5)
}

pub enum PlanResult {
    Solved { path: Path, cost: f64, stats: PlannerStats },
    NoPath { stats: PlannerStats },
    Error(PlanError),
}

pub struct Path { pub states: Vec<State>, pub segments: Vec<SegmentKind> }

pub struct Trajectory { pub points: Vec<TrajPoint>, pub duration: f64 }
pub struct TrajPoint { pub t: f64, pub state: State, pub vel: SmallVec<[f64;8]>, pub acc: SmallVec<[f64;8]> }
// mp-time
pub trait TimeParam {
    fn parameterize(&self, path: &Path, limits: &DynamicLimits) -> Trajectory;
}
pub struct DynamicLimits { pub v_max: Vec<f64>, pub a_max: Vec<f64>, pub j_max: Option<Vec<f64>> }
// Implementors: TrapezoidalProfile, SCurveProfile, ToppLite

// mp-local
pub trait LocalPlanner {
    fn compute_cmd(&mut self, state: &State, traj: &Trajectory, obstacles: &dyn Environment,
                   budget: Duration) -> LocalCmd;       // <= 20 ms (#7.7)
    fn needs_replan(&self) -> Option<ReplanReason>;
}
pub struct LocalCmd { pub v: f64, pub omega: f64, pub valid: bool }   // or joint velocities
pub enum ReplanReason { TrajectoryBlocked, TrajectoryStale, GoalMoved }
#[derive(thiserror::Error)]
pub enum PlanError {
    StartInCollision, GoalInCollision, StartOutOfBounds, GoalOutOfBounds,
    InvalidSpace(String), ResourceExhausted { what: String, cap: u64 },
    Timeout { elapsed_ms: u64 }, InvalidRobotModel(String), InvalidMap(String),
    NumericalDegeneracy(String),
}
