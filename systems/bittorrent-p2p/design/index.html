Core schema & types (§11)

/// 20-byte SHA-1 (v1) info-hash. Copy, Eq, Hash — used as a map key everywhere.
pub struct InfoHashV1(pub [u8; 20]);
/// 32-byte SHA-256 (v2, BEP 52) info-hash; truncated to 20 bytes for handshake/tracker use.
pub struct InfoHashV2(pub [u8; 32]);
/// 20-byte peer id (BEP 20 convention "-SW0010-" + random suffix).
pub struct PeerId(pub [u8; 20]);
/// 20-byte Mainline DHT node id (Kademlia ID space).
pub struct NodeId(pub [u8; 20]);

impl NodeId {
    fn distance(&self, other: &NodeId) -> [u8; 20];   // XOR metric, Kademlia paper, section 2.1
}
pub trait Transport: AsyncRead + AsyncWrite + Send + Unpin {
    fn peer_addr(&self) -> SocketAddr;
    fn local_addr(&self) -> SocketAddr;
    fn shutdown(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<io::Result<()>>;
}
pub enum Value {
    Int(i64),
    Bytes(Vec<u8>),
    List(Vec<Value>),
    Dict(BTreeMap<Vec<u8>, Value>),   // BTreeMap enforces canonical sort order on encode
}
pub struct Metainfo {
    pub info_bytes: bytes::Bytes,               // captured verbatim; never re-bencoded
    pub infohash_v1: Option<InfoHashV1>,
    pub infohash_v2: Option<InfoHashV2>,
    pub announce: Option<String>,
    pub announce_list: Vec<Vec<String>>,        // BEP 12 tiers
    pub url_list: Vec<String>,                  // BEP 19 web seeds
    pub info: InfoDict,
}

pub enum InfoDict {
    SingleV1 { name: String, piece_length: u32, pieces: Vec<[u8; 20]>, length: u64, private: bool },
    MultiV1  { name: String, piece_length: u32, pieces: Vec<[u8; 20]>, files: Vec<FileEntry>, private: bool },
    V2       { name: String, piece_length: u32, meta_version: u32, file_tree: FileTreeNode, private: bool },
}

pub struct FileEntry { pub path: PathBuf, pub length: u64, pub attr: Option<String> }

pub enum FileTreeNode {
    File { length: u64, pieces_root: Option<[u8; 32]> },   // SHA-256 merkle root, BEP 52
    Directory(BTreeMap<String, FileTreeNode>),
}
pub enum Message {
    KeepAlive, Choke, Unchoke, Interested, NotInterested,
    Have(u32), Bitfield(Bytes),
    Request { piece: u32, offset: u32, length: u32 },
    Piece   { piece: u32, offset: u32, block: Bytes },
    Cancel  { piece: u32, offset: u32, length: u32 },
    Port(u16),
    Suggest(u32), HaveAll, HaveNone,
    Reject { piece: u32, offset: u32, length: u32 },
    AllowedFast(u32),
    Extended { id: u8, payload: Bytes },
    HashRequest { pieces_root: [u8; 32], base_layer: u32, index: u32, length: u32, proof_layers: u32 },
    Hashes      { pieces_root: [u8; 32], base_layer: u32, index: u32, length: u32, proof_layers: u32, hashes: Bytes },
    HashReject  { pieces_root: [u8; 32], base_layer: u32, index: u32, length: u32, proof_layers: u32 },
}
pub struct PeerState {
    pub am_choking: bool, pub am_interested: bool,
    pub peer_choking: bool, pub peer_interested: bool,
    pub last_message_received: Instant, pub last_message_sent: Instant,
    pub bytes_received: u64, pub bytes_sent: u64,
}
pub enum PieceStatus { Missing, Partial { in_flight_blocks: u16, total_blocks: u16 }, Complete, Quarantined { failures: u8 } }

pub struct PickerSnapshot {
    pub piece_count: u32, pub block_size: u32,
    pub status: Vec<PieceStatus>, pub availability: Vec<u16>,
    pub priority: Vec<Priority>, pub endgame: bool,
}

pub enum Priority { Skip = 0, Low = 1, Normal = 2, High = 3 }
pub struct AnnounceRequest {
    pub infohash: InfoHashV1, pub peer_id: PeerId, pub port: u16,
    pub uploaded: u64, pub downloaded: u64, pub left: u64,
    pub event: Event, pub numwant: u32, pub key: u32, pub compact: bool,
}
pub enum Event { Started, Completed, Stopped, None }
pub struct AnnounceResponse {
    pub interval: u32, pub min_interval: Option<u32>, pub complete: u32,
    pub incomplete: u32, pub peers: Vec<SocketAddr>, pub failure: Option<String>,
}
pub enum Query {
    Ping { id: NodeId },
    FindNode { id: NodeId, target: NodeId },
    GetPeers { id: NodeId, infohash: InfoHashV1 },
    AnnouncePeer { id: NodeId, infohash: InfoHashV1, port: u16, token: Vec<u8>, implied_port: bool },
}
pub enum Response {
    PingPong { id: NodeId },
    Nodes { id: NodeId, nodes: Vec<NodeContact> },
    PeersExact { id: NodeId, token: Vec<u8>, peers: Vec<SocketAddr> },
    PeersClose { id: NodeId, token: Vec<u8>, nodes: Vec<NodeContact> },
}
pub struct NodeContact { pub id: NodeId, pub addr: SocketAddr }
pub trait Storage: Send + Sync {
    async fn read(&self, piece: u32, offset: u32, length: u32) -> Result<Bytes, Error>;
    async fn write(&self, piece: u32, offset: u32, data: Bytes) -> Result<(), Error>;
    async fn fsync(&self, piece: u32) -> Result<(), Error>;
    async fn verify_resume(&self) -> Result<Bitfield, Error>;
    fn piece_count(&self) -> u32;
}
pub trait Picker: Send + Sync {
    fn next_request(&mut self, peer_id: PeerId, peer_bitfield: &Bitfield) -> Option<BlockRequest>;
    fn record_have(&mut self, peer_id: PeerId, piece: u32);
    fn record_completed_piece(&mut self, piece: u32);
    fn record_failed_piece(&mut self, piece: u32);
    fn snapshot(&self) -> PickerSnapshot;
}
