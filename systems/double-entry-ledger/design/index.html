Core schema & types (§11)

#[repr(C, packed)] struct Account {
  id:               u128,  // Client-assigned; unique per cluster
  debits_pending:   u128,  // Reserved by pending transfers (debit side)
  debits_posted:    u128,  // Settled debits
  credits_pending:  u128,  // Reserved by pending transfers (credit side)
  credits_posted:   u128,  // Settled credits
  user_data:        u128,  // Opaque; application metadata
  reserved:         u32,   // Must be zero
  ledger:           u32,   // Ledger namespace; trial balance per ledger
  code:             u16,   // Application-defined account type
  flags:            u16,   // AccountFlags bitmask
  timestamp:        u64,   // Assigned by primary (ns since epoch)
}  // 128 bytes
#[repr(C, packed)] struct Transfer {
  id:                   u128,
  debit_account_id:     u128,
  credit_account_id:    u128,
  amount:               u128,
  pending_id:           u128,  // 0 if not resolving a pending transfer
  user_data:            u128,
  timeout:              u32,   // Seconds until auto-void; 0 = no timeout
  ledger:               u32,
  code:                 u16,
  flags:                u16,   // TransferFlags bitmask
  timestamp:            u64,
}  // 128 bytes
enum CreateAccountError { u32 }
  variants: None=0, LinkedEventFailed, LinkedEventChainOpen,
            ReservedFlag, ReservedField, IdMustNotBeZero,
            LedgerMustNotBeZero, DebitAccountNotFound,
            Exists, ExistsWithDifferentFlags, ...

enum CreateTransferError { u32 }
  variants: None=0, LinkedEventFailed, LinkedEventChainOpen,
            ReservedFlag, ReservedField, IdMustNotBeZero,
            DebitAccountNotFound, CreditAccountNotFound,
            AccountsMustHaveSameLedger, AmountMustNotBeZero,
            PendingIdMustNotBeZero, PendingTransferNotFound,
            PendingTransferAlreadyPosted, PendingTransferAlreadyVoided,
            ExceedsCredits, ExceedsDebits, Overflow, Exists, ...
struct StateMachine {
  account_store:   AccountStore,   // HashMap<u128, Account> + LSM flush handle
  transfer_store:  TransferStore,  // HashMap<u128, Transfer> + LSM flush handle
  pending_index:   PendingIndex,   // HashMap<u128, Transfer> (pending only)
  dedup_index:     DedupIndex,     // HashMap<u128, EventResult>
  cluster_time:    u64,            // Monotonically advancing ns; set per batch
  commit_timestamp: u64,           // LSN of last committed prepare
}

fn execute_batch(batch: &[Event], timestamp: u64) -> Vec<EventResult>;
  // Processes entire batch atomically within the single-threaded loop
  // Applies TimeoutSweep before batch; LinkedChainEval per chain; BalanceChecker per event

fn digest(&self) -> [u8; 32];
  // SHA256 of sorted (account_id → account_bytes) + (transfer_id → transfer_bytes)
  // Used by VSR to detect state divergence across replicas
struct AppendLog {
  file:       File,        // O_DIRECT opened WAL file
  offset:     u64,         // Current append offset (always sector-aligned)
  lsn:        u64,         // Next LSN to assign
  sector_buf: [u8; 512],   // Sector-aligned write buffer (static)
}

fn append(payload: &[u8]) -> WalLsn;   // Write sector(s), return LSN
fn fsync(&mut self) -> io::Result<()>;  // Called after each PrepareMessage
fn recover(from_lsn: u64) -> impl Iterator<Item=(WalLsn, Vec<u8>)>;
struct Forest {
  accounts:  Tree<u128, Account>,
  transfers: Tree<u128, Transfer>,
  pending:   Tree<u128, Transfer>,
  dedup:     Tree<u128, EventResult>,
  cache:     BlockCache,   // Shared LRU; static 4 GB
}

fn get<K,V>(tree: TreeId, key: K) -> Option<V>;
fn put<K,V>(tree: TreeId, key: K, value: V);
fn flush(tree: TreeId) -> io::Result<()>;  // Write memtable to L0 SSTable
fn compact(tree: TreeId);                  // Trigger level compaction
enum ReplicaRole { Primary, Backup, Recovering }

struct Replica {
  index:          usize,        // 0..cluster_size
  view:           u64,          // Current view number
  op_number:      u64,          // Highest prepare op seen
  commit_number:  u64,          // Highest committed op
  role:           ReplicaRole,
  state_machine:  StateMachine,
  wal:            AppendLog,
  prepare_log:    PrepareLog,   // In-memory: op_number → PrepareMessage
  quorum:         QuorumTracker,
}

fn on_message(msg: VsrMessage);            // Dispatch to role handlers
fn on_client_request(req: ClientRequest);  // Primary only: sequence + prepare
fn apply_prepare(p: PrepareMessage);       // Both: apply to state machine after commit
struct Session {
  client_id:      u128,
  ack_number:     u64,     // Last acknowledged request_id
  window:         SlidingWindow,
}

struct SessionRegistry {
  sessions: HashMap<u128, Session>,  // client_id → Session
  max_sessions: usize,               // Static cap; reject beyond limit
}

fn register(client_id: u128) -> Result<(), SessionError>;
fn route_request(frame: Frame) -> Option<(Session, Vec<Event>)>;
fn route_response(client_id: u128, results: Vec<EventResult>) -> Frame;
struct Simulator {
  rng:          SeededPrng,     // xoshiro256** seeded from CLI or random
  replica_count: usize,
  history_len:   usize,
  fault_rate:    f32,
  model:         ReferenceModel, // Pure-function reference for comparison
}

fn run(iterations: u64) -> SimReport;
  // Per iteration: generate random history; inject faults; run state machine;
  //   check invariants against model; record divergence if found
struct WalLsn(u64);      // WAL log sequence number; monotonic; per-replica
struct OpNumber(u64);    // VSR op number; monotonic; primary-assigned
struct ViewNumber(u64);  // VSR view number; increments on view-change
