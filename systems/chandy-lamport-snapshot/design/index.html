Core schema & types (§11)

// ids.rs
#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
pub struct ProcessId(pub u32);

#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub struct ChannelId { pub from: ProcessId, pub to: ProcessId }

#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub struct SnapshotId { pub initiator: ProcessId, pub seq: u64 }
// message.rs
#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub struct AppPayload(pub Vec<u8>);   // opaque; app-defined encoding

#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub enum Message {
    App(AppPayload),
    Marker(SnapshotId),
}
// event.rs
#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub enum Event {
    Send { channel: ChannelId, msg: Message, clock: VectorClock },
    Receive { channel: ChannelId, msg: Message, clock: VectorClock },
    Internal { process: ProcessId, clock: VectorClock },
}

#[derive(Debug, Clone, PartialEq, Eq, Default, Serialize, Deserialize)]
pub struct VectorClock(pub BTreeMap<ProcessId, u64>);

impl VectorClock {
    pub fn increment(&mut self, p: ProcessId);
    pub fn merge(&self, other: &VectorClock) -> VectorClock;
    pub fn happens_before(&self, other: &VectorClock) -> bool;
    pub fn concurrent(&self, other: &VectorClock) -> bool;
}
// state.rs
#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub enum RecordStatus {
    NotStarted,
    Recording { msgs: Vec<AppPayload> },
    Done     { msgs: Vec<AppPayload> },
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SnapshotState {
    pub local_state_at_first_marker: AppState,
    pub channel_msgs: BTreeMap<ChannelId, Vec<AppPayload>>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ProcessState {
    pub local: AppState,
    pub recording: BTreeMap<(SnapshotId, ChannelId), RecordStatus>,
    pub recorded: BTreeMap<SnapshotId, SnapshotState>,
}
// snapshot.rs
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct GlobalSnapshot {
    pub snapshot_id:     SnapshotId,
    pub process_states:  BTreeMap<ProcessId, AppState>,
    pub channel_states:  BTreeMap<ChannelId, Vec<AppPayload>>,
    pub completed_at:    Option<VectorClock>,   // logical time of last process completing
}

// cut.rs
#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub struct Cut(pub BTreeMap<ProcessId, u64>);   // per-process event-count frontier

impl Cut {
    pub fn is_consistent(&self, history: &EventHistory) -> bool;
    // contract: for every Receive(c,m,vc) in cut, the matching Send(c,m,vc) is in cut or m is in channel_states
}
// stable_property (in cl-detect, re-exported via cl-core)
#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub enum StableProperty {
    Terminated,
    Deadlocked,
    Conserved { quantity: String, expected: i64, actual: i64 },
}

// error.rs
#[derive(Debug, thiserror::Error)]
pub enum CoreError {
    #[error("FIFO violation on channel {channel:?}: got seq {got}, expected {expected}")]
    FifoViolation { channel: ChannelId, got: usize, expected: usize },
    #[error("Duplicate snapshot record for ({snapshot_id:?}, {process:?})")]
    DuplicateRecord { snapshot_id: SnapshotId, process: ProcessId },
    #[error("Inconsistent cut: receive {msg_id} has no matching send in cut")]
    InconsistentCut { msg_id: String },
    #[error("Snapshot incomplete: missing process {process:?}")]
    SnapshotIncomplete { process: ProcessId },
    #[error(transparent)]
    Other(#[from] anyhow::Error),
}
// channel.rs
pub struct FifoChannel {
    pub id:    ChannelId,
    queue:     VecDeque<Message>,    // internal; in send order
    seq_send:  usize,
    seq_recv:  usize,
}

impl FifoChannel {
    pub fn new(id: ChannelId) -> Self;
    pub fn enqueue(&mut self, msg: Message) -> usize;            // returns send seq
    pub fn dequeue(&mut self) -> Option<(usize, Message)>;       // (recv_seq, msg)
    pub fn peek(&self) -> Option<&Message>;
    pub fn in_flight(&self) -> &[Message];                       // snapshot: messages not yet delivered
    pub fn is_empty(&self) -> bool;
}
// app_trait.rs
pub trait AppHandler: Send {
    type State: Clone + Serialize + for<'de> Deserialize<'de> + Send;
    fn handle_message(&mut self, state: &mut Self::State, from: ProcessId, payload: AppPayload)
        -> Vec<(ChannelId, AppPayload)>;   // returns outbound messages
    fn handle_internal(&mut self, state: &mut Self::State)
        -> Vec<(ChannelId, AppPayload)>;
    fn capture_state(&self, state: &Self::State) -> AppState;   // for snapshot recording
}

// process.rs
pub struct Process<H: AppHandler> {
    pub id:       ProcessId,
    pub handler:  H,
    pub state:    H::State,
    outgoing:     BTreeMap<ChannelId, FifoChannel>,
    incoming:     BTreeMap<ChannelId, VecDeque<Message>>,
    pub proc_state: ProcessState,
    pub clock:    VectorClock,
}

impl<H: AppHandler> Process<H> {
    pub fn new(id: ProcessId, handler: H, initial: H::State) -> Self;
    pub fn send(&mut self, channel: ChannelId, payload: AppPayload) -> Result<(), CoreError>;
    pub fn on_receive(&mut self, channel: ChannelId, msg: Message) -> Result<(), CoreError>;
    pub fn on_internal(&mut self) -> Result<(), CoreError>;
    pub fn snapshot_state(&self) -> &ProcessState;
}
// scheduler.rs
pub struct DeliverySchedule {
    pub seed:  u64,
    steps:     Vec<(ChannelId, usize)>,   // ordered (channel, queue-index) steps
}

impl DeliverySchedule {
    pub fn generate(seed: u64, channels: &[ChannelId], n_steps: usize) -> Self;
    pub fn next_delivery(&mut self) -> Option<(ChannelId, usize)>;
}

// history.rs
pub struct EventHistory {
    events: Vec<(ProcessId, Event)>,
}

impl EventHistory {
    pub fn append(&mut self, process: ProcessId, event: Event);
    pub fn events_for(&self, process: ProcessId) -> impl Iterator<Item = &Event>;
    pub fn all_events(&self) -> &[(ProcessId, Event)];
}
// recorder.rs
pub struct Recorder {
    pub snapshot_id: SnapshotId,
    process_snapshots: BTreeMap<ProcessId, SnapshotState>,
}

impl Recorder {
    pub fn record_process(&mut self, p: ProcessId, state: SnapshotState) -> Result<(), CoreError>;
    pub fn is_complete(&self, all_processes: &[ProcessId]) -> bool;
    pub fn assemble(&self, all_processes: &[ProcessId]) -> Result<GlobalSnapshot, CoreError>;
}

// completion.rs
pub struct CompletionDetector {
    snapshot_id:  SnapshotId,
    pending:      BTreeSet<ProcessId>,
}

impl CompletionDetector {
    pub fn new(snapshot_id: SnapshotId, processes: Vec<ProcessId>) -> Self;
    pub fn mark_done(&mut self, p: ProcessId);
    pub fn is_complete(&self) -> bool;
}
// verifier.rs
pub struct Verifier<'a> {
    snapshot: &'a GlobalSnapshot,
    history:  &'a EventHistory,
}

impl<'a> Verifier<'a> {
    pub fn new(snapshot: &'a GlobalSnapshot, history: &'a EventHistory) -> Self;
    pub fn verify(&self) -> Result<Cut, CoreError>;
    // contract: returns Cut iff snapshot is a consistent cut (I1); Err(InconsistentCut{..}) otherwise
}
