Core schema & types (§11)

struct Request {
    id:            RequestId,           // immutable; assigned at ingress
    model:         ModelId,             // required; routing key
    adapter:       Option<AdapterId>,   // LoRA adapter; None = base model
    prompt_tokens: u32,                 // parsed token count; 0 if unknown pre-tokenize
    max_tokens:    u32,                 // upper bound; used for capacity estimation
    prefix_hash:   PrefixHash,          // SHA-256 of prompt token prefix; routing key
    slo:           Slo,                 // TTFT + TPOT + wall-clock deadline
    priority:      Priority,            // P0..P3; drives queue ordering
    idempotent:    bool,                // gates hedging; caller declares
    stream:        bool,                // SSE/chunked relay vs. batch response
}
struct Slo {
    ttft_ms:     u32,          // max time-to-first-token milliseconds; 0 = unconstrained
    tpot_ms:     u32,          // max time-per-output-token milliseconds; 0 = unconstrained
    deadline:    Option<Instant>, // absolute wall-clock deadline; None = no hard deadline
}
enum Priority { P0 = 0, P1 = 1, P2 = 2, P3 = 3 }
struct LoadState {
    queue_depth:   u32,     // backend-reported pending request count
    kv_util:       f32,     // KV cache utilization [0.0, 1.0]
    running_batch: u32,     // current continuous-batch size
    ewma_ttft_ms:  f32,     // EWMA of recent TTFT observations (α=0.1)
    inflight:      u32,     // requests currently being processed (router-tracked)
}
enum Health { Healthy, Degraded, Ejected }
struct Replica {
    id:       BackendId,
    model:    ModelId,
    adapters: HashSet<AdapterId>,    // LoRA adapters loaded; empty = base-model only
    load:     Arc<ArcSwap<LoadState>>,
    health:   Arc<AtomicU8>,          // Health as u8; see 11.6
    addr:     SocketAddr,
}
struct RoutingDecision {
    backend: BackendId,
    reason:  RouteReason,
}

enum RouteReason {
    LeastLoad,
    PrefixHit { match_len: u32 },   // tokens matched in PrefixTree
    AdapterResident,
    Affinity { session_key: u64 },
    Fallback,                        // no preferred backend; random among healthy
}
struct Lease {
    request:  Arc<Request>,
    backend:  BackendId,
    deadline: Instant,              // copied from request.slo.deadline or now + max_ttl
}
struct PrefixTree {
    // radix trie: PrefixHash prefix → BTreeSet<(Instant, BackendId)>
    // recency tracked per (backend, prefix) entry
    // eviction: LRU by last-access Instant when node_count > capacity
}

trait PrefixIndex: Send + Sync {
    fn insert(&self, hash: PrefixHash, backend: BackendId);
    fn longest_match(&self, hash: PrefixHash) -> Option<(BackendId, u32)>;
    fn evict_backend(&self, backend: BackendId);
    fn node_count(&self) -> usize;
}
struct AdapterInventory {
    // adapter_id → Vec<BackendId> index
    // maintained in sync with Replica.adapters
}

trait AdapterIndex: Send + Sync {
    fn backends_for(&self, adapter: &AdapterId) -> SmallVec<[BackendId; 4]>;
    fn register(&self, backend: BackendId, adapters: &HashSet<AdapterId>);
    fn deregister(&self, backend: BackendId);
}
enum InferError {
    BackendUnavailable { backend: BackendId, reason: String },
    AdapterNotFound    { adapter: AdapterId },
    ModelNotFound      { model: ModelId },
    QueueFull          { model: ModelId, queue_len: u32 },
    SloDeadlineExpired { request: RequestId },
    UpstreamTimeout    { phase: TimeoutPhase, elapsed_ms: u64 },
    UpstreamError      { status: u16, body: Bytes },
    CircuitOpen        { backend: BackendId },
    HedgeCancelled     { winner: BackendId },
    ParseError         { detail: String },
    Internal           { detail: String },
}

enum TimeoutPhase { Connect, FirstToken, TokenInterval }
struct CapacityDescriptor {
    backend:       BackendId,
    max_batch:     u32,          // backend-declared max continuous-batch size
    kv_capacity:   u64,          // total KV cache slots (tokens)
    gpu_mem_bytes: u64,          // reported GPU memory (informational)
    adapter_slots: u32,          // max simultaneous loaded adapters
}
struct RouterConfig {
    listen:            ListenConfig,      // bind addr, TLS cert/key paths, H2 settings
    backends:          Vec<BackendSpec>,  // static backend list; overridden by dynamic registry
    routing:           RoutingConfig,     // policy name, prefix_cache_capacity, affinity TTL
    queue:             QueueConfig,       // per-model capacity, shedding policy
    dispatch:          DispatchConfig,    // pool size, timeouts, retry/hedge params
    sched:             SchedConfig,       // fairness quotas, autoscale thresholds
    telemetry:         TelemetryConfig,   // prometheus addr, OTLP endpoint, log level
    health:            HealthConfig,      // probe interval, scrape interval, EWMA α
}
struct SloMetrics {
    ttft_hist:        Histogram,   // TTFT observation buckets (ms)
    tpot_hist:        Histogram,   // TPOT observation buckets (ms)
    queue_depth:      Gauge,       // per-model queue depth
    cache_hit_rate:   Counter,     // prefix cache hits
    cache_miss_rate:  Counter,     // prefix cache misses
    slo_breach_total: Counter,     // requests that missed TTFT or TPOT SLO
    shed_total:       Counter,     // requests shed at admission
}
