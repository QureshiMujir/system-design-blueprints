Core schema & types (§11)

// crates/scheduler-core/src/task/mod.rs  — §11.1
#[derive(Copy, Clone, Eq, PartialEq, Hash, Debug)]
pub struct TaskId(pub u64);

#[derive(Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Debug)]
pub struct Priority(pub i32);
impl Priority { pub const IDLE: Self; pub const DEFAULT: Self; pub const URGENT: Self; }

#[derive(Clone, Debug)]
pub enum Affinity {
    Any { last_cpu: Option<CpuId> },   // any CPU; last-CPU hint honoured if Some
    Pinned { cpus: BitSet },           // only this set; spawn fails if none online
    Preferred { cpus: BitSet },        // prefer set; migrate only on overload
}

#[derive(Clone, Debug)]
pub enum Policy {
    RoundRobin { slice: Duration },
    Mlfq { classes: u8 },                       // class 0 highest
    Cfs { nice: i32 },                          // weight derived from nice
    Edf { deadline: Instant, computation: Duration, period: Option<Duration> },  // RT pool only
    Rms { period: Duration, computation: Duration },                             // RT pool only
}

#[derive(Clone, Debug)]
pub struct SpawnOpts {
    pub policy: Policy,
    pub affinity: Affinity,
    pub stack_size: usize,
    pub name: Option<String>,
}
// crates/scheduler-core/src/task/header.rs  — §11.2
/// One cache line; 64 B on x86_64 / aarch64.
#[repr(C, align(64))]
pub struct TaskHeader {
    pub state: AtomicU8,                  // Spawning→Ready→Running→Blocked→…→Done (I2)
    pub preempt_requested: AtomicBool,    // observed at safepoints (P4)
    pub last_cpu: AtomicU32,              // for cache-locality re-enqueue (Phase 9)
    pub vruntime_ns: AtomicU64,           // CFS only; integer ns (P12)
    pub last_transition: AtomicU64,       // profiler input (P7)
    pub refcount: AtomicUsize,            // runqueue + join handle + pending I/O
    pub class: AtomicU8,                  // RT or BE (I4)
    pub body: *const TaskBody,            // cold fields, off the hot line
}

pub struct TaskBody {
    pub id: TaskId,
    pub name: Option<String>,
    pub policy: Policy,
    pub affinity: Affinity,
    pub stack: Stack,
    pub join_waker: AtomicWaker,
    pub profile: ProfileSlot,
}
// crates/scheduler-core/src/runqueue/mod.rs  — §11.3
/// Implemented by FifoQ, PrioQ, MlfqQ, CfsTree, EdfHeap, RmsHeap, ChaseLevDeque.
pub trait Runqueue: Send + 'static {
    fn push(&self, task: TaskRef);              // total; never blocks, grows on full
    fn pop(&self) -> Option<TaskRef>;           // None iff empty; never spins
    fn len(&self) -> usize;                     // approximate; snapshot read
    fn steal(&self, limit: usize) -> Vec<TaskRef>;  // up to `limit` from the back
    fn snapshot(&self) -> RunqueueSnapshot;     // textual dump for `schedctl inspect`
}
// crates/scheduler-core/src/worker.rs  — §11.4
pub struct Worker {
    pub cpu: CpuId,
    pub local: Box<dyn Runqueue>,
    pub steal_targets: Vec<Arc<dyn Runqueue>>,   // victim order set by topology
    pub parker: Parker,
    pub profile: WorkerProfile,
    pub current: Option<TaskRef>,                // ADR-19: not a thread-local
    pub time_slice: Cell<Duration>,
    pub started_at: Instant,
    pub steal_ctr: AtomicU64,
    pub park_ctr: AtomicU64,
}

// crates/scheduler-core/src/runtime.rs
pub struct Runtime {
    pub workers: Vec<JoinHandle<()>>,
    pub injectors: Arc<InjectorSet>,
    pub topology: Arc<Topology>,
    pub shutdown: Arc<ShutdownToken>,
    pub profile_sink: Arc<dyn ProfileSink>,
}

pub trait RuntimeApi {                                // Compatibility & Versioning: frozen at V1
    fn spawn<F>(&self, f: F) -> JoinHandle<F::Output>
        where F: Future + Send + 'static, F::Output: Send + 'static;
    fn spawn_with<F>(&self, opts: SpawnOpts, f: F) -> Result<JoinHandle<F::Output>, SpawnError>
        where F: Future + Send + 'static, F::Output: Send + 'static;
    fn block_on<F: Future>(&self, f: F) -> F::Output;
    fn shutdown(self, grace: Duration) -> ShutdownReport;
    fn request_stw(&self, deadline: Instant) -> Result<StwToken, StwError>;
    fn resume_from_stw(&self, token: StwToken);
    fn metrics(&self) -> RuntimeMetricsSnapshot;
}
// crates/scheduler-core/src/parker/mod.rs  — §11.5
pub trait Parker: Send + Sync + 'static {
    fn park_until(&self, deadline: Option<Instant>) -> bool;  // true if unparked, false if expired
    fn unpark(&self);                                         // idempotent
}

// crates/scheduler-core/src/profile/mod.rs
#[derive(Clone, Debug)]
pub enum ProfileEvent {
    Spawned    { task: TaskId, at: Instant, on_cpu: CpuId },
    Enqueued   { task: TaskId, at: Instant, on_cpu: CpuId, queue_depth: usize },
    Dispatched { task: TaskId, at: Instant, on_cpu: CpuId, queue_delay: Duration },
    Yielded    { task: TaskId, at: Instant, ran_for: Duration, reason: YieldReason },
    Blocked    { task: TaskId, at: Instant, on: BlockOn },
    Unblocked  { task: TaskId, at: Instant, blocked_for: Duration },
    Migrated   { task: TaskId, at: Instant, from: CpuId, to: CpuId, reason: MigrationReason },
    PreemptReq { task: TaskId, at: Instant, by: CpuId, reason: PreemptReason },
    PreemptAck { task: TaskId, at: Instant, latency: Duration },
    Done       { task: TaskId, at: Instant, total_run: Duration, total_block: Duration },
}
pub enum YieldReason     { SliceExhausted, Cooperative, ParkSelf, GcSafepoint }
pub enum BlockOn         { Io, Lock, Channel, Sleep, JoinHandle }
pub enum MigrationReason { Stolen, Affinity, Overload, RtPreempt }
pub enum PreemptReason   { SliceExpiry, RtPreemption, GcStw, Shutdown }
