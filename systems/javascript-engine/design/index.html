Core schema & types (§11)

struct Engine { /* opaque */ }
impl Engine {
    fn new(config: EngineConfig) -> Engine;                  // owns: bytecode cache, IC stub cache
    fn create_context(&self) -> Context;                      // 1 Engine : N Context (P11)
}

struct Context { /* opaque */ }                               // owns: globals, heap, microtask queue
impl Context {
    fn eval(&mut self, src: Source) -> Result<Value, Error>;  // not Send across threads (P11)
    fn call(&mut self, f: Function, this: Value, args: &[Value]) -> Result<Value, Error>;
    fn run_microtasks(&mut self);                              // I11; host-driven (P10)
    fn define_native_function(&mut self, name: &str, f: NativeFn);
}

#[repr(transparent)]
struct Value(u64);                                             // NaN-boxed (§8.1); Copy, no drop glue
enum ValueTag { Double, Smi, Oddball, Object, String, Symbol, BigInt, Function } // read-only view

struct Object { shape: ShapePtr, slots: [Value; INLINE_SLOTS], overflow: Option<PropertyBag> }
struct Shape { parent: Option<ShapePtr>, key: (Atom, Attrs), slot: u32, version_tag: u64 } // I4: no &mut post-publish

struct FeedbackSlot { state: IcState, entries: SmallVec<[(ShapePtr, u32); 4]> } // §8.3
enum IcState { Uninitialised, Monomorphic, Polymorphic, Megamorphic, Generic } // I5 progress-only

struct Bytecode { ops: Vec<u8>, constants: ConstPool, functions: Vec<FunctionProto>, version: u32 } // I13
struct FunctionProto { param_count: u32, local_count: u32, feedback_vector_len: u32, source_map: SourceMap }
struct Heap { young: SemiSpace, old: OldGen, card_table: CardTable, roots: RootSet } // owns all GC memory
struct SemiSpace { from: Region, to: Region, top: *mut u8 }         // bump-alloc; moved by minor GC (P5)
struct OldGen { pages: Vec<Page>, free_lists: [FreeList; SIZE_CLASSES] }
struct CardTable { bytes: Vec<u8> }                                 // 1 B / 512 B heap (I7, ADR-26)
struct HeapStats { live_bytes: u64, young_capacity: u64, old_capacity: u64, pause_hist: Histogram }

struct DeoptEntry { bytecode_offset: u32, reg_to_slot: Vec<(RegId, SlotId)>, reason: DeoptReason } // I10
enum DeoptReason { ShapeGuardFailed, TypeGuardFailed, CalleeGuardFailed, BoundsGuardFailed }
struct JitStats { tier1_fn_count: u32, tier2_fn_count: u32, deopt_count: u64, megamorphic_count: u64 }

enum Error {
    SyntaxError { message: String, span: Span },
    TypeError { message: String },
    RangeError { message: String },
    ReferenceError { name: Atom },
    InvalidBytecode { offset: u32, reason: &'static str },          // §8.8 malformed-input path
    InternalError { message: String },
}
