Core schema & types (§11)

// NtpTimestamp: u64, 32.32 fixed seconds since 1900-01-01 UTC. + explicit era.
struct NtpTimestamp { bits: u64 }           // upper32=sec, lower32=frac
struct NtpDateTime  { era: i8, ts: NtpTimestamp }   // era-extended (range > 8e9 yr)
impl NtpTimestamp {
    fn from_posix(sec: i64, nanos: u32) -> Self;     // +2_208_988_800 epoch shift
    fn to_posix(self) -> (i64, u32);
    fn sub_signed(self, o: NtpTimestamp) -> NtpDuration;  // era-aware, signed
    fn add(self, d: NtpDuration) -> Self;
    fn frac_secs_f64(self) -> f64;                   // for servo math only
}
struct NtpDuration { ticks: i64 }            // signed 32.32 seconds
struct NtpShort { bits: u32 }                // UQ16.16 (RootDelay/RootDispersion)

// PtpTimestamp: 80-bit on wire (48 sec + 32 ns), TAI epoch (1970-01-01).
struct PtpTimestamp { secs: u64 /*48 used*/, nanos: u32 }   // secs masked to 48b on wire
struct PtpInterval { scaled_ns: i64 }        // ns × 2^16 (correctionField domain)

enum TimeScale { Utc, Tai, Monotonic }
struct LeapState { kind: LeapKind, armed: bool, applied: bool, tai_offset: i32 }
enum LeapKind { None, Insert /*+1*/, Delete /*−1*/, Alarm /*unsync*/ }
struct LeapTable { entries: Vec<(NtpTimestamp /*instant*/, i32 /*tai_offset*/)> }
enum NtpMode { SymActive=1, SymPassive=2, Client=3, Server=4, Broadcast=5, Control=6, Private=7 }
enum NtpLeap { NoWarning=0, AddSec=1, DelSec=2, Alarm=3 }
struct NtpPacket {
    leap: NtpLeap, version: u8, mode: NtpMode,
    stratum: u8, poll: i8, precision: i8,
    root_delay: NtpShort, root_dispersion: NtpShort, reference_id: [u8;4],
    reference_ts: NtpTimestamp,
    origin_ts: NtpTimestamp,   // T1 echo
    receive_ts: NtpTimestamp,  // T2
    transmit_ts: NtpTimestamp, // T3
    ext_fields: Vec<NtpExtField>,
    mac: Option<NtpMac>,
}
struct NtpExtField { field_type: u16, length: u16, value: Bytes }   // RFC 7822, 4-byte padded
enum NtpMac { Symmetric { key_id: u32, digest: Bytes }, Nts(NtsAuthEf) }

// PTP
struct PtpHeader {
    msg_type: PtpMsgType, version: u8, msg_len: u16, domain: u8,
    flags: PtpFlags, correction: PtpInterval,
    source_port: PortIdentity, sequence_id: u16, log_interval: i8,
}
struct PortIdentity { clock_identity: [u8;8], port_number: u16 }
enum PtpMsgType { Sync, DelayReq, PdelayReq, PdelayResp, FollowUp, DelayResp,
                  PdelayRespFollowUp, Announce, Signaling, Management }
struct PtpFlags { two_step: bool, unicast: bool, leap61: bool, leap59: bool,
                  utc_valid: bool, ptp_timescale: bool, /* ... */ }
struct PtpMessage { header: PtpHeader, body: PtpBody }
enum PtpBody { Sync(PtpTimestamp), FollowUp(PtpTimestamp), DelayResp{ rx: PtpTimestamp, req_port: PortIdentity },
               Announce(AnnounceBody), /* ... */ }
enum AssocState { Init, NoPeer, SyncCandidate, Selected, SysPeer, FalseTick, Reject }
struct NtpAssoc {
    addr: SocketAddr, state: AssocState,
    stratum: u8, refid: [u8;4],
    offset: NtpDuration, delay: NtpDuration,
    dispersion: f64, jitter: f64,            // seconds
    poll: i8, reach: u8,                      // 8-bit reachability register
    filter: FilterShiftRegister,
    xmt: NtpTimestamp,                        // last sent T1 (I10 echo check)
    last_rx: NtpTimestamp,
}
struct FilterShiftRegister { samples: [FilterSample; 8], idx: usize }
struct FilterSample { offset: NtpDuration, delay: NtpDuration, disp: f64, t: NtpTimestamp }
struct SelectInterval { lo: f64, hi: f64, mid: f64, assoc: AssocId }   // correctness interval
struct Truechimer { assoc: AssocId, offset: f64, jitter: f64, root_dist: f64 }
struct SystemState {
    leap: NtpLeap, stratum: u8, refid: [u8;4],
    root_delay: NtpShort, root_dispersion: NtpShort,
    offset: f64, jitter: f64, sys_peer: Option<AssocId>,
    poll: i8,
}
enum LoopMode { Pll, Fll, Hybrid }
struct LoopFilter {
    phase: f64,          // last offset (s)
    freq: f64,           // frequency estimate (ppm)
    time_const: f64,     // τ from poll
    mode: LoopMode,
    jitter: f64,
}
enum StepDecision { Slew(NtpDuration), Step(NtpDuration), Panic(NtpDuration) }
struct Servo { kp: f64, ki: f64, integrator: f64, last_offset: f64, state: ServoState }
enum ServoState { Unlocked, Locked, Holdover }

struct OscillatorModel { phase: f64, freq: f64, drift: f64, jitter: f64 }   // s, ppm, ppm/s, s
struct ClockAdjust { freq_ppm: f64, offset: NtpDuration, op: AdjOp }
enum AdjOp { Slew, StepSet, FreqOnly, Query }
struct Uncertainty { max_error: f64, est_error: f64 }   // seconds; interval = [t−max, t+max]

trait ClockBackend {                       // sys_clock OR virtual_clock
    fn now(&self) -> (NtpTimestamp, Uncertainty);
    fn adjust(&mut self, adj: ClockAdjust) -> Result<(), ClockError>;   // I13 sole mutator
    fn precision(&self) -> i8;
}
struct DefaultDS { clock_identity: [u8;8], number_ports: u16, priority1: u8,
                   clock_quality: ClockQuality, priority2: u8, domain: u8, slave_only: bool }
struct ClockQuality { clock_class: u8, clock_accuracy: u8, offset_scaled_log_variance: u16 }
struct CurrentDS { steps_removed: u16, offset_from_master: PtpInterval, mean_path_delay: PtpInterval }
struct ParentDS { parent_port: PortIdentity, gm_identity: [u8;8], gm_clock_quality: ClockQuality,
                  gm_priority1: u8, gm_priority2: u8 }
struct TimePropertiesDS { current_utc_offset: i16, leap61: bool, leap59: bool,
                          utc_valid: bool, ptp_timescale: bool, time_source: u8 }
struct ForeignMaster { source_port: PortIdentity, announce: AnnounceBody, count: u8, last_seen: NtpTimestamp }
struct Bmca { foreign: Vec<ForeignMaster>, erbest: Option<PortIdentity> }   // Erbest = best foreign master
enum PortState { Initializing, Faulty, Disabled, Listening, PreMaster, Master, Passive, Uncalibrated, Slave }
enum WireError { Short, BadVersion, ForbiddenMode, ExtLenOverrun, MacMismatch, BadFieldLen }
enum SyncError  { Falseticker, SanityFail(SanityReason), Unreachable, Bogus, Replay, Kod(KodCode) }
enum SanityReason { Stratum16, ZeroStratum, LeapAlarm, NegDelay, DispOverflow, OriginMismatch }
enum ClockError { NoCapability, ForeignDiscipline, PanicThreshold, AdjtimexFailed(i32) }
enum KodCode { Rate, Deny, Restr, Other([u8;4]) }
