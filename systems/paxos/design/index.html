Core schema & types (§11)

pub struct Ballot { pub round: u64, pub proposer_id: NodeId }   // P3, ADR-08
impl Ballot {
    pub const BOTTOM: Self;
    pub fn next(self) -> Self;                                  // round += 1, same proposer_id
}
impl Ord for Ballot { /* lexicographic: round, then proposer_id — I3, I13 */ }
pub struct NodeId(pub u32);                                     // ADR-09
pub struct Quorum<'a> { members: &'a [NodeId], cluster_size: usize }
impl<'a> Quorum<'a> {
    pub fn try_new(members: &'a [NodeId], cluster_size: usize) -> Option<Self>;  // I6, P10
    pub fn members(&self) -> &[NodeId];
}
pub struct Prepare { pub ballot: Ballot, pub slot: SlotId }
pub struct Promise  { pub ballot: Ballot, pub slot: SlotId, pub accepted: Option<(Ballot, Value)> }
pub struct Accept   { pub ballot: Ballot, pub slot: SlotId, pub value: Value }
pub struct Accepted { pub ballot: Ballot, pub slot: SlotId }
pub struct Learn    { pub slot: SlotId, pub ballot: Ballot, pub value: Value }
pub enum Message { Prepare(Prepare), Promise(Promise), Accept(Accept), Accepted(Accepted),
                   Learn(Learn), Heartbeat(Heartbeat), CatchupRequest(CatchupRequest),
                   CatchupResponse(CatchupResponse), ReconfigPropose(ReconfigPropose) }
pub struct AcceptorState {
    pub promised_ballot: Option<Ballot>,     // I2, I4 — disk-resident
    pub accepted_ballot: Option<Ballot>,     // I2, I5 — disk-resident
    pub accepted_value: Option<Value>,       // I2, I5 — disk-resident, atomic with accepted_ballot
}
impl AcceptorState {
    fn handle_prepare(&mut self, msg: &Prepare) -> Option<Promise>;  // I4; caller must fsync first (I2)
    fn handle_accept(&mut self, msg: &Accept) -> Option<Accepted>;   // I4; caller must fsync first (I2)
}
pub struct ProposerState { pub our_node: NodeId, pub cluster_size: usize,
                            pub current_ballot: Ballot, pub phase: ProposerPhase }
pub enum ProposerPhase {
    Idle,
    Preparing { ballot: Ballot, promises: Vec<Promise>, proposed_value: Value },
    Accepting { ballot: Ballot, value_being_committed: Value, accepts: Vec<Accepted> },
}
impl ProposerState {
    fn start_round(&mut self, value: Value) -> Prepare;                 // P3, I13
    fn handle_promise(&mut self, p: Promise) -> Option<Accept>;         // P2c value-derivation, §16.1
}
pub struct LearnerState { cluster_size: usize,
    accepts: BTreeMap<(SlotId, Ballot), BTreeSet<NodeId>>, chosen: BTreeMap<SlotId, Value> }
impl LearnerState {
    fn observe(&mut self, from: NodeId, msg: &Accepted, value: &Value) -> Option<Learn>;  // I9
}
pub trait StateMachine {
    type Snapshot;
    fn apply(&mut self, slot: SlotId, value: &Value);       // must be deterministic — I11, P11
    fn encode_state(&self) -> Self::Snapshot;               // must be order-stable — P8
    fn decode_state(bytes: &Self::Snapshot) -> Self;
}
pub enum WalRecord {
    Promise  { slot: SlotId, ballot: Ballot },
    Accepted { slot: SlotId, ballot: Ballot, value: Value },
    Snapshot { up_to_slot: SlotId, hash: [u8; 32] },
}
pub struct LogEntry { pub slot: SlotId, pub chosen_ballot: Ballot, pub value: Value }
pub struct PaxosLog { entries: BTreeMap<SlotId, LogEntry>,
                       snapshot_up_to: SlotId, first_uncommitted: SlotId }
pub struct Leader { pub our_node: NodeId, pub leader_ballot: Ballot,
                     pub next_uncommitted: SlotId, prepared: BTreeMap<SlotId, BulkPrepareEntry> }
pub struct BulkPrepareEntry { promises: usize, highest_accepted: Option<(Ballot, Value)> }
pub struct Membership { pub members: Vec<NodeId>, pub epoch: u64 }
pub struct ReconfigPropose { pub slot: SlotId, pub new_membership: Membership }
pub const ALPHA_DEFAULT: u64 = 3;                                 // ADR-16, I10
pub struct Client { leader_hint: Mutex<Option<NodeId>>, transport: Arc<dyn Transport> }
pub enum RpcRequest  { Propose(Value), Read(ReadMode) }
pub enum RpcResponse { Committed(SlotId, Value), NotLeader(Option<NodeId>), Timeout }
pub enum ReadMode { Linearizable, ReadIndex, Lease }              // ADR-19, ADR-18
pub struct TraceEvent<'a> { pub virtual_time: u64, pub node: NodeId, pub kind: TraceKind<'a> }
pub enum TraceKind<'a> { SendPrepare(&'a Prepare), RecvPrepare(&'a Prepare),
    FsyncPromise(&'a AcceptorState), SendPromise(&'a Promise), SendAccept(&'a Accept),
    RecvAccepted(&'a Accepted), Learn(&'a Learn) }
pub enum ProtocolError { StaleBallot, MalformedMessage(&'static str), QuorumUnavailable }
pub enum StorageError  { Io(std::io::Error), CrcMismatch { offset: u64 }, TornWrite { offset: u64 } }
