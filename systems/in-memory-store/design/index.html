Core schema & types (§11)

// crates/store-core/src/object.rs
#[repr(u8)]
pub enum ObjectType { String = 0, List = 1, Set = 2, Hash = 3, Zset = 4, Stream = 5 }

#[repr(u8)]
pub enum Encoding {
    RawString = 0, IntString = 1, EmbString = 2,
    LinkedList = 10, Quicklist = 11, ListpackList = 12,
    Hashtable = 20, ListpackHash = 21,
    HashtableSet = 30, Intset = 31, ListpackSet = 32,
    Skiplist = 40, ListpackZset = 41,
    StreamRadix = 50,
}

pub struct RedisObject {
    pub kind: ObjectType,
    pub encoding: Encoding,
    pub lru_or_lfu: u32,
    pub refcount: u32,
    pub ptr: ObjectPtr,
}

pub enum ObjectPtr {
    String(StringValue), List(Box<dyn ListEncoding>), Hash(Box<dyn HashEncoding>),
    Set(Box<dyn SetEncoding>), Zset(Box<dyn ZsetEncoding>), Stream(Box<StreamValue>),
}
// crates/store-core/src/db.rs
pub struct Database {
    pub id: u8,
    pub keys: Dict<Bytes, RedisObject>,
    pub expires: Dict<Bytes, Instant>,
    pub watched: HashMap<Bytes, Vec<ClientId>>,
    pub blocked_clients: HashMap<Bytes, Vec<ClientId>>,
    pub keyspace_notifications: NotifyMatrix,
    pub stats: DatabaseStats,
}

pub struct Dict<K, V> {
    ht: [HashTable<K, V>; 2],
    rehash_idx: i64,
    pause_rehash: u32,
}
// crates/store-server/src/server.rs
pub struct Server {
    pub config: Config, pub shards: Vec<Arc<Shard>>, pub cluster: Option<ClusterState>,
    pub replication: ReplicationState, pub aof: AofState, pub rdb: RdbState,
    pub eviction: EvictionState, pub expiration: ExpirationState, pub pubsub: PubsubState,
    pub acl: AclState, pub modules: ModuleRegistry, pub stats: ServerStats,
}

pub struct Shard {
    pub id: u16, pub databases: Vec<Database>, pub dispatcher: Mutex<Dispatcher>,
    pub clients: HashMap<ClientId, Connection>, pub blocked_keys: HashMap<Bytes, Vec<ClientId>>,
}

pub struct Connection {
    pub id: ClientId, pub addr: SocketAddr, pub protocol: Protocol,
    pub input_buffer: BytesMut, pub output_buffer: BytesMut, pub auth_state: AuthState,
    pub selected_db: u8, pub multi_state: Option<MultiState>, pub watch_set: HashSet<Bytes>,
    pub subscriptions: SubscriptionSet, pub flags: ConnectionFlags,
    pub last_interaction: Instant, pub created_at: Instant,
}
// crates/store-resp/src/value.rs
pub enum RespValue {
    SimpleString(Bytes), Error(Bytes), Integer(i64), BulkString(Option<Bytes>),
    Array(Option<Vec<RespValue>>), Map(Vec<(RespValue, RespValue)>), Set(Vec<RespValue>),
    Boolean(bool), Double(f64), BigNumber(Bytes),
    Verbatim { fmt: [u8; 3], data: Bytes }, Push(Vec<RespValue>), Null, Streamed(Vec<RespValue>),
}

// crates/store-dispatcher/src/command.rs
pub trait Command: Send + Sync + 'static {
    fn name(&self) -> &'static str;
    fn arity(&self) -> Arity;
    fn flags(&self) -> CommandFlags;
    fn first_key(&self) -> Option<usize>;
    fn last_key(&self) -> Option<usize>;
    fn key_step(&self) -> usize;
    fn acl_categories(&self) -> AclCategories;
    fn execute(&self, ctx: &mut CommandContext, args: &[Bytes]) -> Result<RespValue, CommandError>;
}
// crates/store-core/src/skiplist.rs
pub struct SkiplistNode {
    pub score: f64, pub member: Bytes, pub backward: *mut SkiplistNode,
    pub levels: Vec<SkiplistLevel>,
}
pub struct SkiplistLevel { pub forward: *mut SkiplistNode, pub span: u64 }
pub struct Skiplist { pub head: Box<SkiplistNode>, pub tail: *mut SkiplistNode,
    pub length: u64, pub level: u8 }
// crates/store-cluster/src/lib.rs
pub const SLOT_COUNT: u16 = 16384;
pub struct ClusterState { pub my_id: NodeId, pub nodes: HashMap<NodeId, NodeInfo>,
    pub slots: SlotMap, pub epoch: u64, pub config_epoch: u64, pub known_nodes: KnownNodes }
pub struct SlotMap { pub owners: [Option<NodeId>; SLOT_COUNT as usize],
    pub migrating: HashMap<u16, NodeId>, pub importing: HashMap<u16, NodeId> }

pub trait Cluster: Send + Sync {
    fn slot_for(&self, key: &[u8]) -> u16;
    fn owner(&self, slot: u16) -> Option<NodeId>;
    fn redirect(&self, slot: u16, key: &[u8]) -> Option<RedirectKind>;
    fn handle_gossip(&mut self, msg: GossipMessage);
    fn current_epoch(&self) -> u64;
}
pub enum RedirectKind { Moved(SocketAddr), Ask(SocketAddr) }
// crates/store-persistence/src/lib.rs
pub trait AofWriter: Send + Sync {
    fn append(&mut self, cmd: &[Bytes]) -> io::Result<()>;
    fn flush_to_disk(&mut self) -> io::Result<()>;
    fn current_offset(&self) -> u64;
    fn rewrite(&mut self, snapshot: DatabaseSnapshot) -> io::Result<()>;
}
pub trait RdbWriter: Send + Sync {
    fn save(&mut self, snapshot: DatabaseSnapshot, path: &Path) -> io::Result<()>;
}
pub trait RdbLoader: Send + Sync { fn load(&mut self, path: &Path) -> io::Result<Database>; }

// crates/store-replication/src/lib.rs
pub trait Replication: Send + Sync {
    fn append_to_backlog(&mut self, cmd: &[Bytes]);
    fn handshake(&mut self, addr: SocketAddr) -> io::Result<HandshakeResult>;
    fn psync(&mut self, replid: ReplicationId, offset: u64) -> io::Result<PsyncResult>;
    fn ack(&mut self, offset: u64);
    fn min_replica_offset(&self) -> u64;
}
pub enum PsyncResult { FullResync { replid: ReplicationId, offset: u64 }, Continue,
    ContinueNewReplid(ReplicationId) }

pub struct ReplicationBacklog { pub buffer: Vec<u8>, pub size: usize, pub head: usize,
    pub offset: u64 }
// crates/store-core/src/error.rs
pub enum CommandError {
    WrongType, WrongArity(&'static str), NotAnInteger, NotAFloat, Syntax, NoAuth, NoPerm,
    Readonly, Moved { slot: u16, addr: SocketAddr }, Ask { slot: u16, addr: SocketAddr },
    ClusterDown, Oom, ExecAbort, Loading, Busy, MasterDown, PhaseDeferred(&'static str),
}
// crates/store-expiration/src/lib.rs
pub struct Ttl { pub absolute: Instant }
pub trait ActiveExpire { fn cycle(&mut self, db: &mut Database, budget: Duration) -> ExpireStats; }
pub struct ExpireStats { pub sampled: usize, pub expired: usize, pub elapsed: Duration }
