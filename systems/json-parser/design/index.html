Core schema & types (§11)

enum class ValueTag : uint8_t { Null, Bool, Integer, Double, ShortString, String, Array, Object };
// Ownership: value type, no ownership itself. Mutation: immutable (P10). Concurrency: trivially
// shareable (POD). Persistence: not persisted directly. Serialization: drives Serializer dispatch.
// Compatibility: adding a variant is a breaking ABI change (§27) — enum is closed per release.

struct ValueUnion {
    uint64_t tag_and_flags;   // 3-bit tag + 1-bit inline flag, see §8.7
    uint64_t payload;         // inline value or {block_index:32, block_offset:28, reserved:4}
};
// Ownership: indirect payload references Arena-owned memory; ValueUnion itself owns nothing.
// Mutation: immutable post-construction. Concurrency: read-safe when Document is read-only (I11).
// Lifetime obligation: indirect payload is valid only for the owning Document's lifetime.

struct Value {
    ValueUnion repr;
    // Ownership: see ValueUnion. Mutation: via DocumentBuilder only (ADR-20). Persistence: N/A.
    // Serialization: primary input to Serializer. Compatibility: layout is an internal detail (§27).
};
struct ObjectRef { uint32_t block_index; uint32_t block_offset; uint32_t member_count; };
struct ArrayRef  { uint32_t block_index; uint32_t block_offset; uint32_t element_count; };
struct StringRef { uint32_t block_index; uint32_t block_offset; uint32_t byte_length; };
// Ownership: indirect references into Arena blocks, owned by the Document. Mutation: immutable.
// Concurrency: read-safe under I11. Lifetime obligation: valid only within owning Document lifetime.

struct StringView {
    const char* data;
    uint32_t length;
    bool borrowed_from_input;  // true = zero-copy view into the caller's input buffer (P5)
};
// Ownership: non-owning. Untrusted-input handling: content is validated UTF-8 only after
// Utf8Validator has run over [data, data+length). Lifetime obligation: valid only as long as the
// source buffer (input buffer if borrowed_from_input, else the owning Arena) is alive.

enum class NumberKind : uint8_t { Integer, Double };
struct NumberValue { NumberKind kind; union { int64_t as_int; double as_double; } value; };
// Ownership: value type. Mutation: immutable. Serialization: drives Ryū (double) or decimal
// integer formatting (Integer) in the Serializer. Compatibility: kind enum closed per release.
struct Arena {
    ArenaBlock* head_block;
    ArenaBlock* current_block;
    size_t total_bytes_allocated;
};
struct ArenaBlock { ArenaBlock* next; size_t used; size_t capacity; /* inline byte storage */ };
// Ownership: owns all its blocks; owned exclusively by one Document. Mutation: append-only during
// parse, wholesale-reset on parse failure (P7) or Document destruction. Concurrency: single-writer
// (the parsing thread); read-only post-construction is safe under I11. Persistence: not persisted.

struct Limits {
    uint32_t max_depth;            // default 512
    uint64_t max_document_bytes;    // default ~2 GiB
    uint32_t max_string_bytes;       // default ~256 MiB
    uint32_t max_number_digits;       // default 1024
    uint64_t max_allocation_bytes;     // arena ceiling, derived multiple of input length
};
struct ParseOptions {
    Limits limits;
    bool zero_copy;                  // enable StringView borrowing (P5)
    bool strict_duplicate_keys;       // ADR-16 strict mode toggle
    bool allow_infinity_on_overflow;   // ADR-15-adjacent overflow policy, default false
    uint8_t extensions;                 // ADR-23 bitmask, default 0 (strict RFC 8259)
};
// Ownership: value types, owned by the caller, passed by const reference into parse calls.
// Mutation: immutable once passed to a parse call. Compatibility: new fields are additive-only,
// defaulted (§27 additive compatibility rule).
enum class ErrorCode : uint16_t {
    UnexpectedEof, UnexpectedToken, TrailingData, InvalidEscape, InvalidUnicodeEscape,
    LoneSurrogate, InvalidControlChar, UnterminatedString, InvalidNumber, NumberTooLong,
    NumberOutOfRange, InvalidUtf8, DepthLimitExceeded, DocumentTooLarge, StringTooLong,
    AllocationLimitExceeded, DuplicateKey, OutOfMemory
};
// Ownership: value type. Compatibility: adding a code is additive-safe; removing/renumbering is
// breaking (§27 error-code stability).

struct SourceLocation { uint64_t byte_offset; uint32_t line; uint32_t column; };
struct ParseError { ErrorCode code; SourceLocation location; };
// Ownership: value types. Serialization: human-readable formatting is a library-provided helper,
// not baked into the type. Compatibility: SourceLocation fields are additive-only.

struct ParseResult {
    bool ok;
    union { Document value; ParseError error; };  // variant discipline per P7/I7
};
// Ownership: owns the Document on success (moves out to caller); owns the ParseError on failure.
// Mutation: immutable after construction.

class SaxHandler {
public:
    virtual void on_start_object() = 0;
    virtual void on_end_object() = 0;
    virtual void on_start_array() = 0;
    virtual void on_end_array() = 0;
    virtual void on_key(StringView) = 0;
    virtual void on_string(StringView) = 0;
    virtual void on_number(NumberValue) = 0;
    virtual void on_bool(bool) = 0;
    virtual void on_null() = 0;
    virtual void on_error(ParseError) = 0;
    virtual ~SaxHandler() = default;
};
// Ownership: caller-owned, library holds a non-owning reference during a single streaming parse.
// Untrusted-input handling: StringView params obey the same UTF-8-validated, lifetime-bounded
// contract as DOM StringViews; on_error is terminal, no further calls follow.

enum class SaxEvent : uint8_t {
    StartObject, EndObject, StartArray, EndArray, Key, StringValue, NumberValue, BoolValue,

};

enum class PatchOpKind : uint8_t { Add, Remove, Replace, Move, Copy, Test };
struct PatchOp { PatchOpKind kind; JsonPointer path; JsonPointer from; Value value; };
// Ownership: PatchOp owns a copy of `value` if present (arena-allocated in the target Document's

struct Document {

    const char* input_buffer;      // borrowed or owned, per zero_copy option

};
