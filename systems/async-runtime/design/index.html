Core schema & types (§11)

#[repr(C)]
struct Header {
    state: AtomicU32,        // packed: 3-bit state | 1-bit notified | 27-bit refcount (I1)
    vtable: &'static Vtable, // clone/wake/wake_by_ref/drop fn pointers (P2)
    scheduler: SchedulerHandle,
}
// Ownership: shared via Arc-style manual refcount (ADR-19); Send+Sync (P13).
// Mutation: CAS-only on `state`; no lock.

struct RawTask<F: Future, S> {
    header: Header,
    future: UnsafeCell<ManuallyDrop<F>>,     // pinned in place at construction
    output: UnsafeCell<MaybeUninit<F::Output>>,
    _scheduler: PhantomData<S>,
}
// Ownership: one Box<RawTask> per spawn (ADR-19). Concurrency: guarded by Header.state.
struct JoinHandle<T> { raw: NonNull<Header>, _marker: PhantomData<T> }
// Send+Sync iff T: Send (P13). Drop => Cancelled if not Complete/Joined (P3, I7).

enum JoinError { Cancelled, Panicked(Box<dyn Any + Send + 'static>) } // P14

struct TaskId(u64); // monotonic per-runtime counter; surfaced in rt-trace (P1)

struct Waker { raw: RawWaker }               // std-adopted (ADR-02)
struct RawWaker { data: *const (), vtable: &'static RawWakerVTable } // P2
trait Reactor: Send + Sync {
    fn register(&self, source: RawFd, interest: Interest, waker: Waker) -> SourceToken;
    fn deregister(&self, token: SourceToken);
    fn drive(&self, timeout: Option<Duration>) -> usize;
}
// Concurrency: implementations must be internally synchronized (P10). No persistence.

struct TimerEntry {
    deadline: Instant,
    waker: Waker,
    level: u8, slot: u16,       // HHWT placement (§18)
    links: IntrusiveLinks,      // O(1) cancel
}
struct WorkerHandle {
    deque: ChaseLevDeque<TaskRef>,   // owner-LIFO / remote-FIFO (P6, ADR-04)
    lifo_slot: Option<TaskRef>,      // single-slot fast path (ADR-04)
    parked: AtomicBool,              // I11 lost-wakeup guard
}
struct Injector { ring: BoundedMpmcRing<TaskRef> } // ADR-22, I3

struct Budget(Cell<u32>); // per-task poll counter, default 128 (ADR-23, I5)
struct TaskGroup<'scope> { children: Vec<JoinHandle<()>>, cancel: CancellationToken } // P7, I8
struct JoinSet<T> { children: Vec<JoinHandle<T>> }                                     // §20
struct Scope<'scope, 'env> { children: Vec<JoinHandle<()>>, _marker: PhantomData<&'scope &'env ()> }
