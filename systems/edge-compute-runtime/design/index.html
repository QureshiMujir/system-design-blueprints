Core schema & types (§11)

pub struct WorkerId(u64);
pub struct TenantId(u64);
pub struct IsolateId(u64);
pub struct PoPId(u32);
pub struct Request {
    pub method:  Method,
    pub url:     Url,
    pub headers: HeaderMap,
    pub body:    Option<Bytes>,
}

pub struct Response {
    pub status:  u16,
    pub headers: HeaderMap,
    pub body:    Bytes,
}

pub enum Method {
    Get, Post, Put, Delete, Patch, Head, Options, Connect, Trace,
    Other(Box<str>),
}
pub struct ResourceLimits {
    pub cpu_fuel:     u64,   // Wasmtime fuel units; trap on exhaustion (I1, P7)
    pub memory_bytes: u32,   // max linear memory bytes; converted to max pages
    pub wall_ms:      u32,   // wall-clock timeout in milliseconds
    pub subrequests:  u16,   // max outbound fetch calls per invocation (I5)
}

pub struct FuelMeter {
    pub initial:  u64,
    pub consumed: AtomicU64,  // incremented by Wasmtime fuel callbacks
}

pub struct EpochDeadline {
    pub epoch_target: u64,  // store.set_epoch_deadline(n); traps when epoch >= target
}
pub struct WorkerScript {
    pub id:       WorkerId,
    pub tenant:   TenantId,
    pub wasm:     Bytes,
    pub entry:    HandlerKind,
    pub limits:   ResourceLimits,
    pub bindings: Vec<Binding>,   // capability-scoped host surface (I10, P9)
}

pub enum HandlerKind { Fetch, Scheduled }

pub struct Binding {
    pub name: Box<str>,
    pub kind: BindingKind,
}

pub enum BindingKind {
    KvNamespace(KvNamespace),
    EnvVar { key: Box<str>, value: Box<str> },
    Secret { key: Box<str> },        // value injected at runtime; not persisted in script
}
pub struct ExecutionContext {
    pub request:          Request,
    pub deadline:         EpochDeadline,
    pub fuel_meter:       FuelMeter,
    pub subrequest_count: AtomicU16,
    pub tenant:           TenantId,
    pub isolate_id:       IsolateId,
    pub started_at:       Instant,
}
pub struct Isolate {
    pub id:             IsolateId,
    pub instance:       Instance,      // Wasmtime Instance; owns linear memory
    pub memory_pages:   u32,           // current allocated pages; checked against limit
    pub fuel_remaining: u64,           // mirrored from FuelMeter at last checkpoint
    pub epoch_deadline: EpochDeadline,
    pub tenant:         TenantId,      // tagged at creation; never mutated (I2)
}
pub struct InstancePool {
    pub per_worker: HashMap<WorkerId, Vec<WarmInstance>>,
    pub allocator:  PoolingAllocator,
    pub capacity:   usize,            // max total WarmInstances across all workers
}

pub struct WarmInstance {
    pub instance:    Instance,
    pub reset_token: ResetToken,      // validated before reuse (I4, P10)
}

pub struct ResetToken(u64);           // incremented each time an instance is reset

pub struct ColdStartSnapshot {
    pub module_id:                ModuleId,
    pub initialized_memory_image: Bytes,  // CoW page-set post-init run
    pub globals:                  Vec<GlobalValue>,
    pub format_version:           u8,
}

pub struct ModuleId(u64);             // SHA-256 truncated to 64 bits of compiled WASM
pub enum HostCall {
    Fetch(Request),
    KvGet  { ns: KvNamespace, key: Box<[u8]> },
    KvPut  { ns: KvNamespace, key: Box<[u8]>, val: Bytes, ttl: Option<Duration> },
    CacheGet(CacheKey),
    CachePut { key: CacheKey, resp: Response, ttl: Duration },
    Log    { level: LogLevel, message: Box<str> },
}

pub enum LogLevel { Trace, Debug, Info, Warn, Error }
pub struct KvNamespace {
    pub tenant: TenantId,
    pub name:   Box<str>,    // logical namespace; unique per tenant
}

pub struct KvEntry {
    pub value:      Bytes,
    pub expires_at: Option<SystemTime>,  // None = no expiry; I6: expired not served
    pub version:    u64,                 // monotonic; LWW merge in Phase 11
}
pub struct CacheKey {
    pub url_hash:  [u8; 32],    // SHA-256 of canonical URL
    pub vary_hash: [u8; 16],    // MD5 of Vary-matched request header values
}

pub struct CacheEntry {
    pub key:        CacheKey,
    pub response:   Response,
    pub expires_at: SystemTime,   // hard expiry; I6
    pub stale_until: SystemTime,  // stale-while-revalidate window end
}
pub struct PoP {
    pub id:       PoPId,
    pub location: GeoCoord,       // (lat, lon) for geo-distance fallback
    pub capacity: u32,            // max concurrent requests
    pub healthy:  AtomicBool,     // set by health probe (I7)
}

pub struct GeoCoord { pub lat: f32, pub lon: f32 }

pub struct RoutingDecision {
    pub pop:    PoPId,
    pub reason: RoutingReason,
}

pub enum RoutingReason { AnyCast, GeoFallback, ExplicitPin }
pub struct Invalidation {
    pub kind:       InvalidationKind,
    pub version:    u64,              // monotonic; receivers discard if version ≤ last seen
    pub origin_pop: PoPId,
}

pub enum InvalidationKind {
    KvKey   { ns: KvNamespace, key: Box<[u8]> },
    CacheKey(CacheKey),
}

pub struct VersionVector(HashMap<PoPId, u64>);   // per-origin monotonic clocks
