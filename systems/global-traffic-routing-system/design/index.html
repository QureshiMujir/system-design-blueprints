Core schema & types (§11)

pub struct EndpointId(u32);
pub struct PoolId(u32);
pub struct Endpoint { id: EndpointId, addr: Target, loc: GeoLoc, weight: u16, hc: HealthCheckId, pool: PoolId }
pub enum Target { V4(Ipv4Addr), V6(Ipv6Addr), Cname(String) }
pub struct GeoLoc { lat: f32, lon: f32, country: [u8;2], continent: Continent }
pub struct Pool { id: PoolId, name: String, endpoints: Vec<EndpointId>, policy: Policy, hc: HealthCheck, ttl: Ttl }
pub struct PolicySet { name: String, root: Policy }      // top-level, possibly nested (§8.4)
pub struct HealthCheckId(u32);
pub enum CheckKind { Http{ path:String, expect_status:u16, expect_body:Option<String> }, Tcp{ port:u16 }, Icmp }
pub struct HealthCheck { kind: CheckKind, interval: Duration, timeout: Duration, k_fail: u8, k_pass: u8 }
pub enum HealthState { Healthy, Degraded, Unhealthy, Unknown }              // consensus result (I-CONS)
pub struct VantageVerdict { vantage: VantageId, up: bool, rtt: Option<Duration>, ts: Instant }
pub struct FlapState { penalty: f32, damped: bool, last_flip: Instant }     // I-DAMP
pub trait Consensus { fn fuse(&self, votes:&[VantageVerdict], weights:&VantageWeights) -> HealthState; }  // quorum, I-CONS
pub trait FlapDamper { fn observe(&mut self, flipped:bool) -> bool; }       // returns `damped` (I-DAMP)
pub struct VantageWeights { w: HashMap<VantageId, f32> }                    // per-vantage reliability
pub struct PrefixTrie { /* LPM over geo DB prefixes (I-LPM) */ }
impl PrefixTrie { fn locate(&self, subnet: IpNet) -> Option<&GeoLoc>; }     // longest-prefix match
pub struct EcsOption { family: u8 /*1=v4,2=v6*/, source_prefix_len: u8, scope_prefix_len: u8, addr: IpAddr }
impl EcsOption { fn validate(&self) -> Result<(),GtrError>; }              // I-ECS-SAFE (caps, family, zeroed)
pub fn geoproximity(client:&GeoLoc, ep:&GeoLoc, bias_km: i32) -> f64;       // great-circle + bias (§8.10)
pub struct ClientContext { src: IpAddr, ecs: Option<EcsOption>, qname: String, qtype: RecordType, ts: Instant }
pub enum RecordType { A, AAAA, CNAME }
pub struct SteeringDecision { endpoints: Vec<EndpointId>, ttl: Ttl, ecs_scope: u8, trace: DecisionTrace }
pub struct DecisionTrace { policy_path: Vec<String>, location: Option<GeoLoc>, ecs_scope: u8,
                           chosen: Vec<EndpointId>, health_used: Vec<(EndpointId,HealthState)>,
                           generation: u64, latency_fallback: Option<&'static str>, failsafe: bool }
pub enum Policy { Failover(Vec<Box<Target2>>), Weighted(Vec<(Box<Target2>,u16)>), Latency(Vec<Box<Target2>>),
                  Geolocation{ rules:Vec<(Region,Box<Target2>)>, default:Option<Box<Target2>> },
                  Geoproximity(Vec<(Box<Target2>,i32)>), MultiValue{ targets:Vec<Box<Target2>>, max:u8 } }
pub enum Target2 { Endpoint(EndpointId), Pool(PoolId), Nested(Policy) }     // composite (§8.4)
pub enum Region { Continent(Continent), Country([u8;2]), Subdivision(String) }
pub trait Evaluator {
    fn decide(&self, p:&Policy, ctx:&ClientContext, health:&HealthSnapshot,
              lm:&LatencyMap, geo:&PrefixTrie, seed:u64) -> SteeringDecision;     // deterministic, I-DET
}
pub struct ClientNet(IpNet);                                               // e.g. /24
pub struct LatencyMap { table: HashMap<ClientNet, HashMap<EndpointId, LatencyEst>> }
pub struct LatencyEst { ms: f32, samples: u32, confidence: f32, last_update: Instant }  // decayed (§8.10)
impl LatencyMap { fn lookup(&self, net:ClientNet) -> Option<&HashMap<EndpointId,LatencyEst>>; }
pub fn ingest_rum(map:&mut LatencyMap, sample: RumSample);                  // RUM aggregation
pub struct Ttl(u32);
pub struct ResourceRecord { name:String, rtype:RecordType, ttl:Ttl, rdata:RData }
pub enum RData { A(Ipv4Addr), Aaaa(Ipv6Addr), Cname(String) }
pub struct DnsAnswer { records: Vec<ResourceRecord>, opt_ecs: Option<EcsOption>, rcode: Rcode }
pub enum Rcode { NoError, NxDomain, NoData, FormErr, ServFail }
pub fn build_answer(d:&SteeringDecision, ctx:&ClientContext, soa_min:Ttl) -> DnsAnswer;   // I-WIRE/I-TTL/I-ECS
pub struct AnycastPrefix(IpNet);
pub enum AnnounceIntent { Announce{ prefix:AnycastPrefix, pop:PopId }, Withdraw{ prefix:AnycastPrefix, pop:PopId } }
pub trait BgpSink { fn apply(&self, intent: AnnounceIntent) -> Result<(),GtrError>; }   // real BGP deferred §22.2
pub fn announce_plan(pops:&[(PopId,HealthState)], prefix:AnycastPrefix) -> Vec<AnnounceIntent>;
pub struct Generation(u64);                                                // monotonic, leader-advanced (I-LEADER)
pub struct ResolvedPlan { gen: Generation, pools: Vec<Pool>, health: HealthSnapshot,
                          latency_delta: Vec<(ClientNet,EndpointId,LatencyEst)>, policy_ver: u64 }
pub struct EdgeState { gen: Generation, pools: Arc<Vec<Pool>>, health: Arc<HealthSnapshot>,
                       lm: Arc<LatencyMap>, geo: Arc<PrefixTrie> }          // RCU snapshot (I-SNAP)
pub trait Propagator { fn push(&self, plan:&ResolvedPlan, edges:&[EdgeId]); fn ack(&self, edge:EdgeId, g:Generation); }
pub struct Staleness { leader_gen: Generation, per_edge: HashMap<EdgeId, (Generation, Instant)> }  // I-PROP
pub struct Epoch(u64);
pub trait Leader { fn is_leader(&self)->bool; fn epoch(&self)->Epoch; fn fence(&self, e:Epoch)->bool; }   // I-LEADER
pub struct ControlPlane { /* sharded-by-pool aggregation, decision compute, propagation */ }
pub struct Edge { state: ArcSwap<EdgeState>, responders: (DnsResponder, HttpRedirector) }
pub struct Prober { vantage: VantageId, checks: Vec<(EndpointId, HealthCheck)> }
pub struct GtrConfig { role: Role, pools: PathBuf, geodb: PathBuf, vantages: Vec<VantageId>,
                       coordinator: SocketAddr, staleness_gens: u64, staleness_secs: u64 }
pub enum GtrError { BadArg(String), EcsInvalid(String), GeoMiss, NoHealthData,
                    WireFormat{ off:usize }, PropStale(EdgeId), GenRegress{ have:u64, got:u64 },
                    NotLeader(Epoch), SinkRejected(String), PolicyInvalid(String) }
pub struct DecisionMetric { policy_kind:&'static str, location:&'static str, ecs_scope:u8,
                            chosen_pop:&'static str, health_filtered:u32, ttl:u32, generation:u64,
                            latency_us:u64, failsafe:bool }
pub struct HealthMetric { endpoint:u32, state:&'static str, quorum:u8, vantages_up:u8, flap_penalty:f32, damped:bool }
pub struct PropMetric { edge:u32, leader_gen:u64, edge_gen:u64, lag_secs:f64, stale:bool, apply_us:u64 }
