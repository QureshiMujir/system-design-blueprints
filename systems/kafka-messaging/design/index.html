Core schema & types (§11)

// TopicName(String)              own: broker; mut: immutable; concur: Arc<str>; persist: metadata log; ser: compact string; compat: stable
// PartitionId(u32)               own: broker; mut: immutable; concur: Copy; persist: metadata log; ser: int32; compat: stable
// TopicPartition{topic,partition} own: broker; mut: immutable; concur: Copy+Hash; persist: metadata log; ser: struct; compat: stable
// Offset(u64)                    own: log; mut: monotonic-append-only; concur: atomic read; persist: segment; ser: int64; compat: stable
// LogEndOffset(Offset)           own: replica; mut: advances on append; concur: per-partition lock; persist: derived; ser: int64; compat: stable
// HighWatermark(Offset)          own: leader ISR mgr; mut: advances on ISR catch-up; concur: atomic; persist: checkpoint file; ser: int64; compat: stable
// LogStartOffset(Offset)         own: retention worker; mut: advances on delete; concur: atomic; persist: checkpoint file; ser: int64; compat: stable
// LeaderEpoch(i32)               own: controller; mut: monotonic increment; concur: Copy; persist: epoch checkpoint; ser: int32; compat: stable
struct RecordBatch {               // own: log append path; mut: append-only; concur: none post-write; persist: segment .log; ser: §8.1 layout; compat: magic-versioned
    base_offset: Offset,
    partition_leader_epoch: LeaderEpoch,
    magic: u8,
    crc: Crc32c,
    attributes: u16,
    producer_id: ProducerId,
    producer_epoch: ProducerEpoch,
    base_sequence: SequenceNumber,
    records: Vec<Record>,
}
struct RecordHeader { key: String, value: Vec<u8> } // own: record; mut: immutable; ser: varint-prefixed
struct Record {                    // own: batch; mut: immutable once appended; ser: §8.1.1 varint layout
    offset_delta: i32, timestamp_delta: i64,
    key: Option<Vec<u8>>, value: Option<Vec<u8>>, headers: Vec<RecordHeader>,
}
struct Crc32c(u32);                // own: batch; mut: computed once; ser: uint32; compat: RFC 3385 fixed
enum CompressionType { None, Gzip, Snappy, Lz4, Zstd } // persist: attributes bits 0-2; compat: additive
struct Segment {                   // own: kbroker-log; mut: append-only until rolled; concur: single-writer; persist: .log file
    base_offset: Offset, path: PathBuf, size_bytes: u64,
}
struct SegmentHeader { base_offset: Offset, created_at_ms: i64 } // persist: implicit via filename+first batch
struct IndexEntry { relative_offset: i32, position: i32 } // ser: 8 bytes fixed-width (OffsetIndex)
struct OffsetIndex { entries: Vec<IndexEntry>, mmap: MmapHandle } // own: segment; concur: read-mostly mmap
struct TimeIndex { entries: Vec<(i64, i32)>, mmap: MmapHandle }   // ser: 12 bytes fixed-width
struct Log {                       // own: kbroker-log; concur: per-partition actor/shard; persist: segment dir
    tp: TopicPartition, segments: Vec<Segment>, leo: LogEndOffset, hw: HighWatermark, lso: LogStartOffset,
}
struct LogAppendResult { base_offset: Offset, log_append_time_ms: i64 } // ser: produce response fields
struct FetchRequest { tp: TopicPartition, fetch_offset: Offset, max_bytes: u32, max_wait_ms: u32, min_bytes: u32 }
struct FetchResult { records: Vec<RecordBatch>, high_watermark: HighWatermark, log_start_offset: LogStartOffset }
enum AckLevel { None, Leader, All } // ser: int16 (-1/0/1 wire values); compat: stable, see ADR-10
struct ProducerId(i64);            // own: broker (assigned via InitProducerId); persist: producer-state snapshot
struct ProducerEpoch(i16);         // mut: incremented on InitProducerId re-init; fences zombies (P6)
struct SequenceNumber(i32);        // own: per (ProducerId,ProducerEpoch,partition); mut: monotonic within epoch
struct ReplicaId(i32);             // own: controller/metadata; concur: Copy
enum ReplicaState { Leader, Follower, Observer } // mut: controller-assigned; persist: metadata log
struct Isr { members: Vec<ReplicaId>, last_caught_up_ms: HashMap<ReplicaId,i64> } // own: leader ISR mgr; concur: per-partition lock
struct FetcherCursor { tp: TopicPartition, next_fetch_offset: Offset, leader_epoch: LeaderEpoch } // own: replica fetcher
struct ControllerState { epoch: i32, active_controller: ReplicaId } // own: kbroker-controller; persist: metadata log
enum MetadataRecord { TopicRecord, PartitionRecord, LeaderChangeRecord, ProducerIdsRecord } // ser: metadata-log entries; compat: schema-versioned
struct GroupId(String);            // own: group coordinator; concur: hash-routed
struct MemberId(String);           // own: coordinator-assigned per join; mut: reassigned on rejoin after expiry
struct GenerationId(i32);          // mut: incremented per completed rebalance (I11 gate)
struct Assignment { tp: Vec<TopicPartition> } // own: group leader computed, coordinator-distributed
enum RebalanceState { Empty, PreparingRebalance, CompletingRebalance, Stable } // see §8.9 table
struct OffsetCommit { group: GroupId, tp: TopicPartition, offset: Offset, generation: GenerationId } // persist: offset store
struct TransactionalId(String);    // own: producer-declared, coordinator hash-routed
enum TransactionState { Empty, Ongoing, PrepareCommit, PrepareAbort, CompleteCommit, CompleteAbort } // persist: txn log
struct ControlRecord { marker: TransactionState, producer_id: ProducerId, producer_epoch: ProducerEpoch } // ser: isControlBatch=1
struct RetentionPolicy { retention_ms: Option<i64>, retention_bytes: Option<i64>, cleanup_policy: CleanupPolicy } // persist: topic config
enum CleanupPolicy { Delete, Compact, CompactAndDelete } // compat: additive
struct CompactionCursor { tp: TopicPartition, dirty_ratio: f32, cleaned_offset: Offset } // own: compaction worker
struct Quota { entity: String, produce_bytes_per_sec: u64, fetch_bytes_per_sec: u64 } // own: kbroker-quota
enum BrokerError {                 // ser: int16 Kafka-compatible error codes; compat: additive only
    None, CorruptMessage, NotLeaderOrFollower, OutOfOrderSequence, OffsetOutOfRange,
    IllegalGeneration, UnknownMemberId, InvalidRecord, RequestTimedOut, DuplicateSequence,
}
