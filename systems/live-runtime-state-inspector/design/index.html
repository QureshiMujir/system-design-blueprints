Core schema & types (§11)

// SessionId — owner: SessionManager; immutable once issued; not persisted across Detached.
type SessionId = string & { readonly __brand: "SessionId" };

// NodeId — owner: serializer's WeakRef table; stable within a session; not compatible across
// session boundaries; serializable as an opaque string.
type NodeId = string & { readonly __brand: "NodeId" };

// LogicalClock — owner: hooks (issued at source); monotonic per session; compared, never added.
type LogicalClock = number & { readonly __brand: "LogicalClock" };

// StateSnapshot — owner: serializer; produced per keyframe; immutable once emitted;
// serializes via CBOR (ADR-05); versioned by `ussVersion` (see §27).
interface StateSnapshot {
  readonly sessionId: SessionId;
  readonly clock: LogicalClock;
  readonly roots: readonly NodeId[];
  readonly nodes: ReadonlyMap<NodeId, SnapshotNodeRecord>; // see §8.1 field table
  readonly ussVersion: string;
}
// SerializedValue — tagged union; owner: serializer; immutable; produced per primitive node.
type SerializedValue =
  | { tag: "string"; v: string }
  | { tag: "number"; v: number }
  | { tag: "boolean"; v: boolean }
  | { tag: "bigint"; v: string }   // decimal string form, CBOR bignum on the wire
  | { tag: "null" }
  | { tag: "undefined" }
  | { tag: "native"; kind: string; summary: string }; // Date, Map, Set, Promise, RegExp, etc.

// TruncationMarker — owner: serializer; attached to a node exceeding depth/breadth caps (§4).
interface TruncationMarker {
  readonly reason: "depth" | "breadth" | "node-cap" | "byte-cap";
  readonly limit: number;
}

// CycleRef — owner: serializer; emitted in place of re-descending into a visited node (I3).
interface CycleRef {
  readonly target: NodeId;
}
// DependencyEdge — owner: Dependency Graph Tracker; append-only within a session.
interface DependencyEdge {
  readonly from: NodeId;
  readonly to: NodeId;
  readonly kind: "reads" | "writes" | "subscribes" | "triggers";
  readonly clock: LogicalClock;
}

// EffectRecord — owner: Dependency Graph Tracker; mutation: append/retire only, never edited.
interface EffectRecord {
  readonly nodeId: NodeId;
  readonly lastRunClock: LogicalClock;
  readonly dependsOn: readonly NodeId[];
  readonly disposed: boolean;
}

// AsyncTaskRecord — owner: async-tracking hook (Phase 8); concurrency: single writer per task.
interface AsyncTaskRecord {
  readonly taskId: string;
  readonly startClock: LogicalClock;
  readonly endClock: LogicalClock | null;
  readonly causedBy: NodeId | null;
  readonly status: "pending" | "settled" | "rejected";
}
// EventRecord — owner: hooks; append-only; see §8.4 for the full field table.
interface EventRecord {
  readonly eventId: string;
  readonly clock: LogicalClock;
  readonly kind: "dispatch" | "render" | "effect-run" | "subscription-fire" | "async-boundary";
  readonly subjectNodeId: NodeId;
  readonly payloadSummary: SerializedValue | null;
  readonly sourceAdapter: string;
  readonly sizeBytes: number;
}

// RingBuffer<T> — owner: @lrsi/eventlog; mutation: push-with-eviction only (ADR-10);
// concurrency: single-writer (hooks), multi-reader snapshot view.
interface RingBuffer<T> {
  readonly capacity: number;
  readonly byteCapacity: number;
  push(item: T): void;                 // may evict oldest non-keyframe entry
  snapshotRange(from: LogicalClock, to: LogicalClock): readonly T[];
}
// TransportMessage — owner: @lrsi/transport; wire-serializable via CBOR; versioned (§27).
type TransportMessage =
  | { type: "Hello"; protocolVersion: string; capabilities: readonly string[] }
  | { type: "AttachRequest"; targetId: string }
  | { type: "AttachAck"; sessionId: SessionId; snapshot: StateSnapshot }
  | { type: "Detach"; sessionId: SessionId }
  | { type: "Snapshot"; snapshot: StateSnapshot }
  | { type: "SnapshotDelta"; sessionId: SessionId; patch: readonly DeltaOp[] }
  | { type: "EventBatch"; sessionId: SessionId; events: readonly EventRecord[] }
  | { type: "Heartbeat"; sessionId: SessionId }
  | { type: "Query"; sessionId: SessionId; spec: QuerySpec }
  | { type: "QueryResult"; sessionId: SessionId; result: unknown }
  | { type: "RedactionPolicyUpdate"; rules: readonly RedactionRule[] }
  | { type: "Error"; code: string; message: string };

// DeltaOp — owner: @lrsi/serializer diff step; used by SnapshotDelta and by @lrsi/replay.
type DeltaOp =
  | { op: "add"; path: string; node: SnapshotNodeRecord }
  | { op: "replace"; path: string; node: SnapshotNodeRecord }
  | { op: "remove"; path: string };
// Subscription — owner: @lrsi/backend session; mutation: created/torn down by client request.
interface Subscription {
  readonly id: string;
  readonly sessionId: SessionId;
  readonly spec: QuerySpec;
  readonly active: boolean;
}

// RedactionRule — owner: @lrsi/redact; evaluated pre-transport (I5); serializable, versioned.
interface RedactionRule {
  readonly pathPattern: string;   // glob-like path matcher, see §8.2 path format
  readonly reason: "secret-heuristic" | "user-declared" | "policy";
}

// ScrubCursor — owner: @lrsi/replay; mutation: advanced only by seek operations.
interface ScrubCursor {
  readonly sessionId: SessionId;
  readonly keyframeId: string;
  readonly deltasApplied: number;
}
// QuerySpec — owner: @lrsi/query; compiled, not string-interpreted (P10); immutable once built.
interface QuerySpec {
  readonly select: "nodes" | "edges" | "events";
  readonly predicate: QueryPredicate;
  readonly project?: readonly string[];
}

type QueryPredicate =
  | { kind: "pathMatches"; pattern: string }
  | { kind: "typeIs"; typeTag: string }
  | { kind: "and"; of: readonly QueryPredicate[] }
  | { kind: "or"; of: readonly QueryPredicate[] };

// InstrumentationHooks — owner: framework adapter; installed/uninstalled per §8.2 contract.
interface InstrumentationHooks {
  readonly adapterId: string;
  install(ctx: unknown): void;     // idempotent (P6)
  uninstall(): void;               // must fully restore pre-attach state (I6)
  onMount(node: NodeId): void;
  onUpdate(node: NodeId, prevRef: NodeId | null): void;
  onUnmount(node: NodeId): void;
  onDependencyEdge(edge: DependencyEdge): void;
  onEvent(record: EventRecord): void;
}

// Error types — owner: shared; serializable across the wire boundary.
type LrsiErrorCode =
  | "ATTACH_TIMEOUT" | "PROTOCOL_MISMATCH" | "REDACTION_VIOLATION"
  | "RING_BUFFER_OVERFLOW" | "REPLAY_NONDETERMINISM" | "RESUME_TOKEN_INVALID";
interface LrsiError { readonly code: LrsiErrorCode; readonly message: string; }
