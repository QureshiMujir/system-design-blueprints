Core schema & types (§11)

// urt-ring — the ring substrate
#[repr(C)]
pub struct Sqe {                  // 64 bytes; must match struct io_uring_sqe for the target kernel
  pub opcode: u8, pub flags: u8, pub ioprio: u16, pub fd: i32,
  pub off: u64, pub addr: u64, pub len: u32, pub op_flags: u32,
  pub user_data: u64, pub buf_index: u16, pub personality: u16,
  pub splice_fd: i32, _pad: [u64; 2],
}
#[repr(C)]
pub struct Cqe { pub user_data: u64, pub res: i32, pub flags: u32 } // 16B (32B variant under SETUP_CQE32)

pub struct SubmissionQueue {      // owned by one reactor (single submitter); not Sync
  head: *const AtomicU32,         // kernel-written (we read Acquire)
  tail: *mut AtomicU32,           // we write Release  ← I7
  ring_mask: u32, ring_entries: u32,
  flags: *const AtomicU32,        // IORING_SQ_NEED_WAKEUP / CQ_OVERFLOW
  dropped: *const AtomicU32,
  array: *mut u32,                // ring slot → sqe index (indirection)
  sqes: *mut Sqe,                 // the SQE pool
}
pub struct CompletionQueue {      // owned by one reactor (single consumer)
  head: *mut AtomicU32,           // we write Release (free slots)
  tail: *const AtomicU32,         // kernel-written (we read Acquire)  ← I7
  ring_mask: u32, ring_entries: u32,
  overflow: *const AtomicU32,     // I9
  cqes: *const Cqe,
}
pub struct Uring {                // one per core (thread-per-core, ADR-008)
  fd: RawFd, sq: SubmissionQueue, cq: CompletionQueue, params: IoUringParams, features: Features,
}
impl Uring {
  pub fn setup(entries: u32, params: &mut IoUringParams) -> io::Result<Uring>; // io_uring_setup + mmap
  pub fn sq_space(&self) -> u32;                                               // capacity left (I6)
  pub fn try_fill(&mut self, sqe: Sqe) -> Result<(), Backpressure>;            // fill one SQE or signal SQ full (I6)
  pub fn submit(&mut self) -> io::Result<u32>;                                 // enter(submit, 0)
  pub fn submit_and_wait(&mut self, min_complete: u32) -> io::Result<u32>;     // enter(submit, min_complete, GETEVENTS)
  pub fn reap(&mut self, f: impl FnMut(&Cqe)) -> u32;                          // drain CQ; Acquire load (I7/I9)
  pub fn register(&self, op: RegisterOp) -> io::Result<()>;                    // io_uring_register (files/buffers/probe)
}
// urt-sys — raw layer (the only FFI; all unsafe wrapped here)
#[repr(C)] pub struct IoUringParams { /* sq_entries, cq_entries, flags, sq_thread_cpu, sq_thread_idle, features, sq_off, cq_off */ }
pub unsafe fn io_uring_setup(entries: u32, p: *mut IoUringParams) -> i32;       // SYS_io_uring_setup
pub unsafe fn io_uring_enter(fd: i32, to_submit: u32, min_complete: u32, flags: u32, sig: *const ()) -> i32;
pub unsafe fn io_uring_register(fd: i32, opcode: u32, arg: *const (), nr: u32) -> i32;
pub fn last_os_error() -> io::Error;                                            // errno → io::Error (§Phase1)

// urt-backend — swappable backend (ADR-003)
pub trait IoBackend: Send {
  fn submit(&mut self, sqes: &[Sqe]) -> io::Result<u32>;
  fn submit_and_wait(&mut self, sqes: &[Sqe], min_complete: u32) -> io::Result<u32>;
  fn reap(&mut self, sink: &mut dyn FnMut(Cqe)) -> u32;                         // identical Cqe semantics both backends (§12)
  fn register(&mut self, op: RegisterOp) -> io::Result<()>;
  fn caps(&self) -> Features;
}
pub struct UringBackend { uring: Uring }
pub struct EpollPoolBackend { epfd: RawFd, pool: WorkerPool, synth_cq: SynthCq } // emulation (§Phase 12)
// urt-types — op + buffer + id + features
pub struct OpId(pub u64);          // == Sqe.user_data; dense via slab key (I4)
#[non_exhaustive]
pub enum Op {                      // a submitted operation; OWNS its buffer/fd-ref until CQE (I2)
  Nop,
  Read   { fd: RawFd, buf: IoBufMut, off: u64 },
  Write  { fd: RawFd, buf: IoBuf,    off: u64 },
  Accept { fd: RawFd, addr: Box<SockAddrStorage> },
  Connect{ fd: RawFd, addr: Box<SockAddrStorage> },
  Send   { fd: RawFd, buf: IoBuf },
  Recv   { fd: RawFd, buf: IoBufMut },
  Fsync  { fd: RawFd, datasync: bool },
  Timeout{ ts: Box<KernelTimespec>, flags: u32 },
  PollAdd{ fd: RawFd, mask: u32 },
  Close  { fd: RawFd },
  OpenAt { dir: RawFd, path: CString, flags: i32, mode: u32 },
  Statx  { dir: RawFd, path: CString, out: Box<Statx> },
  Cancel { target: OpId },
  // … READ_FIXED/WRITE_FIXED/READV/WRITEV/SENDMSG/RECVMSG/PROVIDE_BUFFERS per §8.7
}
pub unsafe trait IoBuf  { fn stable_ptr(&self) -> *const u8; fn len(&self) -> usize; } // ptr STABLE while op in flight (I2)
pub unsafe trait IoBufMut { fn stable_mut_ptr(&mut self) -> *mut u8; fn cap(&self) -> usize; }
// safety: the buffer must not move/free while submitted; 'static or owned move enforces it.
pub struct Features(u64);          // bitset: NOP, READ, …, SQPOLL, BUF_RING, MULTISHOT, DEFER_TASKRUN (I12)
#[non_exhaustive]
pub enum UrtError { Setup(io::Error), Submit(io::Error), Op(io::Error), Cancelled, Backpressure, Unsupported(Feature) }

// urt-driver — in-flight tracking + the buffer-ownership contract
pub struct Inflight { slab: Slab<OpState> }                                    // key = OpId (I4); single-reactor owned
pub enum OpStatePhase { Submitted, InKernel, Cancelling, Completed }            // monotone (I5)
pub struct OpState {
  phase: OpStatePhase,
  waker: Option<Waker>,            // set on poll-Pending; woken on CQE (I11)
  result: Option<(i32, u32)>,      // (res, flags) filled by reactor
  buffer_owner: Option<Op>,        // THE pinned buffer/fd-ref; freed only at Completed (I2/I8)
  multishot: bool,                 // slot survives F_MORE CQEs (I3)
}
impl Inflight {
  pub fn submit(&mut self, op: Op) -> OpId;                                     // insert + return user_data
  pub fn on_cqe(&mut self, cqe: &Cqe) -> Option<Waker>;                         // store result, advance phase, return waker
  pub fn cancel(&mut self, id: OpId) -> Option<Sqe>;                            // → Cancelling, build ASYNC_CANCEL (I8)
}
// urt-exec — task / waker / reactor / runtime
pub struct Task { /* boxed future + state + JoinHandle sender */ }              // !Send by default (thread-per-core)
pub struct RawTask;                                                            // header + vtable (poll/drop/schedule)
pub fn make_waker(task: *const RawTask) -> Waker;                              // hand-built RawWakerVTable (I11)
pub struct RunQueue { /* VecDeque<TaskRef> + optional crossbeam steal deque */ }
pub struct Executor { run_queue: RunQueue, reactor: Reactor }
pub struct Reactor { backend: Box<dyn IoBackend>, inflight: Inflight }          // reaps CQE → wakes tasks
impl Reactor {
  pub fn turn(&mut self, block: bool) -> usize;                                 // submit pending + reap + wake; block via submit_and_wait
}
pub struct JoinHandle<T> { /* receives the task's output */ }
pub struct Runtime { executors: Vec<Executor> }                                // one per core (ADR-008)
impl Runtime {
  pub fn new(cfg: RuntimeConfig) -> io::Result<Runtime>;                        // probe + pick backend + setup rings
  pub fn block_on<F: Future>(&self, fut: F) -> F::Output;
  pub fn spawn<F: Future + 'static>(&self, fut: F) -> JoinHandle<F::Output>;
  pub fn spawn_blocking<R>(&self, f: impl FnOnce() -> R + Send + 'static) -> JoinHandle<R>;
}

// urt-fixed — registered resources (I14)
pub struct FixedBuffers { /* registered iovecs; index → (ptr,len) */ }          // pinned for op lifetime; deregister drains
pub struct FixedFiles   { /* registered fd table; index → fd */ }
pub struct BufRing      { /* buf_ring: mmap'd buffer ids for multishot recv (§9) */ }

// urt-time
pub struct Timer { wheel: TimingWheel }                                         // O(1) insert/expire (#204)
pub fn sleep(dur: Duration) -> Sleep;                                           // → TIMEOUT SQE
pub fn timeout<F: Future>(dur: Duration, f: F) -> Timeout<F>;                   // LINK_TIMEOUT or select

// urt-net / urt-fs (resource futures return (result, buffer) — §8.6)
impl TcpStream { pub async fn recv(&self, buf: IoBufMut) -> (io::Result<usize>, IoBufMut);
                 pub async fn send(&self, buf: IoBuf)    -> (io::Result<usize>, IoBuf); }
impl File      { pub async fn read_at(&self, buf: IoBufMut, off: u64) -> (io::Result<usize>, IoBufMut);
                 pub async fn write_at(&self, buf: IoBuf, off: u64)   -> (io::Result<usize>, IoBuf);
                 pub async fn fsync(&self) -> io::Result<()>; }
