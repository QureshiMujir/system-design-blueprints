Core schema & types (§11)

// secret.rs
pub struct SecretId(String);        // namespaced path; "/" separator; e.g. "database/prod/orders-app"
pub struct SecretVersion(u64);      // per-secret monotonic counter; starts at 1
pub enum StagingLabel {
    Current, Pending, Previous,
    Custom(String),                 // custom labels for blue/green or multi-version scenarios
}
pub enum VersionState {
    Pending,                        // created; not yet backend-confirmed
    Active,                         // CURRENT or PREVIOUS; readable
    Deprecated,                     // post-overlap; PREVIOUS before destruction
    Revoked,                        // terminal; consumers get 401
    Destroyed,                      // terminal; ciphertext and DEK deleted
}

// policy.rs
pub struct RotationPolicy {
    trigger:        RotationTrigger,
    ttl:            Duration,           // default lease TTL
    max_ttl:        Duration,           // max renewable TTL
    overlap_window: Duration,           // dual-validity window after promote
    max_versions:   u32,                // versions to retain (default 3)
    engine:         EngineRef,          // "static" | "db-cred" | "api-key" | etc.
    on_failure:     FailureAction,
}
pub enum RotationTrigger { Interval(Duration), Cron(String), OnDemand, OnEvent(EventKind) }
pub enum FailureAction { Rollback, Halt, Retry { max: u32, backoff: Duration } }

// rotation_state.rs
pub enum RotationState {
    Idle, CreatePending, SetInBackend, TestPending, Promote,
    RetirePrevious, RevokeOld, Failed(FailureReason), RolledBack,
}
pub enum FailureReason { EngineError(String), Timeout, LockLost, StorageError }
// lease.rs
pub struct Lease {
    id:          LeaseId,           // UUID v4
    secret_id:   SecretId,
    version:     SecretVersion,     // which version this lease is for
    ttl:         Duration,
    renewable:   bool,
    issued_at:   Timestamp,
    expires_at:  Timestamp,
    consumer:    Identity,
}

// capability.rs
pub struct Capability { path_glob: String, ops: OpSet }  // OpSet = bitflags
bitflags! { pub struct OpSet: u8 { Read=0x01, Write=0x02, Rotate=0x04, Lease=0x08, Admin=0x10 } }

// error.rs
pub enum SrError {
    Sealed, Unauthorized, NotFound(SecretId), VersionConflict,
    AlreadyRotating, EngineError(String), BarrierError, AuditError,
    RateLimited { retry_after: Duration }, InsufficientShares,
    InvalidIdempotencyKey, StorageError(String),
}
// aead.rs
pub trait Aead: Send + Sync {
    fn seal(&self, nonce: &Nonce, aad: &[u8], pt: &[u8]) -> Result<Ciphertext, CryptoError>;
    fn open(&self, nonce: &Nonce, aad: &[u8], ct: &[u8]) -> Result<Zeroizing<Vec<u8>>, CryptoError>;
    fn algorithm(&self) -> AeadAlgorithm;
}
pub enum AeadAlgorithm { Aes256Gcm, ChaCha20Poly1305 }

// keys.rs
pub struct Dek(Zeroizing<[u8; 32]>);    // per-version data-encryption key
pub struct Kek(Zeroizing<[u8; 32]>);    // store-epoch key-encryption key
pub struct RootKey(Zeroizing<[u8; 32]>);// assembled from Shamir shares; never persisted plaintext

// shamir.rs
pub fn split(secret: &[u8], threshold: u8, n: u8) -> Result<Vec<UnsealKeyShare>, CryptoError>;
pub fn combine(shares: &[UnsealKeyShare]) -> Result<Zeroizing<Vec<u8>>, CryptoError>;
pub struct UnsealKeyShare { pub index: u8, pub share: Zeroizing<Vec<u8>> }

// hkdf.rs
pub fn derive_kek(root: &RootKey, context: &[u8]) -> Result<Kek, CryptoError>;
pub fn derive_dek(kek: &Kek, version_context: &[u8]) -> Result<Dek, CryptoError>;  // optional per-version

// constant_time.rs
pub fn constant_time_eq(a: &[u8], b: &[u8]) -> bool;  // always same-length comparison; subtle crate
// barrier.rs
pub struct Barrier {                      // owns active Kek; seals on panic via panic hook
    state:  SealState,
    kek:    Option<Kek>,
    aead:   Box<dyn Aead>,
}
impl Barrier {
    pub fn unseal(&mut self, root: &RootKey) -> Result<(), BarrierError>;
    pub fn seal(&mut self);
    pub fn encrypt(&self, pt: &[u8]) -> Result<EncryptedBlob, BarrierError>;
    pub fn decrypt(&self, blob: &EncryptedBlob) -> Result<Zeroizing<Vec<u8>>, BarrierError>;
    pub fn wrap_dek(&self, dek: &Dek) -> Result<Vec<u8>, BarrierError>;
    pub fn unwrap_dek(&self, wrapped: &[u8]) -> Result<Dek, BarrierError>;
}

// records.rs
pub struct SecretRecord { id: SecretId, current: SecretVersion, policy: RotationPolicy,
    versions: BTreeMap<SecretVersion, VersionRecord> }
pub struct VersionRecord { version: SecretVersion, state: VersionState,
    labels: Vec<StagingLabel>, wrapped_dek: Vec<u8>,    // DEK wrapped by active KEK
    ciphertext_ref: BlobRef, checksum: [u8; 32],        // SHA-256 of plaintext before encryption
    created_at: Timestamp, rotated_at: Option<Timestamp> }
pub struct BlobRef(String);              // path or key in the blob store
pub struct EncryptedBlob { nonce: [u8; 12], ciphertext: Vec<u8> }

// store.rs
pub trait SecretStore: Send + Sync {
    fn get(&self, id: &SecretId, label: StagingLabel) -> Result<Zeroizing<Vec<u8>>, SrError>;
    fn put(&self, id: &SecretId, plaintext: &[u8], policy: RotationPolicy) -> Result<SecretVersion, SrError>;
    fn atomic_version_commit(&self, id: &SecretId, promote: SecretVersion) -> Result<(), SrError>;
    fn set_version_state(&self, id: &SecretId, v: SecretVersion, state: VersionState) -> Result<(), SrError>;
    fn list_versions(&self, id: &SecretId) -> Result<Vec<VersionRecord>, SrError>;
}
// engine.rs
pub struct RotationCtx { secret_id: SecretId, version: SecretVersion, idempotency_key: Uuid, config: EngineConfig }
pub struct PendingSecret { credential: Zeroizing<Vec<u8>>, metadata: HashMap<String, String> }

pub trait RotationEngine: Send + Sync {
    fn name(&self) -> &'static str;
    fn create_pending(&self, ctx: &RotationCtx) -> Result<PendingSecret, EngineError>;   // step 1
    fn set_in_backend(&self, ctx: &RotationCtx, p: &PendingSecret) -> Result<(), EngineError>; // step 2
    fn test_pending(&self, ctx: &RotationCtx, p: &PendingSecret) -> Result<(), EngineError>;   // step 3
    fn promote(&self, ctx: &RotationCtx, p: &PendingSecret) -> Result<(), EngineError>;        // step 4
    fn retire(&self, ctx: &RotationCtx, prev: SecretVersion) -> Result<(), EngineError>;
    fn revoke(&self, ctx: &RotationCtx, v: SecretVersion) -> Result<(), EngineError>;
    fn rollback(&self, ctx: &RotationCtx, p: &PendingSecret) -> Result<(), EngineError>;
}
// record.rs
pub struct AuditRecord {
    seq:       u64,
    prev_hash: [u8; 32],
    hash:      [u8; 32],          // SHA-256(prev_hash ∥ canonical_encode(fields))
    ts:        Timestamp,
    actor:     Identity,
    action:    Action,
    target:    SecretId,
    version:   Option<SecretVersion>,
    outcome:   Outcome,
    trace_id:  TraceId,
}
pub enum Action { Get, Put, Rotate, Promote, Retire, RevokeVersion, RevokeLease, Unseal, Seal, PolicyChange }
pub enum Outcome { Success, Denied { reason: String }, Error { code: String } }
pub fn canonical_encode(r: &AuditRecord) -> Vec<u8>;  // deterministic; seq + ts + actor + action + target + outcome
// routes/data.rs (representative signatures)
pub async fn get_secret(
    State(app): State<AppState>, Path((id, label)): Path<(String, String)>,
    auth: AuthToken,
) -> Result<Json<SecretResponse>, ApiError>;

pub async fn rotate_now(
    State(app): State<AppState>, Path(id): Path<String>,
    auth: AuthToken, idempotency: IdempotencyKey,
) -> Result<Json<RotateResponse>, ApiError>;

pub async fn grant_lease(State(app): State<AppState>, ...) -> Result<Json<LeaseResponse>, ApiError>;
pub async fn revoke_tree(State(app): State<AppState>, Path(prefix): Path<String>, ...) -> Result<Json<RevokeResponse>, ApiError>;
// cache.rs
pub struct SecretCache {                  // backed by memfd + mlock
    fd:     RawFd,
    ptr:    *mut u8,                      // mmap of fd; mlock'd
    len:    usize,
    expiry: Instant,                      // lease expiry; fail-closed after grace
}
impl SecretCache {
    pub fn read(&self) -> Result<&[u8], AgentError>;         // returns slice into locked memory
    pub fn update(&mut self, data: &[u8], expiry: Instant) -> Result<(), AgentError>;
    pub fn is_valid(&self) -> bool;                           // expiry check
}
