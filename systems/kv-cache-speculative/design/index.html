Core schema & types (§11)

BlockId        = NewType("BlockId", int)          # physical, 0..N_BLOCKS-1
LogicalBlockId = NewType("LogicalBlockId", int)   # logical, 0..n_blocks_in_seq-1
BLOCK_SIZE_DEFAULT: int = 16
BLOCK_SIZE_LARGE:   int = 32
NULL_BLOCK: BlockId = BlockId(-1)

@dataclass(frozen=True, slots=True)
class PhysicalBlock:
    id:           BlockId
    ref_count:    int
    is_full:      bool
    content_hash: int | None
@dataclass(slots=True)
class BlockTable:
    physical: list[BlockId]
    last_block_offset: int

    def append(self, block: BlockId) -> None: ...
    def slot_of(self, position: int, block_size: int) -> tuple[BlockId, int]: ...
    def num_blocks(self) -> int: ...
    def num_tokens(self, block_size: int) -> int: ...
SequenceId = NewType("SequenceId", int)

class SequenceStatus(Enum):
    Waiting = auto(); Running = auto(); Swapped = auto()
    Finished = auto(); Aborted = auto()

@dataclass(slots=True)
class Sequence:
    id: SequenceId; request_id: str
    prompt_tokens: list[int]; output_tokens: list[int]
    block_table: BlockTable; status: SequenceStatus
    sampling: "SamplingParams"; speculative: "SpeculativeParams | None"
    parent_seq: SequenceId | None
    arrival_ns: int; priority: int = 0
    n_accepted: int = 0; n_proposed: int = 0

    def num_tokens(self) -> int: ...
    def is_prefill(self) -> bool: ...
    def acceptance_rate(self) -> float: ...
@dataclass(frozen=True, slots=True)
class SamplingParams:
    n: int = 1; max_tokens: int = 256; temperature: float = 1.0
    top_p: float = 1.0; top_k: int = -1
    presence_penalty: float = 0.0; frequency_penalty: float = 0.0
    repetition_penalty: float = 1.0
    stop: tuple[str, ...] = (); seed: int | None = None
    logit_bias: dict[int, float] | None = None

@dataclass(frozen=True, slots=True)
class SpeculativeParams:
    method: Literal["off", "draft", "medusa", "lookahead"]
    num_speculative_tokens: int = 5
    draft_model_id: str | None = None
    medusa_head_path: str | None = None
    lookahead_n_grams: int = 4
@dataclass(slots=True)
class BatchPlan:
    sequence_ids: list[SequenceId]
    input_tokens: torch.Tensor; input_positions: torch.Tensor
    seq_lens: torch.Tensor; cu_seq_lens: torch.Tensor
    block_tables: torch.Tensor; slot_mapping: torch.Tensor
    is_prefill: torch.Tensor
    speculative_lens: torch.Tensor | None
    sampling: "BatchSamplingParams"

@dataclass(slots=True)
class BatchOutput:
    next_tokens: torch.Tensor
    logprobs: torch.Tensor | None
    acceptance: torch.Tensor | None
@dataclass(slots=True)
class SpeculativePlan:
    method: Literal["draft", "medusa", "lookahead"]
    proposed_tokens: torch.Tensor        # [num_seq, K]
    proposed_probs: torch.Tensor         # [num_seq, K]
    spec_block_ids: list[list[BlockId]]

@dataclass(slots=True)
class AcceptanceResult:
    num_accepted: torch.Tensor           # [num_seq], in [0, K+1] (I10)
    accepted_tokens: torch.Tensor        # [num_seq, K+1]
    resampled_token: torch.Tensor        # [num_seq]
class KvCacheManager:
    block_size: int; num_gpu_blocks: int; num_cpu_blocks: int

    def can_allocate(self, num_blocks: int) -> bool: ...       # I8
    def allocate(self, seq: Sequence, num_blocks: int) -> None: ...
    def free(self, seq: Sequence) -> None: ...
    def share_blocks(self, src: Sequence, dst: Sequence, n: int) -> None: ...  # P2
    def cow_block(self, seq: Sequence, lb: LogicalBlockId) -> BlockId: ...     # I6
    def lookup_prefix(self, hashes: list[int]) -> list[BlockId]: ...          # §16.3
    def commit_prefix(self, seq: Sequence) -> None: ...
    def swap_out(self, seq: Sequence) -> None: ...                            # P8
    def swap_in(self, seq: Sequence) -> None: ...
@dataclass(slots=True)
class SchedulerState:
    waiting: list[Sequence]; running: list[Sequence]
    swapped: list[Sequence]; finished: list[Sequence]
    aborted: list[Sequence]; step_id: int = 0

class Scheduler:
    cache: KvCacheManager; policy: "SchedulingPolicy"

    def schedule_step(self) -> BatchPlan: ...
    def commit_step(self, plan: BatchPlan, output: BatchOutput) -> None: ...
    def admit(self, seq: Sequence) -> None: ...
    def abort(self, seq_id: SequenceId) -> None: ...
class MedusaHead(nn.Module):
    num_layers: int; hidden_size: int; vocab_size: int
    def forward(self, hidden_state: torch.Tensor) -> torch.Tensor: ...  # [batch, vocab]

@dataclass(slots=True)
class MedusaConfig:
    num_heads: int = 4; head_layers: int = 1
    tree_topk: tuple[int, ...] = (10, 10, 8, 4)
    tree_depth: int = 4
