Core schema & types (§11)

struct Superblock {                  // 0x204 bytes, block 0. Persistent, CRC-checked.
  magic: u32, format_version: u16, feature_compat/incompat/ro_compat: u16/u32/u32,
  block_size_log2: u8, blocks_total/free: u64, inodes_total/free: u64,
  block_groups: u32, blocks_per_group/inodes_per_group: u32,
  journal_first_block/journal_block_count: u64, journal_seq: u64,
  root_inode: u64, mount_count/mount_count_max: u32,
  last_mount/write/check_time: i64 (ns), state: u16, errors_behaviour: u16,
  uuid: [u8;16], label: [u8;16], creator_os: u32, _reserved: [u8;0x158],
  checksum: u32,
}                                     // Mutation: mkfs (write-once fields) + mount
                                      // (mount_count, times, journal_seq) via journal.
struct Inode {                       // 0x100 bytes (16/4KiB block). Persistent, CRC-checked.
  mode: u16, uid/gid: u32, link_count: u16, flags: u32,
  size_bytes/blocks_used: u64, atime/mtime/ctime/crtime_ns: i64,
  generation: u32, xattr_block: u64,
  extent_root: ExtentHeader, extents_inline: [Extent; 4],
  _reserved: [u8;0x60], checksum: u32,
}                                     // Mutation: any VFS op via Transaction (I1).
                                      // Concurrency: guarded by per-inode RwLock (I8).
struct ExtentHeader { magic: u16, entries: u16, max: u16, depth: u16 }   // 8 B
struct Extent { logical_block: u32, length: u16 (hi bit=unwritten),
                physical_hi: u16, physical_lo: u32 }                     // 12 B
struct ExtentIndex { logical_block: u32, child_block_lo: u32,
                     child_block_hi: u16, _unused: u16 }                 // 12 B
struct DirEntryHeader { inode: u64, rec_len: u16, name_len: u8,
                        file_type: u8 }                                  // 12 B + name
struct JournalDescriptor { magic: u32, sequence: u64, block_count: u32,
                           mode: u8, checksum: u32 }   // + block_count × JournalBlockTag
struct JournalBlockTag { target_block: u64, flags: u32 }
struct JournalCommit { magic: u32, sequence: u64, timestamp_ns: i64, checksum: u32 }
enum Error {                          // cordfs-core; P12 — typed, no strings.
  Corrupt(&'static str), Io(std::io::Error), NoSpace, ReadOnly, JournalFull,
  Invalidated(u64 /*ino*/), NameTooLong(usize), NotADirectory, IsADirectory,
  NotEmpty, SymlinkLoop, PhaseDeferred(&'static str),
}                                     // Serialization: none — process-local only.

trait BlockDevice: Send + Sync {      // cordfs-block. Ownership: one per mount.
  fn read_block(&self, n: u64, buf: &mut [u8]) -> Result<()>;
  fn write_block(&self, n: u64, buf: &[u8]) -> Result<()>;
  fn flush(&self) -> Result<()>;      // fdatasync
  fn size_in_blocks(&self) -> u64;
}                                     // Concurrency: internally synchronized (Phase 1).

trait Journal: Send + Sync {          // cordfs-journal. Ownership: one per mount.
  fn begin(&self) -> Result<Transaction>;
  fn dirty(&self, txn: &mut Transaction, block: u64, after: &[u8]) -> Result<()>;
  fn commit(&self, txn: Transaction) -> Result<()>;   // I1, I6
  fn checkpoint(&self) -> Result<()>;
  fn replay(&mut self) -> Result<u64>; // returns last committed seq (I6, I11)
}

trait Filesystem: Send + Sync {       // cordfs-core, mounted by cordfs-fuse (§19).
  fn lookup(&self, parent: Ino, name: &OsStr) -> Result<Attr>;
  fn getattr(&self, ino: Ino) -> Result<Attr>;
  fn setattr(&self, ino: Ino, set: AttrSet) -> Result<Attr>;
  fn readlink(&self, ino: Ino) -> Result<OsString>;
  fn open(&self, ino: Ino, flags: i32) -> Result<Fh>;
  fn read(&self, fh: Fh, off: u64, size: u32) -> Result<Vec<u8>>;
  fn write(&self, fh: Fh, off: u64, data: &[u8]) -> Result<u32>;
  fn flush(&self, fh: Fh) -> Result<()>;
  fn release(&self, fh: Fh) -> Result<()>;
  fn fsync(&self, fh: Fh, datasync: bool) -> Result<()>;
  fn opendir(&self, ino: Ino, flags: i32) -> Result<Fh>;
  fn readdir(&self, fh: Fh, off: u64) -> Result<Vec<DirEntry>>;
  fn mkdir(&self, parent: Ino, name: &OsStr, mode: u32) -> Result<Attr>;
  fn rmdir(&self, parent: Ino, name: &OsStr) -> Result<()>;
  fn create(&self, parent: Ino, name: &OsStr, mode: u32, flags: i32) -> Result<(Attr, Fh)>;
  fn unlink(&self, parent: Ino, name: &OsStr) -> Result<()>;
  fn link(&self, ino: Ino, new_parent: Ino, new_name: &OsStr) -> Result<Attr>;
  fn symlink(&self, parent: Ino, name: &OsStr, target: &OsStr) -> Result<Attr>;
  fn rename(&self, op: Ino, on: &OsStr, np: Ino, nn: &OsStr) -> Result<()>;
  fn statfs(&self, ino: Ino) -> Result<StatFs>;
  fn setxattr(&self, ino: Ino, name: &OsStr, v: &[u8], flags: i32) -> Result<()>;
  fn getxattr(&self, ino: Ino, name: &OsStr) -> Result<Vec<u8>>;
  fn access(&self, ino: Ino, mask: i32) -> Result<()>;
}                                      // Concurrency: each method acquires the
                                       // relevant inode lock(s) internally (I8).
