Core schema & types (§11)

struct EventEnvelope { event_id, stream_id, stream_version, global_offset,
                       partition, event_type, schema_version, payload, metadata, recorded_at }
struct EventMetadata { correlation_id, causation_id, actor, source, content_type, trace_context, custom }
struct StreamId(String);  // "<category>-<instance>"
fn category(&self) -> &str; fn partition(&self, p_count: u16) -> u16;
enum ExpectedVersion { Any, NoStream, StreamExists, Exact(u64) }
struct AppendResult { stream_version: u64, first_offset: u64, last_offset: u64, partition: u16 }
trait Aggregate {
  type Event; type Command; type Error;
  fn empty() -> Self;                               // seed state
  fn apply(&mut self, e: &Self::Event);             // pure, total fold step
  fn handle(&self, c: Self::Command)                // pure decision
        -> Result<Vec<Self::Event>, Self::Error>;
  fn category() -> &'static str;
}
trait Command { fn target_stream(&self) -> StreamId; fn idempotency_key(&self) -> Option<Uuid>; }
trait EventStore {
  fn append(&self, stream: &StreamId, expected: ExpectedVersion, events: &[RawEvent])
        -> Result<AppendResult, AppendError>;
  fn read_stream(&self, stream: &StreamId, from: u64, max: usize)
        -> Result<StreamSlice, ReadError>;
  fn read_all(&self, partition: u16, from_offset: u64, max: usize)
        -> Result<LogSlice, ReadError>;
  fn head(&self, stream: &StreamId) -> Result<Option<u64>, ReadError>;
}
struct Snapshot { stream_id, version: u64, schema_version: u32, state_blob: Bytes, created_at }
trait SnapshotStore {
  fn save(&self, s: &Snapshot) -> Result<(), SnapshotError>;
  fn load_latest(&self, stream: &StreamId, at_most: u64) -> Result<Option<Snapshot>, SnapshotError>;
}
trait Projection {
  type Store;
  fn name(&self) -> &str;
  fn interested_in(&self) -> EventFilter;            // categories/types
  fn apply(&self, store: &mut Self::Store, e: &EventEnvelope, ckpt: &mut Checkpoint)
        -> Result<(), ProjectionError>;              // idempotent; advances ckpt atomically
}
struct Checkpoint { projection_name: String, partition: u16, offset: u64 }
trait Subscription {
  fn start(&self, from: SubscriptionStart, sink: Sink) -> SubscriptionHandle;  // CatchUp→Live
}
enum SubscriptionStart { Beginning, FromCheckpoint(Checkpoint), Live }
trait Saga {
  type State; type Event; type Command;
  fn empty() -> Self::State;
  fn apply(&self, s: &mut Self::State, e: &Self::Event);                 // fold saga state
  fn react(&self, s: &Self::State, e: &EventEnvelope)
        -> Vec<SagaAction>;   // emit command | set timer | compensate | finish
}
enum SagaAction { Emit(CommandRef), SetTimer(TimerId, Duration), Compensate(CommandRef), Finish }
trait Upcaster { fn matches(&self) -> (EventType, SchemaVer); fn target(&self) -> SchemaVer;
                 fn upcast(&self, raw: RawEvent) -> Result<RawEvent, UpcastError>; }  // pure,total
enum AppendError   { VersionConflict{expected,actual}, EmptyBatch, StreamClosed, Io(IoErr), Codec(CodecErr) }
enum ReadError     { NotFound, CorruptRecord{offset}, Io(IoErr), Codec(CodecErr) }
enum SnapshotError { AheadOfStream, Corrupt, Io(IoErr) }
enum ProjectionError { Decode(CodecErr), Store(StoreErr), CheckpointRegress, Poison{offset} }
enum UpcastError   { MissingUpcaster{ty,ver}, BadTransform }
enum RecoverError  { TornTail{offset}, CrcMismatch{offset}, SegmentMissing }
