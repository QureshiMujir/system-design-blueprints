Core schema & types (§11)

enum Opcode { Continuation = 0x0, Text = 0x1, Binary = 0x2, Close = 0x8, Ping = 0x9, Pong = 0xA }
// Copy, Eq. RFC 6455 §5.2 wire values. No behavior; pure tag.

struct FrameHeader {
    fin: bool,
    rsv: u8,                 // 3 bits, RFC 7692 owns rsv1 post-negotiation (I4)
    opcode: Opcode,
    masked: bool,             // MUST be true inbound (I2), false outbound (I3)
    payload_len: u64,         // post-decode of 7/16/64-bit field
    mask_key: Option<[u8; 4]>,
}
// Copy. No heap allocation. Constructed only by frame::parser.

enum ParseOutcome<'a> {
    Incomplete { needed: usize },
    Frame(FrameHeader, &'a [u8]),   // header + borrowed payload slice, zero-copy
    ProtocolError(WsError),
}
// Lifetime tied to the read buffer; parser never owns payload bytes.

enum WsError {
    InvalidHandshake(String),
    UnmaskedClientFrame,          // → Close 1002 (I2)
    ReservedBitsSet,                // → Close 1002 (I4)
    FragmentationViolation,         // → Close 1002 (I5, I8)
    ControlFrameTooLarge,           // → Close 1002 (I6)
    InvalidCloseCode(u16),           // → Close 1002 (I7)
    OversizedMessage { max: u32, got: u64 },  // → Close 1009 (I9)
    InvalidUtf8,                      // → Close 1007 (I12)
    ExtensionNegotiationFailed(String), // → Close 1010
    InternalError(String),             // → Close 1011
}
// One variant per close-code mapping (P14); exhaustive match required at call sites.
struct HandshakeRequest { key: String, version: u8, extensions: Vec<String> }
// `subprotocols: Vec<String>` field intentionally omitted — PhaseDeferred("§13").

struct HandshakeResponse { accept: String, selected_extension: Option<ExtensionParams> }

enum ConnectionState { Connecting, Upgrading, Open, ClosingLocal, ClosingRemote, Closed }
// Exhaustive; every transition table in §8/§17 maps 1:1 to a match arm.

enum CloseCode {
    Normal = 1000, GoingAway = 1001, ProtocolError = 1002, Unsupported = 1003,
    NoStatus = 1005, Abnormal = 1006, InvalidPayload = 1007, PolicyViolation = 1008,
    MessageTooBig = 1009, ExtensionRequired = 1010, InternalError = 1011,
}
// RFC 6455 §7.4.1 registry subset actually emitted by this implementation.

struct Reassembler { active_opcode: Option<Opcode>, buffer: BytesMut, max_message_size: u32 }
// Owns no socket. `feed()` enforces I5/I8/I9; `finalize()` enforces I12.

struct ExtensionParams {
    server_no_context_takeover: bool,
    client_no_context_takeover: bool,
    server_max_window_bits: u8,
    client_max_window_bits: u8,
}
// RFC 7692 §7.1 negotiated parameters; None means extension not active.

trait FrameSink {
    fn send_frame(&mut self, header: FrameHeader, payload: &[u8]) -> Result<(), WsError>;
}
// Implemented by the writer task only (I11); no other code path may implement it.
