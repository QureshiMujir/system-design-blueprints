Core schema & types (§11)

pub struct PartitionKey(pub Vec<u8>);
impl PartitionKey {
    pub fn new(bytes: impl Into<Vec<u8>>) -> Self;
    pub fn as_bytes(&self) -> &[u8];
}

pub fn encode_composite(tenant_id: &[u8], row_key: &[u8]) -> Result<PartitionKey, KeyError>;

#[derive(Copy, Clone, PartialEq, Eq, PartialOrd, Ord, Hash, Debug)]
pub struct ShardId(pub u32);
pub struct Range { pub start: Bound<PartitionKey>, pub end: Bound<PartitionKey> }
impl Range {
    pub fn unbounded() -> Self;
    pub fn contains(&self, key: &PartitionKey) -> bool;
}

pub enum PartitionMap {
    Range(BTreeMap<PartitionKey, ShardId>),
    Hash { buckets: Vec<ShardId>, hash_fn: HashFn },
    ConsistentHash(ConsistentHashRing),
    JumpHash { buckets: Vec<ShardId> },
    Rendezvous { shards: Vec<ShardId>, hash_fn: HashFn },
}

#[derive(Copy, Clone)]
pub enum HashFn { Xxh3, SipHash24 }
pub struct ConsistentHashRing {
    tokens: Vec<(u64, ShardId)>,          // sorted by token; I9 enforced in `new`
    pub vnode_count: u32,                 // >= 100, default 256 (ADR-06)
    pub bounded_loads_epsilon: f64,       // 0.0 disables; default 0.25 (ADR-23)
}
impl ConsistentHashRing {
    pub fn new(shards: &[ShardId], vnode_count: u32) -> Result<Self, RingError>;
    pub fn route(&self, key: &PartitionKey) -> ShardId;
    pub fn route_bounded(&self, key: &PartitionKey, loads: &[u64]) -> ShardId;
}

pub struct ShardServer {
    pub id: ShardId,
    pub addr: String,
    pub replicas: Vec<String>,
    pub capacity_bytes: u64,
    pub load_bytes: u64,
    pub load_qps: u32,
    pub region: Option<String>,           // geo-aware placeholder, PhaseDeferred("§22")
}
pub trait Vindex: Send + Sync {
    fn lookup(&self, index_value: &[u8]) -> Result<PartitionKey, VindexError>;
    fn reverse(&self, key: &PartitionKey) -> Result<Vec<u8>, VindexError>;
    fn is_unique(&self) -> bool;
}
pub enum ResharingMode { Split, Merge, Move }

pub struct ResharingPlan {
    pub job_id: uuid::Uuid,
    pub source: ShardId,
    pub target: ShardId,
    pub range: Range,
    pub mode: ResharingMode,
    pub rate_limit_bytes_per_sec: u64,    // ADR-12 default 10 MB/s
    pub created_at: chrono::DateTime<chrono::Utc>,
    pub state: JobState,
}

pub enum JobState {
    Pending,
    Scanning { rows_copied: u64, rows_total: u64 },
    Validating,
    Cutover { cutover_token: u64 },
    Completed,
    Aborted { reason: String },
}
pub trait Router: Send + Sync {
    fn route(&self, key: &PartitionKey) -> Result<ShardId, RoutingError>;
    fn route_range(&self, range: &Range) -> Result<Vec<ShardId>, RoutingError>;
    fn version(&self) -> u64;
    fn refresh(&self) -> Result<(), RoutingError>;
}

#[async_trait::async_trait]
pub trait MetadataStore: Send + Sync {
    async fn get_partition_map(&self) -> Result<VersionedPartitionMap, MetaError>;
    async fn put_partition_map(&self, expected_version: u64, new_map: PartitionMap)
        -> Result<u64, MetaError>;                 // CAS (I4); returns new version
    async fn watch_partition_map(&self) -> Result<Watcher, MetaError>;
    async fn get_shard(&self, sid: ShardId) -> Result<ShardServer, MetaError>;
    async fn list_shards(&self) -> Result<Vec<ShardServer>, MetaError>;
    async fn acquire_lease(&self, sid: ShardId, holder: &str, ttl_seconds: u32)
        -> Result<Lease, MetaError>;
    async fn renew_lease(&self, lease: &Lease) -> Result<(), MetaError>;
}

pub struct VersionedPartitionMap { pub version: u64, pub map: PartitionMap }
pub struct Lease { pub sid: ShardId, pub holder: String, pub fencing_token: u64, pub expires_at: u64 }
message RouteRequest  { bytes key = 1; uint64 client_map_version = 2; }
message RouteResponse {
  oneof result {
    uint32 shard_id = 1;
    KeyNotFound not_found = 2;
  }
}
message KeyNotFound { uint64 current_version = 1; }
pub struct TokenBucket {
    rate_bytes_per_sec: u64,
    capacity_bytes: u64,
    available_bytes: AtomicU64,
    last_refill: AtomicU64,               // nanoseconds since epoch (ADR-33: monotonic clock)
}
