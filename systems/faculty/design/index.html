Core schema & types (§11)

export interface FacultyManifest {
  readonly schemaVersion: '1.x';                 // supported: current + previous major (P9)
  readonly id: FacultyId;                        // globally unique slug, e.g. "munger-multidisciplinary"
  readonly version: string;                      // semver, bumped on every publish
  readonly displayName: string;
  readonly author: AuthorIdentity;               // verified via Stripe Connect KYC (§20)
  readonly provenance: Provenance;                // see 11.1.1
  readonly pattern: CognitivePattern;              // see 11.1.1
  readonly behaviour: BehaviourSpec;                // see 11.1.1
  readonly corpus: CorpusSpec;                       // see 11.1.1
  readonly evaluation: EvaluationSpec;                // see 11.1.1
  readonly pricing: PricingSpec;                       // see 11.1.1
  readonly llm: LlmPreference;                          // see 11.1.1
}
interface Provenance {
  inspiredBy?: string;
  rightsBasis: 'public_domain' | 'estate_licensed' | 'original_work' | 'corporate_owned';
  licenseRef?: string;                            // pointer to executed licence on file
}
interface CognitivePattern {
  summary: string;                                // one normative-claim sentence
  primitives: CognitivePrimitive[];                // e.g. "invert", "first_principles_decomposition"
  domains: Domain[];
}
interface BehaviourSpec {
  systemPrompt: string;
  rules: BehaviouralRule[];                       // structured constraints, §8, ADR-20
  counterQuestions: string[];
  refusalPolicies: RefusalPolicy[];
}
interface CorpusSpec {
  sources: CorpusSource[];                        // signed pointers into object storage
  chunkingStrategy: 'semantic' | 'structural' | 'hybrid';   // ADR-22
  embeddingModel: EmbeddingModelRef;                // ADR-17
}
interface EvaluationSpec {
  suiteRef: EvaluationSuiteRef;
  minScore: number;                               // 0..1, gates listing (I4)
  lastRun?: EvaluationResult;                       // 11.4
}
interface PricingSpec {
  monthlyRentalUsd: number;                        // 0 = free, else $5..$50 band
  trialDays?: number;
  enterpriseUpsellAvailable?: boolean;
}
interface LlmPreference {
  preferred: 'claude-sonnet' | 'claude-opus' | 'gpt-4o' | 'llama-3-70b';   // ADR-07
  contextWindowMin: number;
}
type MemoryEntry = {
  id: string;
  userId: string;                                 // or roomId for Room-scoped memory (§23)
  facultyId: FacultyId;
  facultyVersion: string;                          // memory bound to the writing Manifest version
  kind: 'interaction' | 'summary' | 'anchor' | 'preference';
  createdAt: ISOString;
  expiresAt?: ISOString;                            // user-configurable retention
  content: {
    text: string;
    embedding: Float32Array;                       // dims per manifest.corpus.embeddingModel
    salience: number;                               // 0..1 importance score
    sourceSessionId: string;
  };
  flags: { sensitive: boolean; userPinned: boolean; redacted: boolean };
  vaultScope: 'user_only' | 'user_and_platform_audit' | 'user_and_faculty_aggregate';
  __schemaVersion: number;                          // I10
};
type Session = {
  id: string;
  userId: string;
  facultyIds: FacultyId[];                          // length > 1 = stacked session (§22)
  scope: { surface: 'web_app' | 'extension'; domain?: string; taskHint?: string };
  startedAt: ISOString;
  endedAt?: ISOString;
  wearingSeconds: number;                            // HW/WAU accumulator
  events: SessionEvent[];                             // append-only, idempotent (P8)
  __schemaVersion: number;
};

type Room = Session & {
  roomId: string;
  ownerId: string;
  participantIds: string[];                            // ≤ 200, §4
  transcriptCipherRef: string;                            // E2EE ciphertext pointer (§23)
};
type StackManifest = { facultyIds: FacultyId[]; reconcilerPolicy: 'surface_disagreement' };  // I11 caps length ≤ 4

type EvaluationResult = {
  suiteRef: EvaluationSuiteRef;
  score: number;                                      // 0..1
  perCriterion: Record<CognitivePrimitive, number>;
  runAt: ISOString;
  judgeModel: string;                                  // meta-LLM judge identity (§18)
};

type RoyaltyLedgerEntry = {
  id: string;
  facultyId: FacultyId;
  periodStart: ISOString;
  periodEnd: ISOString;
  grossRevenueCents: number;
  authorShareCents: number;                             // 70% of net revenue (§20, ADR-21)
  platformShareCents: number;
  stripeTransferId?: string;
  reconciledAt?: ISOString;                              // null until I5 check passes
};
type FacultyError =
  | { code: 'MANIFEST_SIGNATURE_INVALID'; facultyId: FacultyId }
  | { code: 'MANIFEST_UNKNOWN_FIELD'; field: string }
  | { code: 'MANIFEST_MISSING_REQUIRED'; field: string }
  | { code: 'EVAL_BELOW_THRESHOLD'; score: number; minScore: number }
  | { code: 'PARTITION_VIOLATION'; attemptedKey: string }
  | { code: 'STACK_SIZE_EXCEEDED'; requested: number }
  | { code: 'ROYALTY_RECONCILIATION_MISMATCH'; deltaCents: number };
