Core schema & types (§11)

pub enum DataType {
    Boolean, Int8, Int16, Int32, Int64, UInt8, UInt16, UInt32, UInt64,
    Float32, Float64, Utf8, Binary, Date32, Timestamp(TimeUnit),
    Decimal128 { precision: u8, scale: i8 },
    Dictionary { key: Box<DataType>, value: Box<DataType> },
}
pub enum TimeUnit { Second, Millisecond, Microsecond, Nanosecond }
pub struct Field { pub name: String, pub data_type: DataType, pub nullable: bool }
pub struct Schema { pub fields: Vec<Field> }
pub type SchemaRef = Arc<Schema>;
pub trait Array: Send + Sync {
    fn len(&self) -> usize;
    fn data_type(&self) -> &DataType;
    fn validity(&self) -> Option<&Bitmap>;
    fn slice(&self, offset: usize, length: usize) -> ArrayRef;
}
pub type ArrayRef = Arc<dyn Array>;
pub struct PrimitiveArray<T: PrimitiveType> {
    values: Arc<[T]>, validity: Option<Bitmap>, offset: usize, length: usize,
}
pub struct Utf8Array {
    values: Arc<[u8]>, offsets: Arc<[i32]>, validity: Option<Bitmap>,
    offset: usize, length: usize,
}
pub struct DictionaryArray<K: PrimitiveType> { keys: PrimitiveArray<K>, values: ArrayRef }
pub struct RecordBatch { schema: SchemaRef, columns: Vec<ArrayRef>, num_rows: usize }
pub trait Operator: Send + Sync {
    fn output_schema(&self) -> &SchemaRef;
    fn consume(&mut self, batch: RecordBatch) -> Result<bool, Error>;   // push, hot path (P2)
    fn next(&mut self) -> Result<Option<RecordBatch>, Error>;           // pull, debug-only
    fn finish(&mut self) -> Result<(), Error>;                         // breaker finalization
}
pub struct Morsel { pub batch: RecordBatch, pub partition_id: u32, pub source_offset: u64 }
pub struct Pipeline { pub source: Box<dyn Operator>, pub ops: Vec<Box<dyn Operator>>,
                       pub sink: Box<dyn Operator> }
pub struct Scheduler { topology: NumaTopology, queues: Vec<MorselQueue>,
                        pipelines: Vec<Arc<Pipeline>> }
pub struct NumaTopology { pub nodes: Vec<NumaNode> }
pub struct NumaNode { pub id: u32, pub cpus: Vec<u32>, pub memory_bytes: u64 }
pub struct HashTable<K, V> { buckets: Vec<Bucket<K, V>>, capacity: usize, mask: usize, seed: u64 }
#[repr(C, align(64))]
pub struct Bucket<K, V> { keys: [K; 4], vals: [V; 4], occupied: u8 }
pub enum AggFn { Sum(usize), Count(usize), CountStar, Min(usize), Max(usize), Avg(usize),
                 SumDistinct(usize) }
pub struct AggState { sums: Vec<f64>, counts: Vec<u64>, mins: Vec<f64>, maxs: Vec<f64> }
pub enum Expression {
    Column(usize), Literal(ScalarValue),
    BinaryOp { op: BinaryOp, left: Box<Expression>, right: Box<Expression> },
    UnaryOp { op: UnaryOp, child: Box<Expression> },
    Cast { child: Box<Expression>, to: DataType },
    Case { whens: Vec<(Expression, Expression)>, otherwise: Box<Expression> },
}
pub enum BinaryOp { Add, Sub, Mul, Div, Eq, NotEq, Lt, LtEq, Gt, GtEq, And, Or }
pub enum UnaryOp { Not, Neg, IsNull, IsNotNull }
pub type KernelFn = fn(inputs: &[ArrayRef], out: &mut dyn ArrayBuilder) -> Result<(), Error>;
pub struct KernelRegistry { fns: HashMap<(BinaryOp, DataType, DataType), KernelFn> }
pub struct MemoryBudget { total: usize, operators: Vec<Arc<AtomicUsize>>, spill_threshold: f32 }
pub struct SpillWriter { path: PathBuf, partition_id: u32 }
pub struct SpillFile { path: PathBuf, partition_id: u32, seed: u64 }
pub struct SpillReader { files: Vec<SpillFile> }
pub struct CodegenContext { cache: HashMap<u64, CompiledKernel> }
pub struct CompiledKernel { function_ptr: *const u8, signature: KernelSignature }
pub struct KernelSignature { inputs: Vec<DataType>, output: DataType }
pub enum JoinType { Inner, LeftOuter, RightOuter, FullOuter, LeftSemi, LeftAnti }
pub struct ExchangeOut { key_cols: Vec<usize>, num_partitions: u32,
                          output_queues: Vec<MorselQueue> }
