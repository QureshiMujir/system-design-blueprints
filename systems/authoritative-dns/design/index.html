Core schema & types (§11)

pub struct Name { labels: SmallVec<[Label; 4]> }
pub struct Label { bytes: SmolStr }               // ≤63 bytes, ASCII/Punycode

impl Name {
    pub fn root() -> Self;
    pub fn from_str(s: &str) -> Result<Self, NameError>;
    pub fn parent(&self) -> Option<Self>;
    pub fn is_subdomain_of(&self, other: &Self) -> bool;
    pub fn label_count(&self) -> usize;
    pub fn canonical(&self) -> CanonicalName;      // lowercase, fully qualified
}

#[repr(u16)]
pub enum RecordType {
    A = 1, NS = 2, CNAME = 5, SOA = 6, PTR = 12, MX = 15, TXT = 16, AAAA = 28,
    SRV = 33, DNAME = 39, OPT = 41, DS = 43, SSHFP = 44, RRSIG = 46, NSEC = 47,
    DNSKEY = 48, NSEC3 = 50, NSEC3PARAM = 51, TLSA = 52, SVCB = 64, HTTPS = 65,
    CAA = 257, ANY = 255, AXFR = 252, IXFR = 251, Unknown(u16),
}
pub struct ResourceRecord {
    pub name: Name, pub rtype: RecordType, pub class: RecordClass,
    pub ttl: u32, pub data: RData,
}

pub enum RData {
    A(Ipv4Addr), Aaaa(Ipv6Addr), Cname(Name),
    Mx { preference: u16, exchange: Name },
    Ns(Name),
    Soa { mname: Name, rname: Name, serial: u32, refresh: u32, retry: u32,
          expire: u32, minimum: u32 },
    Ptr(Name), Txt(Vec<Vec<u8>>),
    Srv { priority: u16, weight: u16, port: u16, target: Name },
    Caa { flags: u8, tag: SmolStr, value: Bytes },
    Unknown { rtype: u16, data: Bytes },
}
pub enum RDataDnssec {                             // logically part of RData
    Dnskey { flags: u16, protocol: u8, algorithm: u8, public_key: Bytes },
    Ds { key_tag: u16, algorithm: u8, digest_type: u8, digest: Bytes },
    Rrsig { type_covered: RecordType, algorithm: u8, labels: u8,
            original_ttl: u32, sig_expiration: u32, sig_inception: u32,
            key_tag: u16, signer_name: Name, signature: Bytes },
    Nsec { next_name: Name, types: TypeBitmap },
    Nsec3 { algorithm: u8, flags: u8, iterations: u16, salt: Bytes,
            next_hashed: Bytes, types: TypeBitmap },
    Svcb { priority: u16, target: Name, params: SvcParams },
    Https { priority: u16, target: Name, params: SvcParams },
    Opt(EdnsOptions),
}
pub struct DnsMessage {
    pub header: DnsHeader, pub questions: Vec<Question>,
    pub answers: Vec<ResourceRecord>, pub authorities: Vec<ResourceRecord>,
    pub additionals: Vec<ResourceRecord>,
}
pub struct DnsHeader {
    pub id: u16, pub flags: DnsFlags,
    pub qdcount: u16, pub ancount: u16, pub nscount: u16, pub arcount: u16,
}
pub struct DnsFlags {
    pub qr: bool, pub opcode: Opcode, pub aa: bool, pub tc: bool,
    pub rd: bool, pub ra: bool, pub z: u8, pub ad: bool, pub cd: bool,
    pub rcode: ResponseCode,
}
#[repr(u16)]
pub enum ResponseCode {
    NoError = 0, FormErr = 1, ServFail = 2, NxDomain = 3, NotImp = 4,
    Refused = 5, YxDomain = 6, YxRRSet = 7, NxRRSet = 8, NotAuth = 9,
    NotZone = 10, BadVers = 16, BadKey = 17, BadTime = 18, BadMode = 19,
    BadName = 20, BadAlg = 21, BadTrunc = 22, BadCookie = 23,
}
pub struct ZoneTree { pub origin: Name, pub root: Node, pub serial: u32 }
pub struct Node {
    pub rrsets: HashMap<RecordType, RrSet>,
    pub children: BTreeMap<Label, Node>,
    pub wildcard: Option<Box<Node>>,
}
pub struct RrSet {
    pub rtype: RecordType, pub class: RecordClass, pub ttl: u32,
    pub records: Vec<RData>, pub rrsigs: Vec<RData>,   // RData::Rrsig only
}
pub struct TsigRecord {
    pub algorithm: Name, pub time_signed: u64, pub fudge: u16,
    pub mac: Bytes, pub original_id: u16, pub error: ResponseCode,
}
pub enum KeyRolloverState {
    Steady { key: KeyId },
    PrePublish { old: KeyId, new: KeyId, since: SystemTime },
    DoubleSign { old: KeyId, new: KeyId, since: SystemTime },
    RetireOld { old: KeyId, since: SystemTime },
}
pub struct RrlBucket { pub tokens: f32, pub last_refill: Instant }
pub struct XfrSession {
    pub zone: Name, pub kind: XfrKind, pub peer: IpAddr,
    pub state: XfrState, pub tsig_key: Option<KeyId>,
}
