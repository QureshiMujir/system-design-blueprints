Core schema & types (§11)

pub struct TransactionId(pub u64);          // Copy; INVALID=0, FROZEN=u64::MAX
pub struct CommitTs(pub u64);               // Copy; high 48b physical µs, low 16b logical
pub const ABORTED_XMIN: TransactionId;      // = TransactionId(u64::MAX - 1)
impl TransactionId { pub const INVALID: Self; pub const FROZEN: Self; }
impl CommitTs { pub const NEVER: Self; pub fn physical_us(self) -> u64; pub fn logical(self) -> u16; }
#[repr(C)]
pub struct TupleHeader { pub xmin: TransactionId, pub xmax: TransactionId,
    pub cmin: u32, pub cmax: u32, pub xmin_commit_ts: CommitTs, pub next: VersionPtr }
pub struct VersionPtr(pub u64);             // opaque; backend-interpreted
pub struct TupleVersion<'a> { pub header: TupleHeader, pub payload: &'a [u8] }
pub struct Snapshot { pub xmin_horizon: TransactionId, pub xmax_ceiling: TransactionId,
    pub in_progress: BTreeSet<TransactionId>, pub snapshot_ts: CommitTs }
impl Snapshot { pub fn sees_xid(&self, xid: TransactionId) -> bool; }
pub enum TransactionState { Active, Committing, Committed { commit_ts: CommitTs }, Aborted }
pub struct Transaction { pub id: TransactionId, pub snapshot: Snapshot,
    pub write_set: WriteSet, pub read_set: ReadSet, pub state: TransactionState,
    pub started_us: u64 }
pub struct WriteSet { /* entries: Vec<WriteEntry>, cap 10,000 (ADR-29) */ }
pub struct WriteEntry { pub key: Vec<u8>, pub old_ptr: VersionPtr, pub new_ptr: VersionPtr }
pub struct ReadSet { /* entries: Vec<ReadEntry>, populated only under SSI */ }
pub struct ReadEntry { pub key: Vec<u8>, pub seen_xmin: TransactionId, pub predicate: Option<PredicateScope> }
pub trait Visibility { fn is_visible(&self, hdr: &TupleHeader, snap: &Snapshot) -> bool; }
pub struct StandardMvcc;
pub fn is_visible_full(hdr: &TupleHeader, snap: &Snapshot, my_xid: TransactionId,
    my_cmd: u32, state_oracle: &dyn TxnStateOracle) -> bool;
pub fn find_visible<S: VersionStore, V: Visibility>(store: &S, vis: &V, snap: &Snapshot,
    key: &[u8]) -> Option<TupleVersion<'_>>;
pub fn find_visible_hot<S: VersionStore, V: Visibility>(store: &S, vis: &V, snap: &Snapshot,
    key: &[u8]) -> Option<TupleVersion<'_>>;                 // Phase 10 hot-path variant
pub trait VersionStore: Send + Sync {
    fn chain_head(&self, key: &[u8]) -> Option<VersionPtr>;
    fn load(&self, ptr: VersionPtr) -> Option<TupleVersion<'_>>;
    fn append(&self, key: &[u8], header: TupleHeader, payload: &[u8]) -> Result<VersionPtr, AppendError>;
    fn set_xmax(&self, ptr: VersionPtr, xmax: TransactionId) -> Result<(), SetXmaxError>;
    unsafe fn reclaim(&self, ptr: VersionPtr);                // GC-only, precondition I10
}
pub enum AppendError { OutOfSpace, IndexCorrupt }
pub enum SetXmaxError { WriteConflict { current_xmax: TransactionId } }
pub struct Registry { /* projection: epoch::Atomic<RegistryProjection> (Phase 10) */ }
pub struct RegistryProjection { pub next_xid: TransactionId, pub active_sorted: Vec<TransactionId> }
impl Registry {
    pub fn take_snapshot(&self, oracle: &dyn TimestampOracle) -> Snapshot;
    pub fn retire(&self, id: TransactionId, final_state: TransactionState);
    pub fn compute_horizon(&self) -> TransactionId;
}
pub trait TimestampOracle: Send + Sync {
    fn allocate_xid(&self) -> TransactionId;
    fn assign_commit_ts(&self) -> CommitTs;
    fn now_for_snapshot(&self) -> CommitTs;
}
pub struct CentralizedOracle;               // AtomicU64-backed
pub struct HybridLogicalClock { /* state: AtomicU64 */ }
pub struct CommitRecord { pub xid: TransactionId, pub commit_ts: CommitTs,
    pub write_count: u32, pub writes: Vec<WriteEntryRecord> }
pub struct WriteEntryRecord { pub key: Vec<u8>, pub old_ptr: VersionPtr, pub new_ptr: VersionPtr }
pub trait WalSink: Send + Sync {
    fn append_commit(&self, xid: TransactionId, ts: CommitTs, writes: &WriteSet) -> Result<CommitHandle, WalError>;
    fn append_abort(&self, xid: TransactionId) -> Result<(), WalError>;
    fn append_checkpoint(&self, state: CheckpointState) -> Result<(), WalError>;
}
pub struct CommitHandle;                    // opaque
pub struct DurableToken { /* _seal: () */ } // typestate: only from CommitHandle::wait_durable
pub struct SiReadLock { pub holder: TransactionId, pub key_or_predicate: PredicateScope,
    pub granted_at_xmin: TransactionId }
pub enum PredicateScope { Key(Vec<u8>), Page { table: u32, page: u32 },
    Predicate { table: u32, expr: PredicateExpr } }
pub enum PredicateExpr { Equals { column: u16, value: Vec<u8> },
    Range { column: u16, lo: Vec<u8>, hi: Vec<u8> } }
pub enum IsolationLevel { ReadCommitted, SnapshotIsolation, SerializableSnapshotIsolation }
