Core schema & types (§11)

struct MethodId(u64)                    // stable, content-hash-derived; owned: graph-store; immutable; Copy; persists across incremental runs
struct Method { id: MethodId, blocks: Vec<BasicBlock>, params: Vec<RefNode> }
                                         // owned: ir-model; immutable after IR-read; not soundness-critical itself (input, not output)
enum IrNode { Alloc, Load, Store, Call, Return, Spawn, Phi }
                                         // owned: ir-model; immutable; Copy-like tags; soundness-critical (drives transfer-fn dispatch)
struct Statement { node: IrNode, operands: Vec<RefId>, site: SiteId }
                                         // owned: ir-model; immutable; serializable (conformance fixtures)
struct Allocation { id: AllocId, site: SiteId, method: MethodId }
                                         // owned: connection-graph; immutable identity; soundness-critical (the unit being classified)
struct AllocId(u64)                     // content-hash-derived within method scope; Copy; stable across incremental re-analysis (I8)
enum EscapeState { NoEscape, ArgEscape, GlobalEscape }
                                         // owned: escape-lattice; Copy, Ord (lattice order); monotone-only mutation (I2); the soundness-critical output type
struct EscapeLattice;                   // stateless; provides join(EscapeState, EscapeState) -> EscapeState; pure function, no mutation
struct ConnectionGraph { objects: Vec<ObjectNode>, refs: Vec<RefNode>, fields: Vec<FieldNode>,
                          points_to: Vec<PointsToEdge>, field_edges: Vec<FieldEdge>, deferred: Vec<DeferredEdge> }
                                         // owned: connection-graph crate; mutable during fixpoint, frozen after convergence; per-method; not directly shared across threads
struct ObjectNode { id: AllocId, state: EscapeState }
                                         // mutable only via monotone join (I2); soundness-critical
struct FieldNode { owner: AllocId, field: FieldId }
                                         // immutable once created; field-sensitivity unit (ADR-07)
struct RefNode { id: RefId, method: MethodId }
                                         // immutable; maps 1:1 to an SSA value
struct PointsToEdge { from: RefId, to: AllocId }
                                         // append-only within a fixpoint run; soundness-critical (defines reachability)
struct FieldEdge { from: (AllocId, FieldId), to: AllocId }
                                         // append-only; soundness-critical
struct DeferredEdge { from: RefId, to: RefId }
                                         // append-only; resolved lazily during fixpoint
trait TransferFn { fn apply(&self, g: &mut ConnectionGraph, stmt: &Statement) -> Changed; }
                                         // pure w.r.t. all state outside `g` (I13); one impl per IrNode variant
struct Fixpoint { worklist: VecDeque<AllocId>, iterations: u32 }
                                         // owned: fixpoint-solver; mutable during solve; bounded by I3; not persisted
struct CallGraph { nodes: Vec<MethodId>, edges: Vec<CallSite>, sccs: Vec<Vec<MethodId>> }
                                         // owned: cfg-callgraph/callgraph-dispatch; immutable per compilation unit; shared read-only across summary workers
struct CallSite { caller: MethodId, callee_targets: Vec<MethodId>, actuals: Vec<RefId>, site: SiteId }
                                         // immutable; callee_targets.len() > 1 implies virtual dispatch (ADR-11)
struct MethodSummary { method: MethodId, param_effects: Vec<EscapeTransfer>, return_effect: EscapeTransfer, ir_hash: u64 }
                                         // owned: summary-engine; immutable once computed; persisted (summary-cache); serializable; versioned (ADR-17)
struct SummaryCache { by_hash: HashMap<u64, MethodSummary> }
                                         // owned: summary-cache crate; mutable (insert/evict); persistence: content-hash keyed on disk (I9); not soundness-critical itself but staleness is (I9)
struct Context { call_string: Vec<SiteId> }
                                         // used only in the optional 1-CFA extension (ADR-06); empty for context-insensitive baseline
struct PartialEscape { branch: BranchId, state: EscapeState, materialize_at: Option<SiteId> }
                                         // owned: partial-escape crate; must satisfy I12 (never below baseline classifier); additive, not authoritative
struct ScalarReplaceCandidate { alloc: AllocId, fields: Vec<FieldId> }
                                         // soundness-critical output; requires I6 field-alias-free check before construction
struct StackAllocCandidate { alloc: AllocId }
                                         // soundness-critical output; requires EscapeState::NoEscape
struct LockElisionSite { alloc: AllocId, monitor_site: SiteId }
                                         // soundness-critical output; requires I7 thread-confinement, strictly stronger than StackAllocCandidate's precondition
struct SoundnessCheck { alloc: AllocId, claimed: EscapeState, observed_escaped: bool }
                                         // owned: soundness-oracle; produced at runtime instrumentation time; a `claimed == NoEscape && observed_escaped` record is a release-blocking defect (I1, I11)
struct OptimizationFacts { scalar: Vec<ScalarReplaceCandidate>, stack: Vec<StackAllocCandidate>, locks: Vec<LockElisionSite> }
                                         // owned: optimizer-facts crate; the versioned external interface (ADR-17); immutable once emitted per compilation unit
enum EscapeError { MalformedIr(SiteId), UnresolvedCallTarget(CallSite), FixpointDivergence(MethodId), SchemaVersionMismatch }
                                         // exhaustive error taxonomy; MalformedIr and FixpointDivergence are hard-reject; UnresolvedCallTarget triggers I4's conservative path, not an error exit
