Core schema & types (§11)

struct Micros(i64);   // 1e-6 currency unit; integer money; checked arithmetic
  fn checked_add(self, Micros) -> Result<Micros, MoneyError>;
  fn checked_mul_ratio(self, num:u64, den:u64) -> Result<Micros, MoneyError>;  // for shading/pCTR scaling

struct Bid(Micros);    // advertiser-stated bid (price units)         DP13
struct Value(i128);    // internal expected value = bid×pCTR (scaled); wider to avoid overflow
struct Price(Micros);  // clearing/settlement price the winner pays    DP13
struct Score(i128);    // ranking key = bid×pCTR×quality (fixed-point scaled)
struct BidRequest {
  id: AuctionId, imp: SmallVec<[Imp; 1]>, device: Device, user: User,
  tmax_ms: u32, cur: Currency, at: AuctionType,
  bcat: SmallVec<[Category;8]>, badv: SmallVec<[Domain;8]>,
  recv_at: Instant,            // ingress timestamp (monotonic)
  deadline: Instant,           // recv_at + collection_budget
}
struct Imp { id: ImpId, bidfloor: Micros, bidfloorcur: Currency, format: Format, secure: bool }
struct Bid {
  id: BidId, impid: ImpId, price: Bid, cid: CampaignId, crid: CreativeId,
  cat: SmallVec<[Category;4]>, seat: SeatId,
  nurl: Option<Url>, burl: Option<Url>, lurl: Option<Url>,
  recv_at: Instant,            // arrival (for late-drop, I5)
}
struct ScoredBid {
  bid: BidId, seat: SeatId, cid: CampaignId,
  stated_bid: Bid, pctr: Pctr, quality: Quality,
  score: Score,                 // bid×pctr×quality (fixed-point)
  effective_value: Value,       // bid×pctr
  eligible: bool, reject: Option<FilterReject>,
}

enum AuctionType { FirstPrice, SecondPrice, Gsp { slots:u8 }, Vcg { slots:u8 } }

struct Award { bid: BidId, seat: SeatId, slot: u8, price: Price, reserve: Micros }

struct AuctionResult {
  auction: AuctionId, at: AuctionType,
  awards: SmallVec<[Award; 4]>,     // one per filled slot
  eligible_count: u32, received_count: u32, late_dropped: u32,
  nbr: Option<NoBidReason>,         // set iff no awards
  seed: u64,                        // tie-break seed (determinism, I6)
}
  fn digest(&self) -> [u8;32];      // hash of awards+inputs for replay compare
fn run_auction(
  bids: &[ScoredBid],            // already filtered+scored
  at: AuctionType,
  reserves: &ReserveSet,         // per-slot effective reserve
  seed: u64,
) -> AuctionResult;
// pure: no clock, no IO, no global RNG; deterministic given args (DP1/I6)
// invariants asserted in debug: IR (I2), reserve (I4), monotonicity helper (I3)
struct ReserveSet { per_slot: SmallVec<[Micros;4]>, soft: bool }  // soft=true ⇒ soft-floor semantics
struct ValueDist { /* parametric or histogram CDF F, pdf f */ }
fn virtual_value(v: Micros, dist: &ValueDist) -> i128;   // φ(v)=v−(1−F(v))/f(v)
fn optimal_reserve(dist: &ValueDist) -> Micros;          // r* where φ(r*)=0 (bisection)
fn effective_floor(publisher: Micros, exchange: Micros, dynamic: Micros) -> Micros; // max
struct DeadlineClock { start: Instant, tmax: Duration, reserve: Duration }
  fn collection_deadline(&self) -> Instant;  // start + tmax − reserve
  fn per_bidder_timeout(&self) -> Duration;

struct CircuitBreaker { state: Closed|Open{until:Instant}|HalfOpen, err_rate: Ewma }
  fn allow(&self) -> bool; fn record(&mut self, outcome: Outcome);

async fn fan_out(req: &BidRequest, bidders: &[BidderHandle], clock: &DeadlineClock)
  -> CollectedBids;   // returns bids with recv_at ≤ deadline only (I5)
struct CampaignBudget { id: CampaignId, budget: Micros, horizon: TimeWindow, spent: AtomicMicros }
struct Pacer {
  mode: Throttle | Dual | Pid,
  mu: f64,            // dual multiplier (Dual)
  p_enter: f64,       // admission prob (Throttle)
  pid: PidState,      // (Pid)
  inflight: InflightLedger,  // reserved-but-unconfirmed spend (I11)
}
  fn admit(&self, value: Value, price_hint: Price, now: Instant) -> Admission; // Skip | Bid(shaded)
  fn on_win(&mut self, price: Price, now: Instant);       // reserve in-flight, update controller
  fn on_billing_confirmed(&mut self, auction: AuctionId); // move in-flight → spent
  fn reconcile(&mut self, kv_spent: Micros);              // bound overspend (I7)

enum Admission { Skip(NoBidReason), Bid { shaded: Bid } }   // shaded = value/(1+mu) in bid units
struct CapKey { user: UserId, campaign: CampaignId }
struct ExactWindow { buckets: TtlMap<(CapKey, BucketTs), u32> }  // exact mode
struct CmsCounter { width:usize, depth:usize, table: Box<[u32]>, seeds:[u64;DEPTH] } // over-estimate
struct CappedBloom { bits: BitVec, k: u8 }  // "already at cap" fast-path
  fn estimate(&self, key: &CapKey) -> u32;             // ≥ true count (CMS, DP9)
  fn admit(&self, key: &CapKey, cap: u32, now: Instant) -> bool;  // estimate < cap
  fn record_impression(&mut self, key: &CapKey, now: Instant);
struct OnlineLogistic { w: HashMap<FeatureId, f64>, z:..., n:..., alpha:f64, beta:f64, l1:f64, l2:f64 } // FTRL
  fn predict(&self, feats: &Features) -> Pctr;          // sigmoid(w·x); calibrated
  fn update(&mut self, feats: &Features, clicked: bool); // online SGD/FTRL step
struct Calibrator { bins: Vec<(f64,f64)> }               // isotonic mapping
  fn apply(&self, raw: f64) -> Pctr;
struct WinRateModel { /* P(win | bid, context) from censored data */ }
  fn p_win(&self, bid: Bid, ctx: &Context) -> f64;
  fn suggested_bid(&self, value: Value, target_win: f64, ctx:&Context) -> Bid;
enum JournalRecord {
  Auction(AuctionRecord), Win(WinNotice), Billing(BillingEvent), Loss(LossNotice),
}
struct AuctionRecord {
  auction: AuctionId, ts: u64, at: AuctionType, reserves: ReserveSet,
  bids: SmallVec<[BidRecord;8]>, awards: SmallVec<[Award;4]>,
  eligible_count:u32, late_dropped:u32, nbr: Option<NoBidReason>, digest:[u8;32],
}
struct BillingEvent { auction: AuctionId, bid: BidId, charge: Micros, ts:u64, idem_key:[u8;16] }
  fn append(&mut self, rec: JournalRecord) -> Result<Lsn, JournalError>; // fsync per DP12 boundary
  fn reconcile(&mut self, ledger_cursor: Lsn) -> ReconcileReport;        // exactly-once (I9)
struct AuctionId(Uuid); struct BidId(u64); struct ImpId(u32);
struct CampaignId(u64); struct CreativeId(u64); struct UserId(u128); struct SeatId(u32);
