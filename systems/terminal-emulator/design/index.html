Core schema & types (§11)

// crates/tem-core/src/cell.rs — the unit of grid mutation (I6, I7)
pub struct Cell {
    pub glyph: GlyphCluster,          // owned; ≥1 codepoint (UAX #29, I9)
    pub fg: Color, pub bg: Color, pub ul: Color,
    pub attrs: Attrs,
    pub width: CellWidth,             // I10
    pub hyperlink_id: Option<HyperlinkId>,  // I13
    pub extra: ExtraFlags,
}
pub enum CellWidth { Single, DoubleLeftHalf, DoubleRightHalf }
pub struct Attrs(u32);                // bitflags: BOLD|ITALIC|UNDERLINE|…
pub enum Color { Default, Indexed(u8), TrueColor { r: u8, g: u8, b: u8 } }
// crates/tem-core/src/cursor.rs — cursor + save/restore stack (I8)
pub struct Cursor {
    pub row: u16, pub col: u16,
    pub fg: Color, pub bg: Color, pub attrs: Attrs,
    pub charset: CharsetSlot,
    pub origin_mode: bool, pub autowrap: bool, pub reverse_wraparound: bool,
    pub pending_wrap: bool,           // N.6 pitfall, §15
}
pub enum CharsetSlot { G0, G1, G2, G3 }
// crates/tem-parser/src/state.rs — Paul Williams DEC state machine (§8.2)
pub enum ParserState {
    Ground, Escape, EscapeIntermediate,
    CsiEntry, CsiParam, CsiIntermediate, CsiIgnore,
    DcsEntry, DcsParam, DcsIntermediate, DcsPassthrough, DcsIgnore,
    OscString, SosPmApcString,
}
pub trait Performer {
    fn print(&mut self, c: char);
    fn execute(&mut self, byte: u8);
    fn csi_dispatch(&mut self, params: &Params, intermediates: &[u8], ignore: bool, action: char);
    fn esc_dispatch(&mut self, intermediates: &[u8], ignore: bool, byte: u8);
    fn osc_dispatch(&mut self, params: &[&[u8]], bell_terminated: bool, truncated: bool);
    fn hook(&mut self, params: &Params, intermediates: &[u8], ignore: bool, action: char);
    fn put(&mut self, byte: u8);
    fn unhook(&mut self);
}
pub struct Parser {
    state: ParserState, params: Params, intermediates: SmallVec<[u8; 4]>,
    osc_raw: Vec<u8>, osc_params: Vec<(usize, usize)>, ignoring: bool,
    max_params: usize, max_intermediates: usize,       // ADR-11, I2
    max_osc_bytes: usize, max_dcs_bytes: usize,
}
// crates/tem-grid/src/grid.rs — two screens + scrollback + tables (ADR-03, I4)
pub struct Grid {
    main: Screen, alt: Screen, active: ActiveScreen,
    scrollback: Scrollback, hyperlinks: HyperlinkTable, images: ImageTable,
}
pub enum ActiveScreen { Main, Alt }
pub struct Screen {
    rows: u16, cols: u16, cells: Vec<Vec<Cell>>, cursor: Cursor,
    saved_cursor: Option<Cursor>, scroll_region: ScrollRegion,
    dirty_rows: BitVec, modes: Modes, charsets: [Charset; 4],
}
pub struct ScrollRegion { pub top: u16, pub bottom: u16, pub left: u16, pub right: u16 }
// crates/tem-grid/src/scrollback.rs — ring buffer, main screen only (I4, I5, ADR-04)
pub struct Scrollback { lines: VecDeque<Line>, max_lines: usize, head_global_row: u64 }
pub struct Line { pub cells: Vec<Cell>, pub wrapped: bool }
impl Scrollback {
    pub fn push(&mut self, line: Line);
    pub fn get(&self, global_row: u64) -> Option<&Line>;
    pub fn len(&self) -> usize;
}
// crates/tem-pty/src/pty.rs — cross-platform PTY (ADR-12)
pub trait Pty: Send {
    fn read(&mut self, buf: &mut [u8]) -> io::Result<usize>;
    fn write(&mut self, buf: &[u8]) -> io::Result<usize>;
    fn resize(&mut self, rows: u16, cols: u16) -> io::Result<()>;
    fn child_pid(&self) -> Option<u32>;
    fn kill(&mut self) -> io::Result<()>;
}
pub struct PtyBuilder {
    pub command: PathBuf, pub args: Vec<OsString>, pub env: Vec<(OsString, OsString)>,
    pub rows: u16, pub cols: u16, pub cwd: Option<PathBuf>,
}
// crates/tem-input/src/keys.rs — keyboard event model (§21, ADR-09)
pub struct KeyEvent { pub key: Key, pub mods: Modifiers, pub event: KeyEventKind, pub text: Option<SmolStr> }
pub enum Key { Char(char), Named(NamedKey) }
pub enum KeyEventKind { Press, Release, Repeat }
pub struct Modifiers { pub shift: bool, pub ctrl: bool, pub alt: bool, pub meta: bool,
    pub hyper: bool, pub super_key: bool, pub caps_lock: bool, pub num_lock: bool }
