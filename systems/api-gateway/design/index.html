Core schema & types (§11)

// gwy-core::types — connection/protocol primitives. Signatures only, no bodies.

/// Owned by gwy-bin at startup; immutable after bind. Not Send across restart.
pub struct Listener {
    id: ListenerId,
    bind_addr: SocketAddr,
    tls: Option<TlsContext>,
    protocols: SmallVec<[Protocol; 2]>, // h1, h2, [h3 deferred]
}

/// Owned per-connection by the TLS terminator. Mutation: session ticket cache only.
/// Not persisted; not serialized. Rebuilt per handshake.
pub struct TlsContext {
    cert_chain: Arc<CertifiedKey>,
    alpn_protocols: Vec<Protocol>,
    session_resumption: bool,
}

/// Copy, Eq. Serialized in config docs as a string enum ("http/1.1","h2","h3").
pub enum Protocol { Http1, Http2, Http3 /* PhaseDeferred */ }

/// Owned per-request; not Send across await points that outlive the request.
/// Not persisted. Threaded through the whole filter chain by &mut.
pub struct RequestContext {
    start: Instant,
    deadline: Deadline,
    method: Method,
    uri: Uri,
    headers: HeaderMap,
    body: BodyStream,
    client_addr: SocketAddr,
    matched_route: Option<Arc<RouteMatch>>,
    principal: Option<Principal>,
    trace_ctx: Option<TraceContext>,
}
// gwy-core::deadline / headers / body — bounded-resource primitives.

/// Copy. Monotonic-clock-backed (ADR-33). Never re-originated once set (P3).
pub struct Deadline { expires_at: Instant }
impl Deadline {
    // remaining() -> Duration; is_expired() -> bool. No mutation methods — read-only budget.
}

/// Owned per-request. Canonical-lowercase keys (pitfall §12.6-class avoidance).
/// Bounded: rejects insertion past the configured header-block cap (I2).
pub struct HeaderMap { /* opaque; iteration + get/insert only, capped */ }

/// Streamed by default (P6); buffered only when a filter requests it, capped at
/// the route's max_body_bytes. Not Clone (single-consumer stream).
pub enum BodyStream {
    Streaming(Pin<Box<dyn Stream<Item = Result<Bytes, GatewayError>> + Send>>),
    Buffered(Bytes),
}
// gwy-router::types — immutable once built; rebuilt wholesale on config reload (I4).

/// Owned by ConfigSnapshot. Immutable; concurrent readers via Arc, no locks.
pub struct RouteTable {
    by_host: HashMap<HostKey, MethodTable>,
    version: ConfigVersion,
}

/// Result of a successful match. Cheap to clone (Arc-wrapped route data).
pub struct RouteMatch {
    route_id: RouteId,
    cluster: ClusterRef,
    captures: SmallVec<[(Box<str>, Box<str>); 4]>, // regex capture groups
    filter_bindings: Arc<[FilterBinding]>,
}

/// Serialized in config docs; compiled into RouteTable at reload time.
pub enum RouteAction {
    Forward(ClusterRef),
    Redirect { status: u16, location_template: Box<str> },
    LocalReply { status: u16, body: Bytes },
}
// gwy-filters::types — the filter-chain contract (§8.3). No filter body logic here.

/// Object-safe trait. Implementors: gwy-filters built-ins. Send + Sync required
/// (shared across worker threads per P7). Cross-request state must be its own
/// concern of the implementor (atomics/sharded locks), never plain interior mutability.
pub trait Filter: Send + Sync {
    fn name(&self) -> &str;
    fn priority(&self) -> i32;
    fn on_request(&self, ctx: &mut RequestContext) -> FilterVerdict;
    fn on_response(&self, ctx: &mut RequestContext, resp: &mut ResponseContext) -> FilterVerdict;
}

/// Returned per phase; Continue lets the chain proceed, others short-circuit (§8.3).
pub enum FilterVerdict {
    Continue,
    Reject { status: u16, reason: Box<str> },
    Respond(ResponseContext),
    Suspend(BoxFuture<'static, FilterVerdict>), // async filters (e.g. JWKS fetch)
}

/// Emitted by the rate-limiter filter. Copy. Not persisted; logged in AccessLogRecord.
pub struct RateLimitDecision {
    admitted: bool,
    remaining: u32,
    reset_after: Duration,
}
// gwy-filters::ratelimit — token-bucket primitive (§8.6). Local-scope by default (ADR-15).

/// Owned per (consumer, route) key in a sharded map; mutated via atomics only —
/// no lock on the admit path (P7). Not serialized; rebuilt on process restart.
pub struct TokenBucket {
    capacity: u32,
    refill_rate_per_sec: f64,
    tokens_fp: AtomicU64,      // fixed-point token count
    last_refill_ns: AtomicU64, // monotonic (ADR-33)
}
// gwy-upstream::types — cluster/endpoint/pool. Concurrency: per-worker shards (P7).

/// Owned by ConfigSnapshot; refers into gwy-upstream's live cluster registry by id.
pub struct ClusterRef { cluster_id: ClusterId }

/// Owned by the cluster's endpoint registry. Health/outlier state mutated by the
/// health-checker task only; read by the load balancer without locking (atomics).
pub struct Endpoint {
    addr: SocketAddr,
    weight: u32,
    health: AtomicEndpointHealth,
}

/// Copy, Eq. Serialized for admin/stats dump only (not the wire config format).
pub enum EndpointHealth { Healthy, Degraded, Ejected, Probing }

/// Owned per worker per cluster (P7 sharding — no cross-core pool sharing).
/// Not persisted; rebuilt from a fresh connect cycle on restart.
pub struct ConnectionPool {
    cluster: ClusterId,
    idle: SegQueue<PooledConn>,
    max_size: u32,
}

/// Owned by exactly one in-flight request at a time (checkout/release protocol, I3).
/// Not Clone. Tracks the protocol-cleanliness tag consulted at release (I3).
pub struct PooledConn { conn_id: ConnId, protocol: Protocol, clean: bool }

/// Object-safe trait; implementors: power-of-two-choices (ADR-16, locked), round-robin,
/// consistent-hash (deferred alternative, §36).
pub trait LoadBalancer: Send + Sync {
    fn pick(&self, cluster: &ClusterId, endpoints: &[Endpoint]) -> Option<EndpointRef>;
}
// gwy-resilience::types — retry/breaker (§19). RetryBudget is the shared-state case (P4/I6).

/// Owned by ConfigSnapshot per route; immutable after reload.
pub struct RetryPolicy {
    max_attempts: u8,
    retry_on: RetryOnSet,        // idempotency-scoped (P10)
    per_try_timeout: Duration,
}

/// Owned per cluster; shared across workers via atomics only (no lock on hot path).
/// Refilled by successful primary requests, debited by each retry attempt (I6).
pub struct RetryBudget { tokens_fp: AtomicU64, refill_rate_per_sec: f64, cap: u32 }

/// Copy, Eq. Persisted only in admin/stats dump; not the wire config format.
pub enum CircuitState { Closed, Open { opened_at: Instant }, HalfOpen { probes_in_flight: u8 } }
// gwy-cache::types — RFC 9111. Owned per worker (ADR-20, no cross-node sharing by default).

/// Hash of (method, scheme, host, path, sorted query, Vary-listed header values) — §8.7.
/// Copy, Eq, Hash. Not persisted beyond process lifetime.
pub struct CacheKey(u64);

/// Owned by the per-worker segmented-LRU store; evicted per ADR-21 policy.
/// Serialized only for cross-process cache warm-up tooling (not the wire config format).
pub struct CacheEntry {
    key: CacheKey,
    status: u16,
    headers: HeaderMap,
    body: Bytes,
    response_date: SystemTime,   // wall clock for freshness math against Cache-Control (ADR-33)
    freshness_lifetime: Duration,
    validators: Option<(ETag, LastModified)>,
}
// gwy-config::types — the hot-reloadable snapshot (§8.8, I4/I12).

/// Copy, Ord. Monotonically increasing; enforced at apply time (I12).
pub struct ConfigVersion(u64);

/// The single unit swapped atomically via ArcSwap<ConfigSnapshot> (P2, I4).
/// Immutable once constructed; readers hold a cloned Arc for the request's lifetime.
pub struct ConfigSnapshot {
    version: ConfigVersion,
    routes: RouteTable,
    clusters: HashMap<ClusterId, Arc<ClusterConfig>>,
    filter_registry: HashMap<Box<str>, Arc<dyn Filter>>,
}
// gwy-observe::types + error taxonomy.

/// Emitted async per request (P9); fixed field set, cardinality-bounded labels only.
/// Serialized as one JSON line to the structured access log.
pub struct AccessLogRecord {
    ts: SystemTime,
    route_id: RouteId,
    cluster_id: Option<ClusterId>,
    status: u16,
    latency_gateway_us: u32,
    latency_upstream_us: Option<u32>,
    consumer_bucket: u16, // hashed, not raw consumer id (P9)
    trace_id: Option<TraceId>,
}

/// Exhaustive error taxonomy; every variant maps to a stable HTTP status + metric label.
pub enum GatewayError {
    ParseError { stage: &'static str },       // 400
    DeadlineExceeded { stage: &'static str }, // 504
    BodyTooLarge,                              // 413
    RouteNotFound,                              // 404
    Unauthenticated,                            // 401
    Unauthorized,                                // 403
    RateLimited { retry_after: Duration },       // 429
    UpstreamUnavailable,                          // 503
    CircuitOpen,                                   // 503
    Internal,                                       // 500
}
