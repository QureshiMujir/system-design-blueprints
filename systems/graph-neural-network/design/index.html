    grad: np.ndarray | None          # adjoint, filled by reverse sweep
    # parents + local vjp closures recorded on the active Tape

    # dense ops (each records a vjp): add, sub, mul, div, neg, matmul, sum, mean,

def gather(X: Node, idx: np.ndarray) -> Node: ...        # rows X[idx]; vjp=scatter_add (I2)

class Graph:
    x: np.ndarray            # (N, F) node features
    edge_index: np.ndarray   # (2, E) int; [src; dst]
    edge_attr: np.ndarray | None = None   # (E, Fe)
    y: np.ndarray | None = None           # (N,) or (N,C) labels
    train_mask: np.ndarray | None = None  # (N,) bool
    val_mask: np.ndarray | None = None
    test_mask: np.ndarray | None = None
    @property
    def num_nodes(self) -> int: ...
    @property
    def num_edges(self) -> int: ...
    def validate(self) -> None: ...        # shape + index bounds (I13)

class MessagePassing:
    def __init__(self, aggr: str = "sum"): ...    # "sum" | "mean" | "max"
    def propagate(self, x: Node, edge_index, edge_attr=None, *, N: int) -> Node:
        ...   # gather -> message -> aggregate -> update (8.2)
    def message(self, x_i: Node, x_j: Node, edge_attr) -> Node: ...   # override
    def aggregate(self, msg: Node, index, N: int) -> Node: ...         # segment reduce
    def update(self, aggr: Node, x: Node) -> Node: ...                 # override

def edge_softmax(scores: Node, dst: np.ndarray, N: int) -> Node: ...  # segment softmax (I11)
