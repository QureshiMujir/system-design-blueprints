Core schema & types (§11)

pub enum Arch { X86, X86_64, AArch64, Arm, RiscV64, Other(u16) }
pub enum Endian { Little, Big }
pub enum Fmt { Elf, Pe, Coff }

pub struct ParseError { pub offset: u64, pub kind: ErrKind }      // total parsing (principle 5)
pub enum ErrKind { Oob{len:u64,file:u64}, Overflow, BadMagic, Ambiguous, DanglingLink,
                   BadStrtabIdx, RvaUnmapped(u64), CountCap, BadField(&'static str) }

pub trait Reader<'a> {                                            // bp-read; all reads checked (I1/I2)
    fn span(&self, off: u64, len: u64) -> Result<&'a [u8], ParseError>;   // ⊆ buffer
    fn u16(&self, off: u64, e: Endian) -> Result<u16, ParseError>;
    fn u32(&self, off: u64, e: Endian) -> Result<u32, ParseError>;
    fn u64(&self, off: u64, e: Endian) -> Result<u64, ParseError>;
    fn cstr(&self, off: u64) -> Result<&'a str, ParseError>;      // bounds + NUL (I6)
}

pub struct Section<'a> { pub name: &'a str, pub kind: SecKind, pub flags: SecFlags,
                         pub data: Result<&'a [u8], ParseError>, pub vaddr: u64, pub size: u64,
                         pub align: u64, pub entsize: u64, pub link: u32, pub info: u32 }
pub struct Symbol<'a> { pub name: &'a str, pub value: u64, pub size: u64, pub bind: Bind,
                        pub kind: SymKind, pub section_idx: Option<usize>, pub defined: bool }
pub struct Reloc { pub offset: u64, pub sym_idx: u32, pub kind: u32, pub addend: i64 }
pub struct Import<'a> { pub lib: &'a str, pub name: Option<&'a str>, pub ordinal: Option<u16>, pub iat_rva: u64 }
pub struct Export<'a> { pub name: Option<&'a str>, pub ordinal: u16, pub rva: u64, pub forwarder: Option<&'a str> }

pub struct RvaMap { /* sorted [(vaddr,vsize,file_off)] */ }       // I4
impl RvaMap { pub fn to_offset(&self, rva: u64) -> Result<u64, ParseError>; }

pub struct BinaryImage<'a> {                                     // immutable, Send+Sync (I12)
    pub fmt: Fmt, pub arch: Arch, pub endian: Endian, pub bits: u8, pub entry: u64,
    pub sections: Vec<Section<'a>>, pub symbols: Vec<Symbol<'a>>, pub relocs: Vec<Reloc>,
    pub imports: Vec<Import<'a>>, pub exports: Vec<Export<'a>>, pub rva_map: RvaMap,
    pub diagnostics: Vec<Diag> }
impl<'a> BinaryImage<'a> {
    pub fn parse(buf: &'a [u8]) -> Result<Self, ParseError>;      // entry; detect→dispatch
    pub fn section(&self, name: &str) -> Option<&Section<'a>>;
    pub fn symbol_at(&self, addr: u64) -> Option<&Symbol<'a>>;    // symbolization for #334
}
