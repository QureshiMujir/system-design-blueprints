Core schema & types (§11)

// crates/vm-core/src/addr.rs — signatures only, no bodies
pub struct PhysAddr(u64);              // ≤52 bits valid (I1, I2)
pub struct VirtAddr(u64);              // canonical form required (x86-64)
pub struct PageFrameNumber(u64);       // PhysAddr >> PAGE_SHIFT

pub enum PageSize { Small4K, Large2M, Huge1G }

impl PhysAddr {
    pub fn new(raw: u64) -> Result<Self, AddrError>;      // I1, I2, §8.7
    pub fn is_aligned(&self, size: PageSize) -> bool;
    pub fn frame(&self) -> PageFrameNumber;
}
impl VirtAddr {
    pub fn new(raw: u64) -> Result<Self, AddrError>;      // canonical check, §8.7
    pub fn page_offset(&self) -> u64;
    pub fn vpn_indices(&self, arch: Arch) -> [u16; 4];    // §8.2 index extraction
}

pub struct PteFlags(u64);
impl PteFlags {
    pub const PRESENT: u64 = 1 << 0;
    pub const WRITABLE: u64 = 1 << 1;
    pub const USER: u64 = 1 << 2;
    pub const ACCESSED: u64 = 1 << 5;
    pub const DIRTY: u64 = 1 << 6;
    pub const HUGE: u64 = 1 << 7;
    pub const GLOBAL: u64 = 1 << 8;
    pub const NX: u64 = 1 << 63;        // P6, I6
}
// crates/vm-vma/src/vma.rs — signatures only
pub struct VmAreaStruct {              // P2: immutable until torn down
    start: VirtAddr,
    end: VirtAddr,
    kind: VmaKind,
    prot: Protection,
    flags: VmaFlags,
    file: Option<FileBacking>,
    anon: Option<AnonState>,
}
pub enum VmaKind { Anonymous, FileBacked, Stack, Heap, Vdso, Shared }
pub struct Protection { read: bool, write: bool, exec: bool }  // I6 enforced here

pub struct MmStruct {                  // per-process address-space handle
    root_table: PhysAddr,              // CR3 / TTBR0 equivalent
    vmas: RbTree<VirtAddr, VmAreaStruct>,
    asid: Asid,
}
impl MmStruct {
    pub fn find_vma(&self, addr: VirtAddr) -> Option<&VmAreaStruct>;   // §15
    pub fn insert_vma(&mut self, vma: VmAreaStruct) -> Result<(), VmaError>;  // I4
    pub fn split_vma_at(&mut self, addr: VirtAddr) -> Result<(), VmaError>;
    pub fn merge_adjacent(&mut self, addr: VirtAddr);
}
// crates/vm-buddy/src/lib.rs — signatures only
pub struct BuddyAllocator {            // P3, ADR-04
    free_lists: [FreeList; MAX_ORDER], // MAX_ORDER = 11 (§35 item 5)
    per_cpu_magazines: PerCpu<Magazine>,
    per_node: [NodeState; MAX_NODES],
    total_pages: u64,
    free_pages: AtomicU64,
}
pub struct Magazine { pages: ArrayVec<PageFrameNumber, 64> }

impl BuddyAllocator {
    pub fn alloc(&self, order: u8, node_hint: Option<NodeId>) -> Option<PageFrameNumber>;
    pub fn free(&self, pfn: PageFrameNumber, order: u8);              // I7, I10
    pub fn buddy_of(pfn: PageFrameNumber, order: u8) -> PageFrameNumber;
}
// crates/vm-fault/src/lib.rs — signatures only
pub enum FaultKind {
    NotPresent { for_write: bool },
    Protection { attempted_write: bool, attempted_exec: bool },
    User,
    Reserved,
}
pub struct FaultContext<'a> {
    addr: VirtAddr,
    kind: FaultKind,
    regs: &'a CpuRegs,
    mm: &'a mut MmStruct,
}
pub enum FaultError { Segv, KernelOops, RetryLimitExceeded }  // I12

pub fn handle_fault(ctx: FaultContext) -> Result<(), FaultError>;   // §17
// crates/vm-tlb/src/lib.rs — signatures only
pub struct PendingSet { entries: PerCpu<BitmapRange> }   // ADR-08

pub struct ShootdownRequest { asid: Asid, range: VirtRange, kind: InvalidationKind }
pub enum InvalidationKind { SinglePage, Range, Full }

impl PendingSet {
    pub fn record(&self, cpu: CpuId, req: ShootdownRequest);     // I3
    pub fn flush_epoch(&self, cpu: CpuId);                       // P5
    pub fn is_lazy(&self, cpu: CpuId) -> bool;                   // lazy-TLB skip
}
// crates/vm-replacement, vm-swap, vm-pagecache — signatures only
pub struct PageDescriptor { pfn: PageFrameNumber, refcount: AtomicU32,
    state: PageState, queue: Option<ReplacementQueue> }
pub enum PageState { Free, Active, Inactive, Pinned, Writeback }  // I9
pub enum ReplacementQueue { Active, Inactive }

pub struct SwapEntry { slot: SwapSlot, tier: SwapTier }
pub enum SwapTier { Zswap { compressed_len: u32 }, Disk { lba: u64 } }  // I13

pub struct PageCacheKey { inode: InodeId, offset: u64 }
pub struct PageCacheEntry { key: PageCacheKey, page: PageFrameNumber,
    dirty: AtomicBool, generation: AtomicU64 }               // §8.8

impl ReplacementQueue {
    pub fn promote(&self, page: &PageDescriptor);        // §20
    pub fn scan_inactive(&self, budget: usize) -> Vec<PageFrameNumber>;
}
