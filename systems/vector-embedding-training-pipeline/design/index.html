Core schema & types (§11)

  # vetp.types — core data
@dataclass(frozen=True)
class Example:                       # immutable curated unit; serialized into .vshard
    id: str; query: str; positive: str
    hard_negs: list[str]; hard_neg_ids: list[str]
    weight: float = 1.0; teacher_score: float | None = None; source: str = ""

class Batch:                         # collated, lives on device; ephemeral (one step)
    q_input_ids: Tensor; q_attn_mask: Tensor          # int32/int8 [B, Lq]
    d_input_ids: Tensor; d_attn_mask: Tensor          # [B, Ld]
    hn_input_ids: Tensor; hn_attn_mask: Tensor        # [B, K, Ld] (K may be 0)
    hn_ids: Tensor; ex_ids: Tensor; weight: Tensor    # [B,K]/[B]/[B]

EmbeddingTensor = Tensor             # [B, d] float; INVARIANT ‖row‖₂=1 (I3); dim d fixed (I15)

  # vetp.config — RunConfig (validated, hashable → config_hash, I13)
@dataclass(frozen=True)
class RunConfig:
    data: DataConfig; model: ModelConfig; loss: LossConfig
    optim: OptimConfig; train: TrainConfig; mining: MiningConfig
    dist: DistConfig; eval: EvalConfig; export: ExportConfig; seed: int
    def config_hash(self) -> str: ...                 # stable hash of normalized config
@dataclass(frozen=True)
class ModelConfig:
    encoder_ref: str; shared_towers: bool; pooling: Literal["cls","mean","last"]
    proj_dim: int; embedding_dim: int; max_seq_len: int   # embedding_dim == d (I15)
@dataclass(frozen=True)
class LossConfig:
    kind: Literal["infonce","mnr","triplet"]; temperature: float  # τ>0 (I12)
    use_inbatch: bool; use_gathered: bool; use_xbm: bool; xbm_size: int; use_gradcache: bool
@dataclass(frozen=True)
class MiningConfig:
    enabled: bool; refresh_every: int; top_k: int; margin: int
    use_ann: bool; use_bm25: bool; filter_false_negatives: bool   # I11
  # vetp.data — sharded format + sampling
class ShardEntry: path: str; n_rows: int; crc32: int
class DataManifest:                  # written once with shards; pins data_version (I13)
    data_version: str; schema_version: str; n_examples: int; n_shards: int
    shard_index: list[ShardEntry]; tokenizer_ref: str; split: Literal["train","eval"]
    dedup_stats: dict; filter_stats: dict
class ShardReader:                   # random-access + sequential; crc-checked
    def read_row(self, shard: int, row: int) -> Example: ...
    def iter_shard(self, shard: int) -> Iterator[Example]: ...
class ShardSampler:                  # seeded; exactly-once per epoch (I2); cursor checkpointed
    def __init__(self, manifest: DataManifest, seed: int, world: int, rank: int): ...
    def epoch_permutation(self, epoch: int) -> Iterator[tuple[int,int]]: ...  # (shard,row)
    def state_dict(self) -> dict: ...; def load_state_dict(self, s: dict) -> None: ...
class StreamingDataset:              # decode→tokenize→collate; feeds bounded prefetch (I5)
    def __iter__(self) -> Iterator[Batch]: ...

  # vetp.model — dual encoder
class DualEncoder(Module):           # owns tower params; forward only (autograd handles backward)
    def encode_query(self, ids: Tensor, mask: Tensor) -> EmbeddingTensor: ...   # → normalized (I3)
    def encode_doc(self, ids: Tensor, mask: Tensor) -> EmbeddingTensor: ...
    def forward(self, batch: Batch) -> tuple[EmbeddingTensor, EmbeddingTensor, EmbeddingTensor]: ...
    @property
    def embedding_dim(self) -> int: ...                # constant d (I15)
  # vetp.loss — losses + negatives
class SimilarityMatrix:              # assembled per step; labels point at positives (I10)
    logits: Tensor                   # [B, num_cols] = q·concat(pos, gathered, xbm, mined)/τ
    labels: Tensor                   # [B] positive column index
    mask_false_neg: Tensor           # bool [B, num_cols] excluding true positives (I11)
def info_nce(sim: SimilarityMatrix, weight: Tensor) -> Tensor: ...   # stable CE; fp32 (I8)
class CrossBatchMemory:              # FIFO queue of detached embeddings [M, d]; momentum
    def enqueue(self, emb: EmbeddingTensor) -> None: ...
    def negatives(self) -> Tensor: ...                 # detached (no grad)
class GradCache:                     # two-pass forward for large effective batch (ADR-015)
    def step(self, model: DualEncoder, batch: Batch, loss_fn) -> Tensor: ...

  # vetp.dist — distributed
def all_gather_embeddings(emb: EmbeddingTensor, world: int, rank: int) -> Tensor: ...
    # returns [B*world, d]; gradient flows ONLY to local rank's B rows (I9)
class DistEnv: rank: int; world: int; device: str; backend: Literal["nccl","gloo"]

  # vetp.sampler — hard-negative mining
class NegativeCache:                 # ex_id → [neg_ids]; versioned; persisted between refreshes
    version: int; staleness_steps: int
    def get(self, ex_id: int) -> list[int]: ...
    def refresh(self, mined: dict[int, list[int]]) -> None: ...   # bumps version, staleness=0
class HardNegativeMiner:
    def mine(self, model: DualEncoder, corpus: ShardReader, queries) -> dict[int, list[int]]: ...
    # ANN over current model (+BM25); drops false negatives (I11)
  # vetp.manifest — checkpoint + run manifest
class Checkpoint:                    # atomic (temp+rename); safetensors weights + json sidecar (I6)
    step: int; epoch: int
    model_state: dict; optim_state: dict; sched_state: dict
    sampler_state: dict; rng_state: dict                # torch/cuda/numpy/python RNG
    arch_signature: str                                  # for compat check (I7)
    def save_atomic(self, path: str) -> None: ...
    @staticmethod
    def load(path: str) -> "Checkpoint": ...
class RunManifest:                   # joins model↔data↔code (I13)
    run_id: str; data_version: str; code_hash: str; config_hash: str
    library_versions: dict; created_at: str; best_metric: float | None
class CompatResult: ok: bool; reason: str | None
def check_compat(ckpt: Checkpoint, cfg: RunConfig) -> CompatResult: ...   # I7

  # vetp.eval — metrics + gate
def recall_at_k(ranked: list[list[int]], qrels: dict, k: int) -> float: ...
def mrr(ranked, qrels) -> float: ...
def ndcg_at_k(ranked, qrels, k: int) -> float: ...
def map_score(ranked, qrels) -> float: ...
def alignment(q_emb: Tensor, d_emb: Tensor) -> float: ...     # collapse metric
def uniformity(emb: Tensor) -> float: ...                     # collapse metric
class EvalReport: metrics: dict[str, float]; quality: dict[str, float]
def regression_gate(report: EvalReport, baseline: dict, tol: dict) -> CompatResult: ...

  # vetp.export — handoff
class HandoffManifest:               # contract for #255 / #328
    model_version: str; embedding_dim: int                  # == d (I15)
    normalization: Literal["l2"]; distance: Literal["cosine","ip"]
    vectors_path: str; ids_path: str; n_vectors: int; data_version: str

  # errors (typed; ADR-033)
class VetpError(Exception): ...
class ConfigError(VetpError): ...        # EmbeddingDimMismatch, BadTemperature, ...
class DataError(VetpError): ...          # CrcMismatch, LeakageDetected, ShardCorrupt
class ResumeError(VetpError): ...        # IncompatibleCheckpoint (I7)
class TrainError(VetpError): ...         # NanLoss (I8), OptimDivergence
class MiningError(VetpError): ...        # IndexBuildFailed, CacheStale
class ExportError(VetpError): ...        # ParityFailed (I14), OpsetUnsupported
