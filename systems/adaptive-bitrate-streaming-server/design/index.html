Core schema & types (§11)

struct Timescale(u32);                                  // Copy; ticks/sec; non-zero invariant
struct MediaTime { ticks: i64, ts: Timescale }          // Copy; rational compare across timescales (I3)
impl MediaTime { fn rescale(self, to: Timescale) -> Result<MediaTime, ArithError>; }  // checked (I20)

struct AccessUnit {                                     // owns a Bytes slice (refcounted)
  track: TrackId, rendition: RenditionId, kind: MediaKind, codec: Codec,
  dts: i64, pts: i64, duration: i64, ts: Timescale,     // dts<=pts (I19)
  keyframe: bool, data: Bytes, params: Option<Arc<ParameterSets>>,
}                                                        // Clone = refcount bump; immutable post-construction
enum MediaKind { Video, Audio, Data }
enum Codec { H264, H265, Av1, Vp8, Vp9, Aac, Opus }
struct ParameterSets { sps: Bytes, pps: Bytes, vps: Option<Bytes>, nal_length_size: u8 }  // Arc-shared; immutable
enum RtmpState { Uninit, VersionSent, AckSent, HandshakeDone }
struct ChunkStreamState { csid_ctx: HashMap<u32, ChunkHeaderCtx>, chunk_size: usize, ... } // per-connection, single-task
struct ChunkHeaderCtx { ts: u32, msg_len: u32, msg_type: u8, msg_stream_id: u32, ts_delta: u32 }
enum AmfValue { Number(f64), Bool(bool), Str(String), Object(Vec<(String, AmfValue)>),
                Null, Undefined, EcmaArray(Vec<(String,AmfValue)>), StrictArray(Vec<AmfValue>), Date(f64) }
struct RtmpCommand { name: String, txid: f64, command_object: AmfValue, args: Vec<AmfValue> }
enum IngestError { BadVersion, BadChunkSize, OrphanChunk, MessageTooLarge{got,cap}, TruncatedAmf,
                   NoParameterSets, UnsupportedCodec(u8), Io(...) }   // thiserror; never panics (DP10)

fn parse_chunk(buf: &mut BytesMut, st: &mut ChunkStreamState) -> Result<Option<RtmpMessage>, IngestError>;
fn decode_amf0(buf: &[u8]) -> Result<(AmfValue, usize), IngestError>;
fn demux_flv_tag(tag: &[u8]) -> Result<FlvTag, IngestError>;
fn reframe_to_au(tag: FlvTag, params: &mut Option<Arc<ParameterSets>>) -> Result<Option<AccessUnit>, IngestError>;
struct BoxWriter<'a> { out: &'a mut BytesMut }          // single-pass; tracks size-patch positions
struct InitSegmentSpec { codec: Codec, params: Arc<ParameterSets>, timescale: Timescale,
                         track_id: u32, width: u16, height: u16, encryption: Option<CencConfig> }
struct MediaSegmentSpec<'a> { seq: u64, base_decode_time: i64,            // → tfdt (I10)
                              samples: &'a [SampleEntry], want_sidx: bool }
struct SampleEntry { duration: u32, size: u32, flags: u32, cts_offset: i32, data: Bytes }
enum PackageError { OffsetOverflow, BoxSizeMismatch{box_, got, expected}, TfdtDrift, SampleTooLarge,
                    SubsampleRange, Arith(ArithError) }

fn write_init_segment(spec: &InitSegmentSpec) -> Result<Bytes, PackageError>;
fn write_media_segment(spec: &MediaSegmentSpec) -> Result<Bytes, PackageError>; // self-checks I9/I10
fn selfcheck_parse(seg: &[u8]) -> Result<(), PackageError>;                      // round-trip own output
struct TsMuxer { cc: HashMap<u16, u8>, pcr_pid: u16, last_pcr: Option<i64>, pmt: PmtSpec }
struct PmtSpec { pcr_pid: u16, streams: Vec<(u16 /*pid*/, u8 /*stream_type*/)> }
enum TsError { ContinuityGap, PcrJump, PesTooLarge, Arith(ArithError) }
fn mux_access_unit(m: &mut TsMuxer, au: &AccessUnit, out: &mut BytesMut) -> Result<(), TsError>; // I11
struct Rung { rendition: RenditionId, width: u16, height: u16, bitrate_bps: u32, codec: Codec }
struct LadderSpec { rungs: Vec<Rung>, gop: MediaTime, force_idr_at_segment: bool, mode: LadderMode }
enum LadderMode { Passthrough, Transcode }              // Transcode drives EncoderBoundary (external)
trait EncoderBoundary {                                  // flux NEVER encodes; this is a seam (DP3, §16)
  fn request(&self, spec: &LadderSpec, src: AuStream) -> Result<Vec<RenditionTrack>, EncoderError>;
}                                                        // PhaseDeferred("§16") for the farm driver
struct RenditionTrack { rung: Rung, aus: AuStream }      // aligned & verified before packaging
enum LadderError { IdrMisaligned{rendition, expected_pts, got_pts}, BitrateNotMonotonic, MissingRung }
fn verify_alignment(tracks: &[RenditionTrack], spec: &LadderSpec) -> Result<(), LadderError>; // I3
struct HlsMasterModel { variants: Vec<Variant>, alt_audio: Vec<MediaGroup> }
struct Variant { bandwidth: u32, resolution: (u16,u16), codecs: String, uri: String, audio_group: Option<String> }
struct HlsMediaModel { version: u8, target_duration: u32, media_sequence: u64,    // monotonic (I2)
                       playlist_type: PlaylistType, map_uri: String,
                       segments: VecDeque<SegRef>, parts: Vec<PartRef>, server_control: Option<LlControl>,
                       preload_hint: Option<String>, rendition_reports: Vec<RenditionReport>, endlist: bool }
enum PlaylistType { Vod, Event, Live }
struct SegRef { extinf: MediaTime, uri: String, discontinuity: bool }
struct PartRef { duration: MediaTime, uri: String, independent: bool }            // INDEPENDENT iff starts on IDR
struct MpdModel { kind: MpdKind, periods: Vec<Period>, avail_start: Option<DateTime>, mup: Option<MediaTime>,
                  time_shift_buffer: Option<MediaTime>, avail_time_offset: Option<f64> }  // ATO = LL-DASH
enum MpdKind { Static, Dynamic }
enum ManifestError { MediaSeqRegression{prev,got}, TargetDurationViolation{target,extinf}, EmptyPlaylist }

fn emit_hls_master(m: &HlsMasterModel) -> String;
fn emit_hls_media(m: &HlsMediaModel) -> Result<String, ManifestError>;            // checks I2, I12 on emit
fn emit_mpd(m: &MpdModel) -> Result<String, ManifestError>;
struct SegmentStore { renditions: HashMap<RenditionId, RenditionRing>, dvr_window: MediaTime, spill: Option<Spill> }
struct RenditionRing { init: Arc<Bytes>, segments: VecDeque<StoredSegment>, parts: VecDeque<StoredPart>, total_bytes: usize }
struct StoredSegment { seq: u64, media_time: MediaTime, dur: MediaTime, bytes: Arc<Bytes>, state: SegState }
enum SegState { Forming, Closing, Committed, Advertised, Evicted }                // §8.15
struct BlockingWaiter { msn: u64, part: Option<u32>, tx: oneshot::Sender<Arc<str>>, deadline: Instant }
enum OriginError { NotFound, RangeNotSatisfiable, ReloadTimeout, DvrExceeded }

fn commit_segment(s: &mut SegmentStore, r: RenditionId, seg: StoredSegment) -> Result<(), OriginError>; // I1 publish-before-advertise
fn serve_range(seg: &StoredSegment, range: Option<ByteRange>) -> Result<(Bytes, ContentRange), OriginError>;
async fn await_reload(store: &SegmentStore, r: RenditionId, msn: u64, part: Option<u32>, to: Duration)
    -> Result<Arc<str>, OriginError>;                                            // LL blocking reload (I13)
struct RtpHeader { version:u8, padding:bool, ext:bool, cc:u8, marker:bool, pt:u8,
                   seq:u16, timestamp:u32, ssrc:u32, csrc:Vec<u32> }
struct RtpPacket { header: RtpHeader, extensions: HeaderExtensions, payload: Bytes }  // payload is a Bytes view
struct HeaderExtensions { abs_send_time: Option<u32>, twcc_seq: Option<u16>, mid: Option<String>, rid: Option<String> }
struct SeqNum(u16);  impl SeqNum { fn after(self, o:SeqNum)->bool; fn diff(self,o:SeqNum)->i32; } // wraparound (I5)
struct RtpTimestamp(u32); // 32-bit modular (I6)
enum Rtcp { SenderReport(Sr), ReceiverReport(Rr), Sdes(..), Bye(..),
            Nack{sender:u32, media:u32, fci:Vec<(u16,u16)>}, Pli{sender:u32,media:u32},
            Fir{sender:u32,media:u32,entries:Vec<(u32,u8)>}, Remb{bitrate:u64,ssrcs:Vec<u32>},
            Twcc(TwccFeedback) }
struct TwccFeedback { base_seq:u16, ref_time:i32 /*64ms*/, statuses:Vec<PacketStatus>, deltas:Vec<i32 /*250us*/> }
enum RtpError { Short, BadExtension, BadCsrc, BadRtcpLength, UnknownPayload }      // drop+count, never panic (DP10)

fn parse_rtp(buf: &[u8], ext_map: &ExtMap) -> Result<RtpPacket, RtpError>;
fn serialize_rtp(p: &RtpPacket, ext_map: &ExtMap, out: &mut BytesMut) -> Result<(), RtpError>;
fn parse_rtcp_compound(buf: &[u8]) -> Result<Vec<Rtcp>, RtpError>;
struct JitterBuffer { cap_pkts: usize, max_depth: MediaTime, ring: BTreeMap<SeqNum, RtpPacket>, ... }
fn insert(jb: &mut JitterBuffer, p: RtpPacket) -> InsertResult;     // Duplicate | Late | Buffered | Overflow(drop-to-keyframe)
fn pop_ready(jb: &mut JitterBuffer, now: MediaTime) -> Vec<RtpPacket>;  // ordered, playout-clocked
struct Router { room: RoomId, producers: HashMap<ProducerId, Producer>, consumers: HashMap<ConsumerId, Consumer>,
                subs: HashMap<ProducerId, SmallVec<ConsumerId>> }   // per-room single-writer task
struct Producer { id: ProducerId, ssrc_layers: HashMap<Ssrc, LayerId>, rid_map: HashMap<String, Ssrc>,
                  params: Arc<ParameterSets>, nack_cache: NackCache, jitter: JitterBuffer }
struct Consumer { id: ConsumerId, peer: PeerId, ssrc_out: Ssrc, rewrite: RewriteState,
                  target: LayerId, state: ConsumerState, ext_map: ExtMap }
enum ConsumerState { New, WaitKeyframe, Active, SwitchingUp, Paused, Closed }     // §8.15
struct RewriteState { seq_offset:i32, ts_offset:i64, last_seq_out:u16, last_ts_out:u32, marker_fix:bool }
struct LayerId { spatial:u8, temporal:u8 }
struct RtpParameters { codecs:Vec<CodecParam>, ext_map:ExtMap, encodings:Vec<Encoding>, rtcp_fb:Vec<Fb> } // from signaling (deferred)
enum SfuError { UnknownProducer, UnknownConsumer, NoSuchLayer, SsrcCollision }

fn on_rtp(r: &mut Router, from: ProducerId, p: RtpPacket);                        // fan-out refs to subscribers
fn rewrite_for(c: &mut Consumer, p: &RtpPacket) -> Option<RtpPacket>;             // I5/I6/I7/I8 enforced
fn select_layer(c: &mut Consumer, target: LayerId, km: &mut KeyframeManager);     // I8; PLI on up-switch
struct GccEstimator { delay: TrendlineFilter, overuse: OveruseDetector, loss: LossController,
                      target_bps: u32, min_bps: u32, max_bps: u32, state: RateState }
enum RateState { Hold, Increase, Decrease }
struct TwccRecord { transport_seq: u16, sent_at: Instant, recv_delta_us: Option<i32> }
fn on_twcc(g: &mut GccEstimator, fb: &TwccFeedback, sent: &SentHistory) -> u32;   // returns clamped target (I15)
struct Pacer { bucket: TokenBucket, target_bps: u32 }
fn pace(p: &mut Pacer, queued: &mut VecDeque<RtpPacket>, now: Instant) -> Vec<RtpPacket>; // I16

struct NackCache { ring: VecDeque<(SeqNum, Bytes)>, cap_bytes: usize, cap_pkts: usize, window: MediaTime }
fn handle_nack(c: &NackCache, fci: &[(u16,u16)]) -> NackResult;                   // Resend(Vec) | RequestKeyframe (I14)
struct KeyframeManager { last_request: HashMap<(Ssrc,LayerId), Instant>, min_interval: Duration } // rate-limit (§4)
fn request_keyframe(km: &mut KeyframeManager, ssrc: Ssrc, layer: LayerId) -> Option<Rtcp>; // coalesce; PLI/FIR
struct CencConfig { scheme: CencScheme, default_kid: [u8;16], iv_size: u8, pattern: Option<(u8,u8)>, psshs: Vec<Pssh> }
enum CencScheme { Cenc /*AES-CTR*/, Cbcs /*AES-CBC pattern*/ }
struct Subsample { clear: u16, encrypted: u32 }                                   // sum == sample size (I17)
struct Pssh { system_id: [u8;16], kids: Vec<[u8;16]>, data: Bytes }
trait KeyProvider { fn key_for(&self, kid: &[u8;16]) -> Result<[u8;16], DrmError>; } // license server deferred §24
enum DrmError { SubsampleRange, UnknownKid, BadIv }
fn encrypt_sample(cfg: &CencConfig, key:&[u8;16], iv:&[u8], au:&AccessUnit) -> Result<(Bytes, Vec<Subsample>), DrmError>;
enum FluxError { Ingest(IngestError), Package(PackageError), Ts(TsError), Manifest(ManifestError),
                 Origin(OriginError), Rtp(RtpError), Sfu(SfuError), Drm(DrmError), Ladder(LadderError),
                 Arith(ArithError), Config(ConfigError) }                          // thiserror; blast radius scoped
struct Metrics { /* handles to counters/histograms incl. every Ixx_total (I1..I20) */ }
struct StreamCtx { track: TrackId, rendition: RenditionId, room: Option<RoomId>, ssrc: Option<Ssrc> } // span fields (DP12)
