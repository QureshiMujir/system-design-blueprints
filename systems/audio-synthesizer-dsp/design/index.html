Core schema & types (§11)

class SampleRate(int):                    // newtype, Hz; owner: RenderContext; RT: read-only, no-alloc
    pass                                  // persisted: yes (preset); serialization: plain int; compat: stable

class BlockSize(int):                     // newtype, frames/block; owner: RenderContext; RT: read-only
    pass                                  // persisted: no (runtime/device config); compat: must be power-of-two (ADR-03)

class ChannelLayout(Enum):                // Mono, Stereo only (multichannel PhaseDeferred, anchor Phase 0)
    MONO = 1
    STEREO = 2                            // owner: AudioBuffer; RT: read-only; persisted: yes

class AudioBuffer:                        // planar float32 storage (ADR-02)
    channels: int                         // owner: Graph/RenderContext; mutation: in-place only
    frames: int
    data: "np.ndarray"                    // shape (channels, frames), float32; allocates: no (arena-backed)
    def view(self, channel: int) -> "np.ndarray": ...   // RT: yes, no-copy view; persisted: no

class RenderContext:                      // per-block execution context
    sample_rate: SampleRate
    block_size: BlockSize
    absolute_sample: int                  // monotonic counter; mutation: RT thread only; allocates: no
class NodeId(int):                       // newtype, stable across a graph's lifetime; persisted: yes

class Port:                              // named signal endpoint on a Node
    node_id: NodeId
    name: str
    kind: "ConnectionType"               // Audio | Mod; owner: Graph; RT: read-only after commit

class Connection:                        // directed edge between two Ports
    src: Port
    dst: Port
    kind: "ConnectionType"                // allocates: no; persisted: yes (patch file)

class Node(Protocol):                    // structural protocol every DSP node implements
    def process(self, ctx: RenderContext, inputs: "dict[str, np.ndarray]") -> "dict[str, np.ndarray]": ...
    def static_size(self) -> int: ...     // bytes; queried once at graph-commit (P1)
    def reset(self) -> None: ...          // allocates: no; RT-safe: yes

class Graph:                             // the DAG container
    nodes: "dict[NodeId, Node]"          // owner: control thread builds, RT thread reads via snapshot swap
    edges: "list[Connection]"
    topo_order: "list[NodeId]"           // recomputed only on commit (ADR-04); RT: read-only
    def validate_acyclic(self) -> None: ...   // raises GraphCycleError (I3); allocates: no in RT path
class Waveform(Enum):                    // Sine, Saw, Square, Triangle, Noise
    SINE = 0
    SAW = 1
    SQUARE = 2
    TRIANGLE = 3
    NOISE = 4

class Oscillator:                        // owner: Voice; mutation: RT thread; allocates: no
    waveform: Waveform
    phase: float                          // [0,1); persisted: no (runtime state)
    phase_increment: float
    pwm: float                            // square-wave duty cycle, [0,1]

class BiquadCoeffs:                      // immutable per-block; ADR-06
    b0: float; b1: float; b2: float
    a1: float; a2: float                  // normalized (a0 divided out); persisted: yes (preset)

class FilterState:                       // mutable per-voice or per-bus state
    z1: float; z2: float                  // biquad delay line, or TPT ic1/ic2; allocates: no

class EnvelopeStage(Enum):                // Idle, Attack, Decay, Sustain, Release
    IDLE = 0; ATTACK = 1; DECAY = 2; SUSTAIN = 3; RELEASE = 4

class EnvelopeParams:                     // immutable per-voice-type config; persisted: yes
    attack_samples: float
    decay_samples: float
    sustain_level: float
    release_samples: float
class VoiceId(int):                      // newtype, index into the static voice array

class Voice:                             // owner: VoiceAllocator; RT: mutated per-block; allocates: no
    active: bool
    midi_note: int
    stage: EnvelopeStage
    osc: Oscillator
    filter_state: FilterState
    age_samples: int                     // for oldest-first stealing (ADR-08)

class VoiceAllocator:                    // owner: audio thread; RT-safe: yes (static free list)
    def allocate(self, note: int) -> VoiceId: ...
    def steal(self) -> VoiceId: ...       // enforces P8 fast-release ramp before reuse
    def release(self, voice_id: VoiceId) -> None: ...

class ParamId(int):                      // newtype, stable identifier for an automatable parameter

class ModRoute:                          // (source, destination, depth); ADR-10
    source: str
    destination: ParamId
    depth: float

class ParamEvent:                        // sample-indexed automation record (§8)
    param_id: ParamId
    target_value: float
    sample_index: int
    curve_type: str                      // "step" | "linear" | "exponential-smoothed"

class SmoothedParam:                     // owner: Node; RT: mutated per-sample; allocates: no
    target: float
    current: float
    smoothing_samples: float             // N_smooth (§8)
class MidiMessage:                       // parsed from raw bytes (ADR-11); persisted: no (transient)
    status: int
    data1: int
    data2: int
    sample_index: int                     // scheduling anchor (§8)

class DelayLine:                         // ring buffer; owner: fx node; allocates: no after init
    buffer: "np.ndarray"
    write_pos: int
    max_delay_samples: int

class FdnMatrix:                         // orthogonal/unitary feedback matrix (I9)
    size: int
    coefficients: "np.ndarray"            // shape (size, size); persisted: yes (preset)

class RenderStats:                        // owner: RenderContext; non-RT consumer reads via telemetry queue
    blocks_processed: int
    xrun_count: int
    peak_dbfs: float
    active_voices: int

class GraphCycleError(Exception): ...      // raised at commit time (I3)
class WavParseError(Exception): ...        // raised by io.wav on malformed input (T1)
class MidiParseError(Exception): ...       // raised by midi.parser on malformed input (I11)
