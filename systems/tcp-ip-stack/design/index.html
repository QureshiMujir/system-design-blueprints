Core schema & types (§11)

pub struct Buf { backing: BufBacking, head: usize, tail: usize }
// ownership: moved across layers, never aliased; mutation: cursor-only (head/tail);
// concurrency: !Sync, single-owner per packet; persistence: none, freed on drop

pub enum BufBacking { Slab(SlabRef), Static(&'static [u8]), Owned(Box<[u8]>) }
pub struct SlabRef(u32);              // index into the core::buffer slab (ADR-10)

pub trait BufOps {
    fn read_n(&mut self, n: usize) -> Option<&[u8]>;
    fn skip(&mut self, n: usize) -> Result<(), BufError>;
    fn put(&mut self, bytes: &[u8]) -> Result<(), BufError>;
    fn remaining(&self) -> usize;
    fn split_off_front(&mut self, n: usize) -> Self;
}
// signatures only (§4 No-Code Rule); bodies are the Phase-0 correctness kernel (§12)

#[derive(Copy, Clone, Eq, PartialEq, Hash)]
pub struct MacAddr([u8; 6]);          // Copy; no interior mutability

#[derive(Copy, Clone, Eq, PartialEq, Hash)]
pub struct Ipv4Addr(u32);             // network byte order (ADR-18)

#[derive(Copy, Clone, Eq, PartialEq, Hash)]
pub struct Ipv6Addr([u8; 16]);

#[derive(Copy, Clone, Eq, PartialEq, Hash)]
pub enum IpAddr { V4(Ipv4Addr), V6(Ipv6Addr) }
// sum type (P2): a wildcard/invalid address state is unrepresentable

#[derive(Copy, Clone, Eq, PartialEq, Hash)]
pub struct Endpoint { pub addr: IpAddr, pub port: u16 }
pub struct EthernetFrame<'a> { buf: &'a [u8] }   // §8.1; borrows Buf, zero-copy (P3)
pub struct ArpPacket<'a>     { buf: &'a [u8] }
pub struct Ipv4Packet<'a>    { buf: &'a [u8] }
pub struct Ipv6Packet<'a>    { buf: &'a [u8] }
pub struct IcmpMessage<'a>   { buf: &'a [u8] }
pub struct UdpDatagram<'a>   { buf: &'a [u8] }
pub struct TcpSegment<'a>    { buf: &'a [u8] }
// each: lifetime tied to the backing Buf; immutable view; the inverse is
// `encode(&self, out: &mut Buf)` — see §8 wire layouts

pub trait WireParse<'a>: Sized {
    fn parse(buf: &'a [u8]) -> Result<Self, ParseError>;   // checksum+bounds verified (I13)
}

pub enum ParseError {
    Truncated, ChecksumMismatch, VersionMismatch, LengthOutOfBounds, UnsupportedOption,
}
// no recovery besides reject-and-drop (P11); never panics on attacker input (T8)
#[derive(Copy, Clone, Eq, PartialEq, Debug)]
pub enum TcpState {
    Closed, Listen, SynSent, SynRcvd, Established,
    FinWait1, FinWait2, CloseWait, Closing, LastAck, TimeWait,
}
// section 3.3.2 of RFC 9293; exhaustive (I8); no integer/flag representation (P2, ADR-03)

pub struct Tcb {
    pub local: Endpoint, pub remote: Endpoint, pub state: TcpState,
    pub snd_una: u32, pub snd_nxt: u32, pub snd_wnd: u32,
    pub snd_wl1: u32, pub snd_wl2: u32, pub iss: u32,
    pub rcv_nxt: u32, pub rcv_wnd: u32, pub irs: u32,
    pub mss: u16, pub rto: Duration, pub srtt: Duration, pub rttvar: Duration,
    pub cwnd: u32, pub ssthresh: u32,
    pub timers: TimerHandles, pub retransmit_q: SegmentQueue,
    pub rx_buf: RingBuf, pub tx_buf: RingBuf, pub flags: TcbFlags,
}
// ownership: one Tcb per 5-tuple, owned by SocketTable; mutation: event-loop only (P5);
// persistence: none, freed at TimeWait expiry (I10); never serialized to wire/disk

pub struct TcbFlags(u32);   // NAGLE_ENABLED, DELACK_PENDING, KEEPALIVE_ENABLED,
                            // SACK_PERMITTED, TIMESTAMPS, WINDOW_SCALE, RETRANSMITTED
pub struct TimerHandles { rto: WheelHandle, persist: WheelHandle, keepalive: WheelHandle, time_wait: WheelHandle }
pub struct SegmentQueue { /* ordered by seq, I3 */ }
pub struct RingBuf { /* pre-sized at socket(), I9 */ }

pub trait CongestionControl {
    fn on_ack(&mut self, tcb: &mut Tcb, acked_bytes: u32);   // ADR-06/19; anchor: RFC 5681
    fn on_loss(&mut self, tcb: &mut Tcb);
    fn on_timeout(&mut self, tcb: &mut Tcb);
}
pub struct RouteTable { /* longest-prefix-match trie, §33 capacity #6 */ }
pub struct Route { pub prefix: IpAddr, pub prefix_len: u8, pub gateway: Option<IpAddr>, pub iface: IfaceId, pub metric: u32 }

pub trait RoutingOps {
    fn add_route(&mut self, r: Route) -> Result<(), RouteError>;
    fn del_route(&mut self, prefix: IpAddr, prefix_len: u8) -> Result<(), RouteError>;
    fn lookup(&self, addr: IpAddr) -> Option<&Route>;
}

pub enum Socket { UdpV4(UdpSocket), UdpV6(UdpSocket), TcpV4(TcpSocket), TcpV6(TcpSocket) }
pub struct UdpSocket { local: Endpoint, rx_buf: RingBuf }
pub struct TcpSocket { tcb: Option<Tcb>, listen_backlog: Option<ListenState> }
pub struct ListenState { backlog: u32, syn_queue: SegmentQueue, accept_queue: VecDeque<Tcb> }
// ownership: SocketTable owns all Socket values; mutation: event-loop only;
// concurrency: !Send across the loop boundary by construction (P8)

pub struct SocketTable { /* hash by 5-tuple (I14) + separate listening-socket table */ }

pub trait DriverShim {
    fn rx(&mut self, buf: &mut Buf) -> Result<usize, ShimError>;
    fn tx(&mut self, buf: &Buf) -> Result<(), ShimError>;
    fn mtu(&self) -> u16;
    fn link_state(&self) -> LinkState;
}
// ADR-05: TUN/TAP implements this trait; AA-1 (DPDK/AF_XDP, §35) would implement the
// same trait to swap without touching any layer above
pub fn socket(domain: Domain, ty: SockType, proto: Protocol) -> io::Result<Socket>;
pub fn bind(s: &mut Socket, addr: Endpoint) -> io::Result<()>;
pub fn listen(s: &mut Socket, backlog: u32) -> io::Result<()>;
pub fn accept(s: &mut Socket) -> io::Result<(Socket, Endpoint)>;
pub fn connect(s: &mut Socket, peer: Endpoint) -> io::Result<()>;
pub fn send(s: &mut Socket, buf: &[u8], flags: SendFlags) -> io::Result<usize>;
pub fn recv(s: &mut Socket, buf: &mut [u8], flags: RecvFlags) -> io::Result<usize>;
pub fn sendto(s: &mut Socket, buf: &[u8], peer: Endpoint, flags: SendFlags) -> io::Result<usize>;
pub fn recvfrom(s: &mut Socket, buf: &mut [u8], flags: RecvFlags) -> io::Result<(usize, Endpoint)>;
pub fn close(s: Socket) -> io::Result<()>;
pub fn shutdown(s: &mut Socket, how: Shutdown) -> io::Result<()>;
pub fn getsockopt<T: SocketOption>(s: &Socket, opt: T) -> io::Result<T::Value>;
pub fn setsockopt<T: SocketOption>(s: &mut Socket, opt: T, value: T::Value) -> io::Result<()>;
pub fn getsockname(s: &Socket) -> io::Result<Endpoint>;
pub fn getpeername(s: &Socket) -> io::Result<Endpoint>;
// stable across the v1.x line (§26); deltas from POSIX documented in docs/protocols/
