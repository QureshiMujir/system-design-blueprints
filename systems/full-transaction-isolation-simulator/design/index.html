Core schema & types (§11)

enum TxnState { Created, Running, Waiting, Committing, Committed, Aborted, Replaying }

enum IsolationLevel { ReadUncommitted, ReadCommitted, RepeatableRead, SnapshotIsolation, Serializable }

struct TransactionDescriptor {
  txn_id: u64,                          // owned by txsim-core, Copy
  begin_ts: u64,                        // monotonic (I6)
  state: TxnState,
  isolation_level: IsolationLevel,
  snapshot_vector: SnapshotVector,      // immutable after Created (P5)
  dependency_set: DependencySet,        // SSI-only; empty under 2PL/MVCC/OCC
  lock_set: LockSet,                    // 2PL-only; empty under MVCC/OCC/SSI
  wal_offset: u64,
}

struct SnapshotVector { as_of_ts: u64 }               // Copy, single field per P5
struct DependencySet { rw_edges: Vec<TxnId>, ww_edges: Vec<TxnId> }  // owned by txsim-ssi
struct LockSet { held: Vec<ResourceId> }              // owned by txsim-lockmgr
enum LockMode { Shared, Exclusive, Update, IntentShared, IntentExclusive }

struct LockEntry {
  resource_id: ResourceId,
  granted_queue: Vec<(TxnId, LockMode)>,
  waiting_queue: Vec<(TxnId, LockMode)>,   // FIFO within priority band
  lock_mode: LockMode,
  escalation_state: EscalationState,       // Row, PageIntent, TableEscalated
}

trait Scheduler {
  fn enqueue(&mut self, txn: TxnHandle);
  fn next(&mut self) -> Option<TxnHandle>;
  fn inject_failure(&mut self, fault: FaultDescriptor);
}
struct VersionNode {
  version_id: u64,
  creator_txn: TxnId,
  begin_visibility_ts: u64,
  end_visibility_ts: Option<u64>,        // None = still open (P5)
  tombstone: bool,
  payload_ref: PayloadRef,               // opaque handle into txsim-storage
  next_version: Option<Box<VersionNode>>,// forms a DAG per I5, acyclic
}

struct OccValidationTicket { txn_id: TxnId, read_set_ts: u64, write_set: Vec<ResourceId> }
struct SerializationGraphEdge { from: TxnId, to: TxnId, kind: ConflictKind, immutable_after_commit: bool } // I8
enum ConflictKind { WriteWrite, WriteRead, ReadWrite }
enum WalRecord {
  Begin { txn_id: TxnId, seq: u64 },
  Write { txn_id: TxnId, key: KeyId, value_ref: PayloadRef, seq: u64 },
  Commit { txn_id: TxnId, seq: u64 },
  Abort { txn_id: TxnId, seq: u64 },
  Checkpoint { seq: u64, active_txns: Vec<TxnId> },
}

struct CheckpointRecord { seq: u64, wal_offset: u64, active_snapshot_low_water: u64 }
struct FaultDescriptor { coordinate: CrashCoordinate, kind: FaultKind }   // I14
struct CrashCoordinate { logical_tick: u64, seed: u64 }
enum FaultKind { WalTornWrite, ProcessCrash, DiskFull, NetworkPartitionReplica }
struct ReplayEvent { seq: u64, txn_id: TxnId, payload: WalRecord, replay_hash: u64 }  // I2
struct ReplicaApplyRecord { primary_seq: u64, applied_ts: u64, lag_ms: u64 }
trait MetricsSink {                     // one-way sink, never feeds scheduling decisions (P8)
  fn record_counter(&mut self, name: &'static str, delta: u64);
  fn record_histogram(&mut self, name: &'static str, value_us: u64);
}

trait ConformanceOracle {                // implemented by both white-box (§18) and elle (§21)
  fn verdict(&self, history: &[WalRecord]) -> AnomalyVerdict;
}

enum AnomalyVerdict { Serializable, DirtyRead, DirtyWrite, NonRepeatableRead, Phantom, WriteSkew, LostUpdate }
