Core schema & types (§11)

/// Stable per-page island identity (RenderCtx uid alloc, SSR); echoed in data-isle-id, Manifest+registry key. Copy, Eq, Hash.
pub struct IslandId(u32);
/// Manifest key for a loadable chunk (island closure or shared dep). Copy, Eq, Hash.
pub struct ChunkRef(u32);

/// One interactive subtree. OWNED by RenderCtx during SSR, serialized into its IslandMarker; not sent whole — client reconstructs from marker attrs (id, strategy, chunk, decoded props, adapter).
pub struct Island {
    id: IslandId, component: ComponentName, strategy: HydrationStrategy,
    props: PropBlob,            // encode()d at render; placed in data-props
    chunk: ChunkRef, adapter: AdapterId, ssr_marker: IslandMarker,
}

/// WHEN to hydrate -> exactly one browser trigger. Exhaustive, no_std; serialized into data-strategy + IslandEntry. Adding a variant = wire-major bump.
pub enum HydrationStrategy {
    Load,                              // hydrate on runtime boot
    Idle,                             // requestIdleCallback (+ setTimeout fallback)
    Visible { root_margin: String }, // IntersectionObserver(root_margin)
    Media   { query: String },       // matchMedia(query), hydrate when it matches
    Interaction { events: Vec<String> }, // pre-hydration listener -> hydrate + replay event
    Only,                            // client-only: no SSR HTML, render from scratch
    Never,                          // static, never hydrated (zero JS)
}

/// DOM boundary delimiting an island in SSR output: `<isle-island>` custom element or a comment pair; carries the versioned attribute wire form below.
pub struct IslandMarker { /* data-isle-id, -strategy, -chunk, -props, -adapter */ }

/// Build-time island routing table. Emitted by emit_manifest, consumed by boot(); JSON wire form (schema_version), immutable post-build, shared &Manifest on client.
pub struct Manifest { islands: Map<IslandId, IslandEntry>, chunks: Map<ChunkRef, ChunkEntry> }
/// Per-island routing row: which chunk, when, prop schema, integrity to verify on load.
pub struct IslandEntry { chunk: ChunkRef, strategy: HydrationStrategy, prop_schema: SchemaId, integrity: String }
/// Per-chunk row: fetch URL, transitive chunk deps (load first), size budget, SRI hash.
pub struct ChunkEntry { url: String, deps: Vec<ChunkRef>, bytes: u64, integrity: String }
/// SSR side: render self to escaped Html within a RenderCtx (uid alloc, sink, ssr scope).
pub trait Component { fn render(&self, ctx: &mut RenderCtx) -> Html; }
/// Client side: adopt the existing SSR DOM subtree el (hydrate, NOT re-render), bind props.
pub trait Island: Component { fn hydrate(&self, el: DomEl, props: &PropValue) -> IslandHandle; }
/// Pluggable UI framework: id() keys the registry, mount() drives hydration for non-reference frameworks; bundled reference adapter wraps isle-signal (non-goal: full React/Vue/Svelte).
pub trait Adapter {
    fn id(&self) -> AdapterId;
    fn mount(&self, el: DomEl, comp: &dyn Island, props: &PropValue) -> IslandHandle;
}

/// Escaped, attributed markup buffer. Output of Component::render; APPEND-ONLY.
/// Construct via escaping constructors so injection is impossible by type.
pub struct Html(/* escaped markup buffer */);

/// Build-time failures (isle-graph / directive parse). std side only.
pub enum BuildError { CycleInChunkGraph, UnknownAdapter, BadDirective }
/// Client hydration failures — each is caught + confined to one island's error boundary.
pub enum HydrateError { ChunkLoadFailed, PropDecodeFailed, MarkerMalformed, AdapterMissing, DomMismatch }

/// Structured-clone-like value: the ONLY type crossing the SSR/client boundary as props. no_std+alloc → identical server (encode)/wasm (decode); owned tree, Refs model shared/cyclic subgraphs by index. Adding a variant = PropBlob major bump.
pub enum PropValue {
    Null, Bool(bool), Int(i64), Float(f64), Str(String), Bytes(Vec<u8>),
    Array(Vec<PropValue>), Object(Vec<(String, PropValue)>),
    Ref(u32) /* shared/cyclic node index */, Date(f64) /* ms epoch */,
}
/// Compact versioned wire envelope embedded in data-props. version gates the byte layout.
pub struct PropBlob { version: u16, bytes: Vec<u8> }
pub fn encode(v: &PropValue) -> PropBlob;                       // SSR: serialize once per island
pub fn decode(b: &PropBlob) -> Result<PropValue, HydrateError>; // client: PropDecodeFailed on bad/unknown version
/// Arena owning reactive nodes; dropping the Scope disposes signals/effects (and an IslandHandle's bindings). One Scope per island instance, !Send (thread-local arena).
pub struct Scope(/* arena */);
pub struct Signal<T> { /* get tracks the running effect; set marks dependents dirty */ }
pub struct Memo<T>   { /* derived, cached; recomputes lazily on dep change */ }
pub struct Effect    { /* re-runs on dep change; NO-OP in SSR mode, runs on client */ }
/// Cross-island shared signals, keyed -> the Store bus wiring islands together.
pub struct Store     { /* keyed signals, late-binding subscribers */ }
impl Scope {
    pub fn signal<T>(&self, v: T) -> Signal<T>;
    pub fn memo<T>(&self, f: impl Fn() -> T) -> Memo<T>;
    pub fn effect(&self, f: impl FnMut());   // no-op under RenderCtx ssr-mode scope
}
pub struct ModuleId(u32);  // Copy, Eq, Hash
/// Build-graph node; is_island_root set when a module carries a client: directive.
pub struct Module { id: ModuleId, deps: Vec<ModuleId>, is_island_root: bool }
/// Module + derived chunk partition. Owned by the build, consumed by emit_manifest.
pub struct ChunkGraph { modules: Vec<Module>, chunks: Vec<Chunk> }
/// A loadable unit: an island root's dep-closure, or a hoisted shared chunk (dedup).
pub struct Chunk { id: ChunkRef, modules: Vec<ModuleId>, deps: Vec<ChunkRef> }
/// Actual minify/transpile is OUT OF SCOPE -> default impl = PhaseDeferred. We own the graph only.
pub trait Transpiler { fn transpile(&self, src: &str) -> Result<String, BuildError>; }
pub fn build_graph(entries: &[ModuleId]) -> ChunkGraph;  // island roots -> closures -> chunks -> dedup
pub fn emit_manifest(g: &ChunkGraph) -> Manifest;        // + URLs + integrity hashes
/// Streaming output target; impls back string buffers AND out-of-order stream writers.
pub trait HtmlSink { fn push(&mut self, html: &str); fn flush(&mut self); }
/// Per-request render state: island registry, uid allocator (→ IslandId), streaming sink, ssr-mode signal Scope; `&mut`-borrowed through the render, NOT Send-shared across awaits.
pub struct RenderCtx<'a> { /* island registry, uid alloc, sink, ssr scope */ }
pub fn render_to_string(root: &dyn Component) -> (String, Manifest);          // buffered
pub fn render_streaming(root: &dyn Component, sink: &mut dyn HtmlSink);        // + out-of-order
/// Priority queue + per-strategy trigger registry. OWNS pending islands; single-threaded, cooperatively scheduled on the main thread (rIC/IO/matchMedia/event callbacks enqueue work).
pub struct HydrationScheduler { /* prio queue + trigger registrations */ }
/// JS shim entrypoint (loads wasm): scan IslandMarkers -> registry -> schedule per strategy.
pub fn boot(manifest: &Manifest);
/// Per-island: lazy-import its chunk, decode props, ADOPT existing SSR DOM (not re-render).
fn hydrate_island(marker: &IslandMarker) -> Result<IslandHandle, HydrateError>;
/// Live island: owns its Scope + attached listeners. Drop -> dispose (unsubscribe, free).
pub struct IslandHandle(/* scope + listeners */);
