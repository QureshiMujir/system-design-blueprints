Core schema & types (§11)

// Service identity & registry
type ServiceId = string                              // stable slug, e.g. "payment-service"
type ServiceDescriptor struct {                       // ownership: platform-team catalog
  id ServiceId
  language string
  tier "edge"|"platform"|"business"|"observability"|"admin"
  port int
  healthEndpoint string
}

// Event envelope (CloudEvents-shaped, §8)
type SchemaId = string                                // registry pointer
type SchemaVersion struct { major int; minor int; patch int }   // SemVer 2.0.0, ADR-22
type CompatibilityMode = "backward"|"forward"|"full"|"none"     // §8 decision table
type EventType = string                               // "com.servicesphere.payment.succeeded.v1"
type PartitionKey = string                            // routes to Kafka partition, drives I12
type IdempotencyKey = string                           // hash(producer, aggregate, op, nonce), drives I4
type CorrelationId = string                             // end-to-end request correlation, drives I3
type TraceContext struct {                               // W3C Trace Context; mutation: none (propagated)
  traceId string
  spanId string
  traceFlags byte
}

type EventEnvelope struct {                                // owner: event-bus-service; immutable once published
  id string                                                  // UUIDv7
  source ServiceId
  specversion "1.0"
  type EventType
  time string                                                  // RFC 3339, advisory only (ADR-23)
  datacontenttype "application/json"|"application/protobuf"
  dataschema SchemaId
  subject string                                                 // optional, aggregate id
  partitionkey PartitionKey
  correlationid CorrelationId
  traceparent string
  idempotencykey IdempotencyKey
  data bytes                                                       // schema-validated payload
}
// Identity & authz
type Principal struct {                                // mutation: immutable per request
  subjectId string
  serviceId ServiceId | null                              // set for service-to-service calls
  scopes []string
}
type TokenClaims struct {                                 // RFC 7519 JWT payload, drives I8
  sub string
  scopes []string
  exp int64
  iat int64
  aud ServiceId
}

// Saga & compensation (§8)
type SagaId = string
type SagaStep struct {                                     // owner: saga orchestrator; concurrency: single-writer per SagaId
  stepId string
  action string
  compensation CompensationAction                             // mandatory, enforced by I11
  state "Pending"|"Completed"|"Compensating"|"Compensated"|"CompensationFailed"
}
type CompensationAction struct {
  targetService ServiceId
  operation string
  idempotencyKey IdempotencyKey                                 // compensations are idempotent too (P7)
}

// Outbox & consumer offsets (§8, ADR-08)
type OutboxRecord struct {                                    // persistence: same DB transaction as the aggregate write
  id string
  aggregateId string
  eventEnvelope EventEnvelope
  publishedAt string | null                                       // null until CDC relay confirms publish
}
type ConsumerOffset struct {                                        // persistence: Kafka-managed + local checkpoint (I7)
  consumerGroup string
  topic string
  partition int
  offset int64
}

// Cache, health, SLOs, errors
type CacheKey = string                                                // namespaced, e.g. "user:profile:<id>"
type HealthReport struct { status "healthy"|"degraded"|"unhealthy"|"unknown"; checkedAt string }
type SloDefinition struct { name string; target float64; window string }   // e.g. 99.9% / 30d
type ErrorBudget struct { sloName string; remainingFraction float64 }
type ErrorCode = "VALIDATION"|"UNAUTHORIZED"|"FORBIDDEN"|"NOT_FOUND"
              | "CONFLICT"|"RATE_LIMITED"|"UPSTREAM_TIMEOUT"|"UPSTREAM_UNAVAILABLE"
              | "SCHEMA_INCOMPATIBLE"|"IDEMPOTENCY_CONFLICT"|"INTERNAL"

// each services/<name>/ shares this internal shape:

├── src/                # domain code, per-language convention (§9 ADR-02)

├── utils-python/                    # equivalent behavior contract for Python services (structlog+otel)
└── ui/                                 # @servicesphere/ui — shared React component library + tokens

├── graphql/                       # one <service>.graphql per GraphQL service
├── protobuf/
│   ├── common/                       # shared types: Id, PageRequest, Error, Money, GeoPoint (§11)

└── asyncapi/                          # event catalog: topics, EventType registry, CloudEvents bindings
infra/
├── docker/                       # base Dockerfiles for infra deps (local only)
├── kubernetes/
│   ├── base/                          # namespaces, ingress, cert-manager, storage classes
│   ├── charts/                          # generic per-service Helm chart
│   └── environments/                      # values-dev.yaml, values-staging.yaml, values-prod.yaml
├── argocd/                               # AppProject + ApplicationSet (auto-discovery per service)
└── secrets/
    ├── vault/                                 # policies + AppRole config (ADR-17)
    └── sops/                                    # .sops.yaml + age key rotation runbook

apps/

└── admin-flutter-web/                     # Flutter Web — admin console (parity build)

benchmarks/          # per-service + end-to-end load profiles (§30 methodology)
conformance/          # cross-service conformance suite (§28, gates Phase 11)
fuzz/                  # event-decoder + gateway-request-parser fuzz harnesses (§28)
examples/                # minimal reference consumer/producer per contract type
templates/                # new-service scaffold template per language, ship-checklist pre-wired
scripts/                    # one-off ops scripts (migration runners, seed data)
tools/                        # internal CLIs (contract-diff checker, affected-service detector)
notes/                          # working notes, non-authoritative
papers/                            # citation index for §5 Required Reading sources
legal/                                # license texts, third-party attribution
brand/                                  # logo, name usage guidelines
