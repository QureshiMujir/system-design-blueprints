Core schema & types (§11)

struct PacketRecord
    stream_id: StreamId
    epoch: SessionEpoch
    sequence: PacketSequence
    payload_ref: BufferHandle
    checksum: Crc32c
    timestamp: DiagnosticTimestamp
    persistence_offset: WalOffset
struct DeliveryCursor
    highest_contiguous: PacketSequence
    highest_observed: PacketSequence
    pending_gaps: GapSet
    replay_position: WalOffset
struct RepairWindow
    stream_id: StreamId
    window_start: PacketSequence
    window_end: PacketSequence
    suppression_deadline: MonotonicInstant
struct MembershipView
    receiver_id: ReceiverId
    join_timestamp: DiagnosticTimestamp
    last_heartbeat: MonotonicInstant
    health_state: HealthState
struct CongestionBudget
    allowed_send_rate: BitsPerSecond
    repair_rate: BitsPerSecond
    burst_budget: Bytes
    pacing_interval: Duration
struct WalSegment
    segment_id: SegmentId
    start_sequence: PacketSequence
    end_sequence: PacketSequence
    checksum: Crc32c
    compression_type: CompressionType
struct TimerSlot
    expiration_tick: WheelTick
    callback_ref: TimerCallbackId
    retry_count: u8
enum RuntimeError
    WireDecodeError
    SequenceViolation
    ReplayCorruption
    CongestionViolation
    StorageUnavailable
    MembershipConflict
    RepairStormDetected
