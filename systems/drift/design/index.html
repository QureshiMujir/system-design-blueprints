Core schema & types (§11)

// Advisory: owned by internal/advisory; immutable after ingest; persisted in
// event store as part of Detected event; serialized as OSV-compatible JSON;
// forward-compatible via unknown-field passthrough.
type Advisory struct {
    ID         string
    Ecosystem  string
    Package    string
    Affected   []AffectedRange
    Severity   string // CVSS v3.1 vector
    Summary    string
    References []string
    Withdrawn  *time.Time
    Modified   time.Time
}

// AffectedRange: owned by internal/advisory; immutable; embedded in Advisory;
// no independent persistence; serialization mirrors OSV ranges[].events.
type AffectedRange struct {
    Introduced   string
    Fixed        *string
    LastAffected *string
    Events       []RangeEvent
}
// DependencyNode: owned by internal/graph; mutated only during graph build,
// immutable once a DependencyGraph is sealed for a (repo, SHA); persisted as
// Postgres jsonb; concurrency: single-writer per (repo, SHA) build.
type DependencyNode struct {
    Name            string
    Ecosystem       string
    ResolvedVersion string
    RequestedRange  string
    Depth           int
    Parents         []NodeID
    Children        []NodeID
}

// RiskScore: owned by internal/risk; recomputed per (advisory, node) pair on
// each graph rebuild; not independently persisted, embedded in FixPlan input;
// serialization: flat struct, no versioning needed (recomputed, not replayed).
type RiskScore struct {
    CVSSVector    string
    BaseScore     float64
    Reachable     *bool // nil until Phase 9
    GraphDepth    int
    ComputedAt    time.Time
}
// FixPlan: owned by internal/planner; immutable once emitted (P5); persisted
// keyed by (advisory ID, repo SHA, planner version); serialization must be
// byte-stable for replay (I2); compatibility: additive fields only.
type FixPlan struct {
    AdvisoryID      string
    TargetNode      NodeID
    FromVersion     string
    ToVersion       string
    Strategy        FixStrategy // SemverBump | SourcePatch | Backport
    PlannerVersion  string
    ConstraintCheck ConstraintProof
}

// PatchSet: owned by internal/patch; immutable once generated; persisted keyed
// by DiffHash; serialization: unified diff text plus structured deltas;
// concurrency: generated single-threaded per FixPlan.
type PatchSet struct {
    Files          []DiffHunk
    BaseSHA        string
    DiffHash       string
    ManifestDeltas []VersionDelta
}
// ValidationRun: owned by internal/sandbox; append-only once started, sealed
// on completion; persisted as an EventRecord payload; concurrency: one run
// per sandbox container instance, no shared mutable state across runs.
type ValidationRun struct {
    ID          string
    PatchSetRef string
    StartedAt   time.Time
    EndedAt     *time.Time
    Result      ValidationResult // Passed | Failed | Timeout | SandboxError
    Logs        LogRef
    CPUUsed     time.Duration
    MemPeakMiB  int
}

// Justification: owned by internal/justify; immutable once signed (P3);
// persisted alongside PullRequestDraft; serialization: canonical JSON, signed
// via Sigstore bundle; compatibility: schema version field, additive only.
type Justification struct {
    AdvisoryID       string
    RepoSHA          string
    PatchDiffHash    string
    ValidationRunID  string
    PlannerVersion   string
    RiskScore        RiskScore
    IssuedAt         time.Time
    Signature        SigstoreBundle
    VerificationSteps []string
}
// PullRequestDraft: owned by internal/forge; immutable once handed to
// ForgeAdapter; not persisted beyond the event store's Proposed event;
// serialization: forge-agnostic struct translated per adapter.
type PullRequestDraft struct {
    Repo            string
    BaseBranch      string
    HeadBranch      string
    Title           string
    BodySummary     string
    JustificationRef string
    Labels          []string
}

// SandboxHandle: owned by internal/sandbox; mutable only by the owning
// sandbox-runner process; never persisted (ephemeral, process-local);
// concurrency: exactly one goroutine controls a handle's lifecycle.
type SandboxHandle struct {
    ContainerID string
    CgroupPath  string
    NetNS       string // always isolated, no egress
    ExpiresAt   time.Time
}
// ForgeAdapter: owned by internal/forge; interface, stateless per call;
// implementations hold credentials, never exposed to callers; concurrency:
// implementations must be safe for concurrent use.
type ForgeAdapter interface {
    OpenPullRequest(ctx context.Context, draft PullRequestDraft) (PRRef, error)
    FindExistingPR(ctx context.Context, repo, headBranch string) (*PRRef, error)
}

// EventRecord: owned by internal/eventstore; append-only, never mutated (I8);
// persisted as the canonical Postgres table; serialization: versioned
// envelope wrapping a typed payload; compatibility: payload schema versioned
// independently per event type.
type EventRecord struct {
    Repo        string
    SHA         string
    AdvisoryID  string
    EventType   string // Detected | Planned | Patched | ... | Abandoned
    Payload     json.RawMessage
    RecordedAt  time.Time
}

// Error enums (internal/errors), owned centrally, immutable constants.
type FailureReason string

const (
    MalformedAdvisory   FailureReason = "MalformedAdvisory"
    MalformedManifest   FailureReason = "MalformedManifest"
    NoFixAvailable      FailureReason = "NoFixAvailable"
    PatchApplyFailed    FailureReason = "PatchApplyFailed"
    SandboxUnavailable  FailureReason = "SandboxUnavailable"
    ValidationFailed    FailureReason = "ValidationFailed"
    ValidationTimeout   FailureReason = "ValidationTimeout"
    SigningFailed       FailureReason = "SigningFailed"
    RejectedByReviewer  FailureReason = "RejectedByReviewer"
)
