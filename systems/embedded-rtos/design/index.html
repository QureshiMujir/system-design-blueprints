Core schema & types (§11)

// rtos_kernel/src/tcb.rs
pub struct TaskId(pub u16);                       // [ser postcard] [compat stable]
pub struct Priority(pub u8);                      // 0 highest .. 31 idle

pub enum TaskState { Dormant, Ready, Running, Blocked, Delayed, Suspended }

pub struct TaskControlBlock {                     // [own kernel] [mut critical-section only, I6]
    pub id: TaskId,
    pub state: TaskState,
    pub base_prio: Priority,
    pub cur_prio: Priority,                        // ≥ base_prio while inheriting (I8)
    pub psp: *mut u32,                             // saved stack pointer (I4)
    pub stack_base: *mut u8,
    pub stack_size: u16,
    pub wake_tick: u64,                            // valid iff Delayed (I14)
    pub ready_link: ListLink,                      // intrusive (P13)
    pub wait_link: ListLink,                       // on ≤1 wait list (I12)
    pub held_mutexes: MutexList,
    pub canary: u32,                               // stack-overflow sentinel (I7)
    pub runtime_cycles: u64,                       // [conc updated at switch]
    pub stack_high_water: u16,
}

// rtos_kernel/src/sched.rs
pub trait Scheduler {                              // [conc: called under basepri (P12)]
    fn admit(&mut self, tcb: &'static mut TaskControlBlock);
    fn select_next(&self) -> TaskId;               // O(1) CLZ over bitmap (I2,I3,M3)
    fn block(&mut self, id: TaskId, list: &mut WaitList);
    fn wake(&mut self, id: TaskId);
    fn tick(&mut self, now: u64);                  // move Delayed→Ready (I14)
}
pub struct ReadyList { bitmap: u32, lists: [IntrusiveList; 32] }  // bit set ⟺ non-empty (I3)

// rtos_arch/src/context.rs  (unsafe surface, P4)
pub struct ExceptionFrame { r0:u32,r1:u32,r2:u32,r3:u32,r12:u32,lr:u32,pc:u32,xpsr:u32 }
pub unsafe fn context_switch();                    // PendSV body (I4); #[naked]
pub unsafe fn start_first_task() -> !;             // SVC0, drops privilege optionally

// rtos_arch/src/critical.rs
pub struct CriticalSection;                        // RAII; enter raises basepri (I6,P12)
pub fn enter_critical() -> CriticalSection;
// Drop restores basepri; nesting-counted.
// rtos_sync
pub struct Semaphore { count: u16, max: u16, waiters: WaitList }        // [conc critical]
pub struct Mutex { owner: Option<TaskId>, ceiling: Priority, waiters: WaitList } // PI (I8)
pub struct EventGroup { bits: u32, waiters: WaitList }

// rtos_ipc
pub struct MessageQueue<const N: usize, T: Copy> {  // [own static] [ser N/A] (I11, P14)
    storage: [MaybeUninit<T>; N], head: u16, tail: u16, count: u16,
    send_waiters: WaitList, recv_waiters: WaitList, policy: BackpressurePolicy,
}
pub enum BackpressurePolicy { Block, Timeout(u64), Fail }               // ADR-22

// rtos_mem
pub struct Pool<const BLK: usize, const N: usize> { /* fixed blocks, tagged (I15) */ }
pub enum MemError { PoolExhausted, BadFree, Misaligned, Oom }

// error types
pub enum KernelError { WouldBlock, Timeout, QueueFull, QueueEmpty, Overflow,
                       HoldTimeout, StackOverflow, InvalidState }        // [ser postcard]

// rtos_time
pub struct Instant(u64);                           // monotonic ticks (I10,P7)
pub struct Duration(u64);
pub struct SoftTimer { period: Duration, next: Instant, cb: TimerFn, mode: TimerMode }
pub enum TimerMode { OneShot, Periodic }

// rtos_trace
pub enum TraceEvent { TaskSwitch{from:TaskId,to:TaskId}, IsrEnter{n:u8}, IsrExit{n:u8},
                      Block{id:TaskId}, Wake{id:TaskId}, Alloc{sz:u16}, UserMark{tag:u16} }
pub struct TraceRing<const N: usize> { /* lock-free SPSC, P8 */ }        // [conc lock-free]
