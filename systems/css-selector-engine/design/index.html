/// Copy-on-write refcounted token string. Borrowed until escape forces Owned. no_std+alloc.
pub enum CowRcStr<'i> { Borrowed(&'i str), Owned(Rc<String>) }

/// CSS Syntax L3 token. 'i borrows input; immutable; produced by Tokenizer::next.
pub enum Token<'i> {
    Ident(CowRcStr<'i>), Function(CowRcStr<'i>), Hash(CowRcStr<'i>), IDHash(CowRcStr<'i>),
    QuotedString(CowRcStr<'i>),
    Number{ has_sign: bool, value: f32, int_value: Option<i32> },   // int_value=Some iff integral
    Dimension{ value: f32, int_value: Option<i32>, unit: CowRcStr<'i> },
    Percentage{ unit_value: f32 },
    Delim(char), WhiteSpace(&'i str), Comment(&'i str),
    Colon, Semicolon, Comma,
    IncludeMatch, DashMatch, PrefixMatch, SuffixMatch, SubstringMatch, // ~= |= ^= $= *=
    CDO, CDC,
    ParenthesisBlock, SquareBracketBlock, CurlyBracketBlock,
    CloseParenthesis, CloseSquareBracket, CloseCurlyBracket,
    BadString, BadUrl,                                              // error tokens, never panic
}

/// Single-pass, zero-copy pull lexer. Not Send (borrows input).
pub struct Tokenizer<'i> { /* slice + position */ }
impl<'i> Tokenizer<'i> {
    pub fn new(input: &'i str) -> Self;
    pub fn next(&mut self) -> Option<Token<'i>>;   // None at EOF
    pub fn position(&self) -> usize;               // for error spans
}
pub struct SelectorList(pub Box<[Selector]>);  // comma-separated complex selectors

/// MATCHING ORDER: rightmost compound first; slice iteration = RTL walk; no reversal at match time.
pub struct Selector(Box<[Component]>, Specificity);
impl Selector {
    pub fn specificity(&self) -> Specificity;
    pub fn iter_from(&self, offset: usize) -> impl Iterator<Item=&Component>;
}

/// One simple selector, combinator, or functional pseudo. Immutable.
pub enum Component {
    Combinator(Combinator),               // boundary between compounds, in matching order
    ExplicitUniversalType,                // *
    ExplicitAnyNamespace,                 // *|
    ExplicitNoNamespace,                  // |E  (null namespace)
    DefaultNamespace(NamespaceUrl),       // bound default ns applies to type/universal
    Namespace(NamespacePrefix, NamespaceUrl), // ns|E
    LocalName(LocalName),                 // type selector E
    ID(Atom), Class(Atom),                // #id .class
    AttributeInNoNamespaceExists{ local_name: LocalName, local_name_lower: LocalName },     // [attr]
    AttributeInNoNamespace{ local_name: LocalName, operator: AttrSelectorOperator,
        value: AttrValue, case_sensitivity: CaseSensitivity, never_matches: bool },         // [attr op val]
    AttributeOther(Box<AttrSelectorWithOptionalNamespace>),                                 // [ns|attr ...]
    Negation(Box<[Selector]>),            // :not(S)  non-forgiving
    Is(Box<[Selector]>), Where(Box<[Selector]>),  // :is()/:where() forgiving
    Has(Box<[RelativeSelector]>),         // :has(rel) relational, forward/backward
    Root, Empty, Scope,                   // :root :empty :scope (structural)
    Nth(NthData), NthOf(NthOfData),       // :nth-* and :nth-*(An+B of S)
    NonTSPseudoClass(NonTSPseudoClass),   // stateful/UI/lang/dir/custom
    PseudoElement(PseudoElement),         // ::before etc (matching anchored via Combinator::PseudoElement)
    Host(Option<Selector>), Slotted(Selector), Part(Box<[Atom]>), // shadow DOM = stretch
}

/// Combinators, stored as Component boundaries.
pub enum Combinator {
    Descendant,      // ' '  unbounded  -> backtracks over ancestors
    Child,           // '>'  bounded    -> single parent, no backtrack
    NextSibling,     // '+'  bounded    -> single prev sibling, no backtrack
    LaterSibling,    // '~'  unbounded  -> backtracks over prev siblings
    Part, SlotAssignment, PseudoElement, // shadow / pseudo-element anchoring
}
/// Specificity triple (a,b,c). a=#id, b=.class+[attr]+pseudo-class, c=type+pseudo-element.
/// Universal/combinators/:where() contribute 0. :is()/:not()/:has() take their
/// most-specific argument. Packs to one u32, 8 bits/field, SATURATING on overflow.
pub struct Specificity { pub id: u32, pub class: u32, pub element: u32 }
impl Specificity { pub fn to_u32(self) -> u32; pub fn from_u32(packed: u32) -> Self; }

/// An+B microsyntax payload. a*n + b, 1-based indexing.
pub struct NthData { pub a: i32, pub b: i32, pub ty: NthType }
pub enum NthType { Child, LastChild, OfType, LastOfType, Col, LastCol }
/// :nth-child(An+B of S): An+B filtered to elements matching selector list S.
pub struct NthOfData { pub nth: NthData, pub selectors: Box<[Selector]> }

/// Attribute match operators. Mirror tokenizer match tokens.
pub enum AttrSelectorOperator { Equal, Includes, DashMatch, Prefix, Suffix, Substring } // = ~= |= ^= $= *=

/// Attribute RHS value, interned. never_matches folds compile-time-impossible
/// matches (e.g. [a^=""] , empty Includes) to a constant-false Component.
pub type AttrValue = Atom;

/// Namespace side of an attribute/type constraint.
pub enum NamespaceConstraint { Any, Specific(NamespaceUrl) }

/// Stateful / UI / functional pseudo-classes (T = type, S = structural handled elsewhere).
pub enum NonTSPseudoClass {
    Link, AnyLink, Visited,                         // :visited -> privacy gate (Phase 8/9)
    Hover, Active, Focus, FocusWithin, FocusVisible,
    Enabled, Disabled, Checked, Indeterminate, Default,
    Required, Optional, ReadOnly, ReadWrite, PlaceholderShown, Target,
    Lang(Box<[Atom]>), Dir(Direction), Custom(Atom), // :lang(),:dir(),vendor/unknown
}
/// Pseudo-elements; matched only when MatchingMode::ForStatelessPseudoElement.
pub enum PseudoElement { Before, After, FirstLine, FirstLetter, Selection, Placeholder, Marker, Custom(Atom) }

/// Parse failures. Carried with a byte span. Top-level query API -> SyntaxError.
pub enum ParseError { /* kind: ParseErrorKind, location: usize */ }
pub enum ParseErrorKind {
    UnexpectedToken, EmptySelector, DanglingCombinator, BadPseudo, UnsupportedPseudo,
    BadAttributeSelector, UnclosedBlock, BadNthExpression, NamespaceUnknown,
}
/// Host-tree abstraction. READ-ONLY queries; engine never mutates. Sized+Clone.
pub trait Element: Sized + Clone {
    fn opaque(&self) -> OpaqueElement;          // stable identity; pointer-derived; never deref'd
    fn parent_element(&self) -> Option<Self>;   // tree nav (element-only; skip non-element nodes)
    fn prev_sibling_element(&self) -> Option<Self>;
    fn next_sibling_element(&self) -> Option<Self>;
    fn first_child_element(&self) -> Option<Self>; // :empty / nth support
    fn is_html_element_in_html_document(&self) -> bool; // drives ASCII case-insensitivity + quirks
    fn local_name(&self) -> &LocalName;         // interned; pointer-eq in type-selector matching
    fn namespace(&self) -> &NamespaceUrl;
    fn has_id(&self, id: &Atom, case: CaseSensitivity) -> bool;
    fn has_class(&self, name: &Atom, case: CaseSensitivity) -> bool;
    fn attr_matches(&self, ns: &NamespaceConstraint, local: &LocalName,
                    op: &AttrSelectorOperation) -> bool; // host owns attr storage/lookup
    fn is_root(&self) -> bool;
    fn is_empty(&self) -> bool;    // no element/non-blank-text children
    fn is_link(&self) -> bool;     // href-bearing -> :link/:visited
    fn match_non_ts_pseudo_class(&self, pc: &NonTSPseudoClass,
                                 ctx: &mut MatchingContext) -> bool;
    fn match_pseudo_element(&self, pe: &PseudoElement, ctx: &mut MatchingContext) -> bool;
    fn add_element_unique_hashes(&self, filter: &mut BloomFilter) -> bool; // NO false negatives
}
pub struct OpaqueElement(usize);
/// Per-traversal mutable scratch. NOT Send/Sync; &mut through matches_selector and callbacks.
pub struct MatchingContext<'a> {
    /* bloom_filter, nth_index_cache, matching_mode, visited_handling, quirks_mode,
       scope_element, nesting_level, classes_and_ids_case_sensitivity */
}

pub enum MatchingMode { Normal, ForStatelessPseudoElement }  // gate ::before etc.
pub enum VisitedHandlingMode { AllLinksUnvisited, RelevantLinkVisited, AllLinksVisitedAndUnvisited }
// privacy: API collapses visited+unvisited; querySelector uses AllLinksUnvisited
pub enum CaseSensitivity { CaseSensitive, AsciiCaseInsensitive }
pub enum QuirksMode { Quirks, LimitedQuirks, NoQuirks }      // ASCII-insensitive id/class in Quirks
/// ≤3× 24-bit hashes of ids/classes/local-names left of a combinator. Immutable, Send+Sync.
pub struct AncestorHashes { pub packed: [u32; 3] }
impl AncestorHashes { pub fn new(sel: &Selector) -> Self; }

/// Counting Bloom, 4096 8-bit buckets, 2 slots/insert. Built from ALL ancestors. NO false negatives.
pub struct BloomFilter { /* [u8; 4096] */ }
impl BloomFilter {
    pub fn insert_hash(&mut self, hash: u32);
    pub fn remove_hash(&mut self, hash: u32);     // counting -> reversible on pop
    pub fn might_contain_hash(&self, hash: u32) -> bool;
}

/// Rightmost-simple-selector bucketing; only candidate rules retrieved. Built once, immutable.
pub struct SelectorMap<T> { /* id_hash, class_hash, local_name_hash, namespace_hash, other: Vec<T> */ }
impl<T> SelectorMap<T> {
    pub fn insert(&mut self, entry: T);
    pub fn get_all_matching<E: Element>(&self, el: &E, out: &mut Vec<&T>, ctx: &mut MatchingContext);
}
/// Memoizes (parent, of-type?) → child-index; O(1) amortized `:nth-*`. Per-traversal, &mut.
pub struct NthIndexCache { /* (parent, ty) -> Vec/map of child indices */ }

/// RTL matcher entry. `offset` resumes mid-selector; `hashes` enables Bloom pre-reject.
pub fn matches_selector<E: Element>(
    sel: &Selector, offset: usize, hashes: Option<&AncestorHashes>,
    el: &E, ctx: &mut MatchingContext) -> bool;
/// Classes/ids/attrs/names that, if mutated, force re-match of affected elements.
pub struct InvalidationSet { /* classes, ids, attrs, local_names; descendant flag */ }
/// Per-selector deps built at SelectorMap insert. Maps features → InvalidationSets. Immutable.
pub struct DependencySet { /* feature -> InvalidationSet edges */ }
impl DependencySet { pub fn build(sel: &Selector) -> Self; }
/// Bitflags: minimal re-match work after a mutation.
pub struct RestyleHint( /* bits */ );
impl RestyleHint {
    pub const RESTYLE_SELF: Self;            // re-match this element
    pub const RESTYLE_DESCENDANTS: Self;     // re-match subtree
    pub const RESTYLE_LATER_SIBLINGS: Self;  // re-match '~'/'+' followers
}
