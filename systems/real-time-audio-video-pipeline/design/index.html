Core schema & types (§11)

pub enum SampleFormat { S16LE, F32 }
pub enum PixelFormat { Yuv420P, Nv12, Rgb24 }
pub enum ChannelLayout { Mono, Stereo, Positional(u8 /*count*/) }

pub enum MediaType {
    Audio { format: SampleFormat, rate: u32, layout: ChannelLayout, interleaved: bool },
    Video { pixel: PixelFormat, width: u32, height: u32, fps_num: u32, fps_den: u32 },
    Encoded { codec_id: CodecId /*opaque*/, timebase: Timebase },
}
// Ownership: Copy/Clone (small). Mutation: immutable post-fixate. Concurrency: Send+Sync.
impl MediaType { pub fn intersect(&self, other: &MediaType) -> Option<MediaTypeSet>; pub fn is_fixed(&self) -> bool; }

pub struct Timebase { pub num: u32, pub den: u32 }   // e.g. 1/48000 or 1/90000
pub struct Timestamp(pub i64);                        // ticks in some Timebase
pub fn rescale(ts: Timestamp, from: Timebase, to: Timebase) -> Timestamp; // round-half-even

bitflags! { pub struct FrameFlags: u32 { const KEYFRAME=1; const DISCONT=2; const CORRUPT=4; } }
pub struct Frame {
    pub pts: Option<Timestamp>,
    pub dts: Option<Timestamp>,
    pub duration: Option<i64>,
    pub flags: FrameFlags,
    pub buffer: BufferRef,
}
// PTS monotonic non-decreasing per stream (Invariant 4). DISCONT marks a gap after xrun/loss.

pub struct BufferRef(/* Arc-like refcount into a BufferPool slot */);
impl BufferRef { pub fn as_slice(&self) -> &[u8]; pub fn as_mut_slice(&mut self) -> Option<&mut [u8]>; }
// Ownership: refcounted; last drop returns the slot to its pool (Invariant 7). No alloc on drop.

pub struct BufferPool { /* fixed N slots of fixed size; free-list */ }
impl BufferPool {
    pub fn new(slot_size: usize, count: usize) -> Self;   // allocates once, up front
    pub fn acquire(&self) -> Option<BufferRef>;           // None => exhausted (backpressure, never alloc)
    pub fn free_slots(&self) -> usize;                    // telemetry
}
pub enum MediaError { NegotiationFailed, FormatUnsupported, PoolExhausted, DeviceError(i32), Timeout, PtsRegression }
pub enum Direction { In, Out }
pub struct Port { pub dir: Direction, pub caps: NegotiatedCaps /* Set | Fixed */ }
pub struct Pad { pub port: Port, pub peer: Option<PadId> }

pub trait Node {
    fn kind(&self) -> NodeKind;                                   // Source | Transform | Sink
    fn pads(&self) -> &[Pad];
    fn transform_caps(&self, in_caps: &MediaTypeSet) -> MediaTypeSet; // e.g. resampler
    fn process(&mut self, inputs: &[Option<Frame>], outputs: &mut [Option<Frame>]) -> Result<(), MediaError>;
    // process(): NO allocation, NO blocking on RT segments. inputs/outputs index pads.
}
pub enum NodeKind { Source, Transform, Sink }

pub struct Graph { /* nodes, links, segments */ }
impl Graph {
    pub fn add(&mut self, node: Box<dyn Node>) -> NodeId;
    pub fn link(&mut self, from: PadId, to: PadId) -> Result<(), MediaError>; // negotiates caps
    pub fn partition(&mut self) -> Vec<Segment>;     // group by discipline + thread
    pub fn validate(&self) -> Result<(), MediaError>; // all links fixed (Invariant 9)
}
pub struct Segment { pub nodes: Vec<NodeId>, pub discipline: Discipline, pub thread: ThreadClass }
pub enum Discipline { Pull, Push }
pub enum ThreadClass { RtAudio, RtVideo, Transport, Control }
pub enum BackpressurePolicy { Block /*off-RT only*/, DropOldest, DropNewest }
pub struct BoundedQueue<T> { /* fixed capacity, declared policy */ }
impl<T> BoundedQueue<T> {
    pub fn with_capacity(cap: usize, policy: BackpressurePolicy) -> Self;
    pub fn push(&self, item: T) -> PushResult;   // Accepted | DroppedOld | DroppedNew | Blocked
    pub fn pop(&self) -> Option<T>;
    pub fn depth(&self) -> usize;                // telemetry (Invariant 6)
}
pub enum PushResult { Accepted, DroppedOld, DroppedNew, WouldBlock }

pub struct Deadline { pub at: Timestamp }
pub struct DeadlineAccount { /* counters */ }
impl DeadlineAccount {
    pub fn record(&self, finished_at: Timestamp, deadline: Deadline); // miss if finished > deadline
    pub fn misses(&self) -> u64;                 // Invariant 10
    pub fn overrun_hist(&self) -> &Histogram;
}
pub struct Spsc<T> { /* wait-free single-producer/single-consumer ring, power-of-two cap */ }
impl<T> Spsc<T> {
    pub fn new(capacity_pow2: usize) -> (Producer<T>, Consumer<T>);
}
pub struct Producer<T> { /* !Sync; one thread */ }
pub struct Consumer<T> { /* !Sync; one thread */ }
impl<T> Producer<T> { pub fn push(&mut self, item: T) -> Result<(), T /*full*/>; pub fn is_full(&self) -> bool; }
impl<T> Consumer<T> { pub fn pop(&mut self) -> Option<T>; pub fn is_empty(&self) -> bool; }
// Correctness: no drop/dup under SPSC use (Invariant 8); Acquire/Release ordering; loom-verified.

pub enum RtPriority { AudioHigh, VideoMid, TransportLow }
pub fn spawn_rt<F: FnMut() + Send + 'static>(prio: RtPriority, name: &str, body: F) -> RtHandle;
// sets SCHED_FIFO at the mapped priority; body must honor the RT contract (§4.3).
pub fn lock_all_memory() -> Result<(), MediaError>;  // mlockall(MCL_CURRENT|MCL_FUTURE) + pre-touch pools
pub trait Clock { fn now(&self) -> Timestamp; fn timebase(&self) -> Timebase; }
pub struct WallClock;          // CLOCK_MONOTONIC
pub struct MediaClock { /* advanced by consumed samples */ }
pub struct DeviceClock { /* inferred actual rate */ }
pub struct PresentationClock { /* MediaClock + base offset */ }
impl PresentationClock { pub fn map(&self, pts: Timestamp) -> Timestamp; } // PTS -> wall-relative "now"

pub struct DriftEstimator { /* PI controller on buffer-level error */ }
impl DriftEstimator {
    pub fn update(&mut self, buffer_level_frames: i64, target_frames: i64) -> f64; // -> resampler ratio
    pub fn drift_ppm(&self) -> f64;     // telemetry (Invariant 11)
}
pub trait AudioBackend {
    fn open_capture(&mut self, mt: &MediaType, period: u32, periods: u32) -> Result<(), MediaError>;
    fn open_playback(&mut self, mt: &MediaType, period: u32, periods: u32) -> Result<(), MediaError>;
    fn read(&mut self, into: &mut BufferRef) -> Result<usize /*frames*/, XrunOr<MediaError>>;  // RT
    fn write(&mut self, from: &BufferRef) -> Result<usize, XrunOr<MediaError>>;                // RT
    fn buffer_level(&self) -> i64;     // frames queued in the device (drift input)
}
pub enum XrunOr<E> { Xrun, Err(E) }
// AlsaBackend: locked impl over snd_pcm_*. FileBackend: offline WAV. RT contract: read/write only syscalls.
pub fn recover_xrun(/* pcm handle */) -> Result<(), MediaError>;  // snd_pcm_recover + prepare; count it
pub struct Gain { pub gain: f32 }                       // Node: out = in * gain (smoothed)
pub struct Mixer { pub gains: Vec<f32> }                // Node: out = clamp(Σ in_i * gain_i)
pub struct BiquadCoeffs { pub b0: f32, pub b1: f32, pub b2: f32, pub a1: f32, pub a2: f32 }
pub struct Biquad { coeffs: BiquadCoeffs, z1: f32, z2: f32 } // Node: 2nd-order IIR (DF-II transposed)
pub struct Resampler { /* polyphase FIR bank, phase accumulator */ }
impl Resampler {
    pub fn new(ratio: f64, taps: usize, phases: usize) -> Self;
    pub fn set_ratio(&mut self, ratio: f64);            // drift-comp hook (from DriftEstimator)
    pub fn process(&mut self, input: &[f32], output: &mut [f32]) -> (usize, usize); // (consumed, produced)
}
pub fn set_denormals_off();   // FTZ/DAZ (Invariant 13)
// All process() paths: no alloc, no lock; planar F32 in/out; advanced DSP primitives from #174.
pub struct VideoFrame {
    pub mt: MediaType,                  // Video {...}
    pub planes: [PlaneRef; 3],          // ptr+stride per plane (refcounted into a pool)
    pub pts: Timestamp,
}
pub struct PlaneRef { pub buf: BufferRef, pub stride: usize, pub rows: usize }
pub trait VideoBackend {
    fn open_capture(&mut self, mt: &MediaType) -> Result<(), MediaError>;
    fn dequeue(&mut self) -> Result<VideoFrame, MediaError>;   // V4L2 mmap dqbuf
    fn requeue(&mut self, frame: VideoFrame);                  // qbuf back
}
pub fn convert(src: &VideoFrame, dst_fmt: PixelFormat, dst: &mut VideoFrame) -> Result<(), MediaError>;
pub fn scale(src: &VideoFrame, dst: &mut VideoFrame) -> Result<(), MediaError>; // bilinear
// V4l2Backend: locked. Convert/scale: CPU reference; GPU PhaseDeferred("Deferred Features Register").
pub enum SyncDecision { Present, DropFrame, RepeatLast }
pub struct LipSync { pub bound_ms: i64 /*±20*/ }
impl LipSync {
    pub fn decide(&self, video_pts: Timestamp, master_now: Timestamp, clock: &PresentationClock) -> SyncDecision;
    pub fn error_ms(&self, video_pts: Timestamp, master_now: Timestamp) -> i64; // av_sync_error_ms
}
pub struct JitterBuffer<T> { /* reorder by seq#, adaptive bounded depth */ }
impl<T> JitterBuffer<T> {
    pub fn insert(&mut self, seq: u16, ts: Timestamp, item: T) -> InsertResult; // Inserted | LateDiscard | Overflow
    pub fn pop_due(&mut self, now: Timestamp) -> Option<T>;
    pub fn target_depth(&self) -> usize;   // ceil(jitter_p99 / frame_interval)
}
pub enum InsertResult { Inserted, LateDiscard, Overflow }
pub struct CodecId(pub u32);          // opaque (Invariant 12)
pub struct EncodedFrame { pub pts: Timestamp, pub dts: Timestamp, pub keyframe: bool, pub payload: BufferRef }
pub struct RtpHeader { pub marker: bool, pub payload_type: u8, pub seq: u16, pub timestamp: u32, pub ssrc: u32 }
pub fn packetize(frame: &EncodedFrame, mtu: usize, ssrc: u32, pt: u8, out: &mut Vec<RtpPacket>); // non-RT
pub fn depacketize(pkt: &RtpPacket) -> Result<RtpHeader, MediaError>;
pub struct RtpPacket { pub header: RtpHeader, pub payload: BufferRef }
// codec encode/decode that produces/consumes EncodedFrame.payload = PhaseDeferred("Deferred Features Register") -> #380/#173.
