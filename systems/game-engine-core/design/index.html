Core schema & types (§11)

// Entity identity — value type, hashable, 64-bit packed (index+generation), §8.1
type Entity = int                          // packed (generation << 32) | index
struct EntityMeta:
    generation: u32                        // owner: EntityAllocator; mutation: on despawn only
    location: (ArchetypeId, row: u32) | None  // concurrency: single-writer (sync point)
                                            // persistence: not serialized; rebuilt on load

struct EntityAllocator:
    meta: list[EntityMeta]                 // owner: World; grows monotonically
    free: list[u32]                        // indices available for reuse
    // mutation: spawn/despawn only inside CommandBuffer.apply (I4)
// Component registry, §8.2
type ComponentId = int                     // 12-bit index, §4 ceiling 4096
struct ComponentInfo:
    id: ComponentId
    name: str                              // stable across versions (ADR-17)
    size_bytes: int                        // for dense-column allocation
    is_zst: bool                           // zero-sized marker optimization
    schema_version: int                    // §27 compatibility
    // serialization: name+version pair is the on-disk key, never the raw id
// Archetype storage, §8.2, Phase 1 §13
type ArchetypeId = int
struct Column:
    component_id: ComponentId
    data: NumpyStructuredArray | PyObjectArray   // owner: Archetype; mutation: row-indexed only
struct Archetype:
    id: ArchetypeId
    signature: tuple[ComponentId, ...]     // sorted, immutable identity
    columns: dict[ComponentId, Column]
    entities: list[Entity]                 // row → Entity, I2 dense-contiguous invariant
    add_edge: dict[ComponentId, ArchetypeId]     // memoized, I7
    remove_edge: dict[ComponentId, ArchetypeId]  // memoized, I7
// World — single mutable state owner, §3
struct World:
    entities: EntityAllocator
    archetypes: ArchetypeRegistry          // owner of all Archetype instances
    resources: dict[type, object]          // singleton resources (Time, FrameStats, ...)
    events: dict[EventId, EventQueue]
    change_tick: u32                       // monotonic, bumped once per Update stage
    // concurrency: mutated only via CommandBuffer.apply or the executor's sync points
// Query engine, §8.3, Phase 2 §14
struct QueryFilter:
    with_: tuple[ComponentId, ...]
    without: tuple[ComponentId, ...]
    optional: tuple[ComponentId, ...]
    changed: tuple[ComponentId, ...]       // ADR-06 tick-based
    added: tuple[ComponentId, ...]
struct Query:
    filter: QueryFilter
    matched_archetypes: list[ArchetypeId]  // cached, invalidated on new archetype (ADR-05)
    last_run_tick: u32                     // per-system, per-query bookkeeping
// Scheduler, §8.4, Phase 3/6 §15/§17
struct Access:
    reads: frozenset[ComponentId]
    writes: frozenset[ComponentId]
    resource_reads: frozenset[type]
    resource_writes: frozenset[type]
type SystemId = int
struct SystemDescriptor:
    id: SystemId
    name: str
    access: Access                         // declared statically at registration (P6)
    explicit_before: frozenset[SystemId]
    explicit_after: frozenset[SystemId]
struct Stage:
    name: str
    systems: list[SystemDescriptor]
    parallel_groups: list[list[SystemId]]  // computed at Schedule.build (§8.4)
struct Schedule:
    stages: list[Stage]                    // ordered: First..Last, §3
// Command buffer, §8.5, Phase 5 §17
enum Command:
    Spawn(bundle: dict[ComponentId, bytes])
    Despawn(entity: Entity)
    Insert(entity: Entity, component: ComponentId, value: bytes)
    Remove(entity: Entity, component: ComponentId)
    SetParent(child: Entity, parent: Entity | None)
struct CommandBuffer:
    owner_system: SystemId
    commands: list[Command]                // FIFO, applied at stage sync point (I4)
// Events, §8.7, Phase 7 §18
type EventId = int
struct EventQueue:
    event_type: EventId
    current: list[bytes]                   // serialized event payloads
    previous: list[bytes]
    // mutation: send() appends current; swap_buffers() at Last stage only (I8)
// Resources/assets, §8.8, Phase 8 §19
type AssetId = int
enum LoadState:
    Unloaded | Loading | Loaded | HotReloadPending | Failed | Evicting  // §8.12
struct ResourceHandle:
    index: u32
    generation: u32                        // I9, staleness detection on hot reload
struct AssetSlot:
    asset_id: AssetId
    refcount: u32                          // never negative (I9)
    generation: u32
    state: LoadState
    bytes_ref: object | None
// Scene graph, §8.9, Phase 9 §20
struct Transform:
    local_translation: (f32, f32, f32)
    local_rotation: (f32, f32, f32, f32)   // quaternion
    local_scale: (f32, f32, f32)
    dirty: bool                            // P11
struct GlobalTransform:
    matrix: tuple[f32, ...]                // 4x4, row-major, derived-only (never hand-set)
struct SceneNode:
    parent: Entity | None
    children: list[Entity]
    depth: u32                             // §4 ceiling 64
// Prefab / serialization, §8.10, Phase 10 §21
struct Prefab:
    schema_version: int                    // §27 SemVer-tracked
    components: list[(ComponentInfo.name, dict)]  // name-keyed, not raw ComponentId (ADR-17)
    children: list["Prefab"]
    format: "msgpack" | "json"             // ADR-16
// Profiling / errors, §11 close
struct FrameStats:
    frame_number: u64
    stage_timings_us: dict[str, u32]       // per-stage wall time, ring-buffered
    overrun_count: u32                     // I5 telemetry
    last_exception: str | None             // §8.11 Recovering state
struct TickAccumulator:
    accumulator: f64
    tick_dt: f64
    max_frame_dt: f64                      // I5 clamp input
    max_ticks_per_frame: u32               // I5 clamp bound

// Error types — raised, never silently swallowed (P12)
error StaleHandleError(entity: Entity)                  // I1
error ComponentNotRegisteredError(name: str)            // I11
error ArchetypeMismatchError(archetype: ArchetypeId)    // I2/I7
error CyclicParentError(child: Entity, parent: Entity)  // I12
error PrefabValidationError(reason: str)                // I11
error ScheduleCycleError(cycle: list[SystemId])         // I13
error AssetLoadError(asset_id: AssetId, reason: str)    // §8.12 Failed
error CommandReplayError(command: Command)              // I4
