Core schema & types (§11)

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
pub struct Asn(pub u32);
impl Asn {
    pub fn new(n: u32) -> Result<Self, CoreError>;
    pub fn is_private(&self) -> bool;
    pub fn is_documentation(&self) -> bool;
}
impl fmt::Display for Asn {}  // "AS65000"
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub struct MemberId(pub u32);
#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
pub struct MacAddr(pub [u8; 6]);
impl MacAddr {
    pub fn from_bytes(b: [u8; 6]) -> Self;
    pub fn is_broadcast(&self) -> bool;
    pub fn is_multicast(&self) -> bool;
    pub fn is_unicast(&self) -> bool;
}
impl fmt::Display for MacAddr {}  // colon-hex
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub enum IpAddr { V4([u8; 4]), V6([u8; 16]) }

#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub struct Prefix { pub addr: IpAddr, pub len: u8 }
impl Prefix {
    pub fn contains(&self, other: &Prefix) -> bool;
    pub fn network_addr(&self) -> IpAddr;
    pub fn is_bogon(&self) -> bool;
}
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub struct PortId(pub u32);

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Port {
    pub id: PortId,
    pub member: MemberId,
    pub mac: MacAddr,
    pub ip: IpAddr,
    pub speed_gbps: u32,
    pub up: bool,
}
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PeeringLan {
    pub id: u32,
    pub subnet: Prefix,
    pub members: Vec<MemberId>,
}
impl PeeringLan {
    pub fn contains_member(&self, m: MemberId) -> bool;
}
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Member {
    pub id: MemberId,
    pub asn: Asn,
    pub ports: Vec<PortId>,
    pub policy: PeeringPolicy,
    pub announced: Vec<Prefix>,
    pub peering_kind: PeeringKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
pub enum PeeringKind { Bilateral, Multilateral, Both }
#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct VirtualClock(pub u64);   // nanoseconds from epoch 0
impl VirtualClock {
    pub fn zero() -> Self;
    pub fn advance(&self, nanos: u64) -> Self;
    pub fn elapsed_since(&self, earlier: VirtualClock) -> u64;
}
#[derive(Debug, Clone)]
pub struct SimEvent {
    pub time: VirtualClock,
    pub seq: u64,
    pub kind: IxpEvent,
}
impl PartialOrd for SimEvent {}  // min-heap on (time, seq)

#[derive(Debug, Clone)]
pub enum IxpEvent {
    Frame(Frame),
    SessionUp   { session: BgpSessionId },
    SessionDown { session: BgpSessionId, reason: DownReason },
    RecvUpdate  { session: BgpSessionId, update: BgpUpdate },
    PortDown    { port: PortId },
    PortUp      { port: PortId },
    MacAge      { mac: MacAddr, domain: u32 },
    Tick        { label: &'static str },
}
pub struct SimEngine {
    pub clock: VirtualClock,
}
impl SimEngine {
    pub fn new() -> Self;
    pub fn schedule(&mut self, event: SimEvent);
    pub fn schedule_after(&mut self, delay_ns: u64, kind: IxpEvent) -> SimEvent;
    pub fn step(&mut self) -> Option<SimEvent>;
    pub fn run_until(&mut self, end: VirtualClock);
    pub fn run_to_quiescence(&mut self);
}
pub struct MacTable {
    pub domain_id: u32,
    pub aging_nanos: u64,
}
impl MacTable {
    pub fn learn(&mut self, mac: MacAddr, port: PortId, now: VirtualClock);
    pub fn lookup(&self, mac: MacAddr, now: VirtualClock) -> Option<PortId>;
    pub fn flush_port(&mut self, port: PortId);
    pub fn expire(&mut self, now: VirtualClock) -> Vec<MacAddr>;
    pub fn len(&self) -> usize;
}
pub struct Fabric {
    pub ports: HashMap<PortId, Port>,
    pub mac_tables: HashMap<u32, MacTable>,
}
impl Fabric {
    pub fn ingest_frame(&mut self, ingress: PortId, frame: Frame, now: VirtualClock)
        -> Vec<(PortId, Frame)>;
    pub fn bring_port_down(&mut self, port: PortId, now: VirtualClock);
    pub fn port_counter(&self, port: PortId) -> PortCounters;
}
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Frame {
    pub src_mac: MacAddr,
    pub dst_mac: MacAddr,
    pub ethertype: u16,
    pub payload_len: u32,
}
#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
pub enum SessionState {
    Idle, Connect, OpenSent, OpenConfirm, Established,
}
#[derive(Debug, Clone)]
pub struct BgpSession {
    pub id: BgpSessionId,
    pub local: MemberId,
    pub remote: Endpoint,
    pub state: SessionState,
    pub hold_nanos: u64,
    pub keepalive_nanos: u64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub enum Endpoint { Member(MemberId), RouteServer }

pub type BgpSessionId = u64;
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct BgpRoute {
    pub prefix: Prefix,
    pub as_path: Vec<Asn>,
    pub next_hop: IpAddr,
    pub communities: Vec<Community>,
    pub origin_member: MemberId,
    pub rpki_state: Option<ValidationState>,
}
pub struct RouteServer {
    pub asn: Asn,
    pub sessions: HashMap<MemberId, BgpSessionId>,
    pub rib_per_member: HashMap<MemberId, LocRib>,
    pub filters: Arc<PolicyEngine>,
}
impl RouteServer {
    pub fn recv_update(&mut self, from: MemberId, routes: Vec<BgpRoute>,
                       sim: &mut SimEngine) -> Result<(), RsError>;
    pub fn compute_views(&mut self) -> Result<(), RsError>;
    pub fn view_for(&self, member: MemberId) -> impl Iterator<Item = &RsRoute>;
}
pub struct LocRib {
    pub owner: MemberId,
    pub routes: HashMap<Prefix, RsRoute>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct RsRoute {
    pub prefix: Prefix,
    pub as_path: Vec<Asn>,
    pub next_hop: IpAddr,
    pub communities: Vec<Community>,
    pub origin_member: MemberId,
    pub selected: bool,
}
#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum PeeringPolicy {
    Open,
    Selective     { allowed_peers: Vec<MemberId> },
    Restrictive   { explicit_allow: Vec<MemberId> },
}
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ImportFilter {
    pub prefix_list:  Vec<Prefix>,
    pub max_prefix:   u32,
    pub rpki_mode:    RpkiMode,
    pub irr_as_set:   Option<String>,
    pub bogon_filter: bool,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ExportFilter {
    pub prefix_list:       Vec<Prefix>,
    pub strip_communities: Vec<Community>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
pub enum RpkiMode { Disabled, ValidateDropInvalid, ValidateTagOnly }
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub struct Community(pub u32, pub u32);
impl Community {
    pub const BLACKHOLE: Community = Community(65535, 666);
    pub fn announce_to(peer: MemberId) -> Self;
    pub fn do_not_announce_to(peer: MemberId) -> Self;
}
#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
pub enum ValidationState { Valid, Invalid, Unknown }
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct TrafficFlow {
    pub src:      MemberId,
    pub dst:      MemberId,
    pub rate_mbps: f64,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct TrafficMatrix {
    pub flows: Vec<TrafficFlow>,
}
impl TrafficMatrix {
    pub fn load_toml(path: &Path) -> Result<Self, TrafficError>;
    pub fn total_rate_mbps(&self) -> f64;
}
#[derive(Debug, Clone, Copy, Default, Serialize, Deserialize)]
pub struct PortCounters {
    pub tx_bytes:    u64,
    pub rx_bytes:    u64,
    pub tx_frames:   u64,
    pub rx_frames:   u64,
    pub drops:       u64,
    pub floods_sent: u64,
}
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Scenario {
    pub name:           String,
    pub members:        Vec<Member>,
    pub peering_lan:    PeeringLan,
    pub traffic:        TrafficMatrix,
    pub fault_timeline: Vec<FaultEvent>,
    pub asserts:        Vec<ScenarioAssert>,
    pub sim_end_ns:     u64,
}
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct FaultEvent {
    pub at_ns: u64,
    pub kind:  FaultKind,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum FaultKind {
    PortDown    { port: PortId },
    PortUp      { port: PortId },
    MemberLeave { member: MemberId },
    MemberJoin  { member: MemberId },
}
#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum ScenarioAssert {
    RibContains    { member: MemberId, prefix: Prefix,   at_ns: u64 },
    RibNotContains { member: MemberId, prefix: Prefix,   at_ns: u64 },
    SessionState   { a: MemberId, b: Endpoint, state: SessionState, at_ns: u64 },
    PortUtil       { port: PortId, max_pct: f64,          at_ns: u64 },
    NoPrefixLeak   { via_member: MemberId,                at_ns: u64 },
}
pub trait IntoSimEvent: Clone + Send + 'static {
    fn into_ixp_event(self) -> IxpEvent;
}

pub trait Filter {
    fn evaluate(&self, route: &BgpRoute, ctx: &PolicyContext) -> FilterVerdict;
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum FilterVerdict { Accept, Reject }

pub trait MetricEmitter {
    fn emit(&self, store: &mut MetricStore);
}
