Core schema & types (§11)

/// Object-language types. Immutable; recursive children boxed.
pub enum Type {
    Base(Atom),                          // e.g. Bool, Nat — nullary base type
    Arrow(Box<Type>, Box<Type>),         // S -> T  (function)
    Prod(Box<Type>, Box<Type>),          // S * T   (Phase 10 extension)
    Sum(Box<Type>, Box<Type>),           // S + T   (Phase 10 extension)
    Ref(Box<Type>),                      // Ref T   (Phase 12 stretch: references + store typing)
    All(Box<Type>),                      // forall. T  (System F stretch; binder => uses TyVar/de Bruijn at type level)
    TyVar(usize),                        // de Bruijn type variable (only meaningful under All)
}

/// Object terms in DE BRUIJN form: no names, binders are positional.
/// Var(k) refers to the k-th enclosing binder (0 = innermost).
pub enum Term {
    Var(usize),                          // de Bruijn index (NOT a name)
    Lam(Type, Box<Term>),                // \: T. body  — binder; body is in an extended context
    App(Box<Term>, Box<Term>),
    TmTrue, TmFalse,
    If(Box<Term>, Box<Term>, Box<Term>), // if e1 then e2 else e3
    Zero, Succ(Box<Term>),               // Nat constructors; products/sums/refs ctors added in Phase 10/12
}

/// de Bruijn SHIFT: add `d` to every free Var index >= `cutoff`. TCB.
/// Walking under a Lam increments cutoff by 1. `d` is isize (can be -1 on subst).
/// Invariant: shift(t,0,c) == t; shift(shift(t,a,c),b,c) == shift(t,a+b,c).
pub fn shift(t: &Term, d: isize, cutoff: usize) -> Term;

/// CAPTURE-AVOIDING substitution [s/j]t: replace Var(j) by `s`, shifting `s`
/// under each binder it crosses. TCB. Capture-avoidance is AUTOMATIC given correct shift.
/// Beta uses subst(body, 0, shift(arg,1,0)) then shift(_, -1, 0) — see Phase 4.
pub fn subst(t: &Term, j: usize, s: &Term) -> Term;

/// Structural (de Bruijn) equality == alpha-equivalence. TCB.
/// Total, decidable; no name comparison needed because terms are nameless.
pub fn alpha_eq(a: &Term, b: &Term) -> bool;

/// Value predicate: Lam, TmTrue, TmFalse, Zero, Succ(v) where is_value(v), ... .
/// Drives `step` (a value is stuck = irreducible) and canonical_forms.
pub fn is_value(t: &Term) -> bool;
/// A complete object-language definition (the metalanguage's top-level result).
pub struct LangDef {
    pub sorts:     Vec<Atom>,            // declared sorts, e.g. Tm, Ty, Ctx
    pub grammar:   Vec<Production>,      // term + type grammar w/ binding annotations
    pub judgments: Vec<JudgmentForm>,    // judgment FORMS, e.g. Has-Type(Ctx, Tm, Ty)
    pub rules:     Vec<Rule>,            // inference rules (premises => conclusion)
}

/// One grammar production with binder annotations (which arg positions bind).
pub struct Production { pub sort: Atom, pub ctor: Atom, pub args: Vec<SortRef>, pub binders: Vec<BinderSpec> }

/// A judgment FORM (signature): name + the sorts of its argument positions.
pub struct JudgmentForm { pub form: Atom, pub arg_sorts: Vec<Atom> }

pub struct Lexer<'a> { /* byte cursor over &'a str; yields Token */ }
pub enum Token { /* Ident, LParen, Turnstile, Arrow, Colon, ..., Eof */ }
pub struct Parser<'a> { /* recursive-descent over Token stream */ }

pub struct ParseError { pub kind: ParseErrorKind, pub span: (usize, usize), pub msg: alloc::string::String }
pub enum ParseErrorKind {
    UnexpectedToken,                     // lexical/grammar mismatch
    UnknownSort,                         // reference to undeclared sort
    BadBinder,                           // binder annotation inconsistent w/ production arity
    MalformedRule,                       // premise/conclusion not a valid JudgmentForm instance
}
/// An instance of a judgment form: form name + filled argument terms.
/// e.g. Has-Type with args [ctx-as-term, e, T-as-term].
pub struct Judgment { pub form: Atom, pub args: Vec<Term> }

/// An inference rule: name, premise judgments, one conclusion judgment.
///   premises
///   --------- (name)
///   conclusion
pub struct Rule { pub name: Atom, pub premises: Vec<Judgment>, pub conclusion: Judgment }

/// A DERIVATION TREE: which rule, the subderivations proving each premise,
/// and the concluded judgment. Immutable; children owned (Vec, not Box, since arity varies).
pub struct Derivation { pub rule: Atom, pub subderivations: Vec<Derivation>, pub conclusion: Judgment }

/// Validate a derivation tree against the rule set: every node's (rule, subderivations,
/// conclusion) must instantiate some Rule in `rules` (premises match children's conclusions,
/// modulo the substitution the rule schema requires). Result, never panic.
pub fn check_deriv(d: &Derivation, rules: &[Rule]) -> Result<(), DerivError>;
pub struct DerivError { pub at: Atom, pub kind: DerivErrorKind }   // NoSuchRule | ArityMismatch | PremiseMismatch | ConclusionMismatch

/// EXECUTABLE type checker (algorithmic STLC): infer the unique type of `e` in `ctx`,
/// or a TypeError. `check` = infer + alpha/structural compare. Total on well-formed input.
pub struct Context { /* de Bruijn list (Vec<Type>) of variable types; grows under Lam */ }
pub fn infer(ctx: &Context, e: &Term) -> Result<Type, TypeError>;
pub fn check(ctx: &Context, e: &Term, t: &Type) -> Result<(), TypeError>;
pub enum TypeError { Unbound(usize), Mismatch{ expected: Type, found: Type }, NotAFunction(Type), NotABool, /* ... */ }

/// SMALL-STEP stepper: one reduction (beta + congruence via evaluation contexts).
/// None => `e` is a value OR a STUCK term (progress says well-typed closed terms are never stuck).
pub fn step(e: &Term) -> Option<Term>;
/// lambda-Pi (dependent) terms = types = proofs, all one syntactic category.
/// de Bruijn indices for variables. Immutable; recursive children boxed.
pub enum KTerm {
    Var(usize),                          // de Bruijn index into Ctx
    Sort(Sort),                          // a universe (Type or Kind)
    Pi(Box<KTerm>, Box<KTerm>),          // dependent function type: (x : A) -> B   (A = domain, B = codomain w/ x bound)
    Lam(Box<KTerm>, Box<KTerm>),         // \x : A. body   (A = domain annotation)
    App(Box<KTerm>, Box<KTerm>),         // f a
}

/// The two universes. Discipline: Type : Kind. (No Kind : ? — stratified, no Type:Type.)
pub enum Sort { Type, Kind }

/// Typing context: a STACK (de Bruijn list) of KTerm types, innermost last.
/// GROWS on entry to Pi/Lam, shrinks on exit. The one structure with stack lifetime.
pub struct Ctx { /* Vec<KTerm>; index i (from the right) types Var(i) */ }
impl Ctx {
    pub fn extend(&self, ty: KTerm) -> Ctx;     // push a binder type (returns extended; immutable view)
    pub fn lookup(&self, i: usize) -> Option<&KTerm>;
}

/// THE TRUSTED CHECKER. Infer (synthesize) the type of `t`, or KernelError.
/// Bidirectional: infer_type synthesizes; check_type checks against an expected type
/// via `equal`. Must be TOTAL on all KTerm inputs (terminate, never panic) and SOUND
/// (never assign a type a term does not have).
pub fn infer_type(ctx: &Ctx, t: &KTerm) -> Result<KTerm, KernelError>;
pub fn check_type(ctx: &Ctx, t: &KTerm, ty: &KTerm) -> Result<(), KernelError>;
pub enum KernelError { UnboundVar(usize), NotAFunction(KTerm), Mismatch{ expected: KTerm, found: KTerm },
                       NotASort(KTerm), IllSorted /* universe rule violated */ }

/// WEAK-HEAD NORMAL FORM: reduce only the head redex (beta) until the head ctor is
/// exposed; do NOT reduce under binders or in arguments. Used by `equal`.
pub fn whnf(t: &KTerm) -> KTerm;

/// DEFINITIONAL EQUALITY (conversion): are `a` and `b` equal up to beta (and optionally
/// eta)? Algorithm: whnf both, compare heads, recurse on subterms (extending ctx under binders).
/// This is what makes the kernel a CHECKER not a normalizer — types are compared up to `equal`.
pub fn equal(ctx: &Ctx, a: &KTerm, b: &KTerm) -> bool;
/// Build the PROPOSITION (a KType) for Progress over the given language:
///   forall e T. (|- e : T) -> (value(e) \/ exists e'. e --> e')
/// expressed in lambda-Pi (Pi for forall/implication; Sum/Sigma encoding for \/ and exists).
pub fn progress_statement(lang: &LangDef) -> KTerm;

/// Build the PROPOSITION for Preservation (subject reduction):
///   forall e e' T. (|- e : T) /\ (e --> e') -> (|- e' : T)
pub fn preservation_statement(lang: &LangDef) -> KTerm;

/// A theorem package: `term` is a proof KTerm that the kernel must accept at type `goal`.
/// VALID iff tsp_kernel::check_type(&Ctx::empty(), &p.term, &p.goal).is_ok().
pub struct Proof { pub goal: KTerm, pub term: KTerm }
/// Size-bounded generator of WELL-TYPED object terms (by construction: builds a derivation,
/// reads off the term). Seeded for reproducibility. Bounded depth/size to ensure termination.
pub struct Generator { /* rng seed, size bound, target-type strategy */ }
impl Generator {
    pub fn well_typed(&mut self, ty: &Type, size: usize) -> Option<Term>;   // None if size budget exhausted
}

/// Outcome of a property run: how many cases passed, or a counterexample + which property failed.
pub enum TestResult {
    Pass(u64),                           // u64 = number of cases that passed
    CounterExample(Term, Witness),       // the offending term + Witness describing the failure
}
pub enum Witness {
    ProgressFailed{ stuck: Term },                       // closed well-typed term with step(e)=None and !is_value(e)
    PreservationFailed{ before: Term, after: Term, ty: Type },  // |-before:ty, before-->after, but !(|-after:ty)
}

/// Property oracles over a SINGLE term (used by the harness; combine with Generator + Shrinker).
pub fn check_progress(e: &Term) -> bool;       // is_value(e) || step(e).is_some()   (for closed, well-typed e)
pub fn check_preservation(e: &Term) -> bool;   // step(e) preserves infer(e) (when both defined)

/// Delta-debugging shrinker: given a counterexample, find a structurally minimal one
/// (fewer/smaller subterms) still violating the property. Reduces witness to human-auditable size.
pub struct Shrinker { /* candidate-reduction strategy over Term */ }
