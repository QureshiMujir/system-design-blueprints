Core schema & types (§11)

pub struct Lsn(pub u64);                       // repr(transparent), Copy, Eq, Hash
impl Lsn {
    const ZERO: Lsn; const INVALID: Lsn;       // u64::MAX sentinel
    fn to_be_bytes(self) -> [u8; 8];           // Postgres-compatible wire order
    fn from_be_bytes(b: [u8; 8]) -> Self;
    fn distance_from(self, other: Lsn) -> u64; // saturating; 0 if other > self
}
// impl Ord, PartialOrd (total order, I1); impl Debug ("TL/OFFSET" hex, e.g. "0/1A2B3C4D")
// impl Add<u64>, AddAssign<u64>, Sub<Lsn> -> u64
pub struct TimelineId(pub u32);                // Copy, Ord (I4)
impl TimelineId {
    const INITIAL: TimelineId;                 // = TimelineId(1)
    fn next(self) -> TimelineId;               // bumped on every promotion (P5)
}
pub struct HistoryEntry { timeline: TimelineId, fork_lsn: Lsn, previous: TimelineId }
pub struct HistoryFile { entries: Vec<HistoryEntry> }
impl HistoryFile {
    fn current_timeline(&self) -> TimelineId;
    fn fork_lsn_of(&self, tl: TimelineId) -> Option<Lsn>;
}
pub struct ReplicationSlot {
    name: String,
    kind: SlotKind,
    restart_lsn: Lsn,             // primary will not truncate WAL before this (I6)
    confirmed_flush_lsn: Lsn,     // consumer confirmed durable receipt to here
    catalog_xmin: Option<u64>,    // logical only: oldest catalog snapshot still needed
    timeline: TimelineId,
    last_advance_at: SystemTime,
}
pub enum SlotKind { Physical, Logical { plugin: &'static str } }
pub enum ReplicationMode {
    Async,
    SyncReceive,                  // primary waits for received_lsn >= commit_lsn
    SyncReplay,                   // primary waits for replay_lsn >= commit_lsn (P2)
    Quorum(usize),                // primary waits for N acks; default floor(n/2)+1 (P13)
}
pub struct XLogRecord {
    lsn: Lsn, xid: u64, rmid: u8, info: u8, timeline: TimelineId,
    prev_lsn: Lsn, data: bytes::Bytes, crc32c: u32,
}
pub enum WireMessage {
    XLogData { start_lsn: Lsn, current_lsn: Lsn, server_time_us: i64, records: Vec<XLogRecord> },
    KeepAlive { current_lsn: Lsn, server_time_us: i64, request_reply: bool },
    StandbyStatusUpdate { received_lsn: Lsn, flushed_lsn: Lsn, replay_lsn: Lsn,
                           client_time_us: i64, request_reply: bool },
    HotStandbyFeedback { client_time_us: i64, xmin: u64, catalog_xmin: u64 },
    LogicalEvent(LogicalChangeEvent),
}
pub enum LogicalChangeEvent {
    Begin { xid: u64, commit_lsn: Lsn, commit_time_us: i64 },
    Insert { table_oid: u32, schema_version: u64, new_row: Row },
    Update { table_oid: u32, schema_version: u64, key: Row, before: Option<Row>, after: Row },
    Delete { table_oid: u32, schema_version: u64, key: Row, before: Option<Row> },
    Truncate { table_oid: u32 },
    Message { prefix: String, content: bytes::Bytes },
    Commit { xid: u64, commit_lsn: Lsn, end_lsn: Lsn },
    Schema { table_oid: u32, schema_version: u64, change: SchemaChange },
}
pub struct Row { columns: Vec<ColumnValue> }
pub struct ColumnValue { oid: u32, name: String, data: ColumnData }
pub enum ColumnData {
    Null, Integer(i64), Numeric(String), Text(String), Bytea(bytes::Bytes),
    Timestamp(i64), Bool(bool), Json(serde_json::Value), Array(Vec<ColumnData>),
    Unknown { oid: u32, raw: bytes::Bytes },
}
pub enum SchemaChange {
    AddColumn { name: String, oid: u32 }, DropColumn { name: String },
    AlterColumnType { name: String, from_oid: u32, to_oid: u32 },
    RenameColumn { old: String, new: String },
    AddIndex { name: String }, DropIndex { name: String },
    AddConstraint { name: String }, DropConstraint { name: String },
}
trait WalSender: Send + Sync {
    async fn start(&mut self, slot: &str, start_lsn: Lsn, timeline: TimelineId) -> Result<(), ReplError>;
    async fn next_frame(&mut self) -> Result<WireMessage, ReplError>;
    async fn ack(&mut self, status: StandbyStatusUpdate) -> Result<(), ReplError>;
    fn current_lsn(&self) -> Lsn;
}
trait WalReceiver: Send + Sync {
    async fn connect(&mut self, slot: &str) -> Result<(), ReplError>;
    async fn recv_frame(&mut self) -> Result<WireMessage, ReplError>;
    async fn send_status(&mut self, status: StandbyStatusUpdate) -> Result<(), ReplError>;
    fn received_lsn(&self) -> Lsn; fn flushed_lsn(&self) -> Lsn; fn replay_lsn(&self) -> Lsn;
}
trait LogicalDecoder: Send {
    fn feed(&mut self, record: &XLogRecord) -> Result<Vec<LogicalChangeEvent>, DecodeError>;
}
trait OutputPlugin: Send {
    async fn emit(&mut self, event: &LogicalChangeEvent) -> Result<bytes::Bytes, OutputError>;
    fn name(&self) -> &'static str;
    fn schema_version(&self) -> u64;
}
trait SlotStore: Send + Sync {
    async fn create(&self, slot: ReplicationSlot) -> Result<(), SlotError>;
    async fn drop(&self, name: &str) -> Result<(), SlotError>;
    async fn advance(&self, name: &str, confirmed_flush_lsn: Lsn, restart_lsn: Lsn) -> Result<(), SlotError>;
    async fn get(&self, name: &str) -> Result<ReplicationSlot, SlotError>;
    async fn list(&self) -> Result<Vec<ReplicationSlot>, SlotError>;
    async fn min_restart_lsn(&self) -> Result<Lsn, SlotError>;   // enforces I6
}
pub struct Frame { kind: u8, payload: bytes::Bytes, crc32c: u32 }
pub const MAX_PAYLOAD: usize = 16 * 1024 * 1024;   // 16 MiB (§4, §8.2)

pub enum ReplError { Io(..), Protocol(..), CrcMismatch, SlotNotFound, Timeline(..) }
pub enum SlotError { NotFound, InvariantViolation, Io(..), Corrupt }
pub enum DecodeError { UnknownRmid(u8), Truncated, SchemaMismatch }
pub enum OutputError { Serialize(..), SinkUnavailable }
pub enum FailoverError { NotFenced, LsnMismatch, HistoryWriteFailed }
