Core schema & types (§11)

struct ObjectMeta { name: String, namespace: String, resource_version: u64,
                    generation: u64, labels: BTreeMap<String,String>,
                    annotations: BTreeMap<String,String>, created_at: DateTime<Utc> }

struct Rollout { metadata: ObjectMeta, spec: RolloutSpec, status: RolloutStatus }
// ownership: Arc<RwLock<Rollout>> in store; cloned into reconcile
// persistence: serialized as JSON in store; version-fenced by resource_version

struct RolloutSpec { replicas: u32, selector: LabelSelector, template: PodTemplate,
                     strategy: Strategy, revision_history_limit: u32,
                     min_ready_seconds: u32, progress_deadline_seconds: u32, paused: bool }

enum Strategy { Canary(CanaryStrategy), BlueGreen(BlueGreenStrategy), Rolling(RollingStrategy) }
// concurrency: read-only after admission; mutation only via spec-patch through API

struct CanaryStrategy { steps: Vec<CanaryStep>, traffic_routing: Option<TrafficRouting>,
                        max_surge: IntOrPercent, max_unavailable: IntOrPercent,
                        analysis: Option<AnalysisRef>, canary_service: String,
                        stable_service: String }

enum CanaryStep { SetWeight(u32), Pause { duration: Option<Duration> },
                  Analysis(AnalysisRef), SetCanaryScale(ScaleSpec),
                  Experiment(ExperimentSpec) }
// invariant: SetWeight value in 0..=100

struct BlueGreenStrategy { active_service: String, preview_service: String,
                            auto_promotion_enabled: bool, auto_promotion_seconds: u32,
                            scale_down_delay_seconds: u32,
                            pre_promotion_analysis: Option<AnalysisRef>,
                            post_promotion_analysis: Option<AnalysisRef> }

struct RollingStrategy { max_surge: IntOrPercent, max_unavailable: IntOrPercent,
                          min_ready_seconds: u32 }
struct RolloutStatus { phase: Phase, current_step_index: u32,
                       stable_rs: Option<String>, canary_rs: Option<String>,
                       observed_generation: u64, replicas: u32, updated_replicas: u32,
                       ready_replicas: u32, available_replicas: u32,
                       conditions: Vec<Condition>, message: String,
                       abort: bool, pause_conditions: Vec<PauseCondition>,
                       last_progress_time: Option<DateTime<Utc>> }
// mutation: written by reconcile only; CAS via resource_version

enum Phase { Progressing, Paused, Healthy, Degraded, Aborted }
// transition: machine.rs enforces valid edges

struct Condition { condition_type: String, status: ConditionStatus,
                   reason: String, message: String, last_transition_time: DateTime<Utc> }
struct AnalysisTemplate { metadata: ObjectMeta, metrics: Vec<Metric>,
                           args: Vec<Argument> }

struct Metric { name: String, interval: Duration, count: Option<u32>,
                initial_delay: Option<Duration>, success_condition: Option<String>,
                failure_condition: Option<String>, failure_limit: u32,
                inconclusive_limit: u32, provider: ProviderRef }

struct AnalysisRun { metadata: ObjectMeta, spec: AnalysisRunSpec,
                      status: AnalysisRunStatus }
struct AnalysisRunStatus { phase: AnalysisPhase, metric_results: Vec<MetricResult>,
                            started_at: DateTime<Utc>, completed_at: Option<DateTime<Utc>> }

enum AnalysisPhase { Pending, Running, Successful, Failed, Inconclusive, Error }

struct MetricResult { name: String, phase: AnalysisPhase, measurements: Vec<Measurement>,
                      success_count: u32, failure_count: u32, inconclusive_count: u32 }

struct Measurement { value: serde_json::Value, phase: AnalysisPhase,
                     started_at: DateTime<Utc>, finished_at: DateTime<Utc>,
                     message: Option<String> }

struct TrafficRouting { provider: TrafficProviderRef, stable_weight: u32,
                         canary_weight: u32, managed_routes: Vec<ManagedRoute> }
// invariant: stable_weight + canary_weight == 100

struct Revision { number: u64, pod_template_hash: String, replica_set: String,
                   created_at: DateTime<Utc> }

enum IntOrPercent { Int(u32), Percent(u32) }
#[async_trait] trait Store: Send + Sync {
    async fn get(&self, name: &str) -> Result<Rollout, StoreError>;
    async fn put(&self, rollout: Rollout) -> Result<Rollout, StoreError>; // CAS on resource_version
    async fn delete(&self, name: &str) -> Result<(), StoreError>;
    async fn list(&self) -> Result<Vec<Rollout>, StoreError>;
    async fn watch(&self, name: &str) -> Result<BoxStream<'_, WatchEvent>, StoreError>;
}
enum StoreError { NotFound(String), StaleGeneration{expected:u64,got:u64}, Conflict, Io(io::Error) }

struct RevisionRing { limit: u32, entries: VecDeque<Revision> }
// push() → prune oldest if len > limit

struct EventLog; // append(Event) + query(filter: EventFilter) + replay() → Vec<Event>
enum Action {
    ScaleWorkload { rs: String, replicas: u32 },
    SetWeight { stable: u32, canary: u32 },
    StartAnalysis { template: AnalysisRef, run_name: String },
    AdvanceStep { to_index: u32 },
    Promote { canary_rs: String },
    Abort { reason: String },
    SetCondition { condition: Condition },
}
// all variants: idempotent when applied multiple times

struct ActionSet(Vec<Action>);
// reconcile() → ActionSet; apply_actions() drains sequentially

struct ReconcileContext<'a> { rollout: &'a Rollout, store: &'a dyn Store,
    workload: &'a dyn WorkloadProvider, traffic: &'a dyn TrafficRouter,
    analysis: &'a dyn AnalysisRuntime, events: &'a dyn EventRecorder }
#[async_trait] trait WorkloadProvider: Send + Sync {
    async fn create_or_update(&self, rs: &ReplicaSetSpec) -> Result<String, WorkloadError>;
    async fn scale(&self, rs: &str, replicas: u32) -> Result<(), WorkloadError>;
    async fn ready_count(&self, rs: &str) -> Result<u32, WorkloadError>;
    async fn pod_template_hash(&self, template: &PodTemplate) -> String; // SHA-256
    async fn delete(&self, rs: &str) -> Result<(), WorkloadError>;
}
struct SurgeCalc; // compute_surge(desired:u32, max_surge:IntOrPercent, max_unavail:IntOrPercent) → (max_pods, min_avail)
#[async_trait] trait TrafficRouter: Send + Sync {
    async fn set_weights(&self, stable: u32, canary: u32) -> Result<(), TrafficError>;
    async fn get_weights(&self) -> Result<(u32, u32), TrafficError>;
    async fn set_header_route(&self, rule: HeaderRouteRule) -> Result<(), TrafficError>;
    async fn set_mirror(&self, fraction: u32) -> Result<(), TrafficError>;
    async fn remove_canary_routes(&self) -> Result<(), TrafficError>;
}
struct WeightVerifier; // verify(router) → Option<DriftCorrection>
#[async_trait] trait MetricProvider: Send + Sync {
    async fn query_instant(&self, query: &str, time: DateTime<Utc>) -> Result<serde_json::Value, AnalysisError>;
    async fn query_range(&self, query: &str, start: DateTime<Utc>, end: DateTime<Utc>, step: Duration) -> Result<Vec<(DateTime<Utc>, f64)>, AnalysisError>;
}

#[async_trait] trait AnalysisRuntime: Send + Sync {
    async fn start(&self, run: &AnalysisRun) -> Result<(), AnalysisError>;
    async fn poll(&self, run_name: &str) -> Result<AnalysisRunStatus, AnalysisError>;
    async fn terminate(&self, run_name: &str) -> Result<(), AnalysisError>;
}

struct VerdictEvaluator; // evaluate(expr: &str, value: &serde_json::Value) → bool (Rhai)
#[async_trait] trait EventRecorder: Send + Sync {
    async fn record(&self, event: Event) -> Result<(), EventError>;
}
struct Event { id: Uuid, event_type: EventType, reason: String, message: String,
               involved_object: ObjectRef, timestamp: DateTime<Utc>,
               cloud_event_version: String }
enum EventType { Normal, Warning }
struct WebhookConfig { url: Url, timeout: Duration, max_retries: u32, secret: Option<String> }
