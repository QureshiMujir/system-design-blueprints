Core schema & types (§11)

type PageId = u32;                    // copy; immutable once assigned; persisted; stable across versions; no recovery role itself
type Lsn = u64;                        // copy; monotonic (I9); persisted in every page/record; append-only compat; central to recovery ordering
type TxnId = u64;                       // copy; allocated once per txn; persisted in commit log; stable; recovery reads it from WAL
type Timestamp = i64;                    // copy; wall-clock µs (ADR-38); persisted in TIMESTAMP columns only; no recovery role
struct Page { header: PageHeader, bytes: [u8; 8192] } // owned by a Frame; single-writer under page latch; persisted; versioned by ADR-27; redo target
struct PageHeader { page_id: PageId, lsn: Lsn, checksum: u32, slot_count: u16, free_space_ptr: u16 } // embedded in Page; mutated under latch; persisted; I5/I9 anchor
type SlotId = u16;                      // copy; stable identity component; persisted; no recovery role
struct Slot { offset: u16, length: u16 } // embedded in Page's slot array; mutated under page latch; persisted; I5 anchor
struct TupleId /* Rid */ { page_id: PageId, slot_id: SlotId } // copy; stable row identity; persisted in indexes; recovery-independent
struct Tuple { null_bitmap: NullBitmap, fields: Vec<Datum> } // owned by caller; immutable once built; serialized per §8 tuple encoding; no direct recovery role
struct TupleDesc { columns: Vec<(String, SqlType, bool)> } // owned by Catalog; immutable per schema version; persisted via sys_columns; versioned by ADR-27
enum Datum { Null, Bool(bool), I32(i32), I64(i64), F64(f64), Text(String), Timestamp(Timestamp) } // owned by Tuple; immutable; serialized per §8; no recovery role
enum SqlType { Bool, Int4, Int8, Float8, Numeric(u8,u8), Varchar(u32), Text, Timestamp } // owned by Catalog; immutable; persisted; ADR-27 versioned
struct NullBitmap { bits: Vec<u8> } // owned by Tuple; immutable once built; persisted; no recovery role
struct Schema { columns: Vec<TupleDesc> } // owned by Catalog/LogicalPlan node; immutable; not directly persisted (derived)
type TableId = u32;                      // copy; stable; persisted in sys_tables; recovery-independent
struct TableMeta { table_id: TableId, name: String, first_page_id: PageId } // owned by Catalog; mutated only via DDL txn (P10); persisted; MVCC-versioned
type IndexId = u32;                       // copy; stable; persisted in sys_indexes
struct IndexMeta { index_id: IndexId, table_id: TableId, root_page_id: PageId, is_unique: bool, is_primary: bool } // owned by Catalog; DDL-txn-mutated; persisted; MVCC-versioned
struct Catalog { tables: HashMap<TableId, TableMeta>, indexes: HashMap<IndexId, IndexMeta> } // shared, read-mostly, MVCC-consistent snapshot per txn; persisted via sys_* tables; I11 anchor
struct BufferPool { frames: Vec<Frame>, page_table: HashMap<PageId, usize>, policy: ClockSweep } // shared, latched per-frame; not persisted; owns eviction decisions; I1 gatekeeper
struct Frame { page: Page, pin_count: u32, dirty: bool } // owned by BufferPool; mutated under frame latch; not persisted directly; P6 anchor
struct PinGuard<'a> { frame: &'a Frame } // borrowed, RAII; enforces P6 at compile time; not persisted; no recovery role
struct Pager { file: File, next_page_id: PageId } // owned singleton per heap/index file; single-writer via BufferPool serialization; persisted target; I1/I9 enforcement point
struct HeapFile { pager: Pager, first_page_id: PageId } // owned by table; mutated under buffer-pool-mediated latches; persisted; redo target
struct BTree { pager: Pager, root_page_id: PageId } // owned by index; concurrent via ADR-08; persisted; redo target, I4 anchor
struct BTreeNode { header: PageHeader, entries: Vec<(Datum, PageId)> } // internal-node view over Page; latched; persisted; I4 anchor
struct LeafNode { header: PageHeader, entries: Vec<(Datum, TupleId)>, next_leaf: PageId } // leaf-node view over Page; latched; persisted; I4 anchor
struct InternalNode { header: PageHeader, entries: Vec<(Datum, PageId)> } // alias view of BTreeNode for internal pages; latched; persisted
struct WalRecord { lsn: Lsn, prev_lsn: Lsn, txn_id: TxnId, kind: WalRecordType, page_id: PageId, before: Vec<u8>, after: Vec<u8> } // owned by WalWriter until flushed; immutable once written; persisted append-only; I1/I9/I2 core
enum WalRecordType { Update, Commit, Abort, Checkpoint, Clr } // copy; persisted tag; ADR-11 format; recovery dispatch key
struct WalWriter { buffer: Vec<WalRecord>, last_flushed_lsn: Lsn } // owned singleton; single-writer, group-commit batched (P13); persisted target; I1 enforcement point
struct CheckpointRecord { dirty_page_table: HashMap<PageId, Lsn>, active_txn_table: HashMap<TxnId, Lsn> } // owned by checkpoint process; persisted as a WalRecord payload; I12 anchor
struct RecoveryState { dpt: HashMap<PageId, Lsn>, att: HashMap<TxnId, Lsn>, redo_lsn: Lsn } // owned by RecoveryManager during recovery only; transient; drives redo/undo; I2 anchor
struct Transaction { txn_id: TxnId, snapshot: Snapshot, state: TxnState } // owned by Session; mutated only by its own thread until commit/abort; commit record persisted; P7 anchor
enum TxnState { Active, Committing, Committed, Aborted } // copy; one-way transition; persisted via commit log; I2 anchor
struct Snapshot { xmin_horizon: TxnId, visible_xids: HashSet<TxnId> } // owned by Transaction; immutable after creation (P7); not persisted, rebuilt from ATT on recovery; I3 anchor
struct VersionChain { versions: Vec<(TxnId /*xmin*/, Option<TxnId> /*xmax*/, Tuple)> } // shared per row, append/prune under vacuum coordination; persisted inline in heap pages; I6/I7 anchor
struct UndoRecord { txn_id: TxnId, tuple_id: TupleId, before: Tuple } // owned by transaction manager during active txn; persisted via WAL Update records; supports abort and ARIES undo
enum LockMode { Shared, Exclusive } // copy; persisted only transiently in LockTable; 2PL-path concurrency artifact
struct LockRequest { txn_id: TxnId, tuple_id: TupleId, mode: LockMode } // owned by LockTable queue; not persisted; concurrency-only
struct LockTable { held: HashMap<TupleId, Vec<LockRequest>>, waiting: Vec<LockRequest> } // shared, mutex-guarded; not persisted; deadlock-detection input
struct WaitsForGraph { edges: HashMap<TxnId, Vec<TxnId>> } // rebuilt periodically from LockTable; not persisted; ADR-19 anchor
struct Token { kind: TokenKind, text: String, pos: u32 } // owned by Lexer output stream; immutable; not persisted; syntax-boundary artifact
enum Ast { Select(SelectNode), Insert(InsertNode), Update(UpdateNode), Delete(DeleteNode), CreateTable(CreateTableNode), /* ... */ } // owned by Parser output; immutable; not persisted
struct BoundExpr { ty: SqlType, kind: BoundExprKind } // owned by Binder output; immutable; not persisted; type boundary artifact
enum LogicalPlan { Scan(TableId), Filter(Box<LogicalPlan>, BoundExpr), Project(Box<LogicalPlan>, Vec<BoundExpr>), Join(Box<LogicalPlan>, Box<LogicalPlan>, BoundExpr), Aggregate(Box<LogicalPlan>, Vec<BoundExpr>), Sort(Box<LogicalPlan>, Vec<BoundExpr>), Limit(Box<LogicalPlan>, u64) } // owned by Planner; immutable once built, rewritten by producing new trees; not persisted; I8 anchor
enum PhysicalPlan { SeqScan(TableId), IndexScan(IndexId), Filter(Box<PhysicalPlan>, BoundExpr), Project(Box<PhysicalPlan>, Vec<BoundExpr>), NestedLoopJoin(Box<PhysicalPlan>, Box<PhysicalPlan>), HashJoin(Box<PhysicalPlan>, Box<PhysicalPlan>), SortMergeJoin(Box<PhysicalPlan>, Box<PhysicalPlan>), HashAggregate(Box<PhysicalPlan>), Sort(Box<PhysicalPlan>), Limit(Box<PhysicalPlan>, u64) } // owned by Optimizer output; immutable; not persisted; I8 anchor
trait Executor { fn open(&mut self) -> Result<()>; fn next(&mut self) -> Result<Option<Tuple>>; fn close(&mut self) -> Result<()>; } // implemented per PhysicalPlan node; owns operator-local state; not persisted; P11 anchor
struct CostEstimate { cpu: f64, io: f64, rows: f64 } // owned by Optimizer during search; not persisted; comparative only (P12)
struct Statistics { row_count: u64, histograms: HashMap<String, Histogram>, ndv: HashMap<String, u64> } // owned by Catalog, written only by ANALYZE; persisted; ADR-24 anchor
struct Histogram { buckets: Vec<(Datum, Datum, u64)> } // owned by Statistics; immutable per collection cycle; persisted; equi-depth (ADR-24)
struct Session { conn_id: u64, txn: Option<Transaction>, prepared: HashMap<String, LogicalPlan> } // owned by connection handler thread; mutated by its own thread only; not persisted
enum ProtocolMessage { Startup, Query(String), Parse(String,String), Bind(String,Vec<Datum>), Execute(String), RowDescription(Schema), DataRow(Tuple), CommandComplete(u64), ErrorResponse(String,String,u32), ReadyForQuery } // owned by connection codec; immutable per message; wire-serialized; ADR-29 anchor
struct SqlError { code: String /* SQLSTATE-style */, message: String, position: Option<u32> } // owned by any layer that raises it; immutable; surfaced via ErrorResponse; no recovery role
