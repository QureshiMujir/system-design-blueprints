Core schema & types (§11)

pub type Scalar = f64;
pub struct Vector(pub Vec<Scalar>);
pub struct DenseMatrix { pub rows: usize, pub cols: usize, pub data: Vec<Scalar> } // row-major

pub enum Norm { L1, L2, LInf }
pub fn norm(v: &Vector, kind: Norm) -> Scalar;
pub fn axpy(a: Scalar, x: &Vector, y: &mut Vector);   // y += a x
pub fn dot(x: &Vector, y: &Vector) -> Scalar;

pub enum NmError {
    SingularMatrix,                          // I8 (P9)
    NonConvergence { iters: usize, residual: Scalar }, // P7
    CflViolation { dt: Scalar, dt_max: Scalar },       // I5
    NotSpd,                                  // I2
    DimensionMismatch { expected: usize, got: usize },
    Mesh { msg: String },                    // degenerate element, etc.
    NonFinite,                               // I12
}
// dense direct
pub fn lu_factor(a: &mut DenseMatrix) -> Result<Vec<usize>, NmError>;   // partial pivot
pub fn lu_solve(a: &DenseMatrix, piv: &[usize], b: &Vector) -> Vector;
pub fn cholesky(a: &DenseMatrix) -> Result<DenseMatrix, NmError>;        // SPD (I2)
pub fn qr(a: &DenseMatrix) -> (DenseMatrix, DenseMatrix);                // Householder
pub fn thomas(sub: &[Scalar], diag: &[Scalar], sup: &[Scalar], rhs: &Vector) -> Vector; // O(n)

// sparse
pub struct CsrMatrix { pub n: usize, pub row_ptr: Vec<usize>,
                       pub col_idx: Vec<usize>, pub val: Vec<Scalar> }
pub struct CooBuilder { /* (i,j,v) triplets */ }
impl CooBuilder { pub fn add(&mut self, i: usize, j: usize, v: Scalar); pub fn to_csr(self) -> CsrMatrix; }
pub fn spmv(a: &CsrMatrix, x: &Vector, y: &mut Vector);   // y = A x (P3, rayon rows)
pub fn is_symmetric(a: &CsrMatrix, tol: Scalar) -> bool;  // solver routing (P4)
pub trait Preconditioner { fn apply(&self, r: &Vector) -> Vector; }   // M^{-1} r
pub struct Jacobi; pub struct Ssor { pub omega: Scalar }; pub struct Ilu0 { /* factors */ }

pub struct SolveOptions { pub tol: Scalar, pub max_iter: usize, pub restart: usize }
pub struct SolveReport { pub iters: usize, pub residual: Scalar, pub history: Vec<Scalar> } // P10

pub fn cg(a: &CsrMatrix, b: &Vector, m: &dyn Preconditioner, opt: &SolveOptions)
    -> Result<(Vector, SolveReport), NmError>;     // SPD (P4); true-residual check (I7)
pub fn gmres(a: &CsrMatrix, b: &Vector, m: &dyn Preconditioner, opt: &SolveOptions)
    -> Result<(Vector, SolveReport), NmError>;     // nonsymmetric; Arnoldi+Givens
pub fn bicgstab(/* … */) -> Result<(Vector, SolveReport), NmError>;
pub fn minres(/* … */) -> Result<(Vector, SolveReport), NmError>;
pub struct OdeProblem<'a> {
    pub rhs: &'a dyn Fn(Scalar, &Vector) -> Vector,           // f(t,y)
    pub jac: Option<&'a dyn Fn(Scalar, &Vector) -> DenseMatrix>, // optional ∂f/∂y
    pub y0: Vector, pub tspan: (Scalar, Scalar),
}
pub struct ButcherTableau { pub a: Vec<Vec<Scalar>>, pub b: Vec<Scalar>,
                            pub c: Vec<Scalar>, pub b_hat: Option<Vec<Scalar>> }
pub enum Method { Euler, Rk4, Rk45, BackwardEuler, CrankNicolson, Bdf2 }
pub struct StepControl { pub atol: Scalar, pub rtol: Scalar,
                         pub dt_min: Scalar, pub dt_max: Scalar }   // I10
pub struct OdeSolution { pub t: Vec<Scalar>, pub y: Vec<Vector>,
                         pub n_accept: usize, pub n_reject: usize } // P10

pub fn integrate(p: &OdeProblem, method: Method, ctrl: &StepControl)
    -> Result<OdeSolution, NmError>;   // implicit methods Newton-solve stages
pub fn dense_output(sol: &OdeSolution, t: Scalar) -> Vector;       // interpolation
pub struct Grid1D { pub x: Vec<Scalar>, pub bc: [BoundaryCondition; 2] }
pub struct Grid2D { pub nx: usize, pub ny: usize, pub x: Vec<Scalar>, pub y: Vec<Scalar> }
pub enum BoundaryCondition { Dirichlet(Scalar), Neumann(Scalar),
                             Robin{alpha: Scalar, beta: Scalar, g: Scalar}, Periodic }

pub fn laplacian_1d(g: &Grid1D) -> CsrMatrix;        // tridiagonal SPD (I2)
pub fn laplacian_2d(g: &Grid2D) -> CsrMatrix;        // 5-point
pub fn apply_bc(a: &mut CsrMatrix, b: &mut Vector, g: &Grid1D);  // symmetric (I6)
pub fn mol_rhs(g: &Grid2D, alpha: Scalar) -> impl Fn(Scalar, &Vector) -> Vector; // §8.8
pub struct Mesh { pub nodes: Vec<[Scalar;2]>, pub elements: Vec<[usize;3]>, // triangles
                  pub boundary: Vec<(usize, BoundaryCondition)> }
pub enum ElementKind { P1, P2 }
pub struct QuadratureRule { pub points: Vec<[Scalar;2]>, pub weights: Vec<Scalar> } // I4
pub fn shape_functions(kind: ElementKind, xi: [Scalar;2]) -> (Vec<Scalar>, Vec<[Scalar;2]>); // N, ∇N (I3)
pub fn element_stiffness(mesh: &Mesh, e: usize, kind: ElementKind, q: &QuadratureRule) -> DenseMatrix;
pub fn element_mass(/* … */) -> DenseMatrix;
pub fn assemble(mesh: &Mesh, kind: ElementKind, f: &dyn Fn([Scalar;2]) -> Scalar)
    -> (CsrMatrix, CsrMatrix, Vector);   // (K, M, b); scatter-add (§8.7)
pub fn impose_bc(k: &mut CsrMatrix, b: &mut Vector, mesh: &Mesh);  // symmetric Dirichlet (I6)
pub enum PdeKind { Poisson, Heat, Wave, AdvectionDiffusion, Burgers }
pub struct PdeProblem { pub kind: PdeKind, pub discretization: Discretization,
                        pub bcs: Vec<BoundaryCondition>, pub source: Source,
                        pub time: Option<TimeSpec> }
pub enum Discretization { Fd(GridSpec), Fem(MeshSpec, ElementKind) }
pub struct Solution { pub values: Vector, pub coords: Vec<[Scalar;2]> }
pub fn solve(p: &PdeProblem) -> Result<Solution, NmError>;

// verification (P1, P8)
pub fn error_norm(numeric: &Solution, exact: &dyn Fn([Scalar;2]) -> Scalar, kind: Norm) -> Scalar;
pub fn h1_error(/* … */) -> Scalar;
pub fn mms_source(exact: &dyn Fn([Scalar;2]) -> Scalar, kind: PdeKind) -> Source; // derive f
pub struct ConvergenceStudy { pub h: Vec<Scalar>, pub err: Vec<Scalar>, pub order: Scalar }
pub fn estimate_order(study: &ConvergenceStudy) -> Scalar;   // log-log slope (I1)
pub fn richardson(coarse: Scalar, fine: Scalar, ratio: Scalar, order: Scalar) -> Scalar;
