Core schema & types (§11)

#[derive(Debug, Clone, PartialEq, Eq, Hash, PartialOrd, Ord)]
pub struct Label {
    pub repo:    Option<String>,   // None => implicit workspace repo
    pub package: String,           // e.g. "pkg/sub"
    pub name:    String,           // e.g. "target"
}

impl Label {
    pub fn parse(raw: &str) -> Result<Self, LabelError>;
    pub fn canonical(&self) -> String;           // "//pkg/sub:name"
    pub fn is_relative(&self) -> bool;
}

impl std::str::FromStr for Label { type Err = LabelError; }
impl std::fmt::Display   for Label {}
#[derive(Debug, Copy, Clone, PartialEq, Eq, Hash, PartialOrd, Ord)]
pub struct TargetId(pub u32);
#[derive(Debug, Copy, Clone, PartialEq, Eq, Hash, PartialOrd, Ord)]
pub struct Digest {
    pub hash:       [u8; 32],      // raw sha256 bytes
    pub size_bytes: u64,
}

impl Digest {
    pub fn of_bytes(data: &[u8]) -> Self;
    pub fn hex(&self) -> String;                // lowercase hex of hash
    pub fn verify(&self, data: &[u8]) -> bool;  // sha256(data) == self.hash && len == size_bytes
}

impl std::fmt::Display for Digest {}            // "<hex>/<size>"
impl std::str::FromStr  for Digest { type Err = DigestError; }
#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub enum ArtifactKind { Source, Generated }

#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct ArtifactRef {
    pub path:   PathBuf,
    pub digest: Digest,
    pub kind:   ArtifactKind,
    pub owner:  Option<ActionKey>,   // None for Source; Some for Generated
}
#[derive(Debug, Clone)]
pub struct Action {
    pub mnemonic:  String,                      // human label, e.g. "CcCompile"
    pub command:   Vec<String>,                  // argv[0] = tool path or name
    pub inputs:    Vec<ArtifactRef>,             // all declared input files
    pub outputs:   Vec<PathBuf>,                 // declared output paths (relative to exec root)
    pub env:       BTreeMap<String, String>,     // deterministic; volatile vars stripped
    pub platform:  Digest,                       // digest of Platform proto/json blob
}
#[derive(Debug, Copy, Clone, PartialEq, Eq, Hash, PartialOrd, Ord)]
pub struct ActionKey(pub Digest);

impl ActionKey {
    pub fn compute(action: &Action, toolchain_digest: Digest) -> Self;
    pub fn hex(&self) -> String;
}
#[derive(Debug, Clone)]
pub struct ActionResult {
    pub exit_code:       i32,
    pub output_digests:  BTreeMap<PathBuf, Digest>,   // declared output path -> CAS digest
    pub stdout:          Digest,
    pub stderr:          Digest,
    pub metadata:        ActionMetadata,
}

#[derive(Debug, Clone)]
pub struct ActionMetadata {
    pub execution_duration_ms: u64,
    pub worker_id:             Option<String>,
    pub cache_hit:             bool,
    pub remote:                bool,
}
#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct RuleClass {
    pub name:       String,                         // e.g. "cc_binary"
    pub attrs:      Vec<AttrSchema>,                // ordered attribute definitions
    pub providers:  Vec<TypeId>,                    // provider types this rule produces
}

#[derive(Debug, Clone)]
pub struct AttrSchema {
    pub name:       String,
    pub kind:       AttrKind,                       // String | Label | LabelList | Bool | Int | StringList
    pub required:   bool,
    pub default:    Option<AttrValue>,
}
#[derive(Debug, Clone)]
pub struct Target {
    pub label:      Label,
    pub rule:       RuleClass,
    pub attrs:      BTreeMap<String, AttrValue>,
    pub deps:       Vec<Label>,
    pub toolchains: Vec<Label>,
}
pub trait Provider: std::any::Any + Send + Sync + std::fmt::Debug {
    fn provider_id() -> TypeId where Self: Sized;
    fn as_any(&self) -> &dyn std::any::Any;
}

// Envelope stored in ProviderMap keyed by TypeId
pub struct ProviderMap(HashMap<TypeId, Box<dyn Provider>>);

impl ProviderMap {
    pub fn get<P: Provider>(&self) -> Option<&P>;
    pub fn insert<P: Provider>(&mut self, p: P);
}
#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct ConstraintValue {
    pub setting: Label,    // e.g. //platforms:os
    pub value:   Label,    // e.g. //platforms:linux
}

#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct Platform {
    pub label:       Label,
    pub constraints: BTreeSet<ConstraintValue>,
}

impl Platform {
    pub fn digest(&self) -> Digest;   // stable content-hash of sorted constraint set
    pub fn satisfies(&self, required: &[ConstraintValue]) -> bool;
}
#[derive(Debug, Clone)]
pub struct Toolchain {
    pub lang:            String,
    pub exec_platform:   Digest,
    pub target_platform: Digest,
    pub tools:           BTreeMap<String, ArtifactRef>,  // tool name -> binary ArtifactRef
}

impl Toolchain {
    pub fn digest(&self) -> Digest;    // content-hash of exec+target platform + sorted tool digests
}
#[derive(Debug, Clone)]
pub struct CacheEntry {
    pub action_key: ActionKey,
    pub result:     ActionResult,
    pub created_at: u64,             // unix timestamp (seconds) — informational only, not part of key
}
#[derive(Debug, Clone)]
pub enum SandboxIsolation { SymlinkForest, Namespace, SandboxExec }

#[derive(Debug, Clone)]
pub struct SandboxSpec {
    pub input_root:    PathBuf,              // materialized input directory
    pub allowed_paths: Vec<PathBuf>,         // explicit allow-list (tools, /usr/lib, etc.)
    pub output_paths:  Vec<PathBuf>,         // paths to capture post-execution
    pub env:           BTreeMap<String, String>,
    pub isolation:     SandboxIsolation,
}
#[derive(Debug, Clone)]
pub struct ExecRequest {
    pub action_key:        ActionKey,
    pub input_root_digest: Digest,       // CAS digest of the input-root directory tree
    pub action_digest:     Digest,       // CAS digest of the serialized Action blob
    pub skip_cache:        bool,
}

#[derive(Debug, Clone)]
pub struct ExecResponse {
    pub action_key: ActionKey,
    pub result:     ActionResult,
    pub cached:     bool,
    pub error:      Option<ExecError>,
}
