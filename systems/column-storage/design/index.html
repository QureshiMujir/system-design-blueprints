pub enum DataType {
    Null, Boolean,
    Int8, Int16, Int32, Int64, UInt8, UInt16, UInt32, UInt64,
    Float16, Float32, Float64,
    Decimal128 { precision: u8, scale: i8 },
    Utf8, LargeUtf8, Binary, LargeBinary,
    Date32, Date64,
    Timestamp { unit: TimeUnit, tz: Option<Arc<str>> },
    Duration(TimeUnit), Interval(IntervalUnit),
    List(Arc<Field>), LargeList(Arc<Field>), FixedSizeList(Arc<Field>, i32),
    Struct(Vec<Arc<Field>>),
    Dictionary { keys: Box<DataType>, values: Box<DataType> },
    RunEndEncoded { run_ends: Box<DataType>, values: Box<DataType> },
}
pub enum TimeUnit { Second, Millisecond, Microsecond, Nanosecond }
pub enum IntervalUnit { YearMonth, DayTime, MonthDayNano }
pub struct Field { pub name: Arc<str>, pub data_type: DataType, pub nullable: bool, pub metadata: Vec<(Arc<str>, Arc<str>)> }
pub struct Buffer { bytes: Arc<AlignedBytes>, offset: usize, len: usize }
struct AlignedBytes { ptr: *mut u8, cap: usize, layout: std::alloc::Layout }

impl Buffer {
    pub fn from_vec<T: bytemuck::Pod>(v: Vec<T>) -> Self;
    pub fn as_slice(&self) -> &[u8];
    pub fn slice(&self, offset: usize, len: usize) -> Self;
    pub fn typed<T: bytemuck::Pod>(&self) -> &[T];
}

/// One bit per logical row, LSB-first within each byte (Arrow convention), 64-byte aligned.
pub struct Validity { bits: Buffer, null_count: usize, len: usize }
impl Validity {
    pub fn new_valid(len: usize) -> Self;
    pub fn is_valid(&self, idx: usize) -> bool;
    pub fn null_count(&self) -> usize;
    pub fn len(&self) -> usize;
    pub fn slice(&self, offset: usize, len: usize) -> Self;
}

/// Offsets for variable-length arrays. O = i32 (small) or i64 (large). Exactly len+1 entries (I4).
pub struct Offsets<O: OffsetSize> { buffer: Buffer, _marker: PhantomData<O> }
pub trait OffsetSize: bytemuck::Pod + Copy + Send + Sync + 'static {
    fn from_usize(v: usize) -> Self;
    fn to_usize(self) -> usize;
}
pub trait Array: Send + Sync + std::fmt::Debug {
    fn data_type(&self) -> &DataType;
    fn len(&self) -> usize;
    fn is_empty(&self) -> bool;
    fn null_count(&self) -> usize;
    fn validity(&self) -> Option<&Validity>;
    fn slice(&self, offset: usize, len: usize) -> ArrayRef;
    fn as_any(&self) -> &dyn std::any::Any;
}
pub type ArrayRef = Arc<dyn Array>;

pub struct RecordBatch { schema: Arc<Schema>, columns: Vec<ArrayRef>, num_rows: usize }
impl RecordBatch {
    /// Rejects mismatched column types/lengths at construction (I13).
    pub fn try_new(schema: Arc<Schema>, columns: Vec<ArrayRef>) -> Result<Self, Error>;
    pub fn schema(&self) -> &Arc<Schema>;
    pub fn column(&self, idx: usize) -> &ArrayRef;
    pub fn column_by_name(&self, name: &str) -> Result<&ArrayRef, Error>;
    pub fn num_rows(&self) -> usize;
    pub fn slice(&self, offset: usize, len: usize) -> Self;
    pub fn project(&self, indices: &[usize]) -> Self;   // zero-copy
}
pub enum SelectionVector {
    Full { len: usize },
    Dense { bits: Buffer, len: usize, popcount: usize },
    Roaring { bitmap: roaring::RoaringBitmap, len: usize },

impl SelectionVector {
    pub fn len(&self) -> usize;
    pub fn popcount(&self) -> usize;
    pub fn iter(&self) -> SelectionIter<'_>;
    pub fn intersect(&self, other: &Self) -> Self;
    pub fn union(&self, other: &Self) -> Self;
}

pub enum KernelLevel { Scalar, Avx2, Avx512, Neon }
impl KernelLevel { pub fn detect() -> Self; }   // runtime CPU detection, cached
pub struct Engine { catalog: Arc<dyn Catalog>, storage: Arc<dyn ObjectStore>, kernel_level: KernelLevel, config: EngineConfig }
pub struct EngineConfig {
    pub vector_size: usize,           // default 4096
    pub page_cache_bytes: usize,      // default 256 MiB
    pub prefetch_pages: usize,        // default 4
    pub kernel_level_override: Option<KernelLevel>,
}

#[async_trait::async_trait]
pub trait Catalog: Send + Sync {
    async fn list_tables(&self) -> Result<Vec<Arc<str>>, Error>;
    async fn open_table(&self, name: &str) -> Result<Arc<Table>, Error>;
}
pub struct Table { pub name: Arc<str>, pub schema: Arc<Schema>, pub files: Vec<Arc<TableFile>> }
pub struct TableFile { pub path: Arc<str>, pub format: TableFileFormat, pub footer: Option<ParquetFooter> }
pub enum TableFileFormat { Parquet, Orc, ArrowIpcStream, ArrowIpcFile }

pub struct ScanBuilder { table: Arc<Table>, projection: Option<Vec<Arc<str>>>, filter: Option<expr::Expr>, limit: Option<usize>, batch_size: Option<usize> }
pub trait Scan: Send {
    fn schema(&self) -> &Arc<Schema>;
    fn next_batch(&mut self) -> Result<Option<RecordBatch>, Error>;
}
#[derive(Debug, thiserror::Error)]
pub enum Error {
    #[error("io error: {0}")] Io(#[from] std::io::Error),
    #[error("object store error: {0}")] ObjectStore(Arc<str>),
    #[error("invalid parquet: {0}")] InvalidParquet(Arc<str>),
    #[error("invalid arrow: {0}")] InvalidArrow(Arc<str>),
    #[error("invalid encoding: {0}")] InvalidEncoding(Arc<str>),
    #[error("type mismatch: expected {expected}, got {actual}")] TypeMismatch { expected: Arc<str>, actual: Arc<str> },
    #[error("schema mismatch: {0}")] SchemaMismatch(Arc<str>),
    #[error("out of range")] OutOfRange,
    #[error("phase deferred: {0}")] PhaseDeferred(Arc<str>),
    #[error("kernel unsupported: {kernel} on {dtype}")] KernelUnsupported { kernel: Arc<str>, dtype: Arc<str> },
    #[error("flight error: {0}")] Flight(Arc<str>),
    #[error("internal error: {0}")] Internal(Arc<str>),
}
