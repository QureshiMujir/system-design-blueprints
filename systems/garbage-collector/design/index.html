Core schema & types (§11)

typedef uintptr_t gc_word_t;                 /* machine word; owner: none (value type) */

typedef enum gc_color {                      /* tri-color tag; mutated by tracer + barriers */
    GC_WHITE = 0, GC_GREY = 1, GC_BLACK = 2, GC_OFF_HEAP = 3,
} gc_color_t;

typedef uint8_t gc_age_t;                    /* 0..15 saturating generational age */

typedef struct gc_typeinfo {                 /* host-owned; collector reads only */
    const char *name;
    uint32_t    size;
    uint32_t    n_ptr_slots;
    const uint32_t *ptr_offsets;              /* invariant I14 */
    size_t (*size_of)(const void *obj);       /* signature only, variable-length types */
    void   (*finalize)(void *obj);             /* signature only; NULL if unused */
    uint16_t flags;                            /* GC_TYPE_HAS_FINALIZER | GC_TYPE_WEAK */
    uint16_t hash;
} gc_typeinfo_t;

typedef struct gc_header {                   /* owner: collector; mutated under I1/I4/I9 */
    gc_word_t bits;                           /* [63] fwd? | [62:59] age | [58:57] color | [56:0] typeinfo-ptr-or-fwd-addr */
} gc_header_t;

typedef void *gc_object_t;                   /* opaque payload pointer; header at offset -sizeof(gc_header_t) */

typedef struct gc_thread_ctx gc_thread_ctx_t; /* opaque; owner: collector; one per attached thread */

typedef struct gc_rootset {                  /* owner: collector; concurrency: handshake-guarded (I7) */
    void ***globals; size_t n_globals; size_t cap_globals;
    gc_thread_ctx_t *threads; size_t n_threads;
} gc_rootset_t;
typedef struct gc_event {                    /* owner: collector; emitted read-only to host callback */
    uint64_t start_ns, end_ns;
    uint32_t kind;                            /* gc_collect_kind_t */
    size_t bytes_before, bytes_after, bytes_promoted, bytes_freed;
    size_t roots_scanned, marked_objects, swept_objects, compacted_objects;
    size_t bytes_floating_estimate;
    uint32_t worker_count, worklist_steals, cas_retries, cards_dirtied;
} gc_event_t;

typedef enum gc_collect_kind {
    GC_COLLECT_MINOR = 0, GC_COLLECT_MAJOR = 1,
    GC_COLLECT_ON_OOM = 2, GC_COLLECT_REQUESTED = 3,
} gc_collect_kind_t;

typedef struct gc_config {                   /* owner: host; immutable after gc_new(); not serialized */
    size_t initial_heap_bytes, max_heap_bytes, young_gen_bytes;
    double growth_ratio;
    uint32_t pause_target_ms_p99;
    uint8_t  collector_kind, precise_roots;
    uint32_t worker_threads, card_size_bytes, region_size_bytes;
    uint64_t rng_seed;
    void (*on_event)(const gc_event_t *, void *user);
    void  *on_event_user;
} gc_config_t;

typedef struct gc gc_t;                       /* opaque collector instance handle */
