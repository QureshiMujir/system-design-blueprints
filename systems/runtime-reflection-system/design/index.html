pub trait PartialReflect: Any + Send + Sync {
    fn reflect_ref(&self)  -> ReflectRef<'_>;
    fn reflect_mut(&mut self) -> ReflectMut<'_>;
    fn reflect_owned(self: Box<Self>) -> ReflectOwned;
    fn represented_type_info(&self) -> Option<&'static TypeInfo>;
    fn try_apply(&mut self, value: &dyn PartialReflect) -> Result<(), ApplyError>;
    fn into_partial_reflect(self: Box<Self>) -> Box<dyn PartialReflect>;
    fn as_partial_reflect(&self) -> &dyn PartialReflect;
    fn as_partial_reflect_mut(&mut self) -> &mut dyn PartialReflect;
    fn reflect_partial_eq(&self, value: &dyn PartialReflect) -> Option<bool>;
    fn reflect_hash(&self) -> Option<u64>;
}
pub trait Reflect: PartialReflect {
    fn into_any(self: Box<Self>) -> Box<dyn Any>;
    fn as_any(&self)     -> &dyn Any;
    fn as_any_mut(&mut self) -> &mut dyn Any;
    fn into_reflect(self: Box<Self>) -> Box<dyn Reflect>;
    fn as_reflect(&self)     -> &dyn Reflect;
    fn as_reflect_mut(&mut self) -> &mut dyn Reflect;
    fn set(&mut self, value: Box<dyn Reflect>) -> Result<(), Box<dyn Reflect>>;
}
pub enum ReflectRef<'a> {
    Struct(&'a dyn Struct),
    TupleStruct(&'a dyn TupleStruct),
    Tuple(&'a dyn Tuple),
    List(&'a dyn List),
    Array(&'a dyn Array),
    Map(&'a dyn Map),
    Set(&'a dyn Set),
    Enum(&'a dyn Enum),
    Function(&'a dyn PartialReflect),
    Opaque(&'a dyn PartialReflect),
}
pub enum ReflectMut<'a>  { /* same variants, &'a mut dyn Trait */ }
pub enum ReflectOwned    { /* same variants, Box<dyn Trait> */ }
pub trait Struct: PartialReflect {
    fn field(&self, name: &str) -> Option<&dyn PartialReflect>;
    fn field_mut(&mut self, name: &str) -> Option<&mut dyn PartialReflect>;
    fn field_at(&self, index: usize) -> Option<&dyn PartialReflect>;
    fn field_at_mut(&mut self, index: usize) -> Option<&mut dyn PartialReflect>;
    fn name_at(&self, index: usize) -> Option<&str>;
    fn field_len(&self) -> usize;
    fn iter_fields(&self) -> FieldIter<'_>;
}
pub trait TupleStruct: PartialReflect {
    fn field(&self, index: usize) -> Option<&dyn PartialReflect>;
    fn field_mut(&mut self, index: usize) -> Option<&mut dyn PartialReflect>;
    fn field_len(&self) -> usize;
    fn iter_fields(&self) -> TupleStructIter<'_>;
}
pub trait Tuple: PartialReflect {
    fn field(&self, index: usize) -> Option<&dyn PartialReflect>;
    fn field_mut(&mut self, index: usize) -> Option<&mut dyn PartialReflect>;
    fn field_len(&self) -> usize;
    fn iter_fields(&self) -> TupleIter<'_>;
}
pub trait List: PartialReflect {
    fn get(&self, index: usize)     -> Option<&dyn PartialReflect>;
    fn get_mut(&mut self, index: usize) -> Option<&mut dyn PartialReflect>;
    fn push(&mut self, value: Box<dyn PartialReflect>);
    fn insert(&mut self, index: usize, value: Box<dyn PartialReflect>);
    fn remove(&mut self, index: usize) -> Box<dyn PartialReflect>;
    fn len(&self) -> usize;
    fn iter(&self) -> ListIter<'_>;
}
pub trait Array: PartialReflect {
    fn get(&self, index: usize)     -> Option<&dyn PartialReflect>;
    fn get_mut(&mut self, index: usize) -> Option<&mut dyn PartialReflect>;
    fn len(&self) -> usize;
    fn iter(&self) -> ArrayIter<'_>;
}
pub trait Map: PartialReflect {
    fn get(&self, key: &dyn PartialReflect)     -> Option<&dyn PartialReflect>;
    fn get_mut(&mut self, key: &dyn PartialReflect) -> Option<&mut dyn PartialReflect>;
    fn insert_boxed(&mut self, key: Box<dyn PartialReflect>, value: Box<dyn PartialReflect>);
    fn remove(&mut self, key: &dyn PartialReflect) -> Option<Box<dyn PartialReflect>>;
    fn len(&self) -> usize;
    fn iter(&self) -> MapIter<'_>;
}
pub trait Set: PartialReflect {
    fn contains(&self, value: &dyn PartialReflect) -> bool;
    fn insert_boxed(&mut self, value: Box<dyn PartialReflect>) -> bool;
    fn remove(&mut self, value: &dyn PartialReflect) -> bool;
    fn len(&self) -> usize;
    fn iter(&self) -> SetIter<'_>;
}
pub trait Enum: PartialReflect {
    fn variant_name(&self)  -> &str;
    fn variant_index(&self) -> usize;
    fn variant_type(&self)  -> VariantType;
    fn field(&self, name: &str)     -> Option<&dyn PartialReflect>;
    fn field_mut(&mut self, name: &str) -> Option<&mut dyn PartialReflect>;
    fn field_at(&self, index: usize)     -> Option<&dyn PartialReflect>;
    fn field_at_mut(&mut self, index: usize) -> Option<&mut dyn PartialReflect>;
    fn field_len(&self) -> usize;
    fn iter_fields(&self) -> VariantFieldIter<'_>;
}
pub enum VariantType { Struct, Tuple, Unit }
pub enum ApplyError {

    InvalidField     { name: String },
    Internal(Box<dyn std::error::Error + Send + Sync>),
}

pub struct TypeRegistry {
    registrations: HashMap<TypeId, TypeRegistration>,

}
impl TypeRegistry {
    pub fn register<T: GetTypeRegistration>(&mut self);
    pub fn register_type_data<T, D>(&mut self) where T: Reflect, D: TypeData + FromType<T>;
    pub fn get(&self, type_id: TypeId) -> Option<&TypeRegistration>;
    pub fn get_mut(&mut self, type_id: TypeId) -> Option<&mut TypeRegistration>;
    pub fn get_with_type_path(&self, type_path: &str) -> Option<&TypeRegistration>;
    pub fn iter(&self) -> impl Iterator<Item = &TypeRegistration>;
}
pub struct TypeRegistration {

}

pub enum Access<'a> {
    Field(Cow<'a, str>),      // .field_name
    FieldIndex(usize),        // #0  (tuple-struct positional)
    TupleIndex(usize),        // .0  (tuple field)
    ListIndex(usize),         // [n] (list/array index)
}
pub struct OffsetAccess {
    pub access: Access<'static>,
    pub offset: Option<usize>,   // byte offset in original path string for error reporting
}
pub struct ParsedPath(pub Vec<OffsetAccess>);
impl ParsedPath {
    pub fn parse(path: &str) -> Result<Self, ReflectPathError<'static>>;
}
pub trait GetPath {
    fn reflect_path<'p>(&self, path: impl ReflectPath<'p>)
        -> Result<&dyn PartialReflect, ReflectPathError<'p>>;
    fn reflect_path_mut<'p>(&mut self, path: impl ReflectPath<'p>)
        -> Result<&mut dyn PartialReflect, ReflectPathError<'p>>;
    fn path<'p, T: Reflect>(&self, path: impl ReflectPath<'p>)
        -> Result<&T, ReflectPathError<'p>>;
    fn path_mut<'p, T: Reflect>(&mut self, path: impl ReflectPath<'p>)
        -> Result<&mut T, ReflectPathError<'p>>;
}
pub enum ReflectPathError<'a> {
    InvalidField     { index: usize, field: &'a str },
    InvalidFieldIndex { index: usize, field_index: usize },
    InvalidEnum      { index: usize, variant: &'a str },
    InvalidDowncast  { index: usize, expected: &'static str, actual: ReflectKind },
    IndexOutOfBounds { index: usize, len: usize },
    ParseError       { index: usize, msg: &'static str },
}
pub struct ReflectSerializer<'a> {
    pub value:    &'a dyn PartialReflect,
    pub registry: &'a TypeRegistry,
}
pub struct ReflectDeserializer<'a> {
    pub registry: &'a TypeRegistry,
}
pub struct TypedReflectDeserializer<'a> {
    pub registration: &'a TypeRegistration,
    pub registry:     &'a TypeRegistry,
}
pub struct FunctionInfo {
    pub name:        Option<&'static str>,
    pub args:        Box<[ArgInfo]>,
    pub return_info: ReturnInfo,
}
pub struct ArgInfo  { pub index: usize, pub type_path: &'static str, pub type_id: TypeId }
pub struct ReturnInfo { pub type_path: &'static str, pub type_id: TypeId }

pub struct DynamicFunction {
    pub info: FunctionInfo,
    // private: erased fn pointer Box<dyn Fn(ArgList) -> Return + Send + Sync>
}
impl DynamicFunction {
    pub fn call(&self, args: ArgList) -> Result<Return, FunctionError>;
}

pub struct ArgList { /* owned + borrowed args in call order */ }
pub enum Return { Owned(Box<dyn PartialReflect>), Ref(/* elided */), Mut(/* elided */) }

pub trait IntoFunction<Marker> {
    fn into_function(self) -> DynamicFunction;
}

pub enum FunctionError {
    ArgCountMismatch { expected: usize, received: usize },
    ArgTypeMismatch  { index: usize, expected: &'static str },
    ReturnTypeMismatch,
    CallError(Box<dyn std::error::Error + Send + Sync>),
}
