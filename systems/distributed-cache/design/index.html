Core schema & types (§11)

pub struct SlotId(pub u16);                // range [0, 16384); I13
pub struct NodeId(pub uuid::Uuid);         // stable across restarts
pub struct Epoch(pub u64);                 // monotonic; I2
impl SlotId {
    pub const COUNT: u32 = 16_384;
    pub fn from_key(key: &[u8]) -> Self;   // CRC16(hash_tag(key) or key) mod COUNT, I11
}
pub struct Namespace(pub bytes::Bytes);
pub struct Key { pub namespace: Namespace, pub raw: bytes::Bytes }
pub struct Value { pub bytes: bytes::Bytes, pub flags: u32, pub cas: u64, pub expiry: Option<Expiry> }
pub enum Expiry { Absolute(std::time::SystemTime), Relative(std::time::Duration) }
pub struct SlotTable {
    pub epoch: Epoch,
    pub primary: [NodeId; SlotId::COUNT as usize],
    pub replicas: Vec<Vec<NodeId>>,
    pub migrating: hashbrown::HashMap<SlotId, MigrationState>,
}
pub struct MigrationState { pub source: NodeId, pub destination: NodeId, pub started_at_epoch: Epoch, pub phase: MigrationPhase }
pub enum MigrationPhase { Importing, DoubleWrite, Catchup, Cutover }
#[async_trait::async_trait]
pub trait Engine: Send + Sync + 'static {
    async fn get(&self, key: &Key) -> Result<Option<Value>, EngineError>;
    async fn set(&self, key: Key, value: Value) -> Result<(), EngineError>;
    async fn cas(&self, key: Key, value: Value, expected_cas: u64) -> Result<CasOutcome, EngineError>;
    async fn delete(&self, key: &Key) -> Result<bool, EngineError>;
    fn scan_slot(&self, slot: SlotId) -> BoxStream<'_, KeyValueExpiry>;   // I10
    async fn delete_slot(&self, slot: SlotId) -> Result<u64, EngineError>;
    fn stats(&self) -> EngineStats;
}
pub enum CasOutcome { Set, Mismatch }
pub enum Operation { Get { key: Key, flags: GetFlags }, Set { key: Key, value: Value, policy: WritePolicy },
                      Cas { key: Key, value: Value, expected_cas: u64 }, Delete { key: Key },
                      Cluster(ClusterOp), Hello { version: u8, auth: Option<Auth> }, Ping }
pub enum Response { Value(Value), NotFound, Stored, NotStored, CasMismatch,
                     Moved { slot: SlotId, target: NodeAddress }, Ask { slot: SlotId, target: NodeAddress },
                     Error(WireError) }
pub struct ReplicationRecord { pub seq: u64, pub epoch: Epoch, pub op: ReplicatedOp, pub timestamp: std::time::SystemTime }
pub enum ReplicatedOp { Set { key: Key, value: Value }, Delete { key: Key }, Migrate { slot: SlotId, kvs: Vec<KeyValueExpiry> } }
pub struct HotKeyDetector { cms: CountMinSketch, threshold: u32, decay_period: std::time::Duration,
                             local_cache: lru::LruCache<Key, Value> }
impl HotKeyDetector {
    pub fn observe(&mut self, key: &Key) -> HotKeySignal;
    pub fn local_get(&mut self, key: &Key) -> Option<Value>;
}
pub enum HotKeySignal { Cold, Warm, Hot }

pub struct SingleFlight<K: Eq + Hash + Clone, V: Clone> { inflight: Mutex<HashMap<K, Shared<BoxFuture<'static, V>>>> }
impl<K, V> SingleFlight<K, V> where K: Eq + Hash + Clone + Send + Sync + 'static, V: Clone + Send + Sync + 'static {
    pub async fn run<F, Fut>(&self, key: K, work: F) -> V where F: FnOnce() -> Fut, Fut: Future<Output = V> + Send + 'static;
}
pub enum ClusterEvent { NodeJoined { node: NodeMeta, weight: u32 }, NodeFailed { node: NodeId }, NodeLeft { node: NodeId },
                         SlotMoved { slot: SlotId, from: NodeId, to: NodeId },
                         Failover { slot: SlotId, new_primary: NodeId, demoted: NodeId }, EpochAdvanced(Epoch) }
service Coordinator {
  rpc GetTopology(TopologyReq) returns (TopologyResp);
  rpc WatchTopology(WatchReq) returns (stream TopologyEvent);
  rpc ProposeEvent(ClusterEventReq) returns (ClusterEventResp);
  rpc HealthReport(HealthReportReq) returns (HealthReportResp);
}
pub struct WriteThrough { pub backing: Arc<dyn BackingStore> }
pub struct WriteBack { pub backing: Arc<dyn BackingStore>, pub wal: Arc<dyn Wal>, pub flush_interval: Duration, pub max_dirty_bytes: usize }
pub struct ReadRoutingPolicy { pub min_freshness: Duration, pub primary_fallback: bool, pub balance: Balance }
