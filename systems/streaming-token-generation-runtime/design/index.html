Core schema & types (§11)

// stg-types — identifiers + requests
pub struct RequestId(pub u64);                 // monotone per process; in event ordering (I15)
pub struct SeqId(pub u64);                      // one per in-flight generation; freed on terminal
pub type TokenId = u32;                          // vocab index; executor + tokenizer agree
pub struct GenerationRequest {                   // untrusted input; owned by admission then scheduler
  pub id: RequestId, pub prompt: PromptInput, pub params: SamplingParams,
  pub stream: bool, pub priority: Priority, pub deadline: Option<Instant>, pub seed: Option<u64> }
pub enum PromptInput { Text(String), Tokens(Vec<TokenId>), Chat(Vec<ChatMsg>) }
pub enum Priority { Interactive, Batch, Bulk }   // admission class (I13 FIFO within class)

pub struct SamplingParams {                      // validated; ranges enforced (§8.7)
  pub max_tokens: u32, pub temperature: f32, pub top_k: u32, pub top_p: f32, pub min_p: f32,
  pub repetition_penalty: f32, pub presence_penalty: f32, pub frequency_penalty: f32,
  pub stop: Vec<String>, pub logit_bias: Vec<(TokenId, f32)>, pub n: u32 /*=1 first*/ }
pub enum FinishReason { Stop, Length, Eos, Cancel, Error }

// stg-types — streaming events (wire; serde JSON + prost)
pub enum StreamEvent { Delta(Delta), Done(Done), Error(StreamError) } // terminal-once: I12
pub struct Delta { pub seq: SeqId, pub pos: u32, pub text: String, pub token_id: TokenId }
pub struct Done  { pub seq: SeqId, pub reason: FinishReason, pub usage: Usage }
pub struct Usage { pub prompt_tokens: u32, pub completion_tokens: u32 }

// stg-types — the pluggable backend contracts (principle 9)
pub trait ModelExecutor: Send + Sync {           // the forward pass; STGRT never implements (#44)
  fn forward(&self, batch: &BatchDescriptor, kv: &KvView) -> Result<LogitsBatch>;
  fn kv_layout(&self) -> KvLayout;                // shapes the block allocator (§3)
  fn vocab_size(&self) -> u32;
}
pub trait Tokenizer: Send + Sync {
  fn encode(&self, text: &str) -> Result<Vec<TokenId>>;
  fn decode_incremental(&self, st: &mut DetokState, new: &[TokenId]) -> Result<String>; // §2
  fn eos_id(&self) -> TokenId;
}
pub struct LogitsBatch { /* [seq][vocab] f32; row-per-seq, same order as batch */ }
pub struct KvLayout { pub num_layers: u32, pub kv_heads: u32, pub head_dim: u32, pub dtype: KvDtype }

// stg-sched — sequence + step state
pub enum SeqStatus { Waiting, Prefilling, Running, Preempted, Finished, Cancelled, Errored } // §8.1
pub struct SequenceState {                       // owned by scheduler (single-writer, I2)
  pub seq: SeqId, pub req: RequestId, pub status: SeqStatus,
  pub prompt: Vec<TokenId>, pub generated: Vec<TokenId>, pub pos: u32,   // pos monotone (I1)
  pub block_table: BlockTable, pub params: SamplingParams, pub rng: SeqRng, // per-seq RNG (I14)
  pub cancel: CancelToken, pub detok: DetokState, pub stop: StopMatcher,
  pub admit_seq: u64, pub class: Priority, pub deadline: Option<Instant> }
pub struct StepPlan {                             // one forward call's batch plan
  pub prefills: Vec<PrefillChunk>, pub decodes: Vec<SeqId>,
  pub total_tokens: u32 /*≤ max_num_batched_tokens (I10)*/ }
pub struct PrefillChunk { pub seq: SeqId, pub start: u32, pub len: u32 } // chunked prefill (§11 principle)
// stg-batch — what the executor consumes
pub struct BatchDescriptor {                      // built per step; immutable during forward
  pub entries: Vec<BatchEntry>, pub slot_mapping: SlotMapping }
pub struct BatchEntry { pub seq: SeqId, pub tokens: Vec<TokenId>, pub is_prefill: bool,
                        pub block_table: Vec<BlockId>, pub context_len: u32 }
pub struct SlotMapping { /* (seq,token)→(BlockId, offset): where executor writes K,V (§8.3) */ }
pub struct KvView<'a> { /* borrowed view into the pool the executor reads/writes (lifetime-bound) */ }

// stg-kv — paged KV
pub struct BlockId(pub u32);                      // physical block index in the pool
pub struct BlockTable { pub blocks: Vec<BlockId> }// logical→physical; dense, append-only (I11)
pub struct KvBlockPool {                          // device-mem blocks; single-writer via scheduler
  pub capacity: u32, pub block_size: u32, pub free: Vec<BlockId>, pub ref_count: Vec<u16> } // I4/I6
pub enum AllocResult { Ok(BlockId), NeedPreemption, PoolFull }      // drives §8 step rule 5
pub struct SwapStore { /* host/NVMe-backed swapped blocks; backpressured I/O (§8) */ }
pub struct PrefixCache { /* hash(prefix tokens) → shared BlockTable head; COW on write (§9) */ }

// stg-sample
pub struct SeqRng { /* ChaCha8 seeded per request (I14); no shared state */ }
pub struct StopMatcher { /* multi-token stop-string state across token boundaries (§6) */ }
pub enum SampleOutcome { Token(TokenId), Stop(FinishReason) }

// stg-stream
pub struct DetokState { /* pending partial UTF-8 bytes + token-text buffer (I3) */ }
pub struct StreamSink { /* per-seq bounded mpsc::Sender<StreamEvent>; full ⇒ throttle seq (principle 6) */ }
pub struct CancelToken { /* set on client disconnect; checked at step boundary (I8) */ }

// stg-admit
pub struct AdmissionQueue { /* per-class bounded FIFO (I13); push blocks/rejects when full */ }
pub enum AdmitDecision { Admit(SeqId), Reject(RejectReason) }     // RejectReason::{QueueFull, KvSaturated, ...}

// stg-observe
pub struct StepMetrics { pub step_seq: u64, pub n_prefill: u32, pub n_decode: u32,
                         pub tokens: u32, pub kv_util: f32, pub preempted: u32 } // per-step (I15)
pub struct ReplayTrace { /* requests + executor logits + seeds → deterministic re-run (I7) */ }

// errors (thiserror; no panics on request path — ADR-028)
pub enum StgError { Request(RequestError), Kv(KvError), Sched(SchedError),
                    Exec(ExecError), Stream(StreamError), Admit(AdmitError) }
pub enum RequestError { PromptTooLong, InvalidParam, UnknownModel, BadUtf8 }
pub enum KvError { PoolFull, RefCountMismatch, InvalidBlock, DoubleFree, SwapIo }
pub enum ExecError { BatchMismatch, Backend(String), VocabMismatch }
