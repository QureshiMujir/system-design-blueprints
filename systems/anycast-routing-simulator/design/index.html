Core schema & types (§11)

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
pub struct Asn(pub u32);
impl Asn {
    pub fn new(n: u32) -> Result<Self, CoreError>;   // rejects 0; RFC 7607 reserved
    pub fn is_private(&self) -> bool;                // 64512-65534, 4200000000-4294967294
}
impl Display for Asn { /* "{}", self.0 */ }
#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub struct Prefix { pub addr: IpAddr, pub len: u8 }
impl Prefix {
    pub fn new(addr: IpAddr, len: u8) -> Result<Self, CoreError>;
    pub fn contains(&self, other: &Prefix) -> bool;
}
impl Ord for Prefix { /* addr-then-len lexicographic */ }
#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub struct PoPId(pub u32);
#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub enum IpAddr { V4([u8; 4]), V6([u8; 16]) }
impl IpAddr {
    pub fn family(&self) -> u8;      // 4 or 6
    pub fn to_bits(&self) -> u128;   // for prefix containment arithmetic
}
#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub struct Community(pub u32, pub u32);
#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
pub struct BgpRoute {
    pub prefix:      Prefix,
    pub as_path:     Vec<Asn>,
    pub next_hop:    Asn,
    pub local_pref:  u32,
    pub med:         u32,
    pub communities: Vec<Community>,
    pub origin_pop:  Option<PoPId>,
}
impl BgpRoute {
    pub fn contains_loop(&self, local_asn: Asn) -> bool;
    pub fn path_len(&self) -> usize;
}
#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct UpdateMsg   { pub from: Asn, pub route: BgpRoute }
#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct WithdrawMsg { pub from: Asn, pub prefix: Prefix }
#[derive(Clone, Debug, Serialize, Deserialize)]
pub enum BgpEvent {
    RecvUpdate   { to: Asn, msg: UpdateMsg },
    RecvWithdraw { to: Asn, msg: WithdrawMsg },
    LinkDown     { link: (Asn, Asn) },
    LinkUp       { link: (Asn, Asn) },
    Announce     { origin: Asn, pop: PoPId, route: BgpRoute },
    Withdraw     { origin: Asn, prefix: Prefix },
}
#[derive(Clone, Debug, Eq, PartialEq, Serialize, Deserialize)]
pub struct SimEvent { pub time: u64, pub seq: u64, pub kind: BgpEvent }
impl Ord     for SimEvent { /* (time, seq) ascending -- I1 */ }
impl PartialOrd for SimEvent { /* derived from Ord */ }
pub struct VirtualClock { current: u64 }
impl VirtualClock {
    pub fn new() -> Self;
    pub fn now(&self) -> u64;
    pub fn advance_to(&mut self, t: u64) -> Result<(), ClockError>;
}
pub struct EventQueue { heap: BinaryHeap<Reverse<SimEvent>>, seq_counter: u64 }
impl EventQueue {
    pub fn push(&mut self, time: u64, kind: BgpEvent) -> u64;
    pub fn pop(&mut self)  -> Option<SimEvent>;
    pub fn peek(&self)     -> Option<&SimEvent>;
    pub fn is_empty(&self) -> bool;
    pub fn len(&self)      -> usize;
}
pub struct Engine { clock: VirtualClock, queue: EventQueue, graph: AsGraph }
impl Engine {
    pub fn new(graph: AsGraph) -> Self;
    pub fn schedule(&mut self, at: u64, event: BgpEvent);
    pub fn run_until_quiescent(&mut self) -> ConvergenceReport;
    pub fn run_until(&mut self, time_limit: u64) -> ConvergenceReport;
    pub fn is_quiescent(&self) -> bool;
}
#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub enum Relationship { Customer, Provider, Peer }
impl Relationship {
    pub fn inverse(&self) -> Relationship;
}
#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct Link {
    pub a:          Asn,
    pub b:          Asn,
    pub rel_from_a: Relationship,
    pub latency_ms: u32,
    pub up:         bool,
}
impl Link {
    pub fn rel_from_b(&self) -> Relationship;
    pub fn involves(&self, asn: Asn) -> bool;
}
pub struct AsNode {
    pub asn:       Asn,
    pub neighbors: Vec<Asn>,
    pub policy:    PolicyConfig,
    pub rib:       Rib,
}
impl AsNode {
    pub fn link_to(&self, peer: Asn, graph: &AsGraph) -> Option<&Link>;
    pub fn relationship_to(&self, peer: Asn, graph: &AsGraph) -> Option<Relationship>;
}
pub struct AsGraph {
    nodes: Vec<AsNode>,
    links: Vec<Link>,
    index: HashMap<Asn, usize>,
}
impl AsGraph {
    pub fn node(&self, asn: Asn)         -> Option<&AsNode>;
    pub fn node_mut(&mut self, asn: Asn) -> Option<&mut AsNode>;
    pub fn link_between(&self, a: Asn, b: Asn) -> Option<&Link>;
    pub fn neighbors(&self, asn: Asn)    -> impl Iterator<Item = Asn> + '_;
    pub fn node_count(&self) -> usize;
    pub fn link_count(&self) -> usize;
}
pub struct Rib {
    pub adj_rib_in:  HashMap<(Asn, Prefix), BgpRoute>,
    pub loc_rib:     HashMap<Prefix, BgpRoute>,
    pub adj_rib_out: HashMap<Asn, HashMap<Prefix, BgpRoute>>,
}
impl Rib {
    pub fn install_adj_in(&mut self, from: Asn, route: BgpRoute);
    pub fn remove_adj_in(&mut self, from: Asn, prefix: Prefix) -> Option<BgpRoute>;
    pub fn set_best(&mut self, prefix: Prefix, route: BgpRoute);
    pub fn remove_best(&mut self, prefix: Prefix) -> Option<BgpRoute>;
    pub fn set_adj_out(&mut self, to: Asn, prefix: Prefix, route: BgpRoute);
    pub fn remove_adj_out(&mut self, to: Asn, prefix: Prefix) -> Option<BgpRoute>;
    pub fn candidates(&self, prefix: Prefix) -> impl Iterator<Item = &BgpRoute> + '_;
}
pub fn compare_routes(a: &BgpRoute, b: &BgpRoute) -> Ordering;
pub fn select_best<'a>(candidates: impl Iterator<Item = &'a BgpRoute>) -> Option<&'a BgpRoute>;
#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct PolicyConfig {
    pub local_pref_by_rel: HashMap<Relationship, u32>,
    pub export_rules:      ExportRules,
    pub prepend:           HashMap<Asn, u8>,
    pub med_map:           HashMap<Asn, u32>,
    pub community_actions: Vec<CommunityAction>,
}
impl PolicyConfig {
    pub fn default_gao_rexford() -> Self;
}
#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct ExportRules;
impl ExportRules {
    pub fn export_allowed(learned_from: Relationship, exporting_to: Relationship) -> bool;
}
#[derive(Clone, Debug, Serialize, Deserialize)]
pub enum CommunityAction {
    Match  { community: Community },
    Set    { community: Community },
    Delete { community: Community },
}
#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct AnycastPrefix {
    pub prefix:          Prefix,
    pub announcing_pops: Vec<(Asn, PoPId)>,
}
pub struct Catchment {
    pub prefix: Prefix,
    pub map:    HashMap<Asn, Option<PoPId>>,
}
impl Catchment {
    pub fn derive(prefix: Prefix, graph: &AsGraph) -> Self;
    pub fn serving_pop(&self, client: Asn) -> Option<PoPId>;
    pub fn clients_of(&self, pop: PoPId) -> impl Iterator<Item = Asn> + '_;
    pub fn unreachable_count(&self) -> usize;
}
pub struct FaultInjector;
impl FaultInjector {
    pub fn link_down(engine: &mut Engine, at: u64, a: Asn, b: Asn);
    pub fn link_up(engine: &mut Engine, at: u64, a: Asn, b: Asn);
    pub fn node_down(engine: &mut Engine, at: u64, asn: Asn);
    pub fn pop_withdraw(engine: &mut Engine, at: u64, pop: PoPId, origin_asn: Asn, prefix: Prefix);
}
#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct ConvergenceReport {
    pub events_processed: u64,
    pub updates_sent:     u64,
    pub withdraws_sent:   u64,
    pub converge_time:    u64,
    pub oscillation:      bool,
}
#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct ScenarioSpec {
    pub name:           String,
    pub spec_version:   u32,
    pub topology_ref:   TopologyRef,
    pub announcements:  Vec<AnnounceSpec>,
    pub fault_timeline: Vec<FaultSpec>,
    pub asserts:        Vec<AssertSpec>,
}

#[derive(Clone, Debug, Serialize, Deserialize)]
pub enum AssertSpec {
    CatchmentEq    { asn: Asn, expect_pop: Option<PoPId> },
    ConvergeWithin { ticks: u64 },
    UpdateCountLe  { max: u64 },
    NoOscillation,
    ReachableCount { min: usize },
}
