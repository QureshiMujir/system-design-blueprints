Core schema & types (§11)

#[derive(Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug)]
pub struct PageId(pub u64);           // 0 = file header; 1,2 = meta A/B (§8.1)

#[derive(Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug)]
pub struct Lsn(pub u64);              // monotonic across the WAL

pub const FILE_HEADER_PAGE: PageId = PageId(0);
pub const META_PAGE_A: PageId = PageId(1);
pub const META_PAGE_B: PageId = PageId(2);
pub const FIRST_DATA_PAGE: PageId = PageId(3);
pub const DEFAULT_PAGE_SIZE: usize = 4096;   // ADR-04
pub const MIN_PAGE_SIZE: usize = 512;
pub const MAX_PAGE_SIZE: usize = 65536;

pub fn max_inline_payload(pagesize: usize) -> usize;   // pagesize / 4, ADR-19
#[repr(u8)]
#[derive(Copy, Clone, Eq, PartialEq, Debug)]
pub enum PageKind { Free=0, Meta=1, InternalBPlus=2, LeafBPlus=3, Overflow=4, FreeList=5 }

#[repr(C)]
#[derive(Copy, Clone, Debug)]
pub struct PageHeader {
    pub pgno: u64, pub kind: u8, pub flags: u8,
    pub n_cells: u16, pub free_off: u16, pub free_size: u16,
    pub next_sibling: u64, pub lsn: u64,     // CRC32C trails at [pagesize-4..pagesize]
}

#[repr(C)]
#[derive(Copy, Clone, Debug)]
pub struct Slot { pub cell_off: u16, pub cell_len: u16 }   // §8.1
pub struct BorrowedPage<'pool> { pub(crate) header: PageHeader, pub(crate) bytes: &'pool [u8] }
pub struct PinnedMut<'pool>    { pub(crate) bytes: &'pool mut [u8] }
pub struct OwnedPage(Box<[u8]>);   // in-memory page before pgno assignment (bulk load, split)

pub struct PageRef<'p> { bytes: &'p [u8] }
pub struct PageMut<'p> { bytes: &'p mut [u8] }
#[repr(u8)]
#[derive(Copy, Clone, Eq, PartialEq, Debug)]
pub enum WalRecordKind {
    Begin=1, PageWrite=2, FullPage=3, AllocPage=4, FreePage=5,
    SetMeta=6, Commit=7, Abort=8, Checkpoint=9,
}

#[repr(C)]
#[derive(Copy, Clone, Debug)]
pub struct WalRecordHeader { pub lsn: u64, pub txn_id: u64, pub kind: u8, pub payload_len: u32, pub crc32c: u32 }
pub struct DbOptions {
    pub page_size: usize, pub buffer_pool_pages: usize, pub wal_buffer_bytes: usize,
    pub wal_segment_bytes: u64, pub direct_io: bool, pub fsync_on_commit: bool,
    pub dirty_budget_bytes: usize, pub telemetry: Box<dyn Telemetry + Send + Sync>,
    pub recovery_policy: RecoveryPolicy,
}
pub enum RecoveryPolicy { StrictRedo, BestEffort }   // §35 row 14

pub struct Db { pub(crate) inner: std::sync::Arc<DbInner> }
impl Db {
    pub fn open(path: impl AsRef<std::path::Path>, opts: DbOptions) -> Result<Db>;
    pub fn begin_read(&self) -> Result<ReadTxn<'_>>;
    pub fn begin_write(&self) -> Result<WriteTxn<'_>>;
    pub fn checkpoint(&self) -> Result<()>;
    pub fn vacuum(&self) -> Result<()>;
    pub fn snapshot(&self) -> Result<Snapshot>;
    pub fn bulk_load(&self, items: impl Iterator<Item = (Vec<u8>, Vec<u8>)>) -> Result<()>;
}

pub struct ReadTxn<'db>  { pub(crate) root_pgno: PageId, pub(crate) read_version: u64 }
pub struct WriteTxn<'db> { pub(crate) txn_id: u64, pub(crate) dirty_pages: Vec<PageId>, pub(crate) bytes_dirty: usize }
impl<'db> ReadTxn<'db>  { pub fn get(&self, key: &[u8]) -> Result<Option<Vec<u8>>>; pub fn range(&self, r: impl std::ops::RangeBounds<[u8]>) -> RangeIter<'_>; }
impl<'db> WriteTxn<'db> { pub fn put(&mut self, k: &[u8], v: &[u8]) -> Result<()>; pub fn delete(&mut self, k: &[u8]) -> Result<bool>; pub fn commit(self) -> Result<Lsn>; pub fn abort(self); }
#[derive(thiserror::Error, Debug)]
pub enum Error {
    Io(std::io::Error), PageCorrupt(PageId), WalCorrupt(u64), WalTruncated(Lsn),
    UnexpectedKind(PageId, PageKind), DirtyBudget(usize), KeyTooLarge(usize, usize),
    ValueTooLarge(usize), WriteConflict(PageId), PoolExhausted,
}

pub trait Telemetry: Send + Sync + 'static {
    fn on_commit(&self, txn_id: u64, lsn: Lsn, dirty_pages: usize, bytes: usize);
    fn on_checkpoint(&self, last_lsn: Lsn, pages_flushed: usize);
    fn on_recovery(&self, replayed: usize, last_lsn: Lsn);
    fn on_page_read(&self, pgno: PageId, hit: bool);
    fn on_eviction(&self, pgno: PageId, dirty: bool);
}

pub trait EvictionPolicy: Send + Sync {
    fn on_pin(&self, frame_id: usize);
    fn on_unpin(&self, frame_id: usize);
    fn pick_victim(&self, frames: &[Frame]) -> Option<usize>;   // never returns pin>0 (I3)
}
