Core schema & types (§11)

pub type Complex64 = num_complex::Complex<f64>;

pub struct StateVector {
    pub amps: Vec<Complex64>,   // length == 1<<n (I3); little-endian (P2)
    pub n: usize,               // qubit count
}
impl StateVector {
    pub fn zero(n: usize) -> Result<Self, QError>;     // |0..0>; checks RAM budget (I10)
    pub fn norm_sqr(&self) -> f64;                      // Σ|a|²
    pub fn assert_normalized(&self, tol: f64) -> Result<(), QError>; // I1
    pub fn renormalize(&mut self);
    pub fn inner(&self, other: &StateVector) -> Complex64;           // <self|other>
    pub fn probability(&self, basis: usize) -> f64;
}

// index helpers (P2)
pub fn pair_indices(n: usize, target: usize) -> impl Iterator<Item=(usize,usize)>;
pub fn control_mask(controls: &[usize]) -> usize;

pub enum QError {
    QubitIndex { q: usize, n: usize },         // I4 (P9)
    NonUnitary { tol: f64 },                   // I2
    Normalization { norm_sqr: f64 },           // I1
    InsufficientMemory { needed: u64, budget: u64 }, // I10 (P4)
    DimensionMismatch, QasmParse { line: usize, col: usize, msg: String },
}
pub enum Gate {
    I, X, Y, Z, H, S, Sdg, T, Tdg,
    Rx(f64), Ry(f64), Rz(f64), Phase(f64), U3(f64,f64,f64),
    Cx, Cz, Swap, ISwap, Ccx, Cswap,
    Custom1([[Complex64;2];2]), Custom2([[Complex64;4];4]),
}
impl Gate {
    pub fn matrix(&self) -> GateMatrix;          // 2x2 / 4x4 / 8x8
    pub fn adjoint(&self) -> Gate;               // inverse (params negated) (I9)
    pub fn arity(&self) -> usize;                // 1,2,3
    pub fn is_unitary(&self, tol: f64) -> bool;  // U†U≈I (I2, P1)
}
pub enum GateMatrix { M2([[Complex64;2];2]), M4([[Complex64;4];4]), M8(/*..*/) }
pub fn apply_1q(sv: &mut StateVector, target: usize, u: &[[Complex64;2];2]); // §8.4 (P3)
pub fn apply_controlled(sv: &mut StateVector, controls: &[usize],
                        target: usize, u: &[[Complex64;2];2]);               // §8.5 (I8)
pub fn apply_2q(sv: &mut StateVector, q0: usize, q1: usize, u: &[[Complex64;4];4]);
pub fn apply_kq(sv: &mut StateVector, targets: &[usize], u: &GateMatrix);
pub fn apply_gate(sv: &mut StateVector, op: &Op);   // dispatch on Gate
pub struct FusionBuffer; // accumulates adjacent same-qubit gates into one matrix (P8)
// parallel variants gate disjoint index pairs across rayon threads (ADR-8)
pub struct Op { pub gate: Gate, pub targets: Vec<usize>,
                pub controls: Vec<usize>, pub params: Vec<f64> }
pub struct Circuit {
    pub n_qubits: usize, pub n_clbits: usize, pub ops: Vec<Op>,
}
impl Circuit {
    pub fn builder(n_qubits: usize) -> CircuitBuilder;
    pub fn depth(&self) -> usize;
    pub fn validate(&self) -> Result<(), QError>;  // index range, distinct, registers (I4)
    pub fn adjoint(&self) -> Circuit;              // reversed, gate-adjointed (I9)
}
pub struct CircuitBuilder { /* h(q), x(q), cx(c,t), rx(q,θ), measure(q,c), ... */ }
pub fn prob_one(sv: &StateVector, q: usize) -> f64;                  // §8.6
pub fn marginal(sv: &StateVector, qubits: &[usize]) -> Vec<f64>;     // I6
pub fn sample_shots(sv: &StateVector, shots: usize, rng: &mut Rng)
    -> Vec<usize>;                                                   // CDF sampling (P5,I11)
pub fn measure_collapse(sv: &mut StateVector, q: usize, rng: &mut Rng)
    -> u8;                                                           // mid-circuit (I7)
pub struct ClassicalRegister { pub bits: Vec<u8> }
pub enum Pauli { I, X, Y, Z }
pub struct PauliString { pub paulis: Vec<(usize, Pauli)> }   // qubit -> Pauli
pub struct Observable { pub terms: Vec<(f64, PauliString)> } // Σ wₖ Pₖ
impl Observable {
    pub fn expectation(&self, sv: &StateVector) -> f64;       // <ψ|O|ψ> (real)
    pub fn variance(&self, sv: &StateVector) -> f64;          // <O²>-<O>²
}
// analysis
pub fn fidelity(a: &StateVector, b: &StateVector) -> f64;     // |<a|b>|² (P7,I12)
pub fn partial_trace(sv: &StateVector, keep: &[usize]) -> DensityMatrix; // §8.8
pub fn von_neumann_entropy(rho: &DensityMatrix) -> f64;       // entanglement entropy
pub fn bloch_vector(sv: &StateVector, q: usize) -> [f64;3];

// sim
pub struct Simulator { pub max_qubits_budget: usize, pub threads: usize }
pub enum RunMode { Statevector, Sample{shots: usize}, Expectation(Observable) }
impl Simulator {
    pub fn estimate_bytes(n: usize) -> u64;                   // 2ⁿ·16 (I10)
    pub fn run(&self, c: &Circuit, mode: RunMode, seed: u64) -> Result<RunResult, QError>;
}
pub enum RunResult { State(StateVector), Counts(Vec<(usize,usize)>), Value(f64) }
