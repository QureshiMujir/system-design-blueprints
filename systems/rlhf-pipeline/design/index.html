Core schema & types (§11)

class RunConfig:                       # ownership: single-writer at run start; immutable after load
    model_name: str
    tokenizer_path: str
    loss_family: Literal["ppo", "dpo", "ipo", "orpo", "kto"]
    kl_coefficient: float
    gae_lambda: float
    gae_gamma: float
    ppo_clip_epsilon: float
    schema_version: int              # I9 — persistence: written into checkpoint manifest

class ModelWrapper(Protocol):          # ownership: one instance per model role; mutation: gradient steps only on trainable roles
    def forward(self, input_ids: Tensor) -> ModelOutput: ...
    def parameters(self) -> Iterator[Parameter]: ...
    def is_trainable(self) -> bool: ...
    # concurrency: not thread-safe; one CUDA stream per wrapper instance (P9)

class PreferencePair:                  # persistence: serialized via §8 schema; immutable record
    pair_id: UUID
    prompt: str
    chosen: str
    rejected: str
    source: Literal["HUMAN", "RLAIF", "SYNTHETIC"]
    schema_version: int
    # serialization: JSONL, one record per line; compatibility: additive fields only (I9)

class RolloutRecord:                   # ownership: written once by generation, read-only by PPO (I5)
    batch_id: UUID
    prompt_ids: list[int]
    response_ids: list[int]
    logprobs_policy_old: list[float]
    logprobs_ref: list[float]
    value_estimates: list[float]
    reward_scalar: float
    kl_per_token: list[float]
    # concurrency: single-producer single-consumer per batch_id (I5)
class PolicyLoss(Protocol):            # ownership: stateless pure function object (ADR-10)
    def compute(self, batch: RolloutBatch, policy: ModelWrapper) -> LossOutput: ...
    def required_models(self) -> set[Literal["policy", "value", "reward", "reference"]]: ...
    # compatibility: every implementer (PPOLoss, DPOLoss, IPOLoss, ORPOLoss, KTOLoss) must
    # declare required_models() honestly — enforced by I13

class PreferenceLoss(Protocol):        # ownership: stateless; used by rlhf_dpo (§18)
    def compute(self, pairs: list[PreferencePair], policy: ModelWrapper,
                reference: ModelWrapper | None) -> LossOutput: ...

class CheckpointManifest:               # persistence: JSON, one per checkpoint dir (P10, I10)
    schema_version: int
    global_step: int
    model_checksum: str
    config_snapshot: RunConfig
    rng_state_ref: str

class EvalReport:                       # persistence: JSON, one per eval run (P11)
    reward_score_mean: float
    kl_from_reference: float
    winrate_vs_judge: float
    length_controlled_winrate: float     # I14 — always present
    hacking_score: float | None
    response_length_mean: float          # P12

class GAEBatch:                          # ownership: computed once per rollout batch (I12)
    advantages: Tensor                    # shape [batch, seq_len]
    returns: Tensor
    computed_at_step: int
    # mutation: immutable after construction; reused across all PPO epochs on this batch
