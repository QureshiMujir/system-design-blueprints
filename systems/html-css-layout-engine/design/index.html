Core schema & types (§11)

struct Engine                          // value type; Send, not Sync (P9, I11)
  fn new(cfg: EngineConfig) -> Engine
  fn load_document(&mut self, bytes: &[u8], url: Url) -> DocumentId
  fn layout(&mut self, doc: DocumentId, vp: Viewport) -> LayoutTreeHandle
  fn paint(&self, h: LayoutTreeHandle, surface: &mut dyn Surface) -> DisplayList
  fn hit_test(&self, h: LayoutTreeHandle, x: f32, y: f32) -> Option<NodeId>
  fn dispatch_event(&mut self, h: LayoutTreeHandle, e: Event) -> EventOutcome
  fn recent_parse_errors(&self) -> &[ParseDiagnostic]
  fn force_full_layout(&mut self, doc: DocumentId)   // test-mode only, P8

struct EngineConfig                    // builder; ownership: consumed by Engine::new
  fn font_provider(self, p: impl FontProvider) -> Self
  fn blob_fetcher(self, f: impl BlobFetcher) -> Self
  fn parse_error_sink(self, s: impl ParseErrorSink) -> Self
  fn user_stylesheets(self, sheets: Vec<Stylesheet>) -> Self

struct DocumentId(u32, u32)             // (index, generation) — I5-style safety
struct LayoutTreeHandle(u32, u32)       // invalidated on document mutation
struct Viewport { width: f32, height: f32, dpi: f32, zoom: f32,
                  dark_mode: bool, prefers_reduced_motion: bool, language_tag: LangTag }
enum Node { Element(Element), Text(Text), Comment(Comment),
            ProcessingInstruction(Pi), DocumentType(Doctype),
            DocumentFragment(Fragment), Document(Document) }
struct NodeId(u32, u32)                 // generational; arena-scoped, doc-local
struct Element { qname: QualifiedName, attrs: AttrList, classes: Vec<InternedStr>,
                 id: Option<InternedStr>, state: StatePseudoBits,
                 shadow: Option<ShadowRoot> }
                                          // mutation: via MutationObserver channel only
struct Text { data: InternedStr }        // immutable per node; replaced, not mutated in place
trait BlobFetcher { fn fetch(&self, url: &Url) -> Pin<Box<dyn Future<Output=FetchResult>>>; }
trait FontProvider { fn shape(&self, run: &TextRun) -> ShapedGlyphs;
                     fn metrics(&self, font: FontId) -> FontMetrics; }
struct Stylist { buckets: RuleBuckets, bloom: AncestorBloom, origin: Origin }
struct RuleTree { nodes: InternTable<(RuleTreeNodeId, RuleId), RuleTreeNodeId> }
struct SpecifiedValue(PropertyId, RawValueToken)   // pre-computed-value vocabulary
#[non_exhaustive]
struct ComputedStyle { /* private fields */ }
  fn display(&self) -> Display
  fn font_size(&self) -> Length
  fn width(&self) -> LengthPercentageOrAuto
                                          // no &mut accessor exists — I2
struct SharingKey { rule_node: RuleTreeNodeId, predicate_hash: u64 }   // §15.6, I13
struct BoxId(u32, u32)                   // ≠ NodeId — I1
enum FormattingContext { Block, Inline, Flex, Grid, Table, Multicol, Replaced, Positioned }
struct LayoutBox { id: BoxId, style: Arc<ComputedStyle>, fc: FormattingContext,
                   fragments: Vec<Fragment>, intrinsic: IntrinsicSizeCache,
                   dirty: DirtyBits }
struct DirtyBits { needs_style: bool, needs_box_regenerate: bool,
                   needs_layout: bool, needs_paint: bool }   // I6 lattice
struct Fragment { origin: Point2D, size: Size2D, baseline: Option<f32> }
struct IntrinsicSizeCache { min_content: Option<f32>, max_content: Option<f32> }
enum PaintCommand { SolidRect, Glyph, Image, Border, BoxShadow, Gradient, Stop,
                    PushClip, PopClip, PushStackingContext, PopStackingContext,
                    Transform, Filter, Mask }
struct DisplayList(Vec<PaintCommand>)    // Send + Sync — P10
struct LayerId(u32)
struct ParseDiagnostic { span: ByteSpan, recovery: RecoveryAction, spec_ref: &'static str }
enum LayoutPhaseError { ReentrantFlush { phase: LayoutPhase, site: &'static str } }  // I11
