Core schema & types (§11)

// crates/grpc-core/src/status.rs — owned value, Clone, no interior mutability
pub struct Status { pub code: Code, pub message: String, pub details: Vec<Any>, pub trailers: Metadata }

#[repr(u8)]
pub enum Code { // 17 values per gRPC spec; Copy, exhaustive match required (I12)
    Ok=0, Cancelled=1, Unknown=2, InvalidArgument=3, DeadlineExceeded=4, NotFound=5,
    AlreadyExists=6, PermissionDenied=7, ResourceExhausted=8, FailedPrecondition=9,
    Aborted=10, OutOfRange=11, Unimplemented=12, Internal=13, Unavailable=14,
    DataLoss=15, Unauthenticated=16,
}
impl Status {
    pub fn ok() -> Self;
    pub fn new(code: Code, message: impl Into<String>) -> Self;
    pub fn with_detail<T: prost::Message>(self, detail: T) -> Self; // detail packed via Any
    pub fn is_retryable(&self) -> bool; // consults I6/ADR-11 idempotency policy
}
// crates/grpc-core/src/metadata.rs — owned, BTreeMap for deterministic iteration order
pub struct Metadata { entries: BTreeMap<MetadataKey, Vec<MetadataValue>> } // not Sync-shared; per-call
pub struct MetadataKey(SmolStr); // ASCII-validated on construction
pub enum MetadataValue { Ascii(SmolStr), Binary(Vec<u8>) } // "-bin" suffix keys => Binary
impl Metadata {
    pub fn insert(&mut self, key: impl AsRef<str>, value: MetadataValue) -> Result<(), MetadataError>; // rejects reserved keys, control chars (T11)
    pub fn get_all(&self, key: impl AsRef<str>) -> &[MetadataValue];
    pub fn iter(&self) -> impl Iterator<Item = (&MetadataKey, &MetadataValue)>;
}
// crates/grpc-core/src/deadline.rs — Copy, monotonic clock only (never wall clock)
pub struct Deadline(Instant);
impl Deadline {
    pub fn after(d: Duration) -> Self; // Instant::now() + d
    pub fn remaining(&self) -> Option<Duration>; // None once expired
    pub fn expired(&self) -> bool;
    pub fn to_header(&self) -> String; // "<int><unit>" per §8.7
    pub fn from_header(s: &str) -> Result<Self, ParseError>;
}
pub struct CancellationToken(Arc<Notify>); // Clone, Send+Sync; fires once (P4/I5)
impl CancellationToken {
    pub fn cancel(&self);
    pub async fn cancelled(&self);
    pub fn is_cancelled(&self) -> bool;
}
// crates/grpc-core/src/channel.rs — trait object-safe; impls own connection lifecycle
pub trait Channel: Send + Sync + 'static {
    async fn call<Req: prost::Message, Res: prost::Message + Default>(
        &self, path: &str, request: Req, opts: CallOptions,
    ) -> Result<Response<Res>, Status>;
    fn call_stream_in<Req, Res>(&self, path: &str, opts: CallOptions) -> ClientStream<Req, Res>;
    fn call_stream_out<Req, Res>(&self, path: &str, request: Req, opts: CallOptions) -> ServerStream<Res>;
    fn call_stream_bidi<Req, Res>(&self, path: &str, opts: CallOptions) -> BidiStream<Req, Res>;
}
pub struct CallOptions { // per-call, not shared
    pub deadline: Option<Deadline>, pub metadata: Metadata,
    pub compression: Option<CompressionAlgorithm>, pub retry_policy: Option<RetryPolicy>,
}
pub struct Response<T> { pub message: T, pub metadata: Metadata, pub trailers: Metadata }
// crates/grpc-core/src/service.rs — server-side; Send+Sync, dispatch is &self (shared, no interior lock needed)
pub trait Service: Send + Sync + 'static {
    fn name(&self) -> &'static str;
    fn methods(&self) -> &'static [&'static str];
    async fn dispatch(&self, method: &str, ctx: ServerContext, body: Bytes) -> Result<Bytes, Status>;
}
pub struct ServerContext { // constructed once per call, moved into handler
    pub metadata: Metadata, pub deadline: Option<Deadline>, pub peer: PeerInfo,
    pub cancellation: CancellationToken, pub trace: TraceContext,
}
// crates/grpc-codec/src/varint.rs — free functions, no allocation beyond caller-provided buf
pub fn encode_varint(value: u64, buf: &mut impl bytes::BufMut); // LEB128, §8.2
pub fn decode_varint(buf: &mut impl bytes::Buf) -> Result<u64, CodecError>; // bounds-checked (I3)
pub fn zigzag_encode_i64(value: i64) -> u64;
pub fn zigzag_decode_i64(value: u64) -> i64;
pub fn encode_field_header(tag: u32, wire_type: WireType, buf: &mut impl bytes::BufMut);
pub struct FieldDecoder<'a> { buf: &'a [u8], pos: usize } // borrows, zero-copy scan
impl<'a> FieldDecoder<'a> { pub fn next_field(&mut self) -> Option<Result<FieldHeader, CodecError>>; }
// crates/grpc-proto/src/schema.rs — parse-time owned IR; immutable after resolution
pub struct Schema { pub files: Vec<File>, pub messages: HashMap<FullName, Message>,
                     pub services: HashMap<FullName, ServiceDef>, pub enums: HashMap<FullName, EnumDef> }
pub struct Message { pub name: FullName, pub fields: Vec<Field>, pub nested: Vec<Message>,
                      pub reserved: Vec<ReservedRange> } // I10 enforced during resolve()
pub struct Field { pub name: SmolStr, pub tag: u32, pub ty: FieldType, pub label: FieldLabel }
pub enum FieldType { Primitive(Primitive), Message(FullName), Enum(FullName),
                      Map { key: Primitive, value: Box<FieldType> }, Repeated(Box<FieldType>) }
impl Schema { pub fn resolve(files: Vec<File>) -> Result<Self, Vec<ResolutionError>>; }
// crates/grpc-transport-h2/src/connection.rs — one per TCP connection, owns h2::Connection
pub struct H2Connection { settings: H2Settings, bdp_estimator: BdpEstimator,
                           streams: HashMap<StreamId, StreamState> } // not Send across await without Arc<Mutex<_>>
pub struct H2Settings { pub initial_window_size: u32, pub max_frame_size: u32,
                         pub max_header_list_size: u32, pub max_concurrent_streams: u32 }
pub enum StreamState { Idle, Open, HalfClosedLocal, HalfClosedRemote, Closed } // I2's machine, §8.3
// crates/grpc-loadbalancer, grpc-resolver — trait objects, Send+Sync, hot-swappable
pub trait LoadBalancer: Send + Sync { fn pick(&self, opts: &CallOptions) -> Result<Endpoint, Status>;
                                       fn update_endpoints(&self, endpoints: Vec<Endpoint>); }
pub trait Resolver: Send + Sync { fn resolve(&self) -> Pin<Box<dyn Stream<Item = Vec<Endpoint>> + Send>>; }
// crates/grpc-retry/src/policy.rs — Clone, immutable after construction
pub struct RetryPolicy { pub max_attempts: u32, pub initial_backoff: Duration,
                          pub max_backoff: Duration, pub backoff_multiplier: f64,
                          pub retryable_codes: Vec<Code>, pub idempotency_required: bool } // ADR-11/I6
pub struct RetryThrottle { bucket: AtomicU64, max_tokens: u32, token_ratio: f64 } // lock-free (I7)
impl RetryThrottle { pub fn allow_retry(&self) -> bool; pub fn on_failure(&self); pub fn on_success(&self); }
pub struct HedgingPolicy { pub max_hedges: u32, pub hedge_delay: Duration }
