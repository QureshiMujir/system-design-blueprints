Core schema & types (§11)

// cc-time — I7
type MonotonicNs = u64;                 // ns since arbitrary epoch; monotonic only
struct LogicalClock { now: MonotonicNs } // [own] sim clock; [conc] actor-local; drives cc-sim
trait Clock { fn now(&self) -> MonotonicNs; }  // MonoClock (prod) | LogicalClock (sim)

// cc-api — the pluggable boundary (P8, ADR-05)
struct AckInfo {                        // [ser] internal only
    ack_ts: MonotonicNs, cumulative_ack: u64, sacked: SmallVec<Range<u64>>,
    ecn_ce_count: u64, ts_echo: u32 }
struct RateSample {                     // produced by cc-rate (P9); [own] by controller call
    delivery_rate_bps: u64, rtt: MonotonicNs, is_app_limited: bool,
    acked_bytes: u64, lost_bytes: u64, prior_inflight: u64 }
struct CcContext<'a> { now: MonotonicNs, mss: u32, inflight: u64, tel: &'a Telemetry }
enum CcEvent { Sent(SegmentMeta), Ack(AckInfo), Loss(Range<u64>), Rto, EcnCe }
trait CongestionController {            // [conc] one actor writer (I8)
    fn on_event(&mut self, ev: &CcEvent, rs: &RateSample, ctx: &CcContext) -> CcDecision;
    fn cwnd_bytes(&self) -> u64;        // I1, I6
    fn pacing_rate_bps(&self) -> u64;   // I14
    fn state_tag(&self) -> &'static str;// telemetry label (P7)
}
struct CcDecision { cwnd_bytes: u64, pacing_rate_bps: u64, can_send: bool }
// cc-rate — I3, I4, I5
struct MinRttFilter { window_ns: u64, samples: WindowedMin<MonotonicNs> } // 10 s (M10)
struct BtlBwFilter  { window_rtts: u32, samples: WindowedMax<u64> }        // 10 RTT
struct RateSampler { delivered: u64, delivered_ts: MonotonicNs, app_limited_until: u64 }

// cc-reno / cc-cubic / cc-bbr runtime state (all [own] per-flow, [persist] via checkpoint)
struct RenoState  { cwnd: u64, ssthresh: u64, in_recovery: bool }          // §14
struct CubicState { cwnd: u64, w_max: u64, k_ns: u64, ssthresh: u64,        // §15, RFC 9438
                    epoch_start: MonotonicNs, tcp_cwnd: u64 }
enum BbrPhase { Startup, Drain, ProbeBw { gain_idx: u8 }, ProbeRtt }        // §17
struct BbrState { phase: BbrPhase, btlbw: u64, rtprop: MonotonicNs,
                  pacing_gain: Fixed, cwnd_gain: Fixed, full_bw_cnt: u8,
                  probe_rtt_deadline: MonotonicNs }                          // I13
// cc-loss — RACK-TLP (RFC 8985), RTO (RFC 6298); I12, I15
struct RackState { reo_wnd_ns: u64, xmit_ts: MonotonicNs, end_seq: u64, dsack_seen: bool }
struct RtoTimer  { srtt: u64, rttvar: u64, rto_ns: u64, backoff: u8 }        // I12 bounds

// cc-ecn — RFC 3168 / DCTCP / L4S (RFC 9331); I11
enum EcnCodepoint { NotEct, Ect0, Ect1, Ce }
struct DctcpState { alpha: Fixed, ce_ratio: Fixed, next_rtt_edge: MonotonicNs }

// cc-replay — I9; [persist] append-only, [compat] append-only fields + version tag
struct ReplayFrame { seq: u64, ts: MonotonicNs, kind: u8, len: u32, crc32c: u32 } // ADR-10
struct ReplayHeader { magic: u32, version: u16, seed: u64, algo: u8 }             // [compat]

// error taxonomy (P10) — no silent clamps
enum CcError { BadAck, RateInflation, InvariantBreach(u8), ReplayDivergence, FfiUnsafe }
