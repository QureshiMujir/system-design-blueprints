Core schema & types (§11)

pub const BLOCK_SIZE: usize = 16;
pub type Block = [u8; BLOCK_SIZE];

pub const AES_128_KEY_SIZE: usize = 16;
pub const AES_192_KEY_SIZE: usize = 24;
pub const AES_256_KEY_SIZE: usize = 32;

pub const AES_128_ROUNDS: usize = 10;
pub const AES_192_ROUNDS: usize = 12;
pub const AES_256_ROUNDS: usize = 14;
#[derive(Clone, ZeroizeOnDrop)]           // I1, I2, ADR-13; no Debug/Display/Serialize
pub struct RoundKeys128(pub(crate) [u32; 44]);
#[derive(Clone, ZeroizeOnDrop)]
pub struct RoundKeys192(pub(crate) [u32; 52]);
#[derive(Clone, ZeroizeOnDrop)]
pub struct RoundKeys256(pub(crate) [u32; 60]);
pub trait BlockEncrypt {
    fn encrypt_block(&self, block: &mut Block);
    fn encrypt_blocks(&self, blocks: &mut [Block]);   // dispatches to SIMD when available, P6
}
pub trait BlockDecrypt {
    fn decrypt_block(&self, block: &mut Block);
    fn decrypt_blocks(&self, blocks: &mut [Block]);
}
pub struct Aes128 { round_keys: RoundKeys128, backend: Backend }
pub struct Aes192 { round_keys: RoundKeys192, backend: Backend }
pub struct Aes256 { round_keys: RoundKeys256, backend: Backend }

impl Aes128 { pub fn new(key: &[u8; AES_128_KEY_SIZE]) -> Self; }
impl Aes192 { pub fn new(key: &[u8; AES_192_KEY_SIZE]) -> Self; }
impl Aes256 { pub fn new(key: &[u8; AES_256_KEY_SIZE]) -> Self; }

pub enum Backend { Aesni, Vaes, Neon, Soft }   // I12; selected once by dispatch::pick()
pub trait Aead {
    const KEY_LEN: usize;
    const NONCE_LEN: usize;
    const TAG_LEN: usize = 16;                 // I5, P9 — fixed, never configurable

    fn seal(key: &[u8], nonce: &[u8], aad: &[u8],
            inout: &mut [u8], tag_out: &mut [u8; 16]) -> Result<(), Error>;
    fn open(key: &[u8], nonce: &[u8], aad: &[u8],
            inout: &mut [u8], tag_in: &[u8; 16]) -> Result<(), Error>;
}
pub struct AesKw128;  pub struct AesKw256;      // RFC 3394; input len % 8 == 0 required (I9)
pub struct AesKwp128; pub struct AesKwp256;      // RFC 5649; any length, internal padding
pub enum Error {
    InvalidKeyLength, InvalidNonceLength, InvalidLength,
    CounterOverflow, AuthenticationFailed, UnsupportedParams,
}
