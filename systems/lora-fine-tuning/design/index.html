class AdapterMethod(str, Enum):
    LORA = "lora"; ADALORA = "adalora"; DORA = "dora"; IA3 = "ia3"
    PROMPT = "prompt"; PREFIX = "prefix"; BOTTLENECK = "bottleneck"

class TaskType(str, Enum):
    CAUSAL_LM = "causal_lm"; SEQ2SEQ_LM = "seq2seq_lm"
    SEQ_CLS = "seq_cls"; TOKEN_CLS = "token_cls"

@dataclass
class LoraConfig:
    method: AdapterMethod = AdapterMethod.LORA
    r: int = 8
    alpha: int = 16
    dropout: float = 0.05
    target_modules: tuple[str, ...] = ("q_proj", "v_proj")
    modules_to_save: tuple[str, ...] = ()
    bias: Literal["none", "all", "lora_only"] = "none"
    task_type: TaskType = TaskType.CAUSAL_LM
    base_model_name: str = ""
    fan_in_fan_out: bool = False
    init_lora_weights: Literal["kaiming", "gaussian", "pissa"] = "kaiming"
    use_rslora: bool = False
    use_dora: bool = False
    layers_to_transform: Optional[tuple[int, ...]] = None
    layers_pattern: Optional[str] = None

    def scaling(self) -> float: ...   # α/√r if use_rslora else α/r (ADR-03)
class Adapter(nn.Module, ABC):
    base: nn.Module            # the wrapped frozen layer; never copied, held by reference
    method: ClassVar[AdapterMethod]

    @abstractmethod
    def forward(self, x: Tensor) -> Tensor: ...

    @abstractmethod
    def merge_into_base(self) -> None: ...   # P5: must satisfy I6 within 1e-6

    @abstractmethod
    def trainable_parameters(self) -> Iterator[nn.Parameter]: ...

    @abstractmethod
    def step0_equivalence_delta(self) -> Tensor: ...  # I3 conformance hook
class LoraLinear(Adapter):
    method: ClassVar[AdapterMethod] = AdapterMethod.LORA
    r: int
    alpha: int
    scaling: float
    lora_A: nn.Parameter   # (r, in_features) — Kaiming or PiSSA init
    lora_B: nn.Parameter   # (out_features, r) — zero-init (I3, P13)
    dropout: nn.Module

    def __init__(self, base: nn.Linear, r: int, alpha: int,
                 dropout: float, init: str = "kaiming",
                 use_rslora: bool = False) -> None: ...
    def forward(self, x: Tensor) -> Tensor: ...     # y = base(x) + scaling*dropout(x)@A^T@B^T
    def merge_into_base(self) -> None: ...          # base.weight += scaling * B @ A (§8)
class AdaLoraLinear(Adapter):
    method: ClassVar[AdapterMethod] = AdapterMethod.ADALORA
    P: nn.Parameter        # (out_features, r)
    E: nn.Parameter        # (r,) — learned singular values, pruning target (I11)
    Q: nn.Parameter        # (r, in_features)
    ranknum: nn.Parameter  # non-trainable, current active rank

    def forward(self, x: Tensor) -> Tensor: ...       # y = base(x)+ (α/ranknum)·(x@Q^T*E)@P^T
    def orth_regularization(self) -> Tensor: ...       # ‖P^TP−I‖²+‖QQ^T−I‖² (§20.2)

class DoraLinear(Adapter):
    method: ClassVar[AdapterMethod] = AdapterMethod.DORA
    lora_A: nn.Parameter
    lora_B: nn.Parameter
    magnitude: nn.Parameter   # (out_features,) — init = ‖base.weight‖_c (ADR-09)

    def forward(self, x: Tensor) -> Tensor: ...  # V=W+s·BA; y=linear(x, m*(V/‖V‖_c))

class IA3Linear(Adapter):
    method: ClassVar[AdapterMethod] = AdapterMethod.IA3
    scale: nn.Parameter       # (out_features,), init 1.0

    def forward(self, x: Tensor) -> Tensor: ...  # y = base(x) * scale
class PromptEncoder(nn.Module):
    num_virtual_tokens: int
    embedding: nn.Embedding    # (num_virtual_tokens, hidden_size)

    def __init__(self, num_virtual_tokens: int, hidden_size: int,
                 init_from_vocab: Optional[Tensor] = None) -> None: ...
    def forward(self, batch_size: int, device: torch.device) -> Tensor: ...

class PrefixEncoder(nn.Module):
    num_virtual_tokens: int
    num_layers: int
    num_heads: int
    head_dim: int
    token_embedding: nn.Embedding
    mlp: nn.Sequential          # reparameterization trick (Li & Liang 2021)

    def __init__(self, num_virtual_tokens: int, num_layers: int,
                 num_heads: int, head_dim: int,
                 encoder_hidden: int = 512) -> None: ...
    def forward(self, batch_size: int,
                device: torch.device) -> Tensor: ...  # -> per-layer KV prefix

@dataclass
class AdapterMeta:
    name: str
    method: AdapterMethod
    config: LoraConfig
    base_model: str
    chat_template: str            # P3
    tokenizer_path: str
    trained_at: str                # ISO-8601
    package_version: str
    git_sha: str
    recipe_hash: str               # I12
    seed: int
    final_loss: float
    eval_metrics: dict[str, float]

class AdapterStore:
    _adapters: dict[str, dict[str, Tensor]]
    _meta: dict[str, AdapterMeta]
    _version: int                  # incremented on every mutation (I14)

    def mount(self, meta: AdapterMeta,
              tensors: dict[str, Tensor]) -> None: ...   # I7: rejects tensor-id collision
    def unmount(self, name: str) -> None: ...             # reference-counted (T10)
    def get(self, name: str, request_version: int) -> Optional[dict[str, Tensor]]: ...
    def list_names(self) -> list[str]: ...
@dataclass
class MergePlan:

    weights: tuple[float, ...]         # sums to 1.0 for multi-adapter merge
    target_dtype: Literal["bf16", "fp16", "fp32"]
    quantize_after: Optional[Literal["int8", "nf4"]] = None   # ADR-26, opt-in only

def execute_merge(base: nn.Module, plan: MergePlan) -> nn.Module: ...  # I6-checked
