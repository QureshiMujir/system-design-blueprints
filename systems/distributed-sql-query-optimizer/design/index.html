Core schema & types (§11)

#[derive(Clone, Copy, PartialEq, Eq, Hash, Ord, PartialOrd)] pub struct ColumnId(pub u32);
pub struct TableId(u32); pub struct IndexId(u32); pub struct FuncId(u32);
pub struct GroupId(u32); pub struct ExprId(u32); pub struct ScalarId(u32);
pub struct RuleId(u16); pub struct RegionId(u16); pub struct NodeId(u32); pub struct RangeId(u64);
pub struct PlaceholderIdx(u16);
pub struct ColSet(/* compact bitset over ColumnId: inline u64 + spill bitvec */);
impl ColSet { fn contains(self, c: ColumnId)->bool; fn union(self,o:Self)->Self; fn intersect(..); fn is_subset_of(..); }
pub enum RelExpr {
  Scan(ScanOp), IndexScan(IndexScanOp), Select{input:GroupId, filters:ScalarId}, Project{input:GroupId, projections:Vec<(ColumnId,ScalarId)>, passthrough:ColSet},
  InnerJoin(JoinOp), LeftJoin(JoinOp), SemiJoin(JoinOp), AntiJoin(JoinOp),
  HashJoin(JoinOp), MergeJoin(MergeJoinOp), LookupJoin(LookupJoinOp), InvertedJoin(InvertedJoinOp), ZigzagJoin(ZigzagJoinOp),
  GroupBy{input:GroupId, grouping_cols:ColSet, aggregations:Vec<AggExpr>}, ScalarGroupBy{input:GroupId, aggregations:Vec<AggExpr>},
  Distinct{input:GroupId, cols:ColSet}, Sort{input:GroupId, ordering:OrderingChoice}, Limit{input:GroupId, n:ScalarId, offset:ScalarId},
  UnionAll(SetOp), Union(SetOp), Intersect(SetOp), Except(SetOp), Values{rows:Vec<Vec<ScalarId>>, cols:ColSet}, WithScan{id:CteId},
  Window(WindowOp), Ordinality{input:GroupId},
  Insert(MutationOp), Update(MutationOp), Delete(MutationOp), Upsert(MutationOp),
  Distribute{input:GroupId, distribution:Distribution},  // physical enforcer
  SortEnforcer{input:GroupId, ordering:OrderingChoice},   // physical enforcer
}
pub enum ScalarExpr {
  Column(ColumnId), Const(Datum), Placeholder(PlaceholderIdx),
  And(Vec<ScalarId>), Or(Vec<ScalarId>), Not(ScalarId), Comparison{op:CmpOp, l:ScalarId, r:ScalarId},
  BinOp{op:ArithOp, l:ScalarId, r:ScalarId}, UnaryOp{op:UnOp, arg:ScalarId}, Function{id:FuncId, args:Vec<ScalarId>},
  Cast{expr:ScalarId, typ:DataType}, Case{branches:Vec<(ScalarId,ScalarId)>, els:Option<ScalarId>}, Coalesce(Vec<ScalarId>),
  Tuple(Vec<ScalarId>), Array(Vec<ScalarId>), Subquery(GroupId), Exists(GroupId), AnyAll{cmp:CmpOp, scalar:ScalarId, sub:GroupId},
}
pub struct ScanOp { table:TableId, cols:ColSet, hard_limit:Option<u64>, soft_limit:Option<u64> }
pub struct IndexScanOp { table:TableId, index:IndexId, constraint:Constraint, remaining:ScalarId, reverse:bool, cols:ColSet }
pub struct JoinOp { left:GroupId, right:GroupId, on:ScalarId, kind:JoinKind }
pub struct LookupJoinOp { input:GroupId, table:TableId, index:IndexId, key_cols:Vec<ColumnId>, on:ScalarId, kind:JoinKind }
pub enum Operator { /* flat tag union of every RelExpr + ScalarExpr variant — the memo intern key discriminant */ }
pub struct Memo { groups:Arena<Group>, exprs:Arena<MExpr>, scalars:Arena<ScalarExpr>,
  intern:HashMap<MExprKey,ExprId>, root:GroupId, budget:Cost, stats_epoch:u64 }
pub struct Group { id:GroupId, members:Vec<ExprId>, logical:LogicalProps,
  winners:HashMap<PhysProps,Winner>, explored:bool, state:GroupState }
pub struct MExpr { op:Operator, children:SmallVec<[GroupId;3]>, scalar:Option<ScalarId>, rules_applied:RuleMask }
pub struct Winner { expr:ExprId, cost:Cost, provided:PhysProps }
pub enum GroupState { Unexplored, Exploring, Explored }
struct MExprKey { op:Operator, children:SmallVec<[GroupId;3]>, scalar:Option<ScalarId> }  // intern key (I1)
impl Memo {
  fn add(&mut self, e:RelExpr) -> GroupId;          // interns + runs Normalize factory hook (§16)
  fn group(&self, g:GroupId) -> &Group;
  fn intern_expr(&mut self, op:Operator, ch:&[GroupId], sc:Option<ScalarId>) -> ExprId;  // dedup via MExprKey
  fn set_winner(&mut self, g:GroupId, req:&PhysProps, w:Winner);
}
pub struct LogicalProps { out_cols:ColSet, not_null:ColSet, fd:FuncDepSet, cardinality:Cardinality,
  outer_cols:ColSet, keys:Vec<ColSet>, stats:Statistics }
pub struct FuncDepSet { key:Option<ColSet>, equiv:EquivGroups, constants:ColSet, deps:Vec<FuncDep> }
pub struct FuncDep { from:ColSet, to:ColSet, strict:bool }
pub struct Cardinality { min:u64, max:u64 }            // max == u64::MAX ⇒ unbounded
pub struct PhysProps { ordering:OrderingChoice, distribution:Distribution, limit_hint:f64 }
pub struct OrderingChoice { cols:Vec<OrderingCol> }    // FD-simplified; canonical for hashing
pub struct OrderingCol { col:ColumnId, desc:bool }
pub enum Distribution { Any, Single(RegionId), Set(SmallVec<[RegionId;4]>) }
impl PhysProps { fn satisfies(&self, required:&PhysProps, fds:&FuncDepSet) -> bool; fn any() -> Self; }
impl FuncDepSet { fn closure(&self, cols:ColSet) -> ColSet; fn add_equiv(&mut self, a:ColumnId, b:ColumnId); }
impl OrderingChoice { fn simplify(&mut self, fds:&FuncDepSet); }  // drop constants, collapse equivalents (I6)
pub struct Statistics { row_count:f64, col_stats:HashMap<ColSet,ColumnStatistic>, selectivity:f64, is_estimated:bool }
pub struct ColumnStatistic { distinct_count:f64, null_count:f64, histogram:Option<Histogram>, avg_size:f64 }
pub struct Histogram { buckets:Vec<Bucket> }           // equi-depth
pub struct Bucket { upper_bound:Datum, num_eq:f64, num_range:f64, distinct_range:f64 }
pub fn selectivity(pred:&ScalarExpr, stats:&Statistics) -> f64;   // [0,1]; never panics on missing stats
pub struct Constraint { cols:Vec<ColumnId>, spans:Spans, tight:bool }   // tight ⇒ remaining filter empty
pub struct Spans(Vec<Span>);                                            // disjoint, sorted, normalized
pub struct Span { start:Key, start_boundary:Boundary, end:Key, end_boundary:Boundary }
pub enum Boundary { Inclusive, Exclusive }
pub struct Key(Vec<Datum>);                                             // index-key prefix
pub fn analyze(filter:&ScalarExpr, index:&IndexDesc) -> (Constraint, /*remaining*/ ScalarId);
pub struct Cost { cpu:f64, io:f64, net:f64, total:f64 }   // total is the ONLY comparison key
impl Cost { fn combine(self, child:Cost) -> Cost; fn is_finite(self) -> bool; const INF:Cost; }
impl PartialOrd for Cost { /* by .total; NaN treated as INF */ }
pub trait Coster {
  fn cost(&self, memo:&Memo, expr:ExprId, required:&PhysProps) -> Cost;          // local op cost
  fn lower_bound(&self, memo:&Memo, group:GroupId, required:&PhysProps) -> Cost; // B&B pruning bound (§22)
}
pub struct BaseCoster { coeffs:CostCoeffs }   // cpu/io/seek/net unit costs, loaded from costs.toml
pub enum RuleKind { Normalize, Explore, Implement }
pub struct RuleDef { id:RuleId, name:&'static str, kind:RuleKind }   // generated table from .opt
pub struct RuleMask(u128);                                           // per-MExpr "already fired" bitset
// .opt source shape (compiled at build time by quasar-optgen):
//   [EliminateAndAnd, Normalize] (And (And $a $b) $c) => (And $a $b $c)
//   [GenerateIndexScan, Explore]  (Select (Scan $t) $f) => (IndexScan $t (Constraint $f) ...)
//   [ImplementInnerJoinHash, Implement] (InnerJoin $l $r $on) => (HashJoin $l $r $on)
pub fn dispatch(memo:&mut Memo, expr:ExprId, rule:RuleId) -> SmallVec<[ExprId;4]>;  // generated
pub enum Task {
  OptimizeGroup{g:GroupId, req:PhysProps}, OptimizeExpr{e:ExprId, req:PhysProps},
  ExploreGroup{g:GroupId}, ExploreExpr{e:ExprId},
  ApplyRule{e:ExprId, r:RuleId}, EnforceProps{g:GroupId, req:PhysProps},
}
pub struct Optimizer { memo:Memo, coster:Box<dyn Coster>, stack:Vec<Task>, budget:Cost, spent:Cost, seed:u64, reorder_limit:u8 }
impl Optimizer {
  pub fn new(memo:Memo, coster:Box<dyn Coster>, cfg:OptConfig) -> Self;
  pub fn optimize(&mut self, req:PhysProps) -> Result<PhysicalPlan, OptError>;   // drives the task loop
  fn extract(&self, root:GroupId, req:&PhysProps) -> PhysicalPlan;               // winner tree → plan
}
pub struct PhysicalPlan { root:PhysicalNode }                 // optimizer output, pre-distribution
pub struct FlowSpec { processors:Vec<ProcessorSpec>, streams:Vec<Stream>, gateway:NodeId }
pub struct ProcessorSpec { id:ProcId, node:NodeId, core:ProcessorCore, post:PostProcessSpec, output:Vec<OutputRouter> }
pub enum ProcessorCore { TableReader{spans:Spans, index:IndexId}, HashJoiner, MergeJoiner, Aggregator, Sorter, Distinct, Limit, Values, Noop }
pub struct OutputRouter { router:Router, dest:Vec<ProcId> }
pub enum Router { PassThrough, ByHash(Vec<ColumnId>), Mirror, ByRange(Spans) }
pub struct Stream { source:ProcId, dest:ProcId, kind:StreamKind }
pub enum StreamKind { Local, Remote{ordered:bool} }
pub enum OptError {
  Unsupported{what:String, loc:Span}, NoPlanFound{goal:PhysProps}, StatsUnavailable{table:TableId},
  ConstraintUnsat, PlaceholderUnbound{idx:PlaceholderIdx}, MemoStale{expected:u64, found:u64},
  BudgetExceeded{spent:Cost, budget:Cost}, CatalogMissing{name:String},
}
