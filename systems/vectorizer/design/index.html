Core schema & types (§11)

// crates/vec-deps/src/lib.rs — owned by vec-deps; immutable once built; Send+Sync;
// not persisted; serializable for fixture snapshots (serde, deps/ bucket).
pub struct DependenceGraph { pub edges: Vec<DependenceEdge>, pub reduction_edges: Vec<InstructionId> }

pub struct DependenceEdge {
    pub from: InstructionId,
    pub to: InstructionId,
    pub kind: DependenceKind,
    pub direction: DirectionVector,
    pub distance: Option<DistanceVector>,
}

pub enum DependenceKind { True, Anti, Output, Input }   // RAW, WAR, WAW, RAR
pub struct DirectionVector(pub SmallVec<[Direction; 6]>);   // depth cap 6, §4
pub enum Direction { Equal, Less, Greater, LessEqual, GreaterEqual, Any }
pub struct DistanceVector(pub SmallVec<[i64; 6]>);

pub enum DepTest { Gcd, Banerjee, Omega }                   // ladder step that resolved (ADR-04)
pub fn analyze_dependences(loop_: &Loop, aa: &AliasAnalysis) -> DependenceGraph;
// crates/vec-stride/src/lib.rs — owned by vec-stride; pure function, no mutable state.
pub enum StrideKind { Constant(i64), Linear { base: ScevExpr, step: ScevExpr }, Unknown, Random }
pub fn analyze_stride(place: &Place, loop_: &Loop, se: &ScalarEvolution) -> StrideKind;
// crates/vec-legality/src/lib.rs — owned by vec-legality; the trust boundary (§3).
// Legal{..} is the only value vec-vplan may consume (I8).
pub enum LegalityVerdict { Legal { hints: LegalityHints }, Illegal { reason: IllegalReason, span: Span } }

pub struct LegalityHints {
    pub max_vf: u32,
    pub stride_pattern: StridePattern,
    pub needs_runtime_alias_check: bool,
    pub needs_masking: bool,
    pub reductions: Vec<ReductionKind>,
}

pub enum IllegalReason {
    NegativeDistanceDep(DependenceEdge),
    NonAffineAccess(Place),
    UnknownStride(Place),
    SideEffectInBody(InstructionId),
    ExceptionPossible(InstructionId),
    UnsupportedReduction(InstructionId),
}

pub fn check_legality(loop_: &Loop, deps: &DependenceGraph, strides: &[StrideKind], fast_math: bool) -> LegalityVerdict;
// crates/vec-vplan/src/lib.rs — owned by vec-vplan; constructed only from Legal (I8);
// not persisted; one instance per (VF, UF) candidate, dropped after costing/lowering.
pub struct VPlan { pub vf: u32, pub uf: u32, pub blocks: Vec<VPBasicBlock> }
pub struct VPBasicBlock { pub id: VPBlockId, pub instructions: Vec<VPInstruction> }

pub enum VPInstruction {
    VPLoad { ptr: VPValue, mask: Option<VPValue>, gather: bool },
    VPStore { value: VPValue, ptr: VPValue, mask: Option<VPValue>, scatter: bool },
    VPArith { op: BinOp, lhs: VPValue, rhs: VPValue },
    VPReduce { kind: ReductionKind, init: VPValue, body: VPValue },
    VPBroadcast(VPValue),
    VPGep { base: VPValue, idx: VPValue },
    VPSelect { mask: VPValue, t: VPValue, f: VPValue },
    VPPhi { incoming: Vec<(VPBlockId, VPValue)> },
}

pub fn build_vplan(loop_: &Loop, hints: &LegalityHints, vf: u32, uf: u32) -> VPlan;
// crates/vec-cost/src/lib.rs — CostModel impls are Send+Sync, stateless (I6).
pub trait CostModel: Send + Sync {
    fn instruction_cost(&self, inst: &VPInstruction, vf: u32) -> Cost;
    fn legal_vfs(&self) -> &[u32];
    fn name(&self) -> &str;
}
pub struct Cost { pub throughput: f32, pub latency: f32, pub register_pressure: u32 }
pub struct X86Cost { pub features: X86Features }         // crates/vec-cost-x86
pub struct Aarch64Cost { pub features: Aarch64Features }  // crates/vec-cost-aarch64
pub fn select_vf(hints: &LegalityHints, model: &dyn CostModel, plans: &[VPlan]) -> Option<VF>;
// crates/vec-mask/src/lib.rs — Phase 7; owned by vec-mask; consumes a VPlan, emits a
// masked/gather-lowered variant; no shared state across invocations.
pub struct MaskPlan { pub mask_values: Vec<VPValue>, pub guarded: Vec<InstructionId> }
pub fn if_convert(loop_: &Loop, cond: &CondExpr) -> MaskPlan;
pub fn lower_gather_scatter(vplan: &mut VPlan, stride: &StrideKind) -> GatherScatterReport;
// crates/vec-slp/src/lib.rs — Phase 8; a SeedTree is owned per basic block, dropped
// after packing; not persisted beyond the fixture snapshot.
pub struct SeedTree { pub root: InstructionId, pub levels: Vec<Vec<InstructionId>> }
pub fn find_seeds(block: &BasicBlock) -> Vec<InstructionId>;
pub fn build_tree(seed: InstructionId, block: &BasicBlock, max_depth: u32) -> Option<SeedTree>;
pub fn pack_tree(tree: &SeedTree, model: &dyn CostModel) -> Option<VPlan>;
// crates/vec-diff/src/lib.rs — Phase 9; the differential oracle harness (ADR-13).
pub struct DiffResult { pub fixture: String, pub scalar: Vec<Value>, pub vector: Vec<Value>, pub matched: bool }
pub fn run_differential(fixture: &Fixture, tolerance: FpTolerance) -> DiffResult;
