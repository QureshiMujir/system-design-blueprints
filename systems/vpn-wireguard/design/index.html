Core schema & types (§11)

// wg-core::key
struct PublicKey([u8; 32])          // Copy, Eq, Hash; wire-serialized as-is
struct PrivateKey([u8; 32])         // not Copy; zeroizes on Drop (I9); never serialized
struct PresharedKey([u8; 32])       // Copy, Eq; loaded from config, never logged
fn PrivateKey::generate() -> Self            // CSPRNG + RFC 7748 clamping
fn PrivateKey::public_key(&self) -> PublicKey // Curve25519 scalar-mult
impl FromStr for PublicKey                    // base64 (RFC 4648) decode, config-facing

// wg-core::peer
struct Peer {                       // owned by Tunnel; Arc-shared across tasks
  public_key: PublicKey,            // immutable identity
  preshared_key: Option<PresharedKey>,
  allowed_ips: AllowedIpsList,      // mutated only under config-write lock
  endpoint: RwLock<Option<SocketAddr>>,   // mutated per I5
  persistent_keepalive: Option<Duration>,
  sessions: Mutex<SessionPair>,     // mutex per P11
  last_handshake: RwLock<Option<Instant>>,
  stats: PeerStats,                 // atomics only, no lock
}
struct SessionPair { current: Option<Session>, previous: Option<Session>, next: Option<Session> }
struct PeerStats { tx_bytes: AtomicU64, rx_bytes: AtomicU64, tx_packets: AtomicU64, rx_packets: AtomicU64 }

// wg-core::session
struct Session {                    // one direction-pair of an established tunnel
  local_index: u32, remote_index: u32,      // wire-visible (P3)
  sending_key: SessionKey, receiving_key: SessionKey,   // never serialized, zeroize on Drop
  sending_counter: AtomicU64,       // I1
  receiving_window: ReplayWindow,   // I2
  established_at: Instant,
  last_sent: AtomicU64,             // monotonic-nanos
}
struct SessionKey { key: [u8; 32] } // ChaCha20-Poly1305 key material
struct ReplayWindow { bitmap: [AtomicU64; 32], last_received_counter: AtomicU64 } // 2048 bits
fn ReplayWindow::check_and_update(&self, counter: u64) -> Result<(), ReplayError>  // I2

// wg-core::allowedips
struct AllowedIpsList { v4: AllowedIpsTrie<Ipv4Addr>, v6: AllowedIpsTrie<Ipv6Addr> }
struct AllowedIpsTrie<A> { root: Option<Box<Node<A>>> }   // longest-prefix trie, I13
struct Node<A> { prefix: A, prefix_len: u8, peer: Option<Arc<Peer>>, left: Option<Box<Node<A>>>, right: Option<Box<Node<A>>> }
fn AllowedIpsList::lookup(&self, addr: IpAddr) -> Option<Arc<Peer>>
fn AllowedIpsList::insert(&mut self, range: IpRange, peer: Arc<Peer>)
fn AllowedIpsList::remove(&mut self, range: IpRange) -> Option<Arc<Peer>>
struct IpRange { addr: IpAddr, prefix_len: u8 }            // Copy
// wg-packet::format  (I12 — exact byte offsets, see §8 Tunnel Packet Format)
enum PacketType { HandshakeInitiation = 1, HandshakeResponse = 2, CookieReply = 3, TransportData = 4 }
struct HandshakeInitiation { message_type: u8, reserved: [u8;3], sender_index: u32,
  unencrypted_ephemeral: [u8;32], encrypted_static: [u8;48], encrypted_timestamp: [u8;28],
  mac1: [u8;16], mac2: [u8;16] }                            // 148 bytes total
struct HandshakeResponse { message_type: u8, reserved: [u8;3], sender_index: u32,
  receiver_index: u32, unencrypted_ephemeral: [u8;32], encrypted_nothing: [u8;16],
  mac1: [u8;16], mac2: [u8;16] }                            // 92 bytes total
struct CookieReply { message_type: u8, reserved: [u8;3], receiver_index: u32,
  nonce: [u8;24], encrypted_cookie: [u8;32] }               // 64 bytes total
struct TransportData { message_type: u8, reserved: [u8;3], receiver_index: u32,
  counter: u64 /* + variable ciphertext+tag payload */ }    // ≥32 bytes

// wg-noise::handshake
enum HandshakeRole { Initiator, Responder }
enum HandshakePhase { Initial, InitiatorSentInit, ResponderReceivedInit,
  ResponderSentResponse, InitiatorReceivedResponse, Complete }
struct HandshakeState { role: HandshakeRole, static_key: PrivateKey,
  remote_static_key: PublicKey, preshared_key: Option<PresharedKey>,
  ephemeral_key: Option<PrivateKey>, h: [u8;32], ck: [u8;32], state: HandshakePhase }
fn HandshakeState::new_initiator(static_key: PrivateKey, remote_static_key: PublicKey, psk: Option<PresharedKey>) -> Self
fn HandshakeState::new_responder(static_key: PrivateKey, psks: HashMap<PublicKey, PresharedKey>) -> Self
fn HandshakeState::create_initiation(&mut self) -> HandshakeInitiation
fn HandshakeState::consume_initiation(&mut self, msg: &HandshakeInitiation) -> Result<(), HandshakeError>
fn HandshakeState::create_response(&mut self) -> HandshakeResponse
fn HandshakeState::consume_response(&mut self, msg: &HandshakeResponse) -> Result<(), HandshakeError>
fn HandshakeState::into_session_keys(self) -> (SessionKey, SessionKey)

// wg-cookie
struct CookieMachine { secret: [u8;32], secret_expires_at: Instant, secret_lifetime: Duration }
fn CookieMachine::create_cookie(&self, peer_key: &PublicKey, src: SocketAddr) -> [u8;16]
fn CookieMachine::verify_cookie(&self, peer_key: &PublicKey, src: SocketAddr, mac2: [u8;16]) -> bool
fn CookieMachine::should_challenge(&self) -> bool                     // I8

// wg-tun-userspace
trait Tun: Send + Sync { fn read_packet(&mut self, buf: &mut [u8]) -> io::Result<usize>;
  fn write_packet(&mut self, packet: &[u8]) -> io::Result<()>; fn name(&self) -> &str;
  fn mtu(&self) -> u32; fn set_mtu(&mut self, mtu: u32) -> io::Result<()> }
fn open_tun(name: &str) -> io::Result<impl Tun>       // per-OS impl selected at compile time

// wg-udp
struct UdpTransport { sock: UdpSocket }                // one per Tunnel (P10)
fn UdpTransport::bind(addr: SocketAddr) -> io::Result<Self>
fn UdpTransport::send_to(&self, packet: &[u8], dst: SocketAddr) -> io::Result<()>
fn UdpTransport::recv_from(&self, buf: &mut [u8]) -> io::Result<(usize, SocketAddr)>
