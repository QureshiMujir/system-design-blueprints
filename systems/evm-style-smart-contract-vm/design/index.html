Core schema & types (§11)

// ownership: Copy, stack-allocated; mutation: immutable once constructed;
// concurrency: Send+Sync; persistence: serialized big-endian at ABI boundary;
// determinism: load-bearing — wrapping arithmetic only, never panics
struct U256([u64; 4])                       // 256-bit word, 4x-u64 limbs
struct Address([u8; 20])                    // account identifier
type Word = U256
struct Bytes32([u8; 32])                    // hash/root/topic type
type Bytes = Vec<u8>

// ownership: Copy; determinism: load-bearing, table lookup only
enum Opcode { Stop, Add, Mul, /* ... full set, see §8.2/Appendix A */ Push(u8), Dup(u8), Swap(u8), Log(u8), Call, DelegateCall, StaticCall, Create, Create2, Return, Revert, SelfDestruct, Invalid }

// ownership: owned by ExecutableBytecode; immutable post-decode
struct Instruction { opcode: Opcode, immediate: Option<Bytes>, pc: usize }

// mutation: per-frame, mutable; concurrency: not shared across frames;
// determinism: load-bearing bound (I3)
struct Stack { data: [Word; 1024], top: usize }

// mutation: per-frame, grows monotonically; determinism: load-bearing (expansion gas)
struct Memory { bytes: Vec<u8>, last_charged_words: u64 }

// mutation: decremented monotonically per opcode; determinism: load-bearing (I2)
struct Gas(u64)
struct GasSchedule { fork: ForkId, costs: /* per-opcode static + dynamic fn table */ }
struct AccessList { addresses: HashSet<Address>, storage_keys: HashSet<(Address, U256)> }
// concurrency: single-threaded per frame; determinism: the whole interpreter's contract
struct Interpreter<'a> { code: &'a ExecutableBytecode, pc: usize, stack: Stack, memory: Memory, gas: Gas }

enum ExecutionResult {
    Success { gas_used: Gas, output: Bytes, logs: Vec<Log> },
    Revert  { gas_used: Gas, output: Bytes },
    Halt    { reason: HaltReason, gas_used: Gas },
}

enum HaltReason { OutOfGas, StackOverflow, StackUnderflow, InvalidOpcode, InvalidJump, StaticViolation, DepthExceeded, InsufficientBalance, CreateCollision, CodeSizeExceeded }

// ownership: one per nested call; mutation: owns its Stack/Memory; not shared
struct CallFrame { caller: Address, callee: Address, kind: CallKind, value: U256, gas_limit: Gas, is_static: bool, returndata: Bytes }
enum CallKind { Call, DelegateCall, StaticCall, Create, Create2 }
struct Message { sender: Address, target: Address, value: U256, input: Bytes, gas: Gas, kind: CallKind }
// persistence: durable, committed via trie; determinism: load-bearing (I1)
struct Account { nonce: u64, balance: U256, code_hash: Bytes32, storage_root: Bytes32 }
struct WorldState { accounts: BTreeMap<Address, Account> }   // BTreeMap: ordered iteration, determinism-safe
type StorageKey = U256
type StorageValue = U256

// determinism: load-bearing — node encoding must match spec exactly
enum MptNode { Leaf { path: Vec<Nibble>, value: Bytes }, Extension { path: Vec<Nibble>, child: Bytes32 }, Branch { children: [Option<Bytes32>; 16], value: Option<Bytes> } }
struct Nibble(u8)                            // 4-bit value, 0..=15
struct Rlp;                                  // encode/decode namespace
type StateRoot = Bytes32
// mutation: append-only until revert/commit; concurrency: owned by one execution context
enum JournalEntry { StorageWrite { address: Address, slot: U256, previous: U256 }, BalanceChange { address: Address, previous: U256 }, NonceChange { address: Address, previous: u64 }, AccountCreated { address: Address }, AccountDestroyed { address: Address } }
struct Journal { entries: Vec<JournalEntry> }
struct SnapshotId(usize)

struct Log { address: Address, topics: Vec<Bytes32>, data: Bytes }
struct Receipt { status: bool, gas_used: Gas, logs: Vec<Log>, bloom: [u8; 256] }
struct Transaction { nonce: u64, gas_limit: Gas, to: Option<Address>, value: U256, data: Bytes, /* fee fields per tx-type */ }

// compatibility: EVMC-style boundary; concurrency: host provides thread-safety
trait Precompile { fn run(&self, input: &[u8], gas_limit: Gas) -> ExecutionResult; }
trait Host { fn get_account(&self, addr: &Address) -> Option<Account>; fn set_storage(&mut self, addr: Address, key: U256, value: U256); fn call(&mut self, msg: Message) -> ExecutionResult; }
trait Tracer { fn on_step(&mut self, pc: usize, opcode: Opcode, gas: Gas, stack: &Stack); }   // read-only observer, P10/I13
