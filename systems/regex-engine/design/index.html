#![forbid(unsafe_code)]
pub struct Regex { /* opaque */ }
pub struct RegexSet { /* opaque */ }
pub struct RegexBuilder { /* opaque */ }

pub struct Match<'h> { haystack: &'h [u8], range: std::ops::Range<usize> }
impl<'h> Match<'h> {
    pub fn start(&self) -> usize;
    pub fn end(&self) -> usize;
    pub fn as_str(&self) -> &'h str;   // Unicode mode only
    pub fn as_bytes(&self) -> &'h [u8];
}

pub struct Captures<'h> { /* indexed by name and by group number */ }

impl Regex {
    pub fn new(pattern: &str) -> Result<Self, ParseError>;
    pub fn is_match(&self, haystack: &str) -> bool;
    pub fn find<'h>(&self, haystack: &'h str) -> Option<Match<'h>>;
    pub fn find_iter<'h>(&self, haystack: &'h str) -> impl Iterator<Item = Match<'h>> + 'h;
    pub fn captures<'h>(&self, haystack: &'h str) -> Option<Captures<'h>>;
    pub fn literal_strategy(&self) -> LiteralStrategy<'_>;   // P10
    pub fn engine(&self) -> Engine;                          // P3
}
pub struct RegexBuilder {
    pattern: String,
    case_insensitive: bool, multi_line: bool, dot_all: bool,   // (?i) (?m) (?s)
    unicode: bool,                                             // P5, default true
    longest_match: bool,                                       // P6
    size_limit: usize, dfa_size_limit: usize, backtrack_budget: u64,
    engine_pref: EnginePref,
}
impl RegexBuilder {
    pub fn new(pattern: &str) -> Self;
    pub fn unicode(self, yes: bool) -> Self;
    pub fn longest_match(self, yes: bool) -> Self;
    pub fn size_limit(self, bytes: usize) -> Self;
    pub fn dfa_size_limit(self, bytes: usize) -> Self;
    pub fn backtrack_budget(self, steps: u64) -> Self;
    pub fn engine(self, pref: EnginePref) -> Self;
    pub fn build(self) -> Result<Regex, ParseError>;
}
pub enum EnginePref { Auto, PikeVm, LazyDfa, Backtrack }
pub struct ParseError { pub pattern: String, pub span: std::ops::Range<usize>, pub kind: ParseErrorKind }

#[non_exhaustive]
pub enum ParseErrorKind {
    UnclosedGroup, UnclosedClass, InvalidEscape(char),
    InvalidUnicodeProperty(String), InvalidRepetition, InvalidBackref(u32),
    InvalidLookaround, BackrefRequiresBacktrack, LookaroundRequiresBacktrack,  // ADR-03
    EngineUnsupported { needed: Engine, requested: Engine },
    SizeLimitExceeded,
}

#[non_exhaustive]
pub enum MatchError { Budget, DfaCacheExhausted, InvalidUtf8 { offset: usize } }
pub struct Span(pub std::ops::Range<usize>);
pub enum Ast {
    Empty, Literal { c: char, span: Span }, Class(ClassSet), Anchor(Anchor),
    Repetition { ast: Box<Ast>, kind: RepetitionKind, greedy: bool, span: Span },
    Group { kind: GroupKind, ast: Box<Ast>, span: Span },
    Concat(Vec<Ast>, Span), Alternation(Vec<Ast>, Span),
    Backreference { number: u32, span: Span },
    Lookaround { kind: LookaroundKind, ast: Box<Ast>, span: Span },
}
pub enum RepetitionKind { ZeroOrOne, ZeroOrMore, OneOrMore, Range { min: u32, max: Option<u32> } }
pub enum GroupKind { CapturingNumbered, CapturingNamed(String), NonCapturing, Atomic }
pub enum LookaroundKind { PositiveLookahead, NegativeLookahead, PositiveLookbehind, NegativeLookbehind }
pub enum Hir {
    Empty, Literal(Vec<u8>), Class(ClassUnicode), ClassBytes(ClassBytes),
    Look(Look), Repetition(Repetition), Capture(Capture),
    Concat(Vec<Hir>), Alternation(Vec<Hir>), Backreference(u32), Lookaround(Lookaround),
}
pub struct Repetition { pub min: u32, pub max: Option<u32>, pub greedy: bool, pub sub: Box<Hir> }
pub struct Capture { pub index: u32, pub name: Option<String>, pub sub: Box<Hir> }
pub enum Look { Start, End, StartLF, EndLF, WordAscii, NotWordAscii, WordUnicode, NotWordUnicode }
pub struct Nfa {
    pub instructions: Vec<Instruction>, pub start: StateId,
    pub match_states: Vec<StateId>, pub captures: u32, pub byte_classes: ByteClasses,
}
pub struct StateId(pub u32);
pub enum Instruction {
    Byte { byte: u8, next: StateId },
    Class { class_id: u32, next: StateId },
    Split { next_a: StateId, next_b: StateId },
    Capture { slot: u32, next: StateId },
    Look { look: Look, next: StateId },
    Backref { group: u32, case_insensitive: bool, next: StateId },
    Match { pattern_id: u32 },
}
pub trait Engine { type Match; fn search(&self, input: &Input<'_>, scratch: &mut Scratch)
    -> Result<Option<Self::Match>, MatchError>; }

pub struct PikeVm { program: std::sync::Arc<Nfa> }
pub struct LazyDfa { program: std::sync::Arc<Nfa>, cache: DfaCache }
pub struct Backtrack { program: std::sync::Arc<Nfa>, budget: u64 }

pub struct Input<'h> { pub haystack: &'h [u8], pub start: usize, pub end: usize, pub anchored: Anchored }
pub struct Cursor<'h> { haystack: &'h [u8], offset: usize }
impl<'h> Cursor<'h> {
    pub fn current_byte(&self) -> Option<u8>;
    pub fn next_codepoint(&mut self) -> Option<char>;   // Unicode mode only (P12)
    pub fn position(&self) -> usize;
}

pub struct Scratch { arena: bumpalo::Bump, threads: ThreadList, slots: SlotVec, stack: BacktrackStack }
impl Scratch { pub fn reset(&mut self); }   // I8
pub struct DfaCache { states: std::collections::HashMap<StateSet, StateId>,
    trans: Vec<std::collections::HashMap<u8, StateId>>, bytes: usize, cap: usize }

pub enum LiteralStrategy<'a> { BoyerMoore(&'a [u8]), AhoCorasick(&'a [Vec<u8>]), None }

pub struct RegexSet { /* opaque */ }
impl RegexSet {
    pub fn new<I: IntoIterator<Item = S>, S: AsRef<str>>(patterns: I) -> Result<Self, ParseError>;
    pub fn matches(&self, haystack: &str) -> SetMatches;   // I15
}
pub struct SetMatches { /* bit-vector indexed by pattern */ }
pub struct ConformanceReport { pub schema_version: String, pub engine_version: String,
    pub host: HostInfo, pub buckets: Vec<BucketReport>, pub outcome: Outcome }
pub struct BucketReport { pub name: String, pub outcome: Outcome, pub cases_run: u64,
    pub cases_pass: u64, pub cases_fail: u64, pub cases_deferred: u64, pub duration_ms: u64 }
pub enum Outcome { Pass, Fail, Deferred }
