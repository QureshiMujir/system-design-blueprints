// slam-lie — manifolds (right perturbation: X⊞ξ = X·Exp(ξ))
pub trait LieGroup: Sized + Copy {
    type Tangent;                              // R^DOF (SE2: 3, SE3: 6)
    const DOF: usize;
    fn identity() -> Self;
    fn inverse(&self) -> Self;
    fn compose(&self, rhs: &Self) -> Self;     // X∘Y
    fn exp(xi: &Self::Tangent) -> Self;        // Exp: tangent → group
    fn log(&self) -> Self::Tangent;            // Log: group → tangent
    fn adjoint(&self) -> /* DOF×DOF */ Matrix; // Ad_X
    fn boxplus(&self, xi: &Self::Tangent) -> Self;   // X·Exp(ξ)
    fn boxminus(&self, other: &Self) -> Self::Tangent; // Log(other⁻¹·self)  (or self⁻¹·other — locked §8.1)
}
pub struct SO2 { theta: Real }                 // det=+1 invariant (I1)
pub struct SO3 { q: Quaternion }               // |q|=1 (Hamilton, scalar-first) (I1)
pub struct SE2 { r: SO2, t: Vector2 }          // 3-DOF
pub struct SE3 { r: SO3, t: Vector3 }          // 6-DOF
pub struct Quaternion { w: Real, x: Real, y: Real, z: Real }
pub fn right_jacobian_so3(phi: &Vector3) -> Matrix3;   // Jr
pub fn left_jacobian_se3(xi: &Vector6) -> Matrix6;     // Jl (for adjoint-based Jacobians)

// slam-sensor
pub struct Timestamp { ns: u64 }
pub struct LaserScan {
    angle_min: Real, angle_inc: Real,          // bearings
    ranges: Vec<Real>, valid: Vec<bool>,       // range + validity mask
    range_max: Real, sensor_in_body: SE2,      // sensor pose offset
}
impl LaserScan { fn endpoints(&self) -> Vec<Vector2>; }   // Cartesian (sensor frame)
pub trait BeamModel { fn likelihood(&self, expected: Real, measured: Real) -> Real; } // p_hit/short/max/rand
pub trait InverseSensorModel { fn cell_logodds(&self, hit: bool, range_frac: Real) -> Real; }
pub trait ScanSource { fn next(&mut self) -> Option<(Timestamp, LaserScan, OdomDelta)>; }

// slam-map
pub struct OccupancyGrid {
    origin: Vector2, resolution: Real,         // meters/cell
    width: usize, height: usize,
    logodds: Vec<i16>,                         // clamped [l_min,l_max] (I12)
}
impl OccupancyGrid {
    fn world_to_cell(&self, p: &Vector2) -> Option<(usize,usize)>;
    fn integrate_scan(&mut self, pose: &SE2, scan: &LaserScan, model: &dyn InverseSensorModel); // ray-cast
    fn prob(&self, x: usize, y: usize) -> Real;          // logodds → probability
}
pub struct LandmarkMap { landmarks: Vec<(Vector2, Matrix2 /*cov*/)> }
// slam-match
pub struct MatchResult { rel_pose: SE2, information: Matrix3, score: Real, iters: u32, converged: bool }
pub trait ScanMatcher {
    fn match_scan(&self, source: &LaserScan, target: &MatchTarget, init: &SE2) -> Result<MatchResult, MatchError>;
}
pub struct Icp { mode: IcpMode, max_iters: u32, tol: Real }   // PointToPoint | PointToLine
pub struct Correlative { search_window: SearchWindow, resolutions: Vec<Real> } // multi-resolution
pub struct Ndt { cell_size: Real, max_iters: u32 }
pub enum MatchTarget<'a> { Scan(&'a LaserScan), Grid(&'a OccupancyGrid) }

// slam-odom
pub struct OdomDelta { rel: SE2 }              // relative SE(2) increment (right-composed)
pub trait MotionModel {
    fn predict(&self, x: &SE2, u: &OdomDelta) -> (SE2, Matrix3 /*Σ*/);  // mean + covariance
    fn sample(&self, x: &SE2, u: &OdomDelta, rng: &mut Rng) -> SE2;     // FastSLAM proposal
}
pub struct OdometryModel { alpha: [Real; 4] }  // rot1/trans/rot2 noise params
pub struct VelocityModel { alpha: [Real; 6] }

// slam-ekf
pub struct EkfSlam {
    mu: Vec<Real>,                              // [x_R(3); landmarks(2N)]
    sigma: DMatrix,                             // dense (3+2N)² — SPD (I2)
    ids: Vec<LandmarkId>,
}
impl EkfSlam {
    fn predict(&mut self, u: &OdomDelta, model: &dyn MotionModel);
    fn update(&mut self, z: &Observation) -> UpdateOutcome;   // associate (I10) + Joseph update
    fn augment(&mut self, z: &Observation) -> LandmarkId;     // new landmark
    fn pose(&self) -> SE2;
}

// slam-pf
pub struct Particle { trajectory: Vec<SE2>, pose: SE2, map: OccupancyGrid, weight: Real }
pub struct FastSlam {
    particles: Vec<Particle>, n_eff_thresh: Real, rng: Rng,   // seeded (I5)
}
impl FastSlam {
    fn step(&mut self, u: &OdomDelta, scan: &LaserScan);      // propose+weight+map; resample on N_eff (I14)
    fn best(&self) -> &Particle;                              // max-weight particle
    fn n_eff(&self) -> Real;
}
// slam-graph
pub struct VarId(u64);  pub struct FactorId(u64);
pub enum Variable { Pose2(SE2), Pose3(SE3), Point2(Vector2), Point3(Vector3) }
pub trait Factor {
    fn variables(&self) -> &[VarId];
    fn residual(&self, values: &Values) -> Vec<Real>;        // r(θ)
    fn jacobians(&self, values: &Values) -> Vec<Block>;      // ∂r/∂ξ per variable (on-manifold)
    fn information(&self) -> &DMatrix;                        // Ω (whitening source)
    fn robust_kernel(&self) -> Option<&dyn RobustKernel>;     // loop factors only
}
pub struct FactorGraph { vars: HashMap<VarId, Variable>, factors: Vec<Box<dyn Factor>> }
pub struct Values { map: HashMap<VarId, Variable> }          // current estimate θ
impl FactorGraph {
    fn add_variable(&mut self, id: VarId, v: Variable);
    fn add_factor(&mut self, f: Box<dyn Factor>) -> FactorId;
    fn read_g2o(path: &Path) -> Result<(Self, Values), G2oError>;
    fn write_g2o(&self, values: &Values, path: &Path) -> Result<(), G2oError>;
}

// slam-solver
pub trait RobustKernel { fn weight(&self, e2: Real) -> Real; } // IRLS weight from squared error
pub struct Huber { delta: Real }  pub struct Cauchy { c: Real }  pub struct Dcs { phi: Real }
pub struct SparseCholesky { /* CSC L, fill-reducing perm */ }
impl SparseCholesky {
    fn factorize(h: &CscMatrix) -> Result<Self, LinalgError>; // PD check (I2); AMD ordering (I8)
    fn solve(&self, b: &[Real]) -> Vec<Real>;                 // Hδ=b without explicit inverse
}
pub trait Solver {
    fn optimize(&mut self, graph: &FactorGraph, values: &mut Values) -> SolveReport;
}
pub struct GaussNewton { max_iters: u32, tol: SolveTol }
pub struct Levenberg { max_iters: u32, lambda0: Real, tol: SolveTol }   // default (ADR 28)
pub struct SolveReport { iters: u32, initial_cost: Real, final_cost: Real, grad_norm: Real, converged: bool }
pub enum SolveStrategy { Direct, SchurLandmarks }            // Schur-complement on points (§8.7)
// slam-loop
pub struct ScanContext { rings: usize, sectors: usize, desc: Vec<Vec<Real>> }
pub struct LoopCandidate { from: VarId, to: VarId, descriptor_dist: Real }
pub trait LoopDetector {
    fn add_keyframe(&mut self, id: VarId, scan: &LaserScan, pose: &SE2);
    fn detect(&self, scan: &LaserScan) -> Vec<LoopCandidate>;        // appearance proposal
}
pub struct LoopVerifier { matcher: Box<dyn ScanMatcher>, chi2_gate: Real }
impl LoopVerifier {
    fn verify(&self, cand: &LoopCandidate, cur: &LaserScan, kf: &LaserScan, est: &Values)
        -> Option<LoopConstraint>;                                  // geometric gate (I6)
}

// slam-pipeline
pub struct SlamSystem { /* frontend, backend, loop, map, mode */ }
impl SlamSystem {
    fn submit(&mut self, t: Timestamp, scan: LaserScan, odom: OdomDelta) -> PoseEstimate;
    fn trajectory(&self) -> Vec<SE2>;
    fn map(&self) -> &OccupancyGrid;
    fn relocalize(&mut self, scan: &LaserScan) -> Option<SE2>;      // global scan-context match
    fn save(&self, path: &Path) -> Result<(), IoError>;
    fn load(path: &Path) -> Result<Self, IoError>;
}
pub enum SlamError { NotPositiveDefinite, GaugeUnanchored, BadScan, MatchFailed,
                     SolveDiverged, AssociationAmbiguous, G2oParse, FormatVersion }
