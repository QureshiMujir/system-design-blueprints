Core schema & types (§11)

struct SentPacket
  packet_number: u64
  packet_space: PacketSpaceId
  sent_time: Instant
  ack_eliciting: bool
  bytes_sent: u32
  inflight: bool
  retransmittable_frames: FrameSet
  delivery_state: DeliveryState
  ecn_marking: EcnCodepoint
enum DeliveryState = Sent | Acked | Lost | PtoPending
enum CongestionState = SlowStart | CongestionAvoidance | Recovery | PersistentCongestion
trait RttEstimator
  fn update(sample: Duration, ack_delay: Duration);
  fn pto_base() -> Duration;
  fn smoothed_rtt() -> Duration;
  fn rttvar() -> Duration;
  fn min_rtt() -> Duration;
trait CongestionController
  fn on_packet_sent(bytes: u32, pn: u64);
  fn on_packet_acked(bytes: u32, pn: u64, rtt: Duration);
  fn on_packet_lost(bytes: u32, pn: u64);
  fn on_ecn_ce(pn: u64);
  fn cwnd() -> u64;
  fn pacing_rate() -> u64;
  fn state() -> CongestionState;
struct AckRange
  start: u64
  end: u64
struct PacketSpaceId(u8)   // Initial = 0, Handshake = 1, Application = 2
struct TokenBucket
  capacity_bytes: u32
  tokens_bytes: u32
  refill_rate_bps: u64
  last_refill: Instant
struct ReplayEvent
  seq: u64
  connection_epoch: u32
  ts_monotonic_ns: u64
  kind: EventKind
  payload: EventPayload
struct EcnCounters
  ect0_sent: u64
  ect1_sent: u64
  ce_received: u64
  ect0_acked: u64
  ect1_acked: u64
  ce_acked: u64
struct LossTimer
  packet_space: PacketSpaceId
  deadline: Instant
  kind: LossTimerKind

enum LossTimerKind = PacketThreshold | TimeThreshold | Pto
struct BandwidthSample
  delivered_bytes: u64
  interval: Duration
  is_app_limited: bool
  rate_bps: u64
