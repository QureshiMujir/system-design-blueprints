Core schema & types (§11)

pub struct Image<T> { data: Vec<T>, w: usize, h: usize, stride: usize, channels: u8 } // row-major
pub trait Pixel: Copy { /* u8, u16, f32; saturating cast helpers (I4) */ }
impl<T:Pixel> Image<T> { fn at(&self,x,y,c)->T; fn at_mut(&mut self,x,y,c)->&mut T;
    fn roi(&self, r:Rect)->ImageView<'_,T>;  fn dims(&self)->(usize,usize,u8);
    fn map<U:Pixel>(&self, f:impl Fn(T)->U)->Image<U>; }                  // point op (pure, principle 3)
pub struct ImageView<'a,T>{ /* borrows parent; zero-copy ROI */ }
pub enum BorderMode { Constant(f32), Replicate, Reflect, Reflect101, Wrap } // I5
pub enum Kernel { Separable{ row:Vec<f32>, col:Vec<f32>, anchor:(i32,i32) },
                  Dense{ mat:Mat<f32>, anchor:(i32,i32) } }               // I2/I3
impl Kernel { fn is_separable(m:&Mat<f32>)->Option<(Vec<f32>,Vec<f32>)>;  // rank-1 detect (SVD #349)
              fn gaussian(sigma:f32, radius:Option<usize>)->Kernel;
              fn sobel_x()->Kernel; fn flip(&self)->Kernel; }
pub enum StructElem { Rect(usize,usize), Ellipse(usize,usize), Cross(usize,usize), Custom(Image<u8>) }
pub struct Point { x:f32, y:f32 }  pub struct Size{ w:usize, h:usize }
pub struct Rect { x:i32, y:i32, w:u32, h:u32 }   // overflow-checked ops (I13)
pub struct Line { rho:f32, theta:f32, votes:u32 }            // Hough output (I6)
pub struct LineSeg { a:Point, b:Point }                       // probabilistic Hough
pub struct Circle { center:Point, r:f32, votes:u32 }
pub struct KeyPoint { pt:Point, response:f32, angle:f32, scale:f32 }
pub struct Mat3(/*3×3 homography/affine*/ [[f32;3];3]);
impl Mat3 { fn apply(&self, p:Point)->Point; fn inverse(&self)->Option<Mat3>; }  // I14/I15
pub fn correlate<T:Pixel>(src:&Image<T>, k:&Kernel, border:BorderMode) -> Image<f32>;
pub fn convolve<T:Pixel>(src:&Image<T>, k:&Kernel, border:BorderMode) -> Image<f32>; // = correlate(flip(k)) (I2)
pub fn fft_convolve(src:&Image<f32>, k:&Mat<f32>) -> Image<f32>;                       // #369 (large k)
pub fn box_filter<T>(src:&Image<T>, win:Size, border:BorderMode) -> Image<T>;          // integral O(1)/px
pub fn gaussian_blur<T>(src:&Image<T>, sigma:f32, border:BorderMode) -> Image<T>;      // separable
pub fn median_blur<T>(src:&Image<T>, win:usize) -> Image<T>;                           // histogram
pub fn bilateral(src:&Image<f32>, d:usize, sigma_color:f32, sigma_space:f32) -> Image<f32>;
pub fn sobel<T>(src:&Image<T>, dx:u8, dy:u8, aperture:usize) -> Image<f32>;
pub fn gradient_mag_orient<T>(src:&Image<T>) -> (Image<f32>/*mag*/, Image<f32>/*θ rad, I9*/);
pub fn canny<T>(src:&Image<T>, lo:f32, hi:f32, aperture:usize) -> Image<u8>;           // 1-px, 8-conn (I8)
pub fn log_edges<T>(src:&Image<T>, sigma:f32) -> Image<u8>;                            // Marr-Hildreth
pub fn hough_lines(edges:&Image<u8>, rho_res:f32, theta_res:f32, thresh:u32) -> Vec<Line>;        // I6
pub fn hough_lines_p(edges:&Image<u8>, .., min_len:f32, max_gap:f32) -> Vec<LineSeg>;
pub fn hough_circles(src:&Image<u8>, dp:f32, min_dist:f32, r:(f32,f32)) -> Vec<Circle>;
pub fn morphology<T>(src:&Image<T>, op:MorphOp, se:&StructElem, border:BorderMode) -> Image<T>;    // I10
pub enum MorphOp { Erode, Dilate, Open, Close, Gradient, TopHat, BlackHat, HitMiss }
pub fn distance_transform(bin:&Image<u8>, metric:DistMetric) -> Image<f32>;
pub fn harris<T>(src:&Image<T>, k:f32, win:usize, thresh:f32) -> Vec<KeyPoint>;
pub fn fast<T>(src:&Image<T>, t:u8, n:u8) -> Vec<KeyPoint>;
pub fn orb<T>(src:&Image<T>, n:usize, rng:&mut Rng) -> (Vec<KeyPoint>, BinaryDescriptors);          // I7
pub fn warp_affine<T>(src:&Image<T>, m:&Mat3/*2×3*/, dst:Size, interp:Interp, border:BorderMode)->Image<T>;
pub fn warp_perspective<T>(src:&Image<T>, h:&Mat3, dst:Size, interp:Interp, border:BorderMode)->Image<T>;
pub fn find_homography(src:&[Point], dst:&[Point], method:FitMethod, rng:&mut Rng) -> Option<(Mat3, Vec<bool>/*inliers*/)>; // DLT+RANSAC #349 (I15)
pub enum Interp { Nearest, Bilinear, Bicubic }  pub enum FitMethod { Dlt, Ransac{iters:usize, thresh:f32} }
pub fn connected_components(bin:&Image<u8>, conn:u8) -> CcLabels;       // 4/8-conn (the #387 primitive)
pub fn find_contours(bin:&Image<u8>) -> (Vec<Contour>, Hierarchy);     // Suzuki
pub fn match_template<T>(img:&Image<T>, tmpl:&Image<T>, method:MatchMethod) -> Image<f32>;
pub enum CvError { BadParam{what:&'static str}, DimMismatch{expected:(usize,usize),got:(usize,usize)},
    DtypeMismatch, ImageTooLarge{px:u64}, DegenerateConfig, SingularTransform, EvenKernel, NonPositiveSigma }
pub struct OpTrace { op:&'static str, in_dims:(usize,usize), out_dims:(usize,usize), ms:f32,
    border:&'static str, separable:bool, fft_path:bool, edge_pixels:u32, n_peaks:u32, n_keypoints:u32,
    inliers:u32, seed:u64, n_threads:usize }
pub struct CcLabels { labels: Image<u32>, n: u32, boxes: Vec<Rect>, areas: Vec<u32> }   // connected components (I13)
pub struct Contour { points: Vec<Point>, is_hole: bool }                                  // Suzuki output
pub struct Hierarchy { next: Vec<i32>, prev: Vec<i32>, child: Vec<i32>, parent: Vec<i32> } // contour tree
pub struct Moments { m00:f64, m10:f64, m01:f64, /* … */ hu:[f64;7] }                       // i64/f64 accum (I13)
pub struct BinaryDescriptors { bits: Vec<[u64;4]>, /* 256-bit ORB/BRIEF */ }               // Hamming-matched
pub enum MatchMethod { Ssd, Sad, Ncc, Ccorr }   pub enum DistMetric { L1, L2, Chamfer }
