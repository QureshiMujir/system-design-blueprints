Core schema & types (§11)

enum Method { Get, Post, Put, Delete, Head, Options, Patch, Trace, Connect, Other(String) }
    // own · immut-after-construct · conc: Send+Sync (no shared mutability) · persist: none
    // ser: ASCII tchar per RFC 9110 sec. 5.6.2 · compat: Other(String) absorbs unknown future methods
enum Version { Http10, Http11 }
    // own · immut · conc: Copy · persist: none · ser: "HTTP/1.0"/"HTTP/1.1" literal
    // compat: Phase 8 adds Http2 as a distinct enum (h2_bridge.rs), not a variant here (P11)
enum StatusCode { /* Ok200, NotFound404, ... full 2xx-5xx set the server emits */ }
    // own · immut · conc: Copy · persist: none · ser: 3-digit + reason phrase (RFC 9112 sec. 4)
    // compat: from_u16 total function, rejects out-of-range codes at construction
enum ProtoError { Malformed, Incomplete, InvalidUtf8, UnknownMethod, UnsupportedVersion,
                   InvalidContentLength, BodyTooLarge, ChunkedMalformed(String) }
    // own · immut · conc: Send · persist: none · ser: n/a (internal)
    // compat: exhaustively mapped to StatusCode (I12) — new variants require a new match arm
struct Request { method: Method, path: String, query: Option<String>, version: Version,
                 headers: Vec<(String, String)>, body: bytes::Bytes }
    // own (post-parse, no borrowed lifetimes) · immut after construction · conc: Send+Sync
    // persist: request-scoped only, dies with the connection · ser: parsed from wire, never
    // re-serialised (server does not proxy) · compat: headers lowercased at parse time (P1)
struct Response { status: StatusCode, headers: Vec<(String, String)>, body: bytes::Bytes }
    // own · mut via builder methods (with_header, etc.) until serialise() · conc: Send
    // persist: none · ser: serialise(version: Version) -> Vec<u8>, RFC 9112 sec. 4 wire format
    // compat: version-specific framing (Content-Length vs. chunked) resolved at serialise time
fn parse_request(buf: &[u8]) -> Result<ParseOutcome, ProtoError>;
enum ParseOutcome { Complete(Request, usize /* bytes_consumed */), Incomplete }
    // ParseOutcome: own · conc: Send · persist: none · compat: Incomplete drives the read-more
    // loop in connection.rs — never busy-spins, only re-entered on new socket readiness
const MAX_BODY_SIZE: usize;   // = 1 MiB (§4) — pub const, read by conformance and connection.rs
const MAX_HEADER_BYTES: usize; // = 8 KiB (RFC 9112 sec. 3 recommends a server-defined cap)
enum RouterError { DuplicateRoute { method: Method, pattern: String }, InvalidPattern(String) }
    // own · immut · conc: Send · persist: none · ser: n/a · compat: build-time only (I11)
struct Params(std::collections::HashMap<String, String>);
    // own · mut during trie match, immut once handed to Handler::handle · conc: Send
    // persist: request-scoped · ser: n/a · compat: absent captures simply omitted from the map
enum RouteResult { Match { handler: String, params: Params },
                    MethodNotAllowed { allowed: Vec<Method> }, NotFound }
    // own · immut · conc: Send · persist: none · compat: exhaustive (I3) — no fourth variant
    // without updating dispatcher.rs's match, enforced by the compiler
struct Router { /* trie root, opaque */ }
    // own · immut after Router::build() returns (Arc-shared, ADR-07) · conc: Sync, read-only
    // under load · persist: none, rebuilt at startup · compat: route table is process-lifetime
fn Router::new() -> Router;
fn Router::add_route(&mut self, method: Method, pattern: &str, handler: &str)
    -> Result<(), RouterError>;
fn Router::get(&mut self, pattern: &str, handler: &str) -> Result<(), RouterError>;
fn Router::post(&mut self, pattern: &str, handler: &str) -> Result<(), RouterError>;
fn Router::resolve(&self, method: &Method, path: &str) -> RouteResult;
enum HandlerError { NotFound(String), Panic }
    // own · immut · conc: Send · persist: none · compat: Panic variant only reachable via I8
trait Handler: Send + Sync + 'static { fn handle(&self, req: &Request, params: &Params) -> Response; }
    // implementors: own their own state (if any) · conc: MUST be Send+Sync — shared via Arc
    // across every connection task · persist: handler-defined (static_files.rs reads disk,
    // read-only) · compat: adding a field to a handler struct never breaks the trait contract
fn fn_handler<F>(f: F) -> impl Handler where F: Fn(&Request, &Params) -> Response + Send + Sync + 'static;
struct HandlerRegistry { /* name -> Arc<dyn Handler>, built at startup */ }
    // own · immut after HandlerRegistry::build() (Arc-shared) · conc: Sync, lock-free reads
    // persist: none · compat: register() panics on duplicate name — a startup-time programming
    // error, not a runtime error (deliberate asymmetry with RouterError, which is recoverable)
fn HandlerRegistry::new() -> HandlerRegistry;
fn HandlerRegistry::register(&mut self, name: &str, handler: impl Handler);
fn HandlerRegistry::get(&self, name: &str) -> Option<&(dyn Handler)>;
struct ServerConfig { bind_addr: std::net::SocketAddr, max_requests_per_connection: u32,
                       max_buffer_size: usize, keep_alive_timeout_ms: u64, body_timeout_ms: u64,
                       grace_period_ms: u64, accept_concurrency: usize, tls: Option<TlsConfig> }
    // own · immut after load (config.rs) · conc: Send+Sync, Arc-shared · persist: sourced from
    // env/file at startup, not reloaded (Phase 9 config-reload is deferred, §35)
struct TlsConfig { cert_path: std::path::PathBuf, key_path: std::path::PathBuf,
                    alpn_protocols: Vec<String> }
    // own · immut · conc: Send+Sync · persist: reads cert/key files at startup · compat:
    // rotation requires process restart in Phase 7 (hot-rotation deferred, §35)
enum ConnState { Accepted, AwaitingRequest, Reading, Dispatching, Writing, Closing }
    // own · mut, single-threaded per connection (no cross-task sharing) · conc: !Sync by design
    // — a ConnState instance lives inside exactly one connection task · persist: none
    // compat: state machine table in §8 is the single source of truth for transitions
fn run_listener(listener: tokio::net::TcpListener, router: std::sync::Arc<Router>,
    registry: std::sync::Arc<HandlerRegistry>, cfg: std::sync::Arc<ServerConfig>) -> !;
fn handle_connection(sock: tokio::net::TcpStream, router: std::sync::Arc<Router>,
    registry: std::sync::Arc<HandlerRegistry>, cfg: std::sync::Arc<ServerConfig>);
fn dispatch(req: &Request, router: &Router, registry: &HandlerRegistry) -> Response;
struct ShutdownRegistry { /* tokio::task::JoinSet<()> + tokio_util::sync::CancellationToken */ }
    // own · mut (tasks added/removed) · conc: owned by the main task only, not Arc-shared
    // persist: none · compat: single instance per process, created in main.rs
