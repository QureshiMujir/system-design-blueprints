  reclaim_state: ReclaimState;
  migration_budget: u32;

  physical_addr: PhysAddr;
  owner: OwnerId;
  state: PageState;
  refcount: AtomicU32;
  slab_class: Option<SizeClass>;
  checksum: u32;
  generation: AtomicU32;
Arena:
  arena_id: ArenaId;
  node_id: NodeId;
  free_lists: FreeListHierarchy;
  slab_caches: Vec<SlabCacheRef>;
  remote_free_queue: MpmcQueue<PageHandle>;
  pressure_metrics: PressureSnapshot;

  source_node: NodeId;
  target_node: NodeId;
  page_list: Vec<PageHandle>;
  priority: MigrationPriority;
  enqueue_ts: MonotonicNs;

  sequence: u64;
