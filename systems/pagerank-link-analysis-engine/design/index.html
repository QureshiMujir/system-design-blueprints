Core schema & types (§11)

pub type NodeId = u32;   // dense post-remap id (ADR-17); never mixed with raw labels
pub type Prob   = f64;   // value in [0,1]; invariant, not type-enforced (ADR-05)

/// Rank vector. Newtype guards the sum-to-one invariant (I1) via controlled mutation.
/// Owned by the algorithm; cloned only to double-buffer (P10). Not persisted (recomputed).
#[derive(Debug, Clone)]
pub struct RankVec { pub(crate) inner: Vec<Prob>, n: usize }

impl RankVec {
    pub fn uniform(n: usize) -> Self;                 // 1/n each
    pub fn from_seed_set(n: usize, seeds: &[NodeId]) -> Self;  // 1/|S| on seeds (PPR)
    pub fn l1_normalize(&mut self);                   // enforce I1
    pub fn l1_delta(&self, other: &RankVec) -> f64;   // Σ|a−b| (convergence, P7)
    pub fn l2_delta(&self, other: &RankVec) -> f64;
    pub fn get(&self, node: NodeId) -> Prob;
    pub fn as_slice(&self) -> &[Prob];                // hot-loop read path (pub(crate) inner)
}
#[derive(Debug, Clone, Copy)]
pub struct RawEdge { pub src: NodeId, pub dst: NodeId }   // post-remap

/// Output of graph-io; the only contract graph-repr depends on (P11).
pub struct RawEdgeList {
    pub edges: Vec<RawEdge>,
    pub num_nodes: usize,              // after remap = #interned labels
    pub dangling_nodes: Vec<NodeId>,   // sorted asc (deterministic Roaring build)
    pub node_labels: Option<Vec<String>>,  // present iff remapped from strings
}
/// Out-link adjacency (row i = out-neighbors of i). Used by HITS hub pass.
/// Immutable after build; persisted via binary_format (I12). Host-endian (ADR-08).
#[derive(Debug)]
pub struct CsrGraph {
    pub row_ptr: Vec<u32>,        // len n+1, monotonic (I6 analog)
    pub col_idx: Vec<NodeId>,     // len m; entries < n (I10)
    pub out_deg_recip: Vec<f64>,  // len n; 1/outdeg or 0 for dangling, finite (I4)
    pub n: usize, pub m: usize,
}

/// In-link adjacency (column j = in-neighbors of j). The PageRank kernel's structure (P3).
#[derive(Debug)]
pub struct CscGraph {
    pub col_ptr: Vec<u32>,            // len n+1; col_ptr[n]=m (I6)
    pub row_idx: Vec<NodeId>,         // len m; entries < n (I10)
    pub src_out_deg_recip: Vec<f64>,  // len m; co-located recip of each in-neighbor (P3)
    pub n: usize, pub m: usize,
    pub dangling: DanglingBitVec,
}

impl CscGraph {
    /// Pull SpMV: out[j] = Σ_{i→j} rank[i]·src_out_deg_recip. Distinct buffers (I8).
    pub fn spmv_col(&self, rank: &[f64], out: &mut [f64]);   // signature only (§4)
    pub fn from_raw(raw: &RawEdgeList, csr: &CsrGraph) -> Self;
}

/// Roaring bitmap of zero-out-degree nodes (ADR-07).
pub struct DanglingBitVec { inner: roaring::RoaringBitmap }
impl DanglingBitVec {
    pub fn is_dangling(&self, node: NodeId) -> bool;       // O(1)
    pub fn accumulate_mass(&self, rank: &[f64]) -> f64;    // dmass = Σ_{i∈D} rank[i] (P2,I2)
}
/// Every variant implements this (P8). Object-safe via associated types kept concrete per impl.
pub trait RankAlgorithm {
    type Output; type Config; type Error: std::error::Error;
    fn new(graph: &CscGraph, config: Self::Config) -> Self;
    fn run(&mut self) -> Result<Self::Output, Self::Error>;
    fn iteration_count(&self) -> usize;
    fn convergence_history(&self) -> &[f64];   // L1 delta per iteration (I9 source)
}
#[derive(Debug, Clone)]
pub struct PageRankConfig {
    pub damping: f64,            // (0,1) exclusive (I7); default 0.85
    pub epsilon: f64,            // default 1e-6 (P7)
    pub max_iter: usize,         // default 200 (P4,I15)
    pub norm: ConvergenceNorm,
    pub teleport: TeleportMode,
    pub initial_rank: InitialRank,
}
#[derive(Debug, Clone)] pub enum ConvergenceNorm { L1, L2 }
#[derive(Debug, Clone)] pub enum TeleportMode { Uniform, Custom(Vec<f64>) }  // Custom sums to 1 (I14)
#[derive(Debug, Clone)] pub enum InitialRank  { Uniform, Custom(RankVec) }

#[derive(Debug)]
pub struct PageRankOutput { pub ranks: RankVec, pub iterations: usize,
                            pub converged: bool, pub final_delta: f64 }

#[derive(Debug, Clone)]
pub struct HitsConfig { pub epsilon: f64, pub max_iter: usize, pub norm: ConvergenceNorm }
#[derive(Debug)]
pub struct HitsOutput { pub hub_scores: Vec<f64>, pub authority_scores: Vec<f64>,
                        pub iterations: usize, pub hub_delta: f64, pub auth_delta: f64 }
pub struct HitsAlgorithm {           // needs both directions (Phase 4)
    csr: CsrGraph, csc: CscGraph, hub: Vec<f64>, auth: Vec<f64>,
    config: HitsConfig, history: Vec<(f64, f64)>,   // (hub_delta, auth_delta)/iter
}

pub struct SeedSet { nodes: Vec<NodeId>, weights: Vec<f64> }  // weights sum to 1 (I14)
#[derive(Debug, Clone, PartialEq)]
pub enum ConvergenceState {
    NotStarted,
    Running   { iteration: usize, delta: f64 },
    Converged { iteration: usize, final_delta: f64 },
    TimedOut  { max_iter: usize, last_delta: f64 },   // exit 2 (I15,§19)
}

/// Emitted per iteration to stderr (serde JSON) — the observability invariant I9.
#[derive(serde::Serialize)]
pub struct IterationReport { pub iteration: usize, pub l1_delta: f64, pub l2_delta: f64,
                             pub dangling_mass: f64, pub elapsed_ms: u64 }

pub struct NodeRemapper { label_to_id: HashMap<String, NodeId>, id_to_label: Vec<String> }
impl NodeRemapper {
    pub fn intern(&mut self, label: &str) -> NodeId;   // NFC-normalize, insert-or-get
    pub fn label(&self, id: NodeId) -> &str;
    pub fn num_nodes(&self) -> usize;
}

#[derive(Debug, thiserror::Error)]
pub enum GraphIoError {
    #[error("I/O error: {0}")] Io(#[from] std::io::Error),
    #[error("Malformed edge at line {line}: {detail}")] MalformedEdge { line: usize, detail: String },
    #[error("WebGraph BV decode error: {0}")] WebGraphDecode(String),
    #[error("Node ID overflow: > u32::MAX nodes")] NodeIdOverflow,   // ADR-17 boundary
    #[error("Binary format invalid: {0}")] BadBinary(String),        // I12
}
#[derive(Debug, thiserror::Error)]
pub enum RankError {
    #[error("did not converge in {0} iterations")] NotConverged(usize),  // exit 2
    #[error("out of memory: needed {need} bytes, had {have}")] OutOfMemory { need: usize, have: usize },
    #[error("invalid graph: {0}")] InvalidGraph(String),
}
#[derive(Debug, thiserror::Error)]
pub enum PprError { #[error("seed not found: {0}")] SeedNotFound(String),
                    #[error("empty seed set")] EmptySeedSet }
