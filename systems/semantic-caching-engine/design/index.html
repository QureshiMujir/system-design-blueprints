Core schema & types (§11)

// sc-types — embeddings, keys, entries, wire
pub struct Embedding { vec: Box<[f32]>, dim: u16, model_id: ModelId, embed_ver: EmbedVer } // owned; normalized (I3); bincode in store
pub struct ModelId(pub String);                          // embedding model identity
pub struct EmbedVer(pub u32);                            // embedding-model version — partitions keyspace (I1)
pub struct ModelVersion { name: String, ver: SemVer }    // *generation* model (e.g. gpt-4o@2024-08)
pub struct Namespace(pub String);                        // logical partition / tenant
pub struct CacheKey(pub u128);                           // hash(ns ‖ model_ver ‖ canonical(params)) — exact sub-key
pub struct EntryId(pub u64);                             // dense id; index ↔ store join key (I7)
pub struct EmbeddingId(pub u64);                         // ref from CacheEntry → stored embedding

pub struct CacheEntry {                                  // SOURCE OF TRUTH (principle 2); WAL-durable; bincode
  id: EntryId, key: CacheKey, prompt_hash: u64,
  embed_ref: EmbeddingId, model_ver: ModelVersion,
  response: Bytes, created_ts: Ts, last_access_ts: Ts,
  hits: u64, tokens_saved: u64, cost_saved: f64,
  ttl: Option<Duration>, state: EntryState,
}
pub enum EntryState { Live, Tombstoned, Expired }        // I7/I10
pub struct Ts(pub u64);                                  // monotonic ms (I13)

pub struct LookupRequest { ns: Namespace, model_ver: ModelVersion, prompt: String,
                           params: Params, opts: LookupOpts }   // serde JSON wire
pub struct StoreRequest  { ns: Namespace, model_ver: ModelVersion, prompt: String,
                           response: Bytes, meta: RespMeta }
pub struct LookupOpts { threshold: Option<f32>, verify: Option<bool>, max_candidates: Option<u16>, ttl: Option<Duration> }
pub struct RespMeta { tokens_in: u32, tokens_out: u32, cost: f64, latency_ms: u32 }
pub struct Params { /* canonicalized kv */ }             // hashed into CacheKey
// sc-embed — provider abstraction
pub trait EmbeddingProvider: Send + Sync {
  fn model_id(&self) -> ModelId;
  fn embed_ver(&self) -> EmbedVer;                       // stamps keyspace partition (I1)
  fn dim(&self) -> u16;
  fn embed(&self, text: &str) -> Result<Embedding>;      // MUST L2-normalize (I3)
  fn embed_batch(&self, texts: &[&str]) -> Result<Vec<Embedding>>;
}

// sc-index — ANN
pub trait VectorIndex: Send + Sync {                     // reads concurrent (I8); single writer
  fn insert(&self, id: EntryId, e: &Embedding) -> Result<()>;     // idempotent on id
  fn search(&self, q: &Embedding, k: usize, ef: usize) -> Result<Vec<(EntryId, f32)>>; // desc sim
  fn tombstone(&self, id: EntryId) -> Result<()>;        // soft delete (I7)
  fn compact(&self) -> Result<CompactStats>;             // reclaim tombstones (§9)
  fn rebuild(&self, entries: impl Iterator<Item=(EntryId, Embedding)>) -> Result<()>; // from store (I6)
  fn snapshot(&self) -> IndexSnapshot;                    // consistent read view (I8)
  fn stats(&self) -> IndexStats;                          // size, tombstone_ratio, recall_estimate
}
pub struct HnswParams { m: usize, ef_construction: usize, ef_search: usize, max_layer: u8 } // ADR-002
pub struct IvfPqParams { nlist: usize, m_subq: usize, nbits: u8, nprobe: usize }            // ADR-003
pub struct IndexStats { nodes: u64, tombstones: u64, tombstone_ratio: f64, bytes: u64, embed_ver: EmbedVer }
// sc-store — value store + WAL
pub trait ValueStore: Send + Sync {                      // source of truth (principle 2)
  fn put(&self, e: CacheEntry) -> Result<()>;            // exactly-once / idempotent (I5); WAL-first (I9)
  fn get(&self, id: EntryId) -> Result<Option<CacheEntry>>;
  fn touch(&self, id: EntryId, ts: Ts) -> Result<()>;    // update recency/hits
  fn invalidate(&self, sel: InvalSelector) -> Result<u64>; // by id/ns/model_ver (principle 10)
  fn evict(&self, victim: EntryId) -> Result<()>;        // policy-driven (§6)
  fn checkpoint(&self) -> Result<CheckpointRef>;
}
pub enum InvalSelector { Id(EntryId), Namespace(Namespace), ModelVersion(ModelVersion), Expired }
pub struct WalRecord { seq: u64, op: WalOp, crc32c: u32 } // append-only (I9/I11)
pub enum WalOp { Put(EntryId), Invalidate(InvalSelector), Evict(EntryId), Checkpoint(CheckpointRef) }
pub struct Segment { path: PathBuf, bytes: u64, crc32c: u32 } // round-trip exact (I11)
pub struct CheckpointRef { id: u64, store_snapshot: PathBuf, index_manifest: PathBuf }
// sc-policy — gate + decision
pub struct Threshold { ns: Namespace, embed_ver: EmbedVer, tau: f32, margin: f32, calib_hash: u64 } // sealed (I4)
pub enum Decision { Hit(EntryId, f32), Miss(MissReason), RejectedHit(RejectReason) } // pure fn (I2)
pub enum MissReason { NoCandidate, BelowThreshold, Marginal, Stale, VerifyFailed, Bypass }
pub enum RejectReason { Stale, Tombstoned, Verify, VersionMismatch }
pub struct GateConfig { threshold: Threshold, verify: VerifyMode, max_candidates: u16, widen_ef_on_marginal: bool }
pub enum VerifyMode { Off, ExactRescore, CrossEncoder, Faithfulness }
pub struct ProbeSampler { seed: u64, rate: f32 }         // seeded + bounded (I14)

// sc-admit
pub enum AdmitDecision { Cache(CacheEntry), Bypass(BypassReason) } // I15
pub enum BypassReason { NonDeterministic, LowValue, Shedding, NotCacheable }
pub struct CostModel { token_price: f64, reuse_estimate: f64, storage_cost: f64, false_hit_penalty: f64 }

// sc-observe
pub struct Hll { m: usize, registers: Vec<u8> }          // distinct keys/ns; rel err 1.04/√m
pub struct CountMin { d: usize, w: usize, table: Vec<u32> } // prompt frequency (LFU/hot)
pub struct Reservoir<T> { k: usize, seed: u64, items: Vec<T> } // false-hit probe sample (I14)
pub struct TDigest { /* centroids */ }                   // latency/similarity quantiles
pub struct DriftDetector { window: usize, baseline: f64 } // threshold/false-hit drift

// sc-server
pub struct LookupResponse { decision: DecisionKind, response: Option<Bytes>, similarity: Option<f32>,
                            saved: SavedMetrics, source: ResultSource }
pub enum ResultSource { Cache, Upstream, Bypass }        // I12
pub struct SavedMetrics { tokens_saved: u64, cost_saved: f64, latency_saved_ms: u64 }

// errors (thiserror; no panics on data path — ADR-030)
pub enum ScError { Embed(EmbedError), Index(IndexError), Store(StoreError),
                   Policy(PolicyError), Admit(AdmitError), Wire(WireError), Wal(WalError) }
pub enum IndexError { Unnormalized, DimMismatch, VersionMismatch, NodeCorrupt, Full }
pub enum PolicyError { VersionMismatch, ThresholdRange, NoThresholdForNs }
pub enum WalError { Crc, Truncated, ReplayConflict }
