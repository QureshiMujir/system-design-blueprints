Core schema & types (§11)

// is-core
pub struct UnitName(String);                 // "foo.service" — kind-qualified
pub enum UnitKind {
    Service, Socket, Timer, Path, Target, Slice, Scope, Device, Mount,
}
pub struct Unit {
    pub name: UnitName,
    pub kind: UnitKind,
    pub state: UnitState,                    // §8.3 FSM; owned, mutated only by is-txn
    pub config: UnitConfig,                   // immutable post-parse; is-parser output
    pub deps: Dependencies,                   // owned by is-graph; rebuilt, not patched
    pub cgroup: Option<CgroupPath>,           // Some() once is-cgroup has entered it
    pub pid: Option<Pid>,                     // Some() only while Activating|Active
}
pub enum UnitState {
    Inactive, Activating, Active, Deactivating, Failed, Maintenance,
}
pub struct Dependencies {                     // BTreeSet: P13 determinism
    pub wants: BTreeSet<UnitName>,
    pub requires: BTreeSet<UnitName>,
    pub after: BTreeSet<UnitName>,
    pub before: BTreeSet<UnitName>,
    pub conflicts: BTreeSet<UnitName>,
    pub binds_to: BTreeSet<UnitName>,
}
pub enum RestartPolicy {
    No, Always, OnSuccess, OnFailure, OnAbnormal, OnAbort, OnWatchdog,
}
pub enum InitError {                          // taxonomy referenced by every crate
    ParseError { key: String, value: String, unit: UnitName },
    GraphCycle { members: Vec<UnitName> },
    MissingRequire { unit: UnitName, target: UnitName },
    ActivationTimeout { unit: UnitName, elapsed_ms: u64 },
    CgroupError { unit: UnitName, errno: i32 },
}
// is-parser
pub struct UnitConfig {                       // serialization: parsed, never re-serialized to disk
    pub description: Option<String>,
    pub exec_start: Option<Vec<String>>,
    pub restart: RestartPolicy,
    pub restart_sec: Duration,
    pub start_limit_burst: u32,
    pub start_limit_interval: Duration,
    pub memory_max: Option<u64>,
    pub cpu_weight: Option<u32>,
    pub wanted_by: BTreeSet<UnitName>,
}
pub fn parse_unit(path: &Path) -> Result<UnitConfig, InitError>;   // pure, no I/O side effects beyond read

// is-graph
pub struct Graph { /* adjacency, ownership: is-graph exclusively */ }
pub fn build(units: &[Unit]) -> Result<Graph, InitError>;
pub fn detect_cycles(g: &Graph) -> Vec<Vec<UnitName>>;             // Tarjan SCC, I1

// is-txn
pub struct Transaction {                      // ownership: is-txn; concurrency: single-writer
    pub target: UnitName,
    pub batches: Vec<Vec<UnitName>>,          // topological batches, I6
}
pub fn plan(g: &Graph, target: &UnitName) -> Result<Transaction, InitError>;
pub fn commit(t: Transaction) -> ActivationReport;
// is-supervisor
pub struct SupervisorRecord {                 // persistence: in-memory only, rebuilt on restart
    pub unit: UnitName,
    pub pid: Pid,
    pub restart_count: u32,
    pub window_start: Instant,                // CLOCK_MONOTONIC, ADR-19
}
pub fn spawn(cfg: &UnitConfig, cgroup: &CgroupPath) -> Result<Pid, InitError>;

// is-cgroup
pub struct CgroupPath(PathBuf);
pub struct Limits { pub memory_max: Option<u64>, pub cpu_weight: Option<u32>, pub io_weight: Option<u32> }
pub fn create_and_enter(path: &CgroupPath, limits: &Limits) -> Result<(), InitError>;  // I12 ordering

// is-socket
pub struct SocketUnit { pub listen: SocketAddrSpec, pub accept: bool }
pub fn activate_on_connection(s: &SocketUnit) -> Result<OwnedFd, InitError>;

// is-journal
pub struct JournalEntry {                     // wire/storage record layout
    pub unit: UnitName,
    pub ts_realtime: SystemTime,               // ADR-19
    pub priority: u8,
    pub fields: BTreeMap<String, String>,
}
pub fn append(e: JournalEntry) -> Result<(), InitError>;           // I9 durability contract
