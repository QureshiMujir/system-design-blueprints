#[derive(Copy, Clone, Debug, PartialEq, Eq, Hash, PartialOrd, Ord)]
pub struct TaskId(pub u64);
#[derive(Copy, Clone, Debug, PartialEq, Eq)]
#[repr(u8)]
pub enum TaskState {
    Pending   = 0,  // allocated, not yet enqueued
    Ready     = 1,  // in a deque or injector; not running
    Running   = 2,  // executing on exactly one worker
    Done      = 3,  // FnOnce completed normally
    Panicked  = 4,  // FnOnce panicked; payload held by scope
}
Pending → Ready     : enqueue to deque / injector
Ready   → Running   : worker pops or steals (ownership transfer; I1)
Running → Done      : FnOnce returns normally
Running → Panicked  : FnOnce unwinds; catch_unwind in worker loop (I7)
pub struct Task {
    pub id:    TaskId,
    pub state: AtomicTaskState,
    run:       UnsafeCell<Option<Box<dyn FnOnce() + Send + 'static>>>,
}
pub struct AtomicTaskState(AtomicU8);
impl AtomicTaskState {
    pub fn load(&self, ord: Ordering) -> TaskState;
    pub fn transition(&self, from: TaskState, to: TaskState) -> Result<(), TaskState>;
}
pub(crate) struct RawJob {
    fn_ptr: unsafe fn(*mut ()),

pub struct ChaseLevDeque<T> {
    bottom: AtomicIsize,
    top:    AtomicIsize,
    buffer: Atomic<Arc<CircularBuffer<T>>>,
}
pub enum Steal<T> {
    Success(T),  // item obtained
    Empty,       // deque was empty
    Retry,       // CAS contention; caller should back-off and retry
}
pub struct Stealer<T> {
    deque: Arc<ChaseLevDeque<T>>,
}
impl<T: Send> Clone for Stealer<T> { ... }
impl<T: Send> Stealer<T> {
    pub fn steal(&self) -> Steal<T>;
}
pub(crate) struct CircularBuffer<T> {
    capacity: usize,      // power of 2
    mask:     usize,      // capacity - 1
    slots:    Box<[UnsafeCell<MaybeUninit<T>>]>,
}
impl<T> CircularBuffer<T> {
    pub fn new(cap: usize) -> Self;           // cap must be power of 2
    pub unsafe fn read(&self, idx: isize) -> T;
    pub unsafe fn write(&self, idx: isize, val: T);
    pub fn grow(&self, bottom: isize, top: isize) -> Arc<Self>;
}
pub struct Worker {
    pub id:       usize,
    pub local:    ChaseLevDeque<Task>,
    pub stealers: Box<[Stealer<Task>]>,
    pub rng:      XorShift,
    pub numa:     NumaNode,
    pub stats:    Arc<SchedMetrics>,
}
impl Worker {
    pub fn run_local(&mut self) -> bool;           // pop + execute one task; return true if ran
    pub fn steal_from_victim(&mut self) -> bool;   // attempt steal from RNG-selected victim
    pub fn park_until_notified(&mut self);         // exponential backoff → OS park
}
pub struct WorkerHandle {
    unpark: Arc<AtomicBool>,
    thread: Thread,                               // std::thread::Thread for unpark()
}
impl WorkerHandle {
    pub fn notify(&self);                          // set flag + unpark thread
}
pub struct XorShift { state: u64 }
impl XorShift {
    pub fn new(seed: u64) -> Self;
    pub fn next_usize(&mut self, bound: usize) -> usize;  // uniform in [0, bound)
}
pub struct PoolConfig {
    pub num_threads:   usize,          // default: logical_cpus()
    pub stack_size:    usize,          // default: 2 MiB
    pub affinity:      Option<CpuSet>, // None = OS default
    pub policy:        SchedPolicy,    // default: WorkFirst
    pub max_in_flight: usize,          // backpressure cap (I6, P10)
}
impl Default for PoolConfig { ... }
pub struct CpuSet {
    bits: Vec<u64>,   // one bit per logical CPU; variable length
}
impl CpuSet {
    pub fn from_cpus(cpus: &[usize]) -> Self;
    pub fn all() -> Self;
    pub fn numa_node(node: NumaNode) -> Self;
    pub fn apply_to_current_thread(&self) -> Result<(), AffinityError>;
}
pub struct ThreadPool {
    workers:  Vec<WorkerHandle>,
    injector: Arc<Injector<Task>>,
    topology: Topology,
    config:   PoolConfig,
    state:    Arc<PoolState>,
}
impl ThreadPool {
    pub fn new(config: PoolConfig) -> Self;
    pub fn submit(&self, task: Task) -> Result<(), SubmitError>;   // backpressure: Err if saturated
    pub fn scope<'s, F: FnOnce(Scope<'s>)>(&'s self, f: F);        // blocks until scope done
    pub fn shutdown(self);                                           // drain + join all threads (I10)
}
pub struct Injector<T> {
    queue: crossbeam_deque::Injector<T>,   // or equivalent MPMC impl
}
impl<T: Send> Injector<T> {
    pub fn push(&self, val: T);
    pub fn steal(&self) -> Steal<T>;
    pub fn is_empty(&self) -> bool;
}
#[derive(Copy, Clone, Debug, PartialEq, Eq, Hash)]
pub struct NodeId(pub u32);
pub struct GraphNode {
    pub task:         Task,
    pub pending_deps: AtomicUsize,          // count of unsatisfied in-edges
    pub successors:   Mutex<Vec<NodeId>>,   // edges to decrement on completion
}
pub struct Dependency {
    pub pred: NodeId,
    pub succ: NodeId,
}
pub struct TaskGraph {
    nodes:    Vec<GraphNode>,             // indexed by NodeId
    pool:     Arc<ThreadPool>,
    complete: Latch,                      // signaled when all nodes Done/Panicked
}
impl TaskGraph {
    pub fn add_node(&mut self, task: Task) -> NodeId;
    pub fn add_edge(&mut self, pred: NodeId, succ: NodeId) -> Result<(), CycleError>;
    pub fn submit(self);                  // validate (DFS cycle check), push Ready nodes, wait
    pub fn wait(self);                    // block until complete Latch fires
}
pub struct Scope<'s> {
    pool:    &'s ThreadPool,
    latch:   Arc<Latch>,
    panics:  Arc<Mutex<Vec<Box<dyn Any + Send>>>>,
    _marker: PhantomData<&'s mut &'s ()>,
}
impl<'s> Scope<'s> {
    pub fn spawn<F: FnOnce() + Send + 's>(&self, f: F);
}

pub fn join<A, B, RA, RB>(a: A, b: B) -> (RA, RB)
where
    A: FnOnce() -> RA + Send,

    RB: Send;
pub trait Producer: Send + Sized {
    type Item: Send;
    fn split_at(self, index: usize) -> (Self, Self);
    fn len(&self) -> usize;
    fn into_iter(self) -> impl Iterator<Item = Self::Item>;
}
pub trait ParallelIterator: Sized {
    type Item: Send;
    fn for_each<F: Fn(Self::Item) + Send + Sync>(self, f: F);
    fn map<B: Send, F: Fn(Self::Item) -> B + Send + Sync>(self, f: F) -> impl ParallelIterator<Item = B>;
    fn reduce<F: Fn(Self::Item, Self::Item) -> Self::Item + Send + Sync>(self, identity: Self::Item, f: F) -> Self::Item;
    fn scan<B: Send, F: Fn(&B, Self::Item) -> B + Send + Sync>(self, init: B, f: F) -> Vec<B>;
}
#[derive(Copy, Clone, Debug, PartialEq, Eq)]
pub enum SchedPolicy {
    WorkFirst,      // push continuation to deque; run spawned child immediately
    HelpFirst,      // run continuation inline; push child for steal
    LocalityAware,  // NUMA-biased victim ordering; blends work-first default
}
pub struct Promise<T: Send> {
    inner: Arc<SharedSlot<T>>,
}
pub struct Future<T: Send> {
    inner: Arc<SharedSlot<T>>,
}
impl<T: Send> Promise<T> {
    pub fn fulfill(self, val: T);
    pub fn fulfill_panic(self, payload: Box<dyn Any + Send>);
}
impl<T: Send> Future<T> {
    pub fn wait(self) -> T;   // participates in scheduling; does not park thread (P7)
    pub fn is_ready(&self) -> bool;
}
pub fn future_promise<T: Send>() -> (Future<T>, Promise<T>);
pub struct Latch {
    count:  AtomicUsize,
    parked: Mutex<Vec<Thread>>,
}
impl Latch {
    pub fn new(n: usize) -> Self;
    pub fn count_down(&self);     // decrement; unpark waiters when 0
    pub fn wait(&self);           // park until count == 0
    pub fn is_zero(&self) -> bool;
}
pub struct Barrier {
    n:       usize,
    arrived: AtomicUsize,
    latch:   Latch,
}
impl Barrier {
    pub fn new(n: usize) -> Self;
    pub fn wait(&self);   // last arrival resets + releases all
}
pub struct WaitGroup {
    count: AtomicIsize,
    latch: Arc<Latch>,
}
impl WaitGroup {
    pub fn new() -> Self;
    pub fn add(&self, delta: isize);
    pub fn done(&self);      // add(-1)
    pub fn wait(&self);      // block until count == 0
}
#[derive(Copy, Clone, Debug, PartialEq, Eq, Hash, PartialOrd, Ord)]
pub struct NumaNode(pub u8);
pub struct NumaDomain {
    pub id:       NumaNode,
    pub cpus:     CpuSet,
    pub mem_size: u64,          // bytes of memory on this domain
}
pub struct Topology {
    pub domains:  Vec<NumaDomain>,
    pub cpu_to_node: Vec<NumaNode>,    // indexed by logical CPU id
}
impl Topology {
    pub fn detect() -> Self;           // reads /sys or libnuma; fallback: single domain
    pub fn node_for_cpu(&self, cpu: usize) -> NumaNode;
    pub fn cpus_for_node(&self, node: NumaNode) -> &CpuSet;
}
pub struct StealStats {
    pub steals_attempted: AtomicU64,
    pub steals_won:       AtomicU64,
    pub tasks_run:        AtomicU64,
    pub park_count:       AtomicU64,
    pub idle_ns:          AtomicU64,
}
pub struct SchedMetrics {
    pub per_worker:    Box<[StealStats]>,       // indexed by worker id
    pub global_tasks:  AtomicU64,
    pub latency_hist:  LatencyHistogram,        // fixed-bucket ns histogram
    pub queue_depth:   Box<[AtomicUsize]>,      // per-worker deque depth snapshot
}
impl SchedMetrics {
    pub fn snapshot(&self) -> MetricsSnapshot;  // lock-free read; point-in-time copy
    pub fn reset(&self);
}
pub struct LatencyHistogram {
    buckets: [AtomicU64; 32],   // log2 ns buckets: 0..=1ns, 1..=2ns, 2..=4ns, ...
}
impl LatencyHistogram {
    pub fn record(&self, ns: u64);
    pub fn percentile(&self, p: f64) -> u64;   // approximate; linear interp within bucket
}
