Core schema & types (§11)

struct TaskId(u64)
struct CpuId(usize)
struct CpuMask
    bits: u64                     // one bit per core, cores 0..63
enum Criticality
    Low
    Medium
    High
    Safety
struct TaskSpec
    id: TaskId
    name: String
    period: Duration
    relative_deadline: Duration
    wcet: Duration
    offset: Duration
    affinity: CpuMask
    criticality: Criticality
struct TaskRuntimeState
    current_state: TaskState
    activations: u64
    preemptions: u64
    missed_deadlines: u64
    total_runtime: Duration
    last_start: Option<Instant>
    next_release: Instant
    absolute_deadline: Instant
enum TaskState
    Created
    Ready
    Running
    Blocked
    Preempted
    Sleeping
    Completed
    MissedDeadline
enum SchedulerPolicy
    RateMonotonic
    EarliestDeadlineFirst
trait Scheduler
    fn admit(task: TaskSpec) -> Result<(), AdmissionError>
    fn enqueue(task: TaskId)
    fn dequeue(task: TaskId)
    fn select_next(cpu: CpuId) -> Option<TaskId>
    fn on_tick(now: Instant)
enum AdmissionError
    UtilizationExceeded(f64)
    ResponseTimeExceedsDeadline(TaskId)
    BlockingBoundExceeded(TaskId, Duration)
    InvalidTaskSpec(TaskId)
struct TraceEvent
    timestamp: Instant
    cpu: CpuId
    task: Option<TaskId>
    kind: TraceEventKind
enum TraceEventKind
    TaskReleased
    TaskStarted
    TaskCompleted
    TaskPreempted
    DeadlineMissed
    LockAcquired
    LockReleased
    AdmissionRejected
    Migrated
struct TimerHandle
    id: u64
    expiry: Instant
    cpu_affinity: CpuId
struct WorkloadSpec
    tasks: Vec<TaskSpec>
    policy: SchedulerPolicy
    schema_version: u16
