Core schema & types (§11)

// ann-core — identifiers and shared primitives. Ownership: value types, Copy where possible.
struct VectorId(u64);                          // stable, immutable, host-assigned or index-assigned
struct Dim(u32);                                // fixed per index at creation, never mutated (I4)
enum Metric { L2, InnerProduct, Cosine }        // immutable per index (I11), Copy, serializable
struct Vector { dim: Dim, values: Arc<[f32]> }  // immutable once constructed; Send + Sync; zero-copy clone

// ann-simd — distance kernel contract. Read-only, no persistence, no mutation.
trait DistanceKernel {
    fn distance(&self, a: &[f32], b: &[f32]) -> f32;      // I6: ordering must match Metric semantics
    fn batch_distance(&self, q: &[f32], batch: &[&[f32]], out: &mut [f32]);
}
// ann-flat — the recall oracle (I1). Concurrency: RwLock<Vec<Vector>>, single-writer via ann-mutate.
struct FlatIndex { dim: Dim, metric: Metric, vectors: RwLock<Vec<(VectorId, Vector)>> }

// ann-hnsw — graph types. Mutation: single-writer serialized; per-node RwLock for reads (ADR-17).
struct EntryPoint { node: VectorId, layer: u8 }             // mutated on insert if new max layer
struct Layer(u8);                                            // 0 is base layer, contains every node
struct NeighborList { ids: Vec<VectorId> }                  // sorted by distance, bounded by M (I9)
struct HnswNode { id: VectorId, level: u8, neighbors: Vec<NeighborList> } // one NeighborList per layer ≤ level
struct M(u16);                                               // graph degree bound, range per §4
struct EfSearch(u32);                                        // query-time beam width, range per §4
struct HnswIndex { entry_point: RwLock<EntryPoint>, nodes: RwLock<HashMap<VectorId, HnswNode>> }
// ann-quantize — k-means and PQ. Codebooks immutable after training (P6), versioned (I5, I11).
struct Centroid { values: Arc<[f32]> }                       // immutable once training completes
struct KMeans { k: u32, max_iters: u32, seed: u64 }          // deterministic given seed (ADR-28)
struct Subspace { offset: u32, len: u32 }                    // one of m partitions of Dim
struct Codebook { subspace: Subspace, centroids: Vec<Centroid>, version: u32 } // I5, I11
struct ProductQuantizer { subspaces: Vec<Subspace>, codebooks: Vec<Codebook>, version: u32 }
struct PqCode { codes: Box<[u8]> }                            // one byte per subspace at 8-bit (ADR-09)
struct DistanceTable { per_subspace: Vec<Vec<f32>> }         // built per query, read-only during scan
struct ScalarQuantizer { bits: u8, ranges: Vec<(f32, f32)>, version: u32 }
// ann-ivf — coarse quantizer + lists. Per-cell RwLock (ADR-17); list membership enforced (I10).
struct CoarseQuantizer { centroids: Vec<Centroid>, version: u32 }
struct NProbe(u32);                                           // query-time cells scanned, range per §4
struct InvertedList { cell_id: u32, codes: RwLock<Vec<(VectorId, PqCode)>> }
struct IvfIndex { quantizer: CoarseQuantizer, lists: Vec<InvertedList> }
// ann-filter / ann-mutate / search plumbing. Filters read-only; tombstones mutate under single writer.
struct Predicate { field: String, op: PredicateOp, value: FilterValue }  // host-supplied, opaque payload
enum PredicateOp { Eq, Range, In }
struct Filter { predicates: Vec<Predicate>, estimated_selectivity: f32 } // drives P12 decision
struct Tombstone { id: VectorId, committed_seq: u64 }         // I3, I13 — ordered by op-log sequence
struct SearchParams { k: u32, ef_search: Option<EfSearch>, nprobe: Option<NProbe>, filter: Option<Filter> }
struct Neighbor { id: VectorId, distance: f32 }
struct SearchResult { neighbors: Vec<Neighbor> }              // I12: len ≤ k, no duplicate ids
// ann-persist / ann-eval — versioned on-disk format (ADR-19, I7) and evaluation types.
struct PersistedIndex { format_version: u32, manifest: SegmentManifest, checksum: u128 }
struct MmapRegion { path: PathBuf, len: usize }               // read-only mapping; madvise policy in Phase 10
struct RecallReport { dataset: String, k: u32, recall_at_k: f32, p50_ms: f32, p95_ms: f32, p99_ms: f32, mem_bytes: u64 }

// error taxonomy — ann-core. Every fallible boundary returns one of these, never a bare panic.
enum AnnError {
    DimensionMismatch { expected: Dim, got: Dim },            // I4
    EmptyIndex,
    CodebookVersionMismatch { expected: u32, got: u32 },       // I5, I11
    SegmentFormatVersionMismatch { expected: u32, got: u32 },  // I7, ADR-19
    FilterMatchedNothing,
    KMeansFailedToConverge { iters: u32, residual_inertia: f32 },
}
