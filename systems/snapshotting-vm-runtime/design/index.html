Core schema & types (§11)

// ---- VM value & heap model (svm-types / svm-heap) ----
pub enum Value { Int(i64), Float(f64), Bool(bool), Nil, Ref(ObjRef) }   // Copy; Ref is an id, not a ptr (I4)
pub struct ObjRef(u32);                                                 // index/offset into the heap (position-independent)
#[repr(C)] pub struct ObjHeader { pub tag: u16, pub flags: u8, pub gc_mark: u8, pub size: u32, pub fwd: u32 }
pub enum Obj { Str(Box<[u8]>), Array(Vec<Value>), Map(Vec<(Value,Value)>),
               Closure{ func: FnId, upvalues: Vec<Value> }, Struct{ tag:u16, fields: Vec<Value> },
               ResourceRef(ResourceHandle) }                            // the only external bridge (I7)

pub struct Heap { /* regions: Vec<Region>, alloc cursors, free lists */ }
impl Heap {
    pub fn alloc(&mut self, obj: Obj) -> ObjRef;                        // GC may run (svm-heap)
    pub fn get(&self, r: ObjRef) -> &Obj;
    pub fn gc(&mut self, roots: &RootSet);                             // mark-sweep (#32/#244)
    pub fn roots_iter(&self) -> RootSet;                              // shared with snapshot (principle 11)
}

// ---- interpreter (svm-vm) ----
pub struct Frame { pub func: FnId, pub pc: u32, pub locals: Vec<Value>, pub ret_pc: u32, pub base_sp: u32 }
pub struct Thread { pub id: u32, pub stack: Vec<Value>, pub frames: Vec<Frame>, pub status: ThreadStatus }
pub struct Vm { /* heap, threads, globals, scheduler, logical_clock:u64, rng:RngState, resources */ }
impl Vm {
    pub fn load_module(b: &Bytecode) -> Vm;                            // the only from-source path
    pub fn run(&mut self) -> Result<(), VmError>;                      // dispatch loop; polls safe points (I6)
    pub fn request_safe_point(&mut self) -> SafePointToken;            // STW handshake; quiesce all threads
}
// ---- capture (svm-snap) ----
pub struct CaptureCfg { pub incremental: bool, pub base: Option<Hash32>, pub compress: bool }
pub fn capture(vm: &Vm, tok: SafePointToken, cfg: &CaptureCfg, w: &mut dyn Write)
    -> Result<SnapshotMeta, CaptureError>;                            // read-only on vm (I8); deterministic (I3)
pub struct SnapshotMeta { pub hash: Hash32, pub bytes: u64, pub object_count: u64, pub parent: Option<Hash32> }

// ---- restore (svm-restore) ----
pub fn validate(bytes: &[u8]) -> Result<ValidatedSnapshot, LoadError>; // total; pre-deref checks (I2,I5,I10,I11)
pub fn restore(v: ValidatedSnapshot, rp: &ResourcePolicySet) -> Result<Vm, LoadError>; // relocate+rebuild+reopen
pub enum LoadError { BadMagic, Unsupported{major:u16}, Truncated, IntegrityFail, BadReloc, BadHeader,
                     BaseMismatch, ResourceFail(ResourceHandle), Migrate(MigrateError) }

// ---- resources (svm-resource) ----
pub struct ResourceHandle(u32);
pub enum ResourcePolicy { Reopen, ReopenAtOffset, LazyReconnect, Virtualize, Fail, Drop }   // (I7)
pub struct ResourceDescriptor { pub kind: ResourceKind, pub policy: ResourcePolicy, pub blob: Box<[u8]> }
pub trait Resource {                                                  // CRaC-style hooks
    fn before_checkpoint(&mut self) -> ResourceDescriptor;            // serialize externalizable state
    fn after_restore(&mut self, d: &ResourceDescriptor) -> Result<(), ResourceErr>;  // reopen/reconnect
}

// ---- incremental / COW (svm-cow) ----
pub struct DirtySet { /* per-region page bitmap */ }
impl DirtySet { pub fn mark(&mut self, region:u16, page:u32); pub fn pages(&self) -> impl Iterator<Item=(u16,u32)>; }
pub fn capture_incremental(vm:&Vm, tok:SafePointToken, base:Hash32, dirty:&DirtySet, w:&mut dyn Write)
    -> Result<SnapshotMeta, CaptureError>;                            // overlay over base (principle 8)

// ---- format (svm-format) ----
pub struct SectionDir { /* Vec<{kind, offset, len, crc32c}> */ }
pub struct Header { pub major:u16, pub minor:u16, pub vm_abi:u32, pub page_size:u32,
                    pub base_hint:u64, pub epoch:u64, pub parent:Option<Hash32> }
pub fn content_hash(bytes:&[u8]) -> Hash32;                           // BLAKE3-256 (I10)
