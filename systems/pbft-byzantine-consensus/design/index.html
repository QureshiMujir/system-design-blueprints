Core schema & types (§11)

struct View(u64);              // Copy; immutable; wire: u64 LE via CBOR; compat: append-only
struct SequenceNumber(u64);     // Copy; immutable; monotone per (view); wire: u64
struct Digest([u8; 32]);        // Copy; no Debug/Display (redact() only); wire: 32 raw bytes
struct ReplicaId(u32);          // Copy; stable for cluster lifetime; wire: u32
fn View::next(self) -> View;                       // pure
fn View::primary_of(self, n: u64) -> ReplicaId;    // pure; round-robin (ADR-12)
fn Digest::of(bytes: &[u8]) -> Digest;              // pure; BLAKE3 (ADR-08)
struct Request { operation: Vec<u8>, timestamp: u64, client: ClientId, signature: ClientSig }
struct PrePrepare { view: View, sequence: SequenceNumber, digest: Digest,
                    primary: ReplicaId, authenticator: Auth }
struct Prepare   { view: View, sequence: SequenceNumber, digest: Digest,
                    replica: ReplicaId, authenticator: Auth }
struct Commit    { view: View, sequence: SequenceNumber, digest: Digest,
                    replica: ReplicaId, authenticator: Auth }
struct Checkpoint{ sequence: SequenceNumber, state_digest: Digest,
                    replica: ReplicaId, authenticator: Auth }
struct PreparedProof { pre_prepare: PrePrepare, prepares: Vec<Prepare> }   // len == 2f, I5
struct ViewChange { new_view: View, stable_seq: SequenceNumber,
                    checkpoint_proof: Vec<Checkpoint>, prepared_set: Vec<PreparedProof>,
                    replica: ReplicaId, authenticator: Auth }
struct NewView   { new_view: View, view_change_proofs: Vec<ViewChange>,
                    re_proposed: Vec<PrePrepare>, primary: ReplicaId, authenticator: Auth }
fn Request::digest(&self) -> Digest;   // pure; canonical-CBOR over (operation, timestamp, client)
trait Authenticator: Send + Sync + 'static {
    type Output: Clone + Serialize + DeserializeOwned;
    fn authenticate(&self, payload: &[u8], recipients: &[ReplicaId]) -> Self::Output;
    fn verify(&self, payload: &[u8], auth: &Self::Output, sender: ReplicaId) -> bool;
    fn transferable(&self) -> bool;   // MAC: false; Ed25519/BLS: true (P5)
}
enum Auth { Mac(Vec<MacEntry>), Ed25519([u8; 64]), BlsPartial([u8; 96]), BlsAggregate([u8; 96]) }
struct MacEntry { recipient: ReplicaId, tag: [u8; 32] }
struct Replica<S: StateMachine, A: Authenticator> {
    id: ReplicaId, n: u64, f: u64,                       // n = 3f+1, I2
    current_view: View, stable_seq: SequenceNumber, high_water: SequenceNumber,
    log: Log, state_machine: S, authenticator: A,
    timer: ViewChangeTimer, last_executed: SequenceNumber,
    last_reply_per_client: BTreeMap<ClientId, Reply>,
}
struct Log { /* per-(view,seq) pre_prepares, prepares, commits, checkpoints; append-only */ }
fn Replica::prepared(&self, v: View, n: SequenceNumber, d: Digest) -> bool;         // I3
fn Replica::committed_local(&self, v: View, n: SequenceNumber, d: Digest) -> bool;  // I4
struct Proposal { view: View, height: u64, digest: Digest, parent_qc: Qc }
struct Vote      { view: View, height: u64, digest: Digest, partial_sig: [u8; 96] }
struct Qc        { view: View, height: u64, digest: Digest, aggregate_sig: [u8; 96] }  // I13
trait Pacemaker: Send + 'static {
    fn on_proposal(&mut self, view: View);
    fn on_qc(&mut self, view: View);
    fn on_timeout(&mut self) -> View;
    fn current_leader(&self, view: View) -> ReplicaId;
}
struct Snapshot { sequence: SequenceNumber, root: [u8; 32], chunks: BTreeMap<u64, Chunk> }
struct Chunk([u8; 256]);
fn Snapshot::merkle_proof(&self, index: u64) -> Vec<[u8; 32]>;
fn Snapshot::verify(&self, root: [u8; 32]) -> bool;
enum Action { Discard, Multicast(Message), Equivocation, StateTransferring }
enum BftError { InvalidAuthenticator, SequenceOutOfWindow, DigestMismatch,
                InsufficientQuorum, StaleTimestamp, MalformedNewView }
