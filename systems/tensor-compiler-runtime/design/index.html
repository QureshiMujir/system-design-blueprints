Core schema & types (§11)

pub struct NodeId(pub u32);    // index into Graph.nodes arena
pub struct BufferId(pub u32);  // index into a PrimFunc buffer table
pub struct VarId(pub u32);     // SSA-ish loop/scalar var
pub struct AxisId(pub u32);    // a loop axis within a Schedule
pub struct StageId(pub u32);   // a compute stage within a Schedule
/// The high-level dataflow graph. Arena of nodes + designated I/O.
pub struct Graph {
    pub nodes: Arena<GraphNode>,
    pub outputs: Vec<NodeId>,   // graph result tensors
    pub inputs: Vec<NodeId>,    // placeholders (GraphOp::Input)
}

/// One operator application. `ty` is None until shape inference fills it.
pub struct GraphNode {
    pub op: GraphOp,
    pub args: Vec<NodeId>,      // dataflow operands, DAG edges (acyclic)
    pub ty: Option<TensorType>, // inferred TensorType, verifier-gated
}

/// Operator set. Extensible: `Call(String)` dispatches the op registry.
pub enum GraphOp {
    Input(String),       // named placeholder
    Const(TensorData),   // baked literal tensor
    Add, Mul,            // elementwise (broadcasting)
    MatMul,              // 2D contraction
    Conv2d,              // NCHW default, attrs in registry
    Relu,                // elementwise max(0,x)
    Reshape, Transpose,  // layout/view ops
    Reduce(ReduceKind),  // Sum/Max/Min/Mean over axes
    Call(String),        // registry-resolved extension op
}

/// Full tensor type: static-or-symbolic shape + element dtype.
pub struct TensorType { pub shape: Shape, pub dtype: DType }

pub struct Shape(pub Vec<Dim>);
pub enum Dim {
    Static(i64),       // known extent
    Dynamic(String),   // symbolic dim, resolved at runtime by the VM
}

pub enum DType { F32, F64, I32, I64, I8, Bool }
/// A lowered, schedulable kernel function: typed buffer params + a loop-nest body.
pub struct PrimFunc { pub params: Vec<Buffer>, pub body: Stmt }

/// Statement IR. `For` carries an IterVar; `Allocate` reserves a Buffer.
pub enum Stmt {
    For { var: IterVar, min: Expr, extent: Expr, body: Box<Stmt> },
    Store { buffer: BufferId, index: Expr, value: Expr },
    Block(Vec<Stmt>),
    IfThenElse { cond: Expr, then: Box<Stmt>, els: Option<Box<Stmt>> },
    Allocate { buffer: BufferId },
}

/// Scalar/index expression IR. Pure; no side effects except via Store.
pub enum Expr {
    Load { buffer: BufferId, index: Box<Expr> },
    Var(VarId),
    Const(f64),
    Bin { op: BinOp, lhs: Box<Expr>, rhs: Box<Expr> },
    Call(IntrinId, Vec<Expr>),   // intrinsics: max, exp, fma, etc.
}

/// A typed n-D memory region; shape is symbolic (Vec<Expr>) for dynamic dims.
pub struct Buffer { pub id: BufferId, pub shape: Vec<Expr>, pub dtype: DType, pub scope: Scope }

/// A loop iteration variable + its semantic role.
pub struct IterVar { pub var: VarId, pub kind: IterKind } // Data, Reduce, Thread(deferred)

/// Memory scope. Shared/Thread reserved for the deferred GPU path.
pub enum Scope { Global, Local, Shared }
/// A compiler pass over the Graph IR. Transform passes return a new Graph.
pub trait Pass {
    fn name(&self) -> &str;
    fn run(&self, g: Graph) -> Result<Graph, PassError>;
    fn invalidates(&self) -> &[AnalysisKind]; // which cached analyses this pass dirties
}

/// Ordered pipeline + analysis cache + verifier gate.
pub struct PassManager { /* Vec<Box<dyn Pass>> + analysis cache */ }
impl PassManager {
    pub fn add(&mut self, p: Box<dyn Pass>);
    pub fn run(&self, g: Graph) -> Result<Graph, PassError>; // verifier runs after each pass
}
/// The fusion pass: groups ops, then lowers each group to one PrimFunc.
pub struct FusionPass { /* fusion rules + dominator analysis */ }

/// A maximal fusible subgraph (one elementwise/injective chain + ≤1 reduce).
pub struct FusionGroup {
    pub nodes: Vec<NodeId>,  // member graph nodes, in topo order
    pub root: NodeId,        // group output node
}
/// A PrimFunc + the ordered loop transforms applied to it.
pub struct Schedule { pub func: PrimFunc, pub stages: Vec<Stage> }

/// One compute stage's loop axes under transformation.
pub struct Stage { pub axes: Vec<IterVar> /* loop structure under transform */ }

impl Schedule {
    pub fn split(&mut self, ax: AxisId, factor: i64) -> (AxisId, AxisId); // 1 loop -> outer,inner
    pub fn reorder(&mut self, order: &[AxisId]);                          // permute loop nest
    pub fn fuse(&mut self, a: AxisId, b: AxisId) -> AxisId;               // 2 loops -> 1
    pub fn tile(&mut self, ax: AxisId, ay: AxisId, fx: i64, fy: i64);     // 2D blocking
    pub fn vectorize(&mut self, ax: AxisId);                              // axis -> SIMD lanes
    pub fn unroll(&mut self, ax: AxisId);                                 // fully unroll
    pub fn parallel(&mut self, ax: AxisId);                               // axis -> thread pool
    pub fn compute_at(&mut self, stage: StageId, ax: AxisId);            // nest producer under consumer axis
    pub fn inline(&mut self, stage: StageId);                            // fold producer into consumer
}
/// A compile target: arch + ISA features that gate SIMD codegen.
pub struct Target { pub arch: String, pub features: Vec<String> } // e.g. "x86_64", ["avx2"]

/// Scheduled-TIR -> source backend. CBackend (scalar C) + SimdBackend (portable SIMD).
pub trait CodeGen { fn emit(&self, sched: &Schedule, target: &Target) -> Kernel; }

/// One emitted kernel: name, source text, and its stable ABI signature.
pub struct Kernel { pub name: String, pub source: String, pub signature: KernelSig }

/// The compiled, serializable artifact loaded by the runtime.
pub struct Module { pub kernels: Vec<Kernel>, pub graph: Graph, pub plan: MemoryPlan }
/// Stable C ABI per kernel. Buffers passed as raw pointers + shape/stride metadata.
pub struct KernelSig {
    pub params: Vec<ParamDesc>,  // ordered: inputs then outputs
}
pub struct ParamDesc {
    pub name: String,
    pub dtype: DType,
    pub rank: usize,
    pub role: ParamRole,         // Input | Output | Scalar
}
/// A strided, zero-copy tensor view over ref-counted Storage.
pub struct NDArray {
    pub storage: Storage,
    pub shape: Vec<i64>,
    pub strides: Vec<i64>,  // element strides; enables zero-copy slice/reshape/transpose
    pub dtype: DType,
    pub offset: i64,        // element offset into storage
}

/// A ref-counted device buffer. Send; the unit of memory ownership.
pub struct Storage { ptr: *mut u8, bytes: usize, device: Device }

pub enum Device { Cpu, Gpu /* deferred */ }

/// Pluggable allocation strategy (bump, pool, system).
pub trait Allocator {
    fn alloc(&mut self, bytes: usize) -> Storage;
    fn free(&mut self, s: Storage);
}

/// Liveness-based buffer reuse: which intermediate tensors share Storage.
pub struct MemoryPlan { /* tensor -> storage-slot assignment from liveness analysis */ }

/// Executes a Module's kernels in topological order over NDArray bindings.
pub struct GraphExecutor { /* Module + NDArray binding table + Allocator */ }
/// Register-machine instruction set over NDArray/object registers.
pub enum VmInstruction {
    LoadConst(RegId, ConstId),
    AllocTensor(RegId, ShapeRef),          // runtime shape, resolves Dim::Dynamic
    InvokeKernel(KernelId, Vec<RegId>, RegId), // dispatch a compiled Kernel
    If(RegId, Label, Label),
    Goto(Label),
    Call(FuncId, Vec<RegId>, RegId),       // closures / sub-functions
    Ret(RegId),
}

/// A VM function body + its register frame size.
pub struct VmFunc { pub instrs: Vec<VmInstruction>, pub num_regs: usize }

/// The register-machine interpreter: register file + call stack.
pub struct Interpreter { /* Vec<NDArray|Object> registers, call stack of frames */ }
