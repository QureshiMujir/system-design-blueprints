Core schema & types (§11)

 # src/mac/core/messages.py — identifiers (type-tagged so mypy catches misuse)
RoleId   = Annotated[str, Field(min_length=1, max_length=64)]
WorkerId = Annotated[str, Field(min_length=1, max_length=64)]
TaskId   = UUID
RunId    = UUID
ThreadId = UUID

class Envelope(BaseModel):
    model_config = ConfigDict(frozen=True)
    msg_id: UUID
    run_id: RunId
    thread_id: ThreadId
    parent_msg_id: UUID | None
    sender_role: RoleId
    sender_worker: WorkerId
    recipient_role: RoleId | None       # None => broadcast to the role
    recipient_worker: WorkerId | None    # None => any worker in the role
    sent_at: datetime
    deadline: datetime | None
    payload_kind: str                    # discriminator into PAYLOAD_KINDS
    payload: dict[str, Any]              # validated against the kind's schema
 # Contract-net payloads
class TaskAnnounce(BaseModel):
    task_id: TaskId; description: str; inputs: dict[str, Any]
    eligible_roles: list[RoleId]; deadline_for_bids: datetime
    award_strategy: Literal["lowest-cost", "highest-confidence",
                            "first-acceptable", "manager-choice"]

class Bid(BaseModel):
    task_id: TaskId; bidder: WorkerId
    estimated_cost_usd: float; estimated_wall_seconds: float
    self_assessed_confidence: float; notes: str | None    # confidence in 0..1

class Award(BaseModel):
    task_id: TaskId; awarded_to: WorkerId; deadline: datetime; budget_usd: float

class Result(BaseModel):
    task_id: TaskId; worker: WorkerId
    outcome: Literal["success", "failure", "timeout", "budget-exhausted"]
    output: dict[str, Any]; cost_usd: float; wall_seconds: float
    trace_span_id: str | None
 # Inter-agent dialogue + lifecycle + blackboard payloads
class Speak(BaseModel):       text: str; cites: list[str]      # cites = blackboard keys
class Critique(BaseModel):    target_msg_id: UUID; rating: Literal["accept","revise","reject"]; reasons: list[str]
class Vote(BaseModel):        target_msg_id: UUID; choice: str; weight: float
class Write(BaseModel):       key: str; value: Any; expected_version: int | None   # CAS guard
class Subscribe(BaseModel):   key_prefix: str
class Cancel(BaseModel):      task_id: TaskId; reason: Literal["budget-exhausted","deadline-exceeded","user-cancelled","supervisor-cancelled"]
class Heartbeat(BaseModel):   worker: WorkerId; busy_with: TaskId | None; queue_depth: int

PAYLOAD_KINDS: dict[str, type[BaseModel]]   # "task-announce"→TaskAnnounce, … , "heartbeat"→Heartbeat
 # src/mac/core/errors.py — error hierarchy (all public APIs raise subclasses)
class MacError(Exception): ...
class PhaseDeferred(MacError): ...        # carries the register anchor string
class BudgetExhausted(MacError): ...      # watchdog tripped; clean shutdown
class DeadlineExceeded(MacError): ...     # per-task deadline elapsed
class RoleHallucination(MacError): ...    # sender fields mismatch; dead-lettered
class StuckLoop(MacError): ...            # repeated (sender,recipient,hash) in window
class DuplicateAward(MacError): ...       # two coordinators awarded same task
class NoBidders(MacError): ...            # bid deadline elapsed with zero bids
class Conflict(MacError): ...             # CAS version mismatch on blackboard write
