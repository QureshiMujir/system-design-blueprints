Core schema & types (§11)

struct Point { x: f32, y: f32 }
struct Vec2  { x: f32, y: f32 }
struct Matrix { a:f32, b:f32, c:f32, d:f32, e:f32, f:f32 }  // 2x3 affine

fn Matrix::identity() -> Matrix;
fn Matrix::concat(&self, rhs: &Matrix) -> Matrix;          // self · rhs
fn Matrix::invert(&self) -> Option<Matrix>;                // None if |det|<1e-12 (I5)
fn Matrix::map_point(&self, p: Point) -> Point;
fn Matrix::det(&self) -> f32;
struct Transform { stack: Vec<Matrix>, current: Matrix }
fn save(&mut self);                 // push current
fn restore(&mut self) -> bool;      // pop; false if underflow
fn concat(&mut self, m: &Matrix);   // current = current · m
enum Verb { Move=0, Line=1, Quad=2, Cubic=3, Close=4 }
struct Path { verbs: Arc<[Verb]>, points: Arc<[Point]>, bounds: Rect }
struct PathBuilder { verbs: Vec<Verb>, points: Vec<Point>, last: Point, start: Point }

fn move_to(&mut self, p: Point); fn line_to(&mut self, p: Point);
fn quad_to(&mut self, c: Point, e: Point);
fn cubic_to(&mut self, c1: Point, c2: Point, e: Point);
fn close(&mut self); fn finish(self) -> Result<Path, PathError>;  // rejects non-finite (I-edge)
struct Cell { area: i32, cover: i32 }     // fixed-point Q24.8 (ADR A05)
struct CellGrid<'a> { cells: &'a mut [Cell], width: i32, x_min: i32, ... }  // arena-backed
struct Span<'c> { y: i32, x0: i32, len: u32, coverage: &'c [u8] }  // 0..=255

fn EdgeBuilder::add_line(&mut self, p0: Point, p1: Point);   // accumulate into grid
fn sweep_scanline(grid:&CellGrid, y:i32, rule:FillRule, out:&mut Vec<u8>); // prefix-sum
fn SpanEmitter::emit(coverage:&[u8], y:i32, clip:&ClipMask) -> impl Iterator<Item=Span>;
enum FillRule { NonZero, EvenOdd }
struct Color { r:f32, g:f32, b:f32, a:f32 }       // straight, [0,1]
struct PremulRgba8 { r:u8, g:u8, b:u8, a:u8 }     // premultiplied
struct PremulF16   { r:f16, g:f16, b:f16, a:f16 } // premul, linear-light

fn premultiply(c: Color) -> Color; fn unpremultiply(c: Color) -> Color;
fn srgb_to_linear(x: f32) -> f32;  fn linear_to_srgb(x: f32) -> f32;  // table+exact (I6)
fn dither_8bit(linear: f32, x:i32, y:i32) -> u8;  // Bayer 8x8 (A11)
enum BlendMode { /* SrcOver default + 11 PD + 16 W3C */ }
enum SpreadMode { Pad, Repeat, Reflect }
struct Stop { offset: f32, color: Color }   // 0..=1 ascending
struct StrokeStyle { width:f32, join:Join, cap:Cap, miter_limit:f32, dash:Option<Dash> }
enum Join { Miter, Round, Bevel }  enum Cap { Butt, Round, Square }

trait Shader {                              // pure fn of device coords (P8)
    fn shade_span(&self, y:i32, x0:i32, len:u32, ctm_inv:&Matrix, out:&mut [PremulRgba8]);
}
struct Paint { shader: Box<dyn Shader>, blend: BlendMode, anti_alias: bool,
               stroke: Option<StrokeStyle>, color_space: ColorSpace }
struct PdFactors { fa: f32, fb: f32 }       // per Porter-Duff operator
fn pd_factors(op: PorterDuff, alpha_s:f32, alpha_d:f32) -> PdFactors;
fn blend_separable(mode:BlendMode, cs:Color, cd:Color) -> Color;
fn blend_nonseparable(mode:BlendMode, cs:[f32;3], cd:[f32;3]) -> [f32;3];
fn composite_span_scalar(op:&Paint, src:&[PremulRgba8], cov:&[u8], dst:&mut [PremulRgba8]); // oracle
fn composite_span_simd(...);                 // MUST equal scalar bit-for-bit (P14,I-parity)
enum PixelFormat { Rgba8888Premul, F16x4, A8 }
struct Surface<'b> { fmt:PixelFormat, width:u32, height:u32, stride:usize, bytes:&'b mut [u8] }
fn row_mut(&mut self, y:u32) -> &mut [u8];
struct Arena { buf: Vec<u8>, head: usize, high_water: usize }  // bump; reset() per frame (I12)
fn alloc<T>(&mut self, n:usize) -> &mut [T];  fn reset(&mut self);
struct ClipEntry { rect: Rect, mask: Option<ClipMask> }   // A8 mask if non-rect
struct ClipStack { entries: Vec<ClipEntry>, bounds: Rect }
fn push_rect(&mut self, r:Rect);            // intersect; monotonic shrink (I8)
fn push_path(&mut self, mask:ClipMask, bounds:Rect);
fn pop(&mut self) -> bool;
fn apply(&self, span:&mut Span);            // multiply coverage; clamp x-range (I9)
enum DrawOp {
  Save, Restore, Concat(Matrix),
  ClipRect(Rect), ClipPath(Path, FillRule),
  Fill(Path, FillRule, Paint),
  Stroke(Path, StrokeStyle, Paint),
  DrawImage(ImageRef, Matrix, SamplingMode, Paint),
}
struct Picture { ops: Vec<DrawOp>, bounds: Rect }
fn Picture::playback(&self, canvas:&mut Canvas);   // identical to immediate (P13)
struct Canvas<'s> { surface: Surface<'s>, ctm: Transform, clip: ClipStack, arena: Arena }
fn fill_path(&mut self, p:&Path, rule:FillRule, paint:&Paint);
fn stroke_path(&mut self, p:&Path, style:&StrokeStyle, paint:&Paint);
fn draw_image(&mut self, img:&ImageRef, m:&Matrix, s:SamplingMode, paint:&Paint);
fn clip_rect(&mut self, r:Rect);  fn clip_path(&mut self, p:&Path, rule:FillRule);
fn save(&mut self);  fn restore(&mut self);
enum PathError { PointNotFinite, EmptyPath }
enum MatrixError { Singular }
enum PaintError { GradientNeedsStops, StopsNotSorted, MiterLimitInvalid }
enum SurfaceError { UnsupportedFormat, BufferTooSmall, DimensionTooLarge }
enum ClipError { StackOverflow, Underflow }
