Core schema & types (§11)

struct Document { definitions: Vec<Definition> }
enum Definition { Operation(OperationDefinition), Fragment(FragmentDefinition), TypeSystem(TypeSystemDef) }
enum OperationType { Query, Mutation, Subscription }
struct OperationDefinition { kind: OperationType, name: Option<Name>, variables: Vec<VariableDef>, directives: Vec<Directive>, selection_set: SelectionSet, loc: Location }
struct SelectionSet { selections: Vec<Selection>, loc: Location }
enum Selection { Field(Field), FragmentSpread(FragmentSpread), InlineFragment(InlineFragment) }
struct Field { alias: Option<Name>, name: Name, arguments: Vec<Argument>, directives: Vec<Directive>, selection_set: Option<SelectionSet>, loc: Location }
struct FragmentSpread { name: Name, directives: Vec<Directive>, loc: Location }
struct InlineFragment { type_condition: Option<NamedType>, directives: Vec<Directive>, selection_set: SelectionSet, loc: Location }
struct FragmentDefinition { name: Name, type_condition: NamedType, directives: Vec<Directive>, selection_set: SelectionSet, loc: Location }
enum Value { Variable(Name), Int(i64), Float(f64), String(String), Boolean(bool), Null, Enum(Name), List(Vec<Value>), Object(Vec<(Name,Value)>) } // input AST
enum TypeRef { Named(Name), List(Box<TypeRef>), NonNull(Box<TypeRef>) }   // a type *reference* in query/SDL
struct Argument { name:Name, value:Value, loc:Location }   struct Directive { name:Name, arguments:Vec<Argument>, loc:Location }
struct VariableDef { var:Name, ty:TypeRef, default:Option<Value>, loc:Location }
struct Pos { offset:u32 }   struct Location { line:u32, column:u32 }   // Pos = byte offset; Location for reporting
enum PathSeg { Key(String), Index(usize) }   type Path = Vec<PathSeg>;  // e.g. ["user","pets",1,"name"]
struct GqlError { message:String, locations:Vec<Location>, path:Option<Path>, extensions:Option<Map> }  // path for field errors (I4)
enum RValue { Null, Bool(bool), Int(i64), Float(f64), String(String), Enum(String), List(Vec<RValue>), Object(IndexMap<String,RValue>) } // IndexMap keeps key order (I7)
enum Type { Scalar(Arc<ScalarType>), Object(Arc<ObjectType>), Interface(Arc<InterfaceType>), Union(Arc<UnionType>), Enum(Arc<EnumType>), InputObject(Arc<InputObjectType>), List(Box<Type>), NonNull(Box<Type>) } // resolved; wrapping types §8.3
struct FieldDef { name: String, args: Vec<InputValueDef>, ty: Type, directives: Vec<DirectiveUse>, deprecated: Option<String> }
struct ObjectType { name: String, fields: IndexMap<String,FieldDef>, interfaces: Vec<String> }
struct InterfaceType { name: String, fields: IndexMap<String,FieldDef> }
struct UnionType { name: String, members: Vec<String> }   struct EnumType { name: String, values: Vec<EnumValue> }
struct InputObjectType { name: String, fields: IndexMap<String,InputValueDef> }   struct ScalarType { name: String /* +serialize/parse */ }
struct Schema { query: Arc<ObjectType>, mutation: Option<Arc<ObjectType>>, subscription: Option<Arc<ObjectType>>, types: HashMap<String,Type>, directives: HashMap<String,DirectiveDef> }
impl Schema { fn build_from_sdl(sdl:&str)->Result<Schema,Vec<GqlError>>;  // P3
  fn validate(&self)->Result<(),Vec<GqlError>>;                           // P4, I2
  fn type_of(&self,name:&str)->Option<&Type>;  fn possible_types(&self,abstract_ty:&str)->Vec<&Arc<ObjectType>>; }
struct ValidationContext<'a> { schema:&'a Schema, document:&'a Document, errors: Vec<GqlError> }
trait Rule { fn check(&self, ctx:&mut ValidationContext); }              // one rule per §8.6 row
fn validate(schema:&Schema, doc:&Document) -> Vec<GqlError>;             // runs all rules (P5, I1)
struct ResolveInfo<'a> { field_name:&'a str, return_type:&'a Type, parent_type:&'a str, path:&'a Path, schema:&'a Schema }
type Resolver = Arc<dyn Fn(&dyn Source,&Args,&Context,&ResolveInfo)->BoxFuture<'static,Result<RValue,GqlError>> + Send+Sync>;
struct ResolverRegistry { map: HashMap<(String,String),Resolver> }  // (type,field)
struct Context { /* host per-request: data sources, user, loaders */ }   struct Args(IndexMap<String,RValue>); // coerced (I9)
struct ExecuteRequest<'a> { schema:&'a Schema, document:&'a Document, operation_name:Option<&'a str>, variables: Map, context: Context }
struct ExecutionResult { data: Option<RValue>, errors: Vec<GqlError> }   // I8
async fn execute(req: ExecuteRequest<'_>, reg:&ResolverRegistry) -> ExecutionResult;  // P6/P7
fn collect_fields(object_ty:&ObjectType, ss:&SelectionSet, vars:&Map, schema:&Schema) -> IndexMap<String,Vec<Field>>;  // ordered (I5)
type BatchFn<K,V> = Arc<dyn Fn(Vec<K>)->BoxFuture<'static,Result<Vec<V>,GqlError>> + Send+Sync>;
struct DataLoader<K,V> { batch_fn: BatchFn<K,V>, cache: Mutex<HashMap<K,V>>, queued: Mutex<Vec<K>> }
impl<K:Eq+Hash+Clone,V:Clone> DataLoader<K,V> { fn new(b:BatchFn<K,V>)->Self;
  async fn load(&self,key:K)->Result<V,GqlError>;  async fn load_many(&self,keys:Vec<K>)->Result<Vec<V>,GqlError>; // batched in a tick (I10)
  fn clear(&self,key:&K);  fn clear_all(&self);  fn prime(&self,key:K,value:V); }  // cache invalidation for read-your-writes after a mutation (§21.8); prime seeds a known value
fn inject_introspection(schema:&mut Schema);   // adds __Schema/__Type/... (P10, I11); __typename → concrete type; __schema/__type read live Schema
type SubscribeResolver = Arc<dyn Fn(&Args,&Context)->BoxFuture<'static,Result<BoxStream<'static,RValue>,GqlError>> + Send+Sync>; // source stream (P11)
async fn subscribe(req: ExecuteRequest<'_>, reg:&ResolverRegistry) -> Result<BoxStream<'static,ExecutionResult>,Vec<GqlError>>;  // map → per-event (I14)
struct GraphQLRequest { query: String, variables: Map, operation_name: Option<String> }
struct Limits { max_depth:u32, max_complexity:u32, max_tokens:u32 }   // DoS guards (I12/I13)
fn depth_of(doc:&Document)->u32;   fn complexity_of(doc:&Document, schema:&Schema)->u32;  // field-weight × list-factor (§4.1)
enum GqlErrorKind { Syntax{loc:Location}, Validation, VariableCoercion, ArgumentCoercion, FieldResolver{path:Path}, NonNullViolation{path:Path}, DepthLimit, ComplexityLimit, BadAbiInput }
