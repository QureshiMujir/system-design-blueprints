Core schema & types (§11)

package catalog

type Node struct {
    ID              string
    Node            string            // hostname-style name
    Address         string
    TaggedAddresses map[string]string // lan, wan, lan_ipv4, wan_ipv4
    Datacenter      string
    Meta            map[string]string
    CreateIndex     uint64
    ModifyIndex     uint64
}

type Service struct {
    Name              string
    Tags              []string
    Meta              map[string]string
    Connect           *ConnectConfig // §22, scaffolding only
    EnableTagOverride bool
}

type Weights struct{ Passing, Warning int }

type Instance struct {
    ID, Node, Service string
    Tags              []string
    Meta              map[string]string
    Address           string
    Port              int
    Weights           Weights
    Datacenter        string
    Namespace         string // §19, Phase-0 default namespace only
    CreateIndex       uint64
    ModifyIndex       uint64
}
package catalog

type CheckStatus string

const (
    CheckPassing  CheckStatus = "passing"
    CheckWarning  CheckStatus = "warning"
    CheckCritical CheckStatus = "critical"
)

type CheckKind string

const (
    CheckTTL CheckKind = "ttl"; CheckHTTP CheckKind = "http"
    CheckTCP CheckKind = "tcp"; CheckGRPC CheckKind = "grpc"
    CheckScript CheckKind = "script" // §16, deferred (tracked §37)
    CheckAlias  CheckKind = "alias"  // §16, deferred (tracked §37)
)

type CheckDefinition struct {
    Kind                           CheckKind
    HTTP, TCP, GRPC                string
    GRPCUseTLS                     bool
    Method                         string
    Header                         map[string][]string
    Interval, Timeout, TTL         string
    DeregisterCriticalServiceAfter string
    SuccessBeforePassing           int
    FailuresBeforeCritical         int
}

type Check struct {
    ID, Name, Node, ServiceID, Service string
    Status                             CheckStatus
    Notes, Output                      string
    Definition                         CheckDefinition
    CreateIndex, ModifyIndex           uint64
}
package catalog

type RegisterRequest struct {
    RequestID string // UUID, dedup key, I6/P12
    Node      Node
    Service   Service
    Instance  Instance
    Checks    []Check
}

type QueryOptions struct{ Index uint64; Wait string; Token string; Datacenter string }
type WriteOptions struct{ Token string; Datacenter string }
type Meta struct{ Index uint64; LastContact string; KnownLeader bool }

type KVPair struct {
    Key, Value        string
    Flags             uint64
    Session           string // §17, deferred (tracked §37)
    CreateIndex       uint64
    ModifyIndex       uint64
}
package acl

type Token struct {
    AccessorID string
    SecretID   string
    Policies   []string
    Namespace  string
    CreateIndex, ModifyIndex uint64
}

type Policy struct{ Name, Description, Rules string }

package agent

type Config struct {
    Datacenter   string
    Domain       string // DNS domain, default "discovery.", never hard-coded (§16.6)
    DataDir      string
    BindAddr     string
    Ports        PortConfig
    ACLDefault   string // "allow" | "deny", production default "deny" (I4)
}

type PortConfig struct{ HTTP, DNS, Gossip, GossipWAN, Raft int }
type Registrar interface {
    Register(ctx context.Context, req RegisterRequest) (uint64, error)
    Deregister(ctx context.Context, instanceID string) (uint64, error)
}
type CatalogReader interface {
    Services(ctx context.Context, q *QueryOptions) (map[string][]string, *Meta, error)
    Service(ctx context.Context, name string, q *QueryOptions) ([]Instance, *Meta, error)
}
type HealthReader interface {
    Service(ctx context.Context, name string, passingOnly bool, q *QueryOptions) ([]Instance, *Meta, error)
    State(ctx context.Context, state CheckStatus, q *QueryOptions) ([]Check, *Meta, error)
}
type KV interface {
    Get(ctx context.Context, key string, q *QueryOptions) (*KVPair, *Meta, error)
    CAS(ctx context.Context, p *KVPair, q *WriteOptions) (bool, *Meta, error)
}
type CheckEngine interface {
    AddCheck(def CheckDefinition) error
    Update(id string, status CheckStatus, output string) error
}
type DNSResolver interface {
    Service(name, dc, tag string, qtype uint16) ([]dns.RR, error)
}
