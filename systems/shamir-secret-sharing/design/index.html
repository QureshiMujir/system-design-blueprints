Core schema & types (§11)

/// Element of the prime field F_p (curve scalar field).
/// Invariant: value is always reduced mod p; no negative representation.
#[derive(Clone, Copy, PartialEq, Eq, Zeroize, ZeroizeOnDrop)]
pub struct Scalar(/* [u8; 32] canonical little-endian */);

impl Scalar {
    pub fn zero() -> Self;
    pub fn one() -> Self;
    pub fn from_bytes_mod_order(bytes: &[u8; 64]) -> Self;   // wide reduction
    pub fn from_canonical_bytes(bytes: &[u8; 32]) -> Option<Self>;
    pub fn to_bytes(&self) -> [u8; 32];                       // canonical encoding
    pub fn invert(&self) -> CtOption<Self>;                   // Fermat: self^(p-2)
    pub fn is_zero(&self) -> Choice;                          // constant-time
}
impl Add<Scalar> for Scalar { type Output = Scalar; /* ... */ }
impl Mul<Scalar> for Scalar { type Output = Scalar; /* ... */ }
impl Neg for Scalar    { type Output = Scalar; }
impl Sub<Scalar> for Scalar { type Output = Scalar; }
/// Alias — in the EC instantiation the secret-sharing field equals the curve scalar field.
pub type FieldElement = Scalar;
/// Non-zero evaluation point x_i in F_p identifying one participant.
/// x=0 is reserved: it is the secret (the polynomial constant term).
#[derive(Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]
pub struct ShareId(pub(crate) u32);   // 1-indexed; 0 is invalid

impl ShareId {
    pub fn new(v: u32) -> Result<Self, TssError>;   // rejects 0
    pub fn as_scalar(&self) -> Scalar;               // embed u32 into F_p
}
/// One party's evaluation f(id) of the sharing polynomial.
/// MUST be zeroized after reconstruction or resharing.
#[derive(Clone, Zeroize, ZeroizeOnDrop)]
pub struct Share {
    pub id: ShareId,
    pub value: Scalar,   // f(id); secret portion — never log or serialize unencrypted
}
/// Threshold policy: t-of-n.
/// Invariants: 1 <= t <= n; n >= 1.
#[derive(Clone, Copy, PartialEq, Eq)]
pub struct ThresholdParams {
    pub t: usize,   // reconstruction threshold; polynomial degree = t-1
    pub n: usize,   // total number of shares / parties
}

impl ThresholdParams {
    pub fn new(t: usize, n: usize) -> Result<Self, TssError>;
    pub fn degree(&self) -> usize;   // = t - 1
    pub fn is_sufficient(&self, share_count: usize) -> bool;
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TssError {
    InvalidThreshold { t: usize, n: usize },
    InvalidShareId(u32),              // id = 0
    DuplicateShareId(ShareId),
    InsufficientShares { have: usize, need: usize },
    VerificationFailed { party: ShareId },
    InvalidScalar,                    // non-canonical bytes
    InvalidPoint,                     // not on curve / not in prime-order subgroup
    DkgComplaint { accuser: ShareId, accused: ShareId },
    DkgInsufficientQualified { qualified: usize, threshold: usize },
    PartialSignatureInvalid { party: ShareId },
    AbortWithEvidence(Vec<ShareId>),  // identifiable-abort: list of misbehaving parties
    SerializationError(String),
    InternalError(&'static str),      // invariant violation; always a bug
}
/// State machine states for the distributed key generation protocol.
#[derive(Clone, Copy, PartialEq, Eq, Debug)]
pub enum DkgRound {
    Commit,      // Round 1: broadcast VSS commitments
    Share,       // Round 2: send encrypted sub-shares point-to-point
    Complaint,   // Round 3: broadcast complaints; disqualify bad dealers
    Finalize,    // Round 4: aggregate qualified shares; derive joint key
}
/// Prime-order elliptic-curve group element (Ristretto255 over Ed25519 field).
/// Commitments and public keys are Points.
#[derive(Clone, Copy, PartialEq, Eq, Zeroize)]
pub struct Point(/* RistrettoPoint */);

impl Point {
    pub fn identity() -> Self;
    pub fn generator() -> Self;                          // canonical G
    pub fn second_generator() -> Self;                   // H; NUMS, independent of G
    pub fn compress(&self) -> [u8; 32];                  // canonical encoding
    pub fn decompress(bytes: &[u8; 32]) -> Option<Self>; // None if not on curve
    pub fn is_identity(&self) -> Choice;
}
impl Add<Point> for Point    { type Output = Point; }
impl Neg for Point           { type Output = Point; }
impl Sub<Point> for Point    { type Output = Point; }
impl Mul<Scalar> for Point   { type Output = Point; }   // constant-time scalar mul
/// Feldman VSS commitment: vector of Points g^{a_j} for j in 0..t.
/// Index 0 is the commitment to the secret (g^{a_0} = g^{secret}).
pub type Commitment = Vec<Point>;
/// Pedersen VSS commitment: vector of Points g^{a_j} h^{b_j} for j in 0..t.
/// Perfectly hiding (given discrete-log hardness of G, H).
pub type PedersenCommitment = Vec<Point>;
/// Degree-(t-1) polynomial over F_p with coeffs[0] = secret.
/// Drop impl calls zeroize — do not clone or persist.
#[derive(Zeroize, ZeroizeOnDrop)]
pub struct Polynomial {
    pub(crate) coeffs: Vec<Scalar>,  // coeffs[0] = f(0) = secret; len = t
}

impl Polynomial {
    pub fn new(secret: Scalar, degree: usize, rng: &mut impl CryptoRng) -> Self;
    pub fn eval(&self, x: &Scalar) -> Scalar;     // Horner; constant-time
    pub fn degree(&self) -> usize;                 // coeffs.len() - 1
}
/// A party's long-term key material output from a DKG ceremony.
/// Persisted encrypted; never reconstructed into one place.
#[derive(Clone, Zeroize, ZeroizeOnDrop)]
pub struct KeyShare {
    pub id: ShareId,
    pub secret_share: Scalar,     // f(id) on the joint polynomial; MUST be zeroized after use
    pub public_share: Point,      // g^{secret_share}; used for partial-sig verification
    pub group_public: Point,      // joint public key Y = g^{joint_secret}
    pub params: ThresholdParams,
    pub epoch: u64,               // incremented on each proactive refresh
}
/// FROST per-round nonce commitment pair published in Round 1.
/// (D_i, E_i) = (g^d_i, g^e_i) where d_i, e_i are ephemeral nonces.
#[derive(Clone, Copy, PartialEq, Eq)]
pub struct NonceCommitment {
    pub hiding:  Point,   // D_i = g^{d_i}
    pub binding: Point,   // E_i = g^{e_i}
    pub id: ShareId,
}
/// rho_i = H("frost-binding", i, msg, commitment_list)
/// Derived deterministically per signer per round; never stored.
#[derive(Clone, Copy, Zeroize, ZeroizeOnDrop)]
pub struct BindingFactor(pub(crate) Scalar);
/// A signer's contribution to a threshold signature.
/// For FROST: z_i = d_i + rho_i*e_i + lambda_i*share_i*c
/// For BLS: sigma_i = H(m)^{share_i}
#[derive(Clone, Zeroize, ZeroizeOnDrop)]
pub struct PartialSignature {
    pub id: ShareId,
    pub value: PartialSigValue,
}

#[derive(Clone, Zeroize, ZeroizeOnDrop)]
pub enum PartialSigValue {
    Schnorr(Scalar),  // z_i scalar
    Bls(Point),       // sigma_i group element
}
/// Final threshold signature valid under the joint public key.
#[derive(Clone, Copy, PartialEq, Eq)]
pub struct AggregateSignature {
    pub variant: SigVariant,
}

#[derive(Clone, Copy, PartialEq, Eq)]
pub enum SigVariant {
    Frost { R: Point, z: Scalar },   // standard (R,z) Schnorr signature
    Bls { sigma: Point },            // aggregated BLS point
}
