Core schema & types (§11)

/// Generational arena id into AXTree::nodes. index:u24 + generation:u8 packed in u32.
/// Reused indices bump generation -> stale ids resolve to None (no dangling AXNode).
pub struct AXNodeId(u32);

/// An accessible object. Replaced (not mutated) on recompute to enable diffing.
pub struct AXNode {
    id: AXNodeId,
    role: Role,                         // computed role (explicit -> implicit -> Generic)
    name: AccessibleName,
    description: AccessibleDescription,
    states: AXStateSet,
    props: AXPropertyMap,
    dom_node: Option<OpaqueDomId>,      // None for synthetic/text inline nodes
    parent: Option<AXNodeId>,           // AX parent (differs from DOM via owns/prune)
    children: Vec<AXNodeId>,            // AX order (owned nodes appended at owner)
    ignored: bool,                      // true -> not exposed to AT (pruned/presentational)
    bounds: Option<Rect>,               // hit-test rect; None until layout supplies it
}
/// WAI-ARIA 1.2 role taxonomy. #[non_exhaustive]. ABSTRACT category is intentionally
/// NOT representable as a value -> abstract roles can never be a computed role.
pub enum Role {
    // ABSTRACT (rejected as values; parse_role_tokens skips them): RoleType, Widget,
    //   Composite, Input, Landmark, Range, Section, Sectionhead, Select, Structure, Window, Command.
    // WIDGET:    Button, Checkbox, Gridcell, Link, MenuItem, MenuItemCheckbox, MenuItemRadio,
    //            Option, ProgressBar, Radio, ScrollBar, SearchBox, Separator, Slider, SpinButton,
    //            Switch, Tab, TabPanel, TextBox, Treeitem
    // COMPOSITE: ComboBox, Grid, ListBox, Menu, MenuBar, RadioGroup, TabList, Tree, TreeGrid
    // STRUCTURE: Application, Article, Cell, ColumnHeader, Definition, Document, Feed, Figure,
    //            Generic, Group, Heading, Img, List, ListItem, Math, None, Note, Paragraph,
    //            Presentation, Row, RowGroup, RowHeader, Table, Term, Toolbar, Tooltip, Caption,
    //            Code, Emphasis, Strong, Time
    // LANDMARK:  Banner, Complementary, ContentInfo, Form, Main, Navigation, Region, Search
    // LIVE:      Alert, Log, Marquee, Status, Timer
    // WINDOW:    AlertDialog, Dialog
}
pub enum AXStateValue { True, False, Mixed, Undefined } // tri/quad-state (e.g. aria-checked="mixed")

/// StateKey -> AXStateValue | bool. Sparse; absent key = spec default for the role.
pub struct AXStateSet { /* busy, checked, disabled, expanded, focused, focusable, hidden,
    invalid, multiselectable, pressed, readonly, required, selected, modal, multiline,
    atomic, ... */ }

/// Numeric/string ARIA properties + native equivalents. Sparse map.
pub struct AXPropertyMap { /* level, valuemin, valuemax, valuenow, valuetext, posinset,
    setsize, keyshortcuts, roledescription, placeholder, colindex, rowindex, colcount,
    rowcount, ... */ }
pub struct AccessibleName { text: String, from: NameFrom, sources: Vec<NameSource> }
pub enum NameFrom { Author, Contents, Placeholder, Title, RelatedElement, Prohibited }
// NameSource: per-step provenance for name-trace (which attr/node produced each fragment).
pub struct AccessibleDescription { text: String, from: DescriptionFrom }
pub enum AXRelation { Owns, Controls, LabelledBy, DescribedBy, FlowTo, ActiveDescendant, Details, ErrorMessage } // #[non_exhaustive]

pub struct AXTree {
    nodes: Arena<AXNode>,
    root: AXNodeId,
    dom_to_ax: Map<OpaqueDomId, AXNodeId>, // forward map; lookup for mark_dirty + relation resolution
    dirty: Set<AXNodeId>,                  // recompute frontier
}
/// Minimal mutation deltas; the incremental serialization wire form. #[non_exhaustive].
pub enum AXTreeUpdate {
    NodeAdded { id: AXNodeId, parent: AXNodeId },
    NodeRemoved { id: AXNodeId },
    RoleChanged { id: AXNodeId, role: Role },
    NameChanged { id: AXNodeId, name: AccessibleName },
    StateChanged { id: AXNodeId, states: AXStateSet },
    Reparented { id: AXNodeId, new_parent: AXNodeId },
}
// AXEvent: AT-facing notifications (focus moved, live-region announce, value changed).

pub enum AXError { UnknownRole, AbstractRoleUsed, RelationCycle, OrphanReference } // #[non_exhaustive]
/// Read-only view of one styled DOM node. Implemented by the host (projects 124/260)
/// and by ax-dom's reference arena. Borrowed immutably for the whole build/recompute call.
pub trait DomNode: Sized + Clone {
    fn local_name(&self) -> &LocalName;             // lowercased tag, e.g. "button"
    fn namespace(&self) -> &Namespace;              // HTML / SVG / MathML -> gates implicit role
    fn attr(&self, name: &str) -> Option<&str>;     // non-namespaced attr (type, href, alt...)
    fn aria(&self, name: &str) -> Option<&str>;     // "aria-*"/"role"; None if absent
    fn computed_display_is_none(&self) -> bool;      // true -> whole subtree excluded
    fn computed_visibility_hidden(&self) -> bool;    // visibility:hidden|collapse (nuanced inclusion)
    fn is_focusable(&self) -> bool;                  // tabindex/native -> presentational conflict input
    fn text_content(&self) -> Option<&str>;          // for name-from-content leaves
    fn native_state(&self) -> NativeState;           // checked/disabled/required/readonly/value/open
    fn parent(&self) -> Option<Self>;
    fn children(&self) -> Vec<Self>;                 // DOM order; flattened tree (post-slotting)
    fn assigned_slot(&self) -> Option<Self>;         // shadow slotting -> AX follows flat tree
    fn shadow_root(&self) -> Option<Self>;           // shadow host -> descend into shadow tree
    fn opaque(&self) -> OpaqueDomId;                 // stable identity; key for dom_to_ax + visited set
}
pub fn compute_role<N: DomNode>(n: &N) -> Role;            // explicit -> implicit -> Generic
pub fn parse_role_tokens(value: &str) -> Option<Role>;     // first valid non-abstract token in fallback list
pub fn implicit_role<N: DomNode>(n: &N) -> Option<Role>;   // HTML-AAM tag+attr table; input[type] matrix
pub fn validate_role(role: Role, ctx: &RoleContext) -> Result<(), AXError>;
pub fn compute_name<N: DomNode>(n: &N, role: Role) -> AccessibleName;     // role-gated precedence
pub fn compute_description<N: DomNode>(n: &N) -> AccessibleDescription;

/// Per-traversal recursion guard. visited prevents cyclic aria-labelledby; in_aria_labelledby
/// blocks re-entering a labelledby step (a labelledby traversal does not re-enter labelledby).
struct NameContext { visited: Set<OpaqueDomId>, traversal_root: OpaqueDomId, in_aria_labelledby: bool }
pub fn build_tree<N: DomNode>(root: &N) -> AXTree;          // full DOM->AX walk
pub fn is_included<N: DomNode>(n: &N, role: Role) -> bool;  // inclusion/exclusion predicate
pub fn apply_relations(tree: &mut AXTree);                  // owns reparent + relation resolution
pub fn mark_dirty(tree: &mut AXTree, dom: OpaqueDomId, kind: MutationKind);
pub fn recompute(tree: &mut AXTree) -> Vec<AXTreeUpdate>;   // drains dirty; emits minimal diff
/// The cross-process surface. Real OS IPC is PhaseDeferred behind this trait.
pub trait PlatformBackend { fn apply(&mut self, updates: &[PlatformUpdate]); }

pub fn map_uia(node: &AXNode) -> UiaNode;     // Role -> UIA ControlType (Button -> ControlType.Button)
pub fn map_ia2(node: &AXNode) -> Ia2Node;     // -> IAccessible2+MSAA (Button -> ROLE_PUSH_BUTTON)
pub fn map_atspi(node: &AXNode) -> AtspiNode; // -> AT-SPI2 (Button -> PUSH_BUTTON)
pub fn map_macos(node: &AXNode) -> MacAxNode; // -> NSAccessibility (Button -> NSAccessibilityButtonRole)
