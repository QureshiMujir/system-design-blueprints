Core schema & types (§11)

/// Hash/index handle for a visited state. Dense u64 assigned by StateTable on
/// first insertion; stable for the lifetime of one search. NOT serialized.
#[derive(Clone, Copy, PartialEq, Eq, Hash)]
pub struct StateId(pub u64);

/// Packed valuation of ALL model variables. The dominant memory unit — one per
/// visited state in the StateTable. bits laid out per StateEncoder var->range.
/// Hash + Eq are over the raw bit pattern (canonical: encoder zero-pads unused).
#[derive(Clone, PartialEq, Eq, Hash)]
pub struct State { pub bits: BitVec }   // BitVec = compact in-crate bit buffer

/// Interned proposition id (e.g. "x>3", "pc==crit"). u32 -> 4B keys, dense.
#[derive(Clone, Copy, PartialEq, Eq, Hash)]
pub struct AtomicProp(pub u32);

/// Set of AtomicProps true in a given state. BitSet indexed by AtomicProp.0.
/// Produced by TransitionSystem::label; consumed as Buchi-edge guards.
#[derive(Clone, PartialEq, Eq)]
pub struct Label(pub BitSet);

/// An explicit edge (used by KripkeStructure / explicit CTL labeling, NOT by
/// on-the-fly search which never materializes edges).
#[derive(Clone, Copy)]
pub struct Transition { pub from: StateId, pub to: StateId }
/// The semantic model every engine consumes. Built once by mc-lang::compile.
/// Closures are Fn (pure, reentrant) -> usable from a single-threaded DFS
/// without &mut; capture the StateEncoder by Arc/Rc. No Send/Sync required.
pub struct TransitionSystem {
    pub initial: Vec<State>,                       // 1+ initial states
    pub next:  Box<dyn Fn(&State) -> Vec<State>>,  // successor (next-state) fn
    pub label: Box<dyn Fn(&State) -> Label>,       // proposition labeling
    pub props: Vec<AtomicProp>,                    // declared proposition vocab
}
/// Explicit finite graph form: used by explicit CTL fixpoint labeling (## 19)
/// where the reachable set is materialized. Built by reachable() snapshotting.
pub struct KripkeStructure {
    pub states: Vec<State>,            // index = implicit StateId
    pub trans:  Vec<Transition>,       // explicit successor edges
    pub labels: Vec<Label>,            // labels[i] = label of states[i]
}

/// Open-addressing hash SET of visited State -> StateId. THE dominant memory
/// cost of explicit search. Linear probing, power-of-two capacity, Robin-Hood
/// optional; load factor <= 0.75 then grow x2 + rehash. No deletes (monotone).
pub struct StateTable { /* keys: Vec<Option<State>>, ids assigned on insert */ }

impl StateTable {
    pub fn new() -> Self;
    /// Returns (id, is_new). is_new=false => already visited (cycle/merge point).
    pub fn intern(&mut self, s: &State) -> (StateId, bool);
    pub fn contains(&self, s: &State) -> bool;
    pub fn len(&self) -> usize;        // # distinct visited states
}
/// Counterexample. Pure prefix (safety) => cycle=None. Lasso (liveness) =>
/// cycle=Some: states repeat prefix.last() .. and loop. Replayed by `simulate`.
pub struct Trace {
    pub prefix: Vec<State>,        // path from an initial state
    pub cycle:  Option<Vec<State>> // Some => accepting cycle (omega = prefix.(cycle)^w)
}
/// Guarded-command statement AST (Promela/NuSMV-flavored). #[non_exhaustive].
pub enum Stmt {
    Assign(Var, Expr),     // v := e
    Guard(Expr),           // [g] enabling condition
    Choice(Vec<Stmt>),     // nondeterministic alternative (::)
    Seq(Vec<Stmt>),        // sequential composition
    // .. (atomic, skip, goto: added in ## 13)
}

/// A model module: variable decls, init predicate, guarded transitions, specs.
pub struct Module {
    pub vars:  Vec<VarDecl>,    // typed/ranged variables -> bit budget
    pub init:  Expr,            // initial-state predicate
    pub trans: Vec<Guarded>,    // guarded commands (guard -> update)
    pub specs: Vec<Spec>,       // LTL/CTL properties to check
}

pub enum Spec { Ltl(LtlFormula), Ctl(CtlFormula), Invariant(LtlFormula) }

pub fn parse_model<R: Read>(r: R) -> Result<Module, ParseError>;

/// Maps each variable to a bit range; encode/decode State <-> valuation.
/// Owns the var->[lo,hi) layout that defines State::bits canonical form.
pub struct StateEncoder { /* var_ranges: Vec<(VarId, Range<u32>)>, width: u32 */ }
impl StateEncoder {
    pub fn width(&self) -> u32;                      // total bits
    pub fn encode(&self, val: &Valuation) -> State;  // pack
    pub fn decode(&self, s: &State) -> Valuation;    // unpack (for traces/sim)
}

/// Lowers a Module into the semantic TransitionSystem the engines consume.
/// next() = enabled guarded commands applied; label() = eval AtomicProps.
pub fn compile(m: &Module) -> TransitionSystem;
/// LTL syntax tree. NNF rewriter pushes Not to literals (Release dual of Until).
#[non_exhaustive]
pub enum LtlFormula {
    True, False, Ap(AtomicProp), Not(Box<LtlFormula>),
    And(Box<LtlFormula>, Box<LtlFormula>), Or(Box<LtlFormula>, Box<LtlFormula>),
    Next(Box<LtlFormula>),                                  // X
    Until(Box<LtlFormula>, Box<LtlFormula>),                // U
    Release(Box<LtlFormula>, Box<LtlFormula>),              // R
    Finally(Box<LtlFormula>), Globally(Box<LtlFormula>),    // F / G
}

/// Negation normal form: Not only on Ap; F p = true U p, G p = false R p.
pub fn to_nnf(f: &LtlFormula) -> LtlFormula;

/// (Generalized->degeneralized) Buchi automaton over Label guards.
pub struct BuchiAutomaton {
    pub states:    Vec<BAState>,
    pub init:      Vec<usize>,                 // initial state indices
    pub edges:     Vec<(usize, Label, usize)>, // (from, guard, to)
    pub accepting: BitSet,                     // accepting state indices
}

/// Tableau / GPVW construction. Caller passes the negated property to get
/// Buchi(!phi); a reachable accepting cycle in the product = CEX to phi.
pub fn ltl_to_buchi(f: &LtlFormula) -> BuchiAutomaton;
/// On-the-fly reachable-state count (BFS/DFS over next()), interning into table.
pub fn reachable(ts: &TransitionSystem, table: &mut StateTable) -> usize;

/// Safety: BFS for a SHORTEST counterexample. Ok(()) => invariant holds on all
/// reachable states; Err(Trace) => prefix to a violating state (cycle=None).
pub fn check_invariant(ts: &TransitionSystem, inv: &LtlFormula) -> Result<(), Trace>;

/// Synchronous product TransitionSystem x BuchiAutomaton(!phi). On-the-fly:
/// product states = (State, ba_state); generated lazily during nested_dfs.
pub struct Product { /* ts: &TransitionSystem, ba: &BuchiAutomaton */ }
impl Product {
    pub fn new<'a>(ts: &'a TransitionSystem, ba: &'a BuchiAutomaton) -> Product /* <'a> */;
}

/// Nested DFS accepting-cycle detection (Courcoubetis et al). Outer DFS finds
/// accepting states post-order; inner DFS seeks a path back to the outer stack.
/// Some(lasso) => language non-empty => phi VIOLATED. None => phi holds.
/// O(states+transitions); 2 color bits/state in a StateTable-keyed map.
pub fn nested_dfs(prod: &Product) -> Option<Trace>;
/// Complemented edge into the node table. High bit = complement flag; index in
/// low bits. Halves the table (f and !f share a node). NOT a Box -> just a u32.
#[derive(Clone, Copy, PartialEq, Eq, Hash)]
pub struct BddRef(pub u32);

/// A reduced node: top variable + lo(else)/hi(then) children. Interned in the
/// unique table -> structural sharing; never two equal nodes (canonicity).
#[derive(Clone, Copy, PartialEq, Eq, Hash)]
pub struct BddNode { pub var: u32, pub lo: BddRef, pub hi: BddRef }

/// Owns ALL BDD memory: unique table (de-dup), computed cache (memoize ite),
/// variable order. Single owner per symbolic check; &mut for every op (it may
/// allocate nodes / grow the cache). Not Send/Sync.
pub struct BddManager { /* unique: HashMap<BddNode,BddRef>, cache, var_order: Vec<u32> */ }

impl BddManager {
    pub fn var(&mut self, i: u32) -> BddRef;                       // projection x_i
    pub fn ite(&mut self, f: BddRef, g: BddRef, h: BddRef) -> BddRef; // universal apply
    pub fn and(&mut self, a: BddRef, b: BddRef) -> BddRef;         // = ite(a,b,0)
    pub fn or (&mut self, a: BddRef, b: BddRef) -> BddRef;         // = ite(a,1,b)
    pub fn not(&self,     a: BddRef) -> BddRef;                    // flip complement bit (O(1))
    pub fn restrict(&mut self, f: BddRef, var: u32, val: bool) -> BddRef;
    pub fn exists(&mut self, vars: &BddRef, f: BddRef) -> BddRef;  // existential quant (image)
}
/// CTL syntax tree. #[non_exhaustive]. Mix of E/A path quantifiers x X/F/G/U.
#[non_exhaustive]
pub enum CtlFormula {
    Ap(AtomicProp), Not(Box<CtlFormula>),
    And(Box<CtlFormula>, Box<CtlFormula>), Or(Box<CtlFormula>, Box<CtlFormula>),
    Ex(Box<CtlFormula>), Eg(Box<CtlFormula>),
    Eu(Box<CtlFormula>, Box<CtlFormula>),                  // E[p U q]
    Ax(Box<CtlFormula>), Af(Box<CtlFormula>),
    Ag(Box<CtlFormula>), Au(Box<CtlFormula>, Box<CtlFormula>),
}

/// Transition relation T(s, s') as ONE BDD over current x next (primed) vars.
pub struct TransitionRelation { pub bdd: BddRef /* over current ⊎ next vars */ }

/// EX: pre_image(set) = { s | exists s'. T(s,s') AND set(s') }. Renames primed
/// vars back; the kernel of every CTL fixpoint.
pub fn pre_image(mgr: &mut BddManager, tr: &TransitionRelation, set: BddRef) -> BddRef;

/// Returns the BDD of the SAT-SET (states where f holds). EG = greatest fixpoint
/// of (f AND EX Z); E[pUq] = least fixpoint of (q OR (p AND EX Z)). Sets, never
/// enumerated states -> 10^20 states tractable when the BDD is small.
pub fn check_ctl(mgr: &mut BddManager, tr: &TransitionRelation, f: &CtlFormula) -> BddRef;

/// Bounded model checking: unroll I(s0) AND /\_{i<k} T(s_i,s_{i+1}) AND
/// bounded(!phi) over s0..sk into CNF; solve via the project-277 SAT engine.
/// Some(Trace) of length k => CEX; None => no CEX within bound k (not a proof
/// unless k >= completeness threshold).
pub fn bmc(ts: &TransitionSystem, phi: &LtlFormula, k: usize) -> Option<Trace>;

                                          │ Fn next/label (pure, Rc<StateEncoder>)
 mc-explicit owns: StateTable, color maps, Product<'a>(&ts,&ba) ──> Trace
 mc-symbolic owns: BddManager (unique table + cache + var_order) ──> BddRef sat-set
 mc-bmc:           CNF ──> project-277 SAT engine ──> Trace
 Outputs: Trace (RAM, printed by `simulate`) · counts (usize) · BddRef (manager-local)
 Persisted: nothing at runtime. serde behind `dump` feature: Module + Trace only.
