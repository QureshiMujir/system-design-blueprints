Core schema & types (§11)

// --- secret newtypes (all: ZeroizeOnDrop, no Debug-of-bytes, no cleartext Serialize) ---
struct RootKey([u8;32]);          // zeroizing
struct ChainKey([u8;32]);         // zeroizing
struct MessageKey([u8;32]);       // zeroizing
struct AeadKey([u8;32]);          // zeroizing, transient
struct AeadNonce([u8;12]);        // transient

struct X25519Public([u8;32]);     // public, Copy ok
struct X25519Secret([u8;32]);     // zeroizing, clamped on construction
struct X25519KeyPair { public: X25519Public, secret: X25519Secret }   // zeroizing secret

struct IdentityKeyPair(X25519KeyPair);        // long-term; XEdDSA-capable
struct SignedPreKey { kp: X25519KeyPair, id: u32, sig: XEdDSASig }  // sig over public
struct OneTimePreKey { kp: X25519KeyPair, id: u32 }                 // single-use
struct EphemeralKey(X25519KeyPair);           // per-session initiator key
struct DhKeyPair(X25519KeyPair);              // ratchet keypair alias
type   DhPublic = X25519Public;
struct XEdDSASig([u8;64]);                    // public
struct PreKeyBundle {
  reg_id: u32, ik_b: X25519Public,
  spk_id: u32, spk_b: X25519Public, sig_b: XEdDSASig,
  opk: Option<(u32, X25519Public)>,
}

struct AssociatedData([u8;64]);   // Encode(IK_A) ‖ Encode(IK_B); public, bound into AEAD

struct MsgHeader { dh_pub: DhPublic, pn: u32, n: u32 }   // public

struct RatchetMessage { header: MsgHeader, ciphertext: Vec<u8> }  // ciphertext incl. 16B tag

struct X3dhInitialMessage {
  ik_a: X25519Public, ek_a: X25519Public,
  spk_id: u32, opk_id: Option<u32>,
  body: RatchetMessage,
}

struct HeaderKeys { hks: ChainKey, hkr: ChainKey, nhks: ChainKey, nhkr: ChainKey }  // HE; zeroizing
struct SkippedKeyId { dhr: X25519Public, n: u32 }   // Ord for BTreeMap eviction

struct SessionState {
  rk: RootKey,
  dhs: DhKeyPair,
  dhr: Option<DhPublic>,
  cks: Option<ChainKey>,
  ckr: Option<ChainKey>,
  ns: u32, nr: u32, pn: u32,
  mkskipped: BTreeMap<SkippedKeyId, MessageKey>,   // bounded by MAX_SKIP + global cap
  hdr_keys: Option<HeaderKeys>,                    // Some(..) iff HE enabled
  wire_version: u16,
}

const MAX_SKIP: u32 = 1000;
const SKIP_GLOBAL_CAP: usize = 2000;
trait Aead {
  fn seal(&self, key:&AeadKey, nonce:&AeadNonce, pt:&[u8], ad:&[u8]) -> Vec<u8>;     // ct‖tag
  fn open(&self, key:&AeadKey, nonce:&AeadNonce, ct:&[u8], ad:&[u8]) -> Result<Vec<u8>, Error>; // ct-time tag
}
trait Kdf {
  fn rk(&self, rk:&RootKey, dh:&[u8;32]) -> (RootKey, ChainKey);          // KDF_RK
  fn ck(&self, ck:&ChainKey) -> (ChainKey, MessageKey);                   // KDF_CK
  fn expand_mk(&self, mk:&MessageKey) -> (AeadKey, AeadNonce);            // EXPAND_MK
}
trait Dh    { fn generate(&self) -> X25519KeyPair; fn agree(&self, s:&X25519Secret, p:&X25519Public) -> [u8;32]; }
trait Sign  { fn xeddsa_sign(&self, ik:&IdentityKeyPair, msg:&[u8]) -> XEdDSASig;
              fn xeddsa_verify(&self, ik:&X25519Public, msg:&[u8], sig:&XEdDSASig) -> bool; }
trait Rng   { fn fill(&self, out:&mut [u8]); }     // CSPRNG; panics→Err on entropy failure
trait PreKeyServer {
  fn publish(&mut self, b: PublishedBundle);
  fn fetch(&mut self, reg_id: u32) -> Result<PreKeyBundle, Error>;   // pops one OPK atomically
}
enum Error {
  AuthFailed,          // AEAD tag mismatch — fail closed, no state change (I10)
  SkipLimitExceeded,   // (N - Nr) > MAX_SKIP or global cap (I3)
  BadHeader,           // malformed/oversized wire field (I13)
  NoSession,           // op on Uninitialized/missing session
  PreKeyConsumed,      // OPK reused (I9)
  Replay,              // N already consumed, no skipped key
  BadSignature,        // SPK signature verify failed (I8)
  EntropyFailure,      // CSPRNG draw failed
  StoreCorruption,     // persisted state failed integrity check
  VersionUnsupported,  // wire/protocol version not negotiable
}
