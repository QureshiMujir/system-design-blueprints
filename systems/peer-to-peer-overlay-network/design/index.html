Core schema & types (§11)

pub struct NodeId([u8; 32]);                 // SHA-256 (#88); also used for Key
pub type Key = NodeId;
pub struct Distance([u8; 32]);               // metric-dependent magnitude (XOR result, ring gap, …)

pub trait Metric {                           // I2: valid metric axioms
    fn distance(a: &NodeId, b: &NodeId) -> Distance;       // XOR | ring | prefix
    fn closer(target: &NodeId, x: &NodeId, y: &NodeId) -> std::cmp::Ordering;
}
pub struct Xor;     // a ⊕ b            (Kademlia, default)
pub struct Ring;    // (b - a) mod 2^256 (Chord)
pub struct Prefix;  // shared-prefix len (Pastry)

pub trait RoutingScheme {                    // ADR-4: Kademlia | Chord | Pastry interchangeable
    fn observe(&mut self, n: NodeInfo);                    // ingest a seen node (I5)
    fn closest(&self, target: &NodeId, count: usize) -> Vec<NodeInfo>;  // table query
    fn on_lookup_response(&mut self, from: &NodeId, contacts: &[NodeInfo]);
}
pub struct NodeInfo { id: NodeId, addr: SocketAddr, last_seen: Instant, rtt: Option<Duration> }
pub struct KBucket { entries: ArrayVec<NodeInfo, K>, replacements: ArrayVec<NodeInfo, K> }  // K = 20
pub struct KademliaTable { local: NodeId, buckets: [KBucket; 256] }    // bucket i = MSB-differing bit
impl KademliaTable {
    fn insert(&mut self, n: NodeInfo);                     // I4: LRU-evict preferring live-old
    fn closest(&self, target: &NodeId, k: usize) -> Vec<NodeInfo>;
    fn needs_refresh(&self, now: Instant) -> Vec<usize>;   // buckets idle > t_refresh (P6)
}
pub struct ChordTable { local: NodeId, finger: [Option<NodeInfo>; 256], successors: Vec<NodeInfo>, predecessor: Option<NodeInfo> }
impl ChordTable {
    fn closest_preceding(&self, target: &NodeId) -> NodeInfo;   // next hop (§8.6)
    fn stabilize_step(&mut self) -> StabilizeAction;            // I6 (P7)
}
pub struct PastryTable { local: NodeId, rows: Vec<Vec<Option<NodeInfo>>>, leaf_set: Vec<NodeInfo>, neighborhood: Vec<NodeInfo> }
pub struct Shortlist { target: NodeId, nodes: BTreeSet<DistOrd>, queried: HashSet<NodeId>, k: usize }
pub enum LookupKind { FindNode, FindValue }
pub struct Lookup { kind: LookupKind, shortlist: Shortlist, outstanding: usize /* ≤ α */, alpha: usize }
pub enum LookupOutcome { Nodes(Vec<NodeInfo>), Value(Value), TimedOut(Vec<NodeInfo>) }   // I3/I7/I8
impl Lookup {
    fn next_queries(&mut self) -> Vec<NodeInfo>;           // α closest un-queried (≤ I15)
    fn ingest(&mut self, from:&NodeId, contacts:&[NodeInfo]) -> bool;  // returns "made progress"
    fn converged(&self) -> bool;                            // I7: k-closest all queried, no closer
}
pub fn disjoint_lookup(target:&NodeId, d:usize) -> Vec<NodeInfo>;   // S/Kademlia (P9, I14)
pub enum Rpc { Ping, Store{key:Key, value:Value, ttl:Duration}, FindNode{target:NodeId}, FindValue{key:Key} }
pub enum RpcResp { Pong, StoreAck, Nodes(Vec<NodeInfo>), Value(Value), NodesOrValue(...) }
pub struct Frame<'a> { version:u8, kind:u8, txn_id:u64, sender:NodeId, payload:&'a [u8], checksum:u32 }  // I11
pub trait Transport {
    async fn call(&self, peer:&NodeInfo, rpc:Rpc, timeout:Duration) -> Result<RpcResp, P2pError>;  // I10
    async fn call_parallel(&self, peers:&[NodeInfo], rpc:Rpc, alpha:usize) -> Vec<Result<RpcResp,P2pError>>; // α
}
pub struct Value { bytes: Vec<u8>, seq: u64, integrity: [u8;32] }   // integrity = SHA-256(bytes) (#88/#71, I10)
pub struct StoredValue { value: Value, expires_at: Instant, key: Key }
pub trait DhtStore {
    fn store(&self, key:Key, value:Value, ttl:Duration) -> Result<usize/*replicas*/, P2pError>;  // I9
    fn get(&self, key:Key) -> Result<Option<Value>, P2pError>;                                    // I8
    fn republish_due(&self, now:Instant) -> Vec<Key>;                                             // P6
}
pub struct NodeConfig { id_mode: IdMode, k: usize, alpha: usize, rpc_timeout: Duration, max_retries: u32,
                        t_refresh: Duration, t_republish: Duration, t_expire: Duration,
                        scheme: SchemeKind, secure: bool, bootstrap: Vec<SocketAddr> }
pub enum IdMode { Random, HashOfAddr{nonce_bits:u32} }     // I13 (secure Sybil mode)
pub enum SchemeKind { Kademlia, Chord, Pastry }            // ADR-3/4
pub struct Node { id: NodeId, table: Arc<Mutex<dyn RoutingScheme>>, store: Arc<dyn DhtStore>, transport: Arc<dyn Transport> }
impl Node {
    async fn join(&self, bootstrap:&[SocketAddr]) -> Result<(), P2pError>;     // P5 (self-lookup, §8.15)
    async fn lookup(&self, target:&NodeId) -> Result<Vec<NodeInfo>, P2pError>; // P3
    async fn store(&self, key:Key, value:Vec<u8>) -> Result<usize, P2pError>;  // P4
    async fn get(&self, key:Key) -> Result<Option<Vec<u8>>, P2pError>;         // P4
    async fn leave(&self) -> Result<(), P2pError>;                             // P6 (graceful)
}
pub struct SimNetwork { nodes: HashMap<NodeId, SimNode>, queue: BinaryHeap<TimedMsg>, rng: StdRng,
                        loss: f64, delay: DelayModel }     // I12 deterministic
impl SimNetwork {
    fn step(&mut self) -> Option<SimEvent>;                // pop next timed message, deliver (or drop)
    fn run_until_quiescent(&mut self) -> SimTrace;         // deterministic trace (I12)
    fn churn(&mut self, join_rate:f64, leave_rate:f64);    // inject membership change (P6 test)
    fn true_closest(&self, target:&NodeId, k:usize) -> Vec<NodeId>;   // the ORACLE (I3, principle 14)
}
pub enum P2pError {
    Timeout(NodeInfo), BadFrame{reason:&'static str}, ProtoVersion{ours:u8,theirs:u8},
    InvalidId, ValueIntegrity{key:Key}, NotFound(Key), Underreplicated{key:Key, have:usize, want:usize},
    BootstrapFailed, NoRouteToTarget, IdConstraintFail, RateLimited(SocketAddr), TransportClosed }
pub struct LookupTrace { target:NodeId, kind:&'static str, scheme:&'static str, hops:u32, total_rpcs:u32,
                         alpha:usize, converged:bool, found_value:bool, rounds:u32, p50_rtt_ms:f64, duration_ms:f64 }
pub struct StoreTrace { key:Key, replicas:u32, want:u32, hops:u32, bytes:usize }
pub struct TableTrace { scheme:&'static str, populated_buckets:u32, total_contacts:u32, evictions:u64,
                        refreshes:u64, prefix_skew:f64 /*I14*/ }
pub struct ChurnTrace { joins:u64, leaves:u64, detected_dead:u64, repairs:u64, lookups_failed:u64, availability:f64 }
