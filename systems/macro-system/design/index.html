Core schema & types (§11)

// crates/ms-tokens/src/lib.rs — signatures only, no bodies
pub struct Token { pub kind: TokenKind, pub text: SmolStr, pub span: Span, pub scopes: ScopeSet }
pub enum TokenKind { Ident, Punct(Punct), Literal(LitKind), Group(Delim), Whitespace, Comment }
pub struct ScopeSet(BTreeSet<ScopeId>);          // ordered: source-order determinism (P9)
pub struct ScopeId(pub u64);                      // monotonic (I2)

impl ScopeSet {
    pub fn empty() -> Self;
    pub fn add(&self, s: ScopeId) -> Self;        // I3: append-only within a step
    pub fn remove(&self, s: ScopeId) -> Self;
    pub fn flip(&self, s: ScopeId) -> Self;
    pub fn is_subset_of(&self, other: &ScopeSet) -> bool; // resolver subset check (I5)
    pub fn contains(&self, s: ScopeId) -> bool;
}
// crates/ms-ast/src/lib.rs
pub enum Item { Fn(Function), Struct(Struct), Enum(Enum), MacroDef(MacroDef), Use(UsePath) }
pub struct MacroDef { pub name: Ident, pub kind: MacroKind, pub def_site: ScopeSet, pub span: Span }
pub enum MacroKind { Pattern { arms: Vec<MacroArm> }, Procedural { entry: ProcMacroEntry } }
pub struct MacroArm { pub pattern: PatternTree, pub template: TemplateTree }
pub struct Expr { pub kind: ExprKind, pub span: Span, pub scopes: ScopeSet }
pub enum ExprKind { Path(Path), Lit(Lit), Block(Block), Call(Box<Expr>, Vec<Expr>), MacroInvocation(MacroInvocation) }
pub struct MacroInvocation { pub path: Path, pub args: TokenStream, pub span: Span, pub depth: u32 }
// crates/ms-patterns/src/pattern.rs
pub enum PatternTree { Token(Token), MetaVar { name: SmolStr, kind: FragmentKind, span: Span },
    Repeat { inner: Box<PatternTree>, sep: Option<Token>, quantifier: Quantifier },
    Group { delim: Delim, items: Vec<PatternTree> } }
pub enum FragmentKind { Ident, Expr, Pat, Ty, Path, Tt, Block, Stmt, Literal }
pub enum Quantifier { ZeroOrMore, OneOrMore, ZeroOrOne }
pub trait MacroTransformer {                      // P2: one contract, two kinds
    fn try_expand(&self, invocation: &MacroInvocation, ctx: &ExpansionCtx) -> Result<TokenStream, ExpandError>;
}
// crates/ms-expander/src/lib.rs
pub struct Expander { pub scope_counter: u64, pub iter_cap: u32, pub depth_cap: u32, pub seed: u64 }
pub struct FixpointDriver<'a> { pub expander: &'a mut Expander, pub iterations: u32 }
impl Expander {
    pub fn expand(&mut self, ast: &mut Item) -> Result<(), ExpandError>;
    fn fresh_scope(&mut self) -> ScopeId;          // source-order (P9, I2)
    fn try_expand_invocation(&mut self, inv: &MacroInvocation) -> Result<TokenStream, ExpandError>;
}
impl<'a> FixpointDriver<'a> {
    pub fn drive(&mut self, ast: &mut Item) -> Result<(), ExpandError>; // I4 cap check
}
// crates/ms-resolver/src/lib.rs
pub struct Resolver { pub bindings: Vec<Binding> }
pub struct Binding { pub name: SmolStr, pub scopes: ScopeSet, pub kind: BindingKind, pub id: DefId }
pub enum BindingKind { Local(LocalId), Global(GlobalId), MacroDef(MacroDefId) }
impl Resolver {
    pub fn resolve(&self, ident: &Ident) -> Result<&Binding, ResolveError>; // I5: no silent pick
}
// crates/ms-diagnostics/src/lib.rs and crates/ms-modules/src/lib.rs
pub struct Diagnostic { pub primary: Span, pub definition_site: Option<Span>,
    pub trace: Vec<Span>, pub message: String, pub suggestion: Option<Suggestion> } // I9
pub struct Suggestion { pub replacement: TokenStream, pub reexpand_checked: bool }    // I13
pub struct ModuleGraph { pub macros: HashMap<MacroDefId, MacroDef> }
pub struct Import { pub path: Path, pub macro_id: MacroDefId, pub def_site: ScopeSet } // I6
// crates/ms-procmacro/src/lib.rs
pub struct ProcMacroSandbox { engine: wasmtime::Engine, store: wasmtime::Store<SandboxCtx> }
pub struct SandboxCtx { pub allowed_capabilities: AllowedCaps }
pub struct AllowedCaps { pub read_files: bool, pub spawn_process: bool, pub network: bool,
    pub clock: bool, pub random: bool }             // all default false (I11)
impl ProcMacroSandbox {
    pub fn invoke(&mut self, macro_id: ProcMacroId, input: TokenStream, timeout: Duration)
        -> Result<TokenStream, SandboxError>;        // hermetic per I7
}
