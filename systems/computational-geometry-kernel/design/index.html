Core schema & types (§11)

// ---- number layer (cgk-num) ----
type F = f64;
struct Expansion { /* non-overlapping, magnitude-sorted f64 components (I15) */ }
struct Interval { lo: f64, hi: f64 }                 // directed-rounding bounds
enum  Rational { /* bignum num/den; PhaseDeferred("§13") */ }

fn fast_two_sum(a: F, b: F) -> (F, F);               // requires |a|>=|b|; (sum, exact_err)
fn two_sum(a: F, b: F) -> (F, F);                    // general (Knuth)
fn split(a: F) -> (F, F);                            // Veltkamp 26+26
fn two_product(a: F, b: F) -> (F, F);                // (prod, exact_err); FMA variant available
fn expansion_sum(e: &Expansion, f: &Expansion) -> Expansion;
fn scale_expansion(e: &Expansion, b: F) -> Expansion;
fn expansion_sign(e: &Expansion) -> Sign;

// ---- predicate layer (cgk-predicates) ----
enum Sign { Negative, Zero, Positive }               // Zero only INTERNALLY pre-SoS
fn orient2d(a: Point, b: Point, c: Point) -> Sign;           // staged adaptive (I1)
fn incircle(a: Point, b: Point, c: Point, d: Point) -> Sign; // a,b,c CCW
fn orient3d(a: Pt3, b: Pt3, c: Pt3, d: Pt3) -> Sign;
fn insphere(a: Pt3, b: Pt3, c: Pt3, d: Pt3, e: Pt3) -> Sign;
fn orient2d_sos(a: Point, b: Point, c: Point, ia: u32, ib: u32, ic: u32) -> SignNZ; // never Zero (I12)
// SignNZ = { Negative, Positive }   (SoS guarantees non-zero)

// ---- kernel (cgk-kernel) ----
struct Point { x: f64, y: f64 }                      // 2D site/vertex coordinate (input)
struct Pt3   { x: f64, y: f64, z: f64 }
struct Vector { x: f64, y: f64 }
struct Segment { a: Point, b: Point }
struct Triangle { a: Point, b: Point, c: Point }
struct Bbox { min: Point, max: Point }
struct Constructed(Point);                           // a coordinate produced by a construction (I18)

trait Kernel { type NT; /* predicates + constructions over NT */ }
fn intersection_point(s: Segment, t: Segment) -> Option<Constructed>;  // construction (inexact)
fn circumcenter(t: Triangle) -> Constructed;                            // Voronoi vertex
fn signed_area(ring: &[Point]) -> f64;               // shoelace; sign = winding
fn point_in_polygon(p: Point, ring: &[Point]) -> Containment;          // ray cast + winding
enum Containment { Inside, Outside, Boundary }
// ---- topology (cgk-dcel) ----
type VId = u32; type HId = u32; type FId = u32;      // arena index handles (ADR-11)
struct HalfEdge { origin: VId, twin: HId, next: HId, prev: HId, face: FId }
struct Vertex { coord: Point, incident: HId }
struct Face { boundary: HId }                        // one bounding half-edge
struct Dcel { verts: Vec<Vertex>, edges: Vec<HalfEdge>, faces: Vec<Face> }
struct QuadEdge { /* e[0..3]: {next: HId, data: VId|FId} (I13) */ }

// ---- algorithm outputs ----
struct Hull { verts: Vec<Point> }                    // CCW, collinearity per policy (I6)
struct Triangulation { dcel: Dcel /* triangles via faces */ }
struct VoronoiDiagram { dcel: Dcel /* cells/edges/vertices */ }
struct Arrangement { dcel: Dcel, intersections: Vec<Constructed> }
enum BoolOp { Union, Intersection, Difference, SymmetricDifference }

// ---- errors ----
enum ParseError { NonFinite, Magnitude, Truncated{need:usize,have:usize},
                  UnclosedRing, BadWkb, BadWkt }
enum GeometryError { SelfIntersecting, Degenerate, NotSimple, EmptyInput }
