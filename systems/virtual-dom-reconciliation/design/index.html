Core schema & types (§11)

type VNode<T = unknown> = {
  $$typeof: symbol;
  type: string | Component<T> | ClassComponent<T> | Sentinel;
  key: string | number | null;
  ref: RefObject<unknown> | RefCallback<unknown> | null;
  props: Readonly<T & { children?: VNodeChild }>;
};
// own: caller (created fresh per render, P1). mut: none (frozen). concurrency: read-only,
// shareable across lanes. persist: none (transient, GC'd post-commit). serialize: no (SSR
// serializes host output, not VNode itself). compat: `$$typeof` brand is a stability
// boundary — changing it is a breaking ADR.

type Component<P> = (props: P, ctx?: ComponentContext) => VNodeChild;
// own: caller. mut: n/a (pure contract, P1). concurrency: may run 0..N times per commit
// (P9). persist: none. compat: adding an optional ctx param is non-breaking.

interface ClassComponent<P, S = unknown> {
  render(): VNodeChild;
  componentDidMount?(): void;
  componentDidUpdate?(prevProps: P, prevState: S): void;
  componentWillUnmount?(): void;
  getDerivedStateFromError?(error: unknown): Partial<S> | null;
  componentDidCatch?(error: unknown, info: ErrorInfo): void;
}
// own: instance owned by its fiber (stateNode). mut: instance methods may mutate `this`.
// concurrency: instantiated once per mount; not re-entrant. persist: instance lifetime =
// fiber lifetime. compat: legacy-only surface (ADR-20 predates hooks; retained for compat).

interface Root {
  render(vnode: VNode): void;
  unmount(): void;
}
// own: caller of createRoot/hydrateRoot. mut: internal fiber tree mutates on render() calls.
// concurrency: single-owner-thread (P15). persist: lives until unmount(). compat: stable.

interface HostAdapter<Node, TextNode> {
  createElement(type: string, props: unknown): Node;
  createTextNode(text: string): TextNode;
  insertBefore(parent: Node, child: Node | TextNode, before: Node | TextNode | null): void;
  removeChild(parent: Node, child: Node | TextNode): void;
  commitUpdate(node: Node, oldProps: unknown, newProps: unknown): void;
  setEventListener(node: Node, type: string, listener: EventListener | null): void;
  hydrate(node: Node, vnode: VNode): HydrateResult;
  flush(): void;
}
// own: implementation (DOM backend / test renderer / deferred native). mut: every method is
// a mutation entry point, called only from commitWork (P4, I4). concurrency: called only on
// the root's owning thread. persist: n/a. compat: adding a method is breaking (all
// implementations must add it); prefer default no-op methods for additive change.

interface Fiber {
  type: VNode["type"]; key: VNode["key"]; tag: FiberTag;
  pendingProps: unknown; memoizedProps: unknown; memoizedState: Hook | null;
  updateQueue: UpdateQueue | null;
  child: Fiber | null; sibling: Fiber | null; return: Fiber | null;
  alternate: Fiber | null; stateNode: unknown;
  flags: FiberFlags; subtreeFlags: FiberFlags; deletions: Fiber[] | null;
  lanes: Lanes; childLanes: Lanes;
}
// own: reconciler (one current + one workInProgress per VNode, paired via alternate, §14.1).
// mut: written only during beginWork/completeWork/commitWork of its own lane. concurrency:
// the workInProgress half is exclusive to the in-flight render (P9); the current half is
// read-only during that window (I14 §7). persist: current fiber persists across commits;
// workInProgress is discarded or promoted to current at commit. serialize: never (internal).

interface Hook<S = unknown> {
  kind: "state" | "reducer" | "effect" | "layoutEffect" | "memo" | "callback" | "ref"
      | "context" | "id" | "transition" | "deferredValue" | "syncExternalStore"
      | "insertionEffect" | "imperativeHandle" | "debugValue";
  memoizedState: S; next: Hook | null;
}
// own: the fiber's hook list (per-fiber, positional, P3). mut: update dispatcher rewrites
// memoizedState in place on the workInProgress hook; mount dispatcher allocates. concurrency:
// exclusive to the fiber's current render. persist: lifetime = fiber lifetime.

type Lanes = number; // 31-bit mask: SyncLane | InputContinuousLane | DefaultLane
                      // | TransitionLane(×16) | RetryLane(×4) | SelectiveHydrationLane
                      // | IdleLane | OffscreenLane  (§17.1, ADR-08)
// own: scheduler + per-fiber `lanes`/`childLanes`. mut: OR'd in on update enqueue, cleared
// on commit. concurrency: read by any lane's render pass; mutated only under the scheduler's
// single-threaded tick. compat: adding a lane bit is non-breaking below bit 31.

interface Scheduler {
  scheduleWork(root: Root, lane: Lanes, work: () => void): void;
  flushSync(): void; now(): number; shouldYield(): boolean; setSliceMs(ms: number): void;
}
// own: process-wide singleton (P14 — shared, not global mutable state; coordinates via
// message-passing). concurrency: single active tick at a time; cross-root fairness via
// per-root lane queues. persist: process lifetime.

interface SyntheticEvent<N = Event> {
  nativeEvent: N; target: EventTarget; currentTarget: EventTarget;
  phase: "capture" | "target" | "bubble";
  isPropagationStopped(): boolean; stopPropagation(): void; preventDefault(): void;
}
// own: allocated per dispatch by the responsible event plugin (§18.4); not pooled (ADR
// note, no pooling since React 17 precedent). mut: propagation/prevented flags mutate during
// dispatch walk. concurrency: single dispatch is synchronous end-to-end (SyncLane, §17.1).

interface SSROptions {
  identifierPrefix?: string; signal?: AbortSignal;
  onShellReady?(): void; onShellError?(error: unknown): void; onAllReady?(): void;
  onError?(error: unknown): void; bootstrapScripts?: string[];
}
interface HydrateOptions {
  identifierPrefix?: string;
  onRecoverableError?(error: unknown): void;
  onCaughtError?(error: unknown): void; onUncaughtError?(error: unknown): void;
}
// own: caller-supplied per call (SSR) / per hydrateRoot call. mut: none (options are
// read-only config). serialize: SSROptions never serialized; identifierPrefix is embedded
// in emitted useId() strings for CSR/SSR determinism (I13 §7).
