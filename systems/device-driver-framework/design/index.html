Core schema & types (§11)

// Ownership: framework singleton per bus instance, Arc<dyn Bus>.
// Mutation: interior (RwLock<BTreeMap>) for children — deterministic order (P12).
// Concurrency: Send + Sync; probe() may run from multiple bus-scan threads.
// Persistence: none — rebuilt every boot from firmware enumeration.
pub trait Bus: Send + Sync {
    fn name(&self) -> &str;
    fn probe(&self, child: &Device) -> ProbeResult;
    fn match_drivers(&self, device: &Device, drivers: &[Arc<dyn Driver>]) -> Option<Arc<dyn Driver>>;
    fn iommu_group(&self, device: &Device) -> Option<IommuGroupId>;
}

// Ownership: Arc<Device>, refcounted (I3); freed only at refcount 0.
// Mutation: state via AtomicState (I1); driver binding via RwLock.
// Concurrency: Send + Sync; children iterated in BTreeMap order (P12).
// Persistence: none; DeviceProperties are in-memory TLV only (ADR-04/05).
// Compatibility: DeviceId is stable across probe/unbind/reprobe of the same slot.
pub struct Device {
    pub id: DeviceId,
    pub bus: Weak<dyn Bus>,
    pub parent: Option<Weak<Device>>,
    pub children: RwLock<BTreeMap<DeviceId, Arc<Device>>>,
    pub resources: ResourceSet,
    pub state: AtomicState,
    pub driver: RwLock<Option<Arc<dyn Driver>>>,
    pub properties: DeviceProperties,
    pub refcount: RefCount,
}

// Ownership: registered once at module init; Arc<dyn Driver> shared across devices.
// Mutation: none — Driver is stateless; per-device state lives in Device.
// Concurrency: Send + Sync; probe/remove/suspend/resume serialized per-device (I1).
pub trait Driver: Send + Sync {
    fn name(&self) -> &str;
    fn priority(&self) -> DriverPriority;
    fn probe(&self, device: &Device) -> Result<ProbeOutcome, ProbeError>;
    fn remove(&self, device: &Device) -> RemoveOutcome;
    fn suspend(&self, device: &Device, target: PmState) -> Result<(), PmError>;
    fn resume(&self, device: &Device) -> Result<(), PmError>;
}
// Ownership: exactly one live Resource per (kind, range), enforced by ResourceSet (I2, I12).
// Mutation: immutable once claimed; released only via RAII Drop.
// Concurrency: claims serialized through ddf-mmio/ddf-irq/ddf-dma (single arbiter each).
pub enum Resource {
    Mmio { base: PhysAddr, size: u64 },
    Io { base: u16, size: u16 },
    Irq { number: u32, trigger: IrqTrigger },
    DmaCoherent { mapping: DmaMapping },
    DmaStreaming { mapping: DmaMapping, direction: DmaDirection },
}

// Ownership: one per Device; owns every Resource handle the bound driver holds.
// Mutation: append-only during probe, drained on remove (I7).
pub struct ResourceSet {
    mmio: Vec<Resource>,
    irqs: Vec<Resource>,
    dma: Vec<Resource>,
}

// Ownership: none — pure Copy error enum.
pub enum ResourceError {
    Conflict { existing: ResourceId },
    Overlap { range: (u64, u64) },
    ExhaustedIrqTable,
    ExhaustedDmaSlots,
}

// Ownership: process-wide singleton table, one entry per IRQ vector.
// Concurrency: lock-free read path (top-half dispatch), locked write path (bind/unbind).
pub trait IrqHandler: Send + Sync {
    fn handle(&self) -> IrqResult;
}

pub enum IrqResult { Handled, NotHandled, WakeBottomHalf(BottomHalfRef) }
pub enum IrqTrigger { Edge, Level }
// Ownership: one per active mapping, owned by the requesting ResourceSet.
// Mutation: sync_for_device/sync_for_cpu are the only permitted state changes (I5).
// Concurrency: Send, not Sync — one driver context owns a mapping at a time.
// Persistence: none; torn down on device remove or driver unbind.
pub struct DmaMapping {
    pub dma_addr: DmaAddr,
    pub len: usize,
    pub group: IommuGroupId,
    pub coherent: bool,
}

pub enum DmaDirection { ToDevice, FromDevice, Bidirectional }

// Ownership: static after boot (I6); one group per IOMMU translation context.
// Compatibility: IommuGroupId is stable for the life of the boot.
pub struct IommuGroup {
    pub id: IommuGroupId,
    pub member_devices: Vec<DeviceId>,
}

pub trait DmaAllocator: Send + Sync {
    fn map_coherent(&self, device: &Device, len: usize) -> Result<DmaMapping, DmaError>;
    fn map_streaming(&self, device: &Device, len: usize, dir: DmaDirection) -> Result<DmaMapping, DmaError>;
    fn sync_for_device(&self, mapping: &DmaMapping);
    fn sync_for_cpu(&self, mapping: &DmaMapping);
}

pub enum DmaError { NoIommuGroup, ExhaustedSlots, MisalignedLength }
// Ownership: one queue per Bus instance; bounded ring buffer (§4, ADR-20).
// Mutation: push on bus-detected change, pop by the dispatcher (I10).
// Concurrency: MPSC — many bus-scan producers, one dispatcher consumer.
pub struct HotplugEvent {
    pub bus_id: BusId,
    pub sequence: u64,
    pub device: DeviceId,
    pub kind: HotplugKind,
}

pub enum HotplugKind { Plugged, Unplugged }

pub trait EventQueue: Send + Sync {
    fn push(&self, event: HotplugEvent) -> Result<(), QueueFull>;
    fn next(&self) -> Option<HotplugEvent>;
}

// Ownership: embedded in Device; total function over (state, event) pairs (I9).
pub enum PmState { D0, D1, D2, D3Hot, D3Cold }
pub enum PmEvent { SuspendRequest(PmState), ResumeRequest, DmaDrainComplete }
pub enum PmError { DmaStillInFlight, UnhandledTransition, DeviceQuarantined }

// Ownership: minted once per (Driver, Device) bind at the ddf-core FFI boundary (P8, ADR-16).
// Compatibility: a Capability token cannot be forged or cloned across devices.
pub struct Capability<T> { _marker: PhantomData<T> }

pub enum ProbeOutcome { Bound, Deferred, Quarantined }
pub enum RemoveOutcome { Removed, RemovedWithLeaks(Vec<ResourceId>) }
pub enum ProbeError { ResourceConflict(ResourceError), MalformedDeviceData, DriverInitFailed }

// Ownership: framework singleton; the only writer of Device.state (I1).
pub enum DeviceState {
    Unbound, Probing, Bound, Suspending, Suspended, Resuming, Removing, Freed, Quarantined,
}
