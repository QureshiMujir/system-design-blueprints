Core schema & types (§11)

pub enum WireType {
    Void,
    Bool,
    I8, I16, I32, I64,
    U8, U16, U32, U64,
    F32, F64,
    Enum,           // discriminant: u16; stored in data section
    Struct,         // pointer to struct (ptr section)
    List,           // pointer to list (ptr section)
    Union,          // discriminant u16 + active arm (data or ptr section)
    Text,           // pointer to UTF-8 bytes + NUL terminator
    Bytes,          // pointer to raw byte blob
    Pointer,        // generic untyped pointer (escape hatch; discouraged)
}
pub enum DefaultVal {
    Void,
    Bool(bool),
    I8(i8), I16(i16), I32(i32), I64(i64),
    U8(u8), U16(u16), U32(u32), U64(u64),
    F32(f32), F64(f64),
    Enum(u16),
    Null,           // absent pointer -> null reader
}
pub struct Layout {
    pub data_words: u16,   // size of data section in 8-byte words
    pub ptr_words:  u16,   // size of pointer section in 8-byte words
    pub align:      u8,    // required alignment in bytes (power of 2; max 8)
}
pub struct Field {
    pub id:            FieldId,
    pub name:          Arc<str>,       // interned; schema-owned
    pub ty:            WireType,
    pub type_id:       Option<TypeId>, // Some(_) for Struct/Enum/Union/List referents
    pub offset:        Offset,         // word offset within data or ptr section
    pub default:       DefaultVal,
    pub since_version: u32,            // schema version when field was added
    pub reserved:      bool,           // true = removed field; hole preserved
    pub doc:           Option<Arc<str>>,
}
pub struct TypeDef {
    pub id:     TypeId,
    pub name:   Arc<str>,
    pub kind:   TypeKind,
    pub fields: Vec<Field>,          // sorted by FieldId ascending
    pub layout: Layout,
    pub doc:    Option<Arc<str>>,
}

pub enum TypeKind { Struct, Enum, Union, List { elem: WireType, elem_type: Option<TypeId> } }
pub struct Schema {
    pub types:       Vec<Arc<TypeDef>>,   // all types; index by TypeId
    pub root:        TypeId,
    pub schema_hash: SchemaHash,
    pub version:     u32,                 // monotonic schema evolution counter
    pub namespace:   Arc<str>,
}

impl Schema {
    pub fn finalize(&mut self) -> Result<(), SerError>;   // resolves refs, computes hash, freezes
    pub fn lookup(&self, id: TypeId) -> Option<&TypeDef>;
    pub fn lookup_by_name(&self, name: &str) -> Option<&TypeDef>;
}
pub enum Pointer {
    Null,
    Struct {
        offset:     i32,         // word offset from pointer location (signed; relative)
        data_words: u16,
        ptr_words:  u16,
    },
    List {
        offset:     i32,
        elem:       ListElemSize, // enum: Void|Bit|Byte|TwoBytes|FourBytes|EightBytes|Composite
        count:      u32,
    },
    Far {
        seg:        SegmentId,
        offset:     u32,         // word offset within target segment
        double_far: bool,        // true = landing pad is another pointer (two-hop)
    },
    Cap(u32),                   // capability index (reserved; transport out of scope)
}

pub enum ListElemSize { Void, Bit, Byte, TwoBytes, FourBytes, EightBytes, Composite }
pub struct Segment {
    pub id:    SegmentId,
    pub bytes: Vec<u8>,   // owned; backing store for a builder-produced segment
}

impl Segment {
    pub fn as_slice(&self) -> &[u8];
    pub fn word_count(&self) -> u32;   // bytes.len() / 8; panics if not word-aligned
}

pub struct Message {
    pub segments: Vec<Segment>,   // segments[0] = root segment
    pub root:     Pointer,        // pointer into segments[0] to root struct
}

impl Message {
    pub fn segment(&self, id: SegmentId) -> Option<&Segment>;
    pub fn root_segment(&self) -> &Segment;   // segments[0]; always present
    pub fn total_bytes(&self) -> usize;
}
pub struct Reader<'a, T> {
    buf:    &'a [u8],    // borrowed slice over validated segment bytes
    base:   Offset,      // word offset of this struct's data section start
    schema: &'a TypeDef, // schema for type T (may be erased to DynReader)
    _type:  PhantomData<&'a T>,
}

impl<'a, T: Readable> Reader<'a, T> {
    // Accessor signatures (generated by ser-codegen per field):
    pub fn get_field<F: FieldAccessor>(&self, field_id: FieldId) -> F::Output<'a>;
    pub fn get_struct_field(&self, field_id: FieldId) -> Option<Reader<'a, DynStruct>>;
    pub fn get_list_field(&self, field_id: FieldId) -> ListReader<'a>;
    pub fn get_text_field(&self, field_id: FieldId) -> &'a str;
    pub fn get_bytes_field(&self, field_id: FieldId) -> &'a [u8];
    pub fn to_owned(&self) -> OwnedValue;   // explicit deep-copy (opt-in; allocates)
}

pub trait Readable: Sized { type Schema; }
pub struct DynStruct;  // type-erased reader target
pub struct Builder<'arena, T> {
    arena:   &'arena mut Arena,
    seg:     SegmentId,
    base:    Offset,
    schema:  &'arena TypeDef,
    _type:   PhantomData<T>,
}

impl<'arena, T: Writable> Builder<'arena, T> {
    pub fn set_field<V: FieldValue>(&mut self, field_id: FieldId, val: V);
    pub fn init_struct(&mut self, field_id: FieldId) -> Builder<'arena, DynStruct>;
    pub fn init_list(&mut self, field_id: FieldId, count: u32) -> ListBuilder<'arena>;
    pub fn set_text(&mut self, field_id: FieldId, s: &str);
    pub fn set_bytes(&mut self, field_id: FieldId, b: &[u8]);
    pub fn finish(self) -> Message;   // only on root Builder; seals arena
}

pub struct Arena {
    segments: Vec<Vec<u8>>,
    current:  usize,
    limit:    usize,   // max total bytes; enforced on alloc
}

pub trait Writable: Sized { type Schema; }
pub struct DynStruct;
pub enum SerError {
    SchemaInvalid { reason: Cow<'static, str> },
    OutOfBounds   { offset: u64, limit: u64 },
    Misaligned    { offset: u64, required_align: u8 },
    BadPointer    { kind: PointerKind, seg: Option<SegmentId>, offset: u64 },
    DepthExceeded { depth: u32, limit: u32 },
    Truncated     { expected: u64, found: u64 },
    IncompatibleSchema { expected: SchemaHash, found: SchemaHash },
    Io(std::io::Error),
}

pub enum PointerKind { Struct, List, Far, Cap }

pub enum SchemaError {
    ParseError    { span: Span, message: Cow<'static, str> },
    DuplicateName { span: Span, name: Arc<str> },
    DuplicateFieldId { type_id: TypeId, field_id: FieldId },
    UnresolvedType { span: Span, name: Arc<str> },
    CyclicEmbedding { type_id: TypeId },
    DuplicateDiscriminant { type_id: TypeId, value: u16 },
    NestingTooDeep { span: Span, depth: u32 },
    InvalidDefault { span: Span, field_id: FieldId },
    InvalidUnion  { type_id: TypeId },
    InvalidEmbedding { type_id: TypeId, field_id: FieldId },
}

pub struct Span { pub line: u32, pub col: u32 }
pub struct DecodeConfig {
    pub max_depth:        u32,   // default 64; recursion + pointer-follow limit
    pub max_segments:     u32,   // default 512; amplification defense
    pub max_message_size: u64,   // default 64 MiB; total bytes across all segments
    pub schema:           Arc<Schema>,
    pub strict_utf8:      bool,  // default true; Text fields validated as UTF-8
}
pub struct CompatReport {
    pub result:    CompatResult,
    pub violations: Vec<CompatViolation>,
}

pub enum CompatResult { FullyCompatible, BackwardOnly, ForwardOnly, Incompatible }

pub struct CompatViolation {
    pub kind:    ViolationKind,
    pub type_id: TypeId,
    pub field_id: Option<FieldId>,
    pub message: Cow<'static, str>,
}

pub enum ViolationKind {
    FieldRemoved,
    FieldTypeChanged,
    FieldOffsetChanged,
    FieldDefaultChanged,
    TypeKindChanged,
    SchemaHashMismatch,
}
pub struct TelemetryHandle {
    pub encode_bytes:        Counter,
    pub encode_duration_ns:  Histogram,
    pub decode_bytes:        Counter,
    pub decode_duration_ns:  Histogram,
    pub validate_duration_ns: Histogram,
    pub validation_errors:   Counter,  // per SerError variant label
    pub segment_allocs:      Counter,
}

pub trait MetricsSink: Send + Sync + 'static {
    fn record_counter(&self, name: &'static str, delta: u64, labels: &[(&str, &str)]);
    fn record_histogram(&self, name: &'static str, value_ns: u64, labels: &[(&str, &str)]);
}
