Core schema & types (§11)

ConsentRecord {
  session_id: UUID; party_ids: [PartyId]          // >= 2
  informed: bool; free_verified_by: CouncilRep
  revocable: true                                  // always, up to and during the session
  witnessed_by: CouncilRep; granted_at: Timestamp
  revoked_at: Timestamp?                            // revocation is final and costless
  audit_trail: NonContentAttestation                // proves consent, reveals no inner content
}

Session {
  id: UUID; state: enum { PRE_COUNSEL, CONSENT_CEREMONY, SESSION, AFTER_CARE,
                          COUNCIL_REVIEW, COMPLETED, REFUSED }
  parties: [PartyId]; counsel_id: UUID; substrate_id: UUID
  window: Duration                                  // minutes–tens of minutes
  inner_content: NEVER_PERSISTED                    // C2 — there is no field for it
  review_verdict: ConformanceVerdict?
}

Council {
  members: [CouncilMember]                          // >=3 generations; plural disciplines
  jurisdictions: [Jurisdiction]                     // never a single country
  decisions: PublicLedger                           // all public, legible, slow
}
CouncilMember { id; generation; discipline; term: Bounded; nomination_path; may_publish_dissent: true }

Counsel { id: UUID; apprenticeship: MultiDecade; secrecy: Strict; regulated_by: Council }

Substrate {
  id: UUID; candidate: enum { NEURAL_A, BIOTECH_B, AI_C, HYBRID_D, OPEN_E }
  steward_lineage: StewardId; conformance: ConformanceVerdict
  bidirectional: true; ephemeral: true; reversible: true; refusal_mode: true
  mistranslation_envelope: Disclosed                 // C4
}
SubstrateSteward { id; lineage: enum { UNIVERSITY, FOUNDATION, PUBLIC_CONSORTIUM }; deploys: false }

ReadinessIndicator { name: string; qualitative: true; assessed_by: Council; status: enum { UNMET, EMERGING, MET } }

Endowment { trustees: PublicList; books: Open; extraction: FORBIDDEN; dissolution_routes_to: CouncilNamedCauses }

AmendmentRecord { proposed_by; sharpens_or_adds: true; relaxes: false; public: true; ratified_at: Timestamp? }

RefusalEvent { what_was_refused: string; could_have_deployed: true; refused_by: Council|Steward; public: true }

ConformanceVerdict { substrate_id; terms: [C1..C12]; outcome: enum { CERTIFIED, DECERTIFIED }; non_content: true }
