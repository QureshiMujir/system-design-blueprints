Core schema & types (§11)

struct Lsn(u64);                       // Copy, Eq, Ord — strictly monotonic (I1)
struct TxnId(u64);                     // Copy, Eq, Hash
struct PageId(u64);                    // Copy, Eq, Hash

struct RecordHeader {                  // 32 B fixed (§4)
    lsn: Lsn,
    prev_lsn: Lsn,                     // per-transaction backward chain (§8.5)
    txn_id: TxnId,
    record_type: RecordType,
    payload_len: u32,                  // must equal serialized payload bytes (I13)
}

enum RecordType {
    Begin,
    Update { page_id: PageId },
    Commit,
    Abort,
    Clr { undo_next_lsn: Option<Lsn> }, // I8: strictly backward or None
    CheckpointBegin,
    CheckpointEnd,
}

struct LogRecord {                     // ADR-12 physiological layout
    header: RecordHeader,
    payload: Vec<u8>,                  // opaque to wal-writer/wal-reader (P13)
    crc32: u32,                        // over header+payload (P7)
}
struct DirtyPageTableEntry { page_id: PageId, rec_lsn: Lsn }   // §8.3
struct TransactionTableEntry {
    txn_id: TxnId,
    status: TxnStatus,                 // Active | Committed | Aborting
    last_lsn: Lsn,
    undo_next_lsn: Option<Lsn>,        // I8
}
enum TxnStatus { Active, Committed, Aborting }

struct CheckpointPayload {             // ADR-08 fuzzy checkpoint
    begin_checkpoint_lsn: Lsn,         // I10
    dirty_page_table: Vec<DirtyPageTableEntry>,
    transaction_table: Vec<TransactionTableEntry>,
}

enum WalError {
    CrcMismatch { at_lsn: Lsn },
    LengthMismatch { at_lsn: Lsn, declared: u32, actual: u32 },
    LsnOutOfOrder { expected_min: Lsn, got: Lsn },
    PostCommitWrite { txn_id: TxnId },
    ForceFailed,
    SegmentNotFound { segment_id: u64 },
}
trait DurableWriter {                  // owned exclusively by wal-writer (ADR-19)
    fn append(&mut self, r: LogRecord) -> Lsn;
    fn force(&mut self) -> Result<Lsn, WalError>;   // durability barrier (P1/ADR-04)
}

trait SegmentScanner {                 // owned by wal-reader (P13)
    fn next_record(&mut self) -> Result<Option<LogRecord>, WalError>;
    fn last_valid_lsn(&self) -> Lsn;   // torn-write tail boundary (P8)
}

trait RetentionGate {                  // owned by wal-replicate (P12)
    fn min_required_lsn(&self) -> Lsn; // slowest consumer's watermark
    fn segment_eligible_for_deletion(&self, segment_id: u64) -> bool;
}
