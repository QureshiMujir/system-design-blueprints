Core schema & types (§11)

pub struct StreamId(u32);                 // RFC 9113 §5.1.1: 31-bit; odd=client, even=server
impl StreamId {
    pub const CONNECTION: StreamId;
    pub const fn new(v: u32) -> Result<Self, InvalidStreamId>;
    pub const fn is_client_initiated(self) -> bool;
    pub const fn is_server_initiated(self) -> bool;
}

pub struct QuicStreamId(u64);             // RFC 9000 §2.1: varint; low 2 bits = type
impl QuicStreamId { pub const fn ty(self) -> StreamType; }  // {ClientBidi,ServerBidi,ClientUni,ServerUni}
pub struct FrameHeader { pub length: u32, pub kind: FrameType, pub flags: FrameFlags, pub stream_id: StreamId }

pub enum FrameType {                      // RFC 9113 §6 + extensions
    Data, Headers, Priority, RstStream, Settings, PushPromise, Ping,
    Goaway, WindowUpdate, Continuation, AltSvc, Origin, PriorityUpdate, Unknown(u8),
}

pub struct FrameFlags(u8);                // END_STREAM|ACK|END_HEADERS|PADDED|PRIORITY bits
pub enum StreamState { Idle, ReservedLocal, ReservedRemote, Open, HalfClosedLocal, HalfClosedRemote, Closed }
pub enum StreamEvent { SendHeaders{end_stream:bool}, RecvHeaders{end_stream:bool},
    SendData{end_stream:bool}, RecvData{end_stream:bool}, SendRstStream(ErrorCode), RecvRstStream(ErrorCode) }
impl StreamState {
    /// Table-driven per RFC 9113 §5.1 / §8 of this document; no free-form logic.
    pub fn apply(self, ev: StreamEvent) -> Result<StreamState, StreamStateError>;
}
pub enum ConnectionToIo { SendFrame(Frame), Flush, Close(Goaway) }
pub enum IoToConnection { Frame(Frame), PeerClosed, IoError(std::io::Error) }
pub enum PacketHeader { Long(LongHeader), Short(ShortHeader) }
pub struct LongHeader { pub version: u32, pub kind: LongHeaderKind, pub dcid: ConnectionId,
    pub scid: ConnectionId, pub token: Bytes, pub length: u64, pub packet_number: PacketNumber }
pub enum LongHeaderKind { Initial, ZeroRtt, Handshake, Retry, VersionNegotiation }
pub struct ShortHeader { pub spin_bit: bool, pub key_phase: bool, pub dcid: ConnectionId, pub packet_number: PacketNumber }
pub enum Frame { Padding, Ping, Ack(AckFrame), ResetStream(ResetStreamFrame), StopSending(StopSendingFrame),
    Crypto(CryptoFrame), NewToken(NewTokenFrame), Stream(StreamFrame), MaxData(MaxDataFrame),
    MaxStreamData(MaxStreamDataFrame), MaxStreams(MaxStreamsFrame), DataBlocked(DataBlockedFrame),
    NewConnectionId(NewConnectionIdFrame), RetireConnectionId(RetireConnectionIdFrame),
    PathChallenge(PathChallengeFrame), PathResponse(PathResponseFrame),
    ConnectionClose(ConnectionCloseFrame), HandshakeDone, Datagram(DatagramFrame) }

pub struct ConnectionId { /* 0..=20 octets, RFC 9000 §17.2 */ }
impl ConnectionId {
    pub fn new(slice: &[u8]) -> Result<Self, ConnectionIdLengthError>;
    pub fn as_slice(&self) -> &[u8];
}
pub struct PacketNumber(pub u64);          // 62-bit, RFC 9000 §12.3 truncation rule
impl PacketNumber {
    /// Truncation per RFC 9000 §17.1; algorithm described in §8, not here.
    pub fn encode(self, largest_acked: Option<PacketNumber>) -> (u32, u8 /* len */);
    pub fn decode(truncated: u32, len: u8, largest: PacketNumber) -> PacketNumber;
}

pub struct Settings { pub header_table_size: u32, pub enable_push: bool,
    pub max_concurrent_streams: Option<u32>, pub initial_window_size: u32,
    pub max_frame_size: u32, pub max_header_list_size: Option<u32>,
    pub enable_connect_protocol: bool, pub no_rfc7540_priorities: bool }   // RFC 9113 §6.5.2
pub struct TransportParameters { pub max_idle_timeout: u64, pub max_udp_payload_size: u64,
    pub initial_max_data: u64, pub initial_max_stream_data_bidi_local: u64,
    pub initial_max_stream_data_bidi_remote: u64, pub initial_max_stream_data_uni: u64,
    pub initial_max_streams_bidi: u64, pub initial_max_streams_uni: u64,
    pub active_connection_id_limit: u64, pub disable_active_migration: bool }  // RFC 9000 §18

pub struct FlowControl { pub send_window: i64, pub recv_window: i64, pub recv_pending: i64 }
impl FlowControl {
    pub fn new(initial: u32) -> Self;
    pub fn consume_send(&mut self, n: u32) -> Result<(), FlowControlError>;
    pub fn on_recv_data(&mut self, n: u32);
    pub fn on_app_consumed(&mut self, n: u32, threshold: u32) -> Option<u32>;
}

pub struct Rack { pub reordering_threshold: Duration, pub smoothed_rtt: Duration, pub rttvar: Duration }
pub struct Cubic { pub cwnd: u64, pub ssthresh: u64, pub w_max: f64, pub k: f64 }
pub trait Frame { fn header(&self) -> FrameHeader; fn write_payload(&self, dst: &mut BytesMut);
    fn parse_payload(header: FrameHeader, src: &mut Bytes) -> Result<Self, FrameError> where Self: Sized; }
pub trait Codec { type Frame: Frame;
    fn encode(&mut self, frame: Self::Frame, dst: &mut BytesMut) -> Result<(), CodecError>;
    fn decode(&mut self, src: &mut BytesMut) -> Result<Option<Self::Frame>, CodecError>; }
pub trait Endpoint { fn handle_packet(&mut self, src: SocketAddr, packet: BytesMut, now: Instant) -> Result<Vec<Output>, EndpointError>;
    fn poll_transmit(&mut self) -> Option<Transmit>; fn poll_timeout(&mut self) -> Option<Instant>; }
pub struct Transmit { pub destination: SocketAddr, pub ecn: Option<EcnCodepoint>, pub data: BytesMut, pub segment_size: Option<usize> }
pub trait TlsConnection {
    fn write_handshake(&mut self, level: KeyLevel, dst: &mut BytesMut) -> Result<usize, TlsError>;
    fn read_handshake(&mut self, level: KeyLevel, src: &[u8]) -> Result<(), TlsError>;
    fn read_keys(&mut self, level: KeyLevel) -> Option<Keys>;
    fn write_keys(&mut self, level: KeyLevel) -> Option<Keys>;
    fn alpn_protocol(&self) -> Option<&[u8]>;
    fn quic_transport_parameters(&self) -> Option<&[u8]>;
    fn set_quic_transport_parameters(&mut self, bytes: &[u8]);
    fn key_update(&mut self) -> Result<(), TlsError>;
}
pub enum KeyLevel { Initial, ZeroRtt, Handshake, OneRtt }
pub struct Keys { pub key: Vec<u8>, pub iv: Vec<u8>, pub hp_key: Vec<u8>, pub aead: Aead }
