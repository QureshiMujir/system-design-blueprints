Core schema & types (§11)

pub struct SequenceNumber(pub u64);          // Copy; total order = MVCC clock (I4, I10)
#[repr(u8)]
pub enum ValueKind {
    Value = 0, Deletion = 1, SingleDeletion = 2,
    Merge = 3,            // PhaseDeferred §11: merge operator
    RangeDeletion = 15,   // meta-block only, never in the data stream (I12)
}
pub struct FileId(pub u64);                  // Copy; stable SSTable identity
pub struct Level(pub u8);                    // Copy; L0 overlapping, L1+ non-overlapping (I3)
pub const L0: Level = Level(0);
pub const MAX_LEVELS: usize = 7;
pub type UserKey<'a> = &'a [u8];
pub struct InternalKey(pub Vec<u8>);   // encoding: user_key || (seq<<8|kind) u64 LE (§8.4)
pub struct BlockHandle { pub offset: u64, pub size: u64 }        // 2 varints in the index
pub struct Footer {
    pub metaindex_handle: BlockHandle,
    pub index_handle: BlockHandle,
    pub magic: u64,                 // rejects non-matching files (§8.1)
}
pub const FOOTER_LEN: usize = 48;   // fixed; trails every SSTable file
#[repr(u8)]
pub enum WalRecordKind {
    Put = 1, Delete = 2, SingleDelete = 3, RangeDelete = 4,
    Merge = 5,             // PhaseDeferred §11
    BatchHeader = 8, BatchEnd = 9,
}
pub struct WalRecordHeader { pub seq: u64, pub kind: u8, pub payload_len: u32, pub crc32c: u32 }
pub struct VersionEdit {
    pub log_number: Option<u64>,
    pub prev_log_number: Option<u64>,
    pub next_file_number: Option<u64>,
    pub last_sequence: Option<SequenceNumber>,
    pub deleted_files: Vec<(Level, FileId)>,
    pub added_files: Vec<AddedFile>,
    pub comparator_name: Option<String>,
}
pub struct AddedFile {
    pub level: Level, pub file_id: FileId, pub file_size: u64,
    pub smallest: InternalKey, pub largest: InternalKey,
    pub smallest_seq: SequenceNumber, pub largest_seq: SequenceNumber,
    pub num_entries: u64, pub num_deletions: u64,
    pub raw_key_size: u64, pub raw_value_size: u64,
}
pub struct Version { pub levels: [Vec<FileMetadata>; MAX_LEVELS], pub last_sequence: SequenceNumber, pub log_number: u64 }
pub trait MemTable: Send + Sync {
    fn add(&self, seq: SequenceNumber, kind: ValueKind, key: &[u8], value: &[u8]);
    fn add_range_delete(&self, seq: SequenceNumber, start: &[u8], end: &[u8]);
    fn get(&self, snapshot: SequenceNumber, key: &[u8]) -> MemGetResult<'_>;
    fn iter(&self) -> Box<dyn LsmIterator + '_>;
    fn approximate_memory_usage(&self) -> usize;
}
pub enum MemGetResult<'m> { Found(&'m [u8]), Deleted, NotFound }
pub trait SstWriter {
    fn add(&mut self, seq: SequenceNumber, kind: ValueKind, key: &[u8], value: &[u8]) -> Result<()>;
    fn add_range_delete(&mut self, seq: SequenceNumber, start: &[u8], end: &[u8]) -> Result<()>;
    fn finalise(self) -> Result<AddedFile>;
}
pub trait SstReader: Send + Sync {
    fn get(&self, key: &[u8], snapshot: SequenceNumber) -> Result<SstGetResult>;
    fn iter(&self) -> Box<dyn LsmIterator + '_>;
    fn properties(&self) -> &TableProperties;
}
pub enum SstGetResult { Found(Vec<u8>), Deleted, NotFound }
pub trait LsmIterator: Send {
    fn seek(&mut self, key: &[u8]) -> Result<()>;
    fn seek_to_first(&mut self) -> Result<()>;
    fn seek_to_last(&mut self) -> Result<()>;
    fn key(&self) -> Option<&InternalKey>;
    fn value(&self) -> Option<&[u8]>;
    fn valid(&self) -> bool;
}
pub struct MergingIter<'r> {
    pub children: Vec<Box<dyn LsmIterator + 'r>>,
    pub snapshot: SequenceNumber,
    // heap + range-del tracker state: internal, not part of the public contract
}
pub trait BloomFilter: Send + Sync {
    fn add(&mut self, key: &[u8]);
    fn build(self) -> Vec<u8>;                          // serialised filter bytes
    fn contains_in(filter_bytes: &[u8], key: &[u8]) -> bool;
}
pub struct DbOptions {
    pub write_buffer_size: usize, pub max_immutable_memtables: usize,
    pub block_size: usize, pub bloom_bits_per_key: u32,
    pub level_multiplier: u64, pub compaction_strategy: CompactionStrategy,
    pub compression: CompressionKind, pub direct_io: bool,
    pub rate_limiter_bytes_per_sec: Option<u64>,
    pub telemetry: Box<dyn Telemetry + Send + Sync>,
}
pub enum CompressionKind { None, Snappy, Lz4, Zstd { level: i32 } }
pub enum CompactionStrategy { Leveled, SizeTiered, TieredLeveled, LazyLeveling }
pub struct WriteOptions { pub sync: bool, pub disable_wal: bool }
pub struct ReadOptions<'s> { pub snapshot: Option<&'s Snapshot>, pub verify_checksums: bool }
pub struct Db { /* Arc<DbInner>: wal, memtables, versions, cache, telemetry */ }
pub struct Snapshot { /* seq: SequenceNumber; pins min_live_snapshot on construction (§17.1) */ }
#[derive(thiserror::Error, Debug)]
pub enum Error {
    Io(std::io::Error), BlockCorrupt(FileId, BlockHandle), WalCorrupt(u64),
    WalTruncated(SequenceNumber), ManifestCorrupt(u64), MissingFile(FileId),
    WriteStall { pending_l0: u32, trigger: u32 }, SnapshotExpired(SequenceNumber),
    PhaseDeferred(&'static str, &'static str),
}
pub trait Telemetry: Send + Sync + 'static {
    fn on_put(&self, batch_size: usize);
    fn on_get(&self, hit: bool, levels_touched: u8);
    fn on_flush(&self, file_id: FileId, bytes: u64);
    fn on_compaction(&self, job_id: u64, level_in: Level, level_out: Level, bytes_in: u64, bytes_out: u64);
    fn on_write_stall(&self, pending_l0: u32);
}
