Core schema & types (§11)

type InstructionId = u64;
// ownership: created once by sim-trace, copied thereafter (Copy)
// mutation: immutable after assignment
// concurrency: no shared mutation — pure value type
// persistence: replay-stable, appears verbatim in every ReplayFrame referencing it
// serialization: bincode fixed-width u64 (ADR-07)
// compatibility: never reused within one run (monotonic allocator)
Uop (fields, no method bodies):
  uop_id: InstructionId
  parent_instruction: InstructionId
  opcode: OpcodeId
  src_regs: [PhysRegId; MAX_SRC]
  dst_reg: PhysRegId
  latency: LatencyClass
  flags: UopFlags
RobEntry (fields, no method bodies):
  rob_slot: u16
  uop_ref: InstructionId
  ready: bool
  fault_code: Option<ExceptionCode>
  branch_mask: EpochId
  completion_cycle: CycleId

  dependency_bitmap: BitSet

  rename_map_snapshot: RenameMapId

  epoch: EpochId
CacheLine (fields, no method bodies):
  tag: u64
  state: LineState
  dirty: bool
  last_access: CycleId
  owner: CoreId
ReplayFrame (fields, no method bodies):
  frame_id: u64
  cycle: CycleId
  subsystem: SubsystemId

ArchitecturalRegister (fields): id: ArchRegId
