Core schema & types (§11)

// Key — opaque byte sequence; comparison is lexicographic
pub struct Key(Vec<u8>);
impl Key { pub fn prefix_end(&self) -> Key; }   // next key after all keys with this prefix

// Revision — global monotonic; sub=0 for single-op txn
pub struct Revision { pub main: i64, pub sub: i64 }
impl Ord for Revision { /* lexicographic (main, sub) */ }

// KeyValue — the value stored per key at a revision
pub struct KeyValue {
    pub key: Key, pub create_revision: i64, pub mod_revision: i64,
    pub version: i64, pub value: Vec<u8>, pub lease: i64,
}

// Event — one mutation event, emitted by the apply loop
pub enum EventType { Put, Delete }
pub struct Event { pub ty: EventType, pub kv: KeyValue, pub prev_kv: Option<KeyValue> }

// Request / Response types
pub struct RangeRequest  { pub key: Key, pub range_end: Vec<u8>, pub revision: i64,
                            pub limit: i64, pub sort: SortOrder, pub count_only: bool }
pub struct RangeResponse { pub header: ResponseHeader, pub kvs: Vec<KeyValue>,
                            pub more: bool, pub count: i64 }
pub struct PutRequest    { pub key: Key, pub value: Vec<u8>, pub lease: i64,
                            pub prev_kv: bool, pub ignore_value: bool }
pub struct DeleteRangeRequest { pub key: Key, pub range_end: Vec<u8>, pub prev_kv: bool }
pub struct ResponseHeader { pub cluster_id: u64, pub member_id: u64,
                             pub revision: i64, pub raft_term: u64 }

// Transaction
pub struct TxnRequest  { pub compare: Vec<Compare>, pub success: Vec<Op>, pub failure: Vec<Op> }
pub struct TxnResponse { pub header: ResponseHeader, pub succeeded: bool, pub responses: Vec<Response> }
pub struct Compare     { pub target: CompareTarget, pub key: Key,
                          pub result: CompareResult, pub value: Vec<u8> }
pub enum CompareTarget  { Version, Create, Mod, Value, Lease }
pub enum CompareResult  { Equal, Greater, Less, NotEqual }
pub enum Op             { Range(RangeRequest), Put(PutRequest),
                          Delete(DeleteRangeRequest), Txn(Box<TxnRequest>) }

// Error taxonomy
pub enum DcsError {
    NotLeader { leader_hint: Option<u64> },  KeyNotFound,  Compacted { compact_revision: i64 },
    FutureRev { current: i64 },  LeaseNotFound,  LeaseExpired,
    AuthDenied { reason: String },  ValidationFailed { msg: String },
    Io(std::io::Error),  Raft(RaftError),
}
pub enum Role { Follower, Candidate, Leader }
pub struct LogEntry    { pub term: u64, pub index: u64, pub data: EntryKind }
pub enum EntryKind     { Normal(Vec<u8>), ConfChange(ConfChange), Empty }
pub struct HardState   { pub term: u64, pub vote: u64, pub commit: u64 }
pub struct SoftState   { pub lead: u64, pub raft_state: Role }
pub struct Ready {
    pub entries: Vec<LogEntry>,            // must persist to WAL before sending messages
    pub hard_state: Option<HardState>,     // persist if Some
    pub committed_entries: Vec<LogEntry>,  // apply to state machine
    pub messages: Vec<Message>,            // send to peers
    pub snapshot: Option<Snapshot>,        // install if Some
}
pub enum Message {
    AppendEntries  { term, leader_id, prev_log_index, prev_log_term, entries, leader_commit },
    AppendResp     { term, follower_id, index, reject: bool, reject_hint },
    RequestVote    { term, candidate_id, last_log_index, last_log_term, pre_vote: bool },
    VoteResp       { term, voter_id, granted: bool, pre_vote: bool },
    InstallSnapshot{ term, leader_id, snapshot: Snapshot },
    ReadIndex      { term, request_ctx: Vec<u8> },
    ReadIndexResp  { term, request_ctx: Vec<u8>, index: u64 },
    Heartbeat      { term, leader_id, commit },
    HeartbeatResp  { term, follower_id },
}
pub struct TreeIndex(BTreeMap<Key, KeyIndex>);  // guarded by RwLock in production
pub struct KeyIndex  { pub key: Key, pub modified: Revision, pub generations: Vec<Generation> }
pub struct Generation{ pub created: Revision, pub version: i64, pub revs: Vec<Revision> }
pub trait Backend: Send + Sync {
    fn read_tx(&self)  -> Box<dyn ReadTx + '_>;
    fn write_tx(&mut self) -> Box<dyn WriteTx + '_>;
    fn size_bytes(&self) -> u64;
}
pub trait ReadTx  { fn get(&self, bucket: &str, key: &[u8]) -> Option<Vec<u8>>; fn range(…) → Vec<(K,V)>; }
pub trait WriteTx { fn put(&mut self, bucket: &str, key: &[u8], val: &[u8]); fn delete(…); fn commit(self); }
// Bucket names (constants): "key" (rev→KV bytes), "meta" (compact_rev, cluster_id), "members"
pub struct WatchId(i64);
pub struct WatchRequest { pub id: WatchId, pub key: Key, pub range_end: Vec<u8>,
                           pub start_revision: i64, pub prev_kv: bool, pub filters: WatchFilters }
pub struct WatchFilters  { pub no_put: bool, pub no_delete: bool }
pub struct WatchResponse { pub watch_id: WatchId, pub events: Vec<Event>,
                            pub header: ResponseHeader, pub compact_revision: i64,
                            pub canceled: bool, pub cancel_reason: String }
pub struct LeaseId(i64);
pub struct Lease { pub id: LeaseId, pub granted_ttl: i64, pub remaining_ttl: i64,
                    pub keys: BTreeSet<Key> }
pub struct LeaseCheckpoint { pub id: LeaseId, pub remaining_ttl: i64 }  // log entry payload
pub enum Layer { Default, Env, Service, Instance }  // ascending precedence
pub enum ValueType { Bool, Int, Float, Str, Json, Bytes }
pub struct FieldDef   { pub name: String, pub ty: ValueType, pub required: bool,
                         pub default: Option<Vec<u8>>, pub constraints: Vec<Constraint> }
pub struct ConfigSchema { pub prefix: Key, pub fields: Vec<FieldDef> }
pub enum Constraint   { MinInt(i64), MaxInt(i64), Regex(String), OneOf(Vec<Vec<u8>>) }
// KvService (tonic-generated trait impl)
async fn put(req: PutRequest)                  -> Result<PutResponse, Status>;
async fn range(req: RangeRequest)              -> Result<RangeResponse, Status>;
async fn delete_range(req: DeleteRangeRequest) -> Result<DeleteRangeResponse, Status>;
async fn txn(req: TxnRequest)                  -> Result<TxnResponse, Status>;
// WatchService
async fn watch(stream: WatchRequestStream)     -> WatchResponseStream;
// LeaseService
async fn lease_grant(req: LeaseGrantRequest)   -> Result<LeaseGrantResponse, Status>;
async fn lease_keep_alive(stream: …)           -> LeaseKeepAliveResponseStream;
async fn lease_revoke(req: LeaseRevokeRequest) -> Result<LeaseRevokeResponse, Status>;
// MaintenanceService
async fn compact(req: CompactionRequest)       -> Result<CompactionResponse, Status>;
async fn defrag(req: DefragmentRequest)        -> Result<DefragmentResponse, Status>;
async fn status(req: StatusRequest)            -> Result<StatusResponse, Status>;
async fn snapshot(req: SnapshotRequest)        -> SnapshotResponseStream;
async fn move_leader(req: MoveLeaderRequest)   -> Result<MoveLeaderResponse, Status>;
