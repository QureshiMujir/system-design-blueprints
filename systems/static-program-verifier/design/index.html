Core schema & types (§11)

/// Interned identifier; Copy, equality by interner index. Never compared by string.
pub struct Atom(pub u32);
/// Byte range [lo,hi) into the single source buffer; on every diagnostic + Reason.
pub struct Span { pub lo: u32, pub hi: u32 }
/// Literal / model value. The only inhabited scalar domains: Int, Bool, Ref(Null).
pub enum Value { Int(i128), Bool(bool), Null }
pub enum UnOp  { Neg, Not }
pub enum BinOp { Add, Sub, Mul, Div, Mod, Eq, Ne, Lt, Le, Gt, Ge, And, Or, Imp, Iff }
pub enum Quantifier { Forall, Exists }
/// A multi-trigger: a set of term patterns guiding SMT quantifier instantiation.
pub struct Trigger { pub terms: Vec<Expr> }
/// A whole compilation unit: an ordered list of top-level declarations.
pub struct Program { pub decls: Vec<Decl> }
/// Axiom(Expr) = a global assumed fact (asserted to SMT verbatim; unsound if false).
pub enum Decl { Method(Method), Function(Function), Axiom(Expr) }

/// A verifiable procedure. body=None => trusted (contract only, no proof obligation).
pub struct Method {
    pub name: Atom,
    pub params: Vec<Param>,    // inputs (immutable in body)
    pub returns: Vec<Param>,   // out-parameters (assignable; appear in ensures)
    pub spec: Spec,            // requires/ensures/modifies/decreases
    pub body: Option<Block>,   // None = abstract/trusted method
}
/// A pure (side-effect-free) function; lowers to uninterpreted SMT fn + axiom (## 21).
pub struct Function { pub name: Atom, pub params: Vec<Param>, pub ret: Type, pub body: Option<Expr> }
pub struct Param { pub name: Atom, pub ty: Type }
pub struct Block { pub stmts: Vec<Stmt> }
/// LValue: assignment target. Var for locals/returns; Field for heap object fields.
pub enum LValue { Var(Atom), Field(Box<Expr>, Atom) }
/// All four clause-vectors are conjunctive (empty = trivially true).
pub struct Spec {
    pub requires:  Vec<Expr>,   // assumed on entry
    pub ensures:   Vec<Expr>,   // asserted on exit (may use old())
    pub modifies:  Vec<Expr>,   // frame: heap locations the method may change
    pub decreases: Vec<Expr>,   // termination variant (lexicographic tuple)
}
/// Imperative statement. While carries invariants+decreases (the loop cut, ## 18).
pub enum Stmt {
    VarDecl(Atom, Type, Option<Expr>),   // local introduction (+ optional init)
    Assign(LValue, Expr),                // l := e
    If(Expr, Block, Block),              // then / else (else may be empty)
    While { guard: Expr, invariants: Vec<Expr>, decreases: Vec<Expr>, body: Block },
    Call(Vec<LValue>, Atom, Vec<Expr>),  // outs := callee(args)
    Assert(Expr),                        // user proof obligation
    Assume(Expr),                        // user-supplied assumption (UNSOUND if wrong; warn)
    Return,                              // exit point; triggers ensures check
}
/// Side-effect-free expression. Old(e) = value of e in the pre-state (two-state).
/// Quant binds (Atom,Type) vars with Trigger patterns over a Box<Expr> body.
pub enum Expr {
    Lit(Value),
    Var(Atom),
    Un(UnOp, Box<Expr>),
    Bin(BinOp, Box<Expr>, Box<Expr>),
    Call(Atom, Vec<Expr>),               // pure-Function application only
    Old(Box<Expr>),                      // legal only inside ensures (## 14)
    Quant(Quantifier, Vec<(Atom, Type)>, Vec<Trigger>, Box<Expr>),
    Field(Box<Expr>, Atom),              // heap field access -> Map select (## 24)
}
/// Source-language type, as-written. Map/Seq structural; Named resolved in spv-ast.
pub enum Type { Int, Bool, Ref, Map(Box<Type>, Box<Type>), Seq(Box<Type>), Named(Atom) }
/// Parse failure: rejects ill-formed input before any semantic phase.
pub struct ParseError { pub kind: ParseErrorKind, pub span: Span }
pub enum ParseErrorKind { UnexpectedToken, BadSpec, BadType }
/// Resolved + type-checked program. Same node shapes; identifiers now resolve and
/// every Expr has a checked Type. Immutable; lent to spv-ivl.
pub struct Typed<'p> { pub program: &'p Program, /* resolution + type tables */ }

/// Semantic (well-formedness) failure on a *parseable* program.
pub enum TypeError {
    Undeclared(Atom),                          // use of an unbound name
    Mismatch { expected: Type, got: Type },    // type clash
    NonBoolSpec,                               // a requires/ensures/invariant isn't Bool
    OldOutsideEnsures,                         // old() used outside an ensures clause
    BadModifies,                               // modifies entry isn't a heap location
    ArityMismatch,                             // call/quantifier wrong number of args
}
/// Guarded command. The complete IVL alphabet (no other control constructs).
pub enum IvlCmd {
    Assume(Expr),                    // restrict to states satisfying Expr
    Assert(Expr, Reason),            // proof obligation; Reason = why it exists
    Havoc(Vec<Var>),                 // forget the listed vars (nondeterministic reassignment)
    Assign(Var, Expr),               // deterministic update (becomes assume(=) in DSA, ## 20)
    Seq(Vec<IvlCmd>),                // sequencing c1;..;cn
    Choice(Box<IvlCmd>, Box<IvlCmd>),// nondeterministic choice c1 [] c2 (if-desugaring)
}

/// Provenance of an Assert: drives error classification + source mapping.
pub struct Reason { pub kind: VErrorKind, pub span: Span }

/// Lower one method to a single IvlCmd:
///   Seq[ assume(requires); <desugared body>; assert(ensures, Reason{Postcondition..}) ].
/// if  -> Choice(assume g; then , assume !g; else)
/// while-> invariant cut (## 18); call -> assert pre / havoc modifies / assume post (## 17).
/// Pure, deterministic; &Method in, owned IvlCmd out. body=None => no obligation emitted.
pub fn desugar_method(m: &Method) -> IvlCmd;
/// First-order verification formula. Atom(Pred) wraps an atomic predicate over terms.
/// Forall carries Triggers (SMT instantiation hints); Exists does not (model-search).
pub enum VcFormula {
    True,
    False,
    And(Vec<VcFormula>),
    Or(Vec<VcFormula>),
    Not(Box<VcFormula>),
    Imp(Box<VcFormula>, Box<VcFormula>),
    Forall(Vec<(Var, Type)>, Vec<Trigger>, Box<VcFormula>),
    Exists(Vec<(Var, Type)>, Box<VcFormula>),
    Atom(Pred),
}

/// Atomic predicate: a relation over first-order terms (e.g. Lt(t1,t2), App(f,args), Eq).
/// Terms reuse Var/Value; Pred is the leaf of VcFormula. (Variant set fixed in ## 15.)
pub enum Pred { /* Cmp(BinOp,Term,Term) | App(Atom,Vec<Term>) | BoolVar(Var) | ... */ }

/// Weakest precondition. The five LOCKED rules:
///   wp(assume p,Q)=p=>Q ; wp(assert p,Q)=p AND Q ; wp(havoc x,Q)=forall x.Q ;
///   wp(x:=e,Q)=Q[e/x] ; wp(c1;c2,Q)=wp(c1,wp(c2,Q)) ; wp(c1[]c2,Q)=wp(c1,Q) AND wp(c2,Q).
/// Method VC = wp(desugar_method(m), True). VALID iff program meets spec.
pub fn wp(c: &IvlCmd, post: VcFormula) -> VcFormula;

/// Passive (dynamic-single-assignment) form: each Var assigned once (fresh `version`),
/// Assign -> Assume(=), Choice merged via versioned joins. Prevents exponential wp blowup
/// (Flanagan-Saxe). MUST run BEFORE wp. Pure; &IvlCmd in, owned IvlCmd out.
pub fn passify(c: &IvlCmd) -> IvlCmd;
/// SMT sort. Array models heaps (Ref->object) and Maps/Seqs (## 24). Uninterp = abstract type.
pub enum SmtSort { Int, Bool, Uninterp(Atom), Array(Box<SmtSort>, Box<SmtSort>) }

/// Sorted first-order SMT term: constants, sorted vars, function apps, arithmetic/bool ops,
/// select/store, quantifiers + triggers. The encoding target of VcFormula. Immutable.
pub enum SmtTerm { /* Const(Value) | Var(Var,SmtSort) | App(Atom,Vec<SmtTerm>) | Forall{..} | ... */ }

/// The solver boundary (Phase 9/10). Implementations: an in-process #278 binding OR a
/// Z3 process driver behind SMT-LIB 2. &mut self => the session is STATEFUL + single-threaded:
/// push/pop, incremental assertions, one session per parallel method-worker (never shared).
pub trait SmtSolver {
    /// Assert the conjunction of `assertions` (the negated VC) and check satisfiability.
    fn check(&mut self, assertions: &[SmtTerm]) -> SmtResult;
}

/// Solver verdict. Unsat => VC valid => Verified. Sat(Model) => counterexample exists.
/// Unknown => timeout / incompleteness (quantifiers) — NEVER treated as Verified (## 22).
pub enum SmtResult { Unsat, Sat(Model), Unknown }

/// A satisfying assignment: solver-name -> Value, parsed from get-model output.
/// DSA/skolem names here; spv-verify maps them back to source vars (## 23).
pub struct Model { /* assignments: Vec<(Atom, Value)> */ }
/// Top-level outcome of verifying one method (or program: Vec joined). The ONLY answer
/// surfaced to the user. Verified MUST be unreachable for a buggy program (soundness).
pub enum VerifyResult { Verified, Failed(Vec<VError>) }

/// One verification failure: a proof obligation the solver could not discharge.
pub struct VError {
    pub kind: VErrorKind,             // taken from the failing Assert's Reason
    pub span: Span,                   // source location of that obligation
    pub counterexample: Counterexample,
}

/// Why an obligation failed. Each maps 1:1 to a category of Assert emitted by desugaring.
pub enum VErrorKind {
    AssertionMightNotHold,            // user assert
    PostconditionMightNotHold,        // ensures at return
    PreconditionOfCallMightNotHold,   // callee requires at a Call site
    LoopInvariantOnEntry,             // invariant fails before first iteration
    LoopInvariantNotMaintained,       // invariant not preserved by one arbitrary iteration
    DivisionByZero,                   // a Div/Mod divisor obligation
    ModifiesViolation,                // write outside the modifies frame
    MightNotTerminate,                // decreases variant not well-founded / non-decreasing
}

/// Source-level witness: variable bindings (mapped back from the Model) + the failing point.
pub struct Counterexample {
    pub bindings: Vec<(Atom, Value)>, // surface var name -> concrete value
    pub at: Span,                     // the program point that fails (== VError.span)
}
