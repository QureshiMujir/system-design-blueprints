Core schema & types (§11)

// ── Pool / heap handle (volatile) ────────────────────────────────────────
pub struct Pool {
    base: *mut u8,        // VOLATILE: varies per mmap — never persisted (P7)
    len: usize,
    uuid_lo: u64,         // matches Superblock.uuid_lo
    domain: Domain,       // Adr | Eadr (detected at open, ADR-007)
    backend: Backend,     // Real(fd) | Emu
    alloc: AllocIndex,    // VOLATILE free-list index (rebuilt on open, INV-6)
    // ownership: owns mapping; Drop = clean-shutdown(set bit, flush, fence, munmap)
    // concurrency P0: single-writer; Send not Sync (until Phase 11 full)
}
impl Pool {
    pub fn create(path:&Path, size:usize, opts:CreateOpts) -> Result<Pool, PErr>;
    pub fn open(path:&Path)   -> Result<Pool, PErr>;   // runs recovery if unclean (INV-8)
    pub fn root(&self) -> PMEMoid;                      // persistent anchor (INV-13)
    pub fn set_root(&self, tx:&Tx, oid:PMEMoid);        // must be in tx (P6)
    pub fn check(&self) -> CheckReport;                 // fsck-style, read-only
}
// ── On-media superblock (durable, fixed offset 0) ────────────────────────
#[repr(C)] pub struct Superblock {
    magic: u64, fmt_major: u16, fmt_minor: u16,
    uuid_lo: u64, uuid_hi: u64,
    pool_size: u64, root_off: u64,   // root_off stable forever (INV-13)
    flags: u32,             // bit0 = CLEAN_SHUTDOWN (INV-8)
    csum: u32,              // crc32 over preceding fields (INV-7, P15)
}
// ── Persistent pointer ────────────────────────────────────────────────────
#[repr(C)] pub struct PMEMoid { pub pool_uuid_lo:u64, pub off:u64 } // 16B, NULL={0,0}
pub struct PersistentPtr<T>(u64 /*off*/, PhantomData<T>);           // single-pool fast path
impl<T> PersistentPtr<T> {
    pub const NULL: Self;
    pub unsafe fn as_ref<'p>(&self, p:&'p Pool) -> Option<&'p T>;     // bounds-checked (INV-4)
    pub unsafe fn as_mut<'p>(&self, p:&'p Pool) -> Option<&'p mut T>; // must be inside tx
    // persistence: off stored; deref translates base+off (P7: no raw addr persisted)
}
// ── Object header (precedes every heap object) ────────────────────────────
#[repr(C)] pub struct ObjHdr { size_class:u32, flags:u32, type_tag:u64 } // 16B
// ── Transaction ──────────────────────────────────────────────────────────
pub struct Tx<'p> { pool:&'p Pool, id:u64, state:TxState }
pub enum TxState { Active, Committing, Committed, Aborting, Aborted }
impl<'p> Tx<'p> {
    pub fn begin(p:&'p Pool) -> Tx<'p>;
    pub fn add_range(&self, oid:PMEMoid, off:usize, len:usize) -> Result<(),PErr>; // INV-3,14
    pub fn alloc(&self, size:usize, type_tag:u64) -> Result<PMEMoid,PErr>;  // P6
    pub fn free(&self, oid:PMEMoid) -> Result<(),PErr>;                     // P6
    pub fn redo_write(&self, oid:PMEMoid, off:usize, bytes:&[u8]) -> Result<(),PErr>;
    pub fn commit(self) -> Result<(),PErr>; // INV-1,2: flush+drain+8B-marker+drain+clear-logs
    pub fn abort(self);                     // rollback undo / discard redo (INV-12 idempotent)
    // Drop without commit ⇒ abort (RAII; P8)
}
// ── Log record (durable — undo & redo share shape) ───────────────────────
#[repr(C)] pub struct LogRec {
    csum:u32, valid:u16, kind:u8, rsvd:u8, // kind: Undo|Redo
    seq:u64, off:u64, len:u32,              // payload[len] follows; INV-11 (self-describing)
}
// ── Allocator metadata (durable) + volatile index ─────────────────────────
#[repr(C)] pub struct RunHdr { size_class:u32, nslots:u32, bitmap_off:u64, next_run_off:u64 }
// bitmap: durable (INV-6); free list = volatile AllocIndex rebuilt from bitmaps
pub struct AllocIndex { /* per-class free slot lists; volatile; rebuilt on open */ }
// ── Errors ───────────────────────────────────────────────────────────────
pub enum PErr {
    NotDax, MapSyncUnsupported, BadMagic, ChecksumMismatch, FmtIncompatible{found:u16,need:u16},
    OutOfSpace, OidOutOfBounds, NotInTx, OverlapRange, DoubleFree, Poison, DomainMismatch,
    PhaseDeferred(&'static str),
}
// pflush: ensure [addr, addr+len) is scheduled to the durability domain.
//   ADR  : issues CLWB/CLFLUSHOPT/CLFLUSH over each covered 64B line.
//   eADR : no-op (caches already in domain).
//   Contract: caller MUST follow with pdrain() before relying on durability/order.
pub fn pflush(addr: *const u8, len: usize);

// pdrain: order all prior pflush/NT stores (SFENCE / DSB). Required on BOTH domains (INV-9).
pub fn pdrain();

// pmemcpy_nt: bulk copy via non-temporal stores; bypasses cache. Caller still pdrain()s.
//   Flushes EXACTLY the written range, nothing else (INV-20). Returns dst.
pub fn pmemcpy_nt(dst: *mut u8, src: *const u8, len: usize) -> *mut u8;

// pset8: the atom. Aligned 8B store + pflush(8) — caller pdrain()s to seal commit point.
pub fn pset8(p: *mut u64, v: u64);   // p MUST be 8-aligned (INV-15) else torn under powerfail
