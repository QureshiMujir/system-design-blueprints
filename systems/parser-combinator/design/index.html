pub trait Stream<'a>: Copy {
    type Item: Copy + Eq + 'a;
    type Position: Copy + Ord + 'a;
    type Slice: 'a;
    fn position(&self) -> Self::Position;
    fn is_empty(&self) -> bool;
    fn first(&self) -> Option<Self::Item>;
    fn split_at(&self, n: usize) -> Option<(Self::Slice, Self)>;
    fn strip_prefix(&self, expected: &[Self::Item]) -> Option<Self>;
    fn split_while<F: Fn(Self::Item) -> bool>(&self, pred: F) -> (Self::Slice, Self);
    fn span(&self, range: core::ops::Range<Self::Position>) -> ByteSpan;
}
pub type IResult<I, O, E = ParseError<I>> = Result<(I, O), Err<E>>;

pub enum Err<E> {
    Error(E),
    Failure(E),
    Incomplete(Needed),
}

pub enum Needed {
    Unknown,
    Size(core::num::NonZeroUsize),
}
pub struct ParseError<I> {
    pub input: I,
    pub expected: SmallVec<Expected, 4>,
    pub contexts: SmallVec<Context, 4>,
    pub kind: ErrorKind,
}

pub enum ErrorKind {
    Char(char), Tag(&'static [u8]), TakeWhile, TakeUntil, Eof, Digit, HexDigit, Alpha,
    Alphanumeric, Space, Multispace, OneOf, NoneOf, SeparatedList, ManyMN(usize, usize),
    RecursionLimit, Custom(u32),
}

pub enum Expected { Char(char), Bytes(&'static [u8]), Description(&'static str), EndOfInput }

pub struct Context { pub label: &'static str, pub span: ByteSpan }

pub trait FromError<I, X>: Sized {
    fn from_error_kind(input: I, kind: ErrorKind) -> Self;
    fn append(input: I, kind: ErrorKind, other: Self) -> Self;
    fn from_external(input: I, source: X) -> Self;
}
pub trait Parser<I, O, E> {
    fn parse(&mut self, input: I) -> IResult<I, O, E>;
    fn map<F: FnMut(O) -> O2, O2>(self, f: F) -> Map<Self, F, O> where Self: Sized;
    fn and<P2: Parser<I, O2, E>, O2>(self, p2: P2) -> And<Self, P2> where Self: Sized;
    fn or<P2: Parser<I, O, E>>(self, p2: P2) -> Or<Self, P2> where Self: Sized;
    fn cut(self) -> Cut<Self> where Self: Sized;
    fn labelled(self, label: &'static str) -> Labelled<Self> where Self: Sized;
}
pub struct ByteSpan { pub start: u32, pub end: u32, pub file: FileId }
pub struct FileId(pub u32);
pub struct LineColumn { pub line: u32, pub column: u32 }
pub struct MultiFileSource(/* opaque: file table + line-index cache, internal */);
pub struct Memo<O, E> { pub result: IResult<u32, O, E>, pub generation: u32 }

pub trait MemoTable<O, E> {
    fn get(&self, rule_id: u32, pos: u32) -> Option<&Memo<O, E>>;
    fn insert(&mut self, rule_id: u32, pos: u32, result: &IResult<u32, O, E>);
    fn clear(&mut self);
}
pub enum Assoc { Left, Right, NonAssoc }
pub struct Power(pub u8, pub u8);
pub enum Fixity { Prefix, Infix, Postfix }

pub struct Op<P, B> { pub matcher: P, pub builder: B, pub power: Power, pub assoc: Assoc, pub fixity: Fixity }

pub struct PrattClimber<I, O, E, A> { /* atoms: A; prefix/infix/postfix: Vec<Op<_,_>> */ }

impl<I, O, E, A> PrattClimber<I, O, E, A> {
    pub fn new(atoms: A) -> Self;
    pub fn prefix(self, matcher: impl Parser<I, (), E>, power: Power, builder: impl Fn(O) -> O) -> Self;
    pub fn infix(self, matcher: impl Parser<I, (), E>, power: Power, assoc: Assoc, builder: impl Fn(O, O) -> O) -> Self;
    pub fn postfix(self, matcher: impl Parser<I, (), E>, power: Power, builder: impl Fn(O) -> O) -> Self;
    pub fn parse(&mut self, input: I) -> IResult<I, O, E>;
}
pub struct BoxedParser<'a, I, O, E> { /* inner: Box<dyn FnMut(I) -> IResult<I, O, E> + 'a> */ }

impl<'a, I, O, E> BoxedParser<'a, I, O, E> {
    pub fn new<P: Parser<I, O, E> + 'a>(p: P) -> Self;
}

impl<'a, I, O, E> Parser<I, O, E> for BoxedParser<'a, I, O, E> {
    fn parse(&mut self, input: I) -> IResult<I, O, E>;
}

pub trait Reporter<I, E> {
    type Output;
    fn report(&self, error: &E, source: &MultiFileSource) -> Self::Output;
}
