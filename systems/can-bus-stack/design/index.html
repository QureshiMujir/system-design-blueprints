Core schema & types (§11)

pub enum CanId { Standard(StdId /* 11-bit, 0..=0x7FF */), Extended(ExtId /* 29-bit, 0..=0x1FFF_FFFF */) }
// Ownership: Copy. Mutation: immutable post-construct. Concurrency: Send+Sync (plain data).
// Serialization: canonical hex (3 or 8 nibbles). Compatibility: layout frozen at v1.

pub enum FrameFormat { Classical, Fd }
pub struct Dlc(u8 /* 0..=15 code */);
impl Dlc { pub fn len(self, fmt: FrameFormat) -> u8; pub fn from_len(len: u8, fmt: FrameFormat) -> Result<Dlc, FrameError>; }
// Contract: from_len rejects illegal lengths (e.g. 9 classical, 13 FD). len() is total per DLC table.

pub enum FrameKind { Data, Remote /* classical only */ }
pub struct Frame {
    pub id: CanId,
    pub kind: FrameKind,
    pub format: FrameFormat,
    pub dlc: Dlc,
    pub brs: bool,        // FD only
    pub esi: bool,        // FD only (set by controller)
    data: heapless::Vec<u8, 64>, // ≤8 classical, ≤64 FD; private to enforce DLC<->len
}
impl Frame {
    pub fn data(&self) -> &[u8];
    pub fn classical_data(id: CanId, data: &[u8]) -> Result<Frame, FrameError>; // ≤8
    pub fn fd_data(id: CanId, data: &[u8], brs: bool) -> Result<Frame, FrameError>; // legal FD len
}
pub enum FrameError { LengthTooLong, IllegalFdLength, RemoteWithData, IdOutOfRange }
pub struct BitTiming { pub brp: u16, pub prop_seg: u8, pub phase_seg1: u8, pub phase_seg2: u8, pub sjw: u8 }
impl BitTiming {
    pub fn nbt_tq(&self) -> u16;            // 1 + prop + ps1 + ps2
    pub fn sample_point_permille(&self) -> u16; // 0..1000
    pub fn validate(&self) -> Result<(), TimingError>; // sjw<=ps2, segs in range
}
pub enum TimingError { SjwTooLarge, SegmentOutOfRange, ZeroPhaseSeg }

pub struct BitBuffer { /* fixed-capacity bit vector, MSB-first */ }
impl BitBuffer { pub fn push(&mut self, bit: bool) -> Result<(), Overflow>; pub fn iter(&self) -> impl Iterator<Item=bool>; }

pub struct Stuffer { run_len: u8, last_bit: bool } // dynamic stuffing running state
impl Stuffer {
    pub fn stuff(input: &BitBuffer, out: &mut BitBuffer) -> Result<u16 /*inserted*/, Overflow>;
    pub fn unstuff(input: &BitBuffer, out: &mut BitBuffer) -> Result<(), StuffError>;
}
pub enum StuffError { SixIdenticalBits, Truncated }

pub enum CrcKind { Crc15, Crc17, Crc21 }
pub fn crc(kind: CrcKind, bits: &BitBuffer) -> u32;   // returns 15/17/21-bit value
pub fn fd_stuff_count(data_len: u8) -> (u8 /*count*/, bool /*parity*/);
// Concurrency: all pure or single-owner running state; no shared mutation.
pub enum ControllerState { ErrorActive, ErrorPassive, BusOff }
pub struct ErrorCounters { pub tec: u16, pub rec: u16 }
impl ErrorCounters {
    pub fn on_tx_error(&mut self);     // tec += 8 (with spec caveats)
    pub fn on_rx_error(&mut self);     // rec += 1 / +8 per rule
    pub fn on_tx_success(&mut self);   // tec -= 1 (floor 0)
    pub fn on_rx_success(&mut self);   // rec adjust per spec
    pub fn state(&self) -> ControllerState; // <128 active, >=128 passive, >=256 busoff
}
// Ownership: owned by controller; single-writer. Telemetry: every mutation emits an event (host).

pub enum CtrlEvent { TxComplete(CanId), RxComplete(Frame), ArbitrationLost, ErrorFrame(ErrorKind), StateChange(ControllerState), BusOffRecovered }
pub enum ErrorKind { Bit, Stuff, Crc, Form, Ack }

pub trait Controller {
    fn step_bit(&mut self, sampled: bool) -> Option<bool /*drive next bit*/>; // one bit-time
    fn queue_tx(&mut self, f: Frame) -> Result<MailboxId, TxFull>;
    fn poll_event(&mut self) -> Option<CtrlEvent>;
    fn counters(&self) -> ErrorCounters;
    fn state(&self) -> ControllerState;
}
pub enum AcceptanceFilter {
    Mask { id: u32, mask: u32, extended: bool },
    Range { lo: u32, hi: u32, extended: bool },
    List(heapless::Vec<u32, 16>),
}
impl AcceptanceFilter { pub fn accepts(&self, id: &CanId) -> bool; }

pub struct Mailbox<const N: usize> { /* RX FIFO of N frames */ }
pub enum OverrunPolicy { DropOldest, DropNewest, Overflow }
pub struct TxQueue<const N: usize> { /* priority-ordered by CAN ID ascending */ }
// Concurrency: single-owner in no_std; on host, sim mediates access. No interior mutability across threads.
pub enum BusLevel { Dominant /*0*/, Recessive /*1*/ }
pub trait Transceiver {
    fn drive(&mut self, level: BusLevel);          // put_bit
    fn sample(&self) -> BusLevel;                  // get_bit
    fn set_state(&mut self, state: ControllerState);
}
// Virtual impl: bound to can-bus node slot. SocketCAN impl: PhaseDeferred("§11.5").
// Ownership: one Transceiver per controller. Concurrency: virtual impl accessed only by scheduler.
pub struct VirtualBus { /* nodes, scheduler, trace, seeded rng */ }
pub struct BusNode { id: NodeId, ctrl: Box<dyn Controller>, tc: VirtualTransceiver }
impl VirtualBus {
    pub fn add_node(&mut self, ctrl: Box<dyn Controller>) -> NodeId;
    pub fn step(&mut self) -> BusBitTrace;     // advance all nodes one bit-time, wired-AND resolve
    pub fn run_until_idle(&mut self, max_steps: u64) -> RunReport;
    pub fn inject(&mut self, fault: Fault);    // force-bit, corrupt-crc, drop-ack
}
pub enum Fault { ForceBit { node: NodeId, bit: bool, at_step: u64 }, DropAck { node: NodeId }, CorruptCrc { node: NodeId } }
// Concurrency: single-threaded deterministic; the ONLY multi-node boundary. bus_bit = AND(all drives).
pub enum IsoTpPdu<'a> {
    Single { data: &'a [u8] },
    First { total_len: u32, data: &'a [u8] },
    Consecutive { sn: u8, data: &'a [u8] },
    FlowControl { fs: FlowStatus, bs: u8, stmin: StMin },
}
pub enum FlowStatus { ContinueToSend, Wait, Overflow }
pub enum StMin { Millis(u8 /*0..=127*/), Micros100(u8 /*1..=9 => 100..900us*/) }
pub enum Addressing { Normal, NormalFixed, Extended(u8), Mixed(u8) }
pub enum IsoTpState { Idle, SendingCf { remaining: u32, sn: u8 }, WaitFc, Reassembling { expected_sn: u8, remaining: u32 } }
pub enum IsoTpError { Timeout(TimerId), WrongSequence, BufferOverflow, UnexpectedPdu, Overflow }
pub enum TimerId { NAs, NBs, NCs, NAr, NBr, NCr }
// Ownership: one send + one receive context per addressing pair. Persistence: none (in-memory). Concurrency: single context.
pub struct UdsRequest { pub sid: u8, pub sub_function: Option<u8>, pub data: heapless::Vec<u8, 4095> }
pub enum UdsResponse {
    Positive { sid: u8 /* = req.sid + 0x40 */, data: heapless::Vec<u8, 4095> },
    Negative { sid: u8 /* 0x7F */, rejected_sid: u8, nrc: Nrc },
}
pub enum Nrc { GeneralReject=0x10, ServiceNotSupported=0x11, SubFunctionNotSupported=0x12,
    IncorrectMessageLengthOrInvalidFormat=0x13, ConditionsNotCorrect=0x22, RequestOutOfRange=0x31,
    SecurityAccessDenied=0x33, InvalidKey=0x35, ExceededNumberOfAttempts=0x36,
    RequestCorrectlyReceivedResponsePending=0x78, /* ... */ }
pub enum Session { Default, Programming, Extended }
pub struct SecurityState { level: u8, attempts: u8, locked_until: Option<Tick> }
// Ownership: server owns session+security; request-serialized. Timing: P2/P2* via injected virtual clock.
pub struct J1939Id { pub priority: u8 /*0..7*/, pub edp: bool, pub dp: bool, pub pf: u8, pub ps: u8, pub sa: u8 }
pub struct Pgn(pub u32 /* 18-bit effective */);
impl J1939Id { pub fn pgn(&self) -> Pgn; pub fn is_pdu1(&self) -> bool; pub fn to_ext(&self) -> ExtId; pub fn from_ext(id: ExtId) -> J1939Id; }
pub enum J1939Tp { Bam, Rts { total: u16, packets: u8 }, Cts { count: u8, next: u8 }, EndOfMsgAck, Abort(u8) }
pub struct Name(pub u64); // 64-bit address-claim NAME
// Ownership: stateless decode; per-session TP context; per-node claimed address. Concurrency: per-context.
