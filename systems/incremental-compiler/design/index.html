Core schema & types (§11)

// ---- query-engine ----
struct Revision(u64);                         // [mut] driver-only monotone; [ser] in manifest; [compat] stable
enum  Durability { Low, Medium, High }        // [ser] u8 tag; [compat] additive-only
struct QueryId(u16);                          // [own] static registry; identifies the query fn
struct QueryKey { qid: QueryId, key_hash: u128 } // [ser] 16B fixed; hash of args
struct Fingerprint(u128);                     // [ser] 16B; BLAKE3-128 of value bytes (ADR-12)
struct MemoEntry { key, value:Arc<dyn Any+Send+Sync>, fingerprint, verified_at:Revision,
                   changed_at:Revision, durability, deps:SmallVec<[QueryKey;8]> } // [conc] behind per-key lock (I8)
trait Database { fn revision(&self)->Revision; fn runtime(&self)->&Runtime; }
trait DerivedQuery { type Key; type Value; const ID:QueryId;
                     fn execute(db:&dyn Database, key:Self::Key)->Self::Value; } // [conc] must be pure (P1)
trait InputQuery  { type Key; type Value; fn set(db:&mut dyn Database, k:Self::Key, v:Self::Value); } // [mut] driver-only (P2)
struct Runtime { rev:AtomicU64, memo:MemoTable, dep_stack:ThreadLocal<Vec<QueryKey>>,
                 durabilities:[Revision;3] } // [conc] Sync; tracks last-changed per durability tier (P4)
struct MemoTable { shards:[Shard;N] }         // [conc] sharded lock-free reads, per-key write lock

// ---- dependency-graph ----
enum  Color { Green, Red, Unknown }           // [mut] per-revision validation marking
struct DependencyEdge { from:QueryKey, to:QueryKey, kind:EdgeKind } // to = dependency read by from
enum  EdgeKind { Value, Existence }           // Existence = "input present?" edge for early cutoff
struct CycleError { participants:Vec<QueryKey> } // [pers] emitted as diagnostic (I4)

// ---- syntax-green / syntax-red ----
struct SyntaxKind(u16);                        // token/node kind tag
struct Token { kind:SyntaxKind, len:u32, text:Interned } // [own] interned text (P5)
struct GreenNode { kind:SyntaxKind, text_len:u32, children:Arc<[GreenChild]>, fp:Fingerprint }
enum  GreenChild { Node(Arc<GreenNode>), Token(Token) } // [conc] Arc-shared, immutable (P5)
struct RedNode<'a> { green:&'a GreenNode, offset:u32, parent:Option<&'a RedNode<'a>> } // absolute pos overlay
// ---- resolver / interner ----
struct Interned(u32);                          // [own] global interner id; [compat] session-local
struct SymbolId(u64);                          // [own] stable across edits (P6, I9); content+scope derived
struct Scope { parent:Option<ScopeId>, bindings:FxHashMap<Interned,SymbolId> }
struct Span { file:FileId, lo:u32, hi:u32 }    // [ser] byte offsets; hygiene-stable (I9)

// ---- hir ----
struct HirId(u64); struct ItemId(u64);         // stable semantic ids (P6)
enum  Ty { Int(IntWidth), Bool, Unit, Fn(Box<[Ty]>,Box<Ty>), Error } // Error = poison, propagates
struct Hir { items:Vec<HirId>, bodies:FxHashMap<ItemId,BodyId> }

// ---- cache-store ----
struct CasKey([u8;32]);                        // BLAKE3-256 of value bytes (ADR-12); content address
struct IndexRecord { key:QueryKey, seg_id:u32, seg_off:u64, val_len:u32,
                     fingerprint:Fingerprint, durability:Durability, flags:u8 } // [pers] mmap index
enum  WalKind { Put, Del, Checkpoint }
struct WalRecord { seq:u64, kind:WalKind, key:QueryKey, payload_crc:u32, len:u32 } // [pers] fsync before index (I6)
struct CacheManifest { magic:u32, schema_version:u32, revision:Revision, root_index_off:u64, wal_seq:u64 } // [compat] I10

// ---- scheduler ----
struct TaskId(u64);
struct CancellationToken { flag:Arc<AtomicBool> } // [conc] cooperative (P14, I13)
struct QueryTask { id:TaskId, key:QueryKey, cancel:CancellationToken, span:tracing::Span }

// ---- diagnostics / driver / replay ----
enum  Severity { Error, Warning, Info, Hint }
struct Diagnostic { severity:Severity, span:Span, code:DiagCode, message:String, notes:Vec<Diagnostic> }
struct Session { db:Runtime, cache:CacheStore, sched:Scheduler, diag_sink:DiagSink } // [own] top object (P11)
struct ReplayLog { manifest, inputs:Vec<(FileId,Revision,Interned)>, expected:Vec<(QueryKey,Fingerprint)> } // [pers] I11
enum  QueryError { Cycle(CycleError), Cancelled, Corruption(QueryKey) } // error taxonomy
