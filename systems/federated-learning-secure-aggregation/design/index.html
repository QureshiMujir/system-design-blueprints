class Tensor:                                   # NumPy-backed value + grad (reverse-mode, 112)
    data: np.ndarray; grad: np.ndarray | None; requires_grad: bool
    def backward(self) -> None: ...             # reverse-mode autodiff
    def matmul(self, o: "Tensor") -> "Tensor": ...
    # add, relu, softmax_xent, etc.

@dataclass
class ModelState:                               # canonical param dict (ordered name -> array)
    params: "OrderedDict[str, np.ndarray]"
    version: int                                # monotonic (I11)
    def clone(self) -> "ModelState": ...
    def axpy(self, a: float, other: "ModelState") -> "ModelState": ...   # self + a*other
    def to_bytes(self) -> bytes: ...            # fixed dtype/shape/endianness (I15)
    @staticmethod
    def from_bytes(b: bytes) -> "ModelState": ...

def quantize(x: np.ndarray, B: float, R: int, stochastic: bool) -> np.ndarray: ...   # -> Z_R (§8.3)
def dequantize(q: np.ndarray, B: float, R: int) -> np.ndarray: ...
class SecureRng(Protocol):                      # CSPRNG; refuses fixed seed in prod (I7)
    def bytes(self, n: int) -> bytes: ...
    def randint_vec(self, R: int, m: int) -> np.ndarray: ...
class TranscriptRng(SecureRng): ...             # tests only; seeded, deterministic (P13)

def x25519_keypair(rng: SecureRng) -> tuple[bytes, bytes]: ...        # (pk, sk)  (RFC 7748)
def x25519_agree(sk: bytes, pk: bytes) -> bytes: ...                  # KA; symmetric (I5)
def prg_expand(seed: bytes, R: int, m: int) -> np.ndarray: ...        # AES-CTR/ChaCha -> Z_R^m
def kdf(master: bytes, label: bytes, n: int) -> bytes: ...            # domain-separated

def shamir_share(secret: bytes, t: int, n: int, rng: SecureRng) -> list["Share"]: ...   # (t,n) (I3)
def shamir_reconstruct(shares: list["Share"]) -> bytes: ...          # needs >= t+1 (I3)
@dataclass
class Share: x: int; y: bytes                    # evaluation point (i, p(i))

def ed25519_keypair(rng: SecureRng) -> tuple[bytes, bytes]: ...      # (RFC 8032)
def ed25519_sign(sk: bytes, msg: bytes) -> bytes: ...
def ed25519_verify(pk: bytes, msg: bytes, sig: bytes) -> bool: ...

def ring_add(a: np.ndarray, b: np.ndarray, R: int) -> np.ndarray: ...   # vector mod R
def ring_sub(a: np.ndarray, b: np.ndarray, R: int) -> np.ndarray: ...
class CryptoError(Exception): ...                # InsufficientShares, BadKey, InsecureRngRefused
def pairwise_mask(u: int, cohort: list[int], sk_u: bytes,
                  pubkeys: dict[int, bytes], R: int, m: int) -> np.ndarray: ...   # Σ ±PRG(s_uv)
def self_mask(b_u: bytes, R: int, m: int) -> np.ndarray: ...                      # PRG(b_u)
def mask_input(x_u: np.ndarray, u: int, cohort: list[int], sk_u: bytes,
               pubkeys: dict[int,bytes], b_u: bytes, R: int) -> np.ndarray: ...   # y_u (§8.4–8.5)

class SecAggRound(Enum): ADVERTISE_KEYS=0; SHARE_KEYS=1; MASKED_INPUT=2; UNMASK=3   # (P14)
class DropoutStatus(Enum): SURVIVED=0; DROPPED=1

@dataclass
class RoundConfig: cohort: list[int]; t: int; R: int; m: int; deadline_s: float    # (I15)

class SecAggServer(Protocol):
    def run_round(self, cfg: RoundConfig, transport: "Transport") -> np.ndarray: ...  # -> Σ x_u (I1)
    # internal: collect keys -> route shares -> collect y_u -> (U,D) -> reconstruct -> unmask
def reconstruct_and_unmask(masked_sum: np.ndarray, survivors: list[int], dropped: list[int],
                           self_shares: dict[int,list[Share]], key_shares: dict[int,list[Share]],
                           pubkeys: dict[int,bytes], R: int) -> np.ndarray: ...        # (§8.5, I2)
def check_consistency(dropout_set_hash: bytes, sigs: dict[int,bytes],
                      pubkeys: dict[int,bytes]) -> bool: ...                          # (I9)
class SecAggError(Exception): ...                # InsufficientParticipants, ProtocolViolation, Equivocation
class Transport(Protocol):
    def send(self, to: int, msg: "Message") -> None: ...
    def recv(self, frm: int, timeout_s: float) -> "Message | None": ...      # None on deadline (I10)
    def broadcast(self, msg: "Message") -> None: ...
    def stats(self) -> "CommStats": ...                                      # bytes/rounds (P9)
@dataclass
class CommStats: up_bytes: int; down_bytes: int; rounds: int; per_client: dict[int, tuple[int,int]]

class Compressor(Protocol):
    def encode(self, x: np.ndarray) -> bytes: ...        # quantize / top-k / random-mask (ADR-14)
    def decode(self, b: bytes) -> np.ndarray: ...
class Message(Protocol): tag: int                        # closed set; proto-backed (I15, P14)
class TransportError(Exception): ...                     # Malformed, Timeout, Closed, SchemaMismatch
@dataclass
class ClientUpdate: client_id: int; version: int; payload: np.ndarray; n_u: int    # masked under SecAgg
class Client(Protocol):
    def train_local(self, model: ModelState, cfg: "TrainConfig") -> np.ndarray: ... # -> Δ_u (clipped)
    def participate(self, cfg: RoundConfig, transport: Transport) -> None: ...      # masks + uploads y_u

class RoundState(Enum): SELECT=0; BROADCAST=1; LOCAL_TRAIN=2; COLLECT=3; AGGREGATE=4; UPDATE=5  # (P14)
class Server(Protocol):
    model: ModelState
    def select_cohort(self, q: float) -> list[int]: ...                  # sampling rate q (§8.8)
    def run_round(self) -> "RoundReport": ...                            # one FL round (P7)
    def run_campaign(self, rounds: int) -> ModelState: ...               # until rounds/budget (I13)
@dataclass
class RoundReport: t: int; cohort: int; survivors: int; dropped: int; agg_norm: float; eps_spent: float
class AggregationRule(Protocol):
    def aggregate(self, summed: np.ndarray, n_total: int, server_state: dict) -> ModelState: ...
class FedAvg(AggregationRule): ...               # weighted mean; Σ n_u/n = 1 (I12)
class FedProx(AggregationRule): mu: float        # proximal (client-side term too)
class FedOpt(AggregationRule): kind: str; eta: float; b1: float; b2: float; tau: float  # Adam/Yogi

### robust (cleartext-aggregation mode; needs individual updates -> SecAgg tension, P12)
def krum(updates: list[np.ndarray], f: int, multi: int = 1) -> np.ndarray: ...
def coordinate_median(updates: list[np.ndarray]) -> np.ndarray: ...
def trimmed_mean(updates: list[np.ndarray], beta: float) -> np.ndarray: ...
def norm_bound(update: np.ndarray, M: float) -> np.ndarray: ...          # SecAgg-compatible (client-side)
def clip_l2(g: np.ndarray, C: float) -> np.ndarray: ...                  # /max(1,‖g‖/C) (I14)
def gaussian_noise(shape, sigma: float, C: float, rng: SecureRng) -> np.ndarray: ...  # N(0,σ²C²)
def dpsgd_step(grads: list[np.ndarray], C: float, sigma: float, rng: SecureRng) -> np.ndarray: ...
def distributed_noise_share(sigma: float, C: float, n_clients: int, rng: SecureRng) -> np.ndarray: ...  # under-mask

class RDPAccountant:
    orders: list[float]                          # α grid
    def step(self, q: float, sigma: float) -> None: ...                  # subsampled-Gaussian RDP (§8.8)
    def get_epsilon(self, delta: float) -> float: ...                    # min_α conversion
@dataclass
class BudgetLedger: target_eps: float; delta: float; spent: float; rows: list[dict]
    def can_afford(self, q: float, sigma: float) -> bool: ...            # pre-round gate (I13)
def calibrate_sigma(target_eps: float, delta: float, q: float, rounds: int) -> float: ...  # σ for ε
