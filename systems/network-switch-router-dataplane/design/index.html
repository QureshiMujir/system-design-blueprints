Core schema & types (§11)

struct PacketBuffer {
    data_ptr: NonNull<u8>;      // owned; mutation: exclusive; concurrency: single-owner (I1)
    data_len: u16;              // owned; mutation: exclusive; not serialized
    headroom: u16;               // owned; mutation: exclusive at rewrite stage only
    queue_owner: QueueId;         // owned; mutation: move-only on queue handoff
    generation: u64;               // owned; mutation: write-once at RX; replay determinism key
}                                   // persistence: never persisted directly (pool-local only)

struct FibEntry {
    prefix: Ipv4Cidr;            // owned by ipv4_fib; immutable post-publish (P4, I2)
    adjacency: AdjacencyRef;       // owned; immutable post-publish; shares generation with self (I11)
    metric: u32;                    // owned; immutable post-publish
    generation: u64;                  // owned; monotonic (I9); compatibility: stable binary version tag
    flags: FibFlags;                    // owned; bitset: local | blackhole | ecmp_member
}                                         // serialization: bincode, versioned (ADR-05)

struct Adjacency {
    rewrite_mac: MacAddr;         // owned; immutable post-publish
    egress_queue: QueueId;          // owned; immutable post-publish
    vlan_action: VlanAction;          // owned; enum: none | push(Vid) | pop
    mtu: u16;                           // owned; immutable post-publish
}                                         // concurrency: read-only from worker cores (P4)

struct WorkerRuntime {
    rx_queues: Vec<QueueDescriptor>;   // owned; initialized once at startup (P7)
    worker_graph: PipelineGraph;         // owned; immutable after init
    telemetry: TelemetryHandle;            // owned; lock-free append-only (I12)
    local_mempool: MempoolHandle;             // owned; NUMA-local (P6)
}                                               // lifecycle: init once, crash-only restart (P7)
struct QueueDescriptor {
    ring_ptr: NonNull<Descriptor>;    // owned by dpdk_hal/afxdp_hal; concurrency: SPSC per queue
    producer_index: AtomicU32;          // owned; mutation: producer core only
    consumer_index: AtomicU32;            // owned; mutation: consumer core only
    watermark: u32;                         // owned; mutation: consumer core only; drives backpressure (P3)
    stats: QueueStats;                        // owned; lock-free counters
}                                               // invariant: producer never overtakes consumer mod capacity (I6)

struct AclDagNode {
    kind: AclNodeKind;                // owned; enum: match | branch | permit | deny | mirror
    match_field: Option<FieldSelector>;  // owned; compile-time only, immutable post-compile (I7)
    on_match: Option<NodeId>;              // owned; immutable post-compile
    on_miss: Option<NodeId>;                 // owned; immutable post-compile
}                                              // serialization: bincode; compatibility: DAG-version tagged

struct ArpCacheEntry {
    ip: Ipv4Addr;                     // owned; mutation: single-writer per shard (I8)
    mac: Option<MacAddr>;               // owned; None while unresolved
    retry_count: u8;                      // owned; bounded (I10)
    state: ArpState;                        // owned; enum: pending | resolved | exhausted
}                                             // persistence: not persisted; rebuilt on restart

struct WalRecord {
    seq: u64;                          // owned; monotonic, gapless (I5)
    generation: u64;                     // owned; matches config_store commit id (I9)
    payload: ConfigDelta;                  // owned; bincode-serialized (ADR-05)
    checksum: u32;                           // owned; CRC32C, verified on replay
}                                              // compatibility: forward-compatible via payload version tag

struct ReplayRecord {
    record_header: RecordHeader;         // owned; magic + version + length
    packet_metadata: PacketMeta;            // owned; ingress port, VLAN, queue_id
    rx_timestamp: u64;                         // owned; TSC-calibrated (ADR-20)
    packet_bytes: Vec<u8>;                       // owned; PCAP-NG compatible payload (ADR-26)
    forwarding_decision: Decision;                // owned; permit | deny | drop(reason)
    worker_id: WorkerId;                            // owned; provenance for divergence triage (I4)
}                                                      // persistence: append-only, PCAP-NG + digest sidecar
