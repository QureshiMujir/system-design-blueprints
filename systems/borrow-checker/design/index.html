// crates/bc-ir/src/mir.rs — part 1
pub struct Body {
    pub locals: Vec<LocalDecl>,
    pub blocks: Vec<BasicBlock>,
    pub entry: BlockId,
    pub return_local: LocalId,
    pub args: Vec<LocalId>,
    pub generics: Generics,
}

pub struct LocalDecl {
    pub id: LocalId,
    pub ty: Ty,
    pub mutability: Mutability,
    pub span: Span,
}

pub struct BasicBlock {
    pub id: BlockId,
    pub statements: Vec<Statement>,
    pub terminator: Terminator,
}

pub enum Statement {
    Assign { place: Place, rvalue: Rvalue, span: Span },
    StorageLive(LocalId),
    StorageDead(LocalId),
    Nop,
}
// crates/bc-ir/src/mir.rs — part 2
pub enum Rvalue {
    Use(Operand),
    Ref { region: RegionVar, kind: BorrowKind, place: Place },
    Repeat(Operand, ConstId),
    BinaryOp(BinOp, Operand, Operand),
    UnaryOp(UnOp, Operand),
    Cast(CastKind, Operand, Ty),
    Aggregate(AggregateKind, Vec<Operand>),
    NullaryOp(NullaryOp, Ty),
}

pub enum Terminator {
    Goto { target: BlockId },
    SwitchInt { discr: Operand, targets: Vec<(i128, BlockId)>, default: BlockId },
    Return,
    Unreachable,
    Drop { place: Place, target: BlockId, unwind: Option<BlockId> },
    Call {
        func: Operand,
        args: Vec<Operand>,
        destination: Place,
        target: Option<BlockId>,
        unwind: UnwindAction,
    },
    Assert { cond: Operand, expected: bool, msg: AssertMessage, target: BlockId, unwind: UnwindAction },
}

pub enum UnwindAction {
    Continue,
    Cleanup(BlockId),
    Unreachable,
    Terminate,
}
// crates/bc-ir/src/place.rs
pub struct Place {
    pub local: LocalId,
    pub projection: SmallVec<[ProjectionElem; 4]>,
}

pub enum ProjectionElem {
    Deref,
    Field { index: usize, ty: Ty },
    Index(LocalId),
    Downcast(VariantIdx),
}

pub enum BorrowKind {
    Shared,
    Shallow,
    Unique,
    Mut { kind: MutBorrowKind },
}

pub enum MutBorrowKind {
    Default,
    TwoPhaseBorrow,
    ClosureCapture,
}
// crates/bc-regions/src/region.rs
#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, PartialOrd, Ord)]
pub struct RegionVar(pub u32);

pub enum Region {
    Var(RegionVar),
    Static,
    Empty,
    Concrete(ConcreteRegion),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct ConcreteRegion {
    pub points: BTreeSet<ProgramPoint>,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct ProgramPoint {
    pub block: BlockId,
    pub statement: u32,
    pub kind: PointKind,    // Before | After
}

pub struct RegionInferenceCtxt {
    pub constraints: ConstraintSet,
    pub regions: BTreeMap<RegionVar, ConcreteRegion>,
    pub graph: ConstraintGraph,
}

impl RegionInferenceCtxt {
    pub fn solve(&mut self) -> Result<RegionSolution, RegionError>;
}

pub struct RegionSolution {
    pub assignments: BTreeMap<RegionVar, ConcreteRegion>,
}
// crates/bc-constraints/src/lib.rs
pub enum Constraint {
    Outlives { sup: RegionVar, sub: RegionVar, at: ProgramPoint, cause: Cause },
    Member { region: RegionVar, place: Place, at: ProgramPoint, cause: Cause },
    Use { region: RegionVar, at: ProgramPoint, cause: Cause },
}

pub struct ConstraintSet {
    pub constraints: Vec<Constraint>,
    pub region_universes: BTreeMap<RegionVar, UniverseIndex>,
}

pub struct Cause {
    pub span: Span,
    pub kind: CauseKind,
}

pub enum CauseKind {
    UseOf(Place),
    BorrowOf(Place),
    AssignmentInto(Place),
    CallArgument(usize),
    ReturnValue,
    Drop(Place),
}
// crates/bc-check/src/lib.rs
pub struct BorrowChecker<'a> {
    pub body: &'a Body,
    pub regions: &'a RegionSolution,
    pub borrows: BorrowSet,
}

pub struct BorrowSet {
    pub borrows: Vec<BorrowData>,
    pub by_local: BTreeMap<LocalId, BTreeSet<BorrowIndex>>,
}

pub struct BorrowData {
    pub index: BorrowIndex,
    pub kind: BorrowKind,
    pub region: RegionVar,
    pub place: Place,
    pub created_at: ProgramPoint,
    pub two_phase: Option<TwoPhaseState>,
}

pub enum TwoPhaseState {
    Reserved,
    Active { at: ProgramPoint },
}
// crates/bc-moves/src/lib.rs
pub enum InitState {
    Unallocated,
    Uninitialized,
    DefinitelyInitialized,
    MaybeInitialized,
    DefinitelyMoved,
    Dead,
}

pub struct MoveAnalysis {
    pub states: BTreeMap<ProgramPoint, BTreeMap<LocalId, InitState>>,
}

impl MoveAnalysis {
    pub fn run(body: &Body) -> MoveAnalysis;
}
