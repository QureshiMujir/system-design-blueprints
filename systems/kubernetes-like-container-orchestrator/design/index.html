Core schema & types (§11)

/// GroupVersionKind discriminant, flattened into every object's JSON.
pub struct TypeMeta { pub api_version: String, pub kind: String } // e.g. "v1","Pod"

/// Identity + lineage + lifecycle metadata shared by ALL kinds.
pub struct ObjectMeta {
    pub name: String,                       // unique within (namespace, kind)
    pub namespace: Option<String>,          // None => cluster-scoped (Node, Namespace)
    pub uid: Uuid,                           // server-assigned, immutable, identity across name reuse
    pub resource_version: String,           // == store mod_rev; opaque; equality-compare only
    pub generation: i64,                     // bumped on spec change; status echoes observedGeneration
    pub labels: BTreeMap<String,String>,     // selector key space; BTree => deterministic JSON/hash
    pub annotations: BTreeMap<String,String>,// non-selectable metadata
    pub owner_references: Vec<OwnerReference>,// GC + adoption lineage
    pub finalizers: Vec<String>,             // block deletion until drained
    pub creation_timestamp: Time,
    pub deletion_timestamp: Option<Time>,    // Some => terminating; object still readable until finalizers clear
}

/// Every kind implements Object => generic Registry/Informer over `dyn Object`.
pub trait Object {
    fn type_meta(&self) -> &TypeMeta;
    fn meta(&self) -> &ObjectMeta;
    fn meta_mut(&mut self) -> &mut ObjectMeta;
}
pub struct Pod    { pub meta: ObjectMeta, pub spec: PodSpec, pub status: PodStatus }

pub struct PodSpec {
    pub containers: Vec<Container>,          // >=1; validated non-empty
    pub volumes: Vec<Volume>,
    pub node_name: Option<String>,           // None => unscheduled; set EXACTLY ONCE by Bind
    pub restart_policy: RestartPolicy,       // Always | OnFailure | Never
    pub node_selector: BTreeMap<String,String>,
    pub tolerations: Vec<Toleration>,
    pub affinity: Option<Affinity>,          // node/pod (anti)affinity rules
}

pub struct PodStatus {
    pub phase: PodPhase,
    pub conditions: Vec<Condition>,          // PodScheduled, Ready, ContainersReady, Initialized
    pub container_statuses: Vec<ContainerStatus>,
    pub pod_ip: Option<IpAddr>,              // assigned by NetworkPlugin
}

pub enum PodPhase { Pending, Running, Succeeded, Failed, Unknown }
pub struct ReplicaSet { pub meta: ObjectMeta, pub spec: ReplicaSetSpec, pub status: ReplicaSetStatus }
pub struct ReplicaSetSpec {
    pub replicas: i32,                       // desired N
    pub selector: LabelSelector,             // MUST match template labels (validated)
    pub template: PodTemplateSpec,           // pod stamp; mutating template != rolling (Deployment's job)
}
pub struct ReplicaSetStatus { pub replicas: i32, pub ready_replicas: i32, pub observed_generation: i64 }

/// spec.strategy: RollingUpdate{max_surge, max_unavailable} | Recreate.
/// Owns a sequence of ReplicaSets (one per pod-template revision) via OwnerReferences.
pub struct Deployment { pub meta: ObjectMeta, pub spec: DeploymentSpec, pub status: DeploymentStatus }

/// spec.selector -> EndpointSlice; spec.ports; spec.cluster_ip (VIP, alloc'd from CIDR);
/// spec.type: ClusterIP | NodePort.
pub struct Service { pub meta: ObjectMeta, pub spec: ServiceSpec, pub status: ServiceStatus }

/// Cluster-scoped. status.capacity / allocatable (cpu, memory, pods);
/// status.conditions (Ready, MemoryPressure, DiskPressure); spec.taints.
pub struct Node { pub meta: ObjectMeta, pub spec: NodeSpec, pub status: NodeStatus }
/// Conjunction of equality labels AND set-based expressions. Empty selector matches ALL.
pub struct LabelSelector {
    pub match_labels: BTreeMap<String,String>,
    pub match_expressions: Vec<Requirement>, // op: In | NotIn | Exists | DoesNotExist
}

/// Lineage edge: child -> owner. controller=true marks the single managing controller.
pub struct OwnerReference {
    pub uid: Uuid,                           // owner identity; stale uid => orphan (owner gone)
    pub kind: String,
    pub controller: bool,                    // exactly one controller=true owner per child
    pub block_owner_deletion: bool,          // owner delete blocked until child cleared (finalizer)
}

/// Standard status condition (mirrors k8s metav1.Condition).
pub struct Condition {
    pub type_: String, pub status: ConditionStatus, // True|False|Unknown
    pub reason: String, pub message: String,
    pub last_transition_time: Time, pub observed_generation: i64,
}
pub struct Revision(pub i64);              // global monotonic logical clock; 0 = "before any write"

pub struct KeyValue {
    pub key: Vec<u8>, pub value: Vec<u8>,  // value = JSON-encoded object bytes
    pub create_rev: Revision,              // rev of first Put for this key (since last delete)
    pub mod_rev: Revision,                 // rev of last Put == object.resource_version
    pub version: i64,                      // per-key write count since create (1-based)
    pub lease: Option<LeaseId>,            // Some => key auto-deleted when lease expires
}

pub enum WatchEvent {
    Put(KeyValue, Option<KeyValue>),       // new + prev (None on create)
    Delete(KeyValue),                      // the tombstoned kv (last value)
    Bookmark(Revision),                    // progress marker; no data change up to this rev
}
pub trait Store {
    /// Range scan [key, end) at optional historical rev; returns kvs + current store rev (for watch start).
    async fn range(&self, key: &[u8], end: &[u8], rev: Option<Revision>) -> Result<RangeResult, StoreError>;
    /// Blind write; returns the new global Revision. Use txn() for CAS.
    async fn put(&self, key: &[u8], val: &[u8], lease: Option<LeaseId>) -> Result<Revision, StoreError>;
    /// Delete [key,end); tombstone bumps Revision; returns it.
    async fn delete(&self, key: &[u8], end: &[u8]) -> Result<Revision, StoreError>;
    /// Atomic compare-and-set: if all `compares` hold, apply on_ok else on_else. Basis of optimistic concurrency.
    async fn txn(&self, compares: Vec<Compare>, on_ok: Vec<Op>, on_else: Vec<Op>) -> Result<TxnResult, StoreError>;
    /// Ordered, gap-free, dup-free stream of WatchEvent with rev > start_rev; 410-Expired if start_rev compacted.
    fn watch(&self, key: &[u8], end: &[u8], start_rev: Revision) -> WatchStream;
    /// TTL lease; keys attached to it die on expiry. Used for Node heartbeats.
    async fn lease_grant(&self, ttl: Duration) -> Result<LeaseId, StoreError>;
    async fn lease_keepalive(&self, id: LeaseId) -> Result<(), StoreError>;
    /// Drop history <= rev; subsequent watch(start_rev<=rev) -> 410. Reclaims space.
    async fn compact(&self, rev: Revision) -> Result<(), StoreError>;
}

pub struct Compare { pub key: Vec<u8>, pub target: CompareTarget } // Value | ModRevision | CreateRevision | Version
pub enum Op { Put { key: Vec<u8>, val: Vec<u8>, lease: Option<LeaseId> }, Delete { key: Vec<u8> }, Range { key: Vec<u8> } }

pub struct Lease { pub id: LeaseId, pub ttl: Duration, pub granted_at: Time, pub attached: Vec<Vec<u8>> }
/// Level-triggered, IDEMPOTENT. Reads CURRENT desired+observed from cache, drives toward desired.
/// Never acts on an event payload; the Request carries only the object KEY.
pub trait Reconciler {
    async fn reconcile(&self, req: Request, ctx: &Context) -> Result<Reconcile, ReconcileError>;
}
pub enum Reconcile { Done, RequeueAfter(Duration), Requeue } // requeue feeds the rate limiter

pub struct Request { pub namespace: Option<String>, pub name: String } // identity only — re-read from Lister

/// Reflector(list+watch) -> DeltaFIFO -> applies to a thread-safe indexed cache + fans out to handlers.
/// Shared: many controllers attach to ONE informer per kind (single watch connection).
pub struct SharedIndexInformer<T> { /* reflector + delta FIFO + Arc<RwLock<indexed cache>> + indexers */ }

/// Read-only typed view over the informer cache. Cheap clones via Arc; never blocks the watch loop.
pub struct Lister<T> { /* fn get(ns,name) -> Option<Arc<T>>; fn list(selector) -> Vec<Arc<T>> */ }

/// Rate-limited, de-duplicated, optionally delayed work items. At most one in-flight per key.
pub struct WorkQueue<K> { /* add / add_after / add_rate_limited / get -> K / done(K) / forget(K) */ }

/// Per-key backoff schedule (e.g. exponential 5ms..1000s) for failed reconciles.
pub trait RateLimiter<K> {
    fn when(&self, item: &K) -> Duration; // delay before next attempt
    fn forget(&self, item: &K);           // reset backoff on success
}
/// Per-pod scheduling cycle state: the pod, a node Lister, the scheduler cache, and plugin scratch state.
pub struct SchedulingContext<'a> { /* pod: &Pod, nodes: &Lister<Node>, cache: &Cache, state: PluginState */ }

/// Hard predicate: is this Node feasible for this Pod? Any reject => Node excluded.
pub trait FilterPlugin {
    fn filter(&self, pod: &Pod, node: &Node, ctx: &SchedulingContext) -> FilterResult; // Pass | Reject(reason)
}
/// Soft priority: rank a feasible Node 0..100. Plugin scores are weighted-summed.
pub trait ScorePlugin {
    fn score(&self, pod: &Pod, node: &Node, ctx: &SchedulingContext) -> i64; // 0..=100
}
/// Result of a cycle: the write that assigns the Pod to a Node.
pub struct Binding { pub pod: ObjectRef, pub node: String }
/// One PodWorker per Pod bound to this Node; owns that Pod's lifecycle state machine end-to-end.
pub struct PodWorker { /* pod uid, desired spec, last status, runtime handle, sync channel */ }

/// Outcome of one sync; drives the kubelet's per-pod loop.
pub enum PodSyncResult {
    Synced(PodStatus),     // converged; report this status to the apiserver
    Backoff(Duration),     // container crash -> retry after backoff
    Terminated,            // pod fully stopped + cleaned up
}

/// Liveness / readiness / startup probing.
pub trait ProbeManager {
    async fn probe(&self, p: &Probe, c: &ContainerStatus) -> ProbeResult; // Success | Failure | Unknown
}
pub trait ContainerRuntime {
    /// Create the pod sandbox (shared net/IPC namespaces + pause-equivalent); returns its id.
    async fn run_sandbox(&self, pod: &PodSandboxConfig) -> Result<SandboxId, RuntimeError>;
    /// Pull + verify + unpack an OCI image into the ImageStore; idempotent on digest.
    async fn pull_image(&self, image: &ImageRef, auth: Option<&Auth>) -> Result<ImageId, RuntimeError>;
    /// Create (not start) a container in an existing sandbox from an unpacked image.
    async fn create_container(&self, sb: SandboxId, c: &ContainerConfig) -> Result<ContainerId, RuntimeError>;
    async fn start_container(&self, id: ContainerId) -> Result<(), RuntimeError>;
    /// Stop with grace period (SIGTERM then SIGKILL at deadline).
    async fn stop_container(&self, id: ContainerId, grace: Duration) -> Result<(), RuntimeError>;
    async fn status(&self, id: ContainerId) -> Result<ContainerStatus, RuntimeError>;
}

/// OCI runtime config: root fs, process, namespaces, cgroups v2 path, mounts, capabilities.
pub struct OciSpec { /* root, process(args,env,cwd,caps), linux{namespaces, resources(cgroups v2), mounts}, hooks */ }

/// A running pod sandbox: shared namespaces + the assigned PodIp; containers join it.
pub struct Sandbox { pub id: SandboxId, pub netns: NetnsHandle, pub pod_ip: Option<PodIp>, pub cgroup: CgroupHandle }

/// Content-addressed image storage: digest -> unpacked rootfs layers + config.
pub struct ImageStore { /* layer cache keyed by sha256 digest; ref-counted; gc on unused */ }
/// Programs the dataplane: Service ClusterIP VIP -> healthy EndpointSlice backends (+ NodePort).
pub struct ServiceProxy { /* watches Service + EndpointSlice; reprograms rules on change */ }

/// Pod-network lifecycle (CNI-like): attach/detach a Sandbox to the cluster pod network.
pub trait NetworkPlugin {
    async fn setup_pod_network(&self, sb: &Sandbox) -> Result<PodIp, NetError>;  // alloc IP + wire netns
    async fn teardown_pod_network(&self, sb: &Sandbox) -> Result<(), NetError>;  // release IP + cleanup
}
