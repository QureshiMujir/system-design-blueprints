Core schema & types (§11)

// crates/dlcore/src/decision.rs — ownership: decision-store; mutation: append-only new
// versions only; concurrency: single-writer per repo (local), merged at team-backend;
// persistence: op-log entries; serialization: protobuf (ADR-09); compatibility: additive
// fields only, versioned schema tag.
pub struct DecisionId(ContentHash);

pub struct Decision {
    pub id: DecisionId,
    pub title: String,
    pub rationale: String,
    pub status: DecisionStatus,          // lifecycle state, §8
    pub scope: DecisionScope,            // Anchored | RepoGlobal
    pub anchors: Vec<Anchor>,
    pub edges: Vec<DecisionEdge>,
    pub assertion: Option<AssertionRule>,
    pub repo_id: RepoId,
    pub created_at: LogicalTimestamp,
    pub author: ActorId,
}

pub enum DecisionStatus { Draft, Active, Drifted, Superseded, Retired, OrphanedPendingReview }
pub enum DecisionScope { Anchored, RepoGlobal }
// crates/dlcore/src/anchor.rs — ownership: anchor-resolver (read), decision-store (write
// via repair); mutation: replaced wholesale on repair, never field-patched in place;
// concurrency: read-many/write-single per file during reparse; persistence: embedded in
// Decision; serialization: protobuf; compatibility: AnchorKind is a closed enum, additive
// variants require a schema version bump.
pub struct Anchor {
    pub file_path: RepoPath,
    pub symbol_ref: Option<SymbolRef>,
    pub ast_path: Option<AstPath>,
    pub content_hash: ContentHash,
    pub fuzzy_window: Option<FuzzyWindow>,
    pub kind: AnchorKind,                // resolved precedence tier, §8
}

pub enum AnchorKind { SymbolRef, AstPath, ContentHash, FuzzyWindow, Orphaned }

pub struct AstPath(pub Vec<AstStep>);     // root-relative node-type+index sequence
pub struct AstStep { pub node_type: &'static str, pub child_index: u16 }

pub struct SymbolRef { pub qualified_name: String, pub kind: SymbolKind }
pub struct ContentHash(pub [u8; 32]);     // blake3 digest, ADR-06
// crates/dlcore/src/anchor.rs (cont.) — ResolutionResult: ownership: anchor-resolver
// (produced fresh per call, never cached across edits per P1); mutation: immutable value;
// concurrency: safe to pass across threads (Send); persistence: not persisted, ephemeral;
// serialization: n/a (internal); compatibility: n/a (internal type).
pub enum ResolutionResult {
    Resolved { anchor: Anchor, confidence: Confidence },
    Ambiguous { candidates: Vec<RepairCandidate> },
    Orphaned { reason: OrphanReason },
}

pub enum Confidence { Exact, High, Medium, Low }
pub struct RepairCandidate { pub anchor: Anchor, pub tier: AnchorKind, pub score: f32 }
// crates/dlcore/src/drift.rs — ownership: drift-detector; mutation: append-only, one
// event per detected transition; concurrency: produced per drift-scan worker, merged
// single-threaded into decision-store; persistence: op-log entry; serialization: protobuf;
// compatibility: DriftKind closed enum, additive requires schema bump.
pub struct DriftEvent {
    pub decision_id: DecisionId,
    pub anchor: Anchor,
    pub kind: DriftKind,
    pub detected_at: LogicalTimestamp,
}

pub enum DriftKind { Moved, Renamed, ModifiedInPlace, Deleted, SemanticallyContradicted }
// crates/dlcore/src/decision.rs (cont.) — DecisionEdge: ownership: decision-store;
// mutation: append-only; concurrency: merged via op-log total order (P12); persistence:
// op-log entry; serialization: protobuf; compatibility: EdgeKind closed enum.
pub struct DecisionEdge { pub from: DecisionId, pub to: DecisionId, pub kind: EdgeKind }
pub enum EdgeKind { Supersedes, DependsOn, Contradicts }
// crates/dlcore/src/replay.rs — ReplayCursor: ownership: replay-engine; mutation:
// monotonically advances, never rewinds except for verification re-runs; concurrency:
// single-threaded fold over op-log; persistence: checkpointed periodically; serialization:
// protobuf; compatibility: cursor format versioned alongside op-log schema.
pub struct ReplayCursor { pub repo_id: RepoId, pub op_offset: u64, pub graph_hash: ContentHash }
// crates/sync-client/src/lib.rs — SyncOp: ownership: sync-client (produced), team-backend
// ingestion (consumed); mutation: immutable once created; concurrency: batched, sent
// out-of-order-tolerant via logical clock (I9); persistence: durable until ack'd;
// serialization: protobuf, ≤64KB median (§4); compatibility: additive fields, actor_id and
// lamport_ts are permanently required.
pub struct SyncOp {
    pub decision_id: DecisionId,
    pub op_type: OpType,
    pub lamport_ts: u64,
    pub actor_id: ActorId,
    pub payload: OpPayload,
}
pub enum OpType { Create, Repair, DriftDetected, StatusChange, EdgeAdd }
// crates/compliance-exporter/src/lib.rs — ExportBundle: ownership: compliance-exporter;
// mutation: immutable once generated; concurrency: generated per request, stateless;
// persistence: written to object storage; serialization: SPDX/CycloneDX-adjacent JSON
// (ADR-24); compatibility: schema-versioned, must remain independently re-verifiable (P11).
pub struct ExportBundle {
    pub cursor: ReplayCursor,
    pub decisions: Vec<Decision>,
    pub provenance: Vec<AnchorProvenance>,
    pub bundle_hash: ContentHash,
}
pub struct AnchorProvenance { pub anchor: Anchor, pub resolved_at: LogicalTimestamp, pub confidence: Confidence }
// error enums — ownership: respective module; mutation: immutable; concurrency: Send+Sync;
// persistence: surfaced in telemetry/logs, never persisted as decision state; serialization:
// Display + tracing-compatible; compatibility: additive variants only.
pub enum AnchorError { InvalidPath, MalformedHash, UnsupportedSymbolKind }
pub enum DriftError { OrphanedEventInput, AssertionEvalFailure(String) }
pub enum SyncError { ClockRegression, PayloadTooLarge, TenantIsolationViolation }
pub enum ExportError { CursorMismatch, IncompleteProvenance }
