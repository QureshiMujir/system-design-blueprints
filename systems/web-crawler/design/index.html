/// The fundamental unit of the URL frontier.
/// Every URL that enters the system is wrapped in a UrlRecord.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct UrlRecord {
    /// The canonical, normalized URL (see §8 normalization pipeline).
    pub url: Url,
    /// Discovery depth (0 = seed URL, 1 = link found on seed page, etc.).
    pub depth: u32,
    /// Priority score: higher = fetch sooner. Default = 0.0.
    /// Phase 1: overridden by PageRank estimate or OPIC score.
    pub score: f32,
    /// Extracted host (lowercase, no port) for per-domain scheduling.
    pub domain: Arc<str>,
    /// Current lifecycle status.
    pub status: UrlStatus,
    /// Earliest time this URL should next be fetched (Unix milliseconds).
    /// Set to the scheduling timestamp + crawl_delay * politeness_multiplier.
    pub next_fetch_at: i64,
    /// Number of times this URL has been attempted (for retry backoff).
    pub attempt_count: u32,
    /// SHA-256 of the most recently fetched body (for change detection).
    pub content_hash: Option<[u8; 32]>,
}

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
pub enum UrlStatus {
    Queued,
    Fetching,
    Fetched,
    Failed { reason: FetchErrorKind },
    Disallowed,
    Skipped { reason: SkipReason },
}

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
pub enum SkipReason {
    BloomFilterHit,         // URL already seen (exact dedup)
    ContentTypeMismatch,    // Not text/html
    MaxDepthExceeded,
    DomainBlacklisted,
}
/// Sent from the scheduler to a fetcher worker.
#[derive(Debug)]
pub struct FetchTask {
    pub record: UrlRecord,
    pub user_agent: Arc<str>,
    pub timeout: Duration,
    pub max_redirects: u8,
}

/// Returned from a fetcher worker to the coordinator.
#[derive(Debug)]
pub enum FetchResult {
    Success(FetchResponse),
    Error(FetchError),
}

#[derive(Debug)]
pub struct FetchResponse {
    /// The URL after following all redirects.
    pub final_url: Url,
    pub status: StatusCode,
    pub headers: HeaderMap,
    /// Raw response body bytes (before charset decoding).
    pub body: Bytes,
    /// Character encoding detected from Content-Type header.
    pub charset: Charset,
    /// Wall clock time at fetch start.
    pub fetched_at: DateTime<Utc>,
    /// Round-trip time for this fetch.
    pub latency: Duration,
    /// Redirect chain (original URL → final URL).
    pub redirect_chain: Vec<Url>,
}
#[derive(Debug, thiserror::Error)]
pub enum FetchError {
    #[error("DNS resolution failed for {host}: {source}")]
    Dns { host: String, source: Box<dyn std::error::Error + Send + Sync> },
    #[error("TCP connection failed: {0}")]
    Tcp(#[from] std::io::Error),
    #[error("TLS handshake failed: {0}")]
    Tls(String),
    #[error("Request timed out after {elapsed:?}")]
    Timeout { elapsed: Duration },
    #[error("Too many redirects ({count})")]
    TooManyRedirects { count: u8 },
    #[error("HTTP error {status}: {url}")]
    HttpStatus { status: StatusCode, url: Url },
    #[error("Response body too large ({size} bytes)")]
    BodyTooLarge { size: usize },
}
/// Parsed and cached rules for a single domain.
#[derive(Debug, Clone)]
pub struct RobotsRules {
    /// Selected rules for this crawler's User-agent.
    pub rules: Vec<RobotsRule>,
    /// Crawl-delay in seconds (None = not specified; use config default).
    pub crawl_delay: Option<f64>,
    /// Sitemap URLs declared in robots.txt.
    pub sitemaps: Vec<Url>,
    /// Timestamp when this robots.txt was fetched.
    pub fetched_at: Instant,
    /// Cache TTL. Re-fetch after this duration.
    pub ttl: Duration,
}

#[derive(Debug, Clone)]
pub struct RobotsRule {
    pub kind: RuleKind,
    /// The path prefix/pattern. May include `*` and `$` wildcards.
    pub pattern: String,
    /// Length of the pattern (used for precedence: longer wins).
    pub pattern_len: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RuleKind { Allow, Disallow }

/// Verdict of allow/disallow evaluation.
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RobotsVerdict {
    Allowed,
    Disallowed,
    /// robots.txt fetch failed transiently; treat as disallowed with backoff.
    Unavailable,
}
/// Two-stage deduplication engine.
pub struct DedupEngine {
    bloom: BloomFilter,
    simhash_index: SimHashIndex,
}

/// Counting Bloom filter using k=7 independent hash functions.
/// Backed by a bitvec stored in a memory-mapped file for fast restart.
pub struct BloomFilter {
    bits: MmapMut,
    num_bits: u64,
    num_hashes: u32,
    /// SipHash seeds for each of the k hash functions.
    seeds: [(u64, u64); 7],
}

impl BloomFilter {
    /// Returns false if the URL has definitely not been seen.
    /// Returns true if the URL has probably been seen (1% FPR).
    pub fn probably_seen(&self, url: &[u8]) -> bool;
    pub fn insert(&mut self, url: &[u8]);
    /// Serialize to the mmap file; flushes on graceful shutdown.
    pub fn flush(&mut self) -> io::Result<()>;
}

/// 64-bit SimHash fingerprint for a document.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub struct SimHash(pub u64);

impl SimHash {
    /// Hamming distance between two fingerprints (`popcount(self ^ other)`).
    pub fn distance(self, other: SimHash) -> u32;
    /// True iff `distance(self, other) <= simhash_threshold` (default 3).
    pub fn is_near_duplicate(self, other: SimHash, threshold: u32) -> bool;
}
pub struct WarcWriter {
    /// Current open file handle (gzip writer).
    current: GzEncoder<BufWriter<File>>,
    /// Path to the current WARC file (e.g., data/warc/CC-MAIN-20240115-0001.warc.gz).
    current_path: PathBuf,
    /// Directory for WARC files.
    output_dir: PathBuf,
    /// Number of records written to the current file.
    record_count: u64,
    /// Bytes written to the current file (uncompressed).
    byte_count: u64,
    /// Rotation thresholds.
    config: WarcConfig,
}

#[derive(Debug, Clone)]
pub struct WarcConfig {
    pub max_records_per_file: u64,   // default: 100_000
    pub max_bytes_per_file: u64,     // default: 1_073_741_824 (1 GiB)
    pub compression_level: u32,      // 1–9, default: 6
}

/// A single WARC response record.
pub struct WarcRecord<'a> {
    pub target_uri: &'a Url,
    pub fetched_at: DateTime<Utc>,
    pub http_response: &'a [u8],  // Raw HTTP response including headers and body
    pub record_id: Uuid,
}

impl WarcWriter {
    pub fn write_record(&mut self, record: WarcRecord<'_>) -> Result<(), WarcError>;
    pub fn rotate(&mut self) -> Result<(), WarcError>;
    pub fn flush(&mut self) -> io::Result<()>;
}
pub struct Scheduler {
    /// Per-domain state, keyed by normalized host string.
    domains: DashMap<Arc<str>, DomainState>,
    /// Global config (default crawl delay, max concurrent per domain).
    config: SchedulerConfig,
}

#[derive(Debug)]
struct DomainState {
    bucket: TokenBucket,
    last_fetch: Instant,
    /// Crawl-delay from robots.txt (overrides config default if set).
    crawl_delay: Duration,
    /// Number of currently in-flight fetches for this domain.
    in_flight: u32,
}

/// Token bucket with nanosecond precision.
pub struct TokenBucket {
    capacity: u32,
    tokens: f64,
    refill_rate: f64,      // tokens per second
    last_refill: Instant,
}

impl TokenBucket {
    /// Returns true if a token was consumed.
    pub fn try_consume(&mut self) -> bool;
    /// Returns the duration until a token becomes available.
    pub fn time_until_available(&self) -> Duration;
}

#[derive(Debug, Clone)]
pub struct SchedulerConfig {
    pub default_crawl_delay: Duration,     // 1 second
    pub max_concurrent_per_domain: u32,   // 1
    pub politeness_multiplier: f64,        // 1.0 (can be raised for more politeness)
    pub robots_unavailable_backoff: Duration, // 60 seconds
}
/// Public interface for the URL frontier.
/// Both single-node (Phase 0) and distributed (Phase 1) implementations satisfy this trait.
#[async_trait]
pub trait Frontier: Send + Sync {
    /// Enqueue a URL for crawling. No-op if the URL is already in the frontier.
    async fn enqueue(&self, record: UrlRecord) -> Result<(), FrontierError>;

    /// Dequeue the highest-priority URL that is available to fetch now
    /// (i.e., its domain's crawl-delay has elapsed).
    async fn dequeue(&self) -> Result<Option<UrlRecord>, FrontierError>;

    /// Update the scheduling state after a fetch attempt.
    async fn update(&self, url: &Url, result: &FetchResult) -> Result<(), FrontierError>;

    /// Current number of URLs in the frontier (approximate for distributed).
    async fn len(&self) -> Result<u64, FrontierError>;

    /// Flush all pending writes to the durable store.
    async fn flush(&self) -> Result<(), FrontierError>;
}
#[derive(Debug, Deserialize)]
pub struct CrawlConfig {
    pub seed_urls: Vec<Url>,
    pub max_depth: u32,
    pub max_pages: Option<u64>,
    pub user_agent: String,
    pub domain_whitelist: Vec<String>,
    pub domain_blacklist: Vec<String>,
    pub fetch_timeout: Duration,
    pub max_redirects: u8,
    pub worker_count: usize,
    pub frontier: FrontierConfig,
    pub scheduler: SchedulerConfig,
    pub storage: StorageConfig,
    pub dedup: DedupConfig,
    pub metrics_port: u16,
}

#[derive(Debug, Deserialize)]
pub struct DedupConfig {
    pub expected_urls: u64,          // For Bloom filter sizing (default: 100_000_000)
    pub false_positive_rate: f64,    // (default: 0.01)
    pub bloom_path: PathBuf,         // On-disk Bloom filter file
    pub simhash_threshold: u32,      // Hamming distance for near-duplicate (default: 3)
}
