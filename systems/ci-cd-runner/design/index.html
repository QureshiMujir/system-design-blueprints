Core schema & types (§11)

struct PipelineSpec { api_version: String, kind: String, name: String,
  triggers: Vec<Trigger>, env: Map<String,String>, jobs: Map<JobId,JobSpec>,
  permissions: Permissions, concurrency: Option<ConcurrencyGroup>,
  defaults: Defaults, metadata: Metadata }
  // owned by dsl-parser; immutable post-parse; content-addressed by
  // BLAKE3(serialized); persisted keyed (tenant_id, pipeline_hash)
struct JobId([u8;16])                          // ULID; Copy, Eq, Hash
struct Permissions { contents: AccessLevel, packages: AccessLevel,
  id_token: AccessLevel, deployments: AccessLevel, pull_requests: AccessLevel }
enum AccessLevel { None, Read, Write }
struct JobSpec { id: JobId, name: String, needs: Vec<JobId>,
  runs_on: RunsOn, strategy: Strategy, steps: Vec<StepSpec>,
  env: Map<String,String>, if_: Option<CelExpr>, timeout_min: Option<u32>,
  continue_on_error: bool, environment: Option<EnvironmentRef>,
  container: Option<ContainerSpec>, services: Vec<ServiceSpec>,
  permissions: Option<Permissions>, outputs: Map<String,String>,
  concurrency: Option<ConcurrencyGroup> }
  // immutable after DAG build; needs[] are validated DAG edges (I8)
struct RunsOn { labels: Vec<String>, executor: Option<ExecutorPreference> }
enum ExecutorPreference { Firecracker, Runc, Docker, Kubernetes, HostShell }
struct Strategy { matrix: Option<MatrixSpec>, fail_fast: bool,
  max_parallel: Option<u32> }
struct StepSpec { id: Option<String>, name: String, kind: StepKind,
  if_: Option<CelExpr>, env: Map<String,String>, working_dir: Option<String>,
  timeout_min: Option<u32>, continue_on_error: bool,
  retry: Option<RetrySpec>, idempotent: bool }
enum StepKind {
  Run { shell: Shell, command: String },
  Uses { action: ActionRef, with: Map<String,String> },
  UploadArtifact { name: String, path: String, retention_days: u32 },
  DownloadArtifact { name: String, path: String },
  Cache { key: String, paths: Vec<String>, restore_keys: Vec<String> } }
struct ActionRef { owner: String, name: String, digest: String }
  // digest required, no floating tags accepted (ADR-06 threat mitigation)
struct WorkflowRun { id: RunId, tenant_id: TenantId, pipeline: PipelineHash,
  trigger: Trigger, status: RunStatus, queued_at: Timestamp,
  started_at: Option<Timestamp>, finished_at: Option<Timestamp>,
  jobs: Map<JobId,JobRun>, matrix_expansion: Map<JobId,Vec<MatrixCoord>>,
  event_log: Vec<RunEvent>, provenance: Option<ProvenanceRef> }
  // mutated only by appending to event_log (I3); reconstructed by replay
struct JobRun { id: JobRunId, job_id: JobId, matrix: Option<MatrixCoord>,
  status: JobStatus, runner_id: Option<RunnerId>,
  executor: Option<ExecutorBackendKind>, leased_at: Option<Timestamp>,
  started_at: Option<Timestamp>, finished_at: Option<Timestamp>,
  steps: Vec<StepRun>, outputs: Map<String,String>, exit_code: Option<i32> }
struct StepRun { index: u32, step: StepSpec, status: StepStatus,
  started_at: Option<Timestamp>, finished_at: Option<Timestamp>,
  exit_code: Option<i32>, log_offset: LogOffset, attempts: u32 }
enum RunStatus { Queued, Running, Approved, AwaitingApproval, Failed,
  Succeeded, Cancelled, Stale }
enum JobStatus { Pending, Scheduling, Leased, Running, Succeeded, Failed,
  Skipped, Cancelled }
enum StepStatus { Pending, Running, Succeeded, Failed, Skipped, TimedOut }
struct RunnerHandle { id: RunnerId, tenant_id: Option<TenantId>,
  labels: Vec<String>, pool: Option<PoolId>, version: String,
  status: RunnerStatus, registered_at: Timestamp,
  last_heartbeat: Timestamp, current_lease: Option<JobRunId>,
  capabilities: RunnerCapabilities }
struct RunnerCapabilities { executors: Vec<ExecutorBackendKind>, cpus: u32,
  mem_mb: u64, disk_mb: u64, kernel: String, arch: String,
  features: Vec<String> }
enum RunnerStatus { Registered, Idle, Leased, Running, Draining, Offline,
  Quarantined }                                // I11 terminal on byzantine output
trait ExecutorBackend {
  fn name(&self) -> &'static str
  fn prepare(&self, ctx: &JobContext) -> Result<PreparedJob, ExecutorError>
  fn run_step(&self, prep: &PreparedJob, step: &StepSpec)
    -> Result<StepOutcome, ExecutorError>
  fn collect_artifacts(&self, prep: &PreparedJob)
    -> Result<Vec<ArtifactRef>, ExecutorError>
  fn teardown(&self, prep: PreparedJob) -> Result<(), ExecutorError>
}
struct JobContext<'a> { workspace: &'a Path, env: &'a Map<String,String>,
  secrets: &'a SecretBundle, cache: &'a CacheClient,
  log_sink: &'a dyn LogSink, tenant: TenantId, run: RunId, job: JobRunId }
struct MatrixSpec { axes: Map<String,Vec<YamlValue>>,
  include: Vec<Map<String,YamlValue>>, exclude: Vec<Map<String,YamlValue>> }
struct MatrixCoord(Map<String,YamlValue>)       // Eq, Hash; ≤ 1,024 per I8
struct ArtifactRef { name: String, digest: Blake3Digest, size: u64,
  mime: Option<String>, tenant_id: TenantId, run_id: RunId,
  job_run_id: JobRunId, retention: Retention }
struct Blake3Digest([u8;32])
enum Retention { Days(u32), Until(Timestamp), Permanent }
struct CacheKey { tenant_id: TenantId, namespace: String, primary: String,
  restore_keys: Vec<String> }
struct CacheEntry { key: CacheKey, digest: Blake3Digest, size: u64,
  created_at: Timestamp, last_hit_at: Timestamp, hit_count: u64 }
struct SecretRef { name: String, scope: SecretScope }
enum SecretScope { Repository, Environment(EnvironmentRef), Organisation,
  Tenant }
struct ScopedToken { token_id: String, bearer: String, scope: Vec<String>,
  issued_at: Timestamp, expires_at: Timestamp, run_id: RunId,
  job_run_id: JobRunId, step_index: u32 }                       // I5
struct AuditEvent { id: AuditEventId, timestamp: Timestamp,
  actor: Principal, action: AuditAction, target: AuditTarget,
  outcome: AuditOutcome, trace_id: String,
  hash_chain_prev: Blake3Digest, hash_chain_self: Blake3Digest }
  // append-only; hash_chain_self = BLAKE3(hash_chain_prev || event bytes)
struct LogChunk { run_id: RunId, job_run_id: JobRunId, step_index: u32,
  stream: LogStream, offset: u64, bytes: Vec<u8>, timestamp: Timestamp,
  final_: bool }
enum LogStream { Stdout, Stderr, System }
struct LogOffset { start: u64, end: Option<u64> }
struct ApprovalRequest { id: ApprovalId, run_id: RunId,
  job_run_id: JobRunId, environment: EnvironmentRef,
  required: ApprovalPolicy, status: ApprovalStatus,
  requested_at: Timestamp, decided_at: Option<Timestamp>,
  decisions: Vec<ApprovalDecision> }
enum ApprovalPolicy { AnyOf(Vec<ApproverRef>), AllOf(Vec<ApproverRef>),
  Quorum { approvers: Vec<ApproverRef>, n: u32 } }
enum ApprovalStatus { Pending, Approved, Rejected, Expired }
enum Trigger { Push{branches:Vec<String>,paths:Vec<String>},
  PullRequest{base:Vec<String>,types:Vec<PrType>},
  Schedule{cron:String,tz:Option<String>}, Manual,
  WorkflowDispatch{inputs:Map<String,InputSpec>},
  Webhook{event_name:String} }
struct Webhook { id: WebhookId, source: WebhookSource,
  secret: Blake3Digest, event: String, headers: Map<String,String>,
  body_digest: Blake3Digest, received_at: Timestamp }
struct Token { kind: TokenKind, principal: Principal, scopes: Vec<String>,
  issued_at: Timestamp, expires_at: Timestamp, binding: TokenBinding }
enum TokenKind { Runner, ApiUser, Webhook, Service, IdToken }
enum Principal { Runner{id:RunnerId,tenant:Option<TenantId>},
  User{id:UserId}, Service{id:String}, Tenant{id:TenantId} }
struct ProvenanceRef { run_id: RunId, statement_digest: Blake3Digest,
  slsa_level: u8, signer: SigstoreCert, rekor_uuid: String }
  // signed by controller KMS key (P4, ADR-25); never runner-signed alone
