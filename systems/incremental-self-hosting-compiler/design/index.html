Core schema & types (§11)

struct FileId(u32)                 // dense id into the file table
struct Symbol(u32)                 // interned string id; thread-stable (INV-13)
struct ItemPath { segments: Vec<Symbol> }      // e.g. crate::parser::expr
struct RelSpan { anchor: AstId, lo: u32, hi: u32 }   // offsets relative to enclosing item (INV-6)
struct AbsSpan { file: FileId, lo: u32, hi: u32 }    // resolved on demand for diagnostics only
struct DefId { crate_: u32, path_hash: u64, disambig: u16 }   // stable identity across edits
struct Revision(u64)                       // monotonic (INV-10)
enum Durability { Volatile, Low, High }    // durability tier (ADR-14)
struct Fingerprint([u8; 32])               // BLAKE3 of a query value (ADR-4)
struct QueryKey { query: QueryId, key_hash: u64 }    // identifies a node
struct DepEdge { to: QueryKey }            // recorded input dependency (INV-5)
struct QueryNode {
  key: QueryKey, value: Option<SerializedValue>, fp: Fingerprint,
  changed_at: Revision, verified_at: Revision, deps: Vec<DepEdge>, durability: Durability,
}
trait Query { type Key; type Value: Fingerprintable; fn execute(db,&Key) -> Value; const ID: QueryId; }
fn maybe_changed_after(db, key: QueryKey, since: Revision) -> bool   // red-green verify primitive
fn fetch<Q: Query>(db, key: Q::Key) -> Q::Value                      // memoized demand entrypoint
fn set_input<I: InputQuery>(db, key, value, durability: Durability)  // bumps Revision
fn intern_fingerprint(value: &impl Fingerprintable) -> Fingerprint   // BLAKE3 of canonical bytes
trait Fingerprintable { fn fingerprint(&self, h: &mut Blake3Hasher); }   // canonical, order-stable
// AST (lyra-ast)
struct AstModule { items: Vec<AstId>, arena: AstArena, file: FileId }
// HIR (lyra-hir)
struct HirItem { def: DefId, kind: HirItemKind, generics: Vec<GenericParam> }
enum Ty {                                   // canonical, hash-consed
  Int(IntWidth, Signed), Float(FloatWidth), Bool, Char, Str, Unit,
  Ref(Box<Ty>, Mutbl), Array(Box<Ty>, ConstLen), Slice(Box<Ty>),
  Tuple(Vec<Ty>), Adt(DefId, Vec<Ty>), Param(u32), Fn(Vec<Ty>, Box<Ty>), Error,
}
// LIR (lyra-lir)
struct LirFn { def: DefId, mono_args: Vec<Ty>, blocks: Vec<Block>, regs: u32 }
struct Block { id: BlockId, insts: Vec<Inst>, term: Terminator }
enum Inst { Bin{op,dst,a,b}, Load{dst,addr}, Store{addr,val}, Call{dst,callee,args}, Phi{dst,incoming}, /* … */ }
struct Cgu { name: CguName, members: Vec<(DefId, Vec<Ty>)> }   // members sorted (determinism)
struct CguName([u8;32])                                        // content-address (§8.7)
struct Artifact { name: CguName, bytes: Vec<u8>, kind: Backend }   // bytecode|C-text
enum Backend { Bytecode, C99 }
struct Symbol Name(String)                                     // from lyra-mangle, deterministic
struct Diagnostic { code: DiagCode, severity: Severity, span: AbsSpan, msg: String, notes: Vec<Note> }
enum Severity { Error, Warning, Note }
struct DiagBuf { items: Vec<Diagnostic> }      // sorted by (file, lo, code) before emit (INV-11)
enum CompileError { Cycle(Vec<QueryKey>), CacheCorrupt(QueryKey), VersionSkew{got,want}, Io(IoError) }
struct Stage { name: &'static str, compiler: PathBuf, kind: StageKind }   // stage0/1/2
enum StageKind { HostRust, SelfBytecode, SelfNative }
struct FixpointResult { stage1_hash: [u8;32], stage2_hash: [u8;32], identical: bool, ddc_ok: bool }
struct DiffResult { artifact: CguName, clean_hash:[u8;32], warm_hash:[u8;32], equal: bool }
struct EditScript { ops: Vec<EditOp> }     // insert/delete/replace byte ranges, for replay
