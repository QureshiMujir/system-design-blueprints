Core schema & types (§11)

Name        // interned identifier; ownership: global intern table, immutable, Copy
Ix          // struct Ix(u32) — de Bruijn index; kernel+elaborator; Copy, no persistence
Lvl         // struct Lvl(u32) — de Bruijn level (semantic values); kernel; Copy
MetaVar     // struct MetaVar(u32) — elaborator-only; never crosses into dtc-kernel input

Term {      // core syntax; owned by whichever crate constructed it; immutable once built
  Var(Ix), Global(Name), Lam(Name, Box<Term>), App(Box<Term>, Box<Term>),
  Pi(Name, Box<Term>, Box<Term>), Sigma(Name, Box<Term>, Box<Term>),
  Pair(Box<Term>, Box<Term>), Proj1(Box<Term>), Proj2(Box<Term>),
  U(Lvl), Ind(Name), Ctor(Name, Vec<Term>),
  Elim(Name, Box<Term>, Motive, Vec<Method>, Box<Term>),
  Meta(MetaVar),                  // present only in dtc-elab-internal terms
  Let(Name, Box<Term>, Box<Term>, Box<Term>)
}
Ty = Term   // types are terms; no separate syntactic category (dependent-type hallmark)
Value {                           // NbE semantic domain; TRUSTED (dtc-nbe); not serialized
  VLam(Closure), VPi(Box<Value>, Closure), VSigma(Box<Value>, Closure),
  VPair(Box<Value>, Box<Value>), VU(Lvl), VNeutral(Neutral)
}
Closure { env: Env, body: Term }  // owned by the Value that holds it; not Copy
Env = Vec<Value>                  // evaluation environment; grows with binder depth
Spine = Vec<Value>                // pending eliminations applied to a Neutral head
Neutral {                          // stuck computation; TRUSTED
  NVar(Lvl), NApp(Box<Neutral>, Value), NProj1(Box<Neutral>), NProj2(Box<Neutral>),
  NElim(Name, Box<Neutral>, Motive, Vec<Method>)
}
Normal = Term                     // quote's output: a beta-normal, eta-long Term
Context { bindings: Vec<Binding> }         // typing context; kernel+elaborator, immutable per-scope
Binding { name: Name, ty: Term }            // one bound variable's declared type
Universe = Lvl                              // alias for clarity in universe-checker code
Decl { name: Name, ty: Term, body: Option<Term> }  // a checked top-level declaration
Signature { decls: Vec<Decl>, inductives: Vec<Inductive> }  // TRUSTED, append-only global env
Inductive { name: Name, params: Vec<Binding>, indices: Vec<Binding>, ctors: Vec<Constructor> }
Constructor { name: Name, args: Vec<Binding> }
Eliminator { for_: Name, motive: Term, methods: Vec<Method>, ty: Term }  // generated, TRUSTED
MetaContext { metas: Vec<MetaVariable> }     // elaborator-only, mutable during elaboration
Constraint {                                  // elaborator-only
  Equality(Term, Term), UniverseLe(Lvl, Lvl),
  ImplicitArg(MetaVar), PatternCoverage(CaseTreeId)
}
UnifyResult { Solved, Postponed, Failed(Reason) }
ConvResult { Equal, NotEqual(Term, Term) }    // TRUSTED, produced only by dtc-nbe/dtc-kernel
InferResult = Result<Term, TypeError>          // TRUSTED
CheckResult = Result<(), TypeError>            // TRUSTED
Kernel { signature: Signature }                // TRUSTED, the only type authorized to accept a Decl
Elaborator { meta_ctx: MetaContext, queue: Vec<Constraint> }  // UNTRUSTED
Positivity { Positive, Negative(Name, Term) }  // dtc-inductive, pre-kernel gate
Coverage { Covering, Missing(Vec<Name>) }      // dtc-pattern, pre-elaboration gate
TypeError {                                    // structured; rendered by dtc-diagnostics
  NotAFunctionType, UniverseInconsistency, NonStrictlyPositive,
  NonCovering, NonTerminating, UnificationFailure, ConversionFailure(Term, Term)
}
