Core schema & types (§11)

// pki-types — core certificate model
struct SerialNumber(BigUint)                    // owned; immutable; unique (I6); DER-serializable
struct DistinguishedName { rdn_sequence: Vec<RelativeDistinguishedName> }  // immutable; from untrusted CSR subject
struct TbsCertificate { version, serial: SerialNumber, issuer: DistinguishedName,
    validity: Validity, subject: DistinguishedName, spki: SubjectPublicKeyInfo,
    extensions: Vec<Extension> }                // immutable once signed; canonical-DER (P5)
struct Certificate { tbs: TbsCertificate, sig_alg: AlgorithmIdentifier, signature: Vec<u8> }
                                                 // immutable; persisted in pki-store; shared via Arc
struct Extension { oid: ObjectIdentifier, critical: bool, kind: ExtensionKind }
enum ExtensionKind { BasicConstraints(BasicConstraints), KeyUsage(KeyUsage),
    ExtendedKeyUsage(ExtendedKeyUsage), SubjectAltName(SubjectAltName),
    Aki(Vec<u8>), Ski(Vec<u8>), CrlDistributionPoints(Vec<Url>),
    AuthorityInfoAccess(Vec<AiaEntry>), CertificatePolicies(Vec<PolicyOid>) }
struct BasicConstraints { is_ca: bool, path_len: Option<u8> }   // enforced I11
struct KeyUsage(bitflags)                       // critical bit semantics per RFC 5280 section 4.2.1.3
struct ExtendedKeyUsage(Vec<ObjectIdentifier>)
struct SubjectAltName(Vec<GeneralName>)         // untrusted until RA/ACME validates (I14)
struct Validity { not_before: Timestamp, not_after: Timestamp }  // checked against I1
// pki-types — request/response and CSR
struct Csr { info: CertificationRequestInfo, sig_alg: AlgorithmIdentifier, signature: Vec<u8> }
                                                 // wholly untrusted input; parsed defensively (P6)
struct CertificateRequest { csr: Csr, requester_identity: Identity, profile_id: ProfileId,
    validation_evidence: ValidationEvidence }   // requester_identity/profile_id from RA; SAN untrusted until I14 clears
struct CaConfig { trust_anchor: TrustAnchor, key_handle: KeyHandle, max_path_len: u8 }
struct IssuingProfile { id: ProfileId, version: u32, allowed_san_patterns: Vec<Pattern>,
    allowed_eku: Vec<ObjectIdentifier>, max_ttl: Duration, quota: QuotaPolicy }
                                                 // versioned+hashed (ADR-16); referenced by PolicyDecision
struct PolicyRule { id: RuleId, predicate: PolicyPredicate, action: PolicyAction }
struct PolicyVerdict { decision_id: DecisionId, result: PolicyResult,
    evaluated_rules: Vec<RuleId>, denial_reasons: Vec<DenialReason>, profile_hash: Hash }
                                                 // immutable once produced; must precede signing (I3, P2)
// pki-hsm — key custody
struct KeyHandle { slot_id: SlotId, key_label: String, algorithm: SignatureAlgorithm }
                                                 // never dereferences to raw key bytes; zeroize-on-drop wrapper around session token only (P7)
struct SigningRequest { tbs_bytes: Vec<u8>, key_handle: KeyHandle, fencing_token: FenceToken }
                                                 // tbs_bytes canonical-DER (P5); consumed once (I12)
// pki-crl / pki-ocsp — revocation
struct Crl { tbs_cert_list: TbsCertList, sig_alg: AlgorithmIdentifier, signature: Vec<u8> }
struct RevokedEntry { serial: SerialNumber, revocation_date: Timestamp, reason: RevocationReason }
enum RevocationReason { KeyCompromise, AffiliationChanged, Superseded,
    CessationOfOperation, PrivilegeWithdrawn, CertificateHold }
struct OcspRequest { cert_id: CertId, nonce: Option<Nonce> }   // untrusted; nonce replay checked
struct OcspResponse { status: CertStatus, this_update: Timestamp, next_update: Timestamp,
    signature: Vec<u8> }                        // pre-signed (ADR-10); served read-only by responder
enum CertStatus { Good, Revoked(RevocationReason), Unknown }
// pki-acme — ACME state
struct AcmeAccount { id: AccountId, jwk: PublicKey, contacts: Vec<Uri> }
struct AcmeOrder { id: OrderId, identifiers: Vec<Identifier>, status: OrderStatus,
    authorizations: Vec<AuthorizationId> }
struct AcmeAuthorization { id: AuthorizationId, identifier: Identifier,
    challenges: Vec<AcmeChallenge>, status: AuthzStatus }
struct AcmeChallenge { kind: ChallengeKind, token: String, status: ChallengeStatus }
enum ChallengeKind { Http01, Dns01 }             // tls-alpn-01 rejected, ADR-13
// pki-ct / pki-path / pki-audit — CT, path validation, audit
struct Sct { log_id: LogId, timestamp: Timestamp, signature: Vec<u8> }
struct CtLog { id: LogId, public_key: PublicKey, submission_url: Url }
struct PathBuilder { leaf: Certificate, candidates: Vec<Certificate>, anchors: Vec<TrustAnchor> }
struct TrustAnchor { certificate: Certificate, name_constraints: Option<NameConstraints> }
struct ValidationResult { accepted: bool, chain: Vec<Certificate>,
    policy_set: Vec<PolicyOid>, failure_reason: Option<PathFailure> }
struct AuditEntry { sequence_id: u64, event_type: AuditEventKind, actor: ActorId,
    object_ref: ObjectRef, payload_hash: Hash, previous_hash: Hash, timestamp: Timestamp }
                                                 // sequence_id strictly monotonic (I8); previous_hash chains (P8)
enum PkiError { ValidationError(String), SerializationError(String), HsmUnavailable,
    PolicyDenied(Vec<DenialReason>), QueueBackpressure, PersistenceCorruption,
    RevocationConflict, ProtocolViolation(String) }
