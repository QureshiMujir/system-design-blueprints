Core schema & types (§11)

pub enum Phase { ReqHeaders=1, ReqBody=2, RespHeaders=3, RespBody=4, Logging=5 }

pub enum WafError { Truncated, TooLarge{limit:usize}, BadFraming, BudgetExceeded(Budget),
                    BadEncoding, Internal }                       // → fail policy, never panic (I2)
pub enum Budget { RegexSteps, EvalTime, ParamCount, BodySize, HeaderCount }

pub struct HttpRequest<'a> { pub method:&'a str, pub uri:Uri<'a>, pub version:HttpVersion,
                             pub headers:Headers<'a>, pub body:BodyWindow<'a> }
pub struct HttpResponse<'a>{ pub status:u16, pub headers:Headers<'a>, pub body:BodyWindow<'a> }

pub struct NormalizedFields {                                     // idempotent, fixed-order (I10)
    pub path_canon:String, pub args:Vec<(String,String)>, pub cookies:Vec<(String,String)>,
    pub json:Vec<(String /*path*/, String /*val*/)>, pub multipart:Vec<Part>,
    pub headers_norm:Vec<(String,String)>, pub body_decoded:Vec<u8> }

pub struct Transaction<'a> {                                      // per-request working set
    pub req:HttpRequest<'a>, pub resp:Option<HttpResponse<'a>>,
    pub norm:NormalizedFields, pub vars:VarStore,
    pub matches:Vec<RuleMatch>, pub score_in:i32, pub score_out:i32,
    pub phase:Phase, pub verdict:Verdict, pub client_ip:IpAddr }

pub enum Verdict { Pass, Block{status:u16, rules:Vec<RuleId>}, DetectOnly{rules:Vec<RuleId>} }  // I6
pub struct RuleMatch { pub id:RuleId, pub var:&'static str, pub value:String,
                       pub score:i32, pub confidence:u8, pub msg:String }
pub enum Variable { RequestUri, RequestMethod, RequestHeaders(Option<String>), Args(Option<String>),
                    ArgsGet, ArgsPost, RequestBody, Cookies(Option<String>), RequestFilename,
                    ResponseStatus, ResponseHeaders, ResponseBody, Tx(String), Ip(String), MatchedVars }
pub enum Transformation { None, Lowercase, UrlDecodeUni, HtmlEntityDecode, RemoveNulls,
                          CompressWhitespace, NormalizePath, Base64Decode, JsDecode }
pub enum Operator { Rx(CompiledRegex), Pm(PhraseSet), DetectSqli, DetectXss, Eq(i64), Ge(i64),
                    Gt(i64), Within(Vec<String>), IpMatch(Vec<IpNet>), Streq(String) }
pub enum Action { Block, Deny{status:u16}, Pass, Drop, Redirect(String),
                  Score(i32), SetVar(String,String), Ctl(Ctl), Chain, Skip(u32),
                  Capture, Log, NoLog, Msg(String), Tag(String), Severity(Sev), Confidence(u8) }

pub struct Rule { pub id:RuleId, pub phase:Phase, pub vars:Vec<Variable>,
                  pub tfns:Vec<Transformation>, pub op:Operator, pub actions:Vec<Action>,
                  pub chain:Option<Box<Rule>> }
pub struct RuleSet { pub rules:Vec<Rule>, pub grammar_version:Version }  // compiled, Arc-shared (I12)

pub trait Operator_ { fn eval(&self, input:&str, tx:&mut Transaction) -> Result<bool, WafError>; }  // I4 step-bounded

pub struct Engine { /* Arc<RuleSet> + scoring config + budgets */ }       // Send + Sync, pure per-tx
impl Engine {
    pub fn evaluate(&self, tx:&mut Transaction) -> Verdict;               // runs phases; deterministic (I3)
    pub fn phase(&self, tx:&mut Transaction, p:Phase) -> Result<(), WafError>;
}
pub struct ScoringConfig { pub inbound_threshold:i32, pub outbound_threshold:i32,
                           pub paranoia:u8, pub mode:Mode }                // I7
pub enum Mode { DetectionOnly, Blocking }                                 // principle 5
pub enum FailPolicy { Open, Closed }                                      // principle 8 (ADR-029)

pub trait Normalizer { fn normalize(&self, req:&HttpRequest) -> Result<NormalizedFields, WafError>; }  // I1/I10
pub struct RateLimiter { /* reuse #39: token/sliding-window + reputation */ }  // I11
