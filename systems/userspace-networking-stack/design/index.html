Core schema & types (§11)

// uns-types — shared IDs + flags + stats + errors
pub struct PortId(pub u16);                              // port (device) handle
pub struct QueueId(pub u16);                             // RX/TX queue within a port
pub struct LcoreId(pub u16);                             // logical core (pinned thread)
pub struct NumaNode(pub u16);                            // NUMA node id
bitflags! { pub struct OffloadFlags: u64 {               // mbuf ol_flags (RX + TX)
  const RX_IP_CKSUM_GOOD; const RX_L4_CKSUM_GOOD; const RX_VLAN_STRIPPED;
  const TX_IP_CKSUM;      const TX_TCP_CKSUM;    const TX_UDP_CKSUM; const TX_VLAN_INSERT; } }
pub struct Ptype(pub u32);                               // packed L2/L3/L4 packet-type (parse output)
pub struct PortStats { rx_pkts:u64, tx_pkts:u64, rx_bytes:u64, tx_bytes:u64, rx_drops:u64, tx_drops:u64 }
pub struct XStats { /* extended per-queue/per-core counters, name→u64 */ }

// uns-mbuf — owned until freed to pool OR handed to TX (I5/I6); refcount governs clones
pub struct Mbuf {
  buf_addr: NonNull<u8>,    // start of data buffer (hugepage-backed)
  data_off: u16,            // headroom offset (default 128)
  data_len: u16,            // bytes in THIS segment
  pkt_len:  u32,            // total bytes across chain
  nb_segs:  u16,            // chain length
  next:     Option<NonNull<Mbuf>>,  // next segment (scatter/jumbo)
  refcnt:   u16,            // clone refcount (I5)
  ol_flags: OffloadFlags,   // RX/TX offload + ptype hints
  rss_hash: u32,            // filled by RSS (I10)
  port:     PortId,
  pool:     NonNull<Mempool>,
}
impl Mbuf {                                              // bounds-checked header ops (I8)
  pub fn data(&self) -> &[u8];                pub fn data_mut(&mut self) -> &mut [u8];
  pub fn prepend(&mut self, len: u16) -> Result<&mut [u8]>;  // push header (shrink data_off)
  pub fn adj(&mut self, len: u16) -> Result<()>;             // strip header (grow data_off)
  pub fn append(&mut self, len: u16) -> Result<&mut [u8]>;   // grow into tailroom
  pub fn trim(&mut self, len: u16) -> Result<()>;
  pub fn clone_ref(&mut self) -> *mut Mbuf;                  // refcnt++ (shared)
}
// uns-mempool — lockless backing + per-core magazine; zero heap on data path (I4)
pub struct Mempool { /* backing free-stack + PerCore<Cache>, NUMA-local (I11) */ }
pub struct Cache { local: [*mut Mbuf; CACHE_SIZE], len: usize }  // per-core, no atomics
impl Mempool {
  pub fn create(n_elts: usize, data_room: u16, socket: NumaNode) -> Result<Self>;
  pub fn get(&self) -> Option<*mut Mbuf>;                   // hot: pop local cache
  pub fn get_bulk(&self, out: &mut [*mut Mbuf]) -> usize;   // burst alloc
  pub fn put(&self, m: *mut Mbuf);                          // hot: push local cache
  pub fn put_bulk(&self, ms: &[*mut Mbuf]);
}

// uns-ring — SPSC + MPMC lockless bounded; correct acquire/release (I1); bounded (I2); pow2 (I12)
pub struct Ring<T> { prod_head: AtomicU32, prod_tail: AtomicU32,
                     cons_head: AtomicU32, cons_tail: AtomicU32,
                     mask: u32, slots: Box<[UnsafeCell<MaybeUninit<T>>]> }
unsafe impl<T: Send> Send for Ring<T> {} unsafe impl<T: Send> Sync for Ring<T> {}
impl<T> Ring<T> {
  pub fn new(capacity: usize /* pow2 */) -> Self;
  pub fn enqueue_burst(&self, items: &[T]) -> usize;        // returns #enqueued (≤ items.len)
  pub fn dequeue_burst(&self, out: &mut [T]) -> usize;      // returns #dequeued
}

// uns-pmd — the only I/O entry points; TX returns unsent count (I6/I10)
pub trait PmdOps: Send {
  fn configure(&mut self, n_rx: u16, n_tx: u16, cfg: &PortConfig) -> Result<()>;
  fn start(&mut self) -> Result<()>;
  fn stop(&mut self) -> Result<()>;
  fn rx_burst(&self, q: QueueId, out: &mut [*mut Mbuf]) -> usize;  // ≤ out.len, no interrupt
  fn tx_burst(&self, q: QueueId, pkts: &[*mut Mbuf]) -> usize;     // #sent; [k..n) still owned
  fn stats(&self) -> PortStats;
}
pub trait EthDev: PmdOps { fn port_id(&self) -> PortId; fn numa_node(&self) -> NumaNode; }
// uns-rss — deterministic hash (I10)
pub struct RssKey([u8; 40]);
pub fn toeplitz(key: &RssKey, tuple: &[u8]) -> u32;          // pure
pub struct Indirection { table: Vec<QueueId> /* pow2 */ }
pub struct FlowKey { src_ip:IpAddr, dst_ip:IpAddr, src_port:u16, dst_port:u16, proto:u8 }  // 5-tuple

// uns-classify — lookup-safe updates (I7)
pub struct Lpm4 { /* DIR-24-8: tbl24[2^24], tbl8 pool; RCU-swapped */ }
impl Lpm4 { pub fn lookup(&self, ip: u32) -> Option<NextHopId>;  // 1–2 memory accesses
            pub fn add(&self, prefix: u32, len: u8, nh: NextHopId) -> Result<()>; }  // control-plane, RCU
pub struct CuckooTable<K, V> { /* two tables, bounded probe; RCU/seqlock */ }
pub struct AclTable { /* tuple-space / bit-vector rules */ }
pub struct NextHopId(pub u32);

// uns-flow — conntrack (§P10); persistence: none (in-memory, control-loaded)
pub struct FlowEntry { key: FlowKey, state: FlowState, last_seen: u64, nat: Option<NatMap>, timer: TimerHandle }
pub enum FlowState { New, SynSent, Established, FinWait, Closed, Assured }
pub struct FlowTable { /* cuckoo-keyed PerCore shards + timer wheel */ }

// uns-eal — init + launch
pub struct Eal { /* hugepages, lcores, NUMA topology */ }
impl Eal {
  pub fn init(args: &EalArgs) -> Result<Self>;                          // map hugepages, detect NUMA
  pub fn launch<F: Fn(LcoreId) + Send + Clone>(&self, f: F) -> Result<()>;  // run f on each lcore (pinned)
  pub fn wait_lcores(&self) -> Result<()>;
}
pub struct PerCore<T> { /* one cache-line-padded T per lcore (I14); !Sync access via lcore id */ }

// errors (thiserror; no panic on data path — ADR-032)
pub enum UnsError { Eal(EalError), Pmd(PmdError), Parse(ParseError), Ring(RingError),
                    Mempool(MempoolError), Classify(ClassifyError), Qos(QosError), Flow(FlowError) }
pub enum ParseError { Truncated, BadHeaderLen, BadChecksum, UnknownEtherType, BadVlan }
pub enum RingError { Full, Empty }                          // Full → counted drop (I2)
