Core schema & types (§11)

/// 64-char Base64 alphabet, index = digit value: A-Z(0-25) a-z(26-51) 0-9(52-61) '+'(62) '/'(63).
pub const B64: [u8; 64];
/// Reverse lookup (char byte -> 0..63, 0xFF = invalid). 256-entry table for branch-free decode.
pub(crate) const B64_REV: [u8; 256];

/// Encode one integer as Base64 VLQ, appending chars to `out`.
/// Sign-bit-and-magnitude (NOT two's complement): LSB of the pre-grouped value = sign
/// (1 = negative); remaining bits split into 5-bit groups low->high; bit 5 (value 32) = continuation.
pub fn encode_vlq(value: i64, out: &mut String);

/// Decode one VLQ integer starting at `*pos`; advances `*pos` past consumed chars.
/// Stops at the first group whose continuation bit is clear; reconstructs sign from bit 0.
pub fn decode_vlq(input: &[u8], pos: &mut usize) -> Result<i64, VlqError>;

pub enum VlqError {
    UnexpectedEof,            // continuation bit set but input ended
    InvalidBase64Char(u8),   // byte not in the alphabet
    Overflow,                // accumulated magnitude exceeds i64 / shift > 63
}
/// A point in a text buffer: 0-based line + column (UTF-16 code units per v3 column semantics).
#[derive(Copy, Clone, PartialEq, Eq, Hash)]
pub struct Position { pub line: u32, pub column: u32 }

/// One mapping entry on a generated line. ABSOLUTE values in memory (deltas live only in the
/// encoded "mappings" string). A 1-field segment has source=None, name=None (generated-only).
#[derive(Copy, Clone, PartialEq, Eq, Hash)]
pub struct Segment {
    pub gen_column: u32,        // generated column on this line
    pub source: Option<u32>,    // index into SourceMap.sources; None => generated-only segment
    pub orig_line: u32,         // original line (0-based)   — meaningful iff source.is_some()
    pub orig_column: u32,       // original column (0-based) — meaningful iff source.is_some()
    pub name: Option<u32>,      // index into SourceMap.names; requires source.is_some()
}

/// Decoded mappings: outer Vec index = generated line (0-based); inner = segments on that line,
/// sorted ascending by gen_column. Empty inner Vec => a generated line with no mappings (";;").
#[derive(Clone, PartialEq)]
pub struct DecodedMap(pub Vec<Vec<Segment>>);

/// The v3 JSON shape (envelope). `mappings` holds the encoded VLQ string (decode via smap-codec).
/// `sources`/`sources_content` are positionally aligned (same length; index i => same source).
#[derive(Clone, PartialEq)]
pub struct SourceMap {
    pub version: u8,                          // always 3
    pub file: Option<String>,                 // generated artifact name
    pub source_root: Option<String>,          // prefix prepended to each `sources` entry
    pub sources: Vec<Option<String>>,         // source paths/URLs (None = anonymous source)
    pub sources_content: Vec<Option<String>>, // inline original contents, aligned to `sources`
    pub names: Vec<String>,                   // symbol names referenced by segment.name
    pub mappings: String,                     // encoded VLQ "mappings" string
    pub ignore_list: Vec<u32>,                // x_google_ignoreList: indices into `sources` to skip
}

/// Result of a consumer original lookup: resolved absolute original location (+ optional name).
#[derive(Copy, Clone, PartialEq, Eq, Hash)]
pub struct OriginalLocation { pub source: u32, pub line: u32, pub column: u32, pub name: Option<u32> }
/// Encode a DecodedMap into the "mappings" string. Emits ';' between generated lines, ',' between
/// segments; each field as a relative VLQ delta per the reset rules (gen_column resets each ';').
pub fn encode_mappings(decoded: &DecodedMap) -> String;

/// Decode a "mappings" string into a DecodedMap (tolerant of real-world maps). Validates each
/// segment has 1, 4, or 5 fields; rejects negative absolute columns / out-of-range source indices.
pub fn decode_mappings(mappings: &str) -> Result<DecodedMap, CodecError>;

pub enum CodecError {
    BadFieldCount(usize),  // segment had != {1,4,5} fields
    Vlq(VlqError),         // underlying VLQ decode failed
    NegativeColumn,        // a delta drove an absolute line/column below 0
    SourceIndexOob,        // source/name index out of range vs the map's tables
}

/// Running-delta state. prev_gen_col RESETS to 0 at every ';' (new generated line);
/// the other four CARRY ACROSS lines. Mirrored shape for encode and decode.
pub(crate) struct EncoderState {
    prev_gen_col: u32,                                          // reset per line
    prev_source: u32, prev_orig_line: u32, prev_orig_column: u32, prev_name: u32, // carried across ';'
}
pub(crate) struct DecoderState { /* identical fields + reset rules */ }
/// Serialize a SourceMap to the canonical v3 JSON envelope (version:3, stable key order).
pub fn to_json(map: &SourceMap) -> String;

/// Parse v3 JSON into a SourceMap. Strips the XSSI ")]}'" prefix if present; requires version==3.
pub fn from_json(text: &str) -> Result<SourceMap, JsonError>;

/// Find the last "//# sourceMappingURL=" (or legacy "//@") in generated code and resolve it:
/// external URL, or an inline "data:application/json;base64,..." map decoded in place.
pub fn extract_source_mapping_url(code: &str) -> Option<SourceMapRef>;

pub enum SourceMapRef { Url(String), Inline(SourceMap) }  // data: URL decoded to a SourceMap

pub enum JsonError {
    Parse(String),                 // malformed JSON / bad data: URL / bad base64
    BadVersion(u8),                // version field present but != 3
    MissingField(&'static str),    // required field absent (e.g. "version", "mappings")
}
/// Mutable accumulator implementing the SourceMapGenerator contract. Interns sources + names,
/// buffers segments per generated line, then sorts + dedups + encodes on finalize.
pub struct SourceMapBuilder { /* sources interner, names interner, Vec<Vec<Segment>>, content map */ }

impl SourceMapBuilder {
    pub fn add_source(&mut self, src: &str) -> u32;        // returns stable source index (dedups)
    pub fn add_name(&mut self, name: &str) -> u32;         // returns stable name index (dedups)
    pub fn set_source_content(&mut self, source: u32, content: &str);  // attach inline content
    /// Record one mapping. `orig = Some((source_idx, original_position))` for a mapped segment,
    /// `None` for a generated-only (1-field) segment. `name` requires `orig.is_some()`.
    pub fn add_mapping(&mut self, gen: Position, orig: Option<(u32, Position)>, name: Option<u32>);
    pub fn into_sourcemap(self) -> SourceMap;  // sort by (gen_line, gen_column), dedup, encode_mappings
}
/// Tie-break direction for position lookups when no exact match exists.
pub enum Bias {
    GreatestLowerBound,   // default: greatest mapped position <= query
    LeastUpperBound,      // least mapped position >= query
}

/// Read-only queryable view: holds the DecodedMap plus a by-generated index (sorted (line,col))
/// and a by-original index (sorted (source,line,col)) for reverse lookup.
pub struct SourceMapConsumer { /* decoded, by_gen index, by_orig index, sources/names/content */ }

impl SourceMapConsumer {
    pub fn new(map: SourceMap) -> Result<Self, CodecError>;   // decodes mappings + builds indexes
    /// generated -> original. Binary-searches the by-generated index with `bias`.
    pub fn original_position_for(&self, gen: Position, bias: Bias) -> Option<OriginalLocation>;
    /// original -> generated. Binary-searches the by-original index with `bias`.
    pub fn generated_position_for(&self, source: u32, orig: Position, bias: Bias) -> Option<Position>;
    pub fn source_content_for(&self, source: u32) -> Option<&str>;
    /// All generated positions mapping to a given original position.
    pub fn all_generated_positions_for(&self, source: u32, orig: Position) -> Vec<Position>;
}
/// Compose two stages: orig->mid (as a consumer) and mid->gen (as a DecodedMap + its names).
/// For each gen segment, look up its mid position in `orig_to_mid`; if mapped, emit gen->original
/// (source/line/col/name); if unmapped, emit a 1-field generated-only segment. Names follow the
/// deepest available mapping.
pub fn compose(orig_to_mid: &SourceMapConsumer, mid_to_gen: &DecodedMap, mid_names: &[String]) -> SourceMap;

/// Fold `compose` over an ordered chain orig -> .. -> gen. Associative; collapses to one flat map.
pub fn remap(chain: &[SourceMap]) -> Result<SourceMap, CodecError>;
/// One section of an index map: its starting offset in the generated output + its own SourceMap.
#[derive(Clone, PartialEq)]
pub struct Section { pub offset: Position, pub map: SourceMap }

/// A sectioned ("index") map. Sections MUST be sorted by `offset` and non-overlapping.
#[derive(Clone, PartialEq)]
pub struct IndexedSourceMap {
    pub version: u8,                  // always 3
    pub file: Option<String>,
    pub sections: Vec<Section>,
}

impl IndexedSourceMap {
    /// Merge into a single flat SourceMap: shift each section's generated positions by its offset
    /// (offset.line added to every gen line; offset.column added only to the section's first line).
    pub fn flatten(&self) -> Result<SourceMap, CodecError>;
}
/// Options controlling map generation: filenames, content embedding, mapping resolution.
pub struct MapOptions {
    pub file: Option<String>,    // generated file name
    pub source: Option<String>,  // original source name (single-input editor)
    pub include_content: bool,   // embed original into sources_content
    pub hires: bool,             // per-character vs per-edit-boundary mappings
}

/// Mutable editor over an original buffer. Records ordered edits (insert/remove/overwrite/move)
/// against original byte offsets; emits transformed text + a correct map WITHOUT re-parsing,
/// via interval bookkeeping + column tracking. Builder-style: each method returns &mut Self.
pub struct MagicString { /* original: String, ordered edits, intro/outro buffers */ }

impl MagicString {
    pub fn overwrite(&mut self, start: usize, end: usize, content: &str) -> &mut Self; // replace [start,end)
    pub fn append(&mut self, s: &str) -> &mut Self;   // append after all content (outro)
    pub fn prepend(&mut self, s: &str) -> &mut Self;  // prepend before all content (intro)
    pub fn remove(&mut self, start: usize, end: usize) -> &mut Self;  // delete [start,end)
    pub fn to_string(&self) -> String;                // materialize transformed output
    pub fn generate_map(&self, opts: &MapOptions) -> SourceMap;  // builds via SourceMapBuilder
}
