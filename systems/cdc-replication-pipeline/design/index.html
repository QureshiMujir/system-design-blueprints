Core schema & types (§11)

// Identifiers — Copy, Ord; persistence: fixed-width ints; serialization: fixed-width wire
// encoding; compatible across versions (append-only IDs, never repurposed, P14).
type SourceId = u32;
type Lsn = u64;                 // monotonic per source; the ordering key (P1, I1)
type Offset = Lsn;               // alias: the offset IS the source LSN (ADR-05)
type Txid = u64;
type SchemaId = u32;
type SchemaVersion = u32;        // monotonic per SchemaId

// ChangeEvent — owned by decode-engine at creation, immutable after; Send+Sync;
// persistence: transient (not itself persisted; fields feed Checkpoint/DLQ);
// wire: Protobuf (ADR-12).
struct ChangeEvent {
    source_id: SourceId,
    lsn: Lsn,
    txid: Txid,
    event_seq: u32,
    op: Op,
    table: TableId,
    schema_version: SchemaVersion,
    before: Option<RowImage>,
    after: Option<RowImage>,
    meta: SourceMeta,
    idempotency_key: IdempotencyKey,
    checksum: u64,
}

// Op — Copy, exhaustively matched everywhere; compatibility: a new variant requires a
// major version bump (ADR-24 / §27).
enum Op { Insert, Update, Delete, Truncate, Ddl }

// RowImage / BeforeAfter — owned by ChangeEvent; immutable; serialization: column-name-
// keyed map, forward-compatible (unknown columns preserved, P14).
struct RowImage { columns: Map<ColumnId, Value> }
struct BeforeAfter { before: Option<RowImage>, after: Option<RowImage> }

// SourceMeta — small, Copy-ish; commit_ts is informational only, never used for
// ordering (ADR-24).
struct SourceMeta { source_id: SourceId, table: TableId, commit_ts: Timestamp }
// TableSchema / ColumnDef — owned by schema-registry; mutation: append-only new
// SchemaVersion per change, never edited in place; persistence: durable table in the
// registry's backing store; compatibility: BACKWARD default (ADR-15).
struct ColumnDef { name: String, ty: ColumnType, nullable: bool, default: Option<Value> }
struct TableSchema { table: TableId, version: SchemaVersion, columns: Vec<ColumnDef> }

// SchemaRegistry — shared, internally Arc<RwLock<..>>; concurrency: readers never block
// on a lookup longer than a version swap; persistence: durable, fleet-replicated (ADR-19).
trait SchemaRegistry {
    fn resolve(&self, table: TableId, version: SchemaVersion) -> Option<TableSchema>;
    fn propose(&self, diff: SchemaDiff) -> Result<SchemaVersion, SchemaViolation>;
}

// SnapshotChunk / Watermark / HandoffState — owned by handoff-coordinator;
// SnapshotChunk immutable once selected; Watermark is a point-in-time LSN marker
// injected into the stream (§8); HandoffState mutation is single-writer per table.
struct SnapshotChunk { table: TableId, key_range: (Value, Value), rows_read: u64 }
struct Watermark { table: TableId, lsn: Lsn, kind: WatermarkKind /* Low | High */ }
enum HandoffState { Snapshotting, WindowReconciling, Streaming }

// DecoderStream / ReplicationSlot — owned by capture-postgres/capture-mysql; not Send
// across tasks (single reader task per slot, P4); persistence: the slot itself lives in
// the source DB, not this process.
trait DecoderStream { fn next_frame(&mut self) -> Result<RawFrame, CaptureError>; }
struct ReplicationSlot { name: String, source_id: SourceId, retained_wal_bytes: u64 }
// Sink trait — implemented by sink-db/sink-search/sink-warehouse/sink-log; object-safe;
// concurrency: one apply-batch in flight per (sink, partition), enforced by
// sink-runtime, not the trait itself.
trait Sink {
    fn apply(&self, batch: &[ChangeEvent]) -> ApplyResult;
    fn idempotency_check(&self, key: &IdempotencyKey) -> bool;
}

// ApplyResult — returned per batch; mutation: none (value type); serialization:
// internal only, never on the wire.
enum ApplyResult { Applied(u32), PartialFailure(Vec<(ChangeEvent, ApplyError)>), Rejected(ApplyError) }

// IdempotencyKey — Copy, Hash, Eq; derived deterministically from (table, primary_key,
// lsn), never randomly generated (I2).
struct IdempotencyKey { table: TableId, pk_hash: u64, lsn: Lsn }

// OffsetStore / Checkpoint — offset-store crate; Checkpoint persistence: durable,
// CAS-write only (I8, monotonic); concurrency: one writer per (sink_id, partition) at a
// time (P4).
struct Checkpoint { sink_id: SinkId, source_id: SourceId, committed_lsn: Lsn, updated_at: Timestamp }
trait OffsetStore {
    fn commit(&self, cp: Checkpoint) -> Result<(), OffsetRegression>;
    fn read(&self, sink_id: SinkId) -> Option<Checkpoint>;
}

// Router / Transform — router owned by the pipeline-core runtime; stateless per event
// (bounded per-key dedup state only, §1 non-goals); Transform is a pure fn-object, no
// shared mutable state across calls.
trait Router { fn route(&self, e: &ChangeEvent) -> Vec<SinkId>; }
trait Transform { fn apply(&self, e: ChangeEvent) -> Result<ChangeEvent, TransformError>; }
// DeadLetter — owned by sink-runtime on apply failure; persistence: durable in the
// sink-log-backed DLQ (ADR-14); retained 7 days (§4).
struct DeadLetter { event: ChangeEvent, error: ApplyError, first_seen: Timestamp, retry_count: u32 }

// Backpressure — backpressure crate; concurrency: the credit counter is atomic, shared
// across producer/consumer tasks (ADR-16).
struct Backpressure { credits: AtomicI64, threshold: u64 }

// LagMetric — lag-monitor crate; recomputed on a fixed interval (<=1s); a read-only
// snapshot type, no shared mutable state exposed externally.
struct LagMetric { source_id: SourceId, sink_id: SinkId, lag_events: u64, lag_ms: u64 }

// PipelineState — pipeline-core; mutation: single-writer state machine (§8),
// transitions validated (I12); persistence: durable, stored alongside Checkpoint metadata.
enum PipelineState { Snapshotting, Handoff, Streaming, Paused, Resuming, Failed }

// Error taxonomy — pipeline-core; exhaustive; compatibility via wrapping (ADR-24 / §27).
enum PipelineError {
    CaptureError(CaptureError), SchemaViolation, ReplayMismatch, OffsetRegression,
    SinkBackpressure, SerializationFailure, HandoffReconcileMismatch, InvalidStateTransition,
}
