Core schema & types (§11)

pub struct RsaPublicKey {
    pub n: BigUint,   // modulus; Clone, Eq; not secret; safe to log/serialize
    pub e: BigUint,   // public exponent, normally 65537 (ADR-08); not secret
}
// Copy: no (BigUint owns heap limbs). Send + Sync: yes (immutable after construction).
// Serialization: via rsa-keys (PKCS#1/PKCS#8/JWK/PEM), lossless round-trip (§20.5).

pub struct RsaPrivateKey {
    pub n: BigUint,
    pub e: BigUint,
    pub d: BigUint,       // secret; zeroized on drop (I8)
    pub p: BigUint,       // secret prime factor; zeroized on drop
    pub q: BigUint,       // secret prime factor; zeroized on drop
    pub crt: CrtParams,   // secret; zeroized on drop
}
// Clone: yes (explicit, deliberate). Copy: no (forces ownership reasoning at call sites).
// Debug/Display: NOT implemented (I8, P11, ADR-22). Drop: ZeroizeOnDrop (I8).
// Send + Sync: yes; concurrent read-only use across threads is safe (fields are immutable
// after construction; every operation takes &self).

pub struct CrtParams {
    pub dp: BigUint,      // d mod (p-1)
    pub dq: BigUint,      // d mod (q-1)
    pub q_inv: BigUint,   // q^-1 mod p
}
// Derived once at keygen (§15.2) or key import (§20); never mutated after construction.
pub struct BigUint {
    limbs: Vec<u64>,       // little-endian limb order; private field
}
// fn from_be_bytes(bytes: &[u8]) -> Self;
// fn to_be_bytes(&self, len: usize) -> Vec<u8>;
// fn bit_len(&self) -> usize;
// fn is_zero(&self) -> bool;
// fn is_even(&self) -> bool;
// Clone: deep-copies limbs. Zeroize: implements ZeroizeOnDrop when constructed via the
// `Secret<BigUint>` wrapper used inside RsaPrivateKey/CrtParams; public-key BigUints do not
// pay the zeroizing-drop cost (not secret, ADR-22 scope is private-key fields only).

pub struct SecretExponent(BigUint);   // newtype; only mod_exp_ct accepts this type (P6)
pub struct PublicExponent(BigUint);   // newtype; only mod_exp_pub accepts this type (P6)

pub struct MontgomeryParams {
    pub n: BigUint,
    pub n_inv_neg_mod_64: u64,   // (-n^-1) mod 2^64
    pub r2_mod_n: BigUint,        // R^2 mod n, for entering Montgomery form
}
// fn for_modulus(n: &BigUint) -> Self;   // computed once per key, cached (§4 startup cost)

// fn mod_exp_ct(base: &BigUint, exp: &SecretExponent, modulus: &MontgomeryParams) -> BigUint;
// fn mod_exp_pub(base: &BigUint, exp: &PublicExponent, modulus: &MontgomeryParams) -> BigUint;
// fn mont_mul(a: &BigUint, b: &BigUint, params: &MontgomeryParams) -> BigUint;
// fn mod_inverse(a: &BigUint, modulus: &BigUint) -> Result<BigUint, RsaError>;
pub enum RsaError {
    MessageTooLarge,
    CiphertextTooLarge,
    InvalidSignature,
    FaultInjection,          // I6 trip; never contains the faulty intermediate value
    MalformedEncoding,       // generic decode failure (I9/I10: never more specific)
    KeyGenerationExhausted,  // rejection sampler exceeded a configured retry ceiling
    SerializationError(SerializationErrorKind),
}
// Derives: Debug, Display (safe — carries no secret material by construction), Error.

pub trait SignatureEncoding {
    // fn encode(message_hash: &[u8], em_len_bits: usize, salt_len: usize,
    //           rng: &mut dyn RngCore) -> Result<Vec<u8>, RsaError>;
    // fn verify(em: &[u8], message_hash: &[u8], em_len_bits: usize,
    //           salt_len: usize) -> Result<(), RsaError>;
}
// Implementors: EmsaPss, EmsaPkcs1V15 (§17). Stateless; no interior mutability.

pub trait EncryptionEncoding {
    // fn encode(message: &[u8], label: &[u8], k: usize,
    //           rng: &mut dyn RngCore) -> Result<Vec<u8>, RsaError>;
    // fn decode(em: &[u8], label: &[u8], k: usize) -> Result<Vec<u8>, RsaError>;
    //   // decode NEVER returns a variant distinguishing failure reason (I9/I10)
}
// Implementors: EmeOaep, EmePkcs1V15 (§18).

pub trait KeyFormat {
    // fn to_der(key: &RsaPrivateKey) -> Vec<u8>;
    // fn from_der(bytes: &[u8]) -> Result<RsaPrivateKey, RsaError>;
}
// Implementors: Pkcs1, Pkcs8, Jwk, Pem (§20). Round-trip identity is the §20.5 acceptance test.
