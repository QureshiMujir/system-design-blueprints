Core schema & types (§11)

struct SessionId(Uuid);
// own: daemon, immutable after creation | conc: Send+Sync, read-only after birth
// pers: SessionHeader field | ser: CBOR bytes | compat: stable since v0

struct SessionHeader {
    id: SessionId,
    created_at: Timestamp,
    host: HostInfo,
    capabilities: AdapterCapabilities,
    cfs_version: (u16, u16),
}
// own: writer at session start, read-only thereafter | conc: single-writer, multi-reader
// pers: header region (§8) | ser: CBOR | compat: unknown fields ignored per SemVer minor bump
struct FrameHeader {
    magic: u16,
    sequence: u64,
    kind: RecordKind,
    payload_len: u32,
}
// own: cfs-core writer exclusively | conc: single-writer append, lock-free reader scan
// pers: frame region prefix (§8) | ser: fixed-width binary | compat: frozen since v0, additive only

enum RecordKind {
    SessionStart, SessionEnd, BufferOpen, BufferEdit, BufferClose,
    CursorMove, SelectionChange, TerminalOutput, TerminalInput,
    BuildEvent, TestResult, DebuggerAttach, DebuggerStop, DebuggerVariable,
    VcsCheckpoint, Keyframe, Redaction, BlobRef, Seal,
}
// own: cfs-core, closed enum with reserved tag range for future kinds
// conc: immutable after definition | pers: 1-byte tag | ser: fixed enum discriminant
// compat: new kinds append to reserved range, never renumber (P2)
enum Record {
    SessionStart(SessionHeader), SessionEnd(CloseReason), BufferOpen(BufferState),
    BufferEdit(EditDelta), BufferClose(BufferId), CursorMove(CursorPos),
    SelectionChange(SelectionRange), TerminalOutput(TerminalChunk),
    TerminalInput(TerminalChunk), BuildEvent(BuildOutcome), TestResult(TestResult),
    DebuggerAttach(AdapterCapabilities), DebuggerStop(DebuggerFrame),
    DebuggerVariable(VariableSlice), VcsCheckpoint(VcsPosition),
    Keyframe(Snapshot), Redaction(RedactionAudit), BlobRef(ChunkRef), Seal(BlobDigest),
}
// own: capture-sched constructs, cfs-core persists | conc: constructed off capture threads,
// persisted by single writer | pers: frame payload | ser: CBOR | compat: variant-tagged, additive
struct BlobDigest([u8; 32]); // SHA-256, ADR-04
// own: chunker, content-derived (P6) | conc: Send+Sync, value type
// pers: index digest column, blob key | ser: raw bytes / hex in diagnostics
// compat: stable since v0; algorithm change requires new CFS major version

struct ChunkRef {
    digest: BlobDigest,
    offset: u64,
    length: u32,
}
// own: chunker + blob store | conc: read-shared once written | pers: blob region entry
// ser: CBOR | compat: additive fields only
enum SnapshotKind { Full, Delta { base: BlobDigest } }
// own: differ/replay | conc: value type | pers: Keyframe record tag | ser: CBOR
// compat: stable since v0

struct Snapshot {
    kind: SnapshotKind,
    buffers: Vec<BufferState>,
    debugger: Option<DebuggerFrame>,
    terminal_cursor: TerminalCursorState,
    vcs: VcsPosition,
    taken_at: Timestamp,
}
// own: capture-sched assembles, replay engine reconstructs | conc: constructed once, then immutable
// pers: Keyframe record body | ser: CBOR | compat: additive fields, absent fields = None
struct BufferState {
    buffer_id: BufferId,
    path: Utf8PathBuf,
    content_digest: BlobDigest,
    cursor: CursorPos,
    dirty: bool,
}
// own: IDE adapter emits, capture-sched batches | conc: one active writer per buffer_id
// pers: BufferOpen/Keyframe payload | ser: CBOR | compat: additive fields only

struct EditDelta {
    buffer_id: BufferId,
    base_digest: BlobDigest,   // precondition per I9
    ops: Vec<TextOp>,          // signatures only, no op bodies executed here
    result_digest: BlobDigest,
}
// own: IDE adapter emits | conc: ordered per buffer_id via sequence number (I7)
// pers: BufferEdit payload | ser: CBOR | compat: base_digest precondition enforced at apply time
struct DebuggerFrame {
    thread_id: u64,
    stack: Vec<StackFrameEntry>,
    scopes: Vec<ScopeRef>,
    stop_reason: DapStopReason,
}
// own: DAP adapter, passive observer only (ADR-10) | conc: one per DebuggerStop event
// pers: DebuggerStop payload | ser: CBOR | compat: mirrors DAP capability-gated fields (I10)

struct TerminalChunk {
    stream: PtyStream,          // stdout | stderr | stdin
    at: Timestamp,
    bytes_digest: BlobDigest,   // large payloads chunked (ADR-06), inline for small
}
// own: PTY tap | conc: append-only per stream | pers: TerminalOutput/Input payload
// ser: CBOR | compat: stable since v0

struct TestResult {
    suite: String,
    name: String,
    outcome: TestOutcome,
    duration_ms: u32,
    output_ref: Option<BlobDigest>,
}
// own: build/test adapter | conc: one record per completed test | pers: TestResult payload
// ser: CBOR | compat: additive fields only
struct VcsPosition {
    branch: String,
    commit: String,
    dirty_paths: Vec<Utf8PathBuf>,
    dirty_digest: BlobDigest,
}
// own: VCS poller in IDE adapter | conc: sampled, not event-driven | pers: VcsCheckpoint payload
// ser: CBOR | compat: additive fields only

struct RedactionRule {
    id: RuleId,
    pattern: RulePattern,       // regex | entropy-threshold | allow-list
    severity: RuleSeverity,
}
// own: scrubber, loaded from rule-set config | conc: read-only during a scan pass
// pers: not persisted in-session; audited via RedactionAudit records | ser: TOML config, not CBOR
// compat: rule-set version tracked independently of CFS version (P9)
struct IndexEntry {
    sequence: u64,
    offset: u64,
    kind: RecordKind,
    timestamp: Timestamp,
}
// own: cfs-index builder | conc: rebuildable, disposable (I6) | pers: SQLite page image
// ser: SQLite row | compat: derivable from frame stream at any version

struct ReplayCursor {
    session: SessionId,
    at_sequence: u64,
    nearest_keyframe: u64,
}
// own: replay engine, one per active replay session | conc: single reader per cursor instance
// pers: not persisted, transient | ser: n/a | compat: n/a

struct DiffResult {
    per_buffer: Vec<BufferDiff>,
    structural: Vec<StructuralChange>,
    symmetric: bool,             // asserts P11 round-trip property
}
// own: differ | conc: computed on demand, immutable once returned | pers: not persisted
// ser: CBOR for export | compat: additive fields only

struct AdapterCapabilities {
    supported_kinds: Vec<RecordKind>,
    supports_evaluate_requests: bool,
    protocol_version: (u16, u16),
}
// own: each adapter declares its own instance | conc: exchanged once at negotiate() (P10)
// pers: DebuggerAttach/SessionStart payload | ser: CBOR | compat: SemVer-gated (ADR-02)

enum ChronicleError {
    MalformedFrame { offset: u64, reason: String },
    DigestMismatch { expected: BlobDigest, found: BlobDigest },
    RedactionTimeout { record_kind: RecordKind },
    CapabilityViolation { op: String },
    SealedSessionMutation,
    IndexStale,
}
// own: cfs-core / scrubber / protocol-server, surfaced to CLI and adapters
// conc: value type, Send+Sync | pers: not persisted, logged to telemetry (P12)
// ser: Display + Debug for CLI/log output | compat: additive variants only
