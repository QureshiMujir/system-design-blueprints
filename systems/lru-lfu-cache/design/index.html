Core schema & types

#[derive(Copy, Clone, Eq, PartialEq, Debug)]
pub struct Slot { index: u32, generation: u32 }

impl Slot {
    pub const NONE: Slot;
    pub fn is_none(self) -> bool;
}
#[repr(C)]
pub struct Node<K, V> {
    pub key:          K,
    pub value:        V,
    pub weight:       u32,
    pub expires_at:   u64,        // absolute monotonic ns; 0 = never
    pub prev:         Slot,
    pub next:         Slot,
    pub policy_state: PolicyState,
    pub generation:   u32,
}
#[repr(C)]
pub union PolicyState {
    pub lru: LruState, pub lfu: LfuState, pub lru_k: LruKState,
    pub two_q: TwoQState, pub arc: ArcState, pub tiny_lfu: TinyLfuState,
    pub clock: ClockState, pub clock_pro: ClockProState,
    pub car: CarState, pub sieve: SieveState,
}
pub struct Arena<K, V> { /* Vec<MaybeUninit<Node>>, free: Vec<u32>, next_generation: u32 */ }

impl<K, V> Arena<K, V> {
    pub fn alloc(&mut self, node: Node<K, V>) -> Slot;
    pub fn free(&mut self, slot: Slot) -> Node<K, V>;
    pub fn get(&self, slot: Slot) -> Option<&Node<K, V>>;
    pub fn get_mut(&mut self, slot: Slot) -> Option<&mut Node<K, V>>;
}
pub trait Policy<K, V>: Send + 'static {
    fn name(&self) -> &'static str;
    fn on_hit(&mut self, arena: &mut Arena<K, V>, slot: Slot);
    fn admit(&mut self, key: &K, weight: u32) -> AdmitDecision;   // only W-TinyLFU may Reject (I7)
    fn on_insert(&mut self, arena: &mut Arena<K, V>, slot: Slot);
    fn evict(&mut self, arena: &mut Arena<K, V>) -> Slot;          // live slot, or NONE iff empty (I6)
    fn on_remove(&mut self, arena: &mut Arena<K, V>, slot: Slot);
    fn on_age(&mut self) {}
}

pub enum AdmitDecision { Admit, Reject }
pub struct Cache<K, V> { /* arena, map, policy, budget, ttl, admission, write, refresh, stats */ }

impl<K: Hash + Eq + Clone, V: Clone> Cache<K, V> {
    pub fn get(&mut self, key: &K) -> Option<V>;
    pub fn insert(&mut self, key: K, value: V) -> Option<V>;
    pub fn remove(&mut self, key: &K) -> Option<V>;
    pub fn contains_key(&self, key: &K) -> bool;
    pub fn len(&self) -> usize;
    pub fn stats(&self) -> StatsSnapshot;
}

pub enum PolicyChoice { Lru, Lfu, LruK(u32), TwoQ, Arc, WTinyLfu, Clock, ClockPro, Car, Sieve }
pub enum AdmissionChoice { TinyLfu, None }

pub struct CacheBuilder<K, V> { /* capacity, weigher, ttl_default, policy, admission, write, refresh, shards */ }
pub struct Budget { /* capacity_bytes, used_bytes, weigher */ }
impl Budget { pub fn over(&self) -> bool; pub fn add(&mut self, w: u32); pub fn sub(&mut self, w: u32); }

pub struct TimerWheel { /* levels: Vec<Wheel>, now_ms: u64 */ }
impl TimerWheel { pub fn schedule(&mut self, slot: Slot, expires_at_ms: u64); pub fn tick(&mut self, now_ms: u64) -> Vec<Slot>; }

pub enum WriteMode<K, V> { None, Through { sink: SinkFn<K, V> }, Back { queue: WriteBackQueue<K, V> }, Around { sink: SinkFn<K, V> } }
pub struct RefreshConfig<K, V> { pub window: Duration, pub loader: LoaderFn<K, V> }
