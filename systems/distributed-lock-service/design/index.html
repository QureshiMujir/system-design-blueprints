Core schema & types (§11)

pub struct Zxid(pub u64);                     // (epoch << 32) | counter; monotonic cluster-wide
impl Zxid {
    pub const ZERO: Self;
    pub fn epoch(self) -> u32;
    pub fn counter(self) -> u32;
    pub fn next(self) -> Self;                // panics on overflow (>100 yr headroom at 1e9 ops/s)
}

pub struct SessionId(pub u64);                // high 32 bits: per-leader nonce (P2, §14.7)
pub struct Path(String);                      // canonical: leading '/', no trailing '/', no '..', no '//'
impl Path {
    pub fn parse(raw: &str) -> Result<Self, Error>;   // rejects malformed at construction (§8.7)
    pub fn as_str(&self) -> &str;
    pub fn parent(&self) -> Option<Path>;
    pub fn last_component(&self) -> &str;
}
pub struct Stat {
    pub czxid: Zxid, pub mzxid: Zxid, pub pzxid: Zxid,
    pub ctime: i64, pub mtime: i64,             // advisory only, not load-bearing (§8.1)
    pub version: u32, pub cversion: u32, pub aversion: u32,
    pub ephemeral_owner: Option<SessionId>,     // None => persistent znode
    pub data_length: u32, pub num_children: u32,
}

pub struct Znode {
    pub path: String,
    pub data: Vec<u8>,                          // ≤ MAX_ZNODE_BYTES (I11)
    pub stat: Stat,
    pub acl: Vec<AclEntry>,
    pub children: alloc::collections::BTreeMap<String, ()>,   // index only; children live in Namespace
}
pub struct Namespace {
    nodes: alloc::collections::BTreeMap<String, Znode>,   // keyed by canonical path
    next_zxid: Zxid,
}
impl Namespace {
    pub fn new() -> Self;
    pub fn create(&mut self, path: &Path, data: &[u8], acl: &[AclEntry],
                   ephemeral_owner: Option<SessionId>, sequential: bool) -> Result<Zxid, Error>;
    pub fn delete(&mut self, path: &Path, expected_version: Option<u32>) -> Result<Zxid, Error>;
    pub fn set_data(&mut self, path: &Path, data: &[u8], expected_version: Option<u32>) -> Result<Zxid, Error>;
    pub fn get_data(&self, path: &Path) -> Result<(&[u8], &Stat), Error>;
    pub fn get_children(&self, path: &Path) -> Result<Vec<String>, Error>;
    pub fn exists(&self, path: &Path) -> bool;
}
pub enum SessionState { Connecting, Connected, Jeopardy, Expired }   // §8.2

pub struct Session {
    pub id: SessionId, pub timeout: core::time::Duration, pub state: SessionState,
    pub last_seen_zxid: Zxid, pub owner_replica: u32,
    pub ephemeral_znodes: alloc::collections::BTreeSet<Path>,
}

pub enum WatchKind { NodeCreated, NodeDeleted, NodeDataChanged, NodeChildrenChanged }

pub struct Watch {
    pub session: SessionId, pub kind: WatchKind, pub path: String,
    pub registered_at: Zxid,                    // last-seen zxid at registration; drives reconnect replay
}
pub struct Perms(pub u32);
impl Perms { pub const READ: Self; pub const WRITE: Self; pub const CREATE: Self;
             pub const DELETE: Self; pub const ADMIN: Self; pub const ALL: Self; }

pub struct AclEntry { pub scheme: String, pub id: String, pub perms: Perms }  // §21.1

pub enum Mutation {
    Create   { path: String, data: Vec<u8>, acl: Vec<AclEntry>,
               ephemeral_owner: Option<SessionId>, sequential: bool },
    Delete   { path: String, expected_version: Option<u32> },
    SetData  { path: String, data: Vec<u8>, expected_version: Option<u32> },
    SetAcl   { path: String, acl: Vec<AclEntry>, expected_aversion: Option<u32> },
    NewSession       { id: SessionId, timeout_ms: u32 },
    HeartbeatSession { id: SessionId },
    ExpireSession    { id: SessionId },          // atomically deletes owned ephemerals (I2)
    Sync     { id: SessionId },                  // no-op; reaches the commit boundary (I14)
}
pub enum Error {
    NoNode, NodeExists, BadVersion, NotEmpty, NoAuth, SessionExpired,
    ZnodeTooLarge, NotLeader { hint: u32 }, LimitExceeded, InvalidPath,
    FrameError(FrameErrorKind),
}
pub trait StateMachine: Send + 'static {
    fn apply(&mut self, index: u64, entry: &[u8]) -> Vec<u8>;
    fn snapshot(&self) -> Vec<u8>;
    fn restore(&mut self, snapshot: &[u8]);
}

pub enum Role { Follower, Candidate, Leader }

pub struct RaftNode<S: StateMachine> {
    state_machine: S, log: Log, role: Role, current_term: u64,
    voted_for: Option<u32>, commit_index: u64, last_applied: u64,
    next_index: alloc::collections::BTreeMap<u32, u64>,      // leader-only
    match_index: alloc::collections::BTreeMap<u32, u64>,     // leader-only
}
pub struct Wal { /* file handle, last_offset, last_index */ }
impl Wal {
    pub fn append(&mut self, entry: &LogEntry) -> Result<u64, Error>;     // fsync after write
    pub fn read_from(&self, index: u64) -> Result<Vec<LogEntry>, Error>;
    pub fn truncate_after(&mut self, index: u64) -> Result<(), Error>;
}

pub struct LogEntry { pub term: u64, pub index: u64, pub mutation: Mutation }

pub struct SnapshotWriter { /* chunked writer, per-chunk CRC32C */ }
pub struct SnapshotReader { /* chunked reader, verifies CRC32C per chunk + file-level */ }
pub struct ClientOptions { pub session_timeout: core::time::Duration, pub tls: TlsConfig }

pub struct Client { /* cluster addr list, session, connection, watch dispatcher */ }
impl Client {
    pub async fn connect(cluster: &[std::net::SocketAddr], opts: ClientOptions) -> Result<Self, Error>;
    pub async fn create(&self, path: &str, data: &[u8], mode: CreateMode) -> Result<String, Error>;
    pub async fn delete(&self, path: &str, version: Option<i32>) -> Result<(), Error>;
    pub async fn get_data(&self, path: &str, watch: bool) -> Result<(Vec<u8>, Stat), Error>;
    pub async fn set_data(&self, path: &str, data: &[u8], version: Option<i32>) -> Result<Stat, Error>;
    pub async fn get_children(&self, path: &str, watch: bool) -> Result<Vec<String>, Error>;
    pub async fn exists(&self, path: &str, watch: bool) -> Result<Option<Stat>, Error>;
    pub async fn sync(&self, path: &str) -> Result<(), Error>;
}

pub struct FencingToken(pub u64);
pub struct Lock { /* client handle, resource name, held child path, token */ }
impl Lock {
    pub async fn acquire(&mut self) -> Result<FencingToken, Error>;
    pub async fn release(&mut self) -> Result<(), Error>;
    pub fn is_definitely_held(&self) -> bool;      // false unless session Connected (§19.5)
}
