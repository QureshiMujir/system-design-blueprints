Core schema & types (§11)

// raytracer-math — immutable value types, Copy, no interior mutability.
pub struct Vec3 { pub x: f32, pub y: f32, pub z: f32 }        // direction/displacement, Copy
pub struct Point3 { pub x: f32, pub y: f32, pub z: f32 }      // position, Copy, affine-transformed
pub struct Normal3 { pub x: f32, pub y: f32, pub z: f32 }     // transforms by inverse-transpose
pub struct Interval { pub min: f32, pub max: f32 }            // t-range / bounds, Copy
pub struct Aabb { pub min: Point3, pub max: Point3 }          // Copy, union/intersect ops only
pub struct Transform { pub m: [[f32; 4]; 4], pub m_inv: [[f32; 4]; 4] } // Copy, precomputed inverse
// raytracer-core — ray types and the primitive/interaction contract.
pub struct Ray {                       // Copy; not normalized by convention (ray differentials)
    pub origin: Point3, pub direction: Vec3,
    pub t_min: f32, pub t_max: f32, pub time: f32,
}
pub struct RayDifferential {           // Copy; carries auxiliary rays for mip-level selection
    pub ray: Ray, pub rx_origin: Point3, pub ry_origin: Point3,
    pub rx_direction: Vec3, pub ry_direction: Vec3, pub has_differentials: bool,
}
pub trait Primitive: Send + Sync {     // immutable post-load, shared read-only across threads
    fn intersect(&self, ray: &Ray) -> Option<Intersection>;
    fn intersect_p(&self, ray: &Ray) -> bool;          // boolean-only, cheaper
    fn bounds(&self) -> Aabb;
    fn surface_area(&self) -> f32;
}
pub struct Triangle { pub v0: Point3, pub v1: Point3, pub v2: Point3, pub mesh_id: u32 } // Copy
pub struct Sphere { pub center: Point3, pub radius: f32, pub material_id: u32 }          // Copy
// raytracer-core — interaction records; produced per-hit, consumed by the integrator.
pub struct Intersection { pub t: f32, pub primitive_id: u32 }         // Copy, minimal hit record
pub struct SurfaceInteraction {                                       // owned per-hit, not shared
    pub point: Point3, pub geometric_normal: Normal3,
    pub shading_normal: Normal3, pub uv: (f32, f32),
    pub dpdu: Vec3, pub dpdv: Vec3, pub material_id: u32, pub wo: Vec3,
}
// raytracer-accel — BVH build/traversal state. Read-only after build (P6).
pub struct BvhNode {                    // Copy, 32 bytes (§8 byte layout)
    pub bounds: Aabb, pub left_or_offset: u32,
    pub count_or_right: u16, pub axis: u8,
}
pub struct BvhBuildRecord {             // transient, build-time only, not retained
    pub first_prim: u32, pub count: u32, pub bounds: Aabb, pub centroid_bounds: Aabb,
}
pub struct SahBucket {                  // transient per-axis binning accumulator (build-time)
    pub count: u32, pub bounds: Aabb,
}
// raytracer-bsdf — pure functions of (wo, wi, uniforms); no interior state.
pub trait Bsdf: Send + Sync {
    fn f(&self, wo: Vec3, wi: Vec3) -> Vec3;                     // evaluate
    fn sample_f(&self, wo: Vec3, u: (f32, f32)) -> BsdfSample;   // sample
    fn pdf(&self, wo: Vec3, wi: Vec3) -> f32;                    // matches sample_f's density (I2)
    fn flags(&self) -> BsdfFlags;
}
pub struct BsdfSample { pub wi: Vec3, pub f: Vec3, pub pdf: f32, pub flags: BsdfFlags } // Copy
pub struct BsdfFlags { pub specular: bool, pub reflection: bool, pub transmission: bool } // Copy
pub struct Lambertian { pub albedo: Vec3 }                                  // immutable, Copy
pub struct Microfacet { pub albedo: Vec3, pub alpha: f32, pub metallic: bool } // immutable, Copy
pub struct Dielectric { pub ior: f32, pub alpha: f32 }                      // immutable, Copy
// raytracer-lights — sampling contract mirrors Bsdf; PDFs in solid-angle measure.
pub trait Light: Send + Sync {
    fn sample_li(&self, reference: Point3, u: (f32, f32)) -> LightSample;
    fn pdf_li(&self, reference: Point3, wi: Vec3) -> f32;
    fn is_infinite(&self) -> bool;
}
pub struct LightSample { pub radiance: Vec3, pub wi: Vec3, pub pdf: f32, pub distance: f32 } // Copy
pub struct AreaLight { pub shape_id: u32, pub emission: Vec3 }             // immutable, Copy
pub struct EnvironmentLight { pub map_id: u32 }                            // immutable, Copy; map owned by Texture system
// raytracer-sampling — one Sampler instance per thread, never shared (P12).
pub trait Sampler: Send {           // Send only, not Sync — per-thread ownership
    fn get_1d(&mut self, dim: SampleDimension) -> f32;
    fn get_2d(&mut self, dim: SampleDimension) -> (f32, f32);
    fn clone_for_pixel(&self, pixel: (u32, u32), sample_index: u32) -> Self;
}
pub struct SampleDimension(pub u32); // Copy; indexes a fixed dimension budget per path vertex
// raytracer-render — camera, film, tiling. Film is the sole cross-thread mutable state (P14).
pub trait Camera: Send + Sync {
    fn generate_ray(&self, sample: CameraSample) -> Ray;
}
pub struct CameraSample { pub p_film: (f32, f32), pub p_lens: (f32, f32), pub time: f32 } // Copy
pub struct Film { /* atomic per-pixel sum + sum_sq + count; owns the buffer, Sync via atomics */ }
pub struct Pixel { pub sum: Vec3, pub sum_sq: Vec3, pub count: u32 }        // Copy, per-pixel state
pub struct Tile { pub x0: u32, pub y0: u32, pub x1: u32, pub y1: u32 }      // Copy, scheduling unit
// raytracer-texture, raytracer-core — colour, textures, scene, config, errors.
pub struct Rgb { pub r: f32, pub g: f32, pub b: f32 }         // Copy; linear working space (ADR-10)
pub trait Texture: Send + Sync { fn eval(&self, uv: (f32, f32), duv: (f32, f32)) -> Rgb; }
pub struct MipMap { /* owns a box-filtered level chain; immutable post-build */ }
pub struct Scene {                       // immutable post-load (P6); owns geometry, materials, lights
    pub primitives: Vec<Box<dyn Primitive>>, pub lights: Vec<Box<dyn Light>>,
    pub bvh: /* BvhNode array */ Vec<BvhNode>, pub camera: Box<dyn Camera>,
}
pub struct RenderConfig { pub spp: u32, pub max_depth: u32, pub seed: u64, pub tile_size: u32 } // Copy, serializable (ADR-22)
pub struct RenderStats { pub rays_traced: u64, pub nan_samples: u64, pub build_time_ms: f64 }  // accumulated, not serialized cross-version (§27)
pub enum RenderError { SceneParse(String), GltfImport(String), BvhBuild(String), Io(String) }  // stable variants across MINOR versions (§27)
