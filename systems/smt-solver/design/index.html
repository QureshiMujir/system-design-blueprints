Core schema & types (§11)

pub struct TermRef(pub u32);   // hash-consed handle; structural-equality => handle identity
pub struct SortRef(pub u32);   // same identity contract as TermRef
pub struct Symbol(pub u32);    // interned identifier

pub enum Sort {
    Bool,
    Int,
    Real,
    BitVec(u32),                      // width w >= 1; fixed-width
    Uninterpreted(Symbol),
    Func(Box<[SortRef]>, SortRef),    // (dom...) -> range, arity >= 1
    Array(SortRef, SortRef),          // (index, element); stretch theory
}

pub enum TermKind {
    Const(Symbol), BoolLit(bool), IntLit(BigInt), RealLit(BigRational), BvLit(BitVec),
    Var(Symbol),
    App(Symbol, Box<[TermRef]>),
    Not(TermRef), And(Box<[TermRef]>), Or(Box<[TermRef]>), Implies(TermRef, TermRef),
    Eq(TermRef, TermRef),             // same-sort equality (=> theory atom for EUF/arith)
    Distinct(Box<[TermRef]>),         // pairwise !=; desugared lazily
    Ite(TermRef, TermRef, TermRef),   // cond:Bool, then/else same sort
    Arith(ArithOp, Box<[TermRef]>),   // linearity ENFORCED at sort-check
    Bv(BvOp, Box<[TermRef]>),
    Forall(Box<[(Symbol, SortRef)]>, TermRef), // STUB
    Exists(Box<[(Symbol, SortRef)]>, TermRef), // STUB
}

pub struct Term { pub kind: TermKind, pub sort: SortRef }
pub struct TermBank { /* arena + HashMap<TermKind,TermRef> + Symbol interner */ }
impl TermBank {
    pub fn intern(&mut self, kind: TermKind) -> Result<TermRef, ParseError>;
    pub fn intern_sort(&mut self, s: Sort) -> SortRef;
    pub fn symbol(&mut self, name: &str) -> Symbol;
    pub fn kind(&self, t: TermRef) -> &TermKind;
    pub fn sort_of(&self, t: TermRef) -> SortRef;
}

pub enum Value { Bool(bool), Int(BigInt), Real(BigRational), Bv(BitVec), Uf(/* finite fn graph */), }

pub enum Logic { QF_UF, QF_LIA, QF_LRA, QF_BV, QF_UFLIA, QF_AUFLIA, /* ... */ Unsupported }
pub enum Command {
    SetLogic(Logic), DeclareFun(Symbol, Box<[SortRef]>, SortRef), Assert(TermRef),
    CheckSat, Push(u32), Pop(u32), GetModel, GetValue(Box<[TermRef]>), GetUnsatCore,
    Exit, SetOption(/* opt + value */),
}

pub fn parse_script<R: Read>(r: R, ctx: &mut CommandContext) -> Result<(), ParseError>;

pub enum ParseError {
    LexError, UnbalancedParens, UnknownSymbol, SortMismatch, ArityMismatch, UnsupportedLogic,
}

pub struct CommandContext { /* bank: TermBank, sat: SatCore, atoms: AtomMap,
    theories: TheoryManager, stack: Vec<ScopeLevel>, logic: Logic,
    names: HashMap<Symbol, TermRef> /* named asserts for cores */ */ }
pub struct SatCore { /* trail, watches, VSIDS, analyze, learnt clauses */ }
pub struct AtomMap { /* Vec<Option<TermRef>> by BoolVar + HashMap<TermRef,BoolVar> */ }
pub struct BoolVar(pub u32);

/// Tseitin-encode the Boolean skeleton of `skeleton` into `out`, registering each
/// theory atom in `map`. Fresh BoolVar per non-atomic subformula; equisatisfiable.
pub fn tseitin(skeleton: TermRef, bank: &TermBank, out: &mut SatCore, map: &mut AtomMap);
pub trait TheorySolver {
    fn assert_lit(&mut self, atom: TermRef, polarity: bool) -> Result<(), TheoryConflict>;
    fn check(&mut self, final_check: bool) -> Result<Vec<TheoryPropagation>, TheoryConflict>;
    fn explain(&mut self, lit: TermRef) -> Vec<TermRef>;
    fn push(&mut self);
    fn pop(&mut self);
    fn model(&self, b: &mut ModelBuilder);
}

pub struct TheoryConflict { pub core: Vec<TermRef> }
pub struct TheoryPropagation { pub lit: TermRef, pub polarity: bool }
pub struct TheoryManager { /* Vec<Box<dyn TheorySolver>> + eq-sharing bus + routing */ }
pub enum CheckResult { Sat, Unsat, Unknown }

pub fn check_sat(ctx: &mut CommandContext) -> CheckResult;
pub struct CongruenceClosure { /* union-find + proof forest + App e-graph + pending queue */ }
impl CongruenceClosure {
    pub fn merge(&mut self, a: TermRef, b: TermRef);
    pub fn are_equal(&self, a: TermRef, b: TermRef) -> bool;
    pub fn explain_eq(&self, a: TermRef, b: TermRef) -> Vec<TermRef>;
}

pub struct Simplex { /* tableau: Vec<Row<BigRational>>, basic/nonbasic sets, Bound[] */ }
pub struct Bound { /* lo: Option<BigRational>, hi: Option<BigRational>, reasons */ }
pub struct Tableau { /* rows of BigRational coefficients; pivot ops */ }
pub struct ArithVar(pub u32);

impl Simplex {
    pub fn assert_bound(&mut self, var: ArithVar, b: Bound) -> Result<(), TheoryConflict>;
    pub fn check(&mut self) -> Result<(), TheoryConflict>;
}

/// QF_BV eager solver: maps each width-w BvTerm bit to a SAT BoolVar
pub struct BitBlaster { /* bit -> BoolVar table + clause emission into SatCore */ }
pub struct ModelBuilder { /* var->Value, UF graphs, default values per sort */ }
pub fn evaluate(t: TermRef, model: &ModelBuilder) -> Value;
pub struct UnsatCore(pub Vec<TermRef>);
