Core schema & types (§11)

// fs-types/src/entity.rs — immutable, Clone, Send+Sync, Arrow+serde round-trippable
pub struct EntityName(String);           // newtype, interned at registry load
pub struct FeatureName(String);          // newtype

pub struct EntityId {
    pub entity: EntityName,
    pub value: String,                   // non-empty, validated at construction
}
impl EntityId {
    pub fn key(&self) -> String;         // canonical "entity:value" online-store key
}

pub struct EntityDef {                   // registry row; immutable once fetched in a snapshot
    pub name: EntityName,
    pub join_key: String,                // source column name
    pub join_dtype: DType,
    pub owner: String,
    pub description: String,
    pub tags: BTreeMap<String, String>,
}
impl EntityDef {
    pub fn validate_can_join(&self, source_columns: &[&str]) -> Result<(), SchemaError>;
}
// fs-types/src/dtype.rs — the closed enumeration (P12, I12)
pub enum DType {
    Int32, Int64, Float32, Float64, Bool, String, Bytes,
    VectorFloat32, VectorFloat64, ListInt64, ListString,
}

pub enum FeatureMissReason { NotMaterialized, TtlExpired, SourceError, UnknownFeature, Unauthorized }

pub enum ScalarValue { I32(i32), I64(i64), F32(f32), F64(f64), Bool(bool),
    Str(String), Bytes(Vec<u8>), VecF32(Vec<f32>), VecF64(Vec<f64>),
    ListI64(Vec<i64>), ListStr(Vec<String>) }

pub struct FeatureValue {                // wire+in-memory value; frozen after construction
    pub dtype: DType,
    pub value: Option<ScalarValue>,
    pub is_null: bool,
    pub miss_reason: Option<FeatureMissReason>,
}
// fs-types/src/feature.rs
pub struct FeatureDef {                  // one row in the registry; version-bumped on change (§21)
    pub name: FeatureName,
    pub dtype: DType,
    pub source_ref: String,              // e.g. "parquet:...", "sql:...", "kafka:..."
    pub expression: String,              // column ref or SQL expr
    pub nullable: bool,
    pub default: Option<ScalarValue>,
    pub freshness_sla: Duration,
    pub ttl: Duration,
    pub owner: String,
    pub tags: BTreeMap<String, String>,
    pub version: u32,
}
impl FeatureDef {
    pub fn is_within_freshness(&self, last_materialized: DateTime<Utc>, now: DateTime<Utc>) -> bool;
}

pub struct FeatureView {                 // unit of scheduling (§16)
    pub name: String,
    pub entities: Vec<EntityName>,
    pub features: Vec<FeatureDef>,
    pub source_ref: String,
    pub ttl: Duration,
    pub online: bool,
    pub offline: bool,
    pub owner: String,
}
impl FeatureView { pub fn feature_names(&self) -> Vec<&FeatureName>; }

pub struct FeatureService {              // unit of deployment; the model-integration boundary
    pub name: String,
    pub features: Vec<String>,           // "view:feature" refs
    pub owner: String,
}
impl FeatureService { pub fn required_views(&self) -> BTreeSet<String>; }
// fs-types/src/job.rs, fs-types/src/registry.rs
pub struct MaterializationJob {
    pub job_id: String,
    pub feature_view: String,
    pub start_ts: DateTime<Utc>,
    pub end_ts: DateTime<Utc>,
    pub backend: JobBackend,              // Batch | Stream | OnDemand
    pub triggered_by: TriggerKind,        // Schedule | Event | Manual
}
impl MaterializationJob { pub fn duration(&self) -> Duration; }

pub struct RegistrySnapshot {             // immutable; SDK/materialize refresh periodically
    pub snapshot_id: String,
    pub fetched_at: DateTime<Utc>,
    pub entities: BTreeMap<EntityName, EntityDef>,
    pub views: BTreeMap<String, FeatureView>,
    pub services: BTreeMap<String, FeatureService>,
    pub git_commit: String,
}
impl RegistrySnapshot {
    pub fn view_for_feature(&self, feature_ref: &str) -> Result<&FeatureView, RegistryError>;
    pub fn feature_def(&self, feature_ref: &str) -> Result<&FeatureDef, RegistryError>;
}
// fs-types/src/store.rs — trait boundary every backend implements (mirrors Feast's Protocol, Ref-Impl 1)
pub trait OnlineStore: Send + Sync {
    fn write(&self, view: &str, batch: &RecordBatch) -> Result<(), StoreError>;
    fn read(&self, view: &str, ids: &[EntityId], features: &[FeatureName])
        -> Result<BTreeMap<EntityId, BTreeMap<FeatureName, FeatureValue>>, StoreError>;
    fn delete(&self, view: &str, ids: &[EntityId]) -> Result<(), StoreError>;
}

pub trait OfflineStore: Send + Sync {
    fn write(&self, view: &str, batch: &Table, event_ts_column: &str) -> Result<(), StoreError>;
    fn point_in_time_join(&self, entity_df: &Table, feature_refs: &[String], ts_col: &str)
        -> Result<Table, StoreError>;
}
// fs-types/src/request.rs, fs-types/src/lineage.rs
pub enum AsOf { Latest, At(DateTime<Utc>) }

pub struct GetFeaturesRequest {
    pub entity_ids: Vec<EntityId>,
    pub feature_refs: Vec<String>,
    pub as_of: AsOf,
    pub max_age: Option<Duration>,
    pub trace_id: Option<String>,
}

pub struct GetFeaturesResponse {
    pub entity_ids: Vec<EntityId>,
    pub feature_refs: Vec<String>,
    pub values: Vec<Vec<FeatureValue>>,   // [entity][feature]
    pub served_at: DateTime<Utc>,
    pub snapshot_id: String,
}
impl GetFeaturesResponse {
    pub fn to_arrow(&self) -> RecordBatch;
    pub fn get(&self, entity_idx: usize, feature_ref: &str) -> Option<&FeatureValue>;
}

pub enum LineageEventKind { MaterializationStart, MaterializationComplete,
    MaterializationFailure, Read, Write, SchemaChange }

pub struct LineageEvent {                 // I8; OpenLineage-shape-compatible (ADR-17)
    pub event_id: String,
    pub kind: LineageEventKind,
    pub trace_id: String,
    pub span_id: String,
    pub parent_span_id: Option<String>,
    pub occurred_at: DateTime<Utc>,
    pub feature_view: Option<String>,
    pub feature_refs: Vec<String>,
    pub sources: Vec<String>,
    pub row_count: Option<u64>,
    pub consumer: Option<String>,
    pub git_commit: Option<String>,
    pub error: Option<String>,
}
// fs-types/src/monitor.rs
pub struct FreshnessReport { pub view: String, pub last_materialized: DateTime<Utc>,
    pub sla: Duration, pub now: DateTime<Utc> }
impl FreshnessReport { pub fn lag(&self) -> Duration; pub fn within_sla(&self) -> bool; }

pub enum QualityConstraintKind { NotNull, InRange, DistinctCount, Regex }
pub struct QualityConstraint { pub feature_ref: String, pub kind: QualityConstraintKind,
    pub severity: Severity }             // Severity: Warn | Fail

pub enum DriftMetric { Psi, Ks, Wasserstein, Js }
pub struct DriftReport { pub feature_ref: String, pub metric: DriftMetric, pub value: f64,
    pub threshold: f64 }
impl DriftReport { pub fn drifted(&self) -> bool; }
