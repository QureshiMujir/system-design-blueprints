Core schema & types (§11)

#[derive(Debug, Clone, PartialEq, Eq, Hash, PartialOrd, Ord)]
pub struct StreamId(Arc<str>);

impl StreamId {
    pub fn new(s: impl Into<Arc<str>>) -> Result<Self, InvalidStreamId>;
    pub fn as_str(&self) -> &str;
    pub fn category(&self) -> &str;          // prefix before first '-'; invariant: non-empty
    pub fn is_system(&self) -> bool;         // starts with '$'
}
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub struct EventId(pub Uuid);

impl EventId {
    pub fn new_v7() -> Self;               // time-ordered UUIDv7; preferred
    pub fn from_uuid(u: Uuid) -> Self;
}
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]
pub struct EventNumber(pub u64);

impl EventNumber {
    pub const START: Self = EventNumber(0);
    pub fn next(self) -> Self;             // wrapping-add would be a bug; panic in debug
    pub fn checked_next(self) -> Option<Self>;
}
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]
pub struct GlobalPosition(pub u64);

impl GlobalPosition {
    pub const START: Self = GlobalPosition(0);
    pub fn as_u64(self) -> u64;
}
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ExpectedVersion {
    Any,                       // disable concurrency check; always append
    NoStream,                  // stream must not exist (current version == -1)
    StreamExists,              // stream must exist; any version acceptable
    Exact(EventNumber),        // stream's current last event number must match
}
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ReadDirection { Forward, Backward }

pub struct ReadResult<T> {
    pub events: Vec<T>,
    pub next_position: Option<T>,   // None when is_end
    pub is_end: bool,
}
#[derive(Debug, Clone, Copy)]
pub struct AppendResult {
    pub next_expected_version: EventNumber,   // last written event number
    pub commit_position: GlobalPosition,      // position of last event in $all
}
#[derive(Debug, Clone)]
pub struct EventRecord {
    pub event_id:    EventId,
    pub stream:      StreamId,
    pub number:      EventNumber,
    pub event_type:  Arc<str>,           // domain type string; e.g. "OrderPlaced"
    pub data:        Bytes,              // opaque; store never parses (P10 / ADR-0010)
    pub metadata:    Bytes,              // causation-id, correlation-id, custom fields
    pub content_type: ContentType,       // JSON | Binary | custom MIME
    pub position:    GlobalPosition,     // assigned on commit; zero before commit
    pub created:     SystemTime,         // wall-clock; approximate; not a sort key
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ContentType { Json, Binary, Custom([u8; 32]) }
pub trait UpcastFn: Send + Sync {
    fn event_type(&self) -> &str;
    fn from_version(&self) -> u32;
    fn to_version(&self) -> u32;
    fn upcast(&self, data: Bytes) -> Result<Bytes, UpcastError>;
}
pub struct Segment {
    pub id:         u64,               // monotonic segment sequence number
    pub path:       PathBuf,
    pub base_offset: u64,              // first record's byte offset in this file
    pub size:       u64,               // current written bytes
}

pub const SEGMENT_MAX_BYTES: u64 = 256 * 1024 * 1024;  // 256 MiB; configurable
pub const RECORD_HEADER_LEN: usize = 8;                  // 4-byte len + 4-byte CRC
#[derive(Debug, Clone, Copy)]
pub struct IndexEntry {
    pub stream_hash: u64,          // FNV-1a of StreamId bytes
    pub number:      EventNumber,
    pub offset:      u64,          // byte offset in segment file
    pub position:    GlobalPosition,
}
pub trait StreamEngine: Send + Sync {
    fn append(
        &self,
        stream: &StreamId,
        expected: ExpectedVersion,
        events: Vec<NewEvent>,
    ) -> Result<AppendResult, AppendError>;

    fn read(
        &self,
        stream: &StreamId,
        from: EventNumber,
        count: usize,
        dir: ReadDirection,
    ) -> Result<ReadResult<EventRecord>, ReadError>;

    fn tombstone(&self, stream: &StreamId, expected: ExpectedVersion)
        -> Result<(), AppendError>;
}

pub struct NewEvent {
    pub event_id:    EventId,
    pub event_type:  Arc<str>,
    pub data:        Bytes,
    pub metadata:    Bytes,
    pub content_type: ContentType,
}
#[derive(Debug, Clone)]
pub struct Snapshot {
    pub stream:        StreamId,
    pub version:       EventNumber,      // stream version at snapshot time
    pub state:         Bytes,            // serialized aggregate state; opaque
    pub position:      GlobalPosition,   // $all position at snapshot time
    pub schema_version: u32,             // invalidated if projection schema changes
    pub created:       SystemTime,
}
pub trait SnapshotStore: Send + Sync {
    fn write(&self, snap: Snapshot) -> Result<(), SnapshotError>;
    fn load_nearest(
        &self,
        stream: &StreamId,
        at_or_before: EventNumber,
    ) -> Result<Option<Snapshot>, SnapshotError>;
    fn invalidate(&self, stream: &StreamId, schema_version: u32)
        -> Result<u64, SnapshotError>;   // returns count invalidated
}
#[derive(Debug, Clone)]
pub struct Projection {
    pub name:       Arc<str>,
    pub query:      Arc<str>,               // DSL or handler registration token
    pub checkpoint: Option<GlobalPosition>, // None = never checkpointed
    pub state:      Bytes,                  // opaque; handler-defined
    pub mode:       ProjectionMode,
    pub status:     ProjectionStatus,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ProjectionMode { Continuous, OneTime }

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ProjectionStatus { Running, Paused, Faulted, Rebuilding }
pub trait ProjectionHandler: Send + Sync {
    fn event_types(&self) -> &[&str];          // filter; empty = all types
    fn apply(
        &self,
        state: &mut dyn Any,
        event: &EventRecord,
    ) -> Result<(), ProjectionError>;          // must be deterministic (I8)
    fn initial_state(&self) -> Box<dyn Any + Send>;
}
#[derive(Debug, Clone)]
pub struct Subscription {
    pub id:     Uuid,
    pub filter: SubscriptionFilter,
    pub from:   SubscriptionPosition,
    pub kind:   SubscriptionKind,
}

#[derive(Debug, Clone)]
pub enum SubscriptionFilter {
    All,
    Stream(StreamId),
    Category(Arc<str>),
    EventType(Arc<str>),
}

#[derive(Debug, Clone, Copy)]
pub enum SubscriptionPosition {
    Start,
    End,
    After(GlobalPosition),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SubscriptionKind { Volatile, Persistent }
#[derive(Debug, Clone, Copy)]
pub struct Checkpoint {
    pub subscription_id: Uuid,
    pub position:        GlobalPosition,
    pub committed_at:    SystemTime,
}
#[derive(Debug, Clone)]
pub struct ReplicationEntry {
    pub position: GlobalPosition,       // the assigned $all position
    pub term:     u64,                  // Raft term in which leader appended this
    pub record:   EventRecord,          // the committed event
}
#[derive(Debug, Clone)]
pub struct ClusterNode {
    pub node_id:      NodeId,               // stable UUID
    pub addr:         SocketAddr,
    pub role:         NodeRole,
    pub term:         u64,
    pub commit_index: GlobalPosition,       // highest committed position known
    pub match_index:  GlobalPosition,       // highest position acked to leader (leader view)
    pub last_contact: SystemTime,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum NodeRole { Leader, Follower, Candidate, PreCandidate }
pub trait RaftTransport: Send + Sync {
    async fn send_append_entries(
        &self, target: NodeId, req: AppendEntriesRequest,
    ) -> Result<AppendEntriesResponse, TransportError>;

    async fn send_vote_request(
        &self, target: NodeId, req: VoteRequest,
    ) -> Result<VoteResponse, TransportError>;
}
pub trait StreamRouter: Send + Sync {
    fn owner_of(&self, stream: &StreamId) -> NodeId;
    fn is_local(&self, stream: &StreamId) -> bool;
    fn all_owners(&self) -> Vec<(NodeId, Vec<StreamId>)>;
}
#[derive(Debug, Clone)]
pub struct ScavengePlan {
    pub segments:    Vec<SegmentPlan>,
    pub initiated_at: SystemTime,
    pub initiated_by: Arc<str>,    // operator identity
}

#[derive(Debug, Clone)]
pub struct SegmentPlan {
    pub segment_id: u64,
    pub live_ranges: Vec<Range<u64>>,   // byte ranges to preserve
    pub bytes_to_remove: u64,
}
