// crates/ush-tables/src/trie.rs
/// Two-stage trie: O(1) code-point lookup. index[] selects 64-entry data block; data[] holds values.
/// Entirely static; zero heap allocation; const-constructible from generated arrays.
pub struct CodePointTrie {
    index: &'static [u16],   // block index; len = (0x110000 >> 6) + overflow sentinel
    data:  &'static [u32],   // packed property values; block size = 64 entries
}
impl CodePointTrie {
    pub const fn new(index: &'static [u16], data: &'static [u32]) -> Self;
    #[inline] pub fn get(&self, c: char) -> u32;  // index[(c as usize)>>6]*64 + (c & 63)
}
// Send + Sync: immutable static data; no interior mutability.

#[derive(Clone, Copy, PartialEq, Eq, Debug)]
pub enum NormForm { Nfc, Nfd, Nfkc, Nfkd }

/// Stateless; cloneable; Send + Sync. Holds form selection only.
#[derive(Clone, Copy)]
pub struct Normalizer { pub form: NormForm }

impl Normalizer {
    pub const fn new(form: NormForm) -> Self;
    /// Allocates output String; input is not mutated.
    pub fn normalize_str(&self, input: &str) -> String;
}

/// Entry point; equivalent to Normalizer::new(form).normalize_str(input).
pub fn normalize(form: NormForm, input: &str) -> String;

#[derive(Clone, Copy, PartialEq, Eq, Debug)]
pub enum QuickCheck { Yes, No, Maybe }

/// Returns Yes only if entire input is proven normalized; No/Maybe triggers full pass.
pub fn quick_check(form: NormForm, input: &str) -> QuickCheck;

/// Stable sort of (char, ccc) pairs by ccc within maximal non-zero-ccc runs.
/// ccc==0 entries act as blockers; sort does not cross them.
pub fn canonical_order(buf: &mut Vec<(char, u8)>);

/// Byte-offset cursor into a &str; tracks GCB state for incremental boundary detection.
/// Borrows input; zero-copy. Not Send across threads while borrow is live (tied to 'a).
pub struct GraphemeCursor<'a> {
    text: &'a str,
    pos:  usize,           // current byte offset
    // internal GCB state for regional-indicator pairing + ZWJ sequence tracking
}
impl<'a> GraphemeCursor<'a> {
    pub fn new(text: &'a str) -> Self;
    /// Advance to next grapheme cluster boundary; returns byte range of cluster or None at end.
    pub fn next_boundary(&mut self) -> Option<std::ops::Range<usize>>;
    pub fn is_boundary(&self, offset: usize) -> bool;
}

// crates/ush-bidi/src/classes.rs
#[derive(Clone, Copy, PartialEq, Eq, Debug)]
#[repr(u8)]
pub enum BidiClass {
    L, R, AL, EN, ES, ET, AN, CS, NSM, BN, B, S, WS, ON,
    LRE, RLE, LRO, RLO, PDF,   // explicit embeddings/overrides (legacy)
    LRI, RLI, FSI, PDI          // isolates (Unicode 6.3+)
}

/// Embedding level 0..=125. Even = LTR, odd = RTL.
/// Invariant: 0 <= self.0 <= 125; enforced at construction.
#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Debug)]
pub struct Level(pub u8);
impl Level {
    pub const LTR: Level = Level(0);
    pub const RTL: Level = Level(1);
    pub const MAX: Level = Level(125);
    #[inline] pub fn is_rtl(self) -> bool { self.0 & 1 == 1 }
    /// Raise to next LTR or RTL level as required by explicit rule semantics.
    pub fn raise_to_ltr(self) -> Option<Level>;  // None if would exceed 125
    pub fn raise_to_rtl(self) -> Option<Level>;
}

/// Per-paragraph bidi resolution output. Borrows nothing (owns Vecs).
/// levels[i] and classes[i] correspond to the i-th char (by char index, not byte).
/// Send + Sync. Single-use: produced by process_paragraph, consumed by reorder_line.
pub struct BidiInfo {
    pub levels:     Vec<Level>,
    pub classes:    Vec<BidiClass>,   // resolved (post W/N/I rules)
    pub para_level: Level,
}

/// Full UAX#9 pipeline: class assignment -> X rules -> isolating-run-sequences ->
/// W/N/I rules -> level assignment. base=None triggers P2/P3 auto-detection (first strong).
pub fn process_paragraph(text: &str, base: Option<Level>) -> BidiInfo;

/// L2 reordering: given levels for a line range, return visual-order char indices.
/// Pure function; allocates the output Vec. Input slice is the levels subrange for the line.
pub fn reorder_line(levels: &[Level], line: std::ops::Range<usize>) -> Vec<usize>;

/// Bidi mirroring character for bracket/symbol chars at odd levels. None if no mirror.
pub fn mirror(c: char) -> Option<char>;

/// OpenType 4-byte tag. Shown in prose as e.g. latn, liga, GSUB (no backticks).

pub struct Tag(pub [u8; 4]);
impl Tag {

// crates/ush-shape/src/face.rs
/// Borrows the raw font bytes for its lifetime. Zero-copy; bounds-checked on every access.
/// Immutable after construction; Send + Sync; shareable across shaping calls.
pub struct Face<'a> {
    data:          &'a [u8],
    table_dir:     TableDirectory<'a>,  // sfnt table offsets + lengths
    pub units_per_em: u16,
}
impl<'a> Face<'a> {
    pub fn new(data: &'a [u8]) -> Result<Self, ParseError>;
    pub fn glyph_h_advance(&self, glyph_id: u32) -> u16;
    pub fn glyph_v_advance(&self, glyph_id: u32) -> u16;
}
// crates/ush-shape/src/buffer.rs — glyph buffer types
/// Per-glyph identity and cluster linkage. cluster = index of source char (monotonic).
/// flags: bit 0 = UNSAFE_TO_BREAK; bit 1 = DEFAULT_IGNORABLE; others reserved.
#[derive(Clone, Copy, Default)]
pub struct GlyphInfo {
    pub glyph_id: u32,
    pub cluster:  u32,   // monotonic; merges on ligature; splits on multiple-subst
    pub flags:    u16,
}

/// Accumulated positioning deltas in font design units (1/units_per_em of an em).
#[derive(Clone, Copy, Default)]
pub struct GlyphPosition {
    pub x_advance: i32,
    pub y_advance: i32,
    pub x_offset:  i32,
    pub y_offset:  i32,
}

#[derive(Clone, Copy, PartialEq, Eq, Debug)]
pub enum Direction { Ltr, Rtl, Ttb, Btt }

/// Mutable shaping workspace. Single-threaded; NOT Send while being shaped.
pub struct Buffer {
    pub infos:      Vec<GlyphInfo>,
    pub positions:  Vec<GlyphPosition>,
    pub direction:  Direction,
}
impl Buffer {
    pub fn new(direction: Direction) -> Self;
    pub fn add_str(&mut self, text: &str);
    pub fn merge_clusters(&mut self, start: usize, end: usize);
    pub fn reverse(&mut self);
}
// crates/ush-shape/src/shape.rs — pipeline entry + shaper trait
/// Full shaping pipeline: cmap char->glyph mapping, feature selection, Shaper hooks,
/// GSUB substitution, GPOS positioning. Mutates buf in place.
pub fn shape(face: &Face, buf: &mut Buffer, features: &[(Tag, u32)]);

/// Script-specific shaping hook interface. Stateless; methods take shared &self.
pub trait Shaper {
    fn collect_features(&self, plan: &mut ShapePlan);
    fn setup_masks(&self, face: &Face, buf: &mut Buffer, plan: &ShapePlan);
    fn reorder(&self, face: &Face, buf: &mut Buffer);
}

/// Resolved feature set + script + lang for one shaping call.
pub struct ShapePlan {
    pub script:    Tag,
    pub language:  Tag,
    pub direction: Direction,
    // feature bitmask + ordered lookup list resolved from ScriptList/FeatureList
}
