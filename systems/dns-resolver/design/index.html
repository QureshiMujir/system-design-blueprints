Core schema & types (§11)

pub struct Name { bytes: bytes::Bytes, label_count: u8 }   // canonical-lowercase, no pointers stored

impl Name {
    pub fn root() -> Self;
    pub fn from_ascii(s: &str) -> Result<Self, NameError>;
    pub fn from_wire(buf: &[u8]) -> Result<(Self, usize), NameError>;
    pub fn label_count(&self) -> u8;
    pub fn is_root(&self) -> bool;
    pub fn parent(&self) -> Option<Name>;
    pub fn is_in_bailiwick_of(&self, zone: &Name) -> bool;      // I1
    pub fn cmp_canonical(&self, other: &Name) -> std::cmp::Ordering;  // I9
    pub fn encode(&self, out: &mut bytes::BytesMut);
}
#[repr(u16)]
pub enum RecordType {                    // IANA-assigned values; do not renumber
    A = 1, NS = 2, CNAME = 5, SOA = 6, PTR = 12, MX = 15, TXT = 16, AAAA = 28,
    SRV = 33, OPT = 41, DS = 43, RRSIG = 46, NSEC = 47, DNSKEY = 48, NSEC3 = 50,
    SVCB = 64, HTTPS = 65, CAA = 257, Any = 255,
}

#[repr(u16)]
pub enum Class { IN = 1, CH = 3, HS = 4, None = 254, Any = 255 }
pub enum RData {
    A(std::net::Ipv4Addr), Aaaa(std::net::Ipv6Addr), Cname(Name), Ns(Name), Ptr(Name),
    Mx { preference: u16, exchange: Name },
    Soa { mname: Name, rname: Name, serial: u32, refresh: u32, retry: u32, expire: u32, minimum: u32 },
    Txt(Vec<bytes::Bytes>), Srv { priority: u16, weight: u16, port: u16, target: Name },
    Caa { flags: u8, tag: bytes::Bytes, value: bytes::Bytes },
    Svcb(SvcbRecord), Https(SvcbRecord), Dnskey(DnskeyRecord), Ds(DsRecord),
    Rrsig(RrsigRecord), Nsec(NsecRecord), Nsec3(Nsec3Record), Opt(OptRecord),
    Unknown { rtype: u16, raw: bytes::Bytes },   // RFC 3597
}
pub struct Header { pub id: u16, pub flags: HeaderFlags, pub qd_count: u16, pub an_count: u16, pub ns_count: u16, pub ar_count: u16 }

bitflags::bitflags! {
    pub struct HeaderFlags: u16 {
        const QR=0x8000; const OPCODE_MASK=0x7800; const AA=0x0400; const TC=0x0200;
        const RD=0x0100; const RA=0x0080; const AD=0x0020; const CD=0x0010; const RCODE_MASK=0x000F;
    }
}
pub struct Question { pub name: Name, pub qtype: RecordType, pub qclass: Class }
pub struct ResourceRecord { pub name: Name, pub rtype: RecordType, pub class: Class, pub ttl: u32, pub rdata: RData }
pub struct Message { pub header: Header, pub questions: Vec<Question>, pub answers: Vec<ResourceRecord>,
                      pub authority: Vec<ResourceRecord>, pub additional: Vec<ResourceRecord>, pub edns: Option<OptRecord> }
#[repr(u8)]  pub enum Opcode { Query=0, IQuery=1, Status=2, Notify=4, Update=5 }
#[repr(u16)] pub enum Rcode  { NoError=0, FormErr=1, ServFail=2, NxDomain=3, NotImp=4,
                                Refused=5, YxDomain=6, YxRrSet=7, NxRrSet=8, NotAuth=9,
                                NotZone=10, BadVers=16, BadCookie=23 }

pub struct OptRecord { pub udp_payload_size: u16, pub extended_rcode: u8, pub version: u8,
                        pub flags: OptFlags, pub options: Vec<EdnsOption> }   // flags carries DO bit

pub enum EdnsOption {
    Nsid(bytes::Bytes), Ecs { family: u16, source: u8, scope: u8, address: bytes::Bytes },
    Cookie { client: [u8; 8], server: Option<bytes::Bytes> },
    Padding(usize), ExtendedError { code: u16, text: bytes::Bytes },
    Unknown { code: u16, data: bytes::Bytes },
}
pub struct RrsetEntry { pub key: RrsetKey, pub rrs: Vec<ResourceRecord>, pub inserted_at: std::time::Instant,
                         pub ttl: u32, pub trust: TrustLevel, pub validation: ValidationVerdict }
pub struct NegativeEntry { pub key: RrsetKey, pub rcode: Rcode, pub ttl: u32,
                            pub soa: Option<ResourceRecord>, pub validation: ValidationVerdict }
pub struct RrsetKey { pub name: Name, pub rtype: RecordType, pub class: Class }

#[repr(u8)]
pub enum TrustLevel { AdditionalGlue=1, AdditionalAuthAA=2, Authority=3, AnswerNonAA=4, AnswerAA=5, SecureValidated=6 }
#[repr(u8)]
pub enum ValidationVerdict { Secure, Insecure, Bogus, Indeterminate }   // RFC 4035 §4, I8

pub struct ResolverConfig { pub mode: ResolverMode, pub upstreams: Vec<Upstream>,
                            pub root_hints_path: std::path::PathBuf, pub trust_anchor_path: std::path::PathBuf,
                            pub qname_min: bool, pub aggressive_nsec: bool, pub cookies: bool,
                            pub serve_stale: ServeStalePolicy, pub cache_size_bytes: usize,
                            pub timeout: std::time::Duration, pub retries: u8 }
pub enum ResolverMode { Stub, Recursive }
pub struct Upstream { pub addr: std::net::SocketAddr, pub transport: TransportKind, pub sni: Option<String> }

pub struct Engine { cache: std::sync::Arc<dns_cache::Cache>, resolver: std::sync::Arc<dns_resolver::Resolver>,
                     validator: std::sync::Arc<dns_validator::Validator>, transports: dns_transport::TransportSet,
                     cfg: ResolverConfig }
impl Engine {
    pub fn new(cfg: ResolverConfig) -> Result<Self, EngineError>;
    pub async fn query(&self, q: &Question, opts: QueryOptions) -> Result<Message, EngineError>;
    pub async fn shutdown(&self);
}
pub struct QueryOptions { pub cd: bool, pub do_bit: bool, pub recursion_desired: bool, pub deadline: Option<std::time::Instant> }
#[async_trait::async_trait]
pub trait Transport: Send + Sync + 'static {
    async fn query(&self, msg: &Message, deadline: std::time::Instant) -> Result<Message, TransportError>;
    fn kind(&self) -> TransportKind;
}

pub trait CacheStore: Send + Sync {
    fn insert_rrset(&self, e: RrsetEntry, accept: TrustLevel) -> Result<(), CacheError>;
    fn insert_negative(&self, e: NegativeEntry) -> Result<(), CacheError>;
    fn lookup(&self, k: &RrsetKey) -> Option<std::sync::Arc<RrsetEntry>>;
    fn lookup_negative(&self, k: &RrsetKey) -> Option<std::sync::Arc<NegativeEntry>>;
    fn synthesise_nxdomain(&self, qname: &Name, qtype: RecordType) -> Option<NegativeEntry>;
    fn evict_expired(&self) -> usize;
}

pub trait Validator: Send + Sync {
    fn validate(&self, answer: &[ResourceRecord], authority: &[ResourceRecord], rrsigs: &[RrsigRecord],
                dnskeys: &[DnskeyRecord], ds_chain: &[DsRecord], trust_anchors: &TrustAnchorSet,
                now: std::time::SystemTime) -> ValidationVerdict;
}
