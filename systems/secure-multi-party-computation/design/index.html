Core schema & types (§11)

pub struct Fp(/* Montgomery-form limbs */);          // prime field F_p (P4)
impl Fp {
    pub const MODULUS: U128;
    pub fn add(self, o: Self) -> Self;
    pub fn sub(self, o: Self) -> Self;
    pub fn mul(self, o: Self) -> Self;               // constant-time Montgomery
    pub fn inverse(self) -> CtOption<Self>;          // const-time; None if 0 (I16,P8)
    pub fn pow(self, e: u128) -> Self;
    pub fn random(rng: &mut impl SecureRng) -> Self; // uniform in F_p (P2)
    pub fn to_le_bytes(self) -> Vec<u8>;             // canonical wire form (I12)
    pub fn from_le_bytes(b: &[u8]) -> Result<Self, FieldError>;
}
pub fn batch_inverse(xs: &mut [Fp]);                 // Montgomery batch trick

pub struct Block(pub u128);                          // 128-bit: GF(2^128) label / 128 GF(2) bits
impl Block {
    pub fn xor(self, o: Self) -> Self;               // GF(2) vector add
    pub fn gf128_mul(self, o: Self) -> Self;         // for RO hashing
    pub fn lsb(self) -> u8;                          // signal bit (point-and-permute) (I7)
    pub fn random(rng: &mut impl SecureRng) -> Self;
}
pub trait SecureRng: rand_core::RngCore + rand_core::CryptoRng {}
pub struct ChaChaRng(/* .. */);   impl SecureRng for ChaChaRng {}

pub struct TranscriptRng { seed: [u8;32] }           // tests only; deterministic
impl TranscriptRng { pub fn seeded(seed: [u8;32]) -> Self; }

pub fn prod_rng() -> ChaChaRng;                      // OS-seeded; refuses fixed seed (I11)
pub fn derive_key(master: &[u8], label: &[u8]) -> [u8;32];  // domain-separated KDF
pub struct AdditiveShare { pub party: usize, pub value: Fp }     // or GF(2)
pub struct ShamirShare   { pub x: u64, pub y: Fp }               // (i, p(i))

pub fn share_additive(s: Fp, n: usize, rng: &mut impl SecureRng) -> Vec<AdditiveShare>;
pub fn reconstruct_additive(shares: &[AdditiveShare]) -> Fp;     // Σ (I3)

pub fn share_shamir(s: Fp, t: usize, n: usize, rng: &mut impl SecureRng)
    -> Vec<ShamirShare>;                                         // degree-t poly (I1)
pub fn reconstruct_shamir(shares: &[ShamirShare]) -> Result<Fp, ShareError>;  // Lagrange; needs >t (I2)
pub fn refresh_shamir(shares: &mut [ShamirShare], t: usize, rng: &mut impl SecureRng); // +sharing of 0

// local ops
pub fn add_shares(a: &Share, b: &Share) -> Share;               // free (P1)
pub fn scalar_mul(alpha: Fp, a: &Share) -> Share;               // free (P1)
pub enum ShareError { InsufficientShares{have:usize, need:usize}, DegreeMismatch, DuplicatePoint }
pub trait Channel {
    fn send<T: Message>(&mut self, to: usize, m: &T) -> Result<(), TransportError>;
    fn recv<T: Message>(&mut self, from: usize) -> Result<T, TransportError>;
    fn broadcast<T: Message>(&mut self, m: &T) -> Result<(), TransportError>;  // echo (ADR 19)
    fn stats(&self) -> CommStats;                  // bytes/rounds per peer (P9)
}
pub struct CommStats { pub bytes_sent: u64, pub bytes_recv: u64, pub rounds: u64 }
pub trait Message: serde::Serialize + serde::de::DeserializeOwned {}
pub enum TransportError { Malformed{tag:u16,len:usize}, RoundDesync, Closed, Io(String) }
pub struct BaseOtSender;   pub struct BaseOtReceiver;
pub fn base_ot_send(ch: &mut impl Channel, msgs: &[(Block,Block)], rng: &mut impl SecureRng)
    -> Result<(), OtError>;                         // Naor–Pinkas (§8.4)
pub fn base_ot_recv(ch: &mut impl Channel, choices: &[bool], rng: &mut impl SecureRng)
    -> Result<Vec<Block>, OtError>;                 // learns m_b only (I5)

pub struct IknpSender { /* base-OT seeds */ }       pub struct IknpReceiver { /* .. */ }
pub fn iknp_extend_send(s: &mut IknpSender, ch: &mut impl Channel, m: usize)
    -> Result<Vec<(Block,Block)>, OtError>;         // m OTs (I6)
pub fn iknp_extend_recv(r: &mut IknpReceiver, ch: &mut impl Channel, choices: &[bool])
    -> Result<Vec<Block>, OtError>;
pub fn transpose_bits(matrix: &mut [Block], rows: usize, cols: usize);  // throughput kernel
// R-OT / C-OT variants
pub fn rot(/* .. */); pub fn cot(delta: Block, /* .. */);
pub enum OtError { BaseFailed, ExtensionMismatch, SelfTransfer, Malformed }
pub struct BeaverTriple { pub a: Share, pub b: Share, pub c: Share }   // c=a·b
pub fn gen_triples_ot(n_parties: usize, count: usize, ch: &mut impl Channel,
                      rng: &mut impl SecureRng) -> Result<Vec<BeaverTriple>, TripleError>; // offline
pub struct TripleStore { /* consumable */ }
impl TripleStore {
    pub fn len(&self) -> usize;
    pub fn take(&mut self) -> Result<BeaverTriple, TripleError>;       // exhaustion guard (I14)
}
pub fn verify_triple_open(t: &BeaverTriple, ch: &mut impl Channel) -> Result<bool, TripleError>; // test mode
pub enum TripleError { Exhausted, BadProduct, GenFailed }
pub enum Gate {
    Input { wire: WireId, party: usize },
    Add { out: WireId, a: WireId, b: WireId },
    ScalarMul { out: WireId, a: WireId, k: Fp },
    Mul { out: WireId, a: WireId, b: WireId },        // arithmetic
    Xor { out: WireId, a: WireId, b: WireId },         // boolean
    And { out: WireId, a: WireId, b: WireId },         // boolean
    Not { out: WireId, a: WireId },
    Output { wire: WireId },
}
pub type WireId = u32;
pub enum CircuitKind { Arithmetic, Boolean }
pub struct Circuit { pub kind: CircuitKind, pub n_wires: u32, pub gates: Vec<Gate>,
                     pub inputs: Vec<(usize, WireId)>, pub outputs: Vec<WireId> }
impl Circuit {
    pub fn builder(kind: CircuitKind) -> CircuitBuilder;
    pub fn from_bristol(src: &str) -> Result<Self, CircuitError>;       // Bristol Fashion (ADR 20)
    pub fn topo_order(&self) -> Vec<usize>;
    pub fn mul_depth(&self) -> usize;   // AND-depth / multiplicative depth (round driver)
    pub fn validate(&self) -> Result<(), CircuitError>;
}
pub enum CircuitError { BadWire(WireId), Cycle, KindMismatch, BristolParse{line:usize,msg:String} }
pub struct GarbledGate { pub ciphertexts: [Block; 2] }   // half-gates: 2 per AND
pub struct GarbledCircuit { pub tables: Vec<GarbledGate>, pub out_decode: Vec<u8> }
pub struct Delta(pub Block);                              // free-XOR global; lsb=1 (I7)
pub fn garble(c: &Circuit, delta: Delta, rng: &mut impl SecureRng)
    -> (GarbledCircuit, /*input labels*/ Vec<(Block,Block)>);
pub fn evaluate(gc: &GarbledCircuit, input_labels: &[Block]) -> Vec<Block>; // gate-by-gate (I8)
pub fn decode(out_labels: &[Block], decode: &[u8]) -> Vec<u8>;
// GMW
pub fn gmw_and(a: &Share, b: &Share, store: &mut TripleStore, ch: &mut impl Channel)
    -> Result<Share, BoolError>;                         // consumes triple/OT
pub enum BoolError { GarbleError, DecodeAmbiguous, BadDelta }
pub trait Protocol {
    type Output;
    fn setup(&mut self, parties: &PartySet, cfg: &ProtocolConfig) -> Result<(), ProtoError>;
    fn preprocess(&mut self, rng: &mut impl SecureRng) -> Result<(), ProtoError>;   // offline (P5,I13)
    fn online(&mut self, my_input: &Input) -> Result<(), ProtoError>;               // consumes preproc
    fn output(&mut self) -> Result<Self::Output, ProtoError>;                       // reveals only outputs (I9)
}
pub struct BgwArith; pub struct GmwBool; pub struct YaoGc;     // concrete protocols (ADR 12,13)
pub struct ProtocolConfig { pub kappa: usize, pub stat_sec: usize, pub n: usize, pub t: usize,
                            pub field: FieldConfig }          // (I12)
pub enum ProtoError { ConfigMismatch, PreprocExhausted, RoundDesync, Insecure, Transport(TransportError) }

// applications (mpc-app)
pub fn secure_sum(/* .. */) -> Fp;
pub fn secure_compare(/* .. */) -> bool;   // a>b (DGK/bit-decomp)
pub fn psi(/* .. */) -> Vec<Item>;         // OT-based intersection
pub fn millionaires(/* .. */) -> bool;     // Yao comparator
