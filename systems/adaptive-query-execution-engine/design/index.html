Core schema & types (§11)

// aqe-types — IR + columnar
pub struct Schema { fields: Vec<Field> }                 // immutable; shared Arc; serialized canonical JSON
pub struct Field { name: String, dt: DataType, nullable: bool, deterministic: bool }
pub enum DataType { Int32, Int64, Float64, Utf8, Boolean, Date32, Dictionary(Box<DataType>) }
pub struct RecordBatch { schema: Arc<Schema>, cols: Vec<ArrayRef>, rows: usize } // Arrow-backed; immutable per batch; spilled via IPC
pub type ArrayRef = arrow_array::ArrayRef;               // dictionary passes through undecoded

pub enum PhysicalOp {                                    // running plan node; owned by StageGraph; serialized JSON
  Scan { table: TableId, cols: Vec<ColId>, pred: Option<Expr> },
  Filter { input: OpId, pred: Expr },
  Project { input: OpId, exprs: Vec<Expr> },
  HashJoin { build: OpId, probe: OpId, on: Vec<(ColId,ColId)>, jt: JoinType },
  SortMergeJoin { left: OpId, right: OpId, on: Vec<(ColId,ColId)>, jt: JoinType },
  BroadcastJoin { build: OpId, probe: OpId, on: Vec<(ColId,ColId)>, jt: JoinType },
  HashAggregate { input: OpId, keys: Vec<ColId>, aggs: Vec<AggExpr>, mode: AggMode },
  SortAggregate { input: OpId, keys: Vec<ColId>, aggs: Vec<AggExpr> },
  Exchange { input: OpId, part: Partitioning },
  Sort { input: OpId, keys: Vec<SortKey> },
  Limit { input: OpId, n: u64 },
}
pub enum AggMode { Partial, Final }                      // two-stage agg
pub enum Partitioning { Hash(Vec<ColId>, u32), RoundRobin(u32), Broadcast, SinglePartition }
pub enum JoinType { Inner, Left, Right, Full, Semi, Anti }
pub struct Expr { /* tree */ deterministic: bool }       // pre-bound to ColId/Lit; flag gates reopt (I14)

pub struct PlanVersion(pub u32);                         // monotone (I7); bumped per applied reopt
pub struct PlanIrVersion { major: u16, minor: u16 }      // SemVer of IR (ADR-030)

// aqe-stage
pub struct StageId(pub u32);                             // monotone, total-ordered with PlanVersion (I13)
pub enum StageState { Pending, Scheduled, Running, Materialized, Consumed, Failed, Recomputing } // I3
pub struct Stage { id: StageId, ops: Vec<OpId>, breaker: BreakerKind, inputs: Vec<StageId>, out: Option<MaterializedRef> }
pub struct StageGraph { stages: Vec<Stage>, edges: Vec<(StageId,StageId)>, version: PlanVersion }
                                                         // single-writer (controller); workers read snapshots

// aqe-stats / aqe-sketch  (sealed at Materialized — I4)
pub struct RuntimeStats { stage: StageId, rows: u64, bytes: u64, per_col: Vec<ColStats>,
                          partition_sizes: Vec<u64>, heavy_hitters: CountMinTopK, skew: SkewReport, sealed: bool }
pub struct ColStats { col: ColId, ndv: Hll, nulls: u64, min: Option<Scalar>, max: Option<Scalar> }
pub struct Hll { m: usize, registers: Vec<u8> }          // NDV; merge-able; rel err 1.04/√m
pub struct CountMin { d: usize, w: usize, table: Vec<u32> } // freq; err εN, prob 1−δ
pub struct CountMinTopK { cms: CountMin, top: BinaryHeap<(u64, Scalar)> }
pub struct SkewReport { max_partition: u64, median_partition: u64, ratio: f64, hot_keys: Vec<(Scalar,u64)> }

// aqe-mem
pub struct MemoryBudget { total: u64, used: AtomicU64 }  // global; thread-safe accounting
pub struct Reservation { op: OpId, granted: u64 }        // RAII; drop returns to budget (I6)
pub enum Pressure { Ok, ShouldSpill, MustBackpressure }
// aqe-spool / aqe-shuffle
pub struct SpillSegment { path: PathBuf, rows: u64, bytes: u64, crc32: u32 } // round-trip exact (I11)
pub struct SpillManifest { segments: Vec<SpillSegment> }
pub struct ShuffleFile { stage: StageId, partition: u32, path: PathBuf, bytes: u64, crc32: u32 }
pub struct ShuffleRegistry { files: HashMap<(StageId,u32), ShuffleFile> }     // completeness (I12)

// aqe-adaptive
pub enum ReoptRule { Dpp, SwitchJoin, SkewSplit, CoalescePartitions, AggMode } // ordered (§8.5)
pub struct ReoptEvent { version_from: PlanVersion, version_to: PlanVersion, stage: StageId,
                        rule: ReoptRule, reason: String, accepted: bool }      // append-only (I13)
pub struct DecisionLattice { /* per-stage monotone state */ }                  // enforces I2

// aqe-exec — operator contract (push model)
pub trait Operator: Send {
  fn open(&mut self, ctx: &ExecCtx) -> Result<()>;
  fn push(&mut self, batch: RecordBatch) -> Result<OpStatus>; // OpStatus::{NeedMore, Backpressure, Done}
  fn finish(&mut self) -> Result<Vec<RecordBatch>>;           // flush (breakers emit here)
}
pub enum OpStatus { NeedMore, Backpressure, Done }

// errors (thiserror; no panics on data path — ADR-027)
pub enum AqeError { Plan(PlanError), Mem(MemError), Spill(SpillError), Shuffle(ShuffleError),
                    Reopt(ReoptError), Exec(ExecError) }
pub enum PlanError { UnresolvedColumn, EmptyPartitioning, SchemaMismatch, BadJoinKeys, NondetInKey }
