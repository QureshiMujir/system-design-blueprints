Core schema & types (§11)

struct Svn(u32);                  // monotone; Ord; no public decrement
struct KeyId([u8; 20]);           // SKI-style 160-bit signer id
struct PcrIndex(u8);              // 0..=23
enum HashAlg { Sha256, Sha384 }   // maps hash_alg field
enum SigAlg { Ed25519, EcdsaP256, RsaPss3072 }
struct Digest { alg: HashAlg, bytes: [u8; 48] /* len = alg */ }
struct ImageId(u8);               // 1=SPL..5=RECOVERY
enum VerifyError {                 // closed, exhaustive (Principle 11)
  Malformed(MalformKind), NonCanonical, ChainUnresolved, PathInvalid(PathErr),
  Expired, NotYetValid, EkuMissing, ConstraintViolation, Revoked(RevokeKind),
  RevocationUnknown, Rollback{img:Svn,min:Svn}, DigestMismatch, BadSignature,
  ArenaExhausted, Timeout, PolicyDenied }
enum Verdict { Accept, Reject(VerifyError) }   // never silently anything else
struct ManifestHeader { magic:[u8;4], fmt_ver:u8, hdr_ver:u8, flags:Flags,
  sig_alg:SigAlg, hash_alg:HashAlg, svn:Svn, image_id:ImageId,
  payload_len:u64, payload_digest:Digest, key_id:KeyId }
struct Manifest<'a> { header:ManifestHeader, chain_ref:ChainRef<'a>,
  tlvs:TlvSlice<'a>, signature:&'a [u8], signed_region:&'a [u8] }
fn parse<'a>(buf:&'a [u8], arena:&mut Arena) -> Result<Manifest<'a>, VerifyError>;
struct DerReader<'a> { /* cursor over &'a [u8] */ }
fn read_tlv(&mut self) -> Result<DerTlv<'a>, VerifyError>; // rejects non-minimal length, indefinite-len
trait Verifier { fn verify(&self, msg:&[u8], sig:&[u8], key:&PublicKey) -> bool; } // constant-time
trait Hasher  { fn update(&mut self, b:&[u8]); fn finish(self) -> Digest; }
trait Clock   { fn now_unix(&self) -> u64; }                 // injected (ADR-028)
trait Rng     { fn fill(&mut self, out:&mut [u8]); }         // injected (ADR-029); not used in verify
enum PublicKey { Ed25519([u8;32]), EcdsaP256(EcPoint), RsaPss{ n:&[u8], e:u32 } }
struct Cert<'a> { tbs:&'a [u8], spki:PublicKey, issuer:Name<'a>, subject:Name<'a>,
  not_before:u64, not_after:u64, bc:BasicConstraints, ku:KeyUsage, eku:EkuSet,
  name_constraints:Option<NameConstraints<'a>>, serial:&'a [u8], sig:&'a [u8], sig_alg:SigAlg }
struct PathState { working_key:PublicKey, max_path_len:Option<u8>,
  permitted:SubtreeSet, excluded:SubtreeSet, valid_eku:EkuSet }
fn validate(chain:&[Cert], anchor:&TrustAnchor, clock:&dyn Clock,
            rev:&dyn RevocationSource) -> Result<KeyId, VerifyError>;
struct TrustAnchor { pk:PublicKey, pk_hash:[u8;32], cross_sign_allow:&[PublicKey] }
trait RevocationSource { fn status(&self, cert:&Cert, clock:&dyn Clock) -> RevStatus; }
enum RevStatus { Good, Revoked, Unknown }
struct Dbx<'a> { revoked_keyids:&'a [KeyId], revoked_hashes:&'a [Digest] } // offline denylist
struct Crl<'a> { issuer:Name<'a>, this_update:u64, next_update:u64, revoked:&'a [(Serial,u64)], sig:&'a [u8] }
struct OcspResp<'a> { cert:Serial, status:RevStatus, produced_at:u64, this_update:u64, next_update:u64, sig:&'a [u8] }
trait Otp  { fn pk_hash(&self) -> [u8;32]; fn min_svn_floor(&self) -> Svn; fn is_locked(&self) -> bool; }
trait NvCounter { fn read(&self) -> Svn; fn commit_ge(&mut self, v:Svn) -> Result<(), NvError>; } // monotone CAS
fn effective_min(otp:&dyn Otp, nv:&dyn NvCounter) -> Svn; // max(floor, counter)
struct PcrBank { pcrs:[Digest; 24], alg:HashAlg }
fn extend(&mut self, idx:PcrIndex, m:&Digest);            // PCR[i] = H(PCR[i] ‖ m)
struct EventLog { entries:Vec<Event> }                     // append-only; replayable
fn quote(&self, sel:PcrSelection, nonce:&[u8], ak:&AttestKey) -> Quote;     // COSE_Sign1-shaped
fn seal(&self, secret:&[u8], policy:PcrPolicy) -> SealedBlob;               // bind to future PCRs
fn unseal(&self, blob:&SealedBlob, now:&PcrBank) -> Result<Vec<u8>, SealErr>;
enum AbState { TryA, TryB, ActiveA, ActiveB, Recovery, BrickGuard }
struct Slot { id:char, tries:u8, priority:u8, svn:Svn }
fn select(&self) -> Option<&Slot>;            // highest priority with tries>0
fn on_verify(&mut self, slot:char, v:Verdict);// Reject → tries=0; Accept → boot+WDT
fn mark_good(&mut self, slot:char, nv:&mut dyn NvCounter); // commits svn (I12)

// verifier entry point (the TCB API)
fn verify_stage(buf:&[u8], payload:&mut dyn Read, anchor:&TrustAnchor,
  rev:&dyn RevocationSource, otp:&dyn Otp, nv:&dyn NvCounter,
  clock:&dyn Clock, arena:&mut Arena) -> Verdict;
struct Arena<'a> { buf:&'a mut [u8], top:usize }
fn alloc(&mut self, n:usize, align:usize) -> Result<&mut [u8], VerifyError>; // ArenaExhausted on overflow
