Core schema & types (§11)

// ir — interned identifiers (normative; ADR-04/05)
pub struct ExprId(u32);          // hash-consed expression
pub struct PlanId(u32);          // hash-consed logical plan node (DAG)
pub struct GroupId(u32);         // memo equivalence group
pub struct GroupExprId(u32);     // a group expression within a group
pub struct TableId(u32); pub struct ColId(u32); pub struct IndexId(u32);
// ir/expr — expression model (normative)
pub enum Expr {
    Col(ColRef), Lit(Value),
    Call(ScalarFn, Box<[ExprId]>),
    Case(Box<[(ExprId, ExprId)]>, Option<ExprId>),
    Agg(AggFn, ExprId, /*distinct*/ bool),
    Subquery(SubqueryKind, PlanId),
    Window(WindowFn, /*part*/ Box<[ExprId]>, /*order*/ Box<[OrderKey]>, FrameSpec),
}
pub struct ScalarFnMeta { pub deterministic: bool, pub null_rejecting: bool, pub monotonic: Option<Dir> }
// hash-consing: Interner<Expr> → ExprId; structural equality is ExprId equality (O(1))
// ir/logical — logical plan + derived properties (normative)
pub enum LogicalOp {
    Scan { table: TableId, cols: ColSet, pred: Option<ExprId> },
    Filter { input: PlanId, pred: ExprId },
    Project { input: PlanId, exprs: Box<[(ExprId, Alias)]> },
    Join { left: PlanId, right: PlanId, on: ExprId, kind: JoinKind },
    Aggregate { input: PlanId, group: Box<[ExprId]>, aggs: Box<[AggItem]> },
    Sort { input: PlanId, keys: Box<[OrderKey]> },
    Limit { input: PlanId, n: u64, offset: u64 },
    SetOp { left: PlanId, right: PlanId, op: SetOpKind, all: bool },
    Window { input: PlanId, fns: Box<[WindowSpec]> },
    Values { rows: Box<[Box<[ExprId]>]> },
}
pub struct LogicalProps {              // §8.4 — derived per node/group (I11)
    pub schema: Schema, pub card: f64, pub keys: Vec<KeySet>,
    pub fds: Vec<FuncDep>, pub not_null: ColSet, pub max_card: Option<u64>,
}
pub fn derive_props(op: &LogicalOp, children: &[&LogicalProps]) -> LogicalProps; // I11
// ir/physical — physical plan (normative)
pub enum PhysicalOp {
    SeqScan{table:TableId, pred:Option<ExprId>}, IndexScan{table:TableId, index:IndexId, bounds:Bounds, pred:Option<ExprId>},
    IndexOnlyScan{table:TableId, index:IndexId, cols:ColSet},
    PhysFilter{input:GroupExprId, pred:ExprId}, PhysProject{input:GroupExprId, exprs:Box<[(ExprId,Alias)]>},
    HashJoin{build:GroupExprId, probe:GroupExprId, keys:JoinKeys, kind:JoinKind, residual:Option<ExprId>},
    MergeJoin{left:GroupExprId, right:GroupExprId, keys:JoinKeys, kind:JoinKind},
    NestedLoop{outer:GroupExprId, inner:GroupExprId, kind:JoinKind, pred:Option<ExprId>},
    HashAgg{..}, SortAgg{..}, PhysSort{..}, PhysLimit{..}, PhysSetOp{..}, PhysWindow{..},
    Exchange{input:GroupExprId, dist:Distribution},   // distributed stub (#316)
}
pub struct PhysAnnot { pub card: f64, pub cost: Cost, pub scalar_cost: f64,
                       pub provided: PhysicalProps, pub mem_bytes: u64 }
// memo — the Cascades memo (normative, P2/I2/I3/I10)
pub struct Memo { groups: Vec<Group>, expr_index: HashMap<Fingerprint, GroupExprId> }
pub struct Group {
    id: GroupId,
    logical_exprs: Vec<GroupExprId>,        // logically equivalent expressions
    physical_exprs: Vec<GroupExprId>,
    props: LogicalProps,                    // shared across exprs (I11)
    winners: HashMap<RequiredProps, Winner>,// per-required-property best (I10)
    explored: bool,
}
pub struct GroupExpr { id: GroupExprId, op: OpRef, children: SmallVec<[GroupId; 4]>, applied_rules: RuleMask }
pub struct Winner { expr: GroupExprId, cost: f64, satisfied: PhysicalProps }
impl Memo {
    fn insert(&mut self, op: OpRef, children: &[GroupId]) -> (GroupId, GroupExprId); // dedup by Fingerprint (I3)
    fn merge_groups(&mut self, a: GroupId, b: GroupId);                              // duplicate detection (I2)
    fn group_props(&self, g: GroupId) -> &LogicalProps;
}
// search — Cascades task scheduler (normative, P10/I7/I8)
pub enum Task {
    OptimizeGroup { group: GroupId, req: RequiredProps, bound: f64 },   // bound = branch-and-bound upper bound
    OptimizeExpr  { expr: GroupExprId, req: RequiredProps, bound: f64 },
    ExploreGroup  { group: GroupId },
    ApplyRule     { expr: GroupExprId, rule: RuleId, req: RequiredProps, bound: f64 },
    OptimizeInputs{ expr: GroupExprId, req: RequiredProps, bound: f64 },
    EnforceProps  { group: GroupId, req: RequiredProps, bound: f64 },
}
pub struct Optimizer { memo: Memo, rules: RuleSet, est: Box<dyn CardinalityEstimator>, cost: CostModel, budget: Budget }
impl Optimizer {
    fn optimize(&mut self, root: PlanId, req: RequiredProps) -> Result<PhysicalPlan, OptError>; // I1/I5/I7
}
pub struct Budget { max_steps: u64, max_bytes: u64, max_wall: Duration }   // I7/P8
// rules — rule contract (normative, P4)
pub enum RuleKind { Transform, Implement }
pub trait Rule {
    fn kind(&self) -> RuleKind;
    fn pattern(&self) -> &Pattern;                              // what group-expr shape it matches
    fn apply(&self, m: &Match, memo: &mut Memo) -> Vec<GroupExprId>;  // new exprs (same group); must preserve I1
    fn promise(&self) -> u16;                                   // firing priority (deterministic order, P9)
}
// estimate — the pluggable estimator (normative, P6/P7, ADR-18, §10.9 seam)
pub trait CardinalityEstimator {
    fn scan_card(&self, t: TableId, pred: Option<ExprId>) -> Estimate;          // base + filter
    fn join_card(&self, l: &Estimate, r: &Estimate, on: ExprId) -> Estimate;    // I6
    fn agg_card(&self, input: &Estimate, group: &[ExprId]) -> Estimate;
    fn selectivity(&self, pred: ExprId, props: &LogicalProps) -> Selectivity;   // ∈[0,1] (I12)
}
pub struct Estimate { pub rows: f64, pub confidence: Confidence }   // rows ≥0 finite (I6)
pub struct Selectivity(f64);                                        // clamped [0,1] (I12)
// cost — cost model (normative, P6, ADR-10)
pub struct Cost { pub cpu: f64, pub io: f64, pub mem: f64, pub net: f64 }   // component vector
pub struct CostModel { pub calib: Calibration }                            // weights → scalar
impl CostModel {
    fn op_cost(&self, op: &PhysicalOp, in_cards: &[f64], props: &PhysicalProps) -> Cost;  // per-operator
    fn scalar(&self, c: &Cost) -> f64;                                                     // weighted sum
    fn lower_bound(&self, group: GroupId, req: &RequiredProps) -> f64;                      // admissible (I8)
}
pub struct PhaseDeferredError { pub reference: &'static str }   // PhaseDeferred("Deferred Features Register")
