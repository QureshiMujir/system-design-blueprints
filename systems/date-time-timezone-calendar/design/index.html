Core schema & types (§11)

// absolute time
struct Instant { secs: i64, nanos: u32, scale: TimeScale }   // nanos ∈ [0,1e9)
enum  TimeScale { Utc, Tai, Unix }                            // leap-second semantics differ

// civil (wall-clock) time — no offset
struct CivilDate { y: i32, m: u8, d: u8 }                     // proleptic Gregorian
struct CivilTime { h: u8, mi: u8, s: u8, ns: u32 }           // s may be 60 on a leap second
struct CivilDateTime { date: CivilDate, time: CivilTime }

struct Offset(i32);                                           // seconds east of UTC
struct ZonedDateTime { civil: CivilDateTime, offset: Offset, zone: ZoneRef }

// arithmetic
struct Duration { secs: i64, nanos: u32 }                    // exact, scale-free
struct Period   { years: i32, months: i32, days: i32 }       // calendrical, variable length

// time zone
struct TimeZone { transitions: Vec<Transition>, types: Vec<TtInfo>,
                  posix: Option<PosixTz>, leap: Vec<LeapRecord> }
struct Transition { at: i64 /* unix */, type_idx: u16 }
struct TtInfo { utoff: i32, is_dst: bool, abbr: Abbrev }
struct PosixTz { std: Abbrev, std_off: i32, dst: Option<(Abbrev, i32)>,
                 start: TzRule, end: TzRule }                 // future-transition rule
enum  TzRule { Julian1(u16), Julian0(u16), MonthWeekDay{ m:u8, w:u8, d:u8 }, time: i32 }

enum AmbiguityPolicy { Compatible, Earliest, Latest, Reject } // gap/overlap resolution

// recurrence (RFC 5545)
struct RRule { freq: Freq, interval: u32, count: Option<u32>, until: Option<Instant>,
               by_second: Vec<u8>, by_minute: Vec<u8>, by_hour: Vec<u8>,
               by_day: Vec<WeekdayNum>, by_monthday: Vec<i8>, by_yearday: Vec<i16>,
               by_weekno: Vec<i8>, by_month: Vec<u8>, by_setpos: Vec<i16>, wkst: Weekday }
enum  Freq { Secondly, Minutely, Hourly, Daily, Weekly, Monthly, Yearly }
struct RRuleIter { /* lazy, bounded by count/until + MAX_ITER */ }

enum Weekday { Mon, Tue, Wed, Thu, Fri, Sat, Sun }
enum Month   { Jan, /* … */ Dec }

// errors
enum DtError { ParseError, RangeError, NonexistentTime(CivilDateTime),
               AmbiguousTime(CivilDateTime, [Offset;2]), BadTzif, BadRRule, RecurrenceUnbounded }
