Core schema & types (§11)

enum CellValue {
    Blank,
    Number(f64),            // canonical numeric; dates are serials
    Bool(bool),
    Text(StringId),         // interned (ADR-013)
    Error(ErrorKind),
}
enum ErrorKind { Null, DivZero, Value, Ref, Name, Num, NA, Spill, Calc, Circular }
fn precedence(&self) -> u8;     // 1..=10 per §8.7
struct CellId { sheet: SheetId, row: u32, col: u32 }     // 0-based internal
fn a1_to_id(s: &str, ctx: SheetId) -> Result<CellId, RefError>;
fn id_to_a1(id: CellId) -> String;
fn col_to_index(letters: &str) -> u32;   // bijective base-26
fn index_to_col(i: u32) -> String;
struct Reference {
    sheet: SheetTarget,     // Current | Id(SheetId) | Range(SheetId,SheetId) | Name(NameId)
    col: Axis, row: Axis,   // Axis = { Abs(u32) | Rel(i32) }  (Rel stored as offset for R1C1/relative)
}
struct RangeRef { start: Reference, end: Reference }
struct FormulaAst { root: AstId, arena: Arc<AstArena> }   // arena-allocated nodes, shared
enum Ast { Lit(CellValue), Ref(Reference), Range(RangeRef), Name(NameId),
           Unary(UnOp, AstId), Binary(BinOp, AstId, AstId),
           Call(FuncId, SmallVec<AstId>), Array(Vec<Vec<AstId>>),
           ImplicitIsect(AstId), Error(ErrorKind) }
struct DependencyGraph {
    precedents: SecondaryMap<CellId, SmallVec<DepEdge>>,   // what a formula reads
    dependents: SecondaryMap<NodeRef, SmallVec<CellId>>,   // who reads a cell/range
    range_nodes: HashMap<RangeRef, RangeNodeId>,           // range-node compression (ADR-006)
}
enum DepEdge { Cell(CellId), Range(RangeNodeId), Name(NameId) }
fn add_formula(&mut self, c: CellId, ast: &FormulaAst);
fn remove_formula(&mut self, c: CellId);
fn dependents_of(&self, c: CellId) -> impl Iterator<Item=CellId>;   // for dirty propagation
fn topo_order(&self, dirty: &DirtySet) -> Result<Vec<CellId>, Cycle>;  // Kahn
fn sccs(&self, sub: &[CellId]) -> Vec<Scc>;                          // Tarjan
struct RecalcEngine { mode: CalcMode, max_iter: u32, epsilon: f64, pool: Option<ThreadPool> }
enum CalcMode { Automatic, Manual, Iterative }
fn recalc_full(&self, wb: &mut Workbook) -> RecalcReport;
fn recalc_dirty(&self, wb: &mut Workbook, dirty: DirtySet) -> RecalcReport;
struct RecalcReport { cells_evaluated: u64, dirty_size: u64, scc_count: u32,
                      iterations: u32, duration: Duration, errors: u64 }
struct FunctionRegistry { table: HashMap<FuncId, FnEntry>, volatile: BitSet }
struct FnEntry { id: FuncId, name: &'static str, arity: Arity,
                 coercions: ArgCoercions, volatile: bool, array_aware: bool,
                 eval: fn(&mut EvalContext, &[Value]) -> Value }
fn lookup(&self, name: &str) -> Option<FuncId>;
fn is_volatile(&self, f: FuncId) -> bool;
fn register(&mut self, entry: FnEntry);     // host custom functions
struct EvalContext<'a> {
    snapshot: &'a Snapshot,     // immutable read view (Invariant I9)
    current: CellId,            // for relative refs / implicit intersection anchor
    registry: &'a FunctionRegistry,
    depth: u16,                 // recursion guard (Invariant I13)
    volatile_seen: &'a Cell<bool>,
}
fn read(&self, r: &Reference) -> Value;
fn read_range(&self, r: &RangeRef) -> ArrayView;
struct SpillRange { anchor: CellId, rows: u32, cols: u32 }
enum SpillState { Anchor(SpillRange) | Spilled { anchor: CellId } | Blocked }
fn reserve(&mut self, sr: SpillRange) -> Result<(), SpillCollision>;  // Invariant I7
struct Workbook { sheets: SlotMap<SheetId, Sheet>, names: NameTable,
                  graph: DependencyGraph, registry: FunctionRegistry,
                  interner: Interner, oplog: UndoLog, version: FormatVersion }
struct Sheet { id: SheetId, name: String, store: CellStore, dims: GridDims }
fn set_cell(&mut self, id: CellId, input: CellInput) -> Result<RecalcReport, EditError>;
fn get_value(&self, id: CellId) -> CellValue;
fn snapshot(&self) -> Arc<Snapshot>;          // structural-sharing COW (Invariant I9)
fn insert_rows(&mut self, sheet: SheetId, at: u32, n: u32) -> RecalcReport;  // → fixup + recalc
struct UndoLog { ops: Vec<Op>, cursor: usize, wal: Option<WalWriter> }
enum Op { SetCell{id, before: CellValue, after: CellInput},
          InsertRows{sheet, at, n}, DeleteRows{..}, InsertCols{..}, DeleteCols{..},
          DefineName{..}, Move{..}, Sort{..} }
fn push(&mut self, op: Op);     // also append to WAL if durable
fn undo(&mut self, wb: &mut Workbook); fn redo(&mut self, wb: &mut Workbook);
