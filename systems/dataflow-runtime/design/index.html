Core schema & types (§11)

trait PartialOrder { fn less_equal(&self, other: &Self) -> bool; }
trait Lattice: PartialOrder { fn join(&self, &Self) -> Self; fn meet(&self, &Self) -> Self; fn minimum() -> Self; }
trait Timestamp: PartialOrder + Lattice + Clone + Ord + Hash + 'static {
    type Summary: PathSummary<Self>;   // how this timestamp transforms along a path
}
trait PathSummary<T> { fn results_in(&self, src: &T) -> Option<T>;   // None = unreachable
                       fn followed_by(&self, &Self) -> Option<Self>; }
struct Product<O, I> { outer: O, inner: I }   // nested-scope timestamp (loop coord = inner)
type RootTime = u64;                          // epoch / event-time at root scope
struct Antichain<T> { elements: SmallVec<[T; 4]> }      // minimal incomparable set
  fn insert(&mut self, t: T) -> bool;                   // true if it changed the antichain
  fn less_equal(&self, t: &T) -> bool;                  // some element ⪯ t
struct Frontier<T> { inner: Antichain<T> }              // times that MAY still appear
struct MutableFrontier<T> { ..; changes: ChangeBatch<T> }
struct ChangeBatch<T> { updates: Vec<(T, i64)> }        // compacted +/- deltas
struct Capability<T> { time: T, port: PortId, tracker: Weak<TrackerInner> }
  fn time(&self)->&T; fn downgrade(&mut self, t:&T); fn delayed(&self, t:&T)->Capability<T>;
  // Drop impl: decrements pointstamp count (consumed += 1)
struct Pointstamp<T> { location: LocationId, time: T }
struct ProgressUpdate<T> { point: Pointstamp<T>, delta: i64 }     // +produced / -consumed
struct ProgressTracker<T> {
    counts: HashMap<Pointstamp<T>, i64>,
    reachability: Reachability,            // precomputed path summaries
    per_location_frontier: Vec<MutableFrontier<T>>,
}
  fn update(&mut self, ProgressUpdate<T>);                 // apply a delta
  fn propagate(&mut self) -> Vec<(LocationId, FrontierChange<T>)>; // close pointstamps → frontier changes
trait Operator {
    fn schedule(&mut self, ctx: &mut SchedulerCtx) -> Activity;   // Active | Idle | Blocked
    fn name(&self) -> &str;
}
enum Activity { Active, Idle, Blocked }
struct Stream<S: Scope, D> { scope: S, source: (OperatorId, PortId), _pd: PhantomData<D> }
  fn map<D2>(self, f) -> Stream<S, D2>;        // and filter/flat_map/inspect/exchange/...
struct InputHandle<T, D>  { /* pull batches + input frontier */ }
struct OutputHandle<T, D> { /* session(cap) -> give(t, d) */ }
struct OperatorInfo { id: OperatorId, scope: ScopeId, inputs: Vec<PortId>, outputs: Vec<PortId> }
struct Message<D> { time: T, data: Vec<D>, from: WorkerId }   // T via the edge's scope
trait Push<T> { fn push(&mut self, msg: Option<Message<T>>); } // None = flush/close
trait Pull<T> { fn pull(&mut self) -> Option<&mut Message<T>>; }
struct Bounded<T> { ring: Ring<Message<T>>, credit: Credit }   // backpressure
struct Pact { partition: fn(&dyn Any) -> u64, kind: PactKind } // Exchange routing
enum PactKind { Pipeline, Exchange }
trait StateBackend { type Handle<K,V>; fn keyed<K,V>(&mut self, name:&str) -> Handle<K,V>; }
struct KeyedState<K, V> { partitions: Vec<HashMap<K, V>>, owner: WorkerId }
  fn get(&self, &K)->Option<&V>; fn insert(&mut self, K, V); fn remove(&mut self,&K)->Option<V>;
  fn iter_for_flush(&self) -> impl Iterator<Item=(&K,&V)>;
  fn snapshot(&self, w: &mut SnapshotWriter); fn restore(r: &mut SnapshotReader)->Self;
type Epoch = u64;
struct CheckpointBarrier { epoch: Epoch, ts: Instant }
enum AlignState { Normal, Aligning{seen: BitSet, epoch: Epoch}, Snapshotting{epoch: Epoch} }
struct SnapshotId { epoch: Epoch }
struct SnapshotMeta { epoch: Epoch, source_offsets: Vec<(SourceId, Offset)>, ops: Vec<OperatorId>, crc: u32, version: u16 }
struct Coordinator { in_flight: HashMap<Epoch, AckSet>, last_complete: Option<Epoch> }
  fn trigger(&mut self) -> Epoch; fn ack(&mut self, OperatorId, Epoch); fn complete(&mut self, Epoch);
struct WorkerId(u32);
struct Worker { id: WorkerId, peers: u32, graph: RuntimeGraph, tracker: ProgressTracker<T>,
                activation: ActivationQueue, channels: ChannelEndpoints, coord: Coordinator }
  fn execute(&mut self) -> ;            // run scheduling loop until quiescent/terminated
  fn step(&mut self) -> bool;           // one scheduling iteration (testable)
struct ActivationQueue { ready: VecDeque<OperatorId>, queued: BitSet }
  fn activate(&mut self, OperatorId); fn next(&mut self) -> Option<OperatorId>;
enum GraphError { UnboundedCycle(ScopeId), TypeMismatch, DuplicatePort, EmptyScope }      // build time
enum RuntimeFault { ProgressRegression, NegativeCount, IllegalEmit, ForeignKeyAccess }    // correctness (abort)
enum CheckpointError { AlignTimeout(Epoch), StoreIo(io::Error), CorruptSnapshot(SnapshotId) }
enum ChannelError { Closed, WouldBlock }                                                  // expected control flow
