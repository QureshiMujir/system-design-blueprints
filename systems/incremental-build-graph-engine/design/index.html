Core schema & types (§11)

ActionKey(Digest)
  = blake3( command_bytes || NUL
         || concat_sorted(input_path → input_digest: path_bytes || NUL || digest_bytes)
         || concat_sorted(env k → v: k_bytes || NUL || v_bytes || NUL)
         || tool_digest_bytes )
enum NodeKind {
    Source { path: PathBuf },                          // leaf: file on disk; no deps except via Glob
    Action(ActionId),                                  // internal: drives execution
    Glob   { pattern: String, base: PathBuf },         // expands to set of Source nodes
    Group,                                             // virtual aggregator; no action; pure fan-in
}

struct BuildNode {
    id:               NodeId,
    key:              NodeKey,
    kind:             NodeKind,
    deps:             Vec<NodeId>,          // static declared dependencies (edges)
    declared_outputs: Vec<PathBuf>,         // output paths this node is responsible for
}
enum EdgeKind { Static, Dynamic }

struct Edge {
    from: NodeId,
    to:   NodeId,
    kind: EdgeKind,
}
struct DynamicDep {
    from:            NodeId,
    to:              NodeId,
    discovered_at:   Revision,
}
struct Cmd {
    program:  PathBuf,
    args:     Vec<OsString>,
}

type EnvMap = BTreeMap<String, String>;   // sorted for deterministic ActionKey hashing

struct Action {
    id:      ActionId,
    command: Cmd,
    inputs:  Vec<NodeId>,                 // must be complete; undeclared = sandbox violation
    outputs: Vec<PathBuf>,                // declared outputs; capture enforced post-exec
    env:     EnvMap,
    key:     ActionKey,                   // computed eagerly at graph-construction time
}
enum NodeStatus { Clean, Dirty, Running, Failed }

struct NodeState {
    last_verified: Revision,           // Revision at which node was confirmed up-to-date
    last_changed:  Revision,           // Revision at which output_digest last differed
    output_digest: Option<Digest>,     // blake3 of declared outputs bundle; None if never built
    status:        NodeStatus,
}
struct VerifiedSet {
    inner: DashMap<NodeId, NodeState>,
}
struct MemoTable {
    states:   VerifiedSet,
    in_flight: DashMap<NodeId, Arc<tokio::sync::watch::Receiver<NodeStatus>>>,
}
struct BuildGraph {
    nodes:    Vec<BuildNode>,                          // dense NodeId → BuildNode
    actions:  Vec<Action>,                             // dense ActionId → Action
    edges:    Vec<Edge>,                               // static + dynamic (append-only post-freeze)
    node_idx: HashMap<NodeKey, NodeId>,                // stable-key lookup
    revdep:   Vec<Vec<NodeId>>,                        // reverse-dep index: NodeId → dependents
    topo:     Vec<NodeId>,                             // topological order (Kahn's, recomputed on dynamic-dep add)
    frozen:   bool,
}

impl BuildGraph {
    fn add_node(&mut self, key: NodeKey, kind: NodeKind) -> NodeId;
    fn add_edge(&mut self, from: NodeId, to: NodeId) -> Result<(), GraphError>;
    fn freeze(&mut self) -> Result<(), GraphError>;    // topo sort + cycle check; sets frozen = true
    fn transitive_deps(&self, root: NodeId) -> impl Iterator<Item = NodeId>;
    fn transitive_rdeps(&self, root: NodeId) -> impl Iterator<Item = NodeId>;
    fn affected_set(&self, changed: &[NodeId]) -> BTreeSet<NodeId>;
}
struct ExecMeta {
    duration_ms:  u64,
    exit_code:    i32,
    stdout_digest: Option<Digest>,
    stderr_digest: Option<Digest>,
    host:         String,
    timestamp_ms: u64,
}

struct CacheEntry {
    key:     ActionKey,
    outputs: Vec<(PathBuf, Digest)>,   // declared output path → CAS digest
    meta:    ExecMeta,
}

struct CasObject {
    digest: Digest,
    size:   u64,
    data:   Bytes,                     // lazily loaded; may be memory-mapped
}
trait ActionCache: Send + Sync {
    fn get(&self, key: &ActionKey) -> Result<Option<CacheEntry>, CacheError>;
    fn put(&self, entry: CacheEntry) -> Result<(), CacheError>;
    fn evict(&self, key: &ActionKey) -> Result<bool, CacheError>;
    fn gc(&self, live_keys: &BTreeSet<ActionKey>) -> Result<GcStats, CacheError>;
}
trait CasStore: Send + Sync {
    fn get(&self, digest: &Digest) -> Result<Option<CasObject>, CacheError>;
    fn put(&self, data: Bytes) -> Result<Digest, CacheError>;      // returns computed digest
    fn contains(&self, digest: &Digest) -> Result<bool, CacheError>;
    fn delete(&self, digest: &Digest) -> Result<bool, CacheError>;
}
struct BuildRequest {
    targets:    Vec<NodeId>,
    max_jobs:   Option<usize>,          // overrides scheduler default
    force_rerun: bool,                  // bypass cache; always execute
    dry_run:    bool,                   // eval + cache-check only; no exec
}

struct BuildResult {
    built:    Vec<NodeId>,              // nodes that executed
    cached:   Vec<NodeId>,              // nodes served from cache
    skipped:  Vec<NodeId>,              // nodes that passed early-cutoff
    errors:   Vec<(NodeId, IbgError)>,
    stats:    BuildStats,
}

struct BuildStats {
    wall_ms:            u64,
    cache_hit_rate:     f32,            // cached / (built + cached); NaN if 0 actions
    critical_path_ms:   u64,
    total_actions:      u32,
    recompute_count:    u32,
}
enum IbgError {
    Graph(GraphError),
    Fingerprint(FingerprintError),
    Cache(CacheError),
    Exec(ExecError),
    Sched(SchedError),
    Persist(PersistError),
    Sandbox(SandboxError),
    Remote(RemoteError),
    Io(std::io::Error),
}

enum GraphError {
    CycleDetected { path: Vec<NodeId> },
    NodeNotFound(NodeKey),
    DuplicateKey(NodeKey),
    GraphNotFrozen,
    GraphAlreadyFrozen,
    InvalidEdge { from: NodeId, to: NodeId, reason: &'static str },
}

enum CacheError {
    NotFound(ActionKey),
    Corrupted { digest: Digest, expected: Digest, actual: Digest },
    BackendError(String),
    SchemaMismatch { found: u32, expected: u32 },
}

enum ExecError {
    SpawnFailed { command: String, source: std::io::Error },
    NonZeroExit { code: i32, stderr_digest: Option<Digest> },
    OutputMissing { path: PathBuf },
    OutputDigestMismatch { path: PathBuf, expected: Digest, actual: Digest },
    Timeout { limit_ms: u64 },
}

enum SandboxError {
    UndeclaredInput { path: PathBuf, action: ActionId },
    UndeclaredOutput { path: PathBuf, action: ActionId },
    NonDeterministic { output: PathBuf, run1: Digest, run2: Digest },
    SandboxSetupFailed(String),
}

enum PersistError {
    CorruptState { detail: String },
    SchemaMismatch { found: u32, expected: u32 },
    WriteFailure(std::io::Error),
    MigrationFailed { from: u32, to: u32 },
}
struct StatEntry {
    mtime_ns: i128,
    size:     u64,
    inode:    u64,
    dev:      u64,
}

struct DigestCache {
    inner: DashMap<PathBuf, (StatEntry, Digest)>,
    capacity: usize,
}

impl DigestCache {
    fn get_or_compute(&self, path: &Path) -> Result<Digest, FingerprintError>;
    fn invalidate(&self, path: &Path);
    fn invalidate_all(&self);
}
struct GlobalRevision {
    inner: Arc<AtomicU64>,
}

impl GlobalRevision {
    fn current(&self) -> Revision;
    fn bump(&self) -> Revision;        // fetch_add(1, SeqCst); returns NEW value
    fn snapshot(&self) -> Revision;    // load(SeqCst)
}
struct ResourceBudget {
    cpu_slots:     usize,              // max concurrent `ibg-exec` spawns
    memory_bytes:  Option<u64>,        // advisory; enforced by cgroup if sandbox active
    io_weight:     Option<u32>,        // blkio weight hint (Linux only)
}
struct Scheduler {
    budget:      ResourceBudget,
    ready:       Arc<Mutex<BinaryHeap<ReadyNode>>>,    // max-heap by critical-path rank
    in_flight:   Arc<AtomicUsize>,
    workers:     Vec<WorkerHandle>,
    notify:      Arc<tokio::sync::Notify>,
}

struct ReadyNode {
    id:           NodeId,
    critical_rank: u64,               // longer path = higher priority
}

struct WorkerHandle {
    thread:  tokio::task::JoinHandle<()>,
    steal_q: Arc<SegQueue<NodeId>>,   // crossbeam SegQueue for work-stealing
}
const SCHEMA_VERSION: u32 = 1;

struct SchemaHeader {
    magic:   [u8; 4],    // b"IBG\0"
    version: u32,
    created_at_revision: Revision,
}
enum WatchEventKind { Created, Modified, Deleted, Renamed { from: PathBuf } }

struct WatchEvent {
    path: PathBuf,
    kind: WatchEventKind,
}

struct InvalidationBatch {
    events:       Vec<WatchEvent>,
    affected:     BTreeSet<NodeId>,    // resolved via node_idx + affected_set
    at_revision:  Revision,
}
trait RemoteCacheClient: Send + Sync {
    fn find_missing(&self, digests: &[Digest]) -> BoxFuture<Result<Vec<Digest>, RemoteError>>;
    fn batch_upload(&self, objects: Vec<CasObject>) -> BoxFuture<Result<(), RemoteError>>;
    fn batch_download(&self, digests: &[Digest]) -> BoxFuture<Result<Vec<CasObject>, RemoteError>>;
    fn get_action_result(&self, key: &ActionKey) -> BoxFuture<Result<Option<CacheEntry>, RemoteError>>;
    fn update_action_result(&self, entry: CacheEntry) -> BoxFuture<Result<(), RemoteError>>;
}
struct BuildMetrics {
    cache_hits:        AtomicU64,
    cache_misses:      AtomicU64,
    early_cutoffs:     AtomicU64,
    actions_executed:  AtomicU64,
    actions_failed:    AtomicU64,
    nodes_evaluated:   AtomicU64,
    critical_path_ms:  AtomicU64,    // set once at build end
}
