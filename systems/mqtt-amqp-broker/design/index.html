Core schema & types (§11)

pub struct ClientId(String);                       // validated: ≤65535 B UTF-8, broker-assignable
pub struct Topic(String);                          // publish topic: no +/#/NUL
pub struct TopicFilter(String);                    // sub filter: may contain +/# (validated)
#[repr(u8)] pub enum QoS { AtMostOnce = 0, AtLeastOnce = 1, ExactlyOnce = 2 }
pub struct PacketId(core::num::NonZeroU16);         // 1..=65535 (I4)
pub enum ProtocolVersion { V3_1, V3_1_1, V5_0, Amqp0_9_1 }

pub struct Message {                               // normalized envelope (P13)
    pub topic: Topic,                              // MQTT topic / AMQP routing-key
    pub payload: bytes::Bytes,                     // zero-copy body
    pub qos: QoS,
    pub retain: bool,
    pub dup: bool,
    pub props: Properties,                         // MQTT5 props / AMQP headers (normalized)
    pub expiry: Option<u32>,                       // message-expiry-interval seconds
    pub amqp: Option<AmqpMeta>,                    // exchange, routing-key, delivery-mode, mandatory
}
pub struct AmqpMeta { pub exchange: String, pub routing_key: String,
                      pub delivery_mode: u8 /*1 transient,2 persistent*/, pub mandatory: bool, pub immediate: bool }

pub struct ReasonCode(u8);                          // unified MQTT5 reason / AMQP reply-code mapping
pub enum BrokerError { Codec(CodecError), Net(NetError), Auth(AuthError),
                       Store(StoreError), Proto(ProtoError), Flow(FlowError) }
pub enum MqttPacket {
    Connect(Connect), ConnAck(ConnAck), Publish(Publish),
    PubAck(PubResp), PubRec(PubResp), PubRel(PubResp), PubComp(PubResp),
    Subscribe(Subscribe), SubAck(SubAck), Unsubscribe(Unsubscribe), UnsubAck(UnsubAck),
    PingReq, PingResp, Disconnect(Disconnect), Auth(Auth),     // 15 variants (P14)
}
pub fn decode_mqtt(buf: &mut bytes::BytesMut, ver: ProtocolVersion)
    -> Result<Option<MqttPacket>, CodecError>;     // None = need more bytes; total/bounded (P2)
pub fn encode_mqtt(p: &MqttPacket, ver: ProtocolVersion, out: &mut bytes::BytesMut);
pub fn read_varint(buf: &[u8]) -> Result<(u32, usize), CodecError>;   // ≤4 bytes (I1)
pub fn write_varint(v: u32, out: &mut bytes::BytesMut);               // v ≤ 268_435_455

#[repr(u8)] pub enum FrameType { Method = 1, Header = 2, Body = 3, Heartbeat = 8 }
pub struct AmqpFrame { pub kind: FrameType, pub channel: u16, pub payload: bytes::Bytes }  // frame-end 0xCE checked (I5)
pub enum AmqpMethod { /* connection.* channel.* exchange.* queue.* basic.* tx.* */ }
pub fn decode_amqp(buf: &mut bytes::BytesMut, frame_max: u32) -> Result<Option<AmqpFrame>, CodecError>;
pub fn decode_method(payload: &bytes::Bytes) -> Result<AmqpMethod, CodecError>;  // class-id/method-id + args
pub enum CodecError { MalformedRemainingLength, MalformedPacket{ty:u8}, BadQoS, BadFlags,
                      FrameError, BadFrameEnd, Oversize{limit:u32,got:u32}, ShortBuffer, BadUtf8 }
pub struct Listener { /* TcpListener | TlsAcceptor | WsAcceptor */ }
pub async fn run_listener(l: Listener, core: Arc<RoutingCore>) -> Result<(), NetError>;
pub struct Connection { pub id: u64, pub peer: SocketAddr, pub version: ProtocolVersion, /* .. */ }
pub async fn mqtt_handshake(conn: &mut Connection, pkt: Connect)
    -> Result<ConnAck, NetError>;                  // CONNECT first + once (I18)
pub async fn amqp_handshake(conn: &mut Connection) -> Result<(), NetError>;  // start/tune/open
pub struct KeepAlive { pub secs: u16 }             // disconnect after 1.5× idle (I13)
pub enum NetError { HandshakeFailed(ReasonCode), KeepAliveTimeout, ProtocolViolation, Tls(String), Io(String) }
pub struct Session {
    pub client_id: ClientId,
    pub persistent: bool,
    pub expiry: Option<u32>,                       // MQTT5 session-expiry (None=on disconnect)
    pub subs: SubscriptionSet,
    pub alloc: PacketIdAllocator,                  // (I4)
    pub inflight_out: InflightWindow,              // sent-unacked QoS1/2
    pub inflight_in: DedupWindow,                  // received-unreleased QoS2 (I2)
    pub outbound: tokio::sync::mpsc::Sender<Message>,  // single-writer drain (P6)
    pub channels: HashMap<u16, AmqpChannel>,       // AMQP multiplexing (I17)
}
pub struct PacketIdAllocator { /* bitmap 1..=65535 */ }
impl PacketIdAllocator { pub fn alloc(&mut self) -> Option<PacketId>; pub fn free(&mut self, id: PacketId); }
pub trait SessionStore {                           // backed by brkr-store
    fn load(&self, id: &ClientId) -> Option<Session>;
    fn persist(&self, s: &Session) -> Result<(), StoreError>;
    fn discard(&self, id: &ClientId);
}
pub struct TopicTrie { /* level-segmented radix nodes */ }
impl TopicTrie {
    pub fn insert(&mut self, filter: &TopicFilter, sub: SubRef);     // MQTT
    pub fn insert_binding(&mut self, key: &str, q: QueueRef);        // AMQP topic exchange
    pub fn remove(&mut self, filter: &TopicFilter, sub: SubRef);
    pub fn matches(&self, topic: &Topic, out: &mut Vec<SubRef>);     // O(depth×branch) (I9)
    pub fn matches_amqp(&self, routing_key: &str, out: &mut Vec<QueueRef>);
}
pub struct SubRef { pub client: ClientId, pub qos: QoS, pub sub_id: Option<u32>, pub no_local: bool, pub share_group: Option<String> }
pub fn is_dollar_topic(t: &str) -> bool;            // $SYS / $share exclusion from root #/+
pub enum QoS2RxState { Idle, AwaitRel(PacketId) }   // receiver guardian (I2)
pub enum QoS2TxState { AwaitRec(PacketId), AwaitComp(PacketId) }
pub struct DedupWindow { /* packet-ids reserved PUBLISH..PUBREL */ }
impl DedupWindow { pub fn is_duplicate(&self, id: PacketId) -> bool; pub fn reserve(&mut self, id: PacketId); pub fn release(&mut self, id: PacketId); }
pub struct InflightWindow { pub cap: u16 /* receive-maximum */ }     // (I10)
pub fn on_publish_rx(s: &mut Session, p: &Publish) -> Vec<MqttPacket>;   // emits PUBACK/PUBREC, dedups (I2)
pub fn on_pubrel(s: &mut Session, id: PacketId) -> MqttPacket;          // -> PUBCOMP, release (I2)
pub fn retransmit_on_reconnect(s: &Session) -> Vec<MqttPacket>;         // DUP=1 unacked (P3)
pub enum QoSError { WindowFull, UnknownPacketId, BadState }
pub struct RetainedStore { /* exact-topic -> Message */ }
impl RetainedStore {
    pub fn set(&self, m: Message);                  // empty payload clears (I6)
    pub fn matching(&self, filter: &TopicFilter) -> Vec<Message>;   // on SUBSCRIBE
}
pub struct Will { pub topic: Topic, pub payload: Bytes, pub qos: QoS, pub retain: bool, pub delay: u32 }
pub fn register_will(s: &mut Session, w: Will);
pub fn fire_will(core: &RoutingCore, s: &Session, reason: DisconnectReason);  // ungraceful only (I8)
pub enum DisconnectReason { Graceful, KeepAlive, SocketError, ProtocolViolation, ServerShutdown }

pub struct RoutingCore { pub trie: RwLock<TopicTrie>, pub retain: RetainedStore,
                         pub sessions: SessionRegistry, pub amqp: AmqpModel, pub store: Arc<dyn Store> }
pub fn dispatch(core: &RoutingCore, msg: Message);  // match -> retain update -> fan-out (P13)
pub enum ExchangeKind { Direct, Fanout, Topic, Headers }
pub struct Exchange { pub name: String, pub kind: ExchangeKind, pub durable: bool, pub auto_delete: bool }
pub struct Queue { pub name: String, pub durable: bool, pub exclusive: bool, pub auto_delete: bool,
                   pub messages: BoundedQueue<Delivery>, pub consumers: Vec<Consumer> }
pub struct Binding { pub exchange: String, pub queue: String, pub key: String, pub args: FieldTable }
pub struct Consumer { pub tag: String, pub channel: u16, pub no_ack: bool, pub prefetch: u16 }
pub fn route(model: &AmqpModel, ex: &str, key: &str, headers: &FieldTable) -> Vec<QueueRef>;  // by ExchangeKind (8.13)
pub fn basic_publish(model: &AmqpModel, m: Message) -> Result<(), AmqpReply>;   // mandatory -> return (P10)
pub enum AmqpReply { NotFound(u16/*404*/), AccessRefused(u16/*403*/), NoRoute(u16/*312*/), PreconditionFailed(u16/*406*/) }
pub struct BoundedQueue<T> { pub max_len: usize, pub max_bytes: usize }    // (I16)
pub enum SlowConsumerPolicy { Block, DropOldest, Disconnect }
pub struct Credit { pub receive_maximum: u16, pub prefetch: u16, pub channel_active: bool }  // (I10)
pub fn governor_admit(global_bytes: &AtomicUsize, add: usize, cap: usize) -> bool;            // memory pressure

pub trait Store {                                   // WAL + segments (#69)
    fn append(&self, rec: &WalRecord) -> Result<Lsn, StoreError>;   // returns log sequence number
    fn fsync(&self) -> Result<(), StoreError>;                      // group-commit (I11)
    fn recover(&self) -> Result<RecoveredState, StoreError>;        // replay on boot (P7)
}
pub enum WalRecord { SessionUpsert{..}, SessionDiscard{..}, SubAdd{..}, SubDel{..},
                     RetainSet{..}, RetainClear{..}, InflightSet{..}, InflightClear{..},
                     Enqueue{queue:String, body_ref:SegRef}, Ack{queue:String, tag:u64} }
pub enum StoreError { Io(String), Corrupt{lsn:u64}, Checksum, Full }
