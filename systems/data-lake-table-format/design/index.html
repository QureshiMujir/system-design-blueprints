Core schema & types (§11)

// table-metadata: immutable, Arc-shared, serde JSON, forward-compatible (unknown fields ignored)
struct TableMetadata {
    format_version: u8,
    table_uuid: Uuid,
    location: String,
    current_schema_id: i32,
    schemas: Vec<Schema>,
    partition_specs: Vec<PartitionSpec>,
    current_snapshot_id: Option<i64>,
    snapshot_log: Vec<SnapshotRef>,
}
// mutation: none in place — every change produces a new TableMetadata
// concurrency: read-shared via Arc, never mutated after construction
// persistence: one JSON object per version, written before CAS
// compatibility: format_version gates readable field set (ADR-26, I12)

type MetadataLocation = String; // object-store URI, versioned filename

// snapshot: append-only, immutable
struct Snapshot {
    snapshot_id: i64,
    parent_snapshot_id: Option<i64>,
    sequence_number: i64,
    timestamp_ms: i64,
    manifest_list: MetadataLocation,
    summary_operation: OperationKind, // append|overwrite|delete|replace
}
type SnapshotId = i64;
struct SnapshotRef { snapshot_id: SnapshotId, kind: RefKind /* tag|branch */, retention_ms: Option<i64> }
// manifest: Avro-serialized on disk, in-memory as typed structs
struct ManifestList { manifests: Vec<ManifestFileSummary> }
struct ManifestFileSummary {
    manifest_path: MetadataLocation,
    partition_spec_id: i32,
    added_files_count: i32,
    partition_value_summary: Vec<PartitionFieldSummary>, // list-level pruning input
}
struct ManifestFile { entries: Vec<ManifestEntry>, sequence_number: i64 }
struct ManifestEntry {
    status: EntryStatus, // added|existing|deleted
    sequence_number: i64,
    file: FileRef, // DataFile | DeleteFile
}
// concurrency: manifests are write-once objects; never edited, only superseded
// data/delete files
struct DataFile {
    file_path: String,
    file_format: FileFormat, // Parquet (only supported today)
    partition: PartitionValue,
    record_count: i64,
    column_stats: HashMap<FieldId, ColumnStats>,
}
struct DeleteFile { file_path: String, content_type: ContentType /* PositionDeletes|EqualityDeletes */, applies_to_seq_ge: i64 }
enum FileFormat { Parquet }
enum ContentType { Data, PositionDeletes, EqualityDeletes }
struct ColumnStats { value_count: i64, null_count: i64, lower_bound: Vec<u8>, upper_bound: Vec<u8> }
struct Metrics { column_stats: HashMap<FieldId, ColumnStats> }
// schema and partitioning
struct Schema { schema_id: i32, fields: Vec<NestedField> }
struct NestedField { field_id: FieldId, name: String, required: bool, ty: Type }
type FieldId = i32;
enum TypePromotion { IntToLong, FloatToDouble, DecimalPrecisionIncrease }
struct PartitionSpec { spec_id: i32, fields: Vec<PartitionField> }
struct PartitionField { source_field_id: FieldId, field_id: FieldId, transform: Transform }
enum Transform { Identity, Bucket(u32), Truncate(u32), Year, Month, Day, Hour }
struct PartitionValue { values: Vec<Vec<u8>> } // one encoded value per PartitionField
// commit / concurrency
trait Catalog {
    fn load_pointer(&self, table: &TableIdent) -> Result<MetadataLocation, CatalogError>;
    fn cas_pointer(&self, table: &TableIdent, old: &MetadataLocation, new: &MetadataLocation) -> Result<(), CommitConflict>;
}
struct Commit { base: MetadataLocation, new_metadata: TableMetadata }
struct CommitConflict { expected: MetadataLocation, actual: MetadataLocation }
enum Isolation { Snapshot } // the only level offered (ADR-07)

// scan planning
struct ScanPlan { snapshot_id: SnapshotId, tasks: Vec<FileScanTask> }
struct FileScanTask { data_file: DataFile, delete_files: Vec<DeleteFile>, residual: Predicate }
enum Predicate { And(Box<Predicate>, Box<Predicate>), Or(..), Cmp{ field_id: FieldId, op: CmpOp, value: Vec<u8> }, AlwaysTrue }
struct Residual(Predicate);

// maintenance
struct ExpireSnapshots { keep: Vec<SnapshotId>, older_than_ms: i64 }
struct OrphanFileGc { dry_run: bool }

// object store
trait ObjectStore {
    fn get(&self, path: &str) -> Result<Bytes, StoreError>;
    fn put(&self, path: &str, data: Bytes) -> Result<(), StoreError>;
    fn put_if_absent(&self, path: &str, data: Bytes) -> Result<(), StoreError>; // CAS primitive
    fn list(&self, prefix: &str) -> Result<Vec<String>, StoreError>;
}

// error taxonomy
enum TableError {
    Commit(CommitConflict),
    Schema(SchemaViolation),
    Manifest(ManifestCorruption),
    Store(StoreError),
    Catalog(CatalogError),
    NotFound,
}
