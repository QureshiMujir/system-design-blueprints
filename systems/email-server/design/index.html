Core schema & types (§11)

pub struct Mailbox { pub local: SmolStr, pub domain: SmolStr }
impl Mailbox {
    pub fn parse(s: &str) -> Result<Self, ParseError>;
    pub fn to_string(&self) -> String;
}

pub struct Envelope {
    pub from: Mailbox,
    pub rcpts: Vec<Mailbox>,
    pub helo: SmolStr,
    pub remote_ip: IpAddr,
    pub authenticated_user: Option<UserId>,
    pub tls_info: Option<TlsInfo>,
    pub size: Option<u64>,
    pub received_at: SystemTime,
}

pub struct Message { pub headers: Headers, pub body: Bytes, pub mime: Option<MimeStructure> }
pub struct Headers { pub list: Vec<HeaderField> }
pub struct HeaderField { pub name: SmolStr, pub value: SmolStr }
pub struct MimeStructure { pub content_type: ContentType, pub parts: Vec<MimePart> }
pub struct UserId(pub u64);
pub enum SmtpState {
    Connected,
    Greeted,
    Authenticated(UserId),
    MailFrom { from: Mailbox },
    RcptTo { from: Mailbox, rcpts: Vec<Mailbox> },
    InData { envelope: Envelope, body_buf: BytesMut },
    Closed,
}

pub struct SmtpSession {
    pub state: SmtpState,
    pub stream: TlsOptional<TcpStream>,
    pub remote_ip: IpAddr,
    pub config: Arc<Config>,
}

pub enum SmtpCommand {
    Ehlo(SmolStr), Helo(SmolStr),
    Mail { from: Mailbox, params: Vec<MailParam> },
    Rcpt { to: Mailbox, params: Vec<RcptParam> },
    Data, Rset, Quit, StartTls,
    Auth(SaslMechName, Option<Bytes>),
}
pub enum ImapState {
    NotAuthenticated,
    Authenticated(UserId),
    Selected { mailbox: MailboxId, modseq_high_water: u64 },
    IdleSelected { mailbox: MailboxId },
    Closed,
}

pub struct MailboxIndexRow {
    pub uid: u64,
    pub uidvalidity: u32,
    pub modseq: u64,
    pub flags: FlagSet,
    pub maildir_filename: SmolStr,
}

pub struct QuotaState { pub user: UserId, pub used_bytes: u64, pub limit_bytes: u64 }
pub trait SaslMechanism {
    fn name(&self) -> SaslMechName;
    fn step(&mut self, client_data: &[u8]) -> Result<SaslStep, SaslError>;
}
pub enum SaslStep { Continue(Bytes), Done(UserId), Failed }

pub struct TlsInfo { pub version: TlsVersion, pub cipher: SmolStr, pub sni: Option<SmolStr> }

pub struct DkimSignature { pub selector: SmolStr, pub domain: SmolStr,
    pub canon: (Canon, Canon), pub algorithm: SigAlg, pub signed_headers: Vec<SmolStr> }
pub enum SpfResult { Pass, Fail, SoftFail, Neutral, None, TempError, PermError }
pub struct DmarcVerdict { pub policy: DmarcPolicy, pub aligned: bool }

pub enum SieveAction { Keep, Discard, Redirect(SmolStr), Fileinto(SmolStr),
    Reject(SmolStr), Vacation { reason: SmolStr, days: u32 } }

pub struct QueueEntry { pub id: QueueId, pub envelope: Envelope, pub state: QueueState,
    pub attempts: u32, pub next_attempt_at: SystemTime }
pub enum QueueState { Active, Deferred, Hold, Bounce }

pub struct DsnReport { pub original: QueueId, pub final_recipient: Mailbox,
    pub action: DsnAction, pub status: SmolStr }
