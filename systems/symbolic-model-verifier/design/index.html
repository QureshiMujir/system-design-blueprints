// Part 1 of 2: BddManager core types and Boolean/quantification API
/// Canonical handle: an index into the owning BddManager's node table.
/// Copy. Valid ONLY against its minting manager; comparing/using a Bdd
/// against a different manager is a logic bug (foreign-node read).
/// Equality of two Bdds from the SAME manager == boolean-function equality
/// (the canonicity contract — P1/P2).
pub struct Bdd(u32);

/// The ROBDD package. Single mutable owner of: the node table (level, low,
/// high per node), the unique table (hashcons on (level, low, high) — P2),
/// the computed/op cache (memoizes ite results — P2), and the fixed total
/// variable order. !Sync by design (single-threaded). All structural ops
/// take &mut self; pure queries take &self.
pub struct BddManager { /* node table, unique table, computed cache, var order */ }

impl BddManager {
    /// Terminal nodes. Distinct handles: bdd_true() != bdd_false() (smoke test, Phase 0).
    pub fn bdd_true(&self) -> Bdd;
    pub fn bdd_false(&self) -> Bdd;

    /// Projection function for the variable at `level`: the BDD with high=true,
    /// low=false. Allocates/levels lazily; idempotent under hashcons.
    pub fn ithvar(&mut self, level: u32) -> Bdd;

    /// THE primitive (P2). ite(f,g,h) = (f AND g) OR (not f AND h).
    /// Memoized in the computed cache keyed by (f,g,h). O(|f|*|g|*|h|) with cache.
    pub fn ite(&mut self, f: Bdd, g: Bdd, h: Bdd) -> Bdd;

    /// Boolean connectives — all special cases of ite (P2): and=ite(f,g,⊥), etc.
    pub fn and(&mut self, f: Bdd, g: Bdd) -> Bdd;
    pub fn or(&mut self, f: Bdd, g: Bdd) -> Bdd;
    pub fn xor(&mut self, f: Bdd, g: Bdd) -> Bdd;
    pub fn imp(&mut self, f: Bdd, g: Bdd) -> Bdd;

    /// Negation. O(1): complement edge (or trivial cache hit). &self — never
    /// allocates a node. not(not(f)) == f structurally.
    pub fn not(&self, f: Bdd) -> Bdd;

    /// Cofactor / restrict: f with variable `level` fixed to `val` (P9).
    /// Eliminates that level from the result.
    pub fn restrict(&mut self, f: Bdd, level: u32, val: bool) -> Bdd;

    /// Existential / universal quantification over the variables in `cube`
    /// (a conjunction-of-positive-literals BDD). exists = OR of cofactors;
    /// forall = AND of cofactors. Foundation of image/pre-image (P9).
    pub fn exists(&mut self, f: Bdd, cube: Bdd) -> Bdd;
    pub fn forall(&mut self, f: Bdd, cube: Bdd) -> Bdd;
}
// Part 2 of 2: BddManager relational product, rename, satcount, and node count
impl BddManager {
    /// Relational product: exists cube . (f AND g). Fused — never materializes
    /// the full (f AND g) conjunction. The hot path for image computation
    /// (conjunctive partitioning + early quantification ride on this — P9/P10).
    pub fn and_exists(&mut self, f: Bdd, g: Bdd, cube: Bdd) -> Bdd;

    /// Variable substitution per `map` (present<->next swap for image/preimage).
    /// map MUST be order-respecting or a reorder pass is implied.
    pub fn rename(&mut self, f: Bdd, map: &VarMap) -> Bdd;

    /// #-of-satisfying-assignments over `nvars` variables (P11). u128: handles
    /// up to 2^127 models without overflow (target reachable-state counts ~10^20).
    pub fn satcount(&self, f: Bdd, nvars: u32) -> u128;

    /// Number of distinct BDD nodes reachable from `f` (DAG size, not paths).
    pub fn node_count(&self, f: Bdd) -> usize;

    /// GC: mark from external roots, sweep dead nodes, purge computed cache (P11).
    pub fn ref_inc(&mut self, f: Bdd);
    pub fn ref_dec(&mut self, f: Bdd);
    pub fn gc(&mut self);

    /// Dynamic variable reordering — sifting algorithm (P7). Triggered when
    /// live_nodes > gc_threshold. Preserves all function semantics / handles.
    pub fn sift_all(&mut self);

    /// Graphviz DOT export for debugging BDD structure.
    pub fn to_dot(&self, roots: &[Bdd]) -> String;

    /// Statistics snapshot: node counts, cache hit/miss rates, GC stats.
    pub fn stats(&self) -> BddStats;
}
INVARIANT (ROBDD canonicity):
  fixed total variable order
  + reduction rule R1: no node with low == high (redundant-test elimination)
  + reduction rule R2: no two structurally-identical nodes (isomorphic merge)
  ==> exactly one canonical DAG per boolean function
  ==> Bdd handle equality  <=>  boolean-function equality   (O(1) test)
/// Variable substitution table for rename (present<->next). Maps source levels
/// to target levels (e.g. present var level -> its next-state companion level).
/// Built once from EncodedFsm.present / .next; reused every image step.
pub struct VarMap { /* level -> level mapping, order-aware */ }
// Part 1 of 2: smv-syntax module-level types (SmvModule, VarType, AssignKind)
/// Parsed SMV module (one MODULE). Owned tree; produced by Parser, consumed by
/// the Encoder. Vec-per-section preserves source order (matters for ASSIGN /
/// DEFINE dependency analysis and CircularDefine detection).
pub struct SmvModule {
    pub vars: Vec<VarDecl>,        // VAR section: name : VarType
    pub defines: Vec<DefineDecl>,  // DEFINE: macro := Expr (no extra state)
    pub assigns: Vec<AssignDecl>,  // ASSIGN: init/next/plain := Expr
    pub trans: Vec<Expr>,          // TRANS: boolean relation over present+next
    pub invar: Vec<Expr>,          // INVAR: state invariant constraint
    pub specs: Vec<SpecDecl>,      // CTLSPEC / LTLSPEC / INVARSPEC
}

/// Finite-domain variable type. ALL types are finite (symbolic encoding needs
/// a bounded boolean encoding). No reals/unbounded ints (non-goal: infinite state).
pub enum VarType {
    Boolean,           // 1 bit
    Enum(Vec<Atom>),   // scalar: ceil(log2 N) bits + out-of-range constraint
    Range(i64, i64),   // bounded integer [lo, hi]: ceil(log2(hi-lo+1)) bits
}

/// Kind of an ASSIGN. init(x):=e seeds the initial predicate; next(x):=e
/// defines the transition relation for x; plain x:=e is an invariant define.
pub enum AssignKind { Init, Next, Plain }
// Part 2 of 2: smv-syntax expression and error types (Expr, SpecDecl, ParseError)
/// Expression AST. Next(.) is legal only inside next() / TRANS context
/// (refers to the next-state value); the Encoder rejects Next() in init/invar.
pub enum Expr {
    Var(Atom),
    Const(Value),
    Not(Box<Expr>),
    Bin(BinOp, Box<Expr>, Box<Expr>),   // & | -> <-> = != < <= + - ...
    Case(Vec<(Expr, Expr)>),            // case g1:e1; g2:e2; ... esac
    Next(Box<Expr>),                    // next-state reference
}

/// A specification clause. Ctl/Ltl carry the raw property string parsed later
/// by smv-logic (defers temporal-operator parsing out of the model grammar);
/// InvarSpec is a pure boolean Expr checked as AG over reachable states.
pub enum SpecDecl {
    Ctl(String),
    Ltl(String),
    InvarSpec(Expr),
}

/// Recoverable parse/static-check failure. Carries kind + source span (not shown).
pub struct ParseError { /* kind: ParseErrorKind, span */ }

pub enum ParseErrorKind {
    UnexpectedToken,  // lexical/grammar
    UnknownVar,       // name resolution
    BadType,          // ill-formed VarType
    TypeMismatch,     // e.g. bool used where Range expected
    CircularDefine,   // DEFINE dependency cycle (Phase 3 must detect)
    OutOfRange,       // Range/Enum literal outside its domain
}
/// Boolean encoding of ONE finite-domain variable: the ordered BDD levels its
/// bits occupy, plus its source VarType (for decode / out-of-range constraints).
/// levels.len() == ceil(log2(domain size)); Boolean => 1 level.
pub struct VarEncoding {
    pub levels: Vec<u32>,  // BDD levels (MSB-first) carrying this var's bits
    pub ty: VarType,       // original domain (for StateAssignment decode)
}

/// The fully-built symbolic finite-state machine. IMMUTABLE post-build: every
/// field is a Bdd handle into the shared BddManager. Checkers borrow &EncodedFsm.
pub struct EncodedFsm {
    pub present: VarMap,        // present-state variable map
    pub next: VarMap,          // next-state variable map (companion levels)
    pub init: Bdd,             // initial-state predicate = init() AND INVAR
    pub trans: Bdd,            // transition relation = next() AND TRANS AND next-INVAR
    pub invar: Bdd,            // state invariant (conjoined into init/trans)
    pub fairness: Vec<Bdd>,    // generalized-Buchi fairness constraints (per-set)
    pub ap: Vec<(Atom, Bdd)>,  // atomic-proposition name -> its state-set Bdd
}
/// CTL formula over state-set BDDs. Minimal adequate set {EX, EU, EG, Not, And/Or};
/// AX/AU/AG/AF/EF/AF are DERIVED (NNF rewrite, Phase 7) — not stored variants.
/// Ap holds a state-set Bdd already bound via EncodedFsm.ap (so checking is
/// pure BDD work, no name lookup).
pub enum CtlFormula {
    Ap(Bdd),
    Not(Box<CtlFormula>),
    And(Box<CtlFormula>, Box<CtlFormula>),
    Or(Box<CtlFormula>, Box<CtlFormula>),
    Ex(Box<CtlFormula>),
    Eg(Box<CtlFormula>),
    Eu(Box<CtlFormula>, Box<CtlFormula>),   // E[p U q]
}

/// LTL formula. X/U are primitive; R/F/G provided (F p = true U p; G p = not F not p;
/// p R q = not(not p U not q)). NNF normalization (Phase 9) pushes Not to APs before
/// LTL->Buchi translation.
pub enum LtlFormula {
    Ap(Bdd),
    Not(Box<LtlFormula>),
    And(Box<LtlFormula>, Box<LtlFormula>),
    Or(Box<LtlFormula>, Box<LtlFormula>),
    X(Box<LtlFormula>),
    U(Box<LtlFormula>, Box<LtlFormula>),
    R(Box<LtlFormula>, Box<LtlFormula>),
    F(Box<LtlFormula>),
    G(Box<LtlFormula>),
}
/// Symbolic CTL engine. Borrows the manager mutably (fixpoints allocate nodes)
/// and the FSM immutably. Lifetime 'a ties both borrows to the call site.
pub struct CtlChecker<'a> { /* &mut BddManager + &'a EncodedFsm */ }

impl CtlChecker<'_> {
    /// Evaluate `f` to the state-set Bdd of states where f holds.
    /// EX p = preimage(p); E[p U q] = lfp Z. q OR (p AND EX Z);
    /// EG p = gfp Z. p AND EX Z. Spec HOLDS iff init ⊆ check(f),
    /// i.e. and(init, not(check(f))) == bdd_false().
    pub fn check(&mut self, f: &CtlFormula) -> Bdd;

    /// Reachable least-fixpoint: lfp Z. init OR Img(Z). Cached; shared with checks.
    pub fn reachable(&mut self) -> Bdd;

    /// Shortest counterexample/witness via onion-ring backward sets (Phase 8);
    /// EG uses a lasso witness. None if the spec holds (no counterexample).
    pub fn counterexample(&mut self, f: &CtlFormula) -> Option<Trace>;
}

/// A finite or lasso-shaped trace. states[0] is an initial state; if loopback
/// is Some(i), the trace loops from the last state back to states[i] (a lasso,
/// for EG / LTL fair cycles). loopback == None => a finite (safety) witness.
pub struct Trace {
    pub states: Vec<StateAssignment>,
    pub loopback: Option<usize>,
}

/// Symbolic LTL engine: product of EncodedFsm with the Buchi automaton for
/// (not phi), then Emerson-Lei fair-cycle detection (nested gfp/lfp over EX +
/// acceptance/fairness sets). Model |= A phi iff NO fair cycle is reachable.
pub struct LtlChecker<'a> { /* product + Emerson-Lei */ }
/// Bounded Model Checking unroller. Borrows the FSM immutably; produces CNF.
pub struct Unroller<'a> { /* &'a EncodedFsm */ }
// Builds, for bound k:  I(s0) AND (conj_{i<k} T(s_i, s_{i+1})) AND (or_{i<=k} not p(s_i))
// as a CnfFormula via a Tseitin transform (fresh vars for BDD/Boolean subterms).
// Incremental k supported: clauses for step k reuse prior structure.

/// A CNF formula in DIMACS literal convention: each clause is a Vec<i32> of
/// non-zero literals (+v / -v, 1-based); nvars = highest variable index.
/// The ONLY serialization boundary to a SAT solver. Owned; portable across
/// processes (unlike BDD handles).
pub struct CnfFormula {
    pub clauses: Vec<Vec<i32>>,
    pub nvars: u32,
}

/// Pluggable SAT solver boundary: DIMACS CNF in, assignment out. Lets #277's
/// solver or a bundled DPLL/CDCL satisfy the same trait (non-goal: smv-bmc does
/// NOT ship a competitive solver). &mut self: a solver carries mutable state.
pub trait SatBackend {
    fn solve(&mut self, cnf: &CnfFormula) -> SatResult;
}

/// SAT outcome. Sat carries a full assignment indexed by (var-1); Unsat means
/// no length-<=k counterexample (and, at the completeness threshold, none at all).
pub enum SatResult {
    Sat(Vec<bool>),
    Unsat,
}
