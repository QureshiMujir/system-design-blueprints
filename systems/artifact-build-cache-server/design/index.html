Core schema & types (§11)

// digest.rs
pub struct Digest { pub hash: String, pub size_bytes: i64 }
// Ownership: Clone, PartialEq, Eq, Hash; no interior mutability.
// Persistence: proto-serialized on wire; stored as (hash_bytes[32], i64) in BlobMeta.
// Invariant: hash is lowercase hex; size_bytes >= 0 for stored blobs.
impl Digest {
    pub fn zero_sha256() -> Self;             // e3b0c44... / 0
    pub fn from_hex(hash: &str, size: i64) -> Result<Self, AbcError>;
    pub fn to_resource_path(&self) -> String; // "{hash}/{size}"
}

pub enum DigestFunction { Sha256, Sha512, Blake3 }
impl DigestFunction {
    pub fn hash_len_hex(&self) -> usize;     // 64, 128, 64
    pub fn from_proto(v: i32) -> Result<Self, AbcError>;
}

pub struct InstanceName(pub String);
impl InstanceName {
    pub fn default() -> Self;               // ""
    pub fn validate(&self) -> Result<(), AbcError>;
}
// directory.rs
pub struct FileNode     { pub name: String, pub digest: Digest, pub is_executable: bool }
pub struct DirectoryNode{ pub name: String, pub digest: Digest }
pub struct SymlinkNode  { pub name: String, pub target: String }
pub struct Directory    {
    pub files:       Vec<FileNode>,
    pub directories: Vec<DirectoryNode>,
    pub symlinks:    Vec<SymlinkNode>,
}
impl Directory {
    pub fn validate(&self) -> Result<(), AbcError>;  // sorted + no dup names + no slash in name
    pub fn serialized_digest(&self, f: DigestFunction) -> Result<Digest, AbcError>;
}
// action.rs
pub struct EnvVar   { pub name: String, pub value: String }
pub struct Command  { pub arguments: Vec<String>, pub environment_variables: Vec<EnvVar>,
                      pub output_paths: Vec<String> }
pub struct Action   { pub command_digest: Digest, pub input_root_digest: Digest,
                      pub timeout: std::time::Duration, pub do_not_cache: bool }
// result.rs
pub struct OutputFile      { pub path: String, pub digest: Digest, pub is_executable: bool }
pub struct OutputDirectory { pub path: String, pub tree_digest: Digest }
pub struct ExecMeta        { pub worker: String, pub queued_at: Timestamp, pub exec_completed_at: Timestamp }
pub struct ActionResult    {
    pub output_files:        Vec<OutputFile>,
    pub output_directories:  Vec<OutputDirectory>,
    pub exit_code:           i32,
    pub stdout_digest:       Option<Digest>,
    pub stderr_digest:       Option<Digest>,
    pub execution_metadata:  ExecMeta,
}
impl ActionResult {
    pub fn all_output_digests(&self) -> Vec<&Digest>;  // all blobs that must be in CAS
}
// error.rs
pub enum AbcError {
    NotFound(Digest),
    DigestMismatch { expected: Digest, got: Digest },
    SizeMismatch   { expected: i64, got: i64 },
    Incomplete      { action_digest: Digest, missing: Vec<Digest> },
    InvalidResourceName(String),
    Unauthenticated,
    PermissionDenied { instance: String },
    OutOfSpace { available: u64, needed: u64 },
    Io(std::io::Error),
    Proto(prost::DecodeError),
}
impl AbcError {
    pub fn to_grpc_status(&self) -> tonic::Status;
    pub fn to_http_status(&self) -> http::StatusCode;
}
// compressor.rs
pub enum Compressor { Identity, Zstd }
impl Compressor {
    pub fn from_proto(v: i32) -> Result<Self, AbcError>;
    pub fn to_resource_segment(&self) -> &'static str; // "blobs" or "compressed-blobs/zstd"
}
// traits.rs
pub trait BlobStore: Send + Sync {
    fn put(&self, d: &Digest, r: &mut dyn Read) -> Result<(), AbcError>;
    fn get(&self, d: &Digest) -> Result<Box<dyn Read + Send>, AbcError>;
    fn contains(&self, d: &Digest) -> bool;
    fn find_missing(&self, digests: &[Digest]) -> Vec<Digest>;
    fn delete(&self, d: &Digest) -> Result<(), AbcError>;
    fn usage_bytes(&self) -> u64;
}

// index.rs
pub struct BlobMeta {
    pub digest:     Digest,
    pub size:       i64,
    pub atime:      std::time::SystemTime,
    pub refcount:   u32,
    pub compressor: Compressor,
    pub tier:       Tier,
}

pub enum Tier { Memory, Disk, Remote }

pub struct Watermark { pub high: u64, pub low: u64 }  // bytes

// tiered.rs  — routes put/get across tiers; handles promotion/demotion
pub struct TieredStore { memory: Arc<MemoryTier>, disk: Arc<DiskTier>, remote: Option<Arc<dyn RemoteTier>> }
impl TieredStore {
    pub fn new(cfg: &StoreConfig) -> Self;
    pub fn promote(&self, d: &Digest, from: Tier) -> Result<(), AbcError>;
}
// gc.rs
pub struct GcSweeper {
    store:     Arc<TieredStore>,
    ac:        Arc<AcCache>,
    watermark: Watermark,
    interval:  Duration,
}
impl GcSweeper {
    pub fn run(self) -> tokio::task::JoinHandle<()>;  // background loop
    pub fn trigger_now(&self);
    pub fn evict_lru_until(&self, target_bytes: u64) -> Result<u64, AbcError>;  // returns freed bytes
}
// resource_name.rs
pub struct ParsedResourceName {
    pub instance:    InstanceName,
    pub upload_uuid: Option<Uuid>,
    pub compressor:  Compressor,
    pub digest:      Digest,
}
pub fn parse_resource_name(s: &str, fn_: DigestFunction) -> Result<ParsedResourceName, AbcError>;

// capabilities.rs
pub struct ServerCapabilities {
    pub digest_functions:            Vec<DigestFunction>,
    pub supported_compressors:       Vec<Compressor>,
    pub max_batch_total_size_bytes:  i64,
    pub symlink_absolute_path_strategy: SymlinkStrategy,
}
