Core schema & types (§11)

struct PacketDescriptor {
    ingress_id: u64,
    rx_queue: u16,
    timestamp_ns: u64,
    packet_len: u32,
    capture_len: u32,
    flow_hash: u64,
    checksum_state: ChecksumState,
    payload_ref: PayloadRef, }
struct FlowKey {
    src_ip: IpAddr,
    dst_ip: IpAddr,
    src_port: u16,
    dst_port: u16,
    protocol: L4Protocol,
    vlan_id: u16, }
struct FlowContext {
    key: FlowKey,
    state: FlowState,
    parser_state: ParserStateSlot,
    stream: Option<TcpReassemblyBuffer>,
    counters: FlowCounters,
    created_at_ns: u64,
    last_seen_ns: u64,
    memory_bytes: u32, }
struct TcpReassemblyBuffer {
    next_expected_seq: u32,
    segments: SegmentTree,
    high_water_bytes: u32,
    gap_count: u16,
    overlap_count: u16, }
struct ParserEvent {
    parser_id: ParserId,
    timestamp_ns: u64,
    severity: Severity,
    metadata: EventMetadata,
    payload_refs: SmallVec<PayloadRef>, }
struct RuleMatch {
    rule_id: RuleId,
    flow_key: FlowKey,
    matched_at_ns: u64,
    evidence: MatchEvidence,
    rule_version: u32, }
struct TelemetryEnvelope {
    kind: TelemetryKind,
    payload: TelemetryPayload,
    emitted_at_ns: u64,
    trace_id: Option<TraceId>, }
struct ReplayFrame {
    frame_seq: u64,
    segment_id: SegmentId,
    schema_version: u16,
    body: ReplayFrameBody,
    checksum: u32, }
struct WorkerHandle {
    worker_id: WorkerId,
    numa_node: NumaNodeId,
    kind: WorkerKind,
    shutdown: ShutdownToken, }
struct RuntimeError {
    class: ErrorClass,
    subsystem: SubsystemId,
    context: ErrorContext,
    retryable: bool, }
trait ProtocolParser {
    fn parser_id(&self) -> ParserId;
    fn feed(&mut self, input: ParserInput) -> ParserResult;
    fn reset(&mut self); }
trait RuleEvaluator {
    fn evaluate(&self, events: &[ParserEvent], ctx: &FlowCounters) -> Vec<RuleMatch>;
    fn budget_cost(&self) -> ExecutionBudget; }
