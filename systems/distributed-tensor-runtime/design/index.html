Core schema & types (§11)

pub struct Tensor<T: Scalar> { data: Buffer<T>, shape: Vec<usize>, strides: Vec<isize>, dtype: DType, device: DeviceId }
pub enum DType { F32, F64, Bf16 }
pub struct DeviceId(u32);            // CPU now; GPU id deferred
pub struct DeviceMesh { axes: Vec<(String, usize)>, ranks: Vec<RankId> }   // e.g. [("data",4),("model",2)]
pub struct RankId(u32);
pub enum DimSharding { Replicated, Sharded(AxisName), Partial(ReduceOp) }
pub struct ShardSpec { dims: Vec<DimSharding> }     // one entry per tensor dim
pub struct ShardedTensor<T> { global_shape: Vec<usize>, spec: ShardSpec, local: Tensor<T>, mesh: Arc<DeviceMesh> }
impl ShardSpec { fn validate(&self, global:&[usize], mesh:&DeviceMesh) -> Result<(),DtrError>; }  // I2
pub struct ProcessGroup { axis: AxisName, members: Vec<RankId>, my_index: usize, size: usize }
impl DeviceMesh { fn group_over(&self, axis:&str, my_rank:RankId) -> ProcessGroup; }
pub enum ReduceOp { Sum, Max, Min, Prod }
pub trait Collectives {
    fn all_reduce<T>(&self, buf:&mut [T], op:ReduceOp, pg:&ProcessGroup) -> Result<(),DtrError>;
    fn all_gather<T>(&self, send:&[T], recv:&mut [T], pg:&ProcessGroup) -> Result<(),DtrError>;
    fn reduce_scatter<T>(&self, send:&[T], recv:&mut [T], op:ReduceOp, pg:&ProcessGroup) -> Result<(),DtrError>;
    fn broadcast<T>(&self, buf:&mut [T], root:usize, pg:&ProcessGroup) -> Result<(),DtrError>;
    fn all_to_all<T>(&self, send:&[T], recv:&mut [T], pg:&ProcessGroup) -> Result<(),DtrError>;
    fn barrier(&self, pg:&ProcessGroup) -> Result<(),DtrError>;
}
pub enum CollAlgo { Ring, RecursiveHalvingDoubling, Tree, Auto }   // Auto → cost model (ADR-5)
pub struct NetParams { alpha_s: f64 /*latency*/, beta_s_per_byte: f64 /*1/BW*/, gamma_s_per_byte: f64 /*reduce*/ }
pub struct CollCost { time_s: f64, bytes_moved: usize, steps: usize }
fn cost(algo: CollAlgo, coll: CollKind, n: usize, bytes: usize, net: &NetParams) -> CollCost;   // §8.4
fn select(coll: CollKind, bytes: usize, n: usize, net: &NetParams) -> CollAlgo;                  // crossover
pub struct Frame<'a> { version:u16, kind:FrameKind, stream:u32, seq:u64, payload:&'a [u8], checksum:u32 }
pub enum FrameKind { Rendezvous, DataChunk, Ack, Credit, Control, Bye }
pub trait Transport {
    fn send(&self, peer:RankId, stream:u32, buf:&[u8]) -> Result<(),DtrError>;   // flow-controlled (I8)
    fn recv(&self, peer:RankId, stream:u32) -> Result<Bytes,DtrError>;            // reliable, ordered (I7)
}
pub struct OpNode { id: OpId, op: Op, inputs: Vec<TensorId>, out_spec: ShardSpec, deps: Vec<OpId>, stream: StreamId }
pub enum Op { Matmul, Add, Mul, Reduce(ReduceOp), Collective(CollKind, ProcessGroup), Reshard(ShardSpec) }
pub struct Graph { nodes: Vec<OpNode> }
pub struct Scheduler { /* topo order, stream assignment, overlap (I10/I11) */ }
pub struct World { rank: RankId, world_size: usize, mesh: Arc<DeviceMesh>, transport: Arc<dyn Transport> }
fn bootstrap(coordinator: SocketAddr, expected: usize, mesh_shape: &[(String,usize)]) -> Result<World,DtrError>; // I13
pub trait FailureDetector { fn is_alive(&self, r:RankId)->bool; fn on_dead(&self, cb: impl Fn(RankId)); }       // I12
pub enum DtrError { CollArgMismatch{expected:u64,got:u64}, ShardInvalid(String), DimMismatch,
                    TransportClosed(RankId), Corruption{stream:u32,seq:u64}, FlowControl,
                    RankDead(RankId), BarrierTimeout, ProtoVersion{ours:u16,theirs:u16}, BadArg(String) }
pub struct CollTrace { coll:&'static str, algo:&'static str, dtype:&'static str, bytes:usize, world:usize,
                       steps:usize, duration_ns:u64, bandwidth_gbps:f64, reduce_order_id:u64, overlapped_ns:u64 }
pub struct RuntimeTrace { rank:u32, world:usize, barrier_ms:f64, detect_ms:Option<f64>, retransmits:u64, inflight_peak:usize }
pub struct CollHandle { id: u64, stream: StreamId }                 // returned by post(); await with wait()
pub trait AsyncCollectives {
    fn post_all_reduce<T>(&self, buf:&mut [T], op:ReduceOp, pg:&ProcessGroup) -> CollHandle;  // non-blocking (P9 overlap)
    fn wait(&self, h: CollHandle) -> Result<(),DtrError>;            // block until complete
}
pub struct JobConfig { world_size:usize, mesh: Vec<(String,usize)>, coordinator: SocketAddr,
                       algo: CollAlgo, deterministic: bool, heartbeat_ms: u64, window_bytes: usize }
pub struct LaunchSpec { hosts: Vec<String>, ranks_per_host: usize, binary: PathBuf, env: Vec<(String,String)> }
