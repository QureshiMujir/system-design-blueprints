Core schema & types (§11)

// ff-core::flag — ownership: control plane authors; SDK holds read-only copies.
pub struct FlagKey(String);                    // immutable, Eq+Hash, environment-scoped
pub struct Flag {                              // Clone+Send+Sync; immutable once compiled
    key: FlagKey,
    flag_type: FlagType,
    variants: Vec<Variant>,                    // ownership: control plane; read-only in SDK
    default_variant: VariantId,
    off_variant: VariantId,
    enabled: bool,
    targeting: Vec<Rule>,                      // ordered by priority, see §8.2
    rollout: Option<Rollout>,
    version: ConfigVersion,                    // serialization: protobuf field 7
}
pub enum FlagType { Boolean, String, Number, Json }  // compat: append-only enum (I7)
// ff-core::variant — immutable, Clone, serialized as part of Flag.
pub struct Variant { id: VariantId, name: String, value: Value }
pub struct VariantId(u16);                     // stable across edits; not re-used (§27)
pub struct VariantWeight(u32);                 // basis-points x10; sums to 100_000 (I13)
// ff-core::segment — mutation: control-plane only; concurrency: read-only in evaluator.
pub struct Segment {
    key: String,
    clauses: Vec<Clause>,                      // OR-combined (§8.2)
    included: Vec<UnitId>,
    excluded: Vec<UnitId>,
    version: ConfigVersion,
}
// ff-rules::ast — Clone+Send+Sync; pure data, no evaluation logic embedded.
pub struct Rule { clauses: Vec<Clause>, priority: i32, outcome: RuleOutcome }
pub enum RuleOutcome { FixedVariant(VariantId), Rollout(Rollout) }
pub struct Clause { attribute: String, op: Operator, values: Vec<Value>, negate: bool }
pub enum Operator {                             // append-only; unknown => non-match (I7)
    Eq, NotEq, In, Gt, Lt, Contains, StartsWith, EndsWith, Matches,
    SegmentMatch, SemVerEq, SemVerGt, SemVerLt, DateBefore, DateAfter,
}
// ff-core::context — constructed per-request by the host app; not persisted.
pub struct EvaluationContext { unit_id: UnitId, attributes: BTreeMap<String, Value> }
pub struct UnitId(String);                     // the randomization unit (ADR-07)
// ff-bucketing — pure functions only; no state; frozen per P8/I8.
pub struct BucketKey { salt: Salt, unit_id: UnitId }  // input tuple to the hash
pub struct Salt(String);                        // seed; changing it reshuffles (§8.3)
pub struct Rollout { bucket_by: String, variations: Vec<(VariantId, VariantWeight)>, salt: Salt }
// ff-core::result — returned to the host app; infallible surface (P4).
pub struct EvaluationResult { variant: VariantId, value: Value, reason: EvaluationReason }
pub enum EvaluationReason {                     // append-only; mirrors LaunchDarkly's taxonomy
    FlagOff, TargetMatch { rule_id: Uuid }, Rollout, Default, Error(EvalError),
}
pub enum EvalError { UnknownFlag, ConfigStale, SegmentCycleDetected, MalformedRule }
// ff-config — compiler output; serialization: protobuf (ADR-09), gzip on the wire.
pub struct ConfigDocument {
    version: ConfigVersion,
    environment_id: Uuid,
    flags: Vec<Flag>,
    segments: Vec<Segment>,
    compiled_at: SystemTime,
}
pub struct ConfigVersion(u64);                  // monotonic per environment (I11)
// ff-ingest — SDK-produced, queue-transported; serialization: protobuf.
pub struct ExposureEvent {
    timestamp: SystemTime, unit_id: UnitId, flag_key: FlagKey,
    experiment_id: Option<Uuid>, variant_id: VariantId,
    config_version: ConfigVersion, dedup_key: DedupKey,
}
pub struct DedupKey(u64);                       // hash of (unit_id, flag_key, window)
// ff-control-plane::experiment — mutation: control plane; concurrency: single-writer per row.
pub struct Experiment {
    key: String, flag_key: FlagKey, status: ExperimentStatus,
    primary_metric: MetricSpec, guardrail_metrics: Vec<MetricSpec>,
    analysis_method: AnalysisMethod, allocation_salt: Salt,
}
pub enum ExperimentStatus { Draft, Running, PeekingBlocked, Concluded, RolledOut, Reverted }
pub struct MetricSpec { event_name: String, metric_type: MetricType }
// ff-stats — computed, not authored; persisted to the results store.
pub struct StatsResult {
    experiment_key: String, per_variant: BTreeMap<VariantId, VarianceEstimate>,
    boundary: SequentialBoundary, srm_p_value: f64, decision_ready: bool,   // gates on I12
}
pub struct SequentialBoundary { lower: f64, upper: f64, alpha: f64, evaluated_at_n: u64 }
// ff-control-plane::audit — append-only (I10); mutation: insert-only role grant.
pub struct AuditEntry {
    id: i64, timestamp: SystemTime, actor: Actor, action: String,
    entity_type: String, entity_id: Uuid, diff: serde_json::Value,
    prev_hash: [u8; 32], entry_hash: [u8; 32],                  // hash-chained (§9.2)
}
pub enum Actor { User(Uuid), ApiKey(Uuid), System }
