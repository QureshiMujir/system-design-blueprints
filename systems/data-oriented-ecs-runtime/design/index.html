pub unsafe trait WorldQuery {
    type Item<'w>;
    type Fetch<'w>: Clone;

}
pub unsafe trait QueryData: WorldQuery { type ReadOnly: ReadOnlyQueryData<State = Self::State>; }

pub unsafe trait SystemParam: Sized {
    type State: Send + Sync + 'static;
    type Item<'w, 's>: SystemParam<State = Self::State>;
    fn init_state(world: &mut World, system_meta: &mut SystemMeta) -> Self::State;
    unsafe fn get_param<'w, 's>(state: &'s mut Self::State, system_meta: &SystemMeta,
        world: UnsafeWorldCell<'w>, change_tick: Tick) -> Self::Item<'w, 's>;
    fn apply(state: &mut Self::State, system_meta: &SystemMeta, world: &mut World); // default: no-op
}
