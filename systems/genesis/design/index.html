Core schema & types (§11)

struct GenomeRecord {
    hash: Blake3Hash,             // owner: genesis-store; immutable once written
    root: NodeId,                 // index into node arena
    arena: Vec<Node>,              // owner: genome; mutation: copy-on-write per operator
    node_types: Vec<Type>,         // parallel to arena; set by type-checker
    size: u32,                     // node count, used by parsimony (§21)
    depth: u16,
}
// concurrency: immutable after construction, shared via Arc across threads
// persistence: Cap'n Proto (ADR-04) in genesis-store; keyed by hash (P4)
// serialization: Cap'n Proto wire form; compatibility: additive-only schema evolution (§27)

enum Node {
    IntLit(i64),
    FloatLit(f64),
    BoolLit(bool),
    VarRef { name: Interned },
    Call { func: Interned, args: Vec<NodeId> },
    If { cond: NodeId, then_branch: NodeId, else_branch: NodeId },
    Let { bindings: Vec<(Interned, NodeId)>, body: NodeId },
}
// ownership: owned by GenomeRecord.arena; mutation: never mutated in place (P5)
// concurrency: Send+Sync (no interior mutability); serialization: Cap'n Proto union
enum Type { Int, Float, Bool, Fn(Vec<Type>, Box<Type>) }
// ownership: owned by GenomeRecord.node_types; persistence: derived, not stored (recomputed on load)

trait MutationOperator {
    fn name(&self) -> &'static str;
    fn apply(&self, parent: &GenomeRecord, rng: &mut GenesisRng) -> Option<GenomeRecord>;
    fn is_type_preserving(&self) -> bool;
}
// ownership: stateless, registered in an operator-rate table (P12); concurrency: Sync, invoked
// from worker-local threads (ADR-11); returns None on failed type-compatible target search (P5)

struct FitnessSpec {
    objectives: Vec<Objective>,
    constraints: Vec<Constraint>,
    spec_hash: Blake3Hash,          // content-addressed like genomes (P4)
}
struct Objective { name: Interned, weight: f64, cases: Vec<Case> }
struct Case { name: Interned, input: Vec<Value>, expected: Value, tolerance: f64 }
struct Constraint { name: Interned, kind: ConstraintKind }
enum ConstraintKind { MaxFuel(u64), MaxMemory(u64) }
// persistence: compiled once per run into an EvaluationPlan (I13); immutable for run lifetime (P12)
struct EvaluationPlan {
    spec_hash: Blake3Hash,
    compiled_cases: Vec<CompiledCase>,   // resolved export indices + expected values
    ceilings: ResourceCeilings,
}
// ownership: genesis-fitness; concurrency: Arc-shared read-only across all sandbox invocations

struct SandboxResult {
    exit: ExitTaxonomy,
    case_results: Vec<CaseResult>,
    fuel_consumed: u64,
    peak_memory_bytes: u64,
    wall_clock_ns: u64,
}
enum ExitTaxonomy { Ok, Trap, FuelExhausted, MemoryExceeded, WallClockTimeout, HostError }
// persistence: not durably stored per-candidate (I8 telemetry only); serialization: Protobuf (ADR-09)
// over the scheduler wire protocol between genesis-worker and genesis-api

struct SandboxHandle {
    id: SandboxId,
    engine: Arc<WasmtimeEngine>,   // shared, never per-instance (§13 pitfall)
    backend: SandboxBackend,        // Wasm | OciGvisor
}
// ownership: pool-managed (P10); mutation: none post-construction; concurrency: not Send across
// pool-checkout boundaries — one SandboxHandle serves exactly one in-flight evaluation
struct PopulationRecord {
    generation: u64,                 // monotonic (I12)
    members: Vec<Blake3Hash>,        // genome hashes, not inline genomes (P4)
    fitness: Vec<FitnessValue>,      // parallel to members
}
struct ArchiveCell {
    descriptor: BehaviorDescriptor,  // MAP-Elites grid coordinate (§20)
    occupant: Option<Blake3Hash>,
    occupant_fitness: FitnessValue,
}
// concurrency: archive writes are compare-and-swap gated by I9; persistence: RocksDB column
// family separate from the genome store (ADR-05); serialization: Protobuf snapshot (ADR-24)

struct RunConfig {
    seed: u64,
    population_size: u32,
    max_depth: u16,
    operator_rates: OperatorRateTable,  // versioned, immutable per-run (P12)
    resource_defaults: ResourceCeilings,
    replacement_policy: ReplacementPolicy,   // Generational | SteadyStateAge
}
// persistence: embedded verbatim in every checkpoint (§22); compatibility: additive fields only,
// unknown fields ignored on load per SemVer minor bump (§27)

enum GenesisError {
    TypeCheckReject { reason: TypeCheckFailReason },
    SandboxCeilingBreach { taxonomy: ExitTaxonomy },
    FitnessSpecMalformed { detail: String },
    LineageCycleDetected,
    CheckpointCorrupt,
    DuplicateJob { key: JobKey },
}
// ownership: propagated as Result<T, GenesisError> across crate boundaries; never panics across
// a sandbox trust boundary (P13); serialization: Protobuf error codes over the scheduler wire (ADR-09)
