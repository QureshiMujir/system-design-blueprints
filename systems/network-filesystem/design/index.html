Core schema & types (§11)

// FileHandle: opaque on the wire, structured internally (ADR-8).
struct FileHandle(Bytes);                       // ≤64B (v3) / ≤128B (v4)
struct FhInner { export_id: u16, fsid: u64, inode: u64, generation: u32 }
fn FileHandle::encode(inner: &FhInner) -> FileHandle;
fn FileHandle::decode(&self) -> Result<FhInner, RpcError>;   // validates len + magic

struct Inode {                                  // on-disk + cached
    ino: u64, gen: u32, kind: FileKind, mode: u32, nlink: u32,
    uid: u32, gid: u32, size: u64, blocks: u64, rdev: u64,
    atime: Timespec, mtime: Timespec, ctime: Timespec, fsid: u64,
}
enum FileKind { Reg, Dir, Symlink, Block, Char, Fifo, Socket }
struct Fattr3 { kind: FileKind, mode: u32, nlink: u32, uid: u32, gid: u32,
                size: u64, used: u64, rdev: (u32,u32), fsid: u64, fileid: u64,
                atime: Nfstime3, mtime: Nfstime3, ctime: Nfstime3 }
struct Sattr3 { mode: Option<u32>, uid: Option<u32>, gid: Option<u32>,
                size: Option<u64>, atime: SetTime, mtime: SetTime }
enum SetTime { DontChange, SetToServerTime, SetToClient(Nfstime3) }
struct WccData { before: Option<WccAttr>, after: Option<Fattr3> }
struct WccAttr { size: u64, mtime: Nfstime3, ctime: Nfstime3 }
struct Nfstime3 { seconds: u32, nseconds: u32 }
struct RpcMessage { xid: u32, body: RpcBody }
enum RpcBody { Call(RpcCall), Reply(RpcReply) }
struct RpcCall { rpcvers: u32, prog: u32, vers: u32, proc_: u32,
                 cred: OpaqueAuth, verf: OpaqueAuth /* args follow, decoded by proc */ }
enum RpcReply { Accepted { verf: OpaqueAuth, stat: AcceptStat },
                Denied(RejectStat) }
struct OpaqueAuth { flavor: AuthFlavor, body: Bytes /* ≤400B */ }
enum AuthFlavor { None, Sys, Short, RpcSecGss }
struct AuthSys { stamp: u32, machinename: String, uid: u32, gid: u32, gids: Vec<u32> /* ≤16 */ }
struct GssContext { handle: Bytes, seq_window: u32, service: GssService, // krb5/i/p
                    established: bool, peer_principal: String }
enum GssService { None, Integrity, Privacy }
struct DrcKey { client: ClientAddrOrId, prog: u32, vers: u32, proc_: u32, xid: u32 }
struct DrcEntry { state: DrcState, reply: Option<Bytes>, ts: Instant, size: usize }
enum DrcState { InProgress, Done }
struct Compound { tag: String, minorversion: u32, ops: Vec<NfsOp> }   // ops ≤ cap
enum NfsOp { Sequence(SeqArgs), PutFh(FileHandle), PutRootFh, GetFh, SaveFh, RestoreFh,
             Lookup(String), Access(u32), Getattr(AttrBitmap), Setattr(StateId, FattrSet),
             Read(StateId,u64,u32), Write(StateId,u64,StableHow,Bytes), Commit(u64,u32),
             Open(OpenArgs), Close(StateId,u32), Lock(LockArgs), Locku(LockArgs),
             Lockt(LockArgs), DelegReturn(StateId), Readdir(/*..*/), ExchangeId(/*..*/),
             CreateSession(/*..*/), DestroySession(SessionId), ReclaimComplete, /* ... */ }
struct StateId { seqid: u32, other: [u8; 12] }                  // I12
struct OpenState { stateid: StateId, owner: OpenOwner, share_access: u32, share_deny: u32,
                   fh: FileHandle, deleg: Option<Delegation> }
struct LockState { stateid: StateId, owner: LockOwner, ranges: Vec<ByteRange>, fh: FileHandle }
struct Lease { client: ClientId, expires: Instant, renew_period: Duration }   // I4
struct Delegation { stateid: StateId, kind: DelegKind, recall: RecallState }   // I15
enum DelegKind { Read, Write /* write deferred to PhaseDeferred */ }
enum RecallState { Granted, Recalling { since: Instant }, Returned }
struct SessionId([u8; 16]);
struct Slot { sequenceid: u32, cached: Option<Bytes>, in_use: bool }            // I10
struct Smb2Header { credit_charge: u16, status: u32, command: Smb2Command, credit_req: u16,
                    flags: u32, next_command: u32, message_id: u64, tree_id: u32,
                    session_id: u64, signature: [u8; 16] }
enum Smb2Command { Negotiate, SessionSetup, Logoff, TreeConnect, TreeDisconnect, Create,
                   Close, Flush, Read, Write, Lock, Ioctl, Cancel, Echo, QueryDirectory,
                   ChangeNotify, QueryInfo, SetInfo, OplockBreak }
struct TreeConnect { tree_id: u32, share: ShareName, access: u32, flags: u32 }
struct SmbOpen { file_id: FileId, fh: FileHandle, oplock: OplockLevel,
                 lease: Option<Lease3>, durable: Option<DurableHandle>, share_mode: u32 }
struct FileId { persistent: u64, volatile: u64 }
enum OplockLevel { None, II, Exclusive, Batch, Lease(LeaseState) }
struct Credits { granted: u32, sequence_window: Range<u64> }                    // I11
struct DurableHandle { guid: [u8;16], timeout: Duration, persistent: bool }
struct Vnode { ino: u64, fsid: u64, kind: FileKind, refcount: AtomicU32 }       // metadata lock per-inode
struct PathComponent(SmallString);                                              // single name, no '/'
struct Export { root: PathBuf, fsid: u64, export_id: u16, opts: ExportOpts }
struct ExportOpts { ro: bool, sec: Vec<SecFlavor>, root_squash: Squash,
                    all_squash: bool, anon_uid: u32, anon_gid: u32, subtree_check: bool }
enum Squash { RootSquash, NoRootSquash, AllSquash }
struct DirCookie { cookie: u64, verifier: [u8; 8] }                              // I7
enum NfsStat3 { Ok, Perm, Noent, Io, Acces, Exist, NotDir, IsDir, Inval, Fbig,
                Nospc, Rofs, NameTooLong, NotEmpty, Stale, BadHandle, BadCookie,
                NotSupp, ServerFault, Jukebox /* ... full §8.5 set */ }
enum Nfs4Err { Ok, Perm, Noent, /* ... */ Grace, NoGrace, StaleClientid, StaleStateid,
               OldStateid, BadStateid, Expired, BadSession, SeqMisordered,
               SeqFalseRetry, Resource, DelayDeleg, /* ... */ }
enum NtStatus { Success, AccessDenied, ObjectNameNotFound, ObjectNameCollision,
                SharingViolation, FileIsADirectory, InsufficientResources,
                NotSupported, Pending, /* ... MS-DTYP codes */ }
enum RpcError { GarbageArgs, ProgUnavail, ProgMismatch{lo:u32,hi:u32}, ProcUnavail,
                SystemErr, AuthError(AuthStat) }
