Core schema & types (§11)

#[derive(Copy, Clone, Debug, PartialEq)]
pub struct Point<const D: usize> {
    pub coords: [f64; D],
}
impl<const D: usize> Point<D> {
    pub fn new(coords: [f64; D]) -> Self;
    pub fn coord(&self, axis: usize) -> f64;
}
#[derive(Copy, Clone, Debug, PartialEq)]
pub struct Rect<const D: usize> {
    pub min: Point<D>,
    pub max: Point<D>,
}
impl<const D: usize> Rect<D> {
    pub fn new(min: Point<D>, max: Point<D>) -> Result<Self, GeomError>;
    pub fn degenerate(center: Point<D>) -> Self;   // min == max (zero-area point-rect)
    pub fn is_valid(&self) -> bool;                // min[i] <= max[i] for all i
}
pub type MBR<const D: usize> = Rect<D>;
#[derive(Copy, Clone, Debug, PartialEq, Eq, Hash)]
pub struct NodeId(pub u32);
impl NodeId {
    pub const NULL: NodeId;           // sentinel: 0xFFFF_FFFF
    pub fn is_null(self) -> bool;
}
#[derive(Copy, Clone, Debug, PartialEq, Eq, Hash, PartialOrd, Ord)]
pub struct DataId(pub u64);
pub fn euclidean<const D: usize>(a: Point<D>, b: Point<D>) -> f64;
pub fn mindist<const D: usize>(p: Point<D>, r: Rect<D>) -> f64;
pub fn minmaxdist<const D: usize>(p: Point<D>, r: Rect<D>) -> f64;
pub fn union<const D: usize>(a: Rect<D>, b: Rect<D>) -> Rect<D>;
pub fn intersection<const D: usize>(a: Rect<D>, b: Rect<D>) -> Option<Rect<D>>;
pub fn intersects<const D: usize>(a: Rect<D>, b: Rect<D>) -> bool;
pub fn area<const D: usize>(r: Rect<D>) -> f64;
pub fn enlargement<const D: usize>(base: Rect<D>, insert: Rect<D>) -> f64;
pub fn overlap_area<const D: usize>(a: Rect<D>, b: Rect<D>) -> f64;
pub fn contains_rect<const D: usize>(outer: Rect<D>, inner: Rect<D>) -> bool;
pub fn contains_point<const D: usize>(r: Rect<D>, p: Point<D>) -> bool;
#[derive(Clone, Debug)]
pub struct Entry<const D: usize> {
    pub mbr: Rect<D>,
    pub child: Child,
}
#[derive(Clone, Debug, PartialEq)]
pub enum Child {
    Node(NodeId),
    Data(DataId),
}
#[derive(Clone, Debug)]
pub struct RTreeNode<const D: usize> {
    pub level: u16,          // 0 = leaf; >0 = interior at given height
    pub entries: Vec<Entry<D>>,
}
impl<const D: usize> RTreeNode<D> {
    pub fn is_leaf(&self) -> bool;      // level == 0
    pub fn is_overflow(&self, m_cap: usize) -> bool;   // entries.len() > m_cap
    pub fn is_underflow(&self, m_min: usize) -> bool;  // entries.len() < m_min
}
pub struct Arena<T> {
    nodes: Vec<Option<T>>,
}
impl<T> Arena<T> {
    pub fn alloc(&mut self, val: T) -> NodeId;
    pub fn get(&self, id: NodeId) -> Option<&T>;
    pub fn get_mut(&mut self, id: NodeId) -> Option<&mut T>;
    pub fn free(&mut self, id: NodeId);
}
pub struct RTree<const D: usize> {
    pub arena: Arena<RTreeNode<D>>,
    pub root: NodeId,
    pub height: u16,
    pub m_min: usize,    // minimum entries per non-root node
    pub m_max: usize,    // maximum entries per node
    pub split: SplitStrategy,
    pub size: usize,     // total number of Data entries indexed
}
impl<const D: usize> RTree<D> {
    pub fn new(m_min: usize, m_max: usize, split: SplitStrategy) -> Self;
    pub fn insert(&mut self, mbr: Rect<D>, id: DataId);
    pub fn delete(&mut self, mbr: Rect<D>, id: DataId) -> bool;
    pub fn window_search(&self, query: Rect<D>) -> Vec<DataId>;
    pub fn height(&self) -> u16;
    pub fn len(&self) -> usize;
}
#[derive(Copy, Clone, Debug, PartialEq, Eq)]
pub enum SplitStrategy {
    Quadratic,   // O(M^2) pick-seeds; Guttman 1984
    Linear,      // O(M) pick-seeds; Guttman 1984
    RStar,       // axis/margin/overlap distribution; Beckmann 1990
}
pub struct RStar<const D: usize> {
    inner: RTree<D>,
    reinsert_flag: Vec<bool>,   // per-level: forced reinsertion already done this insert path
}
impl<const D: usize> RStar<D> {
    pub fn new(m_min: usize, m_max: usize) -> Self;
    pub fn insert(&mut self, mbr: Rect<D>, id: DataId);
    pub fn delete(&mut self, mbr: Rect<D>, id: DataId) -> bool;
    pub fn window_search(&self, query: Rect<D>) -> Vec<DataId>;
}
fn choose_subtree<const D: usize>(node: &RTreeNode<D>, entry_mbr: Rect<D>, level: u16) -> usize;
fn rstar_split<const D: usize>(node: &mut RTreeNode<D>, new_entry: Entry<D>, m_min: usize)
    -> RTreeNode<D>;
fn forced_reinsert<const D: usize>(tree: &mut RStar<D>, node_id: NodeId, level: u16);
#[derive(Copy, Clone, Debug)]
pub struct BulkLoad {
    pub leaf_capacity: usize,           // M for leaf nodes
    pub node_capacity: usize,           // M for interior nodes (often same as leaf)
    pub m_min: usize,                   // minimum occupancy after bulk load
}
impl BulkLoad {
    pub fn slice_count(n: usize, m: usize) -> usize;  // ceil(sqrt(n / m))
}
pub fn str_bulk_load<const D: usize>(
    data: &mut [(Rect<D>, DataId)],
    params: BulkLoad,
) -> RTree<D>;
#[derive(Clone, Debug)]
pub struct QuadNode<const D: usize> {
    pub bounds: Rect<D>,
    pub kind: QuadNodeKind<D>,
    pub depth: u32,
}
#[derive(Clone, Debug)]
pub enum QuadNodeKind<const D: usize> {
    Leaf(Vec<(Point<D>, DataId)>),
    Internal([NodeId; 4]),             // NW, NE, SW, SE (D=2)
}
pub struct Quadtree<const D: usize> {
    pub arena: Arena<QuadNode<D>>,
    pub root: NodeId,
    pub capacity: usize,   // leaf split threshold
    pub max_depth: u32,
    pub size: usize,
}
impl<const D: usize> Quadtree<D> {
    pub fn new(bounds: Rect<D>, capacity: usize, max_depth: u32) -> Self;
    pub fn insert(&mut self, p: Point<D>, id: DataId);
    pub fn point_query(&self, p: Point<D>) -> Vec<DataId>;
    pub fn range_query(&self, query: Rect<D>) -> Vec<DataId>;
}
#[derive(Copy, Clone, Debug, PartialEq, Eq, Hash)]
pub struct GeoHash {
    pub bits: u64,          // interleaved longitude/latitude bits (up to 60 bits used)
    pub precision: u8,      // number of base32 characters (1..=12)
}
impl GeoHash {
    pub fn encode(lat: f64, lng: f64, precision: u8) -> Result<Self, GeohashError>;
    pub fn decode(&self) -> Rect<2>;              // cell bounding box (I8)
    pub fn to_string(&self) -> String;            // base32 string of length precision
    pub fn from_str(s: &str) -> Result<Self, GeohashError>;
    pub fn neighbors(&self) -> [GeoHash; 8];      // N,NE,E,SE,S,SW,W,NW
    pub fn prefix_range(&self) -> (String, String); // [prefix+"0", prefix+"z"] inclusive
    pub fn is_prefix_of(&self, longer: &GeoHash) -> bool;
}
#[derive(Clone, Debug, PartialEq)]
pub struct KnnItem {
    pub dist: f64,
    pub payload: DataId,
}
impl PartialOrd for KnnItem { /* ordered by dist ascending */ }
enum KnnEntry {
    Node { dist: f64, id: NodeId },
    Data { dist: f64, id: DataId },
}
pub fn knn_search<const D: usize>(
    tree: &RTree<D>,
    query: Point<D>,
    k: usize,
) -> Vec<KnnItem>;
#[derive(Clone, Debug)]
pub enum Query<const D: usize> {
    Window(Rect<D>),          // return objects whose MBR intersects query rect
    Intersect(Rect<D>),       // alias for Window (intersection semantics explicit)
    Contains(Rect<D>),        // return objects whose MBR is fully contained within query rect
    Knn(Point<D>, usize),     // k nearest objects by euclidean distance
    Nearest(Point<D>),        // single nearest object (Knn with k=1)
}
#[derive(Clone, Debug)]
pub enum QueryResult {
    Ids(Vec<DataId>),
    Ranked(Vec<KnnItem>),
}
pub trait SpatialIndex<const D: usize> {
    fn execute(&self, query: &Query<D>) -> QueryResult;
}

pub const PAGE_SIZE: usize = 4096;   // fixed; must be power of 2

#[derive(Copy, Clone, Debug, PartialEq, Eq, Hash)]
pub struct PageId(pub u32);

pub struct Page {
    pub data: [u8; PAGE_SIZE],
}
pub struct PagedRTree<const D: usize> {
    path: PathBuf,
    pager: Pager,
    cache: LruCache<PageId, Page>,
    root: PageId,
    m_min: usize,
    m_max: usize,
}
impl<const D: usize> PagedRTree<D> {
    pub fn create(path: &Path, m_min: usize, m_max: usize) -> Result<Self, PersistError>;
    pub fn open(path: &Path) -> Result<Self, PersistError>;
    pub fn insert(&mut self, mbr: Rect<D>, id: DataId) -> Result<(), PersistError>;
    pub fn window_search(&mut self, query: Rect<D>) -> Result<Vec<DataId>, PersistError>;
}
pub struct Oracle<const D: usize> {
    pub records: Vec<(Rect<D>, DataId)>,
}
impl<const D: usize> Oracle<D> {
    pub fn insert(&mut self, mbr: Rect<D>, id: DataId);
    pub fn delete(&mut self, mbr: Rect<D>, id: DataId) -> bool;
    pub fn window(&self, q: Rect<D>) -> Vec<DataId>;
    pub fn intersect(&self, q: Rect<D>) -> Vec<DataId>;
    pub fn contains(&self, q: Rect<D>) -> Vec<DataId>;
    pub fn knn(&self, p: Point<D>, k: usize) -> Vec<KnnItem>;
}
