Core schema & types (§11)

/// Dense-vector handle; index of a row in VectorStore. Stable across save/mmap.
/// Own: Copy POD. Mut: value. Conc: Send+Sync+Copy. Persist: u32 LE. Compat: width FROZEN at u32.
pub struct VectorId(pub u32);

/// Metric selector. Cosine assumes L2-normalized vectors (normalized at VectorStore::push).
/// Chosen at build, written to header, checked on open (search metric MUST equal build metric).
/// Own: Copy. Conc: Send+Sync. Persist: 1-byte tag {0=L2,1=InnerProduct,2=Cosine}.
pub enum Metric { L2, InnerProduct, Cosine }

/// Scalar distance; SMALLER = CLOSER. InnerProduct/Cosine = negated similarity so one min-heap
/// orders all metrics. Own: Copy. Ord: f32::total_cmp (finite). Invariant: never NaN (rejected upstream).
pub struct Distance(pub f32);

/// Owning, row-major, contiguous, 32-byte-aligned f32 store. data.len() == dim*len.
/// Own: owns Vec<f32>; the single source of coordinates. Mut: append-only via push() during build,
/// immutable during search. Conc: &VectorStore Send+Sync after freeze. Persist: header{dim,len} +
/// aligned f32 payload mmap'd zero-copy. Compat: layout FROZEN.
pub struct VectorStore { dim: usize, data: Vec<f32>, len: u32 }
impl VectorStore {
    pub fn get(&self, id: VectorId) -> &[f32];     // &data[id*dim .. id*dim+dim], no copy
    pub fn push(&mut self, v: &[f32]) -> VectorId; // normalizes if Cosine; build phase only
    pub fn dim(&self) -> usize; pub fn len(&self) -> u32;
}

/// SIMD-or-scalar distance fn, selected once at startup via runtime CPU feature dispatch (Phase 1).
/// Pure, alloc-free, stateless. Differential gate: |simd - scalar| <= 1e-5 rel error.
/// Conc: Send+Sync. Kernels: scalar + AVX2/AVX-512/NEON L2/InnerProduct/Cosine.
pub trait DistanceKernel { fn distance(&self, a: &[f32], b: &[f32]) -> Distance; }

/// Distance-computations budget tracker; one inc per kernel call. AtomicU64 -> Sync.
/// Primary cost metric for CostModel + Autotuner (not wall-clock alone).
pub struct DistanceCounter { count: AtomicU64 }
/// One node's out-edges: ids + parallel dists, sorted ASC by dist, capped at R.
/// WHY sorted: prune/heuristic scan stops early; neighbors[0] is closest. WHY capped: bounds
/// out-degree -> bounds dist-comps/hop -> bounds query cost. SmallVec inlines common small degree.
/// Own: owns ids/dists. Mut: &mut during build/prune; len <= R. Conc: per-node-locked on concurrent
/// insert (Phase 10); read-shared once frozen.
pub struct NeighborList { ids: SmallVec<[VectorId; R]>, dists: SmallVec<[Distance; R]> }

/// Mutable build form: one NeighborList per VectorId (index = id). Resized during insert/optimize.
pub struct AdjacencyList { neighbors: Vec<NeighborList> }

/// One HNSW layer (level 0 = base, all nodes). Vamana uses a single Layer{level:0}.
pub struct Layer { adjacency: AdjacencyList, level: u8 }

/// MUTABLE graph: the build/optimize working form. Mutated by HnswBuilder/VamanaBuilder and ALL
/// ann-optimize passes. Own: owns layers. Mut: single-writer &mut. Conc: NOT Sync for writes.
/// Persist: header{kind,metric,max_degree,entry_point} + per-layer adjacency.
pub struct Graph {
    layers: Vec<Layer>,   // layers[0] base; HNSW only is multi-layer
    entry_point: VectorId,
    max_degree: u32,      // R / m_max0 — the degree-invariant ceiling
    metric: Metric,
    kind: GraphKind,
}

/// Per-family build constants carried inside the graph. ml = 1/ln(m).
pub enum GraphKind {
    Hnsw   { m: u32, m_max0: u32, ef_construction: u32, ml: f32 },
    Vamana { r: u32, alpha: f32, l: u32 },
}

/// IMMUTABLE optimized form: CSR adjacency. edges[offsets[i]..offsets[i+1]] = out-neighbors of i
/// (sorted). offsets.len()==N+1. Cache-dense contiguous scan; THE search-time structure.
/// Own: owns both Vecs. Mut: NONE after freeze (rebuild to change topology). Conc: Send+Sync (read-only)
/// — the concurrent-search substrate. Persist: offsets:Vec<u32> + edges:Vec<VectorId> mmap'd zero-copy.
/// Compat: CSR layout FROZEN.
pub struct CsrGraph { offsets: Vec<u32>, edges: Vec<VectorId>, entry_point: VectorId }
impl CsrGraph {
    pub fn from_graph(g: &Graph) -> CsrGraph;        // freeze: flatten base layer to CSR
    pub fn neighbors(&self, id: VectorId) -> &[VectorId];
}
/// All build knobs; one seed for deterministic level assignment (differential testing).
/// Own: Copy. Conc: Send+Sync. Persist: recorded in graph header for provenance.
pub struct BuildParams { m: u32, m_max0: u32, ef_construction: u32, alpha: f32, r: u32, l: u32, seed: u64 }

/// HNSW Algorithm 4. candidates sorted ASC by dist to q. Keep e iff d(e,q) < d(e,r) for every already
/// selected r (e closer to q than to any kept neighbor) -> diverse, navigable set vs naive nearest-M.
/// extend = consider neighbors-of-candidates; keep_pruned = backfill to m. Returns capped sorted list.
pub fn select_neighbors_heuristic(
    candidates: &[(VectorId, Distance)], m: u32, extend: bool, keep_pruned: bool,
) -> NeighborList;

/// Vamana RobustPrune. candidates = visited pool from greedy search. Loop: pop closest p*; edge p->p*;
/// drop every v with alpha*d(p*,v) <= d(p,v) (occlusion, relaxation alpha>=1); stop at R or empty.
pub fn robust_prune(
    p: VectorId, candidates: &mut Vec<(VectorId, Distance)>, alpha: f32, r: u32, store: &VectorStore,
) -> NeighborList;
pub struct SearchParams { k: u32, ef: u32, beam_width: u32 } // ef>=k; beam_width=1 = pure greedy

/// Versioned bitset: O(1) reset between queries (bump epoch, no zeroing). Per-query mutable.
pub struct VisitedSet { /* Vec<u32> epoch stamps + current epoch */ }

/// Per-query scratch. REUSED across queries on a thread -> zero-heap-alloc steady state (clear+reuse).
/// candidates = min-heap (Reverse) frontier (expand closest first); results = bounded max-heap of ef/k.
/// Own: owns its three structures. Mut: &mut per query. Conc: NOT Sync — one per thread/searcher.
pub struct SearchContext {
    visited: VisitedSet,
    candidates: BinaryHeap<Reverse<(Distance, VectorId)>>,
    results: BinaryHeap<(Distance, VectorId)>,
}
pub struct SearchResult { pub id: VectorId, pub distance: Distance }

/// Layer-aware greedy over mutable Graph (build-time + HNSW search). Working list size ef, seeded at ep.
pub fn greedy_search(g: &Graph, q: &[f32], ep: VectorId, ef: u32, ctx: &mut SearchContext) -> Vec<SearchResult>;

/// Production search over frozen CsrGraph. Expand closest unexpanded; stop when closest unexpanded >
/// worst-in-W; return top-k. &CsrGraph Sync => N threads, N SearchContexts, zero contention.
pub fn beam_search(g: &CsrGraph, q: &[f32], params: &SearchParams, ctx: &mut SearchContext) -> Vec<SearchResult>;

/// Exact-KNN ground truth for recall@k. ground_truth[i] = true top-k of query i (brute force).
/// Own: owns the truth set. Conc: Send+Sync (read). Used only by recall/tune/bench, never a query path.
pub struct RecallHarness { ground_truth: Vec<Vec<VectorId>> }
impl RecallHarness { pub fn recall_at_k(&self, results: &[Vec<SearchResult>], k: u32) -> f32; }
/// Full graph-quality report (Phase 7). Every field normative:
pub struct GraphDiagnostics {
    mean_out_degree: f32,         // avg |out(v)|
    out_degree_hist: [u32; 64],   // bucketed out-degree distribution
    min_in_degree: u32,           // lowest reverse-edge count (orphan risk)
    orphan_count: u32,            // nodes with in-degree == 0
    connected_components: u32,    // weak components; healthy graph == 1
    unreachable_from_ep: u32,     // nodes not reachable from entry_point via directed edges
    navigability_score: f32,      // sampled expected-hops-to-convergence proxy (lower = better)
    estimated_recall_at_10: f32,  // sampled queries vs RecallHarness exact KNN at fixed ef
}

/// Edge-selection strategy for re-pruning. RobustPrune = alpha-occlusion (keeps long-range edges);
/// HeuristicSelect = HNSW occlusion rule (keep_pruned backfill); RelativeNeighborhood = RNG (alpha=1 limit).
/// Own: Copy. Conc: Send+Sync.
pub enum PruneStrategy { RobustPrune { alpha: f32 }, HeuristicSelect { keep_pruned: bool }, RelativeNeighborhood }

pub fn occlusion_prune(g: &mut Graph, strat: PruneStrategy, store: &VectorStore); // re-prune for navigability
pub fn degree_rebalance(g: &mut Graph, r: u32);                  // enforce out-degree <= r, drop weakest
pub fn reverse_edge_repair(g: &mut Graph, min_in_degree: u32);   // add reverse edges to orphan/low-in-degree
pub fn connect_components(g: &mut Graph, store: &VectorStore);    // splice unreachable subgraphs into EP component
pub fn reselect_entry_point(g: &Graph, store: &VectorStore) -> VectorId; // medoid/centrality choice

/// Predicts cost without running: mean dist-comps/query ~ ef * mean_out_degree * (1 + hop_factor).
pub struct CostModel { /* fitted hop_factor + degree term */ }
pub struct RecallTarget { pub k: u32, pub recall: f32 }          // e.g. recall@10 >= 0.95

/// Bisection over ef to the smallest ef meeting RecallTarget, then grid over (alpha,R,L).
/// Returns the recall/cost Pareto frontier. Conc: trials run in parallel over the immutable index.
pub struct Autotuner { /* search state */ }
pub struct TuneResult { pub params: SearchParams, pub recall: f32, pub mean_dist_comps: f32, pub p99_us: f32 }
/// Trained PQ: dim split into m_subspaces subspaces, ksub centroids each (ksub=256 -> 1 byte/code).
/// codebooks = flat [m_subspaces*ksub*sub_dim] f32 arena. Asymmetric distance via per-query LUT (no decode).
/// Own: owns codebooks. Mut: immutable after train. Conc: &ProductQuantizer Send+Sync (shared read).
/// Persist: header{m_subspaces,ksub} + codebook payload mmap-friendly. Compat: layout FROZEN.
pub struct ProductQuantizer { m_subspaces: u32, ksub: u32, codebooks: Vec<f32> }

pub struct Tombstone { pub id: VectorId }  // logical delete marker
/// Packed deleted-id set; filtered out of results pre-rerank. Own: owns bits. Conc: read-shared by
/// searchers, flipped by the writer. Persist: bitfield over n nodes.
pub struct DeleteSet { bits: BitVec }

/// Serialize frozen CsrGraph + vectors. Header: magic, version, metric, dim, kind, R, crc32.
/// Atomic: write temp + fsync + rename. Compat: version-mismatch / metric / dim mismatch => Err.
pub fn save_graph(g: &CsrGraph, store: &VectorStore, path: &Path) -> io::Result<()>;
/// Zero-copy mmap load; MappedGraph derefs to a CsrGraph view. Verifies magic+version+crc first.
pub fn mmap_graph(path: &Path) -> io::Result<MappedGraph>;
