Core schema & types (§11)

// fusion-ir: identifiers and tensor description
struct TensorId(u64);                  // immutable, Copy, stable across a compile
struct Tensor { id: TensorId, dtype: DType, shape: Shape, layout: Layout }
enum DType { F32, F16, BF16 }          // Copy, no interior mutability
struct Shape(Vec<DimExpr>);            // owned, immutable after graph validation
enum DimExpr { Static(u64), Symbolic(SymbolId) } // symbolic deferred to PhaseDeferred("Phase 16")
struct Layout { strides: Vec<i64>, offset: i64 } // immutable, no aliasing across Tensors
// fusion-ir: operation graph
struct OpGraph { nodes: Vec<Op>, edges: Vec<(TensorId, TensorId)> } // owned, single-writer during build, immutable after
struct Op { id: OpId, kind: OpKind, inputs: Vec<TensorId>, outputs: Vec<TensorId> }
enum OpKind { Elementwise(ElemFn), Broadcast, Reduce(ReduceOp), Contract(ContractDims) }
struct IterationSpace { dims: Vec<DimExpr>, order: Vec<usize> } // Copy-ish, compared structurally by LegalityChecker
// fusion-legality / fusion-planner
struct FusionGroup { id: FusionGroupId, ops: Vec<OpId>, boundary: IterationSpace } // owned by FusionPlan, no shared mutation
struct FusionPlan { groups: Vec<FusionGroup>, rejected: Vec<(FusionGroup, RejectReason)> } // persisted for telemetry (I12)
enum LegalityVerdict { Legal, Rejected(RejectReason) }
enum RejectReason { IncompatibleIterationSpace, RaceyReduction, DtypeMismatch, IllegalContractBoundary }
// fusion-costmodel
struct MemoryTraffic { unfused_bytes: u64, fused_bytes: u64 } // Copy, pure computed value, not persisted independently
struct ArithmeticIntensity(f64)        // flops / bytes, Copy
struct CostModel;                      // stateless; methods take &FusionGroup, &HardwareDescriptor
struct Occupancy { active_warps_per_sm: u32, max_warps_per_sm: u32 } // Copy, derived value
// fusion-schedule
struct Schedule { tiling: Tiling, launch: LaunchConfig, shared_mem: SharedMemPlan } // owned by a Kernel-in-progress, immutable once Generated
struct Tiling { block_m: u32, block_n: u32, block_k: Option<u32> } // Copy
struct TileConfig { tiling: Tiling, unroll: u32, vector_width: u32 } // Copy, hashable (autotune search key)
struct LaunchConfig { grid: (u32, u32, u32), block: (u32, u32, u32), dynamic_shared_bytes: u32 } // Copy
struct SharedMemPlan { bytes_used: u32, budget_bytes: u32 }        // Copy, validated against I4 pre-codegen
// fusion-codegen
struct Kernel { id: KernelId, source: KernelSource, schedule: Schedule } // immutable after Generated state
enum KernelSource { CudaC(String), Ptx(PtxModule) }
struct PtxModule { version: PtxVersion, target_arch: String, body: String } // serializable, versioned (§27)
// fusion-autotune / fusion-verify
struct Autotuner;                      // stateless driver; persists results via TuningConfig
struct TuningConfig { tile: TileConfig, launch: LaunchConfig, seed: u64, hw_descriptor: HardwareDescriptorId } // I6: reproducible
struct VerificationResult { max_abs_error: f64, max_rel_error: f64, within_tolerance: bool } // not persisted beyond CI report
struct Tolerance { max_abs: f64, max_rel: f64 } // Copy, dtype-dependent default (§4)
// fusion-cache / error taxonomy
struct CompileCache { root: PathBuf }  // single-writer per process; content-addressed keys
struct CacheKey { graph_hash: Hash256, schedule_hash: Hash256, hw_descriptor: HardwareDescriptorId } // Copy, Eq, Hash
enum FusionError {
    Legality(RejectReason),
    Budget(BudgetOverflow),
    Codegen(CodegenError),
    Verification(VerificationResult),
    Cache(CacheError),
}
