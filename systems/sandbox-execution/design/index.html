Core schema & types (§11)

pub enum Tier { Process, Wasm, MicroVm, UserKernel }
pub struct SandboxId(pub u64);
pub struct RunId(pub u64);
pub struct Capability(u64); // bitflags over 20 POSIX capabilities(7); see §11 table
pub struct Budget {
    pub wall_clock: Duration, pub cpu_time: Duration, pub memory_bytes: u64,
    pub pids: u32, pub fds: u32, pub io_read_bps: u64, pub io_write_bps: u64,
    pub net_bps: u64, pub stdout_bytes: u64, pub stderr_bytes: u64, pub disk_bytes: u64,
}
pub struct ImageRef { pub content_hash: [u8; 32], pub signature: Option<[u8; 64]>, pub registry_ref: Option<String> }
pub struct RunSpec {
    pub tier: Tier, pub image: ImageRef, pub argv: Vec<String>, pub env: Vec<(String, String)>,
    pub cwd: String, pub uid: u32, pub gid: u32, pub mounts: Vec<MountSpec>,
    pub network: NetworkSpec, pub capabilities: Capability, pub budget: Budget,
    pub seccomp: SeccompPolicy, pub audit: AuditPolicy, pub timeout_kill_signal: i32,
}
pub enum MountSpec {
    Bind { src: PathBuf, dst: PathBuf, ro: bool, recursive: bool },
    Tmpfs { dst: PathBuf, size: u64, mode: u32 },
    Overlay { lower: PathBuf, upper: PathBuf, work: PathBuf, dst: PathBuf },
    DevTmpfs { dst: PathBuf }, Proc { dst: PathBuf }, Sys { dst: PathBuf, ro: bool },
    FsCrypt { dst: PathBuf, key: [u8; 32] }, FsVerity { dst: PathBuf, root_hash: [u8; 32] },
}
pub enum NetworkSpec {
    None, Slirp { dns: IpAddr, mtu: u16 }, Veth { bridge: String, ip4: Ipv4Addr, mtu: u16 },
    MacVlan { parent: String, ip4: Ipv4Addr }, Vsock { cid: u32 },
}
pub enum Action { Allow, Errno(i32), KillProcess, KillThread, Trap, Notify, Log }
pub struct Rule { pub syscall_nr: i32, pub args: [Option<ArgFilter>; 6], pub action: Action }
pub struct SeccompPolicy { pub default_action: Action, pub rules: Vec<Rule>, pub notify_fd: Option<OwnedFd> }
pub struct Outcome {
    pub run_id: RunId, pub exit: ExitStatus, pub wall_clock: Duration, pub cpu_time: Duration,
    pub peak_rss_bytes: u64, pub stdout_truncated: bool, pub stderr_truncated: bool,
    pub stdout: Bytes, pub stderr: Bytes, pub audit: AuditChain,
}
pub enum ExitStatus { Exited(i32), Signaled(i32), KilledByBudget(BudgetReason), KilledByPolicy(PolicyReason), Crashed }
pub trait TierImpl: Send + Sync + 'static {
    fn name(&self) -> Tier;
    async fn prepare(&self, spec: &RunSpec) -> Result<PreparedSandbox, Error>;
    async fn run(&self, prep: PreparedSandbox) -> Result<Outcome, Error>;
    async fn shutdown(&self) -> Result<(), Error>;
}
pub struct PreparedSandbox { pub id: SandboxId, pub tier: Tier /* + tier-private handle, opaque to sandbox-runtime */ }
pub enum AuditRecordKind { LifecycleStart, PrepareComplete, PrepareFailed, MountAttempt,
    SeccompDeny, BudgetBurn, RunStart, LifecycleEnd, ForceKilled, AuditFinalized }
pub struct AuditRecord { pub seq: u64, pub prev_hash: [u8; 32], pub timestamp_ns: u64,
    pub kind: AuditRecordKind, pub payload: Bytes, pub signature: [u8; 64] }
pub struct AuditChain { pub records: Vec<AuditRecord> }
pub enum Error { ImageNotFound, BudgetBelowTierFloor { tier: Tier, floor: u64, requested: u64 },
    SeccompOrderViolation, NamespaceOrderViolation, KvmUnavailable, ImageSignatureInvalid,
    UnprivilegedUsernsDisabled, AuditIntegrityUnknown, OverlayUnsupported }
