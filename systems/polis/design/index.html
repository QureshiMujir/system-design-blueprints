Core schema & types (§11)

Agent {
  id:                   AgentId           // stable for the run; owned by Orchestrator
  cohort:               CohortId          // sub-population analytics key
  psychology:           PsychologyProfile // immutable except via logged mutation (I5)
  memory_ref:           MemoryRef         // pointer to MemoryAdapter store
  action_repertoire:    Set[ActionType]   // closed set, enforced at ACT (I1)
  communication:        ProtocolSpec
  mortality:            MortalitySpec
  spawn_tick:           Tick
  status:               { ALIVE, DORMANT, DEAD }  // §8.3 state machine
  metadata:             Map[str, JSON]
}
// ownership: Orchestrator; mutation: append-only status transitions + logged psychology
// mutation; concurrency: single-writer per shard; persistence: ClickHouse action log +
// TimescaleDB snapshot; serialization: Pydantic v2 model; compatibility: additive-only v1+.
PsychologyProfile {
  big_five:             { O: float, C: float, E: float, A: float, N: float }  // 0..1
  cultural_overlay:     CultureId
  values:               Map[ValueAxis, float]
  beliefs:              List[BeliefAtom]
  memory_architecture:  { capacity: int, recency_bias: float,
                           episodic_share: float, forgetting_curve: FunctionRef }
  decision_policy:      PolicyRef          // LLM + prompt-scaffold identifier
  mutability:           { IMMUTABLE, EVENT_TRIGGERED, GRADUAL }
  provenance:           ProfileProvenance  // synthetic | aggregated | licensed
}
// ownership: Agent Pool; mutation: only via PsychologyMutation event (I5); concurrency:
// read-shared across shard workers, copy-on-mutation; persistence: object storage,
// content-hashed; serialization: Pydantic v2; compatibility: additive-only v1+.
Environment {
  id:                   EnvId
  geography:            GeographyLayer
  resources:            ResourceLayer
  institutions:         List[Institution]
  technology_epoch:     EpochId
  exogenous_schedule:    List[ExogenousEvent]
  dsl_source:            Hash             // content hash of the DSL document (ADR-18)
}
// ownership: Environment Runtime; mutation: only via tick(actions, rng) → EnvironmentDelta;
// concurrency: single-writer per environment per tick; persistence: object storage +
// TimescaleDB snapshot; serialization: YAML + JSON Schema; compatibility: additive-only v1+.
Event {
  scenario_id:          ScenarioId
  run_id:               RunId
  tick:                 Tick
  phase:                { PERCEIVE, DELIBERATE, ACT, COMMUNICATE, WRITE }
  seq:                  int                // total order within (run, tick, phase), I2
  actor:                AgentId | "ENV" | "ORCH"
  type:                 EventType
  payload:              JSON
  caused_by:            List[EventId]      // causal DAG edges, §8.4
}
// ownership: determinism log (orchestrator/determinism); mutation: append-only, never
// mutated post-write; concurrency: append-only log, one writer per shard, merged in
// canonical order; persistence: ClickHouse (hot) + object storage (cold archive);
// serialization: Kafka-style log record; compatibility: additive-only v1+ (I11).
Relationship {
  a:                    AgentId
  b:                    AgentId
  kind:                 { KIN, ALLY, RIVAL, SPOUSE, MENTOR }
  strength:             float              // -1..1
  history_ref:          List[EventId]
  first_seen_tick:      Tick
  last_updated_tick:    Tick
}
// ownership: derived (a reducer over Event, §8.1); mutation: recomputed per tick, never
// hand-edited; concurrency: read-only view for Studio/Lens; persistence: TimescaleDB;
// serialization: Pydantic v2; compatibility: additive-only v1+.
EmergentMetric {
  id:                   MetricId
  scope:                { GLOBAL, COHORT, REGION }
  reducer:              ReducerSpec
  window:               { tick_window: int, sliding: bool }
  output_type:          { SCALAR, HISTOGRAM, GRAPH, CORPUS }
  publish_rate:         { every_tick, every_n_ticks, on_change }
}
// ownership: Emergent Metrics Bus; mutation: reducer re-evaluated per window; concurrency:
// stateless reducer, safely parallel across metrics; persistence: ClickHouse; serialization:
// Pydantic v2; compatibility: additive-only v1+.
Scenario {
  id:                   ScenarioId
  parent:               ScenarioId | null  // fork DAG, I12
  population_template:  PopulationTemplateRef
  environment:          Environment
  initial_relationships: List[Relationship]
  initial_psychology_distribution: PsychologyDistribution
  scripted_interventions: List[ScheduledIntervention]
  rng_seed:             int
  model_pins:           List[ModelArtefactHash]
  budget:               { agent_hours_cap: int, sim_ticks_cap: int }
  ethics_gate:          EthicsGateResult
  fork_diff:            JSONPatch | null
}
// ownership: Scenario Forking Engine; mutation: immutable once dispatched (content-hashed);
// concurrency: creation is single-writer per fork request (I12 cycle check); persistence:
// object storage, hash-addressable; serialization: YAML + JSON blobs; compatibility:
// additive-only v1+.
EthicsGateResult {
  determination:        { PASS, PASS_WITH_CONDITIONS, HOLD_FOR_REVIEW, REJECT }
  triggers:              Set[GateTrigger]  // population/mass-death/persuasion/... (§22)
  reviewer:              ReviewerId | "AUTOMATED"
  conditions:            List[str]
  decided_at_tick:       Tick | null
}
// ownership: Control Plane ethics gate; mutation: immutable once decided; concurrency:
// single-decision-per-scenario; persistence: object storage (attached to Scenario);
// serialization: Pydantic v2; compatibility: additive-only v1+.
MemoryAdapter (interface) {
  read(query: Query, k: int) -> List[MemoryEntry]
  write(event: Event) -> Ack
  consolidate(epoch: Tick) -> ConsolidationReport
  capacity, recency_bias, episodic_share, forgetting_curve  // bound by PsychologyProfile
}
// ownership: Agent (one adapter instance per agent); mutation: write/consolidate only in
// WRITE phase (I3); concurrency: single-writer per agent, cross-agent parallel;
// persistence: vector store (Chroma → managed, ADR-17); compatibility: additive-only v1+.
ResourceLayer / GeographyLayer / InstitutionLayer (interfaces) {
  ResourceLayer.stocks, .flows, .replenishment_rules
  GeographyLayer.adjacency, .distance, .traversal_cost
  InstitutionLayer.currencies, .courts, .schools, .state_transitions
}
// ownership: Environment; mutation: only via Environment.tick(); concurrency: single-writer
// per environment per tick; persistence: object storage + TimescaleDB snapshot;
// compatibility: additive-only v1+.
ShardDescriptor {
  shard_id:             int
  agent_ids:            Set[AgentId]       // agent_id mod n partition (ADR-07)
  node:                 NodeId
  fd_budget_used:       int                // checked against §4 ceiling
}
// ownership: Orchestrator sharder; mutation: recomputed on rebalance (Phase 7, §19);
// concurrency: one ShardDescriptor per worker process; persistence: in-memory + heartbeat
// log; compatibility: additive-only v1+.
ConformanceReport {
  component:            ComponentRef       // psychology | environment | memory-adapter
  fixtures_run:         List[FixtureId]
  fixtures_passed:      List[FixtureId]
  fixtures_failed:      List[FixtureId]
  admitted:             bool               // gates template-store entry, I13
}
// ownership: core/conformance runner; mutation: immutable per run; concurrency: safely
// parallel across components; persistence: CI artefact + object storage; compatibility:
// additive-only v1+.
