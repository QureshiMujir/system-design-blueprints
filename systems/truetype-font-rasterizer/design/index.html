Core schema & types (§11)

struct Tag(u32);                    // 4 ASCII bytes; from_bytes(&[u8;4]); Display
struct Fixed16_16(i32);             // 16.16 signed fixed; mul/div via i64
struct F2Dot14(i16);                // 2.14 signed; range −2.0..+1.999939; → Fixed16_16
struct F26Dot6(i32);                // 26.6 device coord; +-/ ; round(), floor(), ceil()
type GlyphId = u16;                 // index into glyf/loca; < numGlyphs
struct FontRef<'a> {               // zero-copy view over font bytes
  data: &'a [u8],
  dir: TableDirectory,             // parsed once
  head: OnceCell<Head>, maxp: OnceCell<Maxp>, ...   // lazy table views (interior mut)
}
struct TableDirectory { records: Box<[TableRecord]> }   // sorted by tag
struct TableRecord { tag: Tag, check_sum: u32, offset: u32, length: u32 }
struct Reader<'a> { data: &'a [u8], pos: usize }        // bounds-checked cursor
impl Reader<'a> {
  fn u8/u16/u32/i16/tag/f2dot14/fixed(&mut self) -> Result<_, FontError>;
  fn slice(&mut self, len: usize) -> Result<&'a [u8], FontError>;
  fn seek(&mut self, pos: usize) -> Result<(), FontError>;
}
struct Cmap<'a> { best: CmapSubtable<'a> }
enum CmapSubtable<'a> {
  Format0(&'a [u8]), Format2(&'a [u8]), Format4(Cmap4<'a>),
  Format6(&'a [u8]), Format12(Cmap12<'a>), Format14(Cmap14<'a>),
}
impl Cmap<'a> { fn glyph_index(&self, c: char) -> GlyphId; }  // 0 if unmapped
struct Point { x: F26Dot6, y: F26Dot6, on_curve: bool }
struct Contour { start: u32, end: u32 }     // index range into Outline.points (cyclic)
struct Outline {
  points: Vec<Point>,            // device or funit depending on stage (documented)
  contours: Vec<Contour>,
  bbox: BBox,                    // device bbox after scaling
}
enum Glyph { Simple(Outline), Composite(Vec<CompositeComponent>), Empty }
struct CompositeComponent {
  glyph: GlyphId,
  transform: Matrix2x2,          // F2Dot14-derived, stored 16.16
  args: ComponentArgs,           // Xy { dx, dy } | PointMatch { p_self, p_comp }
  flags: u16,
}
struct BBox { x_min: F26Dot6, y_min: F26Dot6, x_max: F26Dot6, y_max: F26Dot6 }
struct Affine { m: Matrix2x2, t: (F26Dot6, F26Dot6) }
struct Matrix2x2 { xx: Fixed16_16, xy: Fixed16_16, yx: Fixed16_16, yy: Fixed16_16 }
struct Scaler {
  units_per_em: u16,
  ppem: u32,                     // pixels/em
  user_affine: Affine,
  // device = round_26_6( user_affine * (funit * ppem / upem) ), y-flipped
}
impl Scaler { fn scale_point(&self, funit: (i32,i32)) -> Point; }
struct HintVm<'a> {
  stack: Vec<i32>,               // operand stack ≤ maxStackElements
  gstate: GraphicsState,
  cvt: Vec<F26Dot6>,             // scaled control values
  storage: Vec<i32>,             // ≤ maxStorage
  functions: FnTable<'a>,        // fpgm-defined function entry points
  twilight: Vec<Point>,          // zone 0
  zone1: *mut Outline,           // glyph zone (logical; safe handle in impl)
  budget: InstrBudget,           // remaining instrs + loop cap
}
struct GraphicsState {
  freedom_vector: (Fixed16_16, Fixed16_16),
  projection_vector: (Fixed16_16, Fixed16_16),
  dual_projection_vector: (Fixed16_16, Fixed16_16),
  round_state: RoundState,       // round-to-grid/half/double/super/off
  loop_count: u32,
  rp0: u32, rp1: u32, rp2: u32,  // reference points
  zp0: u8, zp1: u8, zp2: u8,     // zone pointers (0 twilight,1 glyph)
  control_value_cutin: F26Dot6,
  single_width_cutin: F26Dot6,
  minimum_distance: F26Dot6,
  instruct_control: u8, scan_control: bool, ...
}
struct InstrBudget { remaining: u64, loop_cap: u32 }
struct Raster {
  edges: EdgeList,               // reused buffer
  cells: Vec<Cell>,              // signed-area accumulation, reused
  width: u32, height: u32,       // device bbox extent
  origin: (i32, i32),            // device pixel origin (left, top)
}
struct EdgeList { edges: Vec<Edge> }
struct Edge { x0,y0,x1,y1: F26Dot6, dir: i8 }     // monotone, winding direction ±1
struct Cell { x: i32, cover: i32, area: i32 }      // FreeType-smooth-style cell
struct CoverageBuffer { width: u32, height: u32, data: Vec<u8> }  // 8-bit alpha
struct GlyphBitmap {
  width: u32, height: u32,
  left: i32, top: i32,           // bearing: pen-to-bitmap offset (device px)
  pitch: u32,                    // bytes per row (≥ width; may be padded)
  format: PixelFormat,           // Gray8 | Mono1 | Lcd24 (Phase 12)
  buffer: Vec<u8>,               // pitch*height bytes
}
struct GlyphMetrics {
  advance: F26Dot6,              // horizontal advance (device)
  lsb: F26Dot6,                  // left side bearing
  bbox: BBox,                    // ink box (device)
}
enum PixelFormat { Mono1, Gray8, Lcd24 }
struct GlyphCacheKey {
  glyph_id: GlyphId,
  size: u32,                     // ppem (or fixed-point size encoded)
  transform: TransformKey,       // quantized affine (exact bits)
  hint_mode: HintMode,
}
enum HintMode { None, Bytecode, Auto /* deferred */ }
struct GlyphCache { map: LruMap<GlyphCacheKey, Arc<GlyphBitmap>>, budget_bytes: usize }
struct Atlas { width: u32, height: u32, shelves: Vec<Shelf>, data: Vec<u8> }
struct AtlasSlot { x: u32, y: u32, w: u32, h: u32 }   // location of a glyph in the sheet
enum FontError {
  OutOfBounds { table: Tag, at: usize, need: usize, have: usize },
  BadMagic { found: u32 },
  MissingTable(Tag),
  UnsupportedSfntVersion(u32),
  BadUnitsPerEm(u16),
  BadLoca { reason: LocaErr },          // NonMonotonic | OutOfRange | WrongLength
  BadGlyph { gid: GlyphId, reason: GlyphErr },
  CompositeTooDeep { gid: GlyphId, depth: u16 },
  CompositeCycle { gid: GlyphId },
  CoordOverflow { gid: GlyphId },
  HintBudgetExceeded { gid: GlyphId, instrs: u64 },
  HintStackFault { op: u8 },
  HintBadJump { target: i64 },
  UnsupportedCmap,
  BadSize(u32),
  GlyphTooLarge { gid: GlyphId, w: u32, h: u32 },
  Truncated,
}
