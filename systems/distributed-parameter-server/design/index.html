Core schema & types (§11)

Key                  // u64 or byte-string parameter key; Copy, hashable
ParamId              // u32 identifies a named parameter tensor/table; Copy
Version              // u64 monotonic per-key version counter; Copy, Ord
Clock                // u64 per-worker scalar clock (ADR-06); Copy, Ord
VectorClock          // Map<WorkerId, u64>; used only in replication forensics, not SSP hot path
DenseParam           // { param_id: ParamId, shape: Vec<u64>, data: Vec<f32> }; owned by Shard; mutable under lock
SparseParam          // { param_id: ParamId, entries: Map<Key, Vec<f32>> }; owned by Shard; mutable under lock
KeyRange             // { start: Key, end: Key }; immutable once assigned; Copy
PushRequest          // { worker_id: WorkerId, param_id: ParamId, range_or_keys, gradient: Gradient, sequence: u64 }; serializable (ADR-16)
PullRequest          // { worker_id: WorkerId, param_id: ParamId, range_or_keys, clock: Clock }; serializable
PullResponse         // { version: Version, values: DenseParam | SparseParam }; serializable
Gradient             // { dtype: DType, payload: DenseParam | SparseParam, compressed: Option<Quantized> }
Aggregator           // { param_id: ParamId, pending: Vec<Gradient>, contributor_count: u32 }; mutated only by Aggregation Engine
OptimizerState        // { optimizer_kind: OptimizerKind, moments: Map<Key, MomentState> }; colocated with parameter (ADR-09), persisted
Shard                 // { shard_id: ShardId, range: KeyRange, dense: DenseParam, sparse: SparseParam, optimizer: OptimizerState, wal: WalHandle }
ShardId                // u32; immutable once assigned
HashRing                // { virtual_nodes: BTreeMap<u64, ShardId>, epoch: u64 } (ADR-04, I8)
Placement                // { key: Key, primary: ShardId, replicas: Vec<ShardId> }
Replica                  // { shard_id: ShardId, role: Primary | Secondary, lag: Duration }
ConsistencyModel          // enum { Bsp, Asp, Ssp { bound: u32 } } (ADR-05)
StalenessBound             // u32; used only when ConsistencyModel::Ssp
WorkerId                    // u64; assigned at registration
ServerNode                   // { node_id: NodeId, shards: Vec<ShardId>, epoch: u64 }
Membership                    // { nodes: Vec<ServerNode>, epoch: u64 } (I8)
Rebalance                      // { moving_keys: Vec<Key>, from: ShardId, to: ShardId, epoch: u64 }
Compressor                      // trait: encode(Gradient) -> Quantized; decode(Quantized) -> Gradient
Quantized                        // { bits: u8, scale: f32, payload: Vec<u8>, residual_carried: bool } (I10)
ErrorFeedback                     // { worker_id: WorkerId, param_id: ParamId, residual: Vec<f32> }; persists across rounds
Checkpoint                         // { generation: u64, manifest: Vec<ChunkRef>, committed: bool } (I11)
Coordinator                         // { membership: Membership, ring: HashRing }; never holds parameter data (P10)
PsError                              // enum { StaleVersionRejected, DuplicateSequence, ShardUnavailable, StalenessBoundExceeded, RebalanceInProgress, ChecksumMismatch }
