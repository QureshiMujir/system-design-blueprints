Core schema & types (§11)

pub enum NodeKind { Entity, Activity, Agent }
pub struct NodeId([u8; 32]);                       // content hash (I5)
pub struct DatasetRef { namespace: String, name: String, version: Option<String> }
pub struct ColumnRef  { dataset: DatasetRef, column: String }
pub struct JobRef     { namespace: String, name: String }
pub struct AgentRef   { id: String, kind: AgentKind }    // Person | Service | Pipeline
pub struct Entity   { id: NodeId, dataset: DatasetRef, column: Option<String> }  // column=None → dataset entity
pub struct Activity { id: NodeId, run_id: String, job: JobRef }
pub struct Agent    { id: NodeId, agent: AgentRef }
pub enum Relation { Used, WasGeneratedBy, WasDerivedFrom, WasAssociatedWith, WasInformedBy }
pub struct RunContext { run_id: String, job: JobRef, valid_time: Timestamp, recorded_at: Timestamp,
                        agent: AgentRef, facets: Facets }
pub struct Edge { id: EdgeId, src: NodeId, dst: NodeId, rel: Relation,
                  granularity: Granularity, ctx: RunContext }       // ctx required (I4)
pub enum Granularity { Dataset, Column }
pub struct EdgeId([u8; 32]);                       // hash(src,dst,rel,run_id) (I7)
pub struct Timestamp(i64);                          // ns since epoch
pub struct BiTemporal { valid: Interval, tx: Interval }   // valid-time + transaction-time
pub struct Revision { rev: u64, tx_time: Timestamp, added_nodes: Vec<NodeId>, added_edges: Vec<EdgeId> }
pub struct Snapshot { as_of_tx: Timestamp, as_of_valid: Option<Timestamp> }   // a (valid,tx) cut (I11/I13)
pub trait LineageGraph {
    fn append(&self, rev: &RevisionBatch) -> Result<Revision, LinError>;        // append-only (I3), cycle-checked (I1)
    fn node(&self, id: NodeId, snap: &Snapshot) -> Option<NodeView>;
    fn out_edges(&self, id: NodeId, rel: Relation, snap: &Snapshot) -> EdgeIter; // downstream
    fn in_edges (&self, id: NodeId, rel: Relation, snap: &Snapshot) -> EdgeIter; // upstream
}
pub struct RevisionBatch { nodes: Vec<Node>, edges: Vec<Edge> }   // dedup applied (I5/I7)
pub struct OlRunEvent { event_type: OlEventType, run_id: String, job: JobRef,
                        inputs: Vec<OlDataset>, outputs: Vec<OlDataset>, event_time: Timestamp, facets: Facets }
pub enum OlEventType { Start, Running, Complete, Fail, Abort }
pub struct OlDataset { ds: DatasetRef, schema: Option<Schema>, sql: Option<String>, column_lineage: Option<ColLineageFacet> }
pub trait Capture {
    fn ingest(&self, ev: OlRunEvent) -> Result<IngestOutcome, LinError>;   // idempotent (I7), validated (I2/I15)
}
pub struct IngestOutcome { dedup_hit: bool, edges_added: usize, parse_complete: bool }  // telemetry (I12)
pub struct ColumnLineageRule;  // conservative: output_col ← every referenced input ColumnRef (I9)
pub fn extract_column_lineage(sql: &str, schemas: &SchemaCatalog)
        -> Result<Vec<(ColumnRef /*out*/, Vec<ColumnRef> /*ins*/)>, ParseOutcome>;     // bounded (I15)
pub enum ParseOutcome { Complete, ColumnIncomplete(String) }   // fall back to dataset-level (I12)
pub struct ClosureOpts { max_hops: u32, max_results: usize, deadline: Duration, granularity: Granularity }
pub trait LineageQuery {
    fn upstream  (&self, n: NodeId, o: &ClosureOpts, s: &Snapshot) -> ClosureResult;   // reverse-BFS (I8)
    fn downstream(&self, n: NodeId, o: &ClosureOpts, s: &Snapshot) -> ClosureResult;   // forward-BFS (I8)
    fn impact    (&self, n: NodeId, o: &ClosureOpts, s: &Snapshot) -> ImpactResult;    // +runs/jobs on paths
    fn root_cause(&self, c: ColumnRef, pred: &Predicate, s: &Snapshot) -> RootCauseResult; // ranked
}
pub struct ClosureResult { nodes: Vec<NodeView>, edges: Vec<EdgeView>, truncated: bool, hops_reached: u32 } // (I8/I12)
pub struct MerkleLog { /* hash-chained append log over #71 */ }
impl MerkleLog {
    fn append(&self, event_bytes: &[u8]) -> Result<MerkleRoot, LinError>;   // (I10)
    fn verify(&self, range: LogRange) -> Result<bool, LinError>;            // recompute chain
}
pub struct Signature([u8; 64]);                    // Ed25519 (RFC 8032)
pub fn sign(event: &[u8], sk: &SigningKey) -> Signature;
pub fn verify_sig(event: &[u8], sig: &Signature, pk: &VerifyingKey) -> bool;
pub enum LinError { ProvType(String), CycleDetected{src:NodeId,dst:NodeId}, MissingRunContext,
                    ParseBounded(String), SqlUnsupported(String), DedupConflict,
                    SnapshotInvalid, QueryDeadline, GraphStore(String), MerkleVerify, BadArg(String) }
pub struct IngestTrace { source:&'static str, dedup_hit:bool, parse_complete:bool, edges_added:usize,
                         column_edges:usize, parse_us:u64, append_us:u64, cycle_rejected:bool }
pub struct QueryTrace  { kind:&'static str, hops_reached:u32, nodes_visited:usize, truncated:bool,
                         col_coverage:f64, snapshot_tx:i64, duration_us:u64, result_hash:u64 }
pub struct ServerConfig { graph_endpoint: GraphRef /*#84*/, ingest_workers: usize, ingest_queue_cap: usize,
                          sql_parse_limits: ParseLimits, signing: Option<SigningPolicy>,
                          retention: RetentionPolicy, default_closure: ClosureOpts }
pub struct ParseLimits { max_depth: u32, max_tokens: usize, max_time: Duration }   // (I15)
pub struct RetentionPolicy { keep_versions: u32, valid_time_window: Duration, compact_after: Duration } // (P10)
