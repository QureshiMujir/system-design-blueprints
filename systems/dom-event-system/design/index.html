Core schema & types (§11)

/// The DOM Event. Constructed once per dispatch; FIELDS MUTATED ONLY by evt-dispatch
/// (current_target/event_phase/flags/path). target/related_target/bubbles/cancelable/
/// composed/is_trusted/ty/time_stamp are IMMUTABLE after init. !Send + !Sync.
pub struct Event {
    ty: Atom, target: Option<OpaqueTarget>, related_target: Option<OpaqueTarget>,
    current_target: Option<OpaqueTarget>, event_phase: EventPhase, flags: EventFlags,
    bubbles: bool, cancelable: bool, composed: bool, is_trusted: bool,
    time_stamp: f64, path: Vec<EventPathItem>,   // computed ONCE at dispatch start
}
/// Dispatch phase. Discriminants spec-stable (wire/trace constants). None outside dispatch.
pub enum EventPhase { None = 0, CapturingPhase = 1, AtTarget = 2, BubblingPhase = 3 }
/// Internal state bits. INITIALIZED set by ctor/initEvent; DISPATCH set during dispatch (re-entry
/// guard); CANCELED == defaultPrevented; IN_PASSIVE_LISTENER gates preventDefault no-op.
bitflags EventFlags {
    STOP_PROPAGATION, STOP_IMMEDIATE_PROPAGATION, CANCELED,
    IN_PASSIVE_LISTENER, COMPOSED, INITIALIZED, DISPATCH
}
/// One registered listener. removed flag = tombstone for mid-invoke removal (skipped, not popped).
/// passive: None = "not specified" (resolve to default-passive per type). Dedup key = (ty,callback,capture).
pub struct EventListener {
    ty: Atom, callback: CallbackRef, capture: bool,
    passive: Option<bool>, once: bool, signal: Option<AbortSignalRef>, removed: bool,
}
/// 3rd arg to addEventListener (dictionary). signal: registering with already-aborted signal = no-op (never added).
pub struct AddEventListenerOptions { capture: bool, passive: Option<bool>, once: bool, signal: Option<AbortSignalRef> }
/// 3rd arg to removeEventListener. Only capture significant (rest ignored on removal).
pub struct EventListenerOptions { capture: bool }
/// Opaque host callable. EQUALITY BY IDENTITY ONLY (pointer/handle); never invoked by core —
/// host runs it. Owned by listener list; Clone = handle copy.
pub struct CallbackRef(/* opaque host handle; PartialEq == identity */);
/// Host node-tree interface. Mirrors project 258's Element trait. All methods PURE reads except
/// listeners() which yields a shared-mutable cell. Clone = cheap handle clone (Rc).
pub trait EventTarget: Sized + Clone {
    /// Shared-mutable listener list for THIS target. Borrow discipline: hold RefCell borrow only
    /// across snapshot/mutation, NEVER across a callback invocation (re-entrancy aliasing hazard).
    fn listeners(&self) -> ListenerListRef;
    /// Event-path parent (spec "get the parent"). MAY depend on event (composed/slot). None = path root.
    fn get_the_parent(&self, event: &Event) -> Option<Self>;
    /// Activation behavior hook, or None. Run as default action at end of dispatch if !canceled.
    fn activation_behavior(&self) -> Option<ActivationBehavior>;
    /// Spec "legacy-pre-activation behavior" (radio/checkbox pre-toggle). Default no-op.
    fn legacy_pre_activation_behavior(&self) {}
    /// Spec "legacy-canceled-activation behavior" (undo pre-activation if canceled). Default no-op.
    fn legacy_canceled_activation_behavior(&self) {}
    /// Identity token for path equality, retarget, currentTarget storage. Stable for target lifetime.
    fn opaque(&self) -> OpaqueTarget;
}
pub fn add_event_listener<T: EventTarget>(t: &T, ty: Atom, cb: CallbackRef, opts: AddEventListenerOptions);
pub fn remove_event_listener<T: EventTarget>(t: &T, ty: &Atom, cb: &CallbackRef, capture: bool);
/// Returned by dispatch entry when event unusable.
pub enum EventError { Uninitialized, AlreadyDispatching /* InvalidStateError */ }
/// One entry in the propagation path (spec "struct"). IMMUTABLE after build_event_path.
/// invocation_target = node whose listeners run; shadow_adjusted_target = exposed event.target at
/// this item (None ⇒ reuse previous non-None, i.e. "no retarget here"); related_target / touch_target_list
/// are per-item retargeted values; root_of_closed_tree / slot_in_closed_tree gate composedPath() visibility.
pub struct EventPathItem {
    invocation_target: OpaqueTarget, invocation_target_in_shadow_tree: bool,
    shadow_adjusted_target: Option<OpaqueTarget>, related_target: Option<OpaqueTarget>,
    touch_target_list: Vec<Option<OpaqueTarget>>, root_of_closed_tree: bool, slot_in_closed_tree: bool,
}
/// Build path ONCE: walk get_the_parent target→root, computing retargeting + closed-tree flags.
/// Non-composed events truncate at target's nearest shadow-including root.
pub fn build_event_path<T: EventTarget>(target: &T, event: &Event) -> Vec<EventPathItem>;
/// Spec "retarget(A, B)": climb A's shadow-including ancestors until A not a descendant of B's root.
/// Pure; identity-only. Keeps closed-shadow internals from leaking as target/relatedTarget.
pub fn retarget(a: OpaqueTarget, b: OpaqueTarget) -> OpaqueTarget;
/// composedPath(): saved path filtered by closed-tree visibility relative to `current` (currentTarget).
/// Empty if event not dispatching. Pure read of event.path.
pub fn composed_path(event: &Event, current: OpaqueTarget) -> Vec<OpaqueTarget>;
/// Entry point (spec "dispatch"). Builds path, sets DISPATCH flag, runs capturing → at-target →
/// bubbling, then default action. Returns FALSE iff canceled (== !defaultPrevented). MUST error
/// (Uninitialized / AlreadyDispatching) if event lacks INITIALIZED or already has DISPATCH set.
pub fn dispatch<T: EventTarget>(target: &T, event: &mut Event, legacy_target_override: bool) -> bool;
/// Spec "invoke" for one path item + phase: set current_target/event_phase, take SNAPSHOT of the
/// listener list, then inner_invoke over it. Honors stop-propagation between items.
fn invoke(item: &EventPathItem, event: &mut Event, phase: EventPhase);
/// Spec "inner invoke": iterate the SNAPSHOT. Skip removed; skip wrong-phase/type; remove `once`
/// from REAL list BEFORE running; set IN_PASSIVE_LISTENER around passive callbacks. Returns
/// false (stop iteration) on stopImmediatePropagation.
fn inner_invoke(event: &mut Event, listeners: &[EventListener], phase: EventPhase) -> bool;
/// Host default-action hook. Invoked at end of dispatch iff !canceled. Opaque fn(&Event).
pub struct ActivationBehavior(/* host hook: fn(&Event) */);
/// Owns its signal 1:1. abort() flips signal aborted + runs abort algorithms.
pub struct AbortController { signal: AbortSignalRef }
/// aborted monotonic (false→true, never back). reason = host JsValueRef (DOMException "AbortError"
/// default). abort_algorithms run ONCE on abort then cleared. source/dependent links (WeakVec, no
/// cycles) implement AbortSignal.any follow relationship. !Send + !Sync.
pub struct AbortSignal {
    aborted: bool, reason: JsValueRef, abort_algorithms: Vec<AbortAlgo>,
    source_signals: WeakVec<AbortSignal>, dependent_signals: WeakVec<AbortSignal>,
}
impl AbortSignal {
    /// Pre-aborted standalone signal with the given reason.
    pub fn abort(reason: JsValueRef) -> AbortSignalRef;
    /// Dependent signal aborted (TimeoutError) after ms via an evt-loop Timer task.
    pub fn timeout(ms: u64) -> AbortSignalRef;
    /// Dependent signal that follows the FIRST source to abort (AbortSignal.any).
    pub fn any(sigs: &[AbortSignalRef]) -> AbortSignalRef;
    /// Err(reason) if aborted, else Ok.
    pub fn throw_if_aborted(&self) -> Result<(), JsValueRef>;
}
/// Task provenance (priority/grouping). Trusted UA-originated events use UserInteraction/Dom.
pub enum TaskSource { Dom, UserInteraction, Networking, Timer }
/// A queued unit of work. steps run to completion (no preemption). document scopes the task.
pub struct Task { source: TaskSource, steps: TaskClosure, document: Option<DocRef> }
/// One per thread. performing_checkpoint guards microtask-checkpoint re-entrancy.
pub struct EventLoop {
    task_queues: Map<TaskSource, VecDeque<Task>>, microtasks: VecDeque<Task>,
    performing_checkpoint: bool,
}
impl EventLoop {
    pub fn queue_task(&mut self, s: TaskSource, t: TaskClosure);
    pub fn queue_microtask(&mut self, t: TaskClosure);
    /// Drain microtasks to empty (they MAY enqueue more); reentrancy-guarded.
    pub fn perform_microtask_checkpoint(&mut self);
    /// Run ONE task from one queue to completion, then a microtask checkpoint.
    pub fn tick(&mut self);
    /// UA path: set is_trusted=true, dispatch SYNCHRONOUSLY within the current task.
    pub fn fire_event(&mut self, target: OpaqueTarget, ev: Event);
}
/// Rc handle into reference Node arena. impl EventTarget. Clone = Rc clone (cheap).
/// listeners() returns node's own Rc<RefCell<Vec<EventListener>>>. !Send + !Sync.
pub struct NodeRef(/* Rc into a Node arena */);
/// Shadow root encapsulation mode. Closed hides internals from composedPath()/retarget exposure.
pub enum ShadowMode { Open, Closed }
