Core schema & types (§11)

// core — propositional substrate (normative)
pub struct Var(pub u32);
#[derive(Clone, Copy)]
pub struct Lit(u32);                       // (var<<1)|sign ; cheap negation = ^1
impl Lit { pub fn var(self)->Var; pub fn sign(self)->bool; pub fn neg(self)->Lit; }
#[derive(Clone, Copy, PartialEq)]
pub enum LBool { True, False, Undef }
pub struct ClauseRef(u32);                  // offset into arena (ADR-04/05)
pub struct ClauseHeader { len:u32, lbd:u32, activity:f32, learnt:bool, deleted:bool }
// Clause stored as header + [Lit; len] contiguously in Arena
pub struct Arena { mem: Vec<u32> }          // bump-allocated; compaction relocates refs
// trail — the single source of truth (P2, I1/I4)
pub struct Trail {
    assigns: Vec<LBool>,                    // per Var
    level:   Vec<u32>,                      // decision level per Var
    reason:  Vec<Option<ClauseRef>>,        // implying clause, None = decision
    stack:   Vec<Lit>,                      // assignment order
    lim:     Vec<usize>,                    // stack index at each decision level
}
impl Trail {
    fn assign(&mut self, l: Lit, reason: Option<ClauseRef>);   // push, set level
    fn backjump(&mut self, level: u32);                        // truncate to level (I4)
    fn value(&self, l: Lit) -> LBool;
}
// cdcl — engine contract (normative)
pub enum SolveResult { Sat(Model), Unsat(Proof), Unknown(Reason) }   // P8/I15
pub trait SatSolver {
    fn add_clause(&mut self, lits: &[Lit]) -> Result<(), ()>;        // false = top-level unsat
    fn solve(&mut self, assumptions: &[Lit]) -> SolveResult;         // incremental (P10)
    fn value(&self, v: Var) -> LBool;
    fn unsat_core(&self) -> Vec<Lit>;                                // over assumptions
}
struct Watcher { clause: ClauseRef, blocker: Lit }                   // two-watched (ADR-07)
// theory — DPLL(T) interface (normative, P5/P6/P11/P12)
pub enum TStatus { Sat, Conflict(Vec<Lit>) }       // conflict = explanation clause (negated)
pub trait Theory {
    fn assert_lit(&mut self, l: Lit) -> TStatus;    // notify atom truth; cheap-detect
    fn check(&mut self, full: bool) -> TStatus;     // full at Boolean fixpoint (P11)
    fn propagate(&mut self) -> Vec<(Lit, /*reason*/ Vec<Lit>)>; // theory-implied lits
    fn explain(&mut self, l: Lit) -> Vec<Lit>;      // why l was propagated (sound, I5)
    fn push(&mut self);                              // checkpoint (I6)
    fn pop(&mut self, n: usize);                     // restore O(Δ) (I6)
    fn model(&self) -> TheoryModel;                  // values for SAT (I8)
}
// theories/euf — congruence closure (normative, I11)
pub struct EUF {
    uf: UnionFind,                          // union-by-rank + path compression
    sig: HashMap<Signature, TermId>,        // f(args) signature → representative
    proof_forest: Vec<EdgeReason>,          // for explanations (Nieuwenhuis-Oliveras)
}
impl EUF { fn merge(&mut self, a:TermId, b:TermId, why:Reason); fn find(&self,t:TermId)->TermId;
           fn explain_eq(&self, a:TermId, b:TermId) -> Vec<Lit>; }
// theories/arith — exact simplex (normative, I10; numbers are bignum rationals, ADR-11)
pub struct Rational;                         // p/q, arbitrary precision
pub struct Simplex {
    tableau: Matrix<Rational>,               // basic = Σ coeff·nonbasic
    bounds:  Vec<(Option<Rational>,Option<Rational>)>, // lo/hi per var
    basic:   Vec<usize>, nonbasic: Vec<usize>,
    assign:  Vec<Rational>,                  // current value per var
}
impl Simplex { fn assert_bound(&mut self, v:usize, b:Bound)->TStatus;
               fn check(&mut self)->TStatus;             // pivot to satisfy bounds
               fn pivot(&mut self, enter:usize, leave:usize); }  // Bland's anti-cycle (I10)
// smt — incremental frontend (normative, P10)
pub struct AssertionStack { levels: Vec<Vec<TermId>> }   // push/pop scoping
pub trait SmtContext {
    fn set_logic(&mut self, logic: &str);
    fn declare_fun(&mut self, name:&str, sort:Sort);
    fn assert(&mut self, t: TermId);
    fn push(&mut self, n: usize); fn pop(&mut self, n: usize);
    fn check_sat(&mut self) -> SolveResult;
    fn get_model(&self) -> Model; fn get_value(&self, ts:&[TermId]) -> Vec<Value>;
    fn get_unsat_core(&self) -> Vec<String>;             // named assertions
}
pub struct PhaseDeferredError { pub reference: &'static str }   // PhaseDeferred("Stretch Goals")
