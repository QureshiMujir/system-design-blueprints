pub struct SymbolId(pub u32);

pub struct FileId(pub u32);

pub struct Interned(pub u32);
// interner.rs
pub struct StringInterner { /* arena + hashmap */ }
impl StringInterner {
    pub fn intern(&mut self, s: &str) -> Interned;
    pub fn resolve(&self, id: Interned) -> &str;
    pub fn len(&self) -> usize;
}

pub struct PathInterner { /* wraps StringInterner with normalization */ }
impl PathInterner {
    pub fn intern_path(&mut self, path: &std::path::Path) -> FileId;
    pub fn resolve_path(&self, id: FileId) -> &std::path::Path;
}
// position.rs
#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, PartialOrd, Ord)]
pub struct Position { pub line: u32, pub col: u32 }   // 0-indexed; col in UTF-16 code units (LSP compat)

#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
pub struct Range { pub start: Position, pub end: Position }  // half-open [start, end)

impl Range {
    pub fn contains(&self, pos: Position) -> bool;      // contract: start <= pos < end
    pub fn is_empty(&self) -> bool;                     // contract: start == end
    pub fn overlaps(&self, other: Range) -> bool;       // I3: used in occurrence validator
}
// moniker.rs
#[derive(Clone, Debug, PartialEq, Eq, Hash)]
pub struct PackageId {
    pub manager: Interned,   // e.g. "cargo", "npm", "pypi"
    pub name:    Interned,   // e.g. "serde"
    pub version: Interned,   // semver string; "" for local
}

#[derive(Clone, Debug, PartialEq, Eq, Hash)]
pub struct Moniker {
    pub scheme:     Interned,      // e.g. "rust", "python", "typescript"
    pub package:    PackageId,
    pub descriptor: String,        // SCIP descriptor path; e.g. "module/Type#method()."
}

impl Moniker {
    pub fn is_local(&self) -> bool;          // version == "" and manager == "local"
    pub fn descriptor_segments(&self) -> impl Iterator<Item=DescriptorSegment<'_>>;
}

// Descriptor grammar (SCIP-style EBNF):
// descriptor      = segment { '/' segment } '.'?
// segment         = namespace_seg | type_seg | term_seg | method_seg | param_seg | meta_seg
// namespace_seg   = ident '/'
// type_seg        = ident '#'
// term_seg        = ident '.'
// method_seg      = ident '(' params ')' '.'
// param_seg       = '[' ident ']'
// meta_seg        = '<' ident '>'
// roles.rs
use bitflags::bitflags;
bitflags! {
    pub struct SymbolRole: u32 {
        const Definition  = 0b0000_0001;
        const Reference   = 0b0000_0010;
        const Import      = 0b0000_0100;
        const Write       = 0b0000_1000;
        const Read        = 0b0001_0000;
        const Generated   = 0b0010_0000;
        const Test        = 0b0100_0000;
    }
}
// kind.rs
#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
#[repr(u8)]
pub enum SymbolKind {
    Module, Namespace, Package,
    Class, Struct, Interface, Enum, EnumMember,
    Trait, TypeAlias, Type,
    Function, Method, Constructor, Getter, Setter,
    Field, Variable, Parameter, Local,
    Constant, Property,
    Macro, Attribute,
    Unknown,
}
// occurrence.rs
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Occurrence {
    pub range:  Range,
    pub symbol: SymbolId,
    pub roles:  SymbolRole,
}

impl Occurrence {
    pub fn is_definition(&self) -> bool;   // contract: roles.contains(Definition)
    pub fn is_reference(&self) -> bool;    // contract: roles.contains(Reference)
}
// scie-symbol/src/lib.rs (re-exported from scie-core equivalents + symbol layer)
#[derive(Clone, Debug)]
pub struct SymbolInformation {
    pub symbol:    SymbolId,
    pub moniker:   Moniker,
    pub kind:      SymbolKind,
    pub signature: String,       // language-specific signature text; empty if unavailable
    pub docs:      String,       // extracted docstring; empty if unavailable
    pub enclosing: Option<SymbolId>,   // parent symbol in nesting hierarchy
}

// scie-type/src/type_ref.rs
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum TypeCtor { Named(SymbolId), Func { params: Vec<TypeRef>, ret: Box<TypeRef> }, Tuple(Vec<TypeRef>), Unknown }

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct TypeRef {
    pub symbol: Option<SymbolId>,   // resolved nominal type symbol; None if Unknown
    pub ctor:   TypeCtor,
}

impl TypeRef {
    pub fn unknown() -> Self;                               // contract: ctor == Unknown, symbol == None
    pub fn is_unknown(&self) -> bool;                       // I9: caller checks before trusting member lookup
    pub fn lookup_member(&self, name: Interned, db: &SymbolTable) -> Option<SymbolId>;
}
// scie-graph/src/graph.rs
#[derive(Clone, Debug, Default)]
pub struct CodeGraph {
    pub def_of:  std::collections::HashMap<SymbolId, (FileId, Range)>,
    pub refs_of: std::collections::HashMap<SymbolId, Vec<(FileId, Range)>>,
    pub impls:   std::collections::HashMap<SymbolId, Vec<SymbolId>>,   // trait -> [impls]
    pub calls:   std::collections::HashMap<SymbolId, Vec<SymbolId>>,   // caller -> [callees]
}

impl CodeGraph {
    pub fn add_def(&mut self, sym: SymbolId, loc: (FileId, Range));
    pub fn add_ref(&mut self, sym: SymbolId, loc: (FileId, Range));     // appends to refs_of
    pub fn add_impl(&mut self, trait_sym: SymbolId, impl_sym: SymbolId);
    pub fn add_call(&mut self, caller: SymbolId, callee: SymbolId);
    pub fn check_consistency(&self) -> Vec<GraphConsistencyError>;       // I4 validator
}
// scie-index/src/document.rs
#[derive(Clone, Debug)]
pub struct Document {
    pub file:        FileId,
    pub language:    Interned,
    pub occurrences: Vec<Occurrence>,       // sorted by range.start; non-overlapping (I3)
    pub symbols:     Vec<SymbolInformation>,
    pub scopes:      Option<Vec<Scope>>,    // optional; omitted in compact index mode
}

impl Document {
    pub fn symbol_at(&self, pos: Position) -> Option<&Occurrence>;  // binary search on occurrences
    pub fn outline(&self) -> Vec<&SymbolInformation>;               // filter: Definition role
}
// scie-incremental/src/revision.rs
#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
pub struct Revision(pub u64);   // monotonically increasing; 0 = initial

// scie-incremental/src/dep_edge.rs
#[derive(Clone, Debug, PartialEq, Eq, Hash)]
pub struct DepEdge { pub from: QueryKey, pub to: QueryKey }

#[derive(Clone, Debug, PartialEq, Eq, Hash)]
pub enum QueryKey { ParseFile(FileId), BuildScopes(FileId), ResolveFile(FileId), BuildGraph }

// scie-incremental/src/db.rs
pub trait QueryDatabase {
    fn set_file_source(&mut self, file: FileId, src: String, rev: Revision);
    fn get_document(&self, file: FileId) -> Arc<Document>;   // memoized; recomputes if dirty
    fn invalidate(&mut self, file: FileId) -> Vec<QueryKey>; // returns affected query set
    fn current_revision(&self) -> Revision;
}
