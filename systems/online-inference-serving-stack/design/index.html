Core schema & types (§11)

struct WorkerDescriptor
    worker_id: WorkerId
    node_id: NodeId
    numa_zone: NumaZoneId
    gpu_inventory: Vec<GpuDescriptor>
    memory_capacity_bytes: u64
    active_leases: Vec<LeaseId>
    health_state: HealthState
    runtime_version: SemVer
struct BatchDescriptor
    batch_id: BatchId
    model_key: ModelKey
    request_ids: Vec<RequestId>
    tensor_shape: TensorShape
    creation_timestamp: MonotonicTs
    dispatch_deadline: MonotonicTs
    estimated_memory_bytes: u64
struct ReplayRecord
    replay_id: ReplayId
    scheduler_decision: SchedulerDecisionRef
    worker_assignment: WorkerId
    serialized_request: Bytes
    runtime_metadata: RuntimeMetadata
    execution_hash: Blake3Hash
struct SchedulerLease
    lease_id: LeaseId
    worker_id: WorkerId
    model_key: ModelKey
    expiration_timestamp: MonotonicTs
    resource_vector: ResourceVector
    fencing_token: u64
struct TenantQuota
    tenant_id: TenantId
    max_queue_depth: u32
    max_concurrency: u32
    tokens_per_second: f64
    fairness_weight: f64
struct ModelManifest
    model_key: ModelKey
    version: SemVer
    content_hash: Blake3Hash
    artifact_uri: Uri
    tensor_input_spec: TensorSpec
    tensor_output_spec: TensorSpec
    rollout_state: RolloutState
struct AdmissionTicket
    request_id: RequestId
    tenant_id: TenantId
    enqueue_timestamp: MonotonicTs
    state: RequestState
    shed_reason: Option<ShedReason>
trait RuntimeExecutor
    fn load_model(&mut self, manifest: &ModelManifest) -> Result<(), LoadError>;
    fn unload_model(&mut self, model_key: &ModelKey) -> Result<(), UnloadError>;
    fn execute_batch(&self, batch: &BatchDescriptor) -> Result<BatchOutput, ExecError>;
    fn health_check(&self) -> HealthState;
    fn collect_runtime_metrics(&self) -> RuntimeMetrics;
struct RoutingTableSnapshot
    generation: u64
    tenant_partitions: HashMap<TenantId, PartitionId>
    model_routes: HashMap<ModelKey, Vec<WorkerId>>
    draining_workers: HashSet<WorkerId>
