#[derive(Copy, Clone, Debug, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
pub struct Diff(pub i64);

impl Diff {
    pub fn zero() -> Self;
    pub fn is_zero(self) -> bool;
    pub fn checked_add(self, rhs: Self) -> Option<Self>;
    pub fn negate(self) -> Self;               // flip sign; used by negate operator
}
impl std::ops::Add for Diff { type Output = Self; fn add(self, rhs: Self) -> Self; }
impl std::ops::Neg for Diff { type Output = Self; fn neg(self) -> Self; }
pub trait Timestamp:
    Clone + Ord + PartialOrd + Eq + Hash + Debug + Send + Sync + Serialize + DeserializeOwned
{
    fn minimum() -> Self;                      // bottom element of the lattice
    fn join(&self, other: &Self) -> Self;      // least upper bound
    fn meet(&self, other: &Self) -> Self;      // greatest lower bound
    fn less_equal(&self, other: &Self) -> bool;
    fn advance_by(&self, frontier: &Frontier<Self>) -> Self; // project to frontier
}

#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
pub struct Frontier<T: Timestamp>(Vec<T>);  // invariant: always an antichain

impl<T: Timestamp> Frontier<T> {
    pub fn empty() -> Self;                  // no times — "everything is complete"
    pub fn from_elem(t: T) -> Self;
    pub fn insert(&mut self, t: T);
    pub fn advance_to(&mut self, t: T);
    pub fn dominates(&self, t: &T) -> bool;  // true iff some element <= t
    pub fn is_empty(&self) -> bool;
    pub fn elements(&self) -> &[T];
    pub fn join(&self, other: &Self) -> Self; // antichain-join; I3
}
pub struct Collection<D, T: Timestamp> {
    // opaque; implemented as a channel or iterator of Update<D,T>
}
impl<D: Clone + Ord, T: Timestamp> Collection<D, T> {
    pub fn map<E, F: Fn(D) -> E>(&self, f: F) -> Collection<E, T>;
    pub fn filter<F: Fn(&D) -> bool>(&self, f: F) -> Collection<D, T>;
    pub fn concat(&self, other: &Collection<D, T>) -> Collection<D, T>;
    pub fn negate(&self) -> Collection<D, T>;
    pub fn consolidate(&self) -> Collection<D, T>;              // I2
    pub fn as_of(&self, t: &T) -> Multiset<D>;                 // fold diffs <= t
}
pub struct Capability<T: Timestamp> {
    time: T,
    // token is reference-counted; drop decrements; last drop signals progress
}
impl<T: Timestamp> Capability<T> {
    pub fn time(&self) -> &T;
    pub fn delayed(&self, new_time: &T) -> Capability<T>; // new_time >= self.time
    pub fn downgrade(&mut self, new_time: &T);             // new_time >= self.time
}
impl<T: Timestamp> Drop for Capability<T> { fn drop(&mut self); }  // signals progress
#[derive(Clone, Debug, Serialize, Deserialize)]
pub enum ProgressMsg<T: Timestamp> {

    CapabilityDrop  { operator_id: OpId, released_time: T },
}
pub struct Batch<K, V, T: Timestamp> {
    // immutable; sorted by (key, val, time); stored as columnar arrays
    lower: Frontier<T>,        // times strictly below lower are absent
    upper: Frontier<T>,        // times in [lower, upper) are present
}
impl<K: Ord, V: Ord, T: Timestamp> Batch<K, V, T> {
    pub fn cursor(&self) -> BatchCursor<K, V, T>;
    pub fn len(&self) -> usize;
    pub fn size_bytes(&self) -> usize;
    pub fn lower(&self) -> &Frontier<T>;
    pub fn upper(&self) -> &Frontier<T>;
}
pub trait Trace<K, V, T: Timestamp>: Send + Sync {
    type Batch: BatchLike<K, V, T>;
    type Cursor: CursorLike<K, V, T>;

    fn insert(&mut self, batch: Self::Batch);
    fn cursor(&self) -> Self::Cursor;                          // scan all data
    fn cursor_through(&self, frontier: &Frontier<T>) -> Option<Self::Cursor>; // as-of
    fn advance_by(&mut self, frontier: &Frontier<T>);          // set since-frontier; I7
    fn distinguish_since(&mut self, frontier: &Frontier<T>);   // compaction trigger
    fn get_logical_compaction(&self) -> &Frontier<T>;
}
pub trait CursorLike<K, V, T: Timestamp> {
    fn key_valid(&self, storage: &impl Trace<K,V,T>) -> bool;
    fn val_valid(&self, storage: &impl Trace<K,V,T>) -> bool;
    fn key<'a>(&self, storage: &'a impl Trace<K,V,T>) -> &'a K;
    fn val<'a>(&self, storage: &'a impl Trace<K,V,T>) -> &'a V;
    fn map_times<F: FnMut(&T, Diff)>(&mut self, storage: &impl Trace<K,V,T>, f: F);
    fn step_key(&mut self, storage: &impl Trace<K,V,T>);
    fn step_val(&mut self, storage: &impl Trace<K,V,T>);
    fn seek_key(&mut self, storage: &impl Trace<K,V,T>, key: &K);
}
pub trait Operator: Send {
    fn name(&self) -> &str;
    fn on_input(&mut self, input: &[Update<Row>], frontier: &Frontier<LogicalTime>)
        -> Vec<Update<Row>>;                                  // emit output deltas
    fn on_frontier(&mut self, frontier: &Frontier<LogicalTime>)
        -> Vec<Update<Row>>;                                  // finalize pending output
    fn frontier_hint(&self) -> Frontier<LogicalTime>;         // operator's output frontier
}
