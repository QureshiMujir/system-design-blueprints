Core schema & types (§11)

pub struct Signal { samples: Vec<f32>, sample_rate: u32 }     // mono
impl Signal { fn duration_s(&self)->f32; fn resample(&self, to:u32)->Signal; }
pub struct FeatureMatrix { data: Vec<f32>, t: usize, d: usize } // T×D row-major
impl FeatureMatrix { fn frame(&self, i)->&[f32]; fn t(&self)->usize; fn d(&self)->usize;
    fn frame_time_s(&self, i, hop, sr)->f32; }                  // I1 frame→time
pub struct FrontEndCfg { sr:u32, win_ms:f32, hop_ms:f32, n_fft:usize, n_mel:usize, n_ceps:usize,
    preemph:f32, lifter:f32, deltas:bool, cmvn:CmvnMode }
pub struct RawSignal(Signal);
pub struct Features(FeatureMatrix);
pub enum AmScores { Likelihoods(Array2<f32> /*T×states, log*/), Posteriors(Array2<f32> /*T×(A+1), log*/) }
pub struct Decoded { lattice: Lattice, one_best: Transcript }
pub struct Transcript { words: Vec<WordHyp> }
pub struct WordHyp { word:String, t_beg_s:f32, dur_s:f32, conf:Confidence } // I1/I10
pub struct Alignment { phones: Vec<(Phone,f32,f32)>, words: Vec<(String,f32,f32)> }
pub struct DiagGaussian { mean: Vec<f32>, var: Vec<f32>, log_det: f32 }   // #372
pub struct Gmm { comps: Vec<DiagGaussian>, log_weights: Vec<f32> }
impl Gmm { fn log_likelihood(&self, x:&[f32]) -> f32; }                   // log-sum-exp (I4)
pub struct Hmm { n_states: usize, log_a: Array2<f32> /*transitions*/, emit: Vec<PdfId> } // left-to-right
pub struct FbResult { log_alpha: Array2<f32>, log_beta: Array2<f32>, log_px: f32, scales: Vec<f32> } // I2
pub struct BwAccum { gamma_sum: /*per-state x,x²,count*/ Vec<f64>, xi_sum: Array2<f64> }   // reduce across utts (I11)
pub struct PdfId(u32);                                   // tied-state index
pub struct PhoneticTree { questions: Vec<Question>, nodes: Vec<TreeNode>, leaf_pdf: Vec<PdfId> } // I15
impl PhoneticTree { fn map(&self, ctx:&Triphone, state:usize) -> PdfId; }  // deterministic (I15)
pub struct Lexicon { word_to_phones: HashMap<String, Vec<Vec<Phone>>>, sil: Phone }
impl Lexicon { fn pron(&self, w:&str) -> Option<&[Vec<Phone>]>; fn g2p(&self, w:&str) -> Vec<Phone>; } // I9
pub struct Phone(u16); pub struct Triphone { left:Phone, center:Phone, right:Phone }
pub struct NGramLm { n:u8, /* counts + backoff weights, Kneser-Ney */ }
impl NGramLm { fn log_prob(&self, hist:&[WordId], next:WordId)->f32; fn perplexity(&self, text:&[WordId])->f32;
    fn read_arpa(p:&Path)->Self; fn write_arpa(&self, p:&Path); }
pub struct NeuralAm { cnn:CnnStack, rnn:BiLstm, head:Linear, alphabet:Alphabet } // #112/#349
impl NeuralAm { fn posteriors(&self, feats:&FeatureMatrix) -> Array2<f32>; }      // T×(|A|+1), log (I4)
pub struct Alphabet { syms:Vec<Sym>, blank:usize }                                // CTC blank (I5)
pub struct CtcLoss { blank:usize }
impl CtcLoss { fn forward_backward(&self, logp:&Array2<f32>, label:&[usize]) -> (f32, Array2<f32>); } // I4
pub fn greedy_decode(logp:&Array2<f32>, blank:usize)->Vec<usize>;                 // collapse+strip (I5)
pub fn prefix_beam(logp:&Array2<f32>, lex:&Lexicon, lm:&NGramLm, cfg:BeamCfg)->Vec<(Vec<usize>,f32)>;
pub struct Token { state:u32, log_score:f32, back:Option<u32>, word_hist:LmState }
pub struct DecodeCfg { beam:f32, max_active:usize, lm_weight:f32, word_penalty:f32, engine:AmKind }
pub struct Lattice { arcs:Vec<LatArc>, /* word graph */ }
pub struct LatArc { from:u32, to:u32, word:WordId, am_score:f32, lm_score:f32, t_beg:u32, t_end:u32 }
impl Lattice { fn one_best(&self)->Transcript; fn posteriors(&self)->Vec<f32> /*arc conf, I10*/; }
pub struct Config { am:AmKind, frontend:FrontEndCfg, decode:DecodeCfg, n_jobs:usize }
pub enum AmKind { GmmHmm, NeuralCtc }
pub enum AsrError { BadParam{what:&'static str}, DecodeFailed, DimMismatch{expected:usize,got:usize},
    AudioTooLong{samples:u64}, BadModel, IncompatibleVersion{got:u32,want:u32}, NotFitted, LabelLongerThanInput }
pub struct UttTrace { stage_ms:BTreeMap<&'static str,f32>, rtf:f32, t_frames:usize, active_tokens_peak:usize,
    am_score:f32, lm_score:f32, lattice_density:f32, mean_word_conf:f32, oov_count:u32, seed:u64 }
