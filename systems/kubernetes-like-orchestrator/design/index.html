Core schema & types (§11)

// apimachinery/runtime — the universal object contract (normative)
type Object interface {
    GetObjectKind() schema.ObjectKind        // apiVersion/kind
    DeepCopyObject() Object                   // value semantics; caches hand out copies
}
type Scheme interface {                       // type registry + conversion
    Recognizes(gvk GroupVersionKind) bool
    New(gvk GroupVersionKind) (Object, error)
    Convert(in Object, outVersion GroupVersion) (Object, error)
    DefaultFor(Object)                        // registered defaulters
}
type Codec interface { Encode(Object) ([]byte, error); Decode([]byte) (Object, GroupVersionKind, error) }
// store/mvcc — revisioned KV (normative)
type Revision int64                            // global, monotonic (I1)
type KV struct{ Key []byte; Value []byte; CreateRev, ModRev Revision; Version int64 }
type TxnRequest struct{ Compare []Cmp; Then, Else []Op } // CAS = compare ModRev
type WatchResponse struct{ Header Revision; Events []Event } // Event{PUT|DELETE, KV, PrevKV}
type Watchable interface {
    Range(key, end []byte, atRev Revision) ([]KV, Revision, error)
    Txn(TxnRequest) (committedRev Revision, succeeded bool, err error)
    Watch(key, end []byte, startRev Revision) (<-chan WatchResponse, CancelFunc)
    Compact(rev Revision) error                 // guarded by min-watcher RV (I14)
}
// apiserver/registry — generic REST contract (normative)
type Strategy interface {
    NamespaceScoped() bool
    PrepareForCreate(Object)                    // default, clear status
    ValidateCreate(Object) field.ErrorList
    PrepareForUpdate(new, old Object)           // preserve status/metadata, bump generation (I4)
    ValidateUpdate(new, old Object) field.ErrorList
    Canonicalize(Object)
}
type Store interface {                          // generic over Object
    Create(ctx, Object) (Object, error)         // assigns uid, RV, creationTimestamp (I2)
    Get(ctx, name string, rv string) (Object, error)
    List(ctx, Selector) (ObjectList, error)
    Update(ctx, Object, precondition RV) (Object, error) // CAS on RV (I3)
    Delete(ctx, name string, opts) (Object, error)       // finalizers → soft delete (I6)
    Watch(ctx, Selector, fromRV string) (watch.Interface, error) // (I8)
}
// client/informer — local cache + delta delivery (normative)
type SharedInformer interface {
    AddEventHandler(ResourceEventHandler)       // OnAdd/OnUpdate/OnDelete
    GetStore() cache.Store                       // indexed by namespace/name
    HasSynced() bool                             // initial list complete
    Run(stop <-chan struct{})                    // list→watch→relist-on-Gone
}
type RateLimitingQueue interface {               // backpressure (principle 10)
    Add(item); AddRateLimited(item); AddAfter(item, d)
    Get() (item, shutdown bool); Done(item); Forget(item); Len() int
}
// scheduler/framework — plugin contracts (normative)
type FilterPlugin interface { Filter(state, pod *Pod, node *Node) *Status } // feasibility
type ScorePlugin  interface { Score(state, pod *Pod, node string) (int64, *Status) } // 0..100
type BindPlugin   interface { Bind(state, pod *Pod, node string) *Status }  // CAS nodeName (I5)
type Framework interface { RunFilters(...); RunScores(...); RunBind(...) }
// kubelet — pod sync (normative)
type PodWorker interface {
    UpdatePod(UpdatePodOptions)                  // create|update|kill, serialized per-pod uid
}
type SyncHandler interface {
    SyncPod(pod *Pod, status *PodStatus) (SyncResult, error)   // converge to spec
    KillPod(pod *Pod, gracePeriod int64) error
}
// CRI client
type RuntimeService interface {
    RunPodSandbox(cfg) (sandboxID string, err error)
    CreateContainer(sandboxID string, cfg) (id string, err error)
    StartContainer(id string) error
    StopContainer(id string, timeout int64) error
    ContainerStatus(id string) (Status, error)
}
// controllers — common reconcile shape (normative)
type Reconciler interface {
    Reconcile(ctx, key string) (Result, error)   // idempotent (principle 5); requeue via Result
}
type Result struct{ Requeue bool; RequeueAfter time.Duration }
type Lease struct{ HolderIdentity string; LeaseDurationSeconds int32; RenewTime MicroTime } // I10
// common error taxonomy (normative)
type StatusError struct{ Code int; Reason string }  // NotFound404 Conflict409 Invalid422 Forbidden403 Gone410 ServerTimeout504
type PhaseDeferredError struct{ Ref string }        // Ref is a real §N.10 anchor (see the Deferred Features Register)
