Core schema & types (§11)

pub struct ComputedStyle { /* resolved values for all PropertyId; Arc-shared; inherited subset */ }
pub enum Origin { UserAgent, User, Author }                  // cascade origin precedence, low->high
pub struct MatchedRules { /* declarations for an element: (Origin, Specificity, source order) */ }
pub struct StyleResolver<'m, M: Matcher> { /* matcher: &'m M (258), stylesheets, sharing_cache */ }
pub struct Cascade { /* per-property winner: sort by (Origin,!important,Specificity,order) */ }
pub struct PropertyId(u16);                                  // interned CSS property handle
pub enum BoxKind { Block, Inline, InlineBlock, Anonymous, Replaced, Text, OutOfFlow, None }
pub struct Box { pub kind: BoxKind, pub style: Arc<ComputedStyle>, pub children: Vec<Box> }
pub enum FormattingContext { Block, Inline, Flex, Grid }     // assigned per Box establishing a BFC/IFC
pub struct LayoutRect { pub x: f32, pub y: f32, pub w: f32, pub h: f32 }  // px, frame-local
pub struct Fragment { pub rect: LayoutRect, pub style: Arc<ComputedStyle>, pub kind: BoxKind, pub children: Vec<Fragment> }
pub struct FragmentTree { pub root: Fragment }
pub struct PropertyTrees { pub transform: TransformTree, pub clip: ClipTree, pub effect: EffectTree, pub scroll: ScrollTree }
pub struct PaintPropertyNode { /* parent id + local transform|clip|effect|scroll state */ }
pub struct StackingContext { pub z_index: i32, pub establishes: bool }
pub enum DisplayItem { Rect{..}, Text{..}, Image{..}, Border{..}, BoxShadow{..},
                       BeginClip{ node: ClipId }, EndClip, BeginEffect{ node: EffectId }, EndEffect }
pub struct PaintChunk { pub items: core::ops::Range<usize>, pub property_state: PropertyTreeState, pub visual_rect: LayoutRect }
pub struct DisplayList { pub items: Vec<DisplayItem>, pub chunks: Vec<PaintChunk> }
pub enum CompositingReason { Transform3D, WillChange, Animation, ScrollDependent, Overlap, Video, IFrame }
pub struct CompositorLayer { pub transform_node: TransformId, pub content: core::ops::Range<usize>,
                             pub reasons: Vec<CompositingReason>, pub tiles: Vec<Tile> }
pub struct LayerTree { pub layers: Vec<CompositorLayer> }
pub struct Tile { pub rect: LayoutRect, pub state: RasterState }   // RasterState: Missing | Rastering | Ready
pub struct RasterTask { pub layer: LayerId, pub tile: TileId, pub chunk: PaintChunk }
pub struct CompositorFrame { pub layers: Vec<CompositorLayer>, pub transforms: TransformTree }
pub enum LifecycleStage { Clean, Style, Layout, CompositingInputs, Prepaint, Paint, Commit, Composite }
pub struct DocumentLifecycle { pub stage: LifecycleStage }
pub struct InvalidationFlags { /* bitflags: NEEDS_STYLE, NEEDS_LAYOUT, NEEDS_PAINT, NEEDS_COMPOSITE */ }
pub struct RenderObjectId(u32);                              // stable identity DOM<->box<->fragment
pub struct FrameScheduler { /* vsync_interval, raf_callbacks: Vec<RafCallback>, pending_commit */ }
pub struct Commit { /* snapshot handed main -> compositor thread */ }
pub fn update_lifecycle(doc: &mut DocumentLifecycle, target: LifecycleStage); // runs only dirty stages up to target
NEEDS_STYLE     -> Style  -> Layout -> Paint -> Composite
NEEDS_LAYOUT    ->          Layout -> Paint -> Composite
NEEDS_PAINT     ->                    Paint -> Composite
NEEDS_COMPOSITE ->                             Composite   (compositor-only; cheap, jank-free)
