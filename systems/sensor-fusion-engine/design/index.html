Core schema & types (§11)

// sf-linalg
pub trait Scalar: Copy + /* float ops */ {}
pub struct Vector<S: Scalar, const N: usize> { data: [S; N] }
pub struct Matrix<S: Scalar, const R: usize, const C: usize> { data: [[S; C]; R] }
impl<S,const N:usize> Vector<S,N>{ fn dot(&self,o:&Self)->S; fn norm(&self)->S; }
impl<S,const R:usize,const C:usize> Matrix<S,R,C>{
    fn transpose(&self)->Matrix<S,C,R>;
    fn mul<const K:usize>(&self,o:&Matrix<S,C,K>)->Matrix<S,R,K>;     // dim-checked at compile time
}
pub struct Cholesky<S,const N:usize>{ l: Matrix<S,N,N> }
impl<S,const N:usize> Cholesky<S,N>{
    fn decompose(a:&Matrix<S,N,N>)->Result<Self,LinalgError>;          // fails if not PD  (I3,I9)
    fn solve<const M:usize>(&self, b:&Matrix<S,N,M>)->Matrix<S,N,M>;   // S⁻¹·b without explicit inverse
}
pub enum LinalgError{ NotPositiveDefinite, Singular, DimMismatch }

// sf-math (Hamilton, body→nav, NED default — LOCKED)
pub struct Quaternion<S>{ w:S, x:S, y:S, z:S }                         // |q|=1 invariant (I1)
impl<S> Quaternion<S>{
    fn identity()->Self; fn normalize(&mut self); fn conj(&self)->Self;
    fn mul(&self,o:&Self)->Self;                                       // Hamilton product
    fn exp(phi:&Vector<S,3>)->Self;                                    // R³→S³
    fn log(&self)->Vector<S,3>;
    fn rotate(&self,v:&Vector<S,3>)->Vector<S,3>;                      // body→nav
    fn to_dcm(&self)->Matrix<S,3,3>;
}
pub enum Frame{ Body, NedNav, Enu, Ecef, Llh }                         // single-sourced (I8)
pub struct So3<S>(Matrix<S,3,3>);
pub fn skew<S>(v:&Vector<S,3>)->Matrix<S,3,3>;                         // [·]_×
pub fn llh_to_ecef<S>(llh:&Vector<S,3>)->Vector<S,3>;
pub fn ecef_to_ned<S>(p:&Vector<S,3>, ref_llh:&Vector<S,3>)->Vector<S,3>;
pub fn gravity_ned<S>(lat:S, alt:S)->Vector<S,3>;
// sf-state
pub struct StateVector<S>{ p:Vector<S,3>, v:Vector<S,3>, q:Quaternion<S>,
                           b_g:Vector<S,3>, b_a:Vector<S,3> }          // nominal, dim 16
pub struct ErrorState<S>{ dx:Vector<S,15> }                           // δp,δv,δθ,δb_g,δb_a
pub struct Covariance<S>{ p:Matrix<S,15,15> }                          // SPD invariant (I2,I3)
impl<S> Covariance<S>{
    fn symmetrize(&mut self);                                          // P=(P+Pᵀ)/2
    fn assert_spd(&self)->Result<(),LinalgError>;                      // Cholesky probe
    fn joseph_update(&mut self, k:&Matrix<S,15, /*m*/usize>, h:&/*..*/, r:&/*..*/);
}
impl<S> StateVector<S>{
    fn inject(&mut self, dx:&ErrorState<S>);                          // ⊞: q←q⊗Exp(δθ), renormalize
}
pub struct Snapshot<S>{ x:StateVector<S>, p:Covariance<S>, t:Timestamp, health:Health }
// published lock-free (double-buffer) for Estimate API readers

// sf-sensor
pub struct Timestamp{ ns:u64 }                                        // monotonic core clock (I4)
pub enum Measurement<S>{
    GpsPos{ p_nav:Vector<S,3>, r:Matrix<S,3,3>, lever:Vector<S,3>, t:Timestamp },
    GpsVel{ v_nav:Vector<S,3>, r:Matrix<S,3,3>, t:Timestamp },
    GpsHeading{ psi:S, r:S, t:Timestamp },
    Mag{ m_body:Vector<S,3>, r:Matrix<S,3,3>, t:Timestamp },
    Baro{ alt:S, r:S, t:Timestamp },
    WheelOdom{ v_body:Vector<S,3>, r:Matrix<S,3,3>, t:Timestamp },
    Zupt{ r:Matrix<S,3,3>, t:Timestamp },
    Imu{ gyro:Vector<S,3>, accel:Vector<S,3>, t:Timestamp },          // drives predict, not "update"
    VoLo{ d_p:Vector<S,3>, d_theta:Vector<S,3>, r:Matrix<S,6,6>, t:Timestamp },
}
pub trait SensorModel<S, const M: usize>{
    fn predict_meas(&self, x:&StateVector<S>)->Vector<S,M>;            // h(x)
    fn jacobian(&self, x:&StateVector<S>)->Matrix<S,M,15>;            // H (error-state)
    fn noise(&self)->Matrix<S,M,M>;                                   // R
}
pub trait SensorSource<S>{ fn poll(&mut self)->Option<Measurement<S>>; }
// sf-sync
pub struct MeasurementBuffer<S, const CAP: usize>{ /* ring, ts-ordered */ }
impl<S,const CAP:usize> MeasurementBuffer<S,CAP>{
    fn push(&mut self, m:Measurement<S>)->Result<(),BufferFull>;       // backpressure (drop oldest + count)
    fn drain_until(&mut self, t:Timestamp)->impl Iterator<Item=Measurement<S>>; // in-order
    fn interpolate_imu(&self, t:Timestamp)->Option<Measurement<S>>;
}

// sf-filter
pub trait Filter<S>{
    fn predict(&mut self, imu:&Measurement<S>, dt:S);                  // calls sf-imu mechanization
    fn update<const M:usize>(&mut self, m:&dyn SensorModel<S,M>, z:&Vector<S,M>) -> UpdateOutcome<S>;
    fn snapshot(&self)->Snapshot<S>;
    fn reset_attitude(&mut self, accel:&Vector<S,3>, mag:&Vector<S,3>);
}
pub struct Ekf<S>{ x:StateVector<S>, p:Covariance<S>, /*..*/ }
pub struct Eskf<S>{ x:StateVector<S>, p:Covariance<S>, /*..*/ }        // recommended core
pub struct Ukf<S>{ /* alpha,beta,kappa, sigma scratch (fixed) */ }
pub struct Kf<S>{ /* linear */ }
pub struct Complementary<S>{ kp:S, ki:S, q:Quaternion<S> }
pub enum UpdateOutcome<S>{ Applied{ nis:S }, Rejected{ nis:S, dof:u8 }, Skipped{ reason:SkipReason } }

// sf-gate
pub struct GateResult{ accepted:bool, d2:f64, threshold:f64, dof:u8 }
pub struct ConsistencyStats{ nis_window:RollingChi2, nees_window:RollingChi2,
                             gated_total:u64, accepted_total:u64, diverged:bool }
impl ConsistencyStats{
    fn observe_nis(&mut self, nis:f64, dof:u8);
    fn nis_in_bounds(&self)->bool;                                    // I10
    fn detect_divergence(&self)->bool;
}
pub struct AdaptiveNoise<S>{ /* innovation-based R/Q scaling */ }

// sf-state health / errors
pub struct Health{ status:FilterStatus, quat_norm_err:f32, cov_not_pd:bool,
                   gated_total:u64, nis_ok:bool, observability:ObsFlags }
pub enum FilterStatus{ Uninit, Aligning, Running, Degraded, DeadReckon, Reinit }
pub enum FusionError{ NotPositiveDefinite, TimeRegression, BufferFull,
                      BadMeasurement, ModelDimMismatch }
