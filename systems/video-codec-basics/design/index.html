Core schema & types (§11)

pub struct Plane {
    pub data:   Vec<u8>,   // row-major samples, length == stride * height
    pub width:  u32,
    pub height: u32,
    pub stride: u32,       // bytes per row, >= width; may include padding
}
pub struct PlaneView<'a>  { data: &'a [u8],     width: u32, height: u32, stride: u32 }
pub struct PlaneMut<'a>   { data: &'a mut [u8], width: u32, height: u32, stride: u32 }
pub struct Frame {
    pub planes:      [Plane; 3],   // [0]=Y luma, [1]=Cb chroma, [2]=Cr chroma (4:2:0)
    pub poc:         i32,           // picture order count; display order
    pub frame_type:  FrameType,
}
pub type FrameRef = Arc<Frame>;    // shared ownership in DPB reference lists
pub enum FrameType { I, P, B }
pub enum MbType {
    IntraNxN,                   // 4x4 luma partitions, each with an IntraMode
    Intra16x16(IntraMode),      // single 16x16 luma prediction mode
    Inter(PartShape),           // inter-coded with given partition shape
    Skip,                       // zero MV, zero residual; copied from spatial/temporal predictor
    PCM,                        // raw PCM escape (edge case; no transform/quant)
}
pub struct Macroblock {
    pub mb_type:   MbType,
    pub qp:        Qp,
    pub parts:     Vec<Partition>,      // 1..16 partitions depending on MbType
    pub residual:  Vec<TransformBlock>, // one per 4x4 luma block + chroma blocks
    pub cbp:       u8,                  // coded-block-pattern (which 8x8 sub-blocks have coeffs)
}
pub struct Partition {
    pub ref_idx: u8,            // index into RefList (L0 for P; L0/L1 for B)
    pub mv:      MotionVector,  // quarter-pel MV for this partition
    pub mv_l1:   Option<MotionVector>, // L1 MV for B-slice bi-prediction
}
pub struct MotionVector { pub x: i16, pub y: i16 }  // units: quarter-pel luma samples
pub enum IntraMode {
    Dc, Vertical, Horizontal, DiagDownLeft, DiagDownRight,
    VertRight, HorizDown, VertLeft, HorizUp,  // 9 4x4 modes
    Plane,                                    // additional for 16x16 / chroma
}
pub struct TransformBlock {
    pub coeffs: [i32; 16],   // 4x4 DCT coefficients; 64-element array for 8x8 blocks (future)
    pub is_8x8: bool,
}
pub struct Qp(pub u8);   // invariant: 0 <= value <= 51
pub enum PartShape {
    P16x16, P16x8, P8x16,
    P8x8,                    // enables sub-MB partition (each 8x8 further split)
    P8x4, P4x8, P4x4,       // sub-MB partition shapes (only valid inside P8x8)
}
pub struct MvPredictor { pub mv: MotionVector }  // median of A, B, C neighbors
pub struct MvDelta     { pub dx: i16, pub dy: i16 }  // signed difference vs predictor
pub struct BitWriter { /* internal: Vec<u8> buffer, bit offset */ }
pub struct BitReader<'a> { /* internal: &'a [u8], bit offset */ }

impl BitWriter {
    pub fn write_bit(&mut self, bit: u8);
    pub fn write_bits(&mut self, val: u32, n: u8);
    pub fn write_ue(&mut self, val: u32);          // exp-Golomb unsigned
    pub fn write_se(&mut self, val: i32);          // exp-Golomb signed
    pub fn into_bytes(self) -> Vec<u8>;
}
impl<'a> BitReader<'a> {
    pub fn read_bit(&mut self) -> Result<u8, CodecError>;
    pub fn read_bits(&mut self, n: u8) -> Result<u32, CodecError>;
    pub fn read_ue(&mut self) -> Result<u32, CodecError>;
    pub fn read_se(&mut self) -> Result<i32, CodecError>;
    pub fn bytes_remaining(&self) -> usize;
}
pub struct NalUnit {
    pub nal_type: NalType,
    pub rbsp:     Vec<u8>,   // RBSP payload after emulation-prevention unstuffing
}
pub enum NalType {
    NonIdrSlice = 1, IdrSlice = 5,
    Sei = 6, Sps = 7, Pps = 8,
    AudDelimiter = 9, EndOfSeq = 10, EndOfStream = 11,
}
pub struct SliceHeader {
    pub slice_type:  SliceType,   // I, P, B (mapped from H.264 slice_type values 0..9)
    pub frame_num:   u32,         // wraps at MaxFrameNum (SPS-derived power of two)
    pub poc:         i32,
    pub ref_list_l0: RefList,
    pub ref_list_l1: RefList,     // empty for I/P slices
    pub qp_delta:    i8,          // slice_qp_delta; base QP = PPS.qp + qp_delta
}
pub enum SliceType { I, P, B }
pub struct RefList {
    pub frames: Vec<FrameRef>,   // ordered list of decoded reference pictures (L0 or L1)
}
pub struct Dpb {
    pub frames:    Vec<FrameRef>,         // all decoded pictures currently held
    pub max_size:  usize,                 // capacity = min(sps.max_ref_frames, 4)
}
impl Dpb {
    pub fn insert(&mut self, frame: FrameRef, marking: PicMarking);
    pub fn evict_unused(&mut self);
    pub fn build_ref_list_l0(&self, cur_poc: i32) -> RefList;
    pub fn build_ref_list_l1(&self, cur_poc: i32) -> RefList;
    pub fn flush(&mut self);             // IDR: mark all as unused, drain
}
pub enum PicMarking { ShortTerm(u32), LongTerm(u32), Unused }
pub struct EncodeParams {
    pub width:        u32,
    pub height:       u32,
    pub fps_num:      u32,
    pub fps_den:      u32,
    pub gop_size:     u32,       // IDR interval in frames
    pub b_frames:     u8,        // B-frames between consecutive P-frames (0 = no B)
    pub qp:           Option<Qp>, // constant QP if Some; rate control if None
    pub target_kbps:  Option<u32>,
    pub color_matrix: ColorMatrix,
}
pub struct GopConfig {
    pub idr_interval:     u32,
    pub b_cadence:        u8,        // e.g. 2 = IBBP pattern
    pub hierarchical_b:   bool,
}
pub struct RdCost { pub distortion: u32, pub bits: u32, pub lambda: f32 }
// cost = distortion + lambda * bits; lambda fixed per QP (not persisted)
pub enum CodecError {
    InvalidQp(u8),
    BitstreamOverrun,
    EmulationPreventionError,
    ExpGolombOverflow,
    UnsupportedSyntax(&'static str),
    NalTypeMismatch { expected: NalType, got: NalType },
    DpbFull,
    InvalidMotionVector(i16, i16),
    InvariantViolation(&'static str),
}
pub type CodecResult<T> = Result<T, CodecError>;
