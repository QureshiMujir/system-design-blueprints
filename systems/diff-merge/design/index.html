Core schema & types (§11)

pub struct Token<'a> { pub bytes: &'a [u8], pub hash: u64 }
pub enum Granularity { Byte, Line, Word, Char, Token /* deferred §13.2 */ }
pub trait Tokeniser {
    fn tokenise<'a>(&self, input: &'a [u8]) -> Vec<Token<'a>>;
    fn granularity(&self) -> Granularity;
}
pub struct LineTokeniser;
pub struct ByteTokeniser;
pub struct WordTokeniser { pub include_punctuation: bool }
pub struct CharTokeniser;
pub enum EditOp {
    Equal   { old_range: Range<usize>, new_range: Range<usize> },
    Delete  { old_range: Range<usize>, new_pos: usize },
    Insert  { new_range: Range<usize>, old_pos: usize },
    Replace { old_range: Range<usize>, new_range: Range<usize> },
}
pub struct EditScript {
    pub ops: Vec<EditOp>, pub old_len: usize, pub new_len: usize,
    pub algorithm: AlgorithmId, pub granularity: Granularity,
}
pub enum AlgorithmId { Myers, Patience, Histogram, WuOnp, HuntMcIlroy }
pub struct Patch {
    pub old_path: Option<String>, pub new_path: Option<String>,
    pub old_mode: Option<u32>, pub new_mode: Option<u32>,
    pub raw_header: Vec<RawHeaderLine>, pub hunks: Vec<Hunk>,
}
pub struct Hunk {
    pub old_range: Range<usize>, pub new_range: Range<usize>,   // 1-based (§8.2)
    pub function_context: Option<String>, pub lines: Vec<HunkLine>,
}
pub enum HunkLine { Context(Vec<u8>), Removed(Vec<u8>), Added(Vec<u8>), NoNewlineAtEof }
pub enum RawHeaderLine {
    GitIndex { old_sha: String, new_sha: String, mode: Option<u32> },
    GitRename { from: String, to: String, similarity: u8 },
    GitCopy   { from: String, to: String, similarity: u8 },
    NewFile { mode: u32 }, DeletedFile { mode: u32 }, Binary,
}
pub struct MergeResult { pub blocks: Vec<Block>, pub conflict_count: usize, pub strategy: MergeStrategy }
pub enum Block { Resolved(Vec<u8>), Conflict { ours: Vec<u8>, base: Vec<u8>, theirs: Vec<u8>, labels: ConflictLabels } }
pub struct ConflictLabels { pub ours_label: String, pub base_label: String, pub theirs_label: String }
pub enum MergeStrategy { Weave, Recursive, Ort }
pub struct MergeOpts { pub strategy: MergeStrategy, pub labels: ConflictLabels, pub max_recursion_depth: u32, pub whitespace_policy: WhitespacePolicy }
pub enum WhitespacePolicy { Strict, IgnoreAllSpace }
pub struct SimilarityScore(pub u8); // 0..=100
pub struct SimilarityIndex { pub shingle_size: usize, pub window_size: usize, pub fingerprints: Vec<u64> }
pub struct RenameOpts { pub min_similarity: SimilarityScore, pub max_pairs: usize, pub detect_copies: bool }
pub enum TreeOp {
    Add    { path: String, mode: u32, content: Vec<u8> },
    Delete { path: String, mode: u32 },
    Modify { path: String, old_mode: u32, new_mode: u32, patch: Patch },
    Rename { from: String, to: String, similarity: SimilarityScore, patch: Option<Patch> },
    Copy   { from: String, to: String, similarity: SimilarityScore, patch: Option<Patch> },
    ModeChange { path: String, old_mode: u32, new_mode: u32 },
}
pub struct JsonPatch { pub ops: Vec<JsonOp> }
pub enum JsonOp {
    Add { path: String, value: serde_json::Value }, Remove { path: String },
    Replace { path: String, value: serde_json::Value },
    Move { from: String, path: String }, Copy { from: String, path: String },
    Test { path: String, value: serde_json::Value },
}
pub struct JsonMergePatch(pub serde_json::Value);
pub trait Diff {
    type TokenT<'a>: PartialEq + Eq + Copy where Self: 'a;
    type Error: std::error::Error;
    fn diff<'a>(&self, old: &'a [Self::TokenT<'a>], new: &'a [Self::TokenT<'a>])
        -> Result<EditScript, Self::Error>;
}
pub trait Apply {
    type Error: std::error::Error;
    fn apply(&self, old: &[u8], patch: &Patch) -> Result<Vec<u8>, Self::Error>;
}
pub trait Merge {
    type Error: std::error::Error;
    fn three_way(&self, base: &[u8], ours: &[u8], theirs: &[u8], opts: MergeOpts)
        -> Result<MergeResult, Self::Error>;
}
pub trait StructuredDiff<T> {
    type Patch;
    type Error: std::error::Error;
    fn diff(&self, old: &T, new: &T) -> Result<Self::Patch, Self::Error>;
    fn apply(&self, old: &T, patch: &Self::Patch) -> Result<T, Self::Error>;
}
pub enum MyersError { CostBudgetExceeded, DidNotConverge }
pub enum ParseError { HunkLengthMismatch { expected: usize, actual: usize }, OverlappingHunks, MalformedHeader }
pub enum ApplyError { ContextMismatch { hunk_idx: usize, line: usize }, NoNewlineMismatch }
pub enum MergeError { RecursionDepthExceeded { depth: u32 }, InvalidMergeBase }
pub enum SemanticError { NotImplemented }
