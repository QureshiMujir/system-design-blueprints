pub trait Hasher: Send + Sync + 'static {
    /// Single-shot 128-bit hash of a byte slice with a 128-bit seed.
    fn hash128(&self, bytes: &[u8], seed: u128) -> u128;
}

pub struct XxHash3;
pub struct WyHash;
pub struct Murmur3_128;
pub struct SipHash13;

impl Hasher for XxHash3 { /* signatures provided in src */ }

/// Kirsch-Mitzenmacher double-hashing: k indices in [0,m) via h_i=(h1+i*h2) mod m.
pub fn kirsch_mitzenmacher(h: u128, k: u32, m: u64) -> KmIter;

pub struct Bloom<H: Hasher = XxHash3> { /* bits, m, k, seed:u128, items, hasher:H */ }
impl Bloom {
    pub fn sized_for(n: u64, f: f64) -> Result<Self, Error>;     // m=⌈-n·ln f/(ln2)²⌉, k=round((m/n)·ln2)
    pub fn capacity_for(memory_bytes: usize, f: f64) -> u64;
}

pub struct CountingBloom<H: Hasher = XxHash3, C: Counter = U4> { /* counters, m, k, seed, items, hasher */ }
pub trait Counter: Copy + Send + Sync {
    const MAX: u64;
    fn saturating_add(self, n: u64) -> Self;
    fn saturating_sub(self, n: u64) -> Self;
}
// impl Counter for U4 (MAX=15), u8 (MAX=255), u16 (MAX=65535)
