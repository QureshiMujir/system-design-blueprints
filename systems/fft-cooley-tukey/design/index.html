Core schema & types (§11)

pub trait Float: Copy + Send + Sync { /* f32 | f64 */
    const EPS: Self; const PI: Self; const ZERO: Self; const ONE: Self;
    fn mul_add(self, b: Self, c: Self) -> Self;     // FMA
    fn sin_cos(self) -> (Self, Self);               // for direct-trig twiddles
}
#[repr(C)] pub struct Complex<T: Float> { pub re: T, pub im: T }   // interleaved, C `_Complex`-compatible
impl<T: Float> Complex<T> { fn mul(self, o:Self)->Self; fn conj(self)->Self; fn add(self,o:Self)->Self; fn scale(self,s:T)->Self; }
#[repr(i32)] pub enum Direction { Forward = -1, Inverse = 1 }   // sign of the exponent
pub enum Normalization { None, OneOverN, Unitary /* 1/√N */ }
pub enum Layout { InPlace, OutOfPlace }
pub struct Plan<T: Float> {            // built once, executed many (ADR-6)
    n: usize, dir: Direction, norm: Normalization, layout: Layout,
    tree: PlanTree,                    // codelet decomposition
    twiddles: Arc<TwiddleSet<T>>,      // shared, cached
    scratch: AlignedArena<Complex<T>>, // owned, reused; no alloc in execute (I14)
    isa: Isa, plan_hash: u64,
}
impl<T: Float> Plan<T> {
    fn execute(&self, input: &[Complex<T>], output: &mut [Complex<T>]) -> Result<(), FftError>; // I1,I14
    fn execute_inplace(&self, data: &mut [Complex<T>]) -> Result<(), FftError>;
}
pub enum PlanTree {
    Radix2Dit { stages: u32 }, Radix2Dif { stages: u32 }, Radix4 { stages: u32 },
    SplitRadix { n: usize },
    MixedRadix { n1: usize, n2: usize, inner: Box<(PlanTree, PlanTree)>, twiddle_id: u64 },
    Codelet { radix: u8 },                       // hand kernels r3/r5/r7/r8
    Bluestein { n: usize, m: usize /*≥2n-1, pow2*/, inner: Box<PlanTree> },
    Rader { p: usize, g: usize /*primitive root*/, inner: Box<PlanTree> },
    Real { half: Box<PlanTree> },                // RFFT/IRFFT wrapper
}
pub struct TwiddleSet<T: Float> { tables: HashMap<TwiddleKey, AlignedVec<Complex<T>>> }
struct TwiddleKey { n: usize, kind: TwiddleKind }     // Forward / SplitRadix-W3 / Chirp
pub enum TwiddleKind { Dft, SplitRadixW3, Chirp { m: usize } }
fn gen_trig<T>(n: usize) -> AlignedVec<Complex<T>>;   // W[k]=(cos(-2πk/n), sin(-2πk/n)), direct (I7)
pub struct Cplx<T> { /* vector of complex: AoS or planar re/im per ADR-23 */ }
type Butterfly<T> = unsafe fn(data: *mut Complex<T>, n: usize, stride: usize, tw: *const Complex<T>);
struct CodeletSet<T> { r2_dit: Butterfly<T>, r2_dif: Butterfly<T>, r4: Butterfly<T>, sr: Butterfly<T>, /* r3/r5/r7/r8 */ }
pub enum PlanFlags { Estimate, Measure, Patient }     // ↔ FFTW_ESTIMATE/MEASURE/PATIENT
fn plan_c2c<T>(n: usize, dir: Direction, norm: Normalization, layout: Layout, flags: PlanFlags) -> Result<Plan<T>, FftError>;
fn factor(n: usize) -> Vec<(usize /*prime*/, u32 /*power*/)>;   // trial division
fn cost_estimate(tree: &PlanTree) -> f64;                       // ESTIMATE cost model
fn plan_r2c<T>(n: usize, layout: Layout, flags: PlanFlags) -> Result<RealPlan<T>, FftError>;   // real→Hermitian (N/2+1) (I6)
fn plan_c2r<T>(n: usize, layout: Layout, flags: PlanFlags) -> Result<RealPlan<T>, FftError>;   // Hermitian→real
fn plan_2d<T>(rows: usize, cols: usize, dir: Direction, norm: Normalization, flags: PlanFlags) -> Result<Plan2D<T>, FftError>;
fn fft_convolve<T>(a: &[Complex<T>], b: &[Complex<T>], mode: ConvMode) -> Result<Vec<Complex<T>>, FftError>; // Linear|Circular
fn overlap_add<T>(signal: &[T], kernel: &[T], block: usize) -> Vec<T>;   // streaming FIR (DSP consumer)
pub struct Wisdom { entries: HashMap<WisdomKey, PlanTree> }
struct WisdomKey { n: usize, dir: Direction, layout: Layout, isa: Isa, dtype: DType }   // (I9)
fn export_wisdom(w: &Wisdom, path: &Path) -> io::Result<()>;     // text/binary; FFTW-interop documented
fn import_wisdom(path: &Path) -> io::Result<Wisdom>;
pub struct Machine { l1d:usize, l2:usize, l3:usize, line:usize, vregs:usize, fma_peak_gflops:f64, isa:Isa, cores:usize }
pub enum Isa { ScalarFallback, Avx2Fma, Avx512, Neon }
pub enum DType { F32, F64 }
pub enum FftError { BadSize(usize), NullPtr, Misaligned{need:usize}, BufferLen{expected:usize,got:usize},
                    IllegalFlags, AliasViolation, ResourceCap{requested:usize}, WisdomCorrupt }
pub struct PlanTrace { n:usize, factors:Vec<(usize,u32)>, dir:Direction, layout:Layout, isa:Isa,
                       codelets:Vec<&'static str>, twiddle:&'static str,
                       plan_ns:u64, exec_ns:u64, gflops:f64, accum_order_id:u64, plan_hash:u64 }
