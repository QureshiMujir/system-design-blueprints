Core schema & types (§11)

trait DocumentReplica {
    type Op;                                  // engine-native op/element
    fn apply_local(&mut self, edit: LocalEdit) -> Self::Op;     // UI intent → native, mutate
    fn integrate_remote(&mut self, op: Self::Op);               // remote native → mutate
    fn text(&self) -> &str;                                     // materialized doc
    fn version(&self) -> VersionVector;                         // causal summary
    fn size_report(&self) -> SizeReport;                        // honest memory (Principle 7)
}
enum LocalEdit { Insert{ at: usize, text: String }, Delete{ at: usize, len: usize } }
enum RemoteEffect { Inserted{ at: usize, text: String }, Deleted{ at: usize, len: usize } }
enum Component { Retain(usize), Insert(String), Delete(usize) }
struct Op { components: Vec<Component>, base_len: usize, target_len: usize }
fn transform(a: &Op, b: &Op, prio: Priority) -> (Op, Op)   // (a', b'), TP1
fn compose(a: &Op, b: &Op) -> Op
fn invert(a: &Op, doc: &str) -> Op
enum Priority { Left, Right }                              // deterministic tie-break
struct RevisionLog { entries: Vec<(u64 /*rev*/, Op, SiteId /*origin*/)> }
struct ServerState { doc: Rope, log: RevisionLog, rev: u64 }
struct ClientState { doc: Rope, acked_rev: u64, pending: VecDeque<Op> }
struct RgaId { site: u32, seq: u64 }                  // Ord: by (seq desc, site)
struct PosId { path: SmallVec<[(u64,u32);8]> }         // Logoot/LSEQ; lexicographic Ord
struct Element { id: RgaId, value: char, deleted: bool, origin_left: RgaId }   // RGA
struct LogootEntry { pos: PosId, value: char, deleted: bool }                  // Logoot/LSEQ
type Tombstone = Element;   // with deleted=true (RGA) — kept for ordering until GC
struct Lamport(u64)
struct VectorClock(BTreeMap<SiteId, u64>)        // deterministic iteration
struct VersionVector(BTreeMap<SiteId, u64>)      // = max seq seen per site
struct Dot { site: SiteId, seq: u64 }
struct DVV { dot: Dot, vv: VersionVector }       // dotted version vector (siblings)
struct Presence { site: SiteId, cursor: CursorPos, selection: Option<Range>, name: String, color: u32, last_seen: Instant }
enum CursorPos { Index(usize) /*OT*/, RelativeTo(RgaId, i32) /*CRDT*/ }
struct UndoManager { stacks: HashMap<SiteId, Vec<UndoEntry>> }   // per-user
enum UndoEntry { Ot(Op /*inverse*/), Crdt(Vec<CrdtId> /*to re-tombstone/restore*/) }
struct SizeReport { live_chars: usize, tombstones: usize, meta_bytes: usize, id_bytes: usize, oplog_bytes: usize }
enum CollabError { MalformedOp, BaseLenMismatch, RevisionDesync, CausalHoldback,
                   TombstoneResurrection, LimitExceeded(LimitKind), Codec(CodecError), Transport(TransportError) }
