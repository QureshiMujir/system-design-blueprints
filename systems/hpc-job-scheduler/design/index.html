Core schema & types (§11)

// job.rs
#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
pub struct JobId(pub u64);

#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub struct StepId { pub job: JobId, pub step: u32 }

#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub struct NodeId(pub u32);
// job.rs
#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
pub enum JobState {
    Pending, Configuring, Running, Completing,
    Completed, Failed, Cancelled, Preempted, Requeued,
}
// job.rs
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Job {
    pub id:          JobId,
    pub user:        String,
    pub account:     String,
    pub spec:        ResourceSpec,
    pub state:       JobState,
    pub priority:    u32,
    pub submit_time: SystemTime,
    pub start_time:  Option<SystemTime>,
    pub end_time:    Option<SystemTime>,
    pub alloc:       Option<Allocation>,
    pub exit_code:   Option<i32>,
    pub requeue:     bool,
    pub array_id:    Option<JobId>,   // non-None for array element jobs
}
// resource.rs
#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub struct ResourceSpec {
    pub nodes:         u32,            // number of nodes requested
    pub tasks:         u32,            // total MPI tasks / processes
    pub cpus_per_task: u16,
    pub mem_per_node:  u64,            // bytes
    pub gpus:          u16,            // GPUs per node
    pub time_limit:    Duration,       // wall-clock limit; enforced by controld timer
    pub partition:     String,
    pub qos:           Option<String>,
    pub account:       Option<String>,
    pub features:      Vec<String>,    // node feature constraints (AND-list)
    pub exclusive:     bool,           // whole-node allocation
}
// node.rs
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Node {
    pub id:          NodeId,
    pub hostname:    String,
    pub cpus:        u16,
    pub mem:         u64,            // bytes total
    pub gpus:        u16,
    pub features:    Vec<String>,
    pub state:       NodeState,
    pub alloc_cores: CoreBitmap,
    pub alloc_mem:   u64,
    pub alloc_gpus:  GpuSet,
    pub last_seen:   SystemTime,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
pub enum NodeState { Idle, Allocated, Mixed, Down, Drain, Draining }
// allocation.rs
#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub struct CoreBitmap(pub u128);   // up to 128 cores per node; bit i = core i allocated

#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub struct GpuSet(pub u64);        // up to 64 GPUs per node; bit i = GPU i allocated

impl CoreBitmap {
    pub fn allocate(&mut self, count: u16) -> Result<CoreBitmap, AllocError>;
    pub fn free(&mut self, mask: CoreBitmap);
    pub fn free_count(&self) -> u16;
    pub fn intersects(&self, other: &CoreBitmap) -> bool;
}
impl GpuSet {
    pub fn allocate(&mut self, count: u16) -> Result<GpuSet, AllocError>;
    pub fn free(&mut self, mask: GpuSet);
}
// partition.rs
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Partition {
    pub name:           String,
    pub node_set:       Vec<NodeId>,
    pub max_time:       Duration,
    pub max_nodes:      u32,
    pub priority_tier:  u8,           // additive to multifactor score
    pub preemptible:    bool,
    pub default:        bool,
    pub allow_accounts: Option<Vec<String>>,
}
// allocation.rs
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Allocation {
    pub job_id: JobId,
    pub nodes:  Vec<NodeAllocation>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct NodeAllocation {
    pub node_id: NodeId,
    pub cores:   CoreBitmap,
    pub gpus:    GpuSet,
    pub mem:     u64,
}

// reservation.rs
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Reservation {
    pub job_id:    JobId,             // top pending job this reserves for
    pub start:     SystemTime,        // earliest the top job can start
    pub node_set:  Vec<NodeId>,       // nodes reserved
    pub window:    Duration,          // duration held
}
// account.rs
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Account {
    pub name:          String,
    pub parent:        Option<String>,
    pub shares:        u32,           // relative share weight
    pub usage_decayed: f64,           // TRES-seconds decayed
    pub children:      Vec<String>,   // child account names
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Association {
    pub user:          String,
    pub account:       String,
    pub qos:           Vec<String>,
    pub max_jobs:      Option<u32>,
    pub max_tres:      Option<TresLimits>,
    pub grp_tres:      Option<TresLimits>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct TresLimits {
    pub cpus:  Option<u64>,
    pub mem:   Option<u64>,     // bytes
    pub gpus:  Option<u32>,
    pub nodes: Option<u32>,
}
// priority.rs
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PriorityFactors {
    pub age:       f64,        // clamped [0,1]: age / age_max
    pub fairshare: f64,        // [0,1]: from fair-share tree
    pub partition: f64,        // partition priority_tier / max_tier
    pub qos:       f64,        // QoS priority factor
    pub job_size:  f64,        // log-normalized requested TRES
    pub weights:   PriorityWeights,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PriorityWeights {
    pub age:       u32,
    pub fairshare: u32,
    pub partition: u32,
    pub qos:       u32,
    pub job_size:  u32,
}

// priority = sum(factor_i * weight_i) normalized to u32; I8
// hpcsched-sched (uses core types)
pub struct SchedulePass {
    pub started:    Vec<JobId>,        // jobs transitioned Pending → Configuring
    pub blocked:    Vec<(JobId, BlockReason)>,
    pub pass_time:  Duration,
}

pub struct BackfillPlan {
    pub reservation:   Reservation,
    pub backfilled:    Vec<(JobId, SystemTime, Allocation)>,  // (job, projected_start, alloc)
}

pub enum BlockReason {
    InsufficientNodes, TimeLimitExceeds, QosLimit, AssocLimit, PartitionLimit, NoFit,
}
// messages.rs — ControlMsg discriminants (no method bodies)
#[derive(Debug, Serialize, Deserialize)]
pub enum ControlMsg {
    Submit(SubmitReq),      SubmitAck(SubmitAck),
    Cancel(CancelReq),      CancelAck(CancelAck),
    StatusReq(StatusReq),   StatusResp(StatusResp),
    LaunchStep(LaunchMsg),  LaunchAck(LaunchAck),
    StepEvent(StepEventMsg), NodeRegister(NodeRegisterMsg),
    Heartbeat(HeartbeatMsg), HeartbeatAck,
    DrainReq(DrainReq),
    QueueQuery(QueueQueryReq), QueueResp(QueueResp),
}
#[derive(Debug, Serialize, Deserialize)]
pub struct SubmitReq   { pub script: String, pub spec: ResourceSpec, pub client_token: u64 }
#[derive(Debug, Serialize, Deserialize)]
pub struct SubmitAck   { pub job_id: JobId, pub client_token: u64 }
#[derive(Debug, Serialize, Deserialize)]
pub struct LaunchMsg   { pub job_id: JobId, pub step_id: StepId,
                         pub alloc: Allocation, pub script: String }
#[derive(Debug, Serialize, Deserialize)]
pub struct HeartbeatMsg {
    pub node_id: NodeId, pub free_cores: CoreBitmap,
    pub free_mem: u64,   pub free_gpus: GpuSet,
    pub running_steps: Vec<StepId>, pub timestamp: SystemTime,
}
#[derive(Debug, Serialize, Deserialize)]
pub struct StepEventMsg {
    pub step_id: StepId, pub event: StepEvent,
    pub node_id: NodeId, pub exit_code: Option<i32>,
}
#[derive(Debug, Clone, Copy, Serialize, Deserialize)]
pub enum StepEvent { Started, Completed, Failed }
#[derive(Debug, Serialize, Deserialize)]
pub struct NodeRegisterMsg {
    pub node_id: NodeId,  pub hostname: String,
    pub cpus: u16,        pub mem: u64,
    pub gpus: u16,        pub features: Vec<String>,
}
// record.rs
#[derive(Debug, Serialize, Deserialize)]
pub enum WalRecord {
    JobSubmitted  { job: Job },
    JobAllocated  { job_id: JobId, alloc: Allocation },
    JobState      { job_id: JobId, state: JobState, time: SystemTime },
    NodeState     { node_id: NodeId, state: NodeState, time: SystemTime },
    CheckpointRef { offset: u64 },       // WAL pointer to latest checkpoint
}

// wal.rs
pub struct StateLog { /* path, fd, crc_state */ }
impl StateLog {
    pub fn append(&mut self, record: &WalRecord) -> Result<u64, StateError>; // returns offset; fdatasync before return (P6)
    pub fn replay(&self, from: u64) -> impl Iterator<Item = Result<WalRecord, StateError>>;
}

// checkpoint.rs
pub struct Checkpoint { /* serialized ControllerState snapshot */ }
impl Checkpoint {
    pub fn write_atomic(path: &Path, state: &ControllerState) -> Result<(), StateError>; // write tmp, rename
    pub fn read(path: &Path) -> Result<Checkpoint, StateError>;
}
// record.rs
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct JobAcctRecord {
    pub job_id:      JobId,
    pub user:        String,
    pub account:     String,
    pub partition:   String,
    pub submit_time: SystemTime,
    pub start_time:  SystemTime,
    pub end_time:    SystemTime,
    pub state:       JobState,
    pub exit_code:   i32,
    pub tres:        TresUsage,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct TresUsage {
    pub cpu_seconds:  u64,
    pub mem_byte_sec: u64,   // mem_per_node * nodes * wall_seconds
    pub gpu_seconds:  u64,
    pub node_seconds: u64,
}
pub trait Schedulable {
    fn job_id(&self) -> JobId;
    fn spec(&self) -> &ResourceSpec;
    fn priority(&self) -> u32;
    fn submit_time(&self) -> SystemTime;
}

pub trait ResourceAllocator {
    fn try_allocate(&mut self, spec: &ResourceSpec, nodes: &[Node]) -> Result<Allocation, AllocError>;
    fn free(&mut self, alloc: &Allocation);
    fn fits(&self, spec: &ResourceSpec, node: &Node) -> bool;
}

pub trait PriorityModel {
    fn compute(&self, job: &Job, tree: &Account) -> PriorityFactors;
    fn score(&self, factors: &PriorityFactors) -> u32;
}

pub trait DurableLog {
    fn append(&mut self, record: &WalRecord) -> Result<u64, StateError>;
    fn replay(&self) -> Box<dyn Iterator<Item = Result<WalRecord, StateError>>>;
}
// hpcsched-core::error
#[derive(Debug, thiserror::Error)]
pub enum CoreError {
    #[error("invalid job state transition {from:?} -> {to:?}")] InvalidTransition { from: JobState, to: JobState },
    #[error("resource spec invalid: {0}")] InvalidSpec(String),
}

// hpcsched-alloc
#[derive(Debug, thiserror::Error)]
pub enum AllocError { InsufficientCores, InsufficientMem, InsufficientGpus, NoEligibleNode, FeatureMismatch }

// hpcsched-state
#[derive(Debug, thiserror::Error)]
pub enum StateError { Io(#[from] std::io::Error), Corrupt { offset: u64 }, Crc { expected: u32, got: u32 } }

// hpcsched-proto
#[derive(Debug, thiserror::Error)]
pub enum ProtoError { Encode(#[from] bincode::Error), Decode(#[from] bincode::Error), UnknownVariant, Io(#[from] std::io::Error) }
