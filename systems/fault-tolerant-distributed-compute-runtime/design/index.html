Core schema & types (§11)

// All id types: Copy, Clone, Eq, Hash, Ord, serde::Serialize/Deserialize, Display
pub struct TaskId     { pub job: u64, pub stage: u32, pub partition: u32, pub attempt: u32 }
pub struct StageId    { pub job: u64, pub index: u32 }
pub struct PartitionId(pub u32);           // partition index within a dataset
pub struct ShuffleId  (pub u64);           // globally unique per shuffle dependency
pub struct ExecutorId (pub u64);           // assigned at registration, stable for executor lifetime
pub struct AttemptId  (pub u32);           // monotonically increasing per (stage, partition) pair
pub struct JobId      (pub u64);

pub enum BlockId {
    Rdd       { rdd: u64, partition: PartitionId },
    Shuffle   { shuffle: ShuffleId, map: PartitionId, reduce: PartitionId },
    Broadcast { id: u64 },
}
pub enum TaskState { Pending, Running, Finished, Failed, Lost, Speculative }
pub enum StageState { Pending, Running, Finished, Failed }

pub enum TaskResult<T> {
    DirectValue(T),           // small result returned inline via RPC
    BlockRef(BlockId),        // large result stored in BlockManager; fetched lazily
    Failed { reason: String, retryable: bool },
}
pub struct Dataset<T: Send + 'static> {
    pub id:             u64,
    pub num_partitions: u32,
    pub deps:           Vec<Dependency>,
    pub compute:        Arc<dyn Fn(PartitionId, &[&dyn Any]) -> Box<dyn Iterator<Item = T>> + Send + Sync>,
}
pub enum Dependency {
    Narrow  { parent: Arc<Dataset<dyn Any + Send>> },
    Shuffle { id: ShuffleId, parent: Arc<Dataset<dyn Any + Send>>, partitioner: Partitioner },
}
pub enum Partitioner {
    Hash  { num_partitions: u32 },
    Range { bounds: Vec<Vec<u8>>, num_partitions: u32 },
}
pub enum StageKind { ShuffleMap { shuffle_id: ShuffleId }, Result }

pub struct Stage {
    pub id:       StageId,
    pub kind:     StageKind,
    pub rdd_id:   u64,
    pub parents:  Vec<StageId>,
    pub tasks:    Vec<Task>,           // one per output partition
    pub state:    StageState,
    pub attempt:  u32,
}

pub struct TaskSet {
    pub stage_id: StageId,
    pub tasks:    Vec<Task>,
    pub attempt:  u32,
}

pub enum Task {
    ShuffleMap { id: TaskId, stage: StageId, partition: PartitionId, locality_prefs: Vec<LocalityPref> },
    Result     { id: TaskId, stage: StageId, partition: PartitionId, locality_prefs: Vec<LocalityPref> },
}
pub struct LocalityPref { pub executor: Option<ExecutorId>, pub node: Option<String>, pub rack: Option<String> }
pub struct DAGScheduler {
    stages:              HashMap<StageId, Stage>,
    waiting:             HashSet<StageId>,          // parents not yet finished
    running:             HashSet<StageId>,
    finished:            HashSet<StageId>,
    map_output_tracker:  Arc<MapOutputTracker>,
    block_mgr_master:    Arc<BlockManagerMaster>,
    lineage:             Arc<LineageGraph>,
}
// Key methods (signatures only):
impl DAGScheduler {
    pub fn submit_job(&mut self, action: ResultDataset, job_id: JobId) -> JobHandle;
    pub fn stage_completed(&mut self, stage_id: StageId, results: Vec<MapStatus>);
    pub fn fetch_failed(&mut self, stage_id: StageId, shuffle_id: ShuffleId, executor: ExecutorId);
    pub fn executor_lost(&mut self, executor: ExecutorId);
}
pub struct TaskScheduler {
    pending:         HashMap<StageId, TaskSet>,
    running:         HashMap<TaskId, RunningTask>,
    executor_slots:  HashMap<ExecutorId, u32>,          // remaining free slots
    locality_delay:  Duration,                           // delay scheduling window
}
pub struct RunningTask { pub task_id: TaskId, pub executor: ExecutorId, pub started: Instant }

impl TaskScheduler {
    pub fn offer_resources(&mut self, executor: ExecutorId, free_slots: u32) -> Vec<TaskAssignment>;
    pub fn task_finished(&mut self, task_id: TaskId, result: TaskResult<()>);
    pub fn task_failed(&mut self, task_id: TaskId, reason: String, retryable: bool);
    pub fn speculative_launch(&mut self, task_id: TaskId) -> Option<TaskAssignment>;
}
pub struct TaskAssignment { pub task: Task, pub executor: ExecutorId }
pub enum DriverMessage {
    LaunchTask    { executor: ExecutorId, task: SerializedTask },
    KillTask      { task_id: TaskId, reason: String },
    RegisterAck   { executor_id: ExecutorId, config: ExecutorConfig },
}

pub enum ExecutorMessage {
    Register      { cores: u32, memory_bytes: u64, addr: SocketAddr },
    Heartbeat     { executor_id: ExecutorId, metrics: ExecutorMetrics },
    StatusUpdate  { task_id: TaskId, state: TaskState, result: Option<SerializedTaskResult> },
    BlockReport   { executor_id: ExecutorId, blocks: Vec<(BlockId, StorageLevel)> },
    FetchFailed   { task_id: TaskId, shuffle_id: ShuffleId, map_partition: PartitionId, message: String },
}

pub struct SerializedTask    (pub Vec<u8>);   // bincode-encoded Task
pub struct SerializedTaskResult(pub Vec<u8>); // bincode-encoded TaskResult<Vec<u8>>
pub struct ExecutorMetrics { pub cpu_percent: f32, pub mem_used: u64, pub tasks_running: u32 }
pub struct ExecutorConfig  { pub cores: u32, pub memory_bytes: u64, pub shuffle_port: u16 }
pub struct MapStatus {
    pub shuffle_id:    ShuffleId,
    pub map_partition: PartitionId,
    pub executor:      ExecutorId,
    pub shuffle_port:  u16,
    pub reduce_sizes:  Vec<u64>,           // bytes in each reduce partition's block
}

pub struct MapOutputTracker {
    statuses: RwLock<HashMap<ShuffleId, Vec<Option<MapStatus>>>>,  // indexed by map partition
}
impl MapOutputTracker {
    pub fn register_map_output(&self, status: MapStatus);
    pub fn unregister_shuffle(&self, shuffle_id: ShuffleId);
    pub fn get_map_statuses(&self, shuffle_id: ShuffleId) -> Option<Vec<MapStatus>>;
    pub fn is_complete(&self, shuffle_id: ShuffleId, num_map_partitions: u32) -> bool;
}
pub struct ShuffleBlockId { pub shuffle: ShuffleId, pub map: PartitionId, pub reduce: PartitionId }
pub struct FetchRequest   { pub blocks: Vec<ShuffleBlockId>, pub executor: ExecutorId }
pub struct FetchResponse  { pub block: ShuffleBlockId, pub data: Bytes }
pub struct StorageLevel {
    pub use_memory:    bool,
    pub use_disk:      bool,
    pub replication:   u8,    // 1 = no replication; 2+ = replicate to other executors
}
impl StorageLevel {
    pub const MEMORY_ONLY:    StorageLevel;
    pub const MEMORY_AND_DISK:StorageLevel;
    pub const DISK_ONLY:      StorageLevel;
}

pub struct BlockManager {
    memory_store:     MemoryStore,
    disk_store:       DiskStore,
    master_client:    BlockManagerMasterClient,   // RPC client to driver-side master
    storage_level_map:HashMap<BlockId, StorageLevel>,
}
impl BlockManager {
    pub fn put_bytes(&mut self, id: BlockId, data: Bytes, level: StorageLevel) -> Result<()>;
    pub fn get_bytes(&self, id: BlockId) -> Option<Bytes>;
    pub fn drop_block(&mut self, id: BlockId) -> bool;
    pub fn report_blocks(&self) -> Vec<(BlockId, StorageLevel)>;
}

pub struct BlockManagerMaster {
    locations: RwLock<HashMap<BlockId, Vec<ExecutorId>>>,
}
impl BlockManagerMaster {
    pub fn register_block(&self, id: BlockId, executor: ExecutorId, level: StorageLevel);
    pub fn get_locations(&self, id: BlockId) -> Vec<ExecutorId>;
    pub fn remove_executor(&self, executor: ExecutorId) -> Vec<BlockId>;  // returns lost blocks
}
pub struct LineageGraph {
    deps: HashMap<u64 /* rdd_id */, Vec<Dependency>>,
}
impl LineageGraph {
    pub fn add_dataset(&mut self, rdd_id: u64, deps: Vec<Dependency>);
    pub fn ancestors(&self, rdd_id: u64) -> Vec<u64>;
    pub fn truncate_at(&mut self, rdd_id: u64);   // I8: remove ancestors after checkpoint
}

pub struct RecomputePlan {
    pub lost:   Vec<(u64, PartitionId)>,              // (rdd_id, partition) to recompute
    pub stages: Vec<Stage>,                            // stages to execute in order
    pub reads_checkpoint: Option<CheckpointRef>,       // if recovery starts from a checkpoint
}
impl RecomputePlan {
    pub fn build(graph: &LineageGraph, lost: Vec<(u64, PartitionId)>,
                 available: &AvailableBlocks, checkpoints: &CheckpointRegistry) -> Self;
}
pub struct CheckpointRef { pub rdd_id: u64, pub path: PathBuf }
pub struct AvailableBlocks(pub HashSet<(u64, PartitionId)>);
pub struct ExecutorInfo {
    pub id:           ExecutorId,
    pub addr:         SocketAddr,
    pub cores:        u32,
    pub memory_bytes: u64,
    pub last_seen:    Instant,
    pub state:        ExecutorState,
    pub failure_count:u32,
}
pub enum ExecutorState { Alive, Suspected, Lost, Blacklisted }

pub struct Heartbeat { pub executor_id: ExecutorId, pub ts: SystemTime, pub metrics: ExecutorMetrics }

pub struct SpeculativeTracker {
    runtimes:    HashMap<TaskId, Duration>,
    p75_runtime: Option<Duration>,
    threshold:   f64,           // fraction above p75 to trigger (default 1.5x)
}
impl SpeculativeTracker {
    pub fn record(&mut self, task_id: TaskId, elapsed: Duration);
    pub fn stragglers(&self) -> Vec<TaskId>;   // tasks exceeding threshold
}
pub struct CheckpointMeta {
    pub rdd_id:     u64,
    pub partition:  PartitionId,
    pub path:       PathBuf,
    pub size_bytes: u64,
    pub checksum:   u64,           // xxhash3 of serialized partition data
}

pub trait StorageBackend: Send + Sync {
    fn write(&self, path: &Path, data: &[u8]) -> Result<()>;
    fn read(&self, path: &Path) -> Result<Vec<u8>>;
    fn exists(&self, path: &Path) -> Result<bool>;
    fn delete(&self, path: &Path) -> Result<()>;
}

pub struct CheckpointRegistry {
    checkpoints: HashMap<(u64, PartitionId), CheckpointMeta>,
}
impl CheckpointRegistry {
    pub fn register(&mut self, meta: CheckpointMeta);
    pub fn lookup(&self, rdd_id: u64, part: PartitionId) -> Option<&CheckpointMeta>;
    pub fn all_for_rdd(&self, rdd_id: u64) -> Vec<&CheckpointMeta>;
}
