Core schema & types (§11)

 # evalh.types — core records (immutable dataclasses; serialized canonical JSON via evalh.hashing)
@dataclass(frozen=True)
class Sample:                       # one eval item; immutable; from dataset loader; jsonl-serialized
    id: str; input: Any; reference: Any; split: Split
    slice_keys: dict[str, str]; meta: dict[str, Any]

@dataclass(frozen=True)
class Prompt:                       # rendered prompt for a sample; template_hash feeds cache key (I6)
    sample_id: str; rendered: ChatMessages | str
    template_hash: str; few_shot_ids: tuple[str, ...]   # few-shot from train/dev only (I4)

@dataclass(frozen=True)
class Completion:                   # a model output; pinned in run-cache (content-addressed)
    sample_id: str; text: str; logprobs: Optional[LogProbs]; finish_reason: str
    model_id: str; model_version: str; tokens_in: int; tokens_out: int; cache_key: str

@dataclass(frozen=True)
class Prediction:                   # completion + parsed answer (parser is part of the task)
    sample_id: str; completion: Completion; parsed: Any

@dataclass(frozen=True)
class Score:                        # output of a pure metric (I2); per sample, per metric
    sample_id: str; metric: str; metric_version: str; value: float | bool; detail: dict

@dataclass(frozen=True)
class MetricResult:                 # aggregate — ALWAYS carries a CI (Principle 1); never bare value
    metric: str; version: str; aggregate: float
    ci_low: float; ci_high: float; method: str; n: int; error_n: int
    per_slice: dict[str, "MetricResult"]

@dataclass(frozen=True)
class RunManifest:                  # full provenance (I8); finalized once, append-only (I7)
    run_id: str; harness_version: str; code_hash: str
    task: TaskRef; dataset_hash: str; template_hash: str
    model: ModelRef; decode_params: DecodeParams
    seeds: dict[str, int]; env: EnvFingerprint; budgets: BudgetSpec
    created_at: str; completions_hash: str           # hash over the set of completions
 # evalh.config
@dataclass(frozen=True)
class DecodeParams:                 # OUTPUT-AFFECTING → part of cache key (I6)
    temperature: float; top_p: float; max_tokens: int
    seed: Optional[int]; stop: tuple[str, ...]
@dataclass(frozen=True)
class BudgetSpec:                   # fail-closed ceilings (I5)
    max_tokens: Optional[int]; max_cost_usd: Optional[float]; max_judge_calls: Optional[int]
@dataclass(frozen=True)
class EvalRequest:                  # immutable; constructed by CLI; fully determines a run with seeds
    task: TaskRef; model: ModelRef; shots: int; seed: int
    decode: DecodeParams; budgets: BudgetSpec; bootstrap_seed: int
 # evalh.registry — resolution by (name, version); refuses unversioned/drifted
class Registry(Protocol):
    def resolve_task(self, ref: TaskRef) -> "TaskSpec": ...
    def resolve_metric(self, name: str, version: str) -> "Metric": ...
    def resolve_template(self, name: str, version: str) -> "PromptTemplate": ...
    def resolve_dataset(self, name: str, version: str) -> "Dataset": ...

 # evalh.dataset — content-hash over MATERIALIZED split (I10); read-only
class Dataset(Protocol):
    def load(self, split: Split) -> Iterable[Sample]: ...
    def content_hash(self, split: Split) -> str: ...     # SHA-256 of canonical materialized split
    def slices(self) -> list[str]: ...

 # evalh.prompt
class PromptTemplate(Protocol):
    version: str
    def render(self, sample: Sample, few_shot: list[Sample], style: PromptStyle) -> Prompt: ...
    def hash(self) -> str: ...                            # template content-hash (I10)
def select_few_shot(pool: list[Sample], k: int, seed: int) -> list[Sample]: ...  # deterministic, train/dev only (I4)

 # evalh.model — backend protocol; concurrency: async; may be NON-deterministic (output pinned in cache)
class Model(Protocol):
    model_id: str; model_version: str
    async def generate(self, prompts: list[Prompt], decode: DecodeParams) -> list[Completion]: ...
    def cost(self, tokens_in: int, tokens_out: int) -> float: ...
 # evalh.metric — PURE function of (response, reference, version) (I2); no I/O/clock/RNG
class Metric(Protocol):
    name: str; version: str
    def score(self, pred: Prediction, sample: Sample) -> Score: ...        # deterministic
 # pass@k uses the unbiased estimator (I15): 1 - C(n-c, k)/C(n, k) over n samples, c correct.

 # evalh.judge — a MODEL used as a metric; bias/variance first-class
class Judge(Protocol):
    judge_id: str; judge_version: str
    async def rubric(self, question, response, rubric) -> Score: ...
    async def pairwise(self, question, a, b) -> PairVerdict: ...            # caller MUST swap-&-avg (I9)
@dataclass(frozen=True)
class PairVerdict: winner: Literal["A","B","tie"]; order: Literal["AB","BA"]; raw: dict

 # evalh.runner — exactly-once (I3), resumable (I13); single-writer to journal
class Runner(Protocol):
    async def run(self, task: "Task", model: Model, req: EvalRequest) -> "RunHandle": ...
    def resume(self, run_id: str) -> "RunHandle": ...                       # from journal (I13)
@dataclass
class RunHandle: run_id: str; state: RunState; manifest: Optional[RunManifest]

 # evalh.cache — content-addressed; idempotent; key = output-affecting inputs only (I6)
class RunCache(Protocol):
    def key(self, prompt: Prompt, model: ModelRef, decode: DecodeParams) -> str: ...   # I6
    def get(self, key: str) -> Optional[Completion]: ...
    def put(self, key: str, completion: Completion) -> None: ...            # idempotent; immutable (I7)
 # evalh.stats — aggregate ± CI with a SEEDED Generator (I12); never bare value
def aggregate(scores: list[Score], slices: dict, *, rng: np.random.Generator,
              method: CIMethod) -> MetricResult: ...
def paired_significance(a: list[Score], b: list[Score], *, rng, test: TestKind) -> SigResult: ...
def correct(pvalues: dict[str, float], method: Correction) -> dict[str, float]: ...

 # evalh.regress — gate fires only on SIGNIFICANT paired regression post-correction (I14)
def gate(candidate: MetricResult, baseline: MetricResult, sig: SigResult,
         policy: GatePolicy) -> GateDecision: ...

 # errors (typed hierarchy; no bare except on data path — ADR-029)
class EvalhError(Exception): ...
class PlanError(EvalhError): ...        # FewShotFromEvalSplit, TokenBudgetExceeded
class ProvenanceError(EvalhError): ...  # ContentHashMismatch, IncompleteManifest, MutationAfterFinalize
class BudgetError(EvalhError): ...      # Exceeded
class JudgeError(EvalhError): ...       # UnswappedComparison, ParseFailure
class StatsError(EvalhError): ...       # UnseededRNG, InsufficientSamples
class MetricError(EvalhError): ...      # InsufficientSamples (pass@k n<k), BadReference
class RunnerError(EvalhError): ...      # JournalGap, DoubleScore
class CacheError(EvalhError): ...       # KeySchemaMismatch
