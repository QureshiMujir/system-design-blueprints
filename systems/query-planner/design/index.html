Core schema & types (§11)

// crates/qp-core/src/id.rs
#[derive(Copy, Clone, Eq, PartialEq, Hash, Debug)]
pub struct TableId(pub u32);      // owned by Catalog; never recycled within a query
#[derive(Copy, Clone, Eq, PartialEq, Hash, Debug)]
pub struct ColumnId(pub u32);     // position within a table or intermediate relation
#[derive(Copy, Clone, Eq, PartialEq, Hash, Debug)]
pub struct ExprId(pub u32);       // stable across rebinds; lets rules refer across rewrites
#[derive(Copy, Clone, Eq, PartialEq, Hash, Debug)]
pub struct PlanId(pub u32);       // stable identifier for a LogicalPlan/PhysicalPlan node
#[derive(Copy, Clone, Eq, PartialEq, Hash, Debug)]
pub struct RelationId(pub u8);    // DPccp bit-set index; u8 ⇒ 64-relation bitset cap (§4)
// crates/qp-core/src/catalog.rs
pub trait Catalog: Send + Sync {
    fn lookup_table(&self, name: &QualifiedName) -> Option<TableSchema>;
    fn lookup_index(&self, table: TableId, name: &str) -> Option<IndexSchema>;
    fn statistics(&self, table: TableId) -> Option<&TableStatistics>;
    fn type_of_function(&self, name: &str, args: &[Type]) -> Option<Type>;
}

pub struct TableSchema {
    pub id: TableId, pub name: QualifiedName, pub columns: Vec<ColumnSchema>,
    pub primary_key: Option<Vec<ColumnId>>, pub indexes: Vec<IndexSchema>,
    pub row_count_estimate: u64,
}
pub struct ColumnSchema { pub id: ColumnId, pub name: String, pub ty: Type, pub nullable: bool }
pub struct IndexSchema {
    pub name: String, pub kind: IndexKind, pub columns: Vec<ColumnId>,
    pub unique: bool, pub partial_predicate: Option<Expr>,
}
pub enum IndexKind { BTree, Hash, Bitmap, Brin, Gin, Gist }
// crates/qp-core/src/ty.rs
pub enum Type {
    Boolean, Int32, Int64, Float32, Float64, Decimal { precision: u8, scale: u8 },
    Date, Timestamp, Time, Interval, Varchar(Option<u32>), Char(Option<u32>),
    Binary, Uuid, Array(Box<Type>), Struct(Vec<(String, Type)>), Null,
}
// crates/qp-cost/src/model.rs
pub struct Cost { pub cpu: f64, pub io: f64, pub network: f64 }
// total = cpu + 4.0×io + 16.0×network  (weights from CostModelConfig, tunable, §16.2)

pub struct CostModelConfig {
    pub cpu_per_row: f64, pub cpu_per_byte: f64,
    pub io_per_seq_page: f64, pub io_per_random_page: f64, pub network_per_byte: f64,
    pub hash_build_per_row: f64, pub hash_probe_per_row: f64,
    pub sort_per_row: f64, pub merge_per_row: f64,
}
// Defaults (System R / PostgreSQL family, normalised to 1 CPU op = 0.01 cost units):
// cpu_per_row=0.01, cpu_per_byte=0.0001, io_per_seq_page=1.0, io_per_random_page=4.0,
// network_per_byte=0.001, hash_build_per_row=0.05, hash_probe_per_row=0.02,
// sort_per_row=0.10, merge_per_row=0.03

pub trait CostModel: Send + Sync {
    fn cost(&self, op: &Operator, child_costs: &[Cost]) -> Cost;   // additive, I3
}
// crates/qp-cost/src/card.rs
pub struct Cardinality { pub rows: f64, pub relative_error: f64 }   // I8: error ∈ [0.0, 10.0]
pub struct Selectivity(pub f64);                                     // I7: value ∈ [0.0, 1.0]
// Selectivity::PASSTHROUGH = 1.0, Selectivity::NONE = 0.0
// Selectivity::clamp(x) → Selectivity(x bounded to [0.0, 1.0]) — every producer calls this (I7)
// crates/qp-stats/src/lib.rs
pub trait Statistics: Send + Sync {
    fn table(&self, t: TableId) -> &TableStatistics;
    fn column(&self, t: TableId, c: ColumnId) -> &ColumnStatistics;
    fn correlation(&self, t: TableId, lhs: ColumnId, rhs: ColumnId) -> f64;
}
pub struct TableStatistics {
    pub row_count: u64, pub disk_bytes: u64,
    pub last_analyzed: Option<chrono::DateTime<chrono::Utc>>,   // I11 staleness source
}
pub struct ColumnStatistics {
    pub null_fraction: f64, pub distinct_count: f64, pub avg_width_bytes: u32,
    pub histogram: Histogram, pub mcv: McvList,
}
pub enum Histogram {
    EquiWidth { buckets: Vec<EquiBucket> },
    EquiDepth { boundaries: Vec<Datum>, target_rows_per_bucket: u64 },
    MaxDiff   { boundaries: Vec<Datum>, frequencies: Vec<u64> },
    None,
}
pub struct EquiBucket { pub low: Datum, pub high: Datum, pub count: u64, pub distinct_estimate: f64 }
pub struct McvList { pub values: Vec<Datum>, pub frequencies: Vec<f64> }   // Σ frequencies ≤ 1.0
// crates/qp-rewrite/src/rule.rs
pub trait Rule: Send + Sync {
    fn name(&self) -> &'static str;
    fn apply(&self, plan: &LogicalPlan, ctx: &RewriteCtx) -> Option<LogicalPlan>;  // I1, I13
    fn direction(&self) -> RuleDirection;   // provided-method convention, default TopDown
}
pub enum RuleDirection { TopDown, BottomUp, Both }
pub struct RewriteCtx<'a> { pub catalog: &'a dyn Catalog, pub stats: &'a dyn Statistics, pub cfg: &'a PlannerConfig }
// crates/qp-ir/src/expr.rs, crates/qp-ir/src/logical.rs
pub enum Expr {
    Column(ColumnRef), Literal(Literal), Function(FunctionCall),
    BinaryOp { op: BinOp, lhs: Box<Expr>, rhs: Box<Expr> }, UnaryOp { op: UnOp, expr: Box<Expr> },
    Case { whens: Vec<(Expr, Expr)>, else_: Option<Box<Expr>> }, Cast { expr: Box<Expr>, ty: Type },
    InList { expr: Box<Expr>, list: Vec<Expr>, negated: bool },
    Between { expr: Box<Expr>, low: Box<Expr>, high: Box<Expr>, negated: bool },
    Like { expr: Box<Expr>, pattern: Box<Expr>, escape: Option<char>, negated: bool },
    IsNull { expr: Box<Expr>, negated: bool },
    Subquery(Box<LogicalPlan>), Exists(Box<LogicalPlan>, bool), Scalar(Box<LogicalPlan>),
    AggregateCall(AggregateCall), WindowCall(WindowCall),
}
pub struct ColumnRef { pub relation: RelationId, pub column: ColumnId, pub ty: Type }

pub enum LogicalPlan {
    Scan(Scan), Filter(Filter), Project(Project), Join(Join), Aggregate(Aggregate),
    Sort(Sort), Limit(Limit), Distinct(Distinct), Union(Union), Intersect(SetOp), Except(SetOp),
    Window(Window), Apply(Apply), Values(Values), CteRef(CteRef), With(With), Empty(Empty),
}
pub struct Join { pub left: Box<LogicalPlan>, pub right: Box<LogicalPlan>, pub on: JoinPredicate, pub kind: JoinKind }
pub enum JoinPredicate { Equi(Vec<(Expr, Expr)>), Theta(Expr), Cross }
pub enum JoinKind { Inner, LeftOuter, RightOuter, FullOuter, LeftSemi, LeftAnti, RightSemi, RightAnti }
pub struct Apply { pub outer: Box<LogicalPlan>, pub inner: Box<LogicalPlan>, pub kind: ApplyKind }
// crates/qp-cascades/src/memo.rs
pub struct Memo { groups: Vec<Group>, interned: HashMap<GroupKey, GroupId> }   // I12 dedup
pub struct Group {
    pub id: GroupId, pub logical_props: LogicalProperties,
    pub expressions: Vec<GroupExpression>, pub winner: HashMap<RequiredProperties, WinnerEntry>,
}
pub struct GroupExpression { pub op: Operator, pub children: Vec<GroupId>, pub explored: bool, pub implemented: bool }
pub enum Operator { Logical(LogicalOp), Physical(PhysicalOp) }
pub struct WinnerEntry { pub physical: GroupExpressionId, pub cost: Cost, pub required_props: PhysicalProperties }
// crates/qp-physical/src/plan.rs, crates/qp-core/src/planner.rs
pub enum PhysicalPlan {
    TableScan(TableScan), IndexScan(IndexScan), IndexOnlyScan(IndexOnlyScan), BitmapScan(BitmapScan),
    HashJoin(HashJoin), MergeJoin(MergeJoin), NestedLoopJoin(NestedLoopJoin),
    HashAggregate(HashAggregate), StreamAggregate(StreamAggregate),
    Sort(SortOp), TopK(TopK), Limit(LimitOp), Project(ProjectOp), Filter(FilterOp),
    Materialize(MaterializeOp), Spool(SpoolOp), Exchange(ExchangeOp), Values(ValuesOp),
}
pub struct PhysicalProperties {
    pub sort_keys: Vec<SortKey>, pub partition_keys: Vec<Expr>, pub distinct_keys: Option<Vec<Expr>>,
    pub estimated_rows: f64, pub estimated_width_bytes: u32, pub estimated_cost: Cost,
}

pub struct Planner { cfg: PlannerConfig, catalog: Arc<dyn Catalog>, stats: Arc<dyn Statistics>, rule_set: RuleSet }
impl Planner {
    pub fn plan(&self, ast: &sql_ast::Statement) -> Result<PhysicalPlan, PlannerError>;   // I4, I6
    pub fn explain(&self, plan: &PhysicalPlan, fmt: ExplainFormat) -> String;
}
