Core schema & types (§11)

pub enum DType { F32, F64, BF16, F16, I8, I16, I32, I64, U8, U16, U32, U64, Bool }
impl DType {
    pub fn size_bytes(self) -> usize;
    pub fn is_floating(self) -> bool;
    pub fn promote(a: DType, b: DType) -> DType;   // numpy-style promotion
}

pub enum Device { Cpu, Cuda(u32), Metal(u32) }     // Cuda/Metal: Phase 12, §24

pub struct Shape(pub SmallVec<[u32; MAX_RANK]>);
impl Shape {
    pub fn rank(&self) -> usize;
    pub fn numel(&self) -> u64;
    pub fn dim(&self, axis: i32) -> u32;
    pub fn contiguous_strides(&self, elem_size: usize) -> Strides;
    pub fn broadcast_with(&self, other: &Shape) -> Result<Shape, BroadcastError>; // I4
}

pub struct Strides(pub SmallVec<[i64; MAX_RANK]>);
impl Strides {
    pub fn is_contiguous(&self, shape: &Shape, elem_size: usize) -> bool;
    pub fn expand_to(&self, src: &Shape, dst: &Shape) -> Strides;  // 0-stride, §8.2
}

pub const MAX_RANK: usize = 6;   // §4 constraint; recompiled constant, not runtime
pub struct Storage {
    pub bytes:   Arc<[UnsafeCell<u8>]>,
    pub dtype:   DType,
    pub device:  Device,
    pub version: AtomicU32,          // I5, bumped on every in-place write
}
unsafe impl Send for Storage {}
unsafe impl Sync for Storage {}

impl Storage {
    pub fn alloc(n_elements: u64, dtype: DType, device: Device) -> Arc<Storage>;
    pub fn version(&self) -> u32;
    pub fn bump_version(&self);
    pub fn as_slice<T: Element>(&self) -> &[T];
    pub fn as_mut_slice<T: Element>(&self) -> &mut [T];   // bumps version (I5)
}
pub struct Tensor {
    pub storage:       Arc<Storage>,
    pub shape:         Shape,
    pub strides:       Strides,
    pub offset:        u64,
    pub requires_grad: bool,
    pub grad_fn:       Option<NodeRef>,     // Weak<Node>, I2
    pub grad:          RwLock<Option<Tensor>>,
    pub saved_version: u32,                  // I5
    pub id:            TensorId,
}
pub struct TensorId(pub u64);               // monotonic, process-unique

impl Tensor {
    pub fn zeros(shape: impl Into<Shape>, dtype: DType, device: Device) -> Tensor;
    pub fn from_slice<T: Element>(data: &[T], shape: impl Into<Shape>) -> Tensor;
    pub fn contiguous(&self) -> Tensor;          // copies iff non-contiguous
    pub fn detach(&self) -> Tensor;              // strips grad_fn, shares storage
    pub fn backward(&self) -> Result<(), AgError>;  // I7
    pub fn grad(&self) -> Option<Tensor>;
}
pub trait Backend: Send + Sync {
    fn device(&self) -> Device;
    fn alloc(&self, n_bytes: u64) -> Arc<Storage>;
    fn elementwise_unary(&self, op: UnaryKind, a: &Tensor, out: &Tensor);
    fn elementwise_binary(&self, op: BinaryKind, a: &Tensor, b: &Tensor, out: &Tensor);
    fn reduce(&self, op: ReduceKind, a: &Tensor, axes: &[u32], keepdim: bool, out: &Tensor);
    fn matmul(&self, a: &Tensor, b: &Tensor, out: &Tensor);
    fn conv2d(&self, x: &Tensor, w: &Tensor, b: Option<&Tensor>, cfg: Conv2dCfg, out: &Tensor);
    fn softmax(&self, a: &Tensor, axis: i32, out: &Tensor);
    fn layer_norm(&self, x: &Tensor, w: &Tensor, b: &Tensor, eps: f32, out: &Tensor);
}
pub fn current_backend(device: Device) -> Arc<dyn Backend>;
pub trait Function: Send + Sync + Debug {
    fn name(&self) -> &'static str;
    fn n_inputs(&self) -> usize;
    fn forward(&self, ctx: &mut Ctx, inputs: &[&Tensor]) -> Tensor;
    fn backward(&self, ctx: &Ctx, grad_output: &Tensor) -> SmallVec<[Tensor; 4]>;
}

pub struct Ctx {
    saved_tensors: SmallVec<[Tensor; 4]>,
    saved_scalars: SmallVec<[f64; 4]>,
    saved_shapes:  SmallVec<[Shape; 4]>,
}
impl Ctx {
    pub fn save_for_backward(&mut self, t: &Tensor);
    pub fn save_shape(&mut self, s: &Shape);
    pub fn saved_tensors(&self) -> &[Tensor];
    pub fn saved_shapes(&self) -> &[Shape];
}
pub struct Node {
    pub fn_obj:    Arc<dyn Function>,
    pub ctx:       Ctx,
    pub inputs:    SmallVec<[NodeRef; 4]>,   // Weak parents, I2
    pub output_id: TensorId,
}
pub struct NodeRef(pub Weak<Node>);

pub struct Tape {
    nodes:        Vec<Arc<Node>>,             // sole strong owner, I2
    by_output:    HashMap<TensorId, Weak<Node>>,
    grad_enabled: Cell<bool>,
}
impl Tape {
    pub fn current() -> RefMut<'static, Tape>;   // thread-local, I10
    pub fn record(&mut self, f: Arc<dyn Function>, ctx: Ctx,
                  inputs: &[&Tensor], output: &Tensor) -> NodeRef;
    pub fn fresh(&mut self);                     // §8.5 state reset
    pub fn grad_enabled(&self) -> bool;
}
pub struct NoGradGuard { prev: bool }
impl NoGradGuard { pub fn new() -> Self; }       // RAII, restores on Drop
pub struct Parameter { pub tensor: Tensor, pub name: String }

pub trait Module: Send + Sync {
    fn forward(&self, x: &Tensor) -> Tensor;
    fn parameters(&self) -> Vec<&Parameter>;
    fn train(&mut self);
    fn eval(&mut self);
}

pub trait Optimizer: Send + Sync {
    fn step(&mut self, params: &mut [&mut Parameter]);
    fn zero_grad(&mut self, params: &mut [&mut Parameter]);
    fn state_dict(&self) -> StateDict;
    fn load_state_dict(&mut self, sd: StateDict);
}
