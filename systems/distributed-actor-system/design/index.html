Core schema & types (§11)

// act-types: identity, messages, errors (I2/I11/I13/I14)
pub struct Pid { pub node: NodeId, pub incarnation: u32, pub local_id: u64 }
pub struct NodeId { pub name: SmolStr, pub addr: SocketAddr, pub uid: u64 }
pub struct ActorRef { /* pid + cached ResolveHint */ }                          // location-transparent (I11)
impl ActorRef {
    pub fn send<M: Message>(&self, msg: M) -> Result<(), SendError>;
    pub fn try_send<M: Message>(&self, msg: M) -> Result<(), SendError>;
    pub fn pid(&self) -> Pid;
}
pub trait Message: Send + 'static { fn type_tag() -> TypeTag; }                 // Send-by-value (I2)
pub struct Envelope { pub to: Pid, pub from: Pid, pub kind: MsgKind, pub seq: u64, pub payload: Bytes }
#[non_exhaustive] pub enum MsgKind { User(TypeTag), System(SystemMsg), Monitor(MonitorMsg) }
#[non_exhaustive] pub enum SystemMsg { Link(Pid), Unlink(Pid), Exit{from:Pid,reason:ExitReason}, Stop, Restart }
#[non_exhaustive] pub enum ExitReason { Normal, Shutdown, Killed, Panic(String), NodeDown(NodeId) }
#[non_exhaustive] pub enum SendError { MailboxFull, NoSuchActor, NodeDown(NodeId), Serialize(SerErr) }
#[non_exhaustive] pub enum FrameError { Truncated, OversizeLen(u32), BadMagic, UnknownType(u8), UnknownTag(TypeTag) }
// act-actor: trait + context (I1/I5/I7/I10)
pub trait Actor: Send + 'static {
    type Msg: Message;
    fn handle(&mut self, msg: Self::Msg, ctx: &mut Context<Self>);             // EXACTLY one at a time (I1)
    fn init(&mut self, _ctx: &mut Context<Self>) {}                            // run on spawn + on restart (I10)
    fn terminate(&mut self, _reason: &ExitReason, _ctx: &mut Context<Self>) {}
}
pub struct Context<'a, A: Actor> { /* self ref, scheduler handle, supervisor handle */ }
impl<'a, A: Actor> Context<'a, A> {
    pub fn spawn<B: Actor>(&mut self, actor: B) -> ActorRef;                    // becomes a child (I5)
    pub fn this(&self) -> ActorRef;
    pub fn link(&mut self, other: &ActorRef);                                  // bidirectional fate-share (I7)
    pub fn monitor(&mut self, other: &ActorRef);
    pub fn stop(&mut self);
    pub fn become_(&mut self, behavior: Behavior<A>);                          // Hewitt axiom 3
}

// act-mailbox: lock-free bounded MPSC + backpressure (I4/I8)
pub struct Mailbox { /* lock-free bounded MPSC + signal lane */ }
impl Mailbox {
    pub fn enqueue(&self, env: Envelope) -> Result<Enqueued, MailboxFull>;     // wait-free SP fast path
    pub fn enqueue_signal(&self, sig: SystemMsg);                              // priority lane (I7)
    pub fn dequeue(&self) -> Option<Envelope>;                                 // single consumer
    pub fn len(&self) -> usize;
}
#[non_exhaustive] pub enum BackpressurePolicy { Block, DropNewest, DropOldest, Fail }   // I8
pub struct Enqueued { pub became_ready: bool }

// act-runtime: M:N scheduler (I3/I16, ADR-002)
pub struct Runtime { /* worker pool + run-queues + timer wheel */ }
impl Runtime {
    pub fn new(cfg: RuntimeCfg) -> Runtime;
    pub fn spawn_root<A: Actor>(&self, actor: A) -> ActorRef;
    pub fn shutdown(self) -> ShutdownReport;                                   // drain → stop → join (I16)
}
pub struct RuntimeCfg { pub workers: usize, pub reduction_budget: u32, pub mailbox_cap: usize, pub policy: BackpressurePolicy }
pub struct Reductions(u32);                                                    // preemption budget (I3)
// act-supervise: supervision tree
#[non_exhaustive] pub enum Strategy { OneForOne, OneForAll, RestForOne }
#[non_exhaustive] pub enum RestartPolicy { Permanent, Transient, Temporary }
pub struct ChildSpec { pub id: ChildId, pub factory: Box<dyn Fn()->BoxActor+Send>, pub restart: RestartPolicy }
pub struct SupervisorSpec { pub strategy: Strategy, pub intensity: Intensity, pub children: Vec<ChildSpec> }
pub struct Intensity { pub max_restarts: u32, pub within: Duration }           // I9
pub struct MonitorRef { pub target: Pid }                                      // fires Down{target,reason} (I12)

// act-transport: binary wire protocol (I6)
pub struct Frame<'a> { pub ty: FrameType, pub header: Header, pub payload: &'a [u8] }
#[non_exhaustive] pub enum FrameType { Hello, Msg, Ping, PingReq, Ack, Gossip, Bye, ExitSig }
pub trait Codec { fn encode(env:&Envelope, out:&mut BytesMut); fn decode(buf:&[u8]) -> Result<Frame,FrameError>; }
pub struct Handshake { pub proto_ver: u16, pub node: NodeId, pub incarnation: u32, pub caps: Caps }
pub struct Connection { /* one TCP per peer; per-conn FIFO (I4) */ }

// act-cluster: SWIM membership (I13)
#[non_exhaustive] pub enum MemberState { Alive, Suspect, Dead }
pub struct Member { pub node: NodeId, pub state: MemberState, pub incarnation: u32 }
pub struct Swim { /* member list + gossip + ping scheduler */ }
impl Swim {
    pub fn join(&mut self, seed: SocketAddr) -> Result<(), JoinError>;
    pub fn members(&self) -> Vec<Member>;
    pub fn leave(&mut self);                                                   // graceful (I16)
}

// act-detector: phi-accrual (I12)
pub struct PhiDetector { /* per-peer arrival-time window */ }
impl PhiDetector {
    pub fn heartbeat(&mut self, peer: &NodeId, now: Instant);
    pub fn phi(&self, peer: &NodeId, now: Instant) -> f64;
    pub fn is_suspect(&self, peer: &NodeId, now: Instant, threshold: f64) -> bool;
}

// act-delivery: AMO/ALO semantics (I14/I15)
#[non_exhaustive] pub enum Delivery { AtMostOnce, AtLeastOnce }
pub struct DedupWindow { /* (sender,seq) seen-set, bounded */ }
pub struct AckTracker { /* unacked outbound, redelivery timers */ }
