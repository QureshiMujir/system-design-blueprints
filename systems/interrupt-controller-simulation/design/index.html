struct InterruptVector
  vector_id: u32;
  priority_class: PriorityClass;
  trigger_mode: TriggerMode;
  delivery_mode: DeliveryMode;
enum InterruptState
  Idle, Pending, Delivered, Active, EoiPending, Completed, RetryPending, Failed,
struct DeliveryDescriptor
  source_controller: ControllerId;
  destination: CpuMask;
  retry_budget: u16;
  priority: u8;
struct ReplayRecord
  sequence: u64;
  timestamp: LogicalTime;
  payload: EventPayload;
trait EventQueueHandle
  fn push(event: InterruptEvent) -> Result<(), QueueFull>;
  fn pop() -> Option<InterruptEvent>;
trait LogicalClock
  fn now() -> LogicalTime;
  fn advance(delta: TickDelta);
struct TopologyGraph
  cpus: Vec<CpuNode>;
  controllers: Vec<ControllerNode>;
  routing_table: RoutingTable;
trait InterruptController
  fn enqueue(event: InterruptEvent);
  fn acknowledge(vector: u32);
  fn eoi(vector: u32);
  fn snapshot() -> Snapshot;

  generation: u64;
  checksum: Sha256;
