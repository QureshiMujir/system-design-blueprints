Core schema & types (§11)

// Identity and time — shared across both stores
TxnId(u64)              // owned by txn manager; immutable once assigned
Timestamp(u64)          // monotonic (I8); Copy; used for xmin/xmax and merge_ts
Snapshot { ts: Timestamp }   // Copy; read-only view handle; OLTP+OLAP path

RowId { page_id: u32, slot: u16 }   // row-store path only; stable across in-place update

// Row store (delta) — OLTP write path
Tuple { row_id: RowId, columns: Vec<Value>, version: MvccVersion }
    // owned by RowStore; mutated via version chain append, never in-place overwrite
DeltaStore   // owner of live Tuples; interior-mutable under per-page latch; WAL-backed
// Columnar main store — OLAP read path
ColumnStore { segments: Vec<SegmentId> }   // append-only (I4); immutable segments
Column { encoding: Encoding, data: Bytes, zone_map: (Value, Value) }
Encoding { Dictionary(dict: Vec<Value>), Rle(runs: Vec<(Value,u32)>), BitPacked(width: u8) }
Compression { ratio: f32, scheme: Encoding }   // read-only after segment seal
// Delta-merge
DeltaMerge { merge_ts: Timestamp, merge_set: RowIdRange, target_segment: SegmentId }
    // single-writer (one merge worker), spans row+column path, WAL-logged
MergePolicy { size_threshold_bytes: u64, time_threshold: Duration }   // ADR-06, Copy
// MVCC — spans both stores
MvccVersion { home: Home, xmin: Timestamp, xmax: Option<Timestamp>, chain_next: Option<Box<MvccVersion>> }
Home { RowDelta, Column(SegmentId) }
VersionChain { head: MvccVersion }   // owned per RowId or per merged-row key
Visibility(fn(&MvccVersion, Timestamp) -> bool)   // pure; I1
// Query path — router and execution
QueryRouter { fn route(&self, q: &Query) -> AccessPath }
AccessPath { PointRow, ScanColumn, Hybrid }   // OLTP=PointRow, OLAP=ScanColumn|Hybrid
VectorizedScan { batch_size: usize = 2048 }   // OLAP path only
Vector<T> { data: Vec<T>, null_mask: BitVec }   // Batch = Vec<Vector<Value>>; not persisted directly
Aggregate { AggKind, accumulator: Value }
Predicate { column: ColumnId, op: CompareOp, literal: Value }
// Resource isolation, freshness, durability
ResourceGovernor { caps: ResourceCap, enforce: fn(Workload) -> Decision }
ResourceCap { cpu_pct: u8, mem_pct: u8 }   // ADR-13; Copy; serializable to config
Freshness { commit_ts: Timestamp, visible_ts: Timestamp }   // lag = visible_ts - commit_ts (I10)
Wal { fn append(&mut self, record: WalRecord) -> Lsn }   // durability layer; I5, I12
Checkpoint { at_lsn: Lsn, manifest_ref: SegmentId }
GcCursor { watermark: Timestamp }   // I7; only advances past oldest live snapshot
// Error taxonomy
HtapError {
    WalIoError, ChecksumMismatch, SnapshotExpired, MergeConflict,
    RowColumnDivergence, GovernorCapExceeded, SchemaMismatch, TimestampRegression,
}
