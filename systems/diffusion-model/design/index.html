Core schema & types (§11)

class NoiseSchedule(Protocol):
    def alpha_bar(self, t: Tensor) -> Tensor: ...      # I1, I5; pure, no state
    def sigma(self, t: Tensor) -> Tensor: ...           # derived from alpha_bar
    def snr(self, t: Tensor) -> Tensor: ...             # ᾱ_t/(1-ᾱ_t)
// ownership: immutable post-construction; concurrency: read-only, shareable across
// workers; persistence: config dict in CheckpointRecord.schedule_cfg; serialization:
// JSON-safe scalars only; compatibility: versioned by name (ADR-04)

@dataclass(frozen=True)
class Preconditioning:
    c_skip: Callable[[Tensor], Tensor]
    c_out: Callable[[Tensor], Tensor]
    c_in: Callable[[Tensor], Tensor]
    c_noise: Callable[[Tensor], Tensor]
    def __post_init__(self) -> None: ...                # enforces I6
// ownership: immutable; mutation: none; concurrency: read-only; persistence: name-keyed
// in CheckpointRecord; serialization: name only, functions reconstructed from registry
class DenoiserBackbone(Protocol):
    def forward(self, x: Tensor, noise_level: Tensor,
                cond: ConditioningTokens | None) -> Tensor: ...
// ownership: nn.Module, trained; mutation: parameters mutate under optimizer.step();
// concurrency: DDP-wrapped, one replica per rank; persistence: safetensors state_dict;
// serialization: flat tensor buffers + shape header; compatibility: architecture id
// pinned in CheckpointRecord (ADR-06)

class UNetBackbone(DenoiserBackbone): ...                # Phase 0
class DiTBackbone(DenoiserBackbone): ...                 # Phase 8, AdaLN-Zero (P12)

@dataclass
class DiffusionModel:
    schedule: NoiseSchedule
    precond: Preconditioning
    backbone: DenoiserBackbone
    prediction_target: PredictionTarget                  # P6
// ownership: composition root, one per checkpoint; mutation: backbone only (P1/P6
// fixed at construction); concurrency: shared read-only handle across samplers (P2)
class PredictionTarget(Enum):
    EPS = "eps"; X0 = "x0"; V = "v"; EDM = "edm"          # §8 conversion identities

class Sampler(Protocol):
    def step(self, model: DiffusionModel, x_t: Tensor,
              t: Tensor, guidance: "GuidanceConfig | None") -> Tensor: ...
// ownership: stateless w.r.t. model (P2); mutation: own trajectory buffer only;
// concurrency: one instance per sampling call, never shared across calls (I13);
// persistence: none (recreated per sample); compatibility: output contract (§11
// tensor table) fixed across all registered samplers (I3)

class AncestralSampler(Sampler): ...                      # Phase 0
class DDIMSampler(Sampler): ...                           # Phase 3
class DPMSolverSampler(Sampler): ...                      # Phase 3
class HeunSampler(Sampler): ...                           # Phase 5
@dataclass(frozen=True)
class GuidanceConfig:
    scale: float                                          # w in §8 CFG eq., ≥0
    rescale: float = 0.0                                  # Lin et al. 2024 rescale trick
// ownership: sampling-call-scoped value object (P8); mutation: none; concurrency:
// immutable, freely shared; persistence: not stored in checkpoint (I7); serialization:
// CLI/config only

class EmaState:
    decay: float                                          # ADR-08, e.g. 0.9999
    def update(self, model_params: Iterable[Tensor]) -> None: ...  # I2
    def shadow_params(self) -> Iterable[Tensor]: ...
// ownership: training-loop-owned, one per run; mutation: update() only, no grad path
// (I2); concurrency: single-writer (rank 0 aggregate under DDP); persistence:
// safetensors alongside raw weights; serialization: same tensor layout as backbone
class VAEEncoder(Protocol):
    def encode(self, x: Tensor) -> "LatentDist": ...       # Phase 6
class VAEDecoder(Protocol):
    def decode(self, z: Tensor) -> Tensor: ...
class AutoencoderKL(VAEEncoder, VAEDecoder):
    scale_factor: float                                    # I8, P9
    version: str                                           # pinned in CheckpointRecord
// ownership: frozen after Phase 6 training, loaded read-only by diffusion training;
// mutation: none post-freeze; concurrency: read-only, shareable; persistence:
// safetensors; compatibility: version+scale_factor pair checked at load (I8)
@dataclass
class TrainConfig:
    resolution: int; batch_size: int; max_steps: int
    schedule_name: str; prediction_target: PredictionTarget
    precision: Literal["bf16", "fp16", "fp32"]              # ADR-03
    seed: int                                                # I4, ADR-17
// ownership: run-scoped, immutable once training starts; persistence: embedded in
// CheckpointRecord for reproducibility (ADR-24)

@dataclass
class Batch:
    images: Tensor | None; latents: Tensor | None
    t: Tensor; noise: Tensor
    labels: Tensor | None; text_embeds: Tensor | None
// ownership: per-step, GC'd after train_step; shapes/dtypes per §8 tensor table

@dataclass
class EvalReport:
    fid: float; clip_score: float | None
    nfe: int; wall_clock_s: float; n_samples: int
// ownership: eval-run-scoped; persistence: appended to Appendix-B-tracked metrics log

@dataclass
class CheckpointRecord:
    step: int; train_config: TrainConfig
    schedule_cfg: dict; vae_ref: str | None; text_encoder_ref: str | None
    format: Literal["safetensors"]                          # I12
// ownership: written once per checkpoint interval, immutable after write; persistence:
// co-located safetensors file + JSON sidecar; compatibility: format field pinned (ADR-15)
