Core schema & types (§11)

// ── bsm-origin ──────────────────────────────────────────────────────────────
pub struct Origin { pub scheme: Scheme, pub host: Host, pub port: u16, pub opaque: Option<OpaqueId> } // RFC6454
pub struct Site   { pub scheme: Scheme, pub registrable_domain: String }   // eTLD+1 via PSL (ADR-011)
pub struct SiteInstance { pub site: Site, pub browsing_instance: BrowsingInstanceId } // isolation key (I4)
impl Origin {
    pub fn parse(url: &str) -> Result<Origin, OriginError>;     // canonicalizes (I9)
    pub fn same_origin(&self, other: &Origin) -> bool;          // tuple equality (SOP, I5)
    pub fn same_site(&self, other: &Origin) -> bool;            // eTLD+1 equality
    pub fn site(&self) -> Site;                                 // isolation key
}

// ── bsm-types: process & capability ───────────────────────────────────────────
pub enum ProcessRole { Browser, Broker, Renderer, Network, Gpu, Utility }   // trust tiers (§8.1)
pub enum CapKind { FileRead, FileWrite, NetConnect, GpuSubmit, IpcBind }
pub struct Rights(u32);                                          // bitset; subset-only delegation (principle 4)
pub struct Capability {                                          // unforgeable (I6, principle 4)
    pub id: CapId,            // broker-side CapTable key (NOT renderer-chosen) — opaque nonce
    pub kind: CapKind,
    pub resource: ResourceRef,
    pub rights: Rights,
    pub epoch: u64,           // bumped on revoke ⇒ stale use denied
}
pub struct CapHandle(u32);                                       // index passed over IPC, not the Capability itself
pub struct CapTable { /* per-process; maps CapId→Capability; minted+revoked by broker only */ }
impl CapTable {
    pub fn mint(&mut self, kind: CapKind, res: ResourceRef, rights: Rights) -> Capability; // broker-only
    pub fn revoke(&mut self, id: CapId);                         // epoch++ (revocation)
    pub fn lookup(&self, id: CapId, epoch: u64) -> Option<&Capability>; // miss/epoch-mismatch ⇒ deny (I6)
}

// ── bsm-ipc: typed messages ───────────────────────────────────────────────────
pub struct IpcMessage {                                         // §8.4 frame, fully validated before use (I7)
    pub channel: ChannelId, pub seq: u64, pub kind: MsgKind,
    pub payload: Vec<u8>,                                        // bounded ≤ MAX_FRAME (I2)
    pub fds: Vec<OwnedFd>,                                       // ≤ MAX_FDS, via SCM_RIGHTS (ADR-008)
}
pub trait Channel {                                             // SEQPACKET socketpair (ADR-007)
    fn send(&self, msg: &IpcMessage) -> Result<(), IpcError>;
    fn recv(&self) -> Result<IpcMessage, IpcError>;             // validates framing; never panics (I7)
    fn bind_interface(&self, cap: &Capability) -> Result<Interface, IpcError>; // no ambient channels (principle 4)
}

// ── bsm-broker: requests / verdicts ───────────────────────────────────────────
pub enum BrokerOp {                                             // the entire privileged surface (audited, principle 9)
    FileOpen { /* server resolves path; renderer never opens by path */ },
    NetFetch { initiator: Origin, /* broker-tracked, not renderer-trusted (I2) */ },
    GpuSubmit { /* validated command buffer ref */ },
}
pub struct BrokerRequest { pub op: BrokerOp, pub requested_rights: Rights } // renderer→broker (hostile input)
pub enum Verdict { Allow, Deny(DenyReason), Kill(KillReason) }              // deny-by-default (I3, principle 3)
pub struct BrokerResponse { pub verdict: Verdict, pub cap: Option<CapHandle>, pub err: Option<BrokerError> }
pub trait Broker {
    fn handle(&mut self, caller: &SiteInstance, req: BrokerRequest) -> BrokerResponse; // the mediation point (I12)
}

// ── bsm-sandbox: the OS cage ──────────────────────────────────────────────────
pub struct SandboxPolicy {                                      // compiled from bsm-policy (§8.2)
    pub syscall_allow: SeccompProgram,   // seccompiler BPF (ADR-003)  — I8
    pub fs_rules: LandlockRuleset,       // ADR-004
    pub namespaces: NamespaceFlags,      // CLONE_NEWUSER|NEWPID|NEWNET|NEWIPC (ADR-005)
    pub drop_caps: bool, pub no_new_privs: bool,
}
pub trait Os {                                                  // OS seam (ADR-015)
    fn install_sandbox(&self, policy: &SandboxPolicy) -> Result<(), SandboxError>; // point of no return (§8.5)
    fn spawn_child(&self, role: ProcessRole, policy: &SandboxPolicy) -> Result<ChildHandle, SpawnError>;
}

// ── bsm-net / bsm-isolation: web-platform policy ──────────────────────────────
pub enum CorpDecision { Allow, Block }                          // §8.6 (I5)
pub fn corp_check(resource: &Origin, requester: &Origin, header: CorpHeader) -> CorpDecision;
pub fn orb_check(resource: &Origin, requester: &Origin, mime: Mime, nosniff: bool) -> CorpDecision;
pub struct CrossOriginIsolation { pub coop_same_origin: bool, pub coep_require_corp: bool } // SAB gate (§8.6)

pub trait SiteIsolationAllocator {                              // process-per-SiteInstance (I4, §8.9)
    fn process_for(&mut self, si: &SiteInstance) -> ProcessId;  // existing or new (OOPIF for cross-site)
    fn on_commit(&mut self, frame: FrameId, dest: &Origin) -> CommitDecision; // swap process if cross-site (I4)
}

// ── Engine / launcher (the harness entry) ─────────────────────────────────────
pub struct Engine { /* owns broker, allocator, OS seam, policy */ }
impl Engine {
    pub fn launch(policy: PolicyConfig) -> Result<Engine, EngineError>;   // spawns broker; loads policy (fail-closed)
    pub fn spawn_renderer(&mut self, si: SiteInstance) -> Result<ProcessId, EngineError>; // sandboxed, deny-by-default
    pub fn shutdown(self);
}
