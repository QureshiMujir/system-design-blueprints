Core schema & types (§11)

#sdg.spec — recipe (validated, immutable once parsed)
class GenerationSpec(BaseModel, frozen=True):       # canonicalized → RecipeHash (I1)
    schema_version: SemVer
    task: TaskType                                   # sft|preference|multiturn|eval|pretrain|distill
    generator: GeneratorRef                          # model name+version+fingerprint+backend
    seeds: SeedRef                                   # pool ids + taxonomy id
    strategy: Strategy                               # tagged union (self_instruct|evol|distill|…)
    templates: TemplateSet
    sampling: SamplingParams                         # temperature/top_p/top_k/max_tokens/seed(mandatory)
    target_count: int
    taxonomy: Optional[TaxonomyRef]
    gates: list[GateConfig]                          # ordered chain
    privacy: Optional[PrivacyConfig]
    splits: SplitConfig                              # fracs sum to 1.0 + leakage_barrier
    exports: list[ExportFormat]
class GeneratorRef(BaseModel, frozen=True):
    name: str; version: str; fingerprint: Fingerprint; backend: Backend  # local|api|runtime330
def recipe_hash(spec: GenerationSpec) -> Hash: ...   # SHA-256 over RFC-8785 canonical JSON
#sdg.types — records (immutable; gates append verdicts via copy-with, never mutate — I5)
RecordId = NewType("RecordId", str)                  # content-addressed: hash(content)+run-local seq
RunId    = NewType("RunId", str)
Hash     = NewType("Hash", str)                      # "sha256:…" / "blake3:…"
class Record(BaseModel, frozen=True):                # frozen → mutation guard (I5)
    id: RecordId
    kind: RecordKind                                  # Instruction|InstructionResponse|PreferencePair|Dialogue|Document|EvalItem
    content: Content                                  # tagged union per kind
    provenance: Provenance                            # complete or record rejected at assembly (I3)
    verdicts: tuple[GateVerdict, ...]                 # append-only (new tuple per gate)
    dedup: DedupInfo
    split: Optional[Split]                            # None until ASSEMBLE; Quarantine = rejected
    content_hash: Hash                                # stable identity; recomputed = invariant check
class Provenance(BaseModel, frozen=True):
    recipe_hash: Hash; model_fp: Fingerprint; model_version: str
    strategy: str; seed: int; parent: Optional[RecordId]   # parent for evolved/scrubbed lineage
    gen_request_hash: Hash; created_run: RunId
class GateVerdict(BaseModel, frozen=True):
    gate: GateId; score: float; passed: bool; threshold: float
    reason: str; scorer_fp: Optional[Fingerprint]     # judge/reward scorer fingerprint (audit)
class DedupInfo(BaseModel, frozen=True):
    content_hash: Hash; minhash: Optional[MinHashSig]; cluster_id: Optional[ClusterId]
    is_exact_dup: bool; near_dup_of: Optional[RecordId]
class Split(str, Enum): TRAIN="train"; DEV="dev"; TEST="test"; QUARANTINE="quarantine"
#sdg.types — manifests (durable, content-addressed)
class GenManifestEntry(BaseModel, frozen=True):       # sealed per request (I4)
    request_hash: Hash; prompt_blob: Hash; sampling: SamplingParams; model_fp: Fingerprint
    seed: int; response_blob: Hash; prompt_tokens: int; completion_tokens: int
    cost: float; latency_ms: float; attempt: int; run: RunId; ts_logical: int
class DatasetManifest(BaseModel, frozen=True):        # the dataset's identity (I1)
    dataset_hash: Hash                                # Merkle over sorted record content_hashes
    recipe_hash: Hash; schema_version: SemVer; created_run: RunId
    splits: dict[Split, SplitStats]
    model_fps: list[Fingerprint]; gate_configs: list[GateConfig]; seeds: list[int]
    contamination_report: ContaminationReport         # 0 train hits asserted (I7)
    lineage_root: Hash
class Lineage(BaseModel, frozen=True):
    record: RecordId; seed_ids: list[str]; gen_request: Hash
    verdicts: tuple[GateVerdict, ...]; parent: Optional[RecordId]; split: Split
#sdg.store
class RunStore(Protocol):                             # ACID; WAL; single-writer per run (I2/I13)
    def begin_run(self, recipe_hash: Hash) -> RunId: ...
    def put_record(self, run: RunId, rec: Record) -> None: ...
    def append_verdict(self, rec: RecordId, v: GateVerdict) -> None: ...
    def checkpoint(self, run: RunId, stage: StageId, ck_hash: Hash) -> None: ...
    def last_checkpoint(self, run: RunId) -> tuple[StageId, Hash] | None: ...
class BlobStore(Protocol):                            # hash = address; read verifies (I12)
    def put(self, data: bytes) -> Hash: ...
    def get(self, h: Hash) -> bytes: ...
#sdg.generate
class Generator(Protocol):                            # pluggable; records manifest (I4)
    async def complete(self, req: GenRequest) -> GenResponse: ...
    @property
    def fingerprint(self) -> Fingerprint: ...
class BudgetLedger(Protocol):                         # hard ceiling (I11)
    def charge(self, tokens: int, cost: float) -> BudgetState: ...
    @property
    def remaining(self) -> float: ...
#sdg.dedup / sdg.filter / sdg.pipeline protocols
class NearDupIndex(Protocol):                         # order-independent + idempotent (I6)
    def add(self, rec: RecordId, sig: MinHashSig) -> None: ...
    def query(self, sig: MinHashSig, jaccard_thr: float) -> list[RecordId]: ...
class Embedder(Protocol):
    def embed(self, texts: list[str]) -> Float32Array: ...
class NgramScanner(Protocol):                         # contamination (I7); Bloom + exact
    def contains(self, ngram: bytes) -> bool: ...
class Gate(Protocol):                                 # appends verdict; never mutates (I5)
    gate_id: GateId
    def evaluate(self, rec: Record, ctx: GateCtx) -> GateVerdict: ...
    blocking: bool
class Stage(Protocol):                                # idempotent; content-addressed (I2)
    stage_id: StageId
    def input_hash(self, run: RunId) -> Hash: ...
    def run(self, run: RunId, ctx: PipelineCtx) -> Hash: ...
class StageState(str, Enum):
    PENDING="pending"; RUNNING="running"; CHECKPOINTED="checkpointed"
    CONSUMED="consumed"; FAILED="failed"; RESUMING="resuming"
#errors (typed; quarantine vs fatal — ADR-028)
class SdgError(Exception): ...
class SpecError(SdgError): ...        # SplitFractions, UnpinnedModel, MissingSeed
class GenerateError(SdgError): ...    # BackendUnavailable, BudgetExhausted, MalformedOutput
class DedupError(SdgError): ...       # IndexCorrupt, Nondeterminism
class ValidateError(SdgError): ...    # SchemaInvalid, ConstraintUnsatisfied, LabelInconsistent
class AssembleError(SdgError): ...    # SplitStraddle (I9), RejectedInDataset (I8)
class PrivacyError(SdgError): ...     # DpBudgetExceeded (I15), PiiUnscrubbed
