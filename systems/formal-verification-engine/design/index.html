Core schema & types (§11)

// syntax — terms, sorts, spans (normative; hash-consed, I5/I7)
pub struct TermId(u32);                         // index into the intern table
pub struct Span { file: FileId, lo: u32, hi: u32 } // byte range for diagnostics (I8)
pub enum Sort {
    Bool, Int, Real, BitVec(u32),
    Array(Box<Sort>, Box<Sort>),                // (Array index elem)
    Adt(AdtId), Uninterp(SymId),
}
pub enum Expr {                                  // hash-consed via TermId
    Lit(Lit), Var(VarId),
    Old(TermId),                                 // pre-state value (I18)
    App(SymId, Vec<TermId>),                     // function application (EUF/theory)
    Unop(Unop, TermId), Binop(Binop, TermId, TermId),
    Ite(TermId, TermId, TermId),
    Select(TermId, TermId), Store(TermId, TermId, TermId),
    Quant(Quant, Vec<(VarId, Sort)>, Vec<Trigger>, TermId), // triggers (I10)
    Match(TermId, Vec<MatchArm>),
}
pub struct Trigger(pub Vec<TermId>);            // pattern terms covering bound vars
// vir — guarded commands + CFG (normative; the single IR all engines consume)
pub enum GCmd {
    Assume(TermId),
    Assert { cond: TermId, kind: ObligationKind, span: Span },  // labeled (I8)
    Havoc(Vec<VarId>),
    Assign(VarId, TermId),                       // removed by passify (I4)
}
pub enum ObligationKind {                        // governs counterexample messages (P9)
    Pre, Post, InvEntry, InvPres, Decreases,
    Assertion, Bounds, Frame, DivNonzero,
}
pub struct Block { label: BlockId, cmds: Vec<GCmd>, term: Terminator }
pub enum Terminator { Goto(Vec<BlockId>), Return }   // Goto = nondeterministic choice □
pub struct Cfg { entry: BlockId, blocks: Vec<Block>, cutpoints: Vec<BlockId> }
// passify — passive form (normative; acyclic + single-def, I4/I5)
pub struct Versioned { base: VarId, version: u32 }   // DSA version
pub struct PassiveCfg { entry: BlockId, blocks: Vec<PassiveBlock> } // no Assign, acyclic
// invariant: each Versioned defined exactly once (audited), CFG has no cycles
// vcgen — VC generation contract (normative, P6/I6/I8)
pub struct Vc { formula: TermId, obligations: Vec<(TermId, ObligationKind, Span)> }
pub trait VcGen {
    fn wp(&self, c: &GCmd, post: TermId) -> TermId;           // §8.2 table
    fn gen_proc(&self, p: &PassiveCfg) -> Vc;                 // block-equation form (§8.4)
}
// strongest-postcondition variant `SpGen` used only for differential testing (ADR-06)
// encode + smt — backend contract (normative, ADR-14/15; I9/I10/I15)
pub enum SmtResult { Unsat(UnsatCore), Sat(Model), Unknown(Reason) }   // I15 maps everything else to Unknown
pub trait SmtBackend {                            // Z3/CVC5 process OR embedded core
    fn declare(&mut self, sym: SymId, sort: &Sort);
    fn assert(&mut self, formula: TermId);        // emit SMT-LIB2 (deterministic, I7)
    fn push(&mut self); fn pop(&mut self, n: usize);          // incremental (P10)
    fn check_sat(&mut self, budget: Budget) -> SmtResult;     // timeout → Unknown (I15)
    fn get_model(&mut self) -> Model;             // on Sat (for cex)
    fn get_unsat_core(&mut self) -> UnsatCore;    // on Unsat (diagnostics)
}
pub struct Budget { wall_ms: u32, mem_mb: u32 }
pub struct Model { values: Vec<(SymId, Value)> } // SMT model
// cex — counterexample (normative, P9/I11)
pub struct CounterExample {
    failing: (ObligationKind, Span),             // which obligation, where
    valuations: Vec<(String, Value)>,            // source-level variable values (DSA-reversed)
    trace: Vec<TraceStep>,                       // path through the CFG (branches, loop iters)
}
pub struct TraceStep { block: BlockId, taken: Branch, iter: Option<u32> }
// invariant: `valuations` come from a Model satisfying ¬VC; re-checked for spuriousness (I11)
// absint — abstract-interpretation framework (normative, P11/I12/I13)
pub trait Lattice: Clone + PartialOrd {
    fn bottom() -> Self; fn top() -> Self;
    fn join(&self, other: &Self) -> Self;        // ⊔
    fn meet(&self, other: &Self) -> Self;        // ⊓
    fn widen(&self, other: &Self) -> Self;       // ▽ — guarantees termination (I13)
    fn narrow(&self, other: &Self) -> Self;      // △ — recover precision
}
pub struct Interval { lo: Bound, hi: Bound }     // the Phase-0 domain
pub trait Transfer<L: Lattice> {                 // MUST be monotone (I12, property-tested)
    fn transfer(&self, c: &GCmd, pre: &L) -> L;
}
pub struct Fixpoint;                              // worklist iterator w/ widen/narrow → inferred invariants
// the top-level verification outcome + verifier driver
pub enum Verdict { Verified, Refuted(CounterExample), Unknown(Reason) } // §1 contract
pub trait Verifier {
    fn verify_proc(&mut self, p: &ProcDecl, budget: Budget) -> Verdict;
    fn verify_program(&mut self, prog: &Program, budget: Budget) -> Vec<(String, Verdict)>;
    fn bmc(&mut self, p: &ProcDecl, k: u32) -> Verdict;   // §11 (I16)
    fn k_induct(&mut self, p: &ProcDecl, k: u32) -> Verdict; // Phase 10 (I17)
    fn infer_invariants(&mut self, p: &ProcDecl) -> Vec<TermId>; // Phase 11 absint
}
pub struct PhaseDeferredError { pub reference: &'static str }  // PhaseDeferred("Deferred Features Register")
