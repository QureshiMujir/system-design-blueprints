Core schema & types (§11)

pub struct Inst { pub raw: u32 }                  // raw 32-bit RV64IM word (no compressed C)

pub struct RegId(pub u8);                         // x0..x31; x0 hardwired zero, never renamed
pub struct CsrFile { /* minimal subset: mcause, mepc, mtval, mstatus */ }

#[non_exhaustive]
pub enum Opcode { Add, Sub, Mul, Div, Lw, Sw, Beq, Jal, Jalr, /* full RV64IM */ }

pub struct DecodedInst {
    pub op: Opcode,
    pub rd: RegId, pub rs1: RegId, pub rs2: RegId,
    pub imm: i64,                                 // sign-extended, format-specific (I/S/B/U/J)
    pub is_branch: bool, pub is_load: bool, pub is_store: bool,
    pub fu: FuKind,                               // routing hint -> ooo-execute
}

pub struct ArchState {
    pub xregs: [u64; 32],                         // xregs[0] MUST always read 0
    pub pc: u64,
    pub csr: CsrFile,
}

pub fn decode(raw: u32) -> DecodedInst;           // pure; total over RV64IM, illegal -> Opcode trap marker
pub fn execute_functional(d: &DecodedInst, st: &mut ArchState); // golden step; advances pc, writes xregs/csr
pub struct Prediction { pub taken: bool, pub target: u64 }

pub trait BranchPredictor {
    fn predict(&self, pc: u64) -> Prediction;
    fn update(&mut self, pc: u64, taken: bool, target: u64);   // called at branch resolve (execute/commit)
}

pub struct Btb { /* set-associative pc -> target; tag + valid */ }
pub struct Ras { /* return-address stack; push on Jal(rd=x1), pop on Jalr(rs1=x1) */ }
pub struct Gshare { /* GHR XOR pc -> 2-bit saturating counters */ }
// TageLite: Gshare-compatible direction predictor, multi-table tagged geometric history (Phase 3)

pub struct FetchUnit { /* speculative pc, fetch_width, Box<dyn BranchPredictor>, Btb, Ras */ }
pub struct FetchBuffer { /* bounded in-order queue<(DecodedInst, predicted_target, seq)> */ }
pub struct PhysReg(pub u16);                      // index into PhysRegFile

pub struct Rat { map: [PhysReg; 32] }             // arch reg -> current phys reg; entry 0 pinned to a zero-phys

pub struct FreeList { /* FIFO queue of free PhysReg + head index */ }

pub struct PhysRegFile {
    regs: Vec<u64>,                               // value per phys reg
    ready: Vec<bool>,                             // src_ready source-of-truth; set on writeback
}

pub struct RenameCheckpoint { rat: Rat, free_head: usize }  // snapshot at each branch rename

pub struct RenameUnit { /* maps rs1/rs2 via Rat; allocs dst from FreeList; records old_phys */ }
pub struct RsEntry {
    pub rob_idx: RobIdx,
    pub op: Opcode,
    pub src1: PhysReg, pub src2: PhysReg,
    pub src1_ready: bool, pub src2_ready: bool,   // wakeup target bits
    pub dst: PhysReg,
    pub fu: FuKind,
    pub issued: bool,                             // set on select; entry freed after issue
}

pub struct Scheduler { entries: Vec<RsEntry> /* + age order + per-FU select */ }
pub struct IssueQueue { /* age-ordered window feeding IssuePorts */ }
pub struct IssuePort { /* binds a select slot to one FuKind */ }
#[non_exhaustive]
pub enum FuKind { Alu, Mul, Div, Branch, Agu, LoadStore }

pub struct FunctionalUnit {
    pub kind: FuKind,
    pub latency: u32,                             // cycles to result
    pub pipelined: bool,                          // true: accept 1/cycle; false (Div): iterative, blocks
    pub busy_until: Cycle,                         // next-free cycle for non-pipelined FU
}

pub struct ExecResult {
    pub dst: PhysReg, pub value: u64,
    pub rob_idx: RobIdx,
    pub mispredict: bool,                          // branch FU sets on direction/target miss
}

pub struct BypassNetwork { /* forward ExecResult.dst+value to same/next-cycle dependent issues */ }
pub struct LsqEntry {
    pub rob_idx: RobIdx,
    pub is_load: bool,
    pub addr: Option<u64>,                        // None until AGU computes
    pub value: Option<u64>,
    pub addr_ready: bool,
    pub committed: bool,                          // stores write memory only after commit
}

pub struct LoadStoreQueue { loads: Vec<LsqEntry>, stores: Vec<LsqEntry> }  // age-ordered

pub struct MemRequest  { pub addr: u64, pub is_write: bool }
pub struct MemResponse { /* value (load), latency, hit/miss */ }

pub struct Cache     { /* set-assoc, MSHRs, hit/miss latency, levels via CacheLevel */ }
pub struct CacheLevel { /* one L1/L2 stage: sets, ways, line size, latency */ }
pub struct Tlb       { /* set-assoc va -> pa */ }
pub struct RobIdx(pub u16);                       // ring index; also age key across RS/LSQ

pub struct RobEntry {
    pub pc: u64,
    pub dst_arch: RegId, pub dst_phys: PhysReg,
    pub old_phys: PhysReg,                         // FREED to FreeList at commit (recovery accounting)
    pub done: bool,                                // set at writeback
    pub exception: Option<Trap>,
    pub is_store: bool,
    pub mispredict: bool,
}

pub struct ReorderBuffer { entries: Vec<RobEntry>, head: usize, tail: usize }  // circular, in-order
pub struct CommitUnit { /* commit up to commit_width/cycle from head while done; handle trap/mispredict */ }
pub struct Cycle(pub u64);

pub struct ConfigParams {
    pub fetch_width: u32, pub issue_width: u32, pub commit_width: u32,
    pub rob_size: usize, pub rs_size: usize, pub lsq_size: usize,
    pub phys_regs: usize,                          // ≥ 32 + rob_size
    pub predictor: PredictorKind,                  // Gshare | TageLite
}

pub struct Stats {
    pub cycles: u64, pub committed: u64,
    pub mispredicts: u64, pub cache_misses: u64,
}                                                  // ipc = committed as f64 / cycles

pub struct OooCore { /* FetchUnit, RenameUnit, Scheduler, FUs, LoadStoreQueue, Cache, Tlb,
                        ReorderBuffer, CommitUnit, PhysRegFile, ConfigParams, Stats,
                        Cycle, golden ArchState */ }

impl OooCore {
    pub fn tick(&mut self);                        // one cycle; REVERSE stage order
    pub fn run(&mut self, max_cycles: u64) -> Stats;
}
DecodedInst --rename--> {dst_phys, old_phys, src1/src2 PhysReg} --dispatch--> RobEntry + RsEntry (+ LsqEntry if mem)
RsEntry.src*_ready  <--wakeup-- ExecResult.dst  <--writeback-- FunctionalUnit  <--select-- Scheduler
RobEntry.old_phys   --commit--> FreeList            RenameCheckpoint --flush--> Rat (+ FreeList head)
ArchState (golden)  ==co-sim==  committed RobEntry effects        Stats.ipc = committed / cycles
