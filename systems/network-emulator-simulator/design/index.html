Core schema & types (§11)

struct PacketDescriptor
  packet_id: u128
  flow_id: u128
  protocol: TransportKind
  ingress_tick: u64
  payload_len: u32
  flags: PacketFlags
struct EventEnvelope
  event_id: u128
  logical_tick: u64
  actor_id: u32
  event_type: EventKind
  payload_ref: PayloadRef
struct SchedulerShard
  shard_id: u16
  timing_wheel: TimingWheelHandle
  local_queue: BoundedQueue<EventEnvelope>
  inflight_budget: u32
  metrics_handle: MetricsHandle
struct LinkModel
  bandwidth_bps: u64
  latency_distribution: LatencyModel
  drop_probability: f32
  corruption_probability: f32
  queue_capacity: u32
struct ReplayRecord
  event_hash: [u8; 32]
  seed: u64
  scheduler_state_hash: [u8; 32]
  timestamp: u64
struct FaultRule
  selector: TopologySelector
  trigger: TriggerCondition
  action: FaultAction
  duration: TickSpan
struct MetricSample
  metric_id: u32
  timestamp: u64
  labels: LabelSet
  value: f64
struct TopologyManifest
  manifest_hash: [u8; 32]
  nodes: Vec<NodeDescriptor>
  links: Vec<LinkModel>
  route_tables: Vec<RouteTable>
  compiled_at_tick: u64
struct WorkerLease
  lease_token: u128
  shard_id: u16
  worker_id: u32
  expires_at_tick: u64
