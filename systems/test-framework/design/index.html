Core schema & types (§11)

pub struct TestId {
    pub file: Arc<str>,
    pub suite: Arc<[Arc<str>]>,
    pub name: Arc<str>,
    pub param_index: u32,
    pub fingerprint: [u8; 16],   // BLAKE3(file, suite, name, param_index), truncated
}
impl TestId { pub fn render(&self) -> String; }
pub type TestBody = Arc<dyn for<'a> Fn(&'a mut TestContext)
    -> Pin<Box<dyn Future<Output = ()> + Send + 'a>> + Send + Sync>;

pub struct TestCase {
    pub id: TestId,
    pub body: TestBody,
    pub fixtures: Arc<[FixtureRef]>,
    pub timeout: Option<Duration>,
    pub retries: u32,
    pub skip: Option<SkipReason>,
    pub tags: Arc<[Arc<str>]>,
}
pub enum SkipReason { Static(Arc<str>), Conditional(Arc<str>),
    Quarantined { since: DateTime<Utc>, ticket: Arc<str> } }
pub struct Suite {
    pub id: SuiteId, pub name: Arc<str>, pub file: Arc<str>,
    pub children: Arc<[SuiteOrTest]>,
    pub before_all: Arc<[Hook]>, pub after_all: Arc<[Hook]>,
    pub before_each: Arc<[Hook]>, pub after_each: Arc<[Hook]>,
    pub fixtures: Arc<[FixtureDefinition]>,
}
pub struct FixtureDefinition {
    pub id: FixtureId, pub scope: FixtureScope,
    pub depends_on: Arc<[FixtureId]>,
    pub type_id: TypeId,   // for the Any downcast on retrieval
}
pub enum FixtureScope { Session, Module, Class, Function }

pub struct FixtureGraph {
    pub fn build(defs: Vec<FixtureDefinition>) -> Result<Self, FixtureGraphError>;
    pub fn setup_order(&self) -> &[FixtureId];
    pub fn teardown_order(&self) -> &[FixtureId];   // I2: reverse of setup_order
}
pub enum FixtureGraphError {
    Cycle { nodes: Vec<FixtureId> },
    UnknownDependency { parent: FixtureId, missing: FixtureId },
    ScopeViolation { child: FixtureId, parent: FixtureId },
}
pub struct AssertionResult {
    pub passed: bool, pub matcher: Arc<str>,
    pub actual: Value, pub expected: Value,
    pub diff: Option<DiffArtifact>,   // computed lazily by the reporter
    pub location: SourceLocation,     // (file, line, column) — I8
}
pub enum Value {
    Null, Bool(bool), Int(i128), Float(f64), String(Arc<str>), Bytes(Arc<[u8]>),
    Array(Arc<[Value]>), Object(Arc<[(Arc<str>, Value)]>),
    Custom { type_name: Arc<str>, repr: Arc<str> },
}
pub trait Matcher: Send + Sync {
    fn name(&self) -> &str;
    fn matches(&self, actual: &Value) -> AssertionResult;
}
pub struct DiffArtifact { pub ops: Vec<DiffOp>, pub granularity: Granularity }
pub enum DiffOp { Equal(Arc<str>), Insert(Arc<str>), Delete(Arc<str>),
    PathChanged { path: Arc<str>, from: Value, to: Value } }
pub enum Granularity { Line, Word, Char, Structural }
pub trait Reporter: Send + Sync {
    fn on_run_start(&mut self, plan: &RunPlan);
    fn on_test_finish(&mut self, id: &TestId, outcome: &TestOutcome);
    fn on_run_finish(&mut self, report: &RunReport);
}
pub enum TestOutcome {
    Passed { duration: Duration, assertions: u32, retries: u32 },
    Failed { failures: Vec<AssertionResult>, duration: Duration, retries: u32 },
    Skipped { reason: SkipReason },
    TimedOut { after: Duration },
    Crashed { signal: Option<i32>, stderr: Arc<str> },
}
pub struct Snapshot {
    pub key: SnapshotKey, pub version: u16,
    pub digest: [u8; 32],           // BLAKE3(normalize(content))
    pub content: SnapshotContent,
}
pub struct SnapshotKey { pub test_id: TestId, pub label: Arc<str>, pub index: u32 }
pub enum SnapshotContent { Text(Arc<str>), Binary(Arc<[u8]>), Json(Arc<serde_json::Value>) }
pub trait Generator: Send + Sync {
    type Value;
    fn generate(&self, src: &mut ConjectureSource) -> Result<Self::Value, GenError>;
}
pub struct ConjectureSource { /* buffer: Vec<u8>, cursor: usize */ }
impl ConjectureSource {
    pub fn draw_bits(&mut self, n: u32) -> u32;
    pub fn checkpoint(&self) -> Checkpoint;
    pub fn restore(&mut self, cp: Checkpoint);
}
pub struct Shrinker { /* candidates: VecDeque<Vec<u8>>, seen: HashSet<[u8;32]> */ }
pub trait ShrinkStrategy: Send + Sync {
    fn name(&self) -> &str;
    fn propose(&self, buffer: &[u8]) -> Vec<Vec<u8>>;
}
pub struct FlakeRecord {
    pub test_id: TestId, pub runs: u64, pub failures: u64,
    pub beta_alpha: f64, pub beta_beta: f64,
    pub flake_score: f64, pub quarantined: bool,
}
impl FlakeRecord {
    pub fn pass_probability(&self) -> f64;   // alpha / (alpha + beta)
    pub fn update(&mut self, passed: bool, now: DateTime<Utc>);
}
pub struct CoverageSnapshot { pub file_map: Vec<FileCoverage>, pub summary: CoverageSummary }
pub struct FileCoverage { pub path: Arc<str>, pub lines: Vec<LineCoverage>,
    pub branches: Vec<BranchCoverage>, pub regions: Vec<RegionCoverage> }
pub struct LineCoverage { pub line: u32, pub hits: u64 }
pub struct BranchCoverage { pub line: u32, pub branch: u32, pub taken: u64, pub not_taken: u64 }
pub struct RunPlan {
    pub tests: Vec<TestCase>, pub fixture_graph: FixtureGraph,
    pub config: RunConfig, pub seed: u128, pub source_tree_hash: [u8; 32],
}
pub struct RunConfig {
    pub workers: usize, pub isolation: IsolationMode,
    pub reporter: Vec<ReporterKind>, pub coverage: bool,
    pub update_snapshots: bool, pub timeout: Duration, pub retries: u32,
}
pub enum IsolationMode { Spawn, Fork, InProcess }
pub struct Worker { pub id: WorkerId, pub pid: u32, pub channel: MsgPackChannel,
    pub current: Option<TestId>, pub crashes: u32 }
pub struct MsgPackChannel { /* encoder/decoder over rmp_serde, inflight: u32 */ }
pub enum IpcMessage { RunnerToWorker(RunnerMsg), WorkerToRunner(WorkerMsg) }
pub enum RunnerMsg { Hello { worker_id: WorkerId, plan_hash: [u8; 32] },
    Run { test: TestCase }, Stop }
pub enum WorkerMsg { Ready { worker_id: WorkerId }, TestFinish { test_id: TestId,
    outcome: TestOutcome }, Crash { signal: Option<i32>, stderr: String } }
pub struct Sandbox { pub tmp_dir: TempDir, pub env_scrubbed: HashMap<String, String>,
    pub rlimit_cpu: u64, pub rlimit_mem: u64, pub clock: VirtualClock }
pub struct VirtualClock { /* epoch: Instant, offset: Duration */ }
impl VirtualClock { pub fn advance(&mut self, d: Duration); pub fn now(&self) -> Instant; }
pub struct MockRegistry { pub fn record(&mut self, call: MockCall); }
pub struct MockCall { pub name: Arc<str>, pub args: Vec<Value>, pub returned: Value }
