Core schema & types (§11)

pub struct TenantId(pub u32);                       // mandatory partition key (I10)
pub struct TraceId(pub [u8; 16]);                   // canonical (I1)
pub struct SpanId(pub [u8; 8]);
pub struct ResourceHash(pub u64);                   // canonical sorted-attr hash (I2)
pub struct Resource { attrs: SortedAttrs }          // service.name, k8s.*, host.* ...
pub struct Scope { name: String, version: Option<String> }
pub type Labels = SortedMap<Bytes, Bytes>;          // length-bounded (I6)
pub enum MetricKind { Counter, Gauge, Histogram, Summary }
pub struct DataPoint { labels: Labels, ts: u64, value: f64, exemplars: Vec<Exemplar> }
pub struct HistogramPoint { labels: Labels, ts: u64, buckets: Vec<(f64,u64)>, sum: f64, count: u64, exemplars: Vec<Exemplar> }
pub struct Exemplar { value: f64, ts: u64, trace_id: TraceId, span_id: SpanId, filtered: Labels }  // I3
pub struct Span { trace_id: TraceId, span_id: SpanId, parent: Option<SpanId>, name: String,
                  kind: SpanKind, start: u64, end: u64, status: SpanStatus, attrs: Labels, events: Vec<SpanEvent> }
pub struct LogRecord { ts: u64, severity: Severity, body: Bytes, attrs: Labels,
                       trace_id: Option<TraceId>, span_id: Option<SpanId> }                       // I5 bridge
pub enum Signal { Metric(Vec<DataPoint>), Trace(Vec<Span>), Log(Vec<LogRecord>) }
pub struct IngestBatch { tenant: TenantId, resource: ResourceHash, scope: Scope, signal: Signal }
pub trait Receiver { fn accept(&self, raw: &[u8]) -> Result<IngestBatch, ObsError>; }   // OTLP decode+validate (I7)
pub trait Pipeline {
    fn normalize(&self, b: IngestBatch) -> Result<IngestBatch, ObsError>;               // I1/I2
    fn enqueue(&self, b: IngestBatch) -> Result<(), ObsError>;                           // bounded; backpressure (I4)
}
pub trait MetricsStore {
    fn append(&self, t: TenantId, p: &DataPoint) -> Result<SeriesId, ObsError>;          // I6 cardinality, I15 dedup
    fn push_exemplar(&self, s: SeriesId, bucket: BucketId, ex: &Exemplar);               // I3
    fn query(&self, t: TenantId, sel: &Selector, range: TimeRange, snap: Snapshot) -> SeriesSet; // I12
}
pub trait TraceStore {
    fn ingest(&self, t: TenantId, span: &Span) -> Result<(), ObsError>;
    fn by_id(&self, t: TenantId, id: TraceId, snap: Snapshot) -> Option<Trace>;          // the pivot lookup
    fn search(&self, t: TenantId, q: &TraceQuery, snap: Snapshot) -> Vec<TraceRef>;
}
pub trait LogStore {
    fn append(&self, t: TenantId, rec: &LogRecord) -> Result<(), ObsError>;              // index labels incl trace_id
    fn query(&self, t: TenantId, q: &LogQuery, range: TimeRange, snap: Snapshot) -> LogStream;
    fn by_trace(&self, t: TenantId, id: TraceId, snap: Snapshot) -> LogStream;           // I5 join
}
pub trait Correlate {
    fn exemplar_trace(&self, s: SeriesId, b: BucketId, snap: Snapshot) -> ExemplarResult;     // I3
    fn trace_logs(&self, t: TenantId, id: TraceId, snap: Snapshot) -> LogStream;              // I5
    fn log_trace(&self, t: TenantId, id: TraceId, snap: Snapshot) -> Option<Trace>;           // I5
    fn derive_span_metrics(&self, spans: &SpanWindow) -> Vec<DataPoint>;                       // I8 (reproducible)
    fn resource_join(&self, rh: ResourceHash) -> CrossSignalRefs;                              // I2
}
pub enum ExemplarResult { Trace(Trace), SampledOut, NotFound }                                 // never dangles (I3)
pub struct Pivot { from: SignalRef, links: Vec<CorrelationLink> }                              // metric→trace→logs
pub enum SampleDecision { Keep, Drop }
pub trait TailSampler {
    fn buffer(&self, span: Span);                                          // accumulate per trace_id (bounded, I4)
    fn decide(&self, trace_id: TraceId) -> SampleDecision;                 // whole-trace, atomic (I9)
}
pub struct TailPolicy { keep_errors: bool, keep_slow_ms: u64, probabilistic: f64, keep_if_exemplar_ref: bool }
pub struct DecisionWindow { max_ms: u64 }                                  // ≤ 30s (§4)
pub enum Query { Metric(PromExpr), Log(LogExpr), Trace(TraceQuery), Pivot(PivotSpec) }   // one engine (principle 2)
pub struct Snapshot { tenant: TenantId, blocks: Vec<BlockId>, head_epoch: u64 }          // I11/I12
pub trait QueryEngine {
    fn plan(&self, q: &Query, snap: &Snapshot) -> Plan;
    fn exec(&self, plan: &Plan) -> Result<QueryResult, ObsError>;          // deterministic per snapshot (I12)
}
pub enum QueryResult { Series(SeriesSet), Logs(LogStream), Traces(Vec<Trace>), Pivot(PivotResult) }
pub struct CardinalityLimits { max_active_series: u64, max_label_len: u32, max_labels: u16, max_values_per_label: u64 }
pub trait Limiter {
    fn check(&self, t: TenantId, labels: &Labels) -> Result<(), ObsError>;   // I6; reject/relabel before storage
    fn active_series(&self, t: TenantId) -> u64;
}
pub struct RelabelRule { action: RelabelAction, label: Bytes, /* drop|keep|rename|hash */ }
pub struct Rule { expr: Query, for_: Duration, severity: Severity }
pub struct SloAlert { sli: Query, objective: f64 /*e.g. 0.999*/, windows: Vec<BurnWindow> } // multi-window
pub struct BurnWindow { long: Duration, short: Duration, burn_rate: f64 }                    // fast + slow burn
pub enum AlertState { Ok, Pending, Firing }
pub struct FiringAlert { rule: String, since: u64, exemplar: Option<TraceId> }               // carries a trace link
pub enum ObsError { OtlpInvalid(String), IdNormalize, ResourceInvalid, CardinalityExceeded{tenant:TenantId,series:u64},
                    Backpressure, TornBlock, TenantMismatch, OtlpVersion{ours:u16,theirs:u16},
                    BlockVersion{ours:u16,theirs:u16}, QueryParse(String), BadArg(String) }
pub struct IngestTrace { tenant:u32, signal:&'static str, accepted:u64, sheded:u64, dropped_dup:u64,
                         card_rejected:u64, queue_depth:usize, accept_to_durable_ms:f64 }
pub struct QueryTrace { tenant:u32, kind:&'static str, snapshot_id:u64, blocks_touched:u32,
                        duration_ms:f64, correlation_lookup_ms:f64, cache_hit:bool }
pub struct PlatformConfig { tenant_defaults: TenantConfig, listen_otlp_grpc: SocketAddr, listen_otlp_http: SocketAddr,
                            listen_query: SocketAddr, storage_dir: PathBuf, plane: Plane }
pub enum Plane { AllInOne, Ingest, Query, Compactor }                       // ADR-3 split
pub struct TenantConfig { cardinality: CardinalityLimits, retention: Retention, sampling: TailPolicy, rate_limit: u64 }
pub struct Retention { metrics_days: u32, logs_days: u32, traces_days: u32, downsample: Vec<(Duration,Duration)> }
