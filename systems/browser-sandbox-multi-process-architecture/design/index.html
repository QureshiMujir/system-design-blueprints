Core schema & types (§11)

/// Stable per-process identity. Allocated by the browser; u32 monotonic, never reused
/// within a session (avoids ABA on crash+restart). Copy: pure data, no authority.
pub struct ProcessId(pub u32);

/// Demux key WITHIN one MessagePipe: many logical frames over one socket (Phase 10).
/// RouteId is namespaced PER pipe — same numeric value on two pipes is unrelated.
pub struct RouteId(pub u32);

/// Kind of child. Drives the default CapabilitySet profile + sandbox Policy (Phase 8).
pub enum ProcessType { Browser, Renderer, Gpu, Network, Utility, Storage }

/// Coarse trust class. Browser = the only Privileged process (the TCB); all children
/// are Sandboxed. Used as an assertion gate, NOT the fine-grained authority model.
pub enum ProcessRole { Privileged, Sandboxed }

/// Security principal for isolation: scheme + eTLD+1 (Phase 9). String form e.g.
/// "https://example.com". Equality of SiteId == same isolation principal.
pub struct SiteId(pub String);

/// A group of SiteInstances that may script each other (same browsing context group).
pub struct BrowsingInstanceId(pub u32);

/// Local name for a Handle in a Message's handle table (index discipline, Phase 2).
pub struct HandleId(pub u32);

/// One least-privilege grant. The ATOM of authority. Each variant is a narrow,
/// self-describing right the broker can validate a BrokerOp against.
pub enum Capability {
    OpenFile { path_glob: String, mode: AccessMode }, // glob, NOT arbitrary path
    ConnectSocket { host: String, port: u16 },        // exact endpoint
    SharedMemory { max_bytes: u64 },                  // cap on memfd size
    ReadClipboard,
    GpuChannel,
    // ... least-privilege grants; closed enum — adding authority is a code change
}

/// Access mode inside Capability::OpenFile. ReadOnly is the renderer default.
pub enum AccessMode { ReadOnly, ReadWrite }

/// The total authority of ONE process. Set semantics (no duplicates). Immutable after
/// process launch in the steady state; grant/revoke (Phase 8) mutate the browser-side
/// copy under the event loop, never the child's. The child cannot grow its own set.
pub struct CapabilitySet(/* HashSet<Capability>, per-ProcessId, browser-owned */);

impl CapabilitySet {
    pub fn allows(&self, op: &BrokerOp) -> bool;  // the central authority predicate
}

/// Flat error taxonomy crossing every crate. Maps to CLI exit codes + metrics labels.
pub enum SbxError {
    Ipc,           // framing/socket/EOF on a MessagePipe
    Sandbox,       // seccomp/namespace/pivot_root setup failed
    PolicyDenied,  // broker refused: op not in caller CapabilitySet
    SpawnFailed,   // clone/exec/zygote-fork failed
    ChildCrashed,  // observed via SIGCHLD / IPC disconnect
    BadHandle,     // missing/duplicate/oversized handle table entry
}
/// Owned OS file descriptor. SINGLE-OWNER, close-on-drop (RAII). Moved (never copied)
/// in Rust; the underlying fd is DUPLICATED into the peer via SCM_RIGHTS on send.
/// THE capability-passing primitive: a Handle IS authority to use whatever it refers to.
pub struct Handle { /* RawFd; impl Drop { close } */ }
impl Handle {
    pub fn from_raw(fd: RawFd) -> Self;     // takes ownership
    pub fn into_raw(self) -> RawFd;         // relinquishes (no close)
    pub fn try_clone(&self) -> Result<Handle, SbxError>; // dup(2)
}

/// Message kind discriminant on the wire.
pub enum MsgKind { Request, Response, Event, Control }

/// Fixed-size, little-endian wire prefix. Parsed FIRST; bounds payload + handle count.
pub struct MessageHeader {
    pub route: RouteId,     // demux target
    pub kind: MsgKind,
    pub req_id: u64,        // correlates Response to Request
    pub num_handles: u16,   // count of fds expected in the SCM_RIGHTS aux data
}

/// One IPC unit. payload = opaque serialized body; handles = passed fds (ancillary data).
pub struct Message {
    pub header: MessageHeader,
    pub payload: Vec<u8>,
    pub handles: Vec<Handle>,  // len MUST == header.num_handles after recv
}
[ MessageHeader  fixed 24 bytes, little-endian ]
    route:u32@0 | kind:u16@4 | _pad:u16@6 | req_id:u64@8 | payload_len:u32@16 | num_handles:u16@20 | _pad:u16@22
[ payload       payload_len bytes (length-prefixed; bounded by MAX_PAYLOAD) ]
handles travel OUT-OF-BAND in SCM_RIGHTS ancillary data, NOT in payload bytes.
/// Typed request/response contract bound to an Endpoint. Codegen-style: each Interface
/// defines msg ids + (de)serialization, giving a Mojo-like statically-typed pipe.
pub trait Interface {
    const NAME: &'static str;
    fn dispatch(&mut self, msg: Message) -> Result<Option<Message>, SbxError>;
}

/// A typed end of a message pipe speaking Interface T. !Sync: owned by one task/thread.
pub struct Endpoint<T: Interface> { /* MessagePipe + PhantomData<T> */ }
impl<T: Interface> Endpoint<T> {
    pub fn send_request(&mut self, m: Message) -> Result<u64, SbxError>; // returns req_id
}

/// Bidirectional UDS-backed channel (one socketpair end). !Sync, Send. Blocking + async
/// modes; carries Messages incl. handles. The browser holds one MessagePipe per child.
pub struct MessagePipe { /* OwnedFd of a SOCK_SEQPACKET unix socket */ }
impl MessagePipe {
    pub fn send(&mut self, m: Message) -> Result<(), SbxError>;
    pub fn recv(&mut self) -> Result<Message, SbxError>;
}

/// Connected pair (socketpair(2)): one end stays in the browser, the other is passed to
/// the child as fd 3 at launch (Phase 3). Both ends Send-movable across fork.
pub fn pair() -> (MessagePipe, MessagePipe);
pub enum Prot { ReadOnly, ReadWrite }

/// A memfd-backed shared buffer. handle is the passable fd (SCM_RIGHTS) to the memfd;
/// len is the sealed size. Ownership: the Handle owns the fd (close-on-drop); the
/// backing pages are reclaimed when ALL holders unmap + close.
pub struct SharedBuffer {
    pub handle: Handle,
    pub len: usize,
}
impl SharedBuffer {
    pub fn create(len: usize) -> Result<Self, SbxError>;       // memfd_create + ftruncate
    pub fn seal(&self, flags: SealFlags) -> Result<(), SbxError>; // F_SEAL_SHRINK|GROW|WRITE
    pub fn map(&self, prot: Prot) -> Result<Mapping, SbxError>;
}

/// Seal flags applied via fcntl(F_ADD_SEALS). Producer seals WRITE+GROW+SHRINK before
/// passing a ReadOnly view → receiver cannot be tricked by a buffer that resizes mid-read.
pub struct SealFlags(/* bitset: Write|Grow|Shrink|FutureWrite */);

/// An mmap'd region. Deref<Target=[u8]> (ro) / DerefMut (rw). munmap on Drop. !Sync.
pub struct Mapping { /* ptr + len + prot */ }

/// Single-producer single-consumer ring over a SharedBuffer. Lock-free via atomic
/// head/tail in the buffer header; for bulk/streaming transfers (e.g. frame data).
pub struct ShmRing { /* SharedBuffer + header offsets */ }
impl ShmRing {
    pub fn push(&mut self, bytes: &[u8]) -> Result<(), SbxError>; // producer side
    pub fn pop(&mut self, out: &mut Vec<u8>) -> Result<bool, SbxError>; // consumer side
}
/// Per-syscall outcome in a seccomp filter. KillProcess for any syscall the policy
/// never expects (fail-closed); Errno to spoof a benign failure; Trap → SIGSYS for
/// broker-redirectable calls; Log for dev-mode auditing.
pub enum Disposition { Allow, Errno(i32), Trap, KillProcess, Log }

/// One rule: syscall nr + outcome + optional argument constraint (e.g. only O_RDONLY
/// open flags, only AF_UNIX socket family). arg_filter compiles to extra BPF comparisons.
pub struct SyscallRule {
    pub nr: i64,
    pub disposition: Disposition,
    pub arg_filter: Option<ArgFilter>,
}

/// Constraint over seccomp_data.args[0..6] (index, op, value). Limited to scalar compares
/// (BPF cannot deref pointers → cannot inspect strings; path checks belong to the broker).
pub struct ArgFilter { pub index: u8, pub op: CmpOp, pub value: u64 }
pub enum CmpOp { Eq, Ne, Ge, Le, MaskedEq }

/// Compiles SyscallRule[] → a classic-BPF program over `struct seccomp_data
/// { nr, arch, instruction_pointer, args[6] }`. FIRST instruction MUST validate `arch`
/// (reject anything != AUDIT_ARCH_X86_64) to block the x32/ia32 ABI bypass.
pub struct SeccompFilter { /* Vec<sock_filter> */ }
impl SeccompFilter {
    pub fn from_rules(rules: &[SyscallRule]) -> Result<Self, SbxError>;
    pub fn install(&self) -> Result<(), SbxError>; // requires no_new_privs already set
}

/// clone(2) namespace flag set. user ns FIRST (unprivileged → fake root) enables the rest.
pub struct Namespaces { /* CLONE_NEWUSER|NEWPID|NEWNET|NEWNS|NEWIPC|NEWUTS bitset */ }

/// Resource ceilings (setrlimit) — DoS bound + escape-surface reduction.
pub struct Rlimits { pub as_bytes: u64, pub nofile: u64, pub nproc: u64, pub cpu_secs: u64 }

/// The complete confinement spec for one child. Built per ProcessType (Phase 8).
pub struct Policy {
    pub namespaces: Namespaces,
    pub seccomp: SeccompFilter,
    pub rlimits: Rlimits,
    pub root: PathBuf,        // pivot_root target: a minimal, mostly-read-only rootfs
    pub drop_caps: bool,      // drop ALL capabilities (capset to empty)
    pub no_new_privs: bool,   // PR_SET_NO_NEW_PRIVS — MUST be true before seccomp install
}

/// Backend-agnostic confinement entrypoint. apply() runs in the CHILD after fork/clone,
/// before exec'ing/running untrusted code. Ordering is load-bearing (see Phase 4/5).
pub trait Sandbox { fn apply(&self, policy: &Policy) -> Result<(), SbxError>; }

/// The PRIMARY backend. macOS (Seatbelt) + Windows (job/restricted-token/AppContainer)
/// are PhaseDeferred stubs returning SbxError::Sandbox.
pub struct LinuxSandbox;
/// A pre-forked, pre-sandboxed parent holding warm COW pages (runtime + common libs).
/// Owns a fork-server socket; lives for the session. Spawning a renderer = fork from here
/// (cheap, COW-shared) rather than a fresh exec.
pub struct Zygote { /* fork-server MessagePipe + base Policy */ }

/// Request to mint a child. process_type selects the FINAL (tightened) policy; channel is
/// the IPC pipe end (Handle) the new child will own; policy is the to-be-sealed confinement.
pub struct ForkRequest {
    pub process_type: ProcessType,
    pub policy: Policy,
    pub channel: Handle,
}

/// Runs inside the zygote; services ForkRequests by fork() then SEALING (tightening) the
/// seccomp policy in the child before it touches untrusted work.
pub struct ForkServer;
impl ForkServer {
    pub fn serve(req: ForkRequest) -> Result<ProcessId, SbxError>; // forks; child seals + runs
}
/// What a child is asking the broker to do on its behalf. Validated against the caller's
/// CapabilitySet before ANY privileged action. Closed enum = closed authority surface.
pub enum BrokerOp {
    OpenFile { path: String, mode: AccessMode },
    Connect  { host: String, port: u16 },
    MapShm   { bytes: u64 },
    // ...
}

/// A request as received over IPC. `from` is the AUTHENTICATED ProcessId (the browser
/// stamps it from the receiving pipe — NOT taken from renderer-supplied bytes; a renderer
/// cannot impersonate another ProcessId).
pub struct BrokerRequest { pub from: ProcessId, pub op: BrokerOp }

/// Result. Handle = the brokered fd (passed back via SCM_RIGHTS). Denied carries a reason
/// string for audit (never echoes secret state). Deny-by-default: unmatched op → Denied.
pub enum BrokerResponse {
    Handle(Handle),
    Denied(String),
    Error(SbxError),
}

/// Maps ProcessId → CapabilitySet; the single chokepoint that decides allow/deny. Owned
/// by the browser, mutated only under the event loop. Every BrokerOp passes through here.
pub struct PolicyEngine { /* HashMap<ProcessId, CapabilitySet> */ }
impl PolicyEngine {
    pub fn validate(&self, req: &BrokerRequest) -> Result<(), SbxError>; // Err(PolicyDenied)
}

pub trait Broker { fn service(&mut self, req: BrokerRequest) -> BrokerResponse; }
// Implementors: FileBroker, SocketBroker, ResourceBroker.
pub struct LaunchOptions {
    pub process_type: ProcessType,
    pub policy: Policy,
    pub use_zygote: bool,   // true → ask ForkServer; false → direct clone/exec
}

/// A live child: its authenticated pid + the browser-side MessagePipe end. Dropping it
/// does NOT kill the child (lifecycle is managed via CrashHandler/ProcessHost).
pub struct ChildProcess { pub pid: ProcessId, pub channel: MessagePipe }

pub struct ProcessLauncher { /* zygote handle OR direct clone/exec path */ }
impl ProcessLauncher {
    /// Builds child cmd+env, sets up the IPC channel (pair()), applies sandbox (in child),
    /// performs the hello handshake, returns the connected ChildProcess.
    pub fn launch(&self, opts: LaunchOptions) -> Result<ChildProcess, SbxError>;
}
/// A (site, browsing-instance) pair = the process-assignment principal. id is browser-local.
pub struct SiteInstance {
    pub id: u32,
    pub site: SiteId,                       // scheme + eTLD+1
    pub browsing_instance: BrowsingInstanceId,
}

/// SiteInstances that may script each other (window.open w/o noopener) → may co-locate.
pub struct BrowsingInstance {
    pub id: BrowsingInstanceId,
    pub instances: Vec<SiteInstance>,
}

/// Outcome of asking "what process for this SiteInstance?"
pub enum ProcessReusePolicy {
    ProcessPerSiteInstance,   // the default: 1 SiteInstance → 1 renderer
    ReuseExistingForSite,     // same site already hosted → reuse (under limit)
    LimitReached,             // process-count cap hit → must consolidate
}

/// Derives the SiteId and maps site → process. eTLD+1 via a bundled public-suffix list.
pub struct SiteIsolationPolicy { /* PSL + HashMap<SiteId, ProcessId> */ }
impl SiteIsolationPolicy {
    pub fn site_for(&self, url: &str) -> SiteId;  // trust-sensitive: defines the boundary
}
/// Per-renderer bookkeeping in the browser. Holds the assigned SiteInstance + every
/// RouteId multiplexed onto that renderer's single MessagePipe.
pub struct RenderProcessHost {
    pub pid: ProcessId,
    pub site_instance: SiteInstance,
    pub routes: Vec<RouteId>,
}

/// Registry of ALL child hosts keyed by ProcessId. The single source of truth for which
/// processes exist + their capabilities (cross-refs PolicyEngine).
pub struct ProcessHost { /* HashMap<ProcessId, RenderProcessHost/...> */ }

/// The browser process itself. Owns everything; runs the single-threaded event loop.
pub struct BrowserProcess {
    /* ProcessHost, ProcessLauncher, brokers (PolicyEngine), SiteIsolationPolicy,
       CrashHandler, RouteId router */
}
impl BrowserProcess {
    pub fn navigate(&mut self, tab: u32, url: &str) -> Result<RouteId, SbxError>;
    // site_for(url) → SiteInstance → ProcessReusePolicy → (reuse|launch) → route
}
/// A unit of "content" the renderer hosts; identified by its RouteId on the shared pipe.
pub struct RenderFrame { pub route: RouteId, pub url: String }

/// Sandboxed child. Talks to the broker for all OS resources. Includes a TEST crash hook
/// (force SIGSEGV / forbidden syscall) for blast-radius + red-team tests (Phase 11/13).
/// Explicitly a STAND-IN — real rendering is #260/#305.
pub struct RendererProcess { /* MessagePipe to browser, broker Endpoint, frames */ }
/// Minimal crash artifact (NOT a full Breakpad minidump): enough to triage + report.
pub struct Minidump {
    pub pid: ProcessId,
    pub signal: i32,             // e.g. SIGSEGV, SIGSYS (seccomp kill)
    pub registers: Vec<u64>,     // best-effort snapshot
    pub maps: String,            // /proc/<pid>/maps snapshot
}

/// What to do after a child dies.
pub enum RestartPolicy {
    Never,
    Restart { max: u32, window_secs: u32 },  // rate-limited to avoid crash loops
    ReloadTab,                               // re-navigate fresh renderer
}

/// SIGCHLD/exit watcher in the browser. On child death: build Minidump, pick RestartPolicy,
/// emit the "sad tab" signal. A renderer crash MUST NOT take down the browser.
pub struct CrashHandler { /* SIGCHLD self-pipe + per-pid RestartPolicy */ }
impl CrashHandler {
    pub fn on_sigchld(&mut self) -> Vec<(ProcessId, Minidump)>; // reaps, never blocks loop
}
