Core schema & types (§11)

pub struct Message<'a> { pub object: ObjectId, pub opcode: u16, pub args: &'a [u8], pub fds: FdSlice<'a> }
pub struct MessageHeader { pub object: u32, pub size: u16, pub opcode: u16 }   // word0 | word1 (§8.1)
#[derive(Clone,Copy)] pub struct Fixed(i32);   // 24.8; Fixed::from_f64 / to_f64 / from_int
pub enum Arg<'a> { Int(i32), Uint(u32), Fixed(Fixed), Str(&'a str), Obj(ObjectId),
                   NewId{iface:&'a str,version:u32,id:ObjectId}, Array(&'a [u8]), Fd(BorrowedFd<'a>) }
#[derive(Clone,Copy,PartialEq,Eq,Hash)] pub struct ObjectId(NonZeroU32);   // 1..=0xFFFFFFFF
impl ObjectId { fn is_client_side(self)->bool { self.0.get() < 0xFF00_0000 } }   // (I2)
pub struct Resource<I: Interface> { id: ObjectId, version: u32, client: ClientId, data: I::Data }
pub trait Interface { const NAME:&'static str; const MAX_VERSION:u32;
    type Data; fn dispatch(res:&Resource<Self>, op:u16, args:ArgReader, cx:&mut Cx) -> Result<(),Error>; }
pub struct ClientId(u32);
pub struct Client { conn: Connection, objects: ObjectMap, server_id_next: u32 /*≥0xFF000000*/,
                    caps: ResourceCaps, counters: ClientCounters }
pub struct ObjectMap { map: HashMap<ObjectId, AnyResource>, tombstones: HashSet<ObjectId> }   // (I2,I3)
pub struct Cx<'a> { display:&'a mut Display, client:ClientId, scene:&'a mut Scene,
                    seat:&'a mut Seat, backend:&'a mut dyn Backend, serial:&'a mut SerialPool }
pub struct Surface { id:ObjectId, role:Option<Role>, current:SurfaceState, pending:SurfaceState,
                     subsurfaces:Vec<SubsurfaceRef>, outputs:SmallVec<OutputId>, mapped:bool }
pub struct SurfaceState { buffer:Option<BufferRef>, buffer_delta:(i32,i32),
    surface_damage:Region, buffer_damage:Region, opaque:Region, input:Region,
    frame_callbacks:Vec<ObjectId>, transform:Transform, scale:i32, viewport:Option<Viewport> }
pub enum Role { XdgToplevel(ToplevelId), XdgPopup(PopupId), Subsurface(ParentId),
                LayerSurface(Layer), Cursor, DndIcon }
pub fn commit(s:&mut Surface, cx:&mut Cx);   // the ONLY pending→current swap point (I5)
pub enum BufferKind { Shm{ pool:ShmPoolRef, offset:usize, stride:usize, format:Fourcc },
                      Dmabuf{ planes:SmallVec<DmabufPlane>, modifier:u64, format:Fourcc } }
pub struct Buffer { id:ObjectId, kind:BufferKind, width:i32, height:i32, released:bool }
pub fn shm_access<'b>(b:&'b Buffer)->Result<&'b [u8],Error>;     // bounds-checked vs pool (I8), read-only
pub fn dmabuf_import(b:&Buffer, r:&mut dyn Renderer)->Result<Texture,Error>;  // EGLImage (zero-copy)
pub fn release(b:&mut Buffer, cx:&mut Cx);                       // wl_buffer.release (I6)
pub struct Scene { outputs:Vec<Output>, layout:Layout, layers:Layers, focus_stack:Vec<SurfaceId> }
pub struct Output { id:OutputId, mode:Mode, scale:i32, transform:Transform, pos:Point,
                    damage:Region, needs_repaint:bool, current_fb:Option<FbId>, buffer_age:u8 }
pub struct Node { surface:SurfaceId, pos:Point, transform:Transform, clip:Option<Rect>, z:i32 }
pub fn output_damage(o:&Output)->&Region;            // conservative superset (I10)
pub fn schedule_repaint(o:&mut Output, when:VblankTarget);  // vblank-relative (principle 6)
pub trait Backend {
    fn outputs(&mut self)->&mut [Box<dyn OutputBackend>];
    fn dispatch(&mut self, events:&mut EventSink);          // vblank/hotplug/session events → loop
    fn session_active(&self)->bool;
}
pub trait OutputBackend {
    fn mode(&self)->Mode;
    fn test_commit(&mut self, state:&AtomicState)->Result<(),KmsError>;   // KMS TEST_ONLY (I15)
    fn commit(&mut self, state:AtomicState, flip:FlipToken)->Result<(),KmsError>;  // atomic, PAGE_FLIP_EVENT
    fn import_scanout(&mut self, buf:&Buffer)->Option<PlaneAssignment>;   // direct scanout (§19)
    fn set_cursor(&mut self, tex:Option<&Texture>, hot:Point);           // hw cursor plane
}
pub trait Renderer {
    type Texture;
    fn begin(&mut self, target:&mut RenderTarget, damage:&Region);   // glScissor(damage) / sw clip (I10)
    fn upload(&mut self, buf:&Buffer)->Result<Self::Texture,RenderError>;   // shm→tex or dmabuf→EGLImage
    fn draw_textured(&mut self, tex:&Self::Texture, src:Rect, dst:Rect, xform:Transform, alpha:f32);
    fn fill(&mut self, rect:Rect, color:Rgba);
    fn end(&mut self)->RenderResult;        // returns sync fence (implicit-sync token)
}
pub struct Seat { caps:SeatCaps, pointer:Option<PointerState>, keyboard:Option<KeyboardState>,
                  touch:Option<TouchState>, kbd_focus:Option<SurfaceId>, ptr_focus:Option<SurfaceId>,
                  keymap:XkbKeymap }
pub struct SerialPool(u32);    // monotonic; alloc() -> Serial (I7)
pub fn pointer_motion(seat:&mut Seat, scene:&Scene, abs:Point, t:Time, cx:&mut Cx);   // hit-test → enter/motion
pub fn keyboard_key(seat:&mut Seat, key:u32, st:KeyState, t:Time, cx:&mut Cx);        // → focused surface only (I7)
pub fn set_focus(seat:&mut Seat, surface:Option<SurfaceId>, cx:&mut Cx);              // leave old, enter new
pub struct Display { globals:Vec<Global>, clients:Slab<Client>, next_serial:SerialPool }
pub struct Global { name:u32, interface:&'static str, version:u32, bind:BindFn }
pub trait EventSource { fn fd(&self)->RawFd; fn ready(&mut self, ev:Readiness, cx:&mut Cx)->Result<(),Error>; }
pub fn run(loop_:&mut EventLoop, cx:&mut Cx) -> !;   // epoll: accept, read+dispatch, repaint, flush
pub enum Error { Wire(WireError), Protocol{iface:&'static str,code:u32,object:ObjectId},
                 ShmOob, BadObject(ObjectId), BadVersion{op:u16,have:u32}, Cap(CapKind),
                 Fd(FdError), Kms(KmsError), Render(RenderError) }
pub struct RepaintTrace { output:OutputId, seq:u64, damage_px:u64, nodes:u32, direct_scanout:bool,
    sample_to_render_ns:u64, render_ns:u64, commit_ns:u64, flip_latency_ns:u64, skipped:bool }
pub struct ProtocolTrace { client:ClientId, iface:&'static str, opcode:u16, object:ObjectId, serial:u32 }
