Core schema & types (§11)

// Identity types. Immutable once assigned. Serialized as strings on the wire.
pub struct NamespaceId(String);      // ownership: cluster; mutation: never; determinism: n/a — input only
pub struct WorkflowId(String);       // ownership: caller-chosen; mutation: never; determinism-sensitive: yes (part of replay key)
pub struct RunId(String);            // ownership: server (UUIDv4); mutation: never; new per run/continue-as-new
pub struct EventId(i64);             // ownership: shard owner; mutation: append-only increment; I2

// Execution identity triple. Used as the primary lookup key everywhere.
pub struct WorkflowExecution {       // persistence: `executions` table row; concurrency: single shard-owner writer
    pub namespace: NamespaceId,
    pub workflow_id: WorkflowId,
    pub run_id: RunId,
}
// The append-only event. Determinism-sensitive: replay position depends on
// exact ordering; `attributes` content feeds the I1 comparator.
pub struct HistoryEvent {
    pub event_id: EventId,               // monotonic, I2
    pub event_time: SystemTime,          // server-stamped; workflow code never reads this directly
    pub event_type: EventType,           // ADR-08 wire tag
    pub attributes: EventAttributes,     // boxed enum, one variant per EventType
    pub task_token: Option<Vec<u8>>,     // present on Started-class events
}

// ~29 variants; see §8.1 catalogue. Additive-only after GA (§27).
pub enum EventType { /* WorkflowExecutionStarted, ActivityTaskScheduled, ... */ }

// One variant per EventType; payload shape frozen per §8.1 table.
pub enum EventAttributes { /* WorkflowExecutionStartedAttrs { .. }, ... */ }

pub struct History {                 // ownership: history service; mutation: append-only
    pub events: Vec<HistoryEvent>,
}

// Replay-cursor. Determinism-sensitive: tracks replay-vs-live boundary (P1).
pub struct HistoryCursor {
    pub position: usize,
    pub is_replay: bool,
}
// A workflow's emitted intent, prior to becoming a HistoryEvent. Determinism-
// sensitive: sequence number is compared against history on replay (P6, I1).
pub struct Command {
    pub seq: u32,
    pub command_type: CommandType,
    pub attributes: EventAttributes, // reuses event-attribute shapes pre-append
}

pub enum CommandType {
    ScheduleActivity, StartTimer, CancelTimer, CancelActivity,
    RecordMarker, StartChildWorkflowExecution, SignalExternalWorkflowExecution,
    CompleteWorkflowExecution, FailWorkflowExecution, CancelWorkflowExecution,
    ContinueAsNewWorkflowExecution,
}
// Dispatched to a workflow worker. Mutation: consumed once per poll; not
// itself persisted (history is). Concurrency: exactly one live holder (I3).
pub struct WorkflowTask {
    pub task_token: WorkflowTaskToken,
    pub execution: WorkflowExecution,
    pub workflow_type: String,
    pub history: History,               // full or sticky-partial
    pub previous_started_event_id: EventId,
    pub started_event_id: EventId,
    pub attempt: u32,
}

// Opaque, HMAC-signed (§8.11). Compatibility: format is an internal contract,
// never exposed to workflow code.
pub struct WorkflowTaskToken(Vec<u8>);
// Dispatched to an activity worker. Determinism: N/A — activities are
// ordinary, non-replayed code (P3). Delivery: at-least-once.
pub struct ActivityTask {
    pub task_token: ActivityTaskToken,
    pub execution: WorkflowExecution,
    pub activity_id: String,
    pub activity_type: String,
    pub input: Payload,
    pub timeouts: TimeoutSpec,
    pub retry_policy: RetryPolicy,
    pub attempt: u32,
    pub heartbeat_details: Option<Payload>,
}

pub struct ActivityTaskToken(Vec<u8>);
pub struct ScheduledEventId(EventId); // links an ActivityTask back to its ActivityTaskScheduled event
// Task-queue identity. Ownership: matching service; mutation: queue contents
// are in-memory and non-durable (P2) — only the schedule/dispatch decision is.
pub struct TaskQueue { pub namespace: NamespaceId, pub name: String }

pub struct RetryPolicy {              // serialization: ADR-08; compatibility: additive fields only
    pub initial_interval: Duration,
    pub backoff_coefficient: f64,
    pub max_interval: Duration,
    pub max_attempts: u32,
    pub non_retryable_errors: Vec<String>,
}

pub struct TimeoutSpec {              // ADR-15: four-timeout model
    pub schedule_to_start: Duration,
    pub start_to_close: Duration,
    pub schedule_to_close: Duration,
    pub heartbeat: Duration,
}
// Durable timer. Persistence: `timers` table row; determinism: workflow code
// only ever sees TimerStarted/TimerFired events, never the row directly.
pub struct Timer { pub timer_id: TimerId, pub fire_time: SystemTime }
pub struct TimerId(String);

// Signal/query surface types. Determinism: signal payload delivery order is
// I10; query handlers are read-only (P10, I11).
pub struct Signal { pub name: String, pub payload: Payload }
pub struct Query { pub name: String, pub payload: Payload }
pub struct QueryResult { pub payload: Payload }
// Child-workflow and continue-as-new linkage (§8.9). Mutation: set once at
// child-start / continue-as-new time, immutable thereafter.
pub struct ChildWorkflowRef {
    pub execution: WorkflowExecution,
    pub cancellation_policy: ChildCancellationPolicy,
}
pub enum ChildCancellationPolicy { Abandon, RequestCancel, Terminate }

pub struct ContinueAsNewSpec { pub new_run_id: RunId, pub input: Payload }
// Versioning primitives (§8.10, P11). Immutable once recorded per run (I13).
pub struct VersionMarker { pub change_id: PatchId, pub version: u32 }
pub struct PatchId(String);

// Worker-side sticky cache entry (P12, I12). Never persisted server-side.
pub struct StickyCacheEntry {
    pub execution: WorkflowExecution,
    pub build_id: String,
    pub replay_state: ReplayState,
    pub last_started_event_id: EventId,
}
pub struct ReplayState { /* opaque to server; owned by workflow-sdk-worker */ }
// Opaque payload wrapper (P14). Engine never deserializes into app types.
pub struct Payload { pub content_type: String, pub data: Vec<u8> }

// Visibility (§8.12) and namespace quota (P13, T6) types.
pub struct VisibilityRecord {
    pub execution: WorkflowExecution,
    pub workflow_type: String,
    pub state: ExecutionState,
    pub start_time: SystemTime,
    pub close_time: Option<SystemTime>,
    pub search_attrs: BTreeMap<String, Payload>,
}

pub struct NamespaceQuota {
    pub namespace: NamespaceId,
    pub max_open_executions: u64,
    pub max_starts_per_second: u32,
    pub max_history_events: u64,
}
// Execution state (§8.6) and error taxonomy. Compatibility: additive-only
// enum variants after GA (§27).
pub enum ExecutionState {
    Running, Completed, Failed, Cancelled, Terminated, TimedOut, ContinuedAsNew,
}

pub enum WfError {
    NonDeterminismError { expected: CommandType, actual: CommandType },
    ActivityError { activity_id: String, cause: String },
    ChildWorkflowError { child: WorkflowExecution, cause: String },
    QueryMutationError,
    ExecutionClosedError,
    HistoryGapError { expected_event_id: EventId },
}
