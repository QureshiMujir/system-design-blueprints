Core schema & types (§11)

pub enum AuthError {
    InvalidRequest(String), InvalidClient, InvalidGrant, UnauthorizedClient,
    UnsupportedGrantType(String), InvalidScope(String), AccessDenied,
    InvalidToken, ExpiredToken, RevokedToken, ReplayedToken,
    Crypto(CryptoError), Storage(StorageError),
    PhaseDeferred(&'static str), Internal,
}

pub trait Clock: Send + Sync + 'static {
    fn now(&self) -> SystemTime;
    fn now_unix(&self) -> u64;
}
pub struct SystemClock;
pub struct MockClock { /* test-only, §27 */ }
pub enum Algorithm {
    HS256, HS384, HS512, RS256, RS384, RS512,
    PS256, PS384, PS512, ES256, ES384, EdDSA,
    // `none` intentionally absent — I5
}
pub struct AcceptedAlgorithms(pub Vec<Algorithm>);

pub struct JsonWebKey {
    pub kty: String, pub kid: String,
    pub usage: Option<String>, pub alg: Option<Algorithm>,
    pub material: KeyMaterial,
}
pub enum KeyMaterial {
    Rsa { n: String, e: String, d: Option<String>, p: Option<String>, q: Option<String> },
    Ec  { crv: String, x: String, y: String, d: Option<String> },
    Okp { crv: String, x: String, d: Option<String> },
    Oct { k: String },
}
pub struct JsonWebKeySet { pub keys: Vec<JsonWebKey> }
pub struct Claims {
    pub iss: String, pub sub: String, pub aud: Audience,
    pub exp: u64, pub nbf: Option<u64>, pub iat: u64, pub jti: String,
    pub extra: serde_json::Map<String, serde_json::Value>,
}
pub enum Audience { One(String), Many(Vec<String>) }

pub struct ClaimsValidator<'a> {
    pub clock: &'a dyn Clock, pub issuer: &'a str,
    pub audience: &'a [String], pub leeway: Duration,
}

pub struct JwsHeader {
    pub alg: Algorithm, pub typ: Option<String>, pub kid: Option<String>,
    pub crit: Option<Vec<String>>, pub jku: Option<String>,
    pub jwk: Option<JsonWebKey>, pub x5c: Option<Vec<String>>,
}
pub enum AuthorizationCodeState { Issued, Redeemed, Expired, Rejected }

pub struct PkceVerifier(SecretString);   // 43..128 chars, [A-Za-z0-9._~-]
pub struct PkceChallenge(String);
pub fn verify_s256(verifier: &PkceVerifier, challenge: &PkceChallenge) -> Result<(), AuthError>;
pub struct UserRow          { id: Uuid, email: String, email_verified: bool,
                               created_at: DateTime<Utc>, deactivated_at: Option<DateTime<Utc>> }
pub struct CredentialRow    { user_id: Uuid, kind: HashKind, hash: PhcString }
pub struct OAuthClientRow   { client_id: String, client_secret_hash: Option<PhcString>,
                               redirect_uris: Vec<String>, is_public: bool }
pub struct AuthorizationCodeRow { code_hash: [u8; 32], client_id: String, sub: Uuid,
                               scope: String, code_challenge: PkceChallenge,
                               expires_at: DateTime<Utc>, state: AuthorizationCodeState }
pub struct RefreshTokenRow  { token_hash: [u8; 32], family_id: Uuid, parent_id: Option<Uuid>,
                               sub: Uuid, client_id: String, expires_at: DateTime<Utc> }
pub struct JwksKeyRow       { kid: String, alg: Algorithm, public_jwk: JsonWebKey,
                               encrypted_private_jwk: Option<Vec<u8>>, state: JwksKeyState }
pub struct AuditLogRow      { id: Uuid, ts: DateTime<Utc>, event_type: AuditEventType,
                               actor: Option<Uuid>, resource: Option<String>, outcome: Outcome }
pub trait UserRepository: Send + Sync {
    async fn create(&self, email: &str) -> Result<UserRow, StorageError>;
    async fn find_by_email(&self, email: &str) -> Result<Option<UserRow>, StorageError>;
}
pub trait OAuthCodeRepository: Send + Sync {
    async fn issue(&self, code: &AuthorizationCodeRow) -> Result<(), StorageError>;
    async fn redeem_once(&self, code_hash: &[u8; 32])
        -> Result<AuthorizationCodeRow, StorageError>;           // I1
}
pub trait RefreshTokenRepository: Send + Sync {
    async fn rotate(&self, presented_hash: &[u8; 32])
        -> Result<RefreshTokenRow, StorageError>;                // I7
    async fn revoke_family(&self, family_id: Uuid, reason: &str) -> Result<(), StorageError>;
}
pub trait JwksProvider: Send + Sync {
    async fn current_signer(&self) -> Result<Arc<dyn Signer>, CryptoError>;
    async fn verifier(&self, kid: &str, alg: Algorithm) -> Result<Arc<dyn Verifier>, CryptoError>;
    async fn jwks_document(&self) -> Result<JsonWebKeySet, CryptoError>;                 // I8
}
pub struct TotpConfig { pub digits: u32, pub period_sec: u32,
                         pub algorithm: HashAlgo, pub lookback_periods: u32 }
pub trait TotpVerifier: Send + Sync {
    fn verify(&self, secret: &[u8], code: &str, now: u64) -> Result<TotpMatch, MfaError>;
}
pub trait WebauthnVerifier: Send + Sync {
    async fn verify_assertion(&self, cred_id: &[u8], client_data: &[u8],
        auth_data: &[u8], sig: &[u8]) -> Result<WebauthnMatch, MfaError>;                // I11
}
pub trait OAuthEngine: Send + Sync {
    async fn exchange_code(&self, req: &TokenRequestCode) -> Result<TokenResponse, AuthError>;
    async fn exchange_refresh(&self, req: &TokenRequestRefresh) -> Result<TokenResponse, AuthError>;
    async fn introspect(&self, req: &IntrospectRequest, rs: &str) -> Result<IntrospectResponse, AuthError>;
}
pub trait OidcEngine: Send + Sync {
    async fn issue_id_token(&self, ctx: &IdTokenContext) -> Result<String, AuthError>;
    async fn discovery_document(&self) -> Result<serde_json::Value, AuthError>;
}
