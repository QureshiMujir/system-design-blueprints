Core schema & types (§11)

struct EdgeId(u32)                          // Own: Copy; Mut: immutable; Conc: Send+Sync; Ser: none; Tgt: derived from target writes
struct PrevLoc(u32)                         // Own: Copy; Mut: mutated per-exec by target instrumentation; Tgt: yes (written by target)

struct CoverageMap {                        // Own: exclusive per executor instance
  bitmap: SharedBytes,                      // Mut: written by target (Tgt: yes), read+reset by engine
  map_size: usize,                          // Pers: config, not per-run state
}

trait Bitmap {                              // Conc: single-writer(target)/single-reader(engine) per exec cycle
  fn reset(&mut self);
  fn diff_against_virgin(&self, virgin: &Bitmap) -> CoverageDelta;
}

struct Input(Vec<u8>)                       // aka TestCase; Own: owned bytes; Ser: raw file on disk (ADR-21)
type TestCase = Input;

struct QueueEntry {                         // Pers: on-disk sidecar metadata (ADR-21)
  id: InputId,
  coverage_hash: CoverageHash,
  favored: bool,
  energy: Energy,
  discovery_lineage: ParentId,
}

struct Corpus {                             // Own: exclusive per campaign; Conc: guarded by campaign lock
  queue: Vec<QueueEntry>,
  store_path: PathBuf,                      // Pers: canonical on-disk location
}

struct Coverage(CoverageDelta)              // Mut: computed fresh per exec, immutable after
struct Feedback { is_new_edge: bool, is_new_bucket: bool }  // §8.3 output

trait IsInteresting {                       // pure function of (Coverage, VirginMap) -> Feedback
  fn evaluate(&self, cov: &Coverage, virgin: &CoverageMap) -> Feedback;
}
enum MutationKind { Bitflip, Arithmetic, InterestingValue, Havoc, Splice, Dictionary, StructureAware }
struct Havoc { stack_depth: u32 }           // Mut: stochastic, seeded (Compat: seed format versioned, §27)
struct Dictionary { tokens: Vec<Token> }
struct Token(Vec<u8>)                       // Ser: dictionary file format (ADR-09, §27)

trait Mutator {                             // Conc: Send, stateless across calls except PRNG state
  fn mutate(&mut self, input: &Input, kind: MutationKind) -> Input;
}

struct PowerSchedule { alpha: f64 }         // §8.5 equation parameter
struct Energy(u32)                         // Mut: recomputed per schedule pass; I9 floor enforced

trait Scheduler {
  fn assign_energy(&self, entry: &QueueEntry, schedule: &PowerSchedule) -> Energy;
  fn select_next(&self, corpus: &Corpus) -> InputId;
}

trait Executor {                            // Tgt: yes — the sole boundary crossing into untrusted code
  fn run(&mut self, input: &Input, timeout: Duration) -> RunResult;
}

struct ForkServer {                         // Conc: one per worker; Pers: none (ephemeral child state, I5)
  control_pipe: RawFd,
  status_pipe: RawFd,
}

struct RunResult {
  exit_kind: ExitKind,
  coverage: Coverage,
  duration: Duration,
}
enum ExitKind { Ok, Crash, Timeout, Oom }   // I8: exhaustive, never ambiguous

struct Crash {                              // Pers: append-only artifact (ADR-09-style, I3)
  crash_id: CrashId,
  input: Input,
  dedup_key: DedupKey,
  build_id: BuildId,
  runtime_seed: Seed,
}

enum DedupKey { StackHash(u64), CoverageHash(u64) }  // §8.8

trait CrashDedup {
  fn classify(&self, run: &RunResult, stderr: &[u8]) -> Option<DedupKey>;
  fn merge_or_new(&mut self, key: DedupKey) -> CrashGroupId;
}

struct StackHash(u64)                       // Ser: computed from unwound relative offsets (§8.8 note)

trait Minimizer {
  fn cmin(&self, corpus: &Corpus) -> Corpus;          // I6: coverage-preserving
  fn tmin(&self, crash: &Crash) -> Crash;             // preserves DedupKey
}

trait Sanitizer {                           // Tgt: yes — compiled into target, not the engine
  fn parse_report(&self, stderr: &[u8]) -> Option<SanitizerReport>;
}
struct SanitizerReport { bug_type: String, top_frame: String }

trait Harness {                             // ADR-18 ABI boundary; Tgt: yes
  fn run_one(&mut self, data: &[u8]) -> ExitKind;
}
struct CampaignStats {                      // Pers: periodic snapshot (ADR-24)
  execs_per_sec: f64,
  edges_covered: u32,
  corpus_size: usize,
  unique_crashes: u32,
  time_since_last_new_coverage: Duration,
}

enum FuzzError {                            // error taxonomy
  ForkServerDesync,
  BitmapOverflow,
  TargetBuildMismatch,
  ReplayMismatch,
  MinimizationCoverageLoss,
  DictionaryEncodingInvalid,
  TimeoutExceeded,
  OomExceeded,
}
