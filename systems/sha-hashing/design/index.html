Core schema & types (§11)

pub trait Hasher: Sized {
    const BLOCK_SIZE: usize;
    const OUTPUT_SIZE: usize;
    fn new() -> Self;
    fn update(&mut self, data: &[u8]);
    fn finalize(self) -> Output<Self>;          // consumes self — I2, P2
    fn digest(data: &[u8]) -> Output<Self>;     // default: new→update→finalize
}
// Output<H>: owned [u8; H::OUTPUT_SIZE]; Copy+Eq; no Debug (avoids leaking via logs, T6-adjacent)
pub struct Output<H: Hasher>(pub [u8; { H::OUTPUT_SIZE }]);

pub trait Xof: Sized {
    const BLOCK_SIZE: usize;                    // rate, in bytes
    fn new() -> Self;
    fn update(&mut self, data: &[u8]);
    fn finalize_xof(self) -> XofReader;          // consumes self
}
// XofReader: owns post-absorb sponge state; read() squeezes RATE bytes/permutation call
pub struct XofReader { /* sponge state, ready to squeeze — signature only */ }
impl XofReader { pub fn read(&mut self, out: &mut [u8]); }
// crates/sha-core/src/sha2/sha256.rs — ownership: owned, Send, not Sync (single-thread use)
pub struct Sha256 {
    h: [u32; 8],            // chaining state — mutation: per compress() call
    block: [u8; 64],        // partial-block buffer — mutation: per update() byte
    block_len: usize,       // 0..64 — invariant I3
    total_len: u64,         // bits processed — feeds length field (I4)
    backend: Sha2Backend,   // dispatch pick — immutable post-new()
}
// Sha224 = Sha256 with IV_224 + 28-byte truncation (identical layout, ADR-03)
pub enum Sha2Backend { Scalar, ShaNi, ArmSha2 }   // no persistence; not serializable
// crates/sha-core/src/sha3/sponge.rs — const-generic over rate (I12)
pub struct Keccak<const RATE: usize> {
    state: [[u64; 5]; 5],   // 1600 bits total — I12
    block: [u8; RATE],      // absorb buffer
    block_len: usize,
    pad_byte: u8,           // 0x06 SHA-3 / 0x1F SHAKE / 0x04 cSHAKE — I5
    backend: Sha3Backend,
}
pub enum Sha3Backend { Scalar, KeccakAvx2, ArmSha3 }
// type aliases: Sha3_256 = Keccak<136>; Shake128 = Keccak<168>; etc. — §8, §17
// crates/sha-mac/src/hmac.rs — ownership: owns H::BLOCK_SIZE-sized outer_pad; key zeroized (ADR-19)
pub struct Hmac<H: Hasher> {
    inner: H,                             // has absorbed (key⊕ipad) already
    outer_pad: Zeroizing<[u8; { H::BLOCK_SIZE }]>,   // ADR-19 — never Debug
}
impl<H: Hasher> Hmac<H> {
    pub fn new(key: &[u8]) -> Self;       // I7 normalization
    pub fn update(&mut self, data: &[u8]);
    pub fn finalize(self) -> Output<H>;
    pub fn verify(self, tag: &[u8]) -> Result<(), MacError>;  // I8 constant-time (ADR-18)
}
// crates/sha-mac/src/hkdf.rs — pure functions, no persistent type (RFC 5869)
pub fn extract<H: Hasher>(salt: &[u8], ikm: &[u8]) -> Output<H>;      // I10
pub fn expand<H: Hasher>(prk: &[u8], info: &[u8], out: &mut [u8]) -> Result<(), HkdfError>;

// crates/sha-mac/src/hmac_drbg.rs — SP 800-90A state machine
pub struct HmacDrbg<H: Hasher> {
    k: Zeroizing<Output<H>>,
    v: Zeroizing<Output<H>>,
    reseed_counter: u64,     // I13 — enforced ≤ 2^48
}
impl<H: Hasher> HmacDrbg<H> {
    pub fn instantiate(entropy: &[u8], nonce: &[u8], personalization: &[u8]) -> Self;
    pub fn reseed(&mut self, entropy: &[u8], additional: &[u8]);
    pub fn generate(&mut self, out: &mut [u8], additional: &[u8]) -> Result<(), DrbgError>;
}

pub enum Error { Io(/* sha-cli only */), Conformance(String), Backend(&'static str) }
pub enum MacError { VerificationFailed }        // never carries the computed tag (no timing/logging leak)
pub enum HkdfError { OutputTooLong }             // RFC 5869 Sec. 2.3: out.len() ≤ 255 * OUTPUT_SIZE
pub enum DrbgError { ReseedRequired }            // I13
