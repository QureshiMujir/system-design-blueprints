Core schema & types (§11)

struct Packet { data: Arc<[u8]>, meta: PacketMeta }
  // owned: Arc-shared immutable buffer; mutation: none post-decode;
  // concurrency: safe to share across cores; persistence: ephemeral;
  // serialization: n/a (in-process only); fast path: yes (zero-copy).

struct PacketMeta { ts_ns: u64, iface: InterfaceId, caplen: u32, wirelen: u32 }
  // owned by Packet; immutable; fast path: yes.

struct EthHeader { dst: [u8;6], src: [u8;6], ethertype: u16 }
  // immutable view into Packet buffer; fast path: yes.

struct IpHeader { version: u8, ttl: u8, proto: u8, src: IpAddr, dst: IpAddr, frag_off: u16, flags: u8 }
  // immutable view; fast path: yes; feeds Defragmenter.

struct TcpHeader { src_port: u16, dst_port: u16, seq: u32, ack: u32, flags: u8, window: u16 }
  // immutable view; fast path: yes; feeds TcpReassembler.

struct UdpHeader { src_port: u16, dst_port: u16, length: u16 }
  // immutable view; fast path: yes.

struct FiveTuple { src_ip: IpAddr, dst_ip: IpAddr, src_port: u16, dst_port: u16, proto: u8 }
  // Copy, Hash, Eq; used as FlowTable key; fast path: yes.
struct Flow { key: FiveTuple, state: TcpState, counters: FlowCounters, created_ns: u64 }
  // owned by FlowTable shard; mutable under shard lock; concurrency: sharded (ADR-13);
  // persistence: ephemeral, optional checkpoint; fast path: yes (I5).

struct FlowTable { shards: Vec<Shard<Flow>> }
  // owned by the pipeline; mutation: insert/evict per shard; concurrency: per-shard locking;
  // persistence: ephemeral; fast path: yes.

struct Fragment { ip_id: u16, offset: u16, more_fragments: bool, payload: Arc<[u8]> }
  // owned by Defragmenter buffer; immutable once captured; fast path: yes.

struct Defragmenter { policy: DefragPolicy, buffers: Map<FragKey, FragBuffer> }
  // mutable state bounded by I6; concurrency: per-flow-shard; not on fast path once buffer
  // bound exceeded (falls back to eviction path); persistence: ephemeral.

struct TcpReassembler { policy: ReassemblyPolicy, streams: Map<FiveTuple, Stream> }
  // mutable, bounded (I6); concurrency: per-flow-shard; fast path: yes (I1).

struct Segment { seq: u32, payload: Arc<[u8]>, flags: u8 }
  // immutable; owned by TcpReassembler's ordered map; fast path: yes.

struct Normalizer { profile: HostProfile }
  // stateless per-packet given a profile; concurrency: read-only shared config;
  // fast path: yes (evasion-normalization boundary, I1).

struct Stream { bytes: Vec<u8>, contiguous_to: u32 }
  // owned by TcpReassembler; mutable append-only up to contiguous_to; fast path: yes.
trait ProtoParser { fn feed(&mut self, stream: &Stream) -> ParseOutcome; }
  // implementors: HttpParser, DnsParser, TlsMetaParser; mutable per-flow parser state;
  // concurrency: owned by the flow's worker core; fast path: yes but bounded (ADR-14).

struct HttpMsg { method: Option<Str>, uri: Option<Str>, headers: Vec<(Str,Str)> }
  // owned, ephemeral, produced per request; serialization: for alert evidence only.

struct DnsMsg { qname: Str, qtype: u16, rcode: u8 }
  // owned, ephemeral; serialization: for alert evidence only.

struct AhoCorasick { states: Vec<AcState>, outputs: Vec<PatternId> }
  // immutable once built from a PatternDb snapshot; concurrency: read-only shared across
  // cores (Arc); persistence: rebuilt from PatternDb on rule reload; fast path: yes (ADR-08).

struct PatternDb { patterns: Vec<(PatternId, Vec<u8>)>, version: u64 }
  // immutable snapshot; versioned (ADR-20); serialization: loaded from signed rule bundle.

struct Rule { header: RuleHeader, options: Vec<RuleOption>, sid: u64, rev: u32 }
  // immutable once compiled into a snapshot; serialization: compiled from DSL text (§8 EBNF).

struct RuleHeader { action: Action, proto: Proto, src: NetSpec, dst: NetSpec, dir: Direction }
  // immutable; part of Rule.

enum RuleOption { Msg(Str), Content(Vec<u8>, ContentMod), Pcre(BoundedRegex), Flow(FlowState), Sid(u64), Rev(u32) }
  // immutable; PCRE variant carries a compiled, budget-bounded regex (ADR-10).
struct Matcher { automaton: Arc<AhoCorasick>, snapshot_gen: u64 }
  // immutable snapshot handle, atomically swapped (I8); concurrency: lock-free read via
  // Arc swap; fast path: yes.

struct MatchResult { pattern_id: PatternId, offset: u32, rule_ids: Vec<u64> }
  // owned, ephemeral, produced per match; feeds RuleEngine evaluation.

struct AnomalyDetector { baselines: Map<FeatureKey, Baseline> }
  // mutable, bounded (I7); concurrency: per-flow-shard or per-feature shard; fast path: yes
  // but off the signature critical path (parallel evaluation).

struct Baseline { mean: f64, variance: f64, window_start_ns: u64, samples: u32 }
  // mutable, rate-limited updates (I7, P10); persistence: optional periodic snapshot.

struct EntropyEstimate { bins: Vec<u32>, total: u64 }
  // ephemeral per evaluation window; recomputed from Baseline-tracked bin counts.

enum Verdict { Allow, Alert, Drop, Reset }
  // Copy; produced once per packet/flow decision; consumed by Enforcement (I2, I3).

struct Alert { id: Uuid, sid: Option<u64>, severity: Severity, flow: FiveTuple, evidence: Vec<u8>, ts_ns: u64 }
  // owned, append-only once emitted (I4); serialization: structured JSON (ADR-21).

struct Correlator { incidents: Map<IncidentId, Vec<Uuid>> }
  // mutable, append-only linkage (I11); concurrency: single correlator per sensor
  // (multi-sensor correlation is PhaseDeferred, §1).

enum IdsError { Malformed(DecodeStage), BufferExceeded(FlowKey), ParserFault(ProtoKind), RuleCompileError(Str), EnforcementFault(Reason) }
  // structured error taxonomy (P15: no stringly-typed failures); immutable once constructed;
  // every variant maps to a named telemetry counter in §7/§24/§25.
