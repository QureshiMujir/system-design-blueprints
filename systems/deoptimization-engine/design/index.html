Core schema & types (§11)

// deopt-core: assumptions, guards, traps, deopt semantics — hot compile-time path
// Assumption is immutable once registered (I7); mutation only via new registration.
struct Assumption { id: AssumptionId, kind: AssumptionKind, class_dep: Option<ClassId> }
enum AssumptionKind { TypeSpeculation(TypeId), MonomorphicCall(MethodId), NullCheckElision, BoundsCheckElision, ClassHierarchyLeaf(ClassId), IntegerRange(RangeBounds) }
struct Guard { id: GuardId, assumption: AssumptionId, frame_state: FrameStateId, trap: UncommonTrapId }
struct UncommonTrap { id: UncommonTrapId, reason: DeoptReason, action: DeoptAction }
enum DeoptReason { TypeMismatch, MegamorphicCall, NullCheckFailed, BoundsCheckFailed, ClassHierarchyChanged, IntegerOverflow, ExplicitInvalidation }
enum DeoptAction { ReinterpretAndRecompile, ReinterpretOnly, Blacklist(AssumptionKind) }
// deopt-core: safepoints and debug info — cold deopt-time path, hot-path adjacent
struct SafepointId(u64)
struct Safepoint { id: SafepointId, pc: MachinePc, debug_info: DebugInfoId }
struct DebugInfo { bytecode_index: BytecodeIndex, inline_frames: Vec<InlineFrameDescriptor>, values: Vec<ValueLocation>, live_refs: BitSet }
enum ValueLocation { Register(RegId), Stack(StackOffset), Constant(ConstValue), Rematerialize(RematRecipeId) }
struct StateMap { entries: Vec<(SafepointId, DebugInfo)>, checksum: u64 }
struct BytecodeIndex(u32)
// deopt-core: virtual objects and rematerialization — cold deopt-time path
struct VirtualObject { id: VirtualObjectId, class_id: ClassId, fields: Vec<(FieldId, ValueLocation)> }
struct RematRecipe { id: RematRecipeId, target: VirtualObject, deps: Vec<RematRecipeId> }
struct ScalarReplacement { original_alloc: AllocSiteId, recipe: RematRecipeId }
// deopt-core: frames — the interpreter/optimized seam representation
struct InterpreterFrame { function: MethodId, bytecode_index: BytecodeIndex, locals: Vec<Value>, operand_stack: Vec<Value>, caller: Option<Box<InterpreterFrame>> }
struct OptimizedFrame { method: MethodId, machine_pc: MachinePc, registers: RegisterSnapshot, stack_slots: Vec<StackSlot> }
struct FrameState { id: FrameStateId, safepoint: SafepointId, debug_info: DebugInfoId }
// deopt-core: the deoptimizer, OSR, and compiled-method bookkeeping
struct Deoptimizer { /* capture, lookup, rebuild, resume — drives §8.3 */ }
struct OsrEntry { id: OsrEntryId, loop_header: BytecodeIndex, entry_state_map: DebugInfoId }
struct CompiledMethod /* = Nmethod */ { id: MethodId, tier: Tier, code: CodeRegion, state_map: StateMap, guards: Vec<Guard> }
enum Tier { Interpreted, Profiling, Tier1, Optimized, Deoptimized, Reprofiled, Recompiled, Blacklisted }
// deopt-core: dependency tracking, invalidation, profiling, error taxonomy
struct Dependency { assumption: AssumptionId, dependent: MethodId }
struct DependencyTracker { by_assumption: HashMap<AssumptionId, HashSet<MethodId>> }
struct Invalidation { assumption: AssumptionId, cause: InvalidationCause, affected: Vec<MethodId> }
struct ProfileData { method: MethodId, invocation_count: u64, backedge_count: u64, type_feedback: TypeFeedbackTable }
struct RecompilationController { /* reacts to Reprofiled state, issues recompiles */ }
struct DeoptCounter { method: MethodId, reason: DeoptReason, count: u64 }
enum DeoptError { MissingDebugInfo(SafepointId), ChecksumMismatch(SafepointId), RematCycle(RematRecipeId), UnmappableOsrState(OsrEntryId), StaleDependency(MethodId) }
