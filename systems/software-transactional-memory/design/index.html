Core schema & types (§11)

struct TVar<T> { inner: Arc<Inner<T>> }
struct Inner<T> { value: AtomicPtr<T>, vlock: AtomicU64, waiters: WaitSet }
struct Tx<'w> { world: &'w World, read_set: SmallVec<[ReadEntry; 8]>,
                write_set: SmallVec<[WriteEntry; 8]>, start_version: u64,
                aborted_count: u32, parent: Option<&'w Tx<'w>> }
struct ReadEntry { id: TVarId, observed_version: u64 }
struct WriteEntry { id: TVarId, new_value: *mut () }
struct World { clock: AtomicU64, reclaimer: crossbeam_epoch::Collector }
enum AbortReason { Conflict, Retry, UserAbort(Box<dyn Any + Send>) }
struct VLock(u64)                        // §8.3 bit layout
type TVarId = usize                      // Inner<T> heap address, stable (I6)

fn atomically<F, R>(f: F) -> R where F: Fn(&mut Tx) -> R
fn try_atomically<F, R>(f: F) -> Result<R, AbortReason> where F: Fn(&mut Tx) -> R
impl<T> TVar<T> { fn new(v: T) -> TVar<T> }
impl<'w> Tx<'w> { fn read<T>(&mut self, tvar: &TVar<T>) -> T
                   fn write<T>(&mut self, tvar: &TVar<T>, v: T)
                   fn retry(&mut self) -> !
                   fn or_else<F, G, R>(&mut self, a: F, b: G) -> R }
