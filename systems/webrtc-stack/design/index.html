Core schema & types (§11)

// crates/rtc-core/src/peerconnection.rs — ownership: one per call; mutation: via FSM only
pub struct PeerConnection {
    pub config: RtcConfiguration,
    pub state: ConnectionState,                 // I12
    pub ice_agent: IceAgent,
    pub dtls: DtlsConnection,
    pub srtp: SrtpContext,
    pub rtp_sessions: HashMap<Ssrc, RtpSession>, // I4
    pub data_channels: HashMap<u16, DataChannel>,
    pub local_description: Option<SessionDescription>,
    pub remote_description: Option<SessionDescription>,
    pub local_certificate: Certificate,
    pub remote_fingerprint: Option<CertFingerprint>, // I8
}

pub enum ConnectionState { New, Connecting, Connected, Disconnected, Failed, Closed } // I12 edges only

pub struct RtcConfiguration {
    pub ice_servers: Vec<IceServer>,
    pub ice_transport_policy: IceTransportPolicy, // All | Relay
    pub bundle_policy: BundlePolicy,              // Balanced | MaxCompat | MaxBundle
    pub certificates: Vec<Certificate>,
}

pub struct IceServer { pub urls: Vec<String>, pub username: Option<String>, pub credential: Option<String> }
// crates/rtc-ice/src/agent.rs — ownership: PeerConnection-owned; not Send across sessions (P10)
pub struct IceAgent {
    pub role: IceRole,                     // Controlling | Controlled, I9
    pub state: IceState,                   // §8.2 table
    pub local_credentials: Credentials,
    pub remote_credentials: Option<Credentials>,
    pub local_candidates: Vec<LocalCandidate>,
    pub remote_candidates: Vec<RemoteCandidate>,
    pub candidate_pairs: Vec<CandidatePair>,
    pub selected_pair: Option<CandidatePair>,  // I1 gate
}

pub enum IceRole { Controlling, Controlled }
pub enum IceState { New, Gathering, Checking, Connected, Completed, Disconnected, Failed, Closed }

pub struct LocalCandidate {
    pub foundation: String,
    pub component_id: u32,
    pub priority: u32,                      // I9: pure function, immutable post-gather
    pub address: SocketAddr,
    pub kind: CandidateKind,                // Host | ServerReflexive | PeerReflexive | Relayed
    pub related_address: Option<SocketAddr>,
}

pub struct CandidatePair { pub local: LocalCandidate, pub remote: RemoteCandidate, pub priority: u64, pub state: PairState, pub nominated: bool }
pub enum PairState { Frozen, Waiting, InProgress, Succeeded, Failed }
// crates/rtc-stun/src/message.rs — ownership: stack-allocated per parse; serialization: RFC 8489 TLV
pub struct StunMessage { pub message_type: u16, pub transaction_id: [u8; 12], pub attributes: Vec<StunAttribute> }

pub enum StunAttribute {
    XorMappedAddress(SocketAddr), Username(String), MessageIntegrity([u8; 20]),
    Fingerprint(u32), IceControlled(u64), IceControlling(u64), UseCandidate,
    Priority(u32), Realm(String), Nonce(Vec<u8>),
    ErrorCode { code: u16, reason: String }, AlternateServer(SocketAddr),
}

#[repr(u16)]
pub enum StunMessageType {
    BindingRequest = 0x0001, BindingResponse = 0x0101, BindingErrorResponse = 0x0111,
    AllocateRequest = 0x0003, AllocateResponse = 0x0103, RefreshRequest = 0x0004,
    SendIndication = 0x0016, DataIndication = 0x0017, ChannelBindRequest = 0x0009,
}
// crates/rtc-dtls/src/connection.rs — ownership: PeerConnection-owned; wraps rustls record layer
pub struct DtlsConnection {
    pub role: DtlsRole,                    // Client | Server, per SDP a=setup
    pub state: DtlsHandshakeState,         // I2 gate
    pub local_certificate: Certificate,
    pub remote_certificate: Option<Certificate>,
    pub negotiated_srtp_profile: Option<SrtpProfile>,
    pub epoch: u16,
    pub next_seq: u64,
}
pub enum DtlsRole { Client, Server }
pub enum DtlsHandshakeState { Init, SentClientHello, ReceivedServerHello, WaitFinished, Established, Closed }
pub enum SrtpProfile { Aes128GcmSha256, AeadAes256Gcm, AesCm128HmacSha1_80 } // ADR-07
// crates/rtc-srtp/src/context.rs — ownership: per-direction keys; concurrency: Send, no shared mutable state
pub struct SrtpContext {
    pub local_keys: SrtpSessionKeys,
    pub remote_keys: SrtpSessionKeys,
    pub replay_window: ReplayWindow,       // I3, P7 fixed-size bitmask
    pub rollover_counter: u32,             // ROC per RFC 3711 3.3.1
}
pub struct SrtpSessionKeys { pub master_key: [u8; 16], pub master_salt: [u8; 14], pub profile: SrtpProfile }
pub struct ReplayWindow { pub highest_seq: u16, pub bitmask: u64 } // fixed 64-packet window (P7)
// crates/rtc-rtp/src/packet.rs — ownership: Bytes-backed, zero-copy (P11)
pub struct RtpHeader {
    pub version: u8, pub padding: bool, pub extension: bool, pub csrc_count: u8,
    pub marker: bool, pub payload_type: u8, pub sequence_number: u16,
    pub timestamp: u32, pub ssrc: u32, pub csrc_list: Vec<u32>,
}
pub struct RtpPacket { pub header: RtpHeader, pub payload: Bytes }
pub struct RtcpFeedback { pub kind: FeedbackKind, pub media_ssrc: u32 } // NACK|PLI|FIR|Twcc
pub enum FeedbackKind { Nack(Vec<u16>), Pli, Fir { seq_nr: u8 }, Twcc(Vec<TwccArrival>) }
// crates/rtc-sctp/src/association.rs — ownership: one per DTLS connection; I5 TSN monotonicity
pub struct SctpAssociation {
    pub state: AssociationState,           // §8.5 table
    pub local_verification_tag: u32,
    pub remote_verification_tag: u32,
    pub next_tsn: u32,                     // I5
    pub streams: HashMap<u16, SctpStream>,
    pub rto: Duration,                     // ADR-23, RFC 4960 6.3 default 1s
}
pub enum AssociationState { Closed, CookieWait, CookieEchoed, Established, ShutdownPending, ShutdownSent }
pub struct DataChannel { pub stream_id: u16, pub label: String, pub reliability: Reliability, pub ordered: bool }
pub enum Reliability { Reliable, PartialRtx(u16), PartialTimed(Duration) }
// crates/rtc-sdp/src/model.rs — ownership: parsed per offer/answer; immutable after parse
pub struct SessionDescription { pub origin: Origin, pub media: Vec<MediaDescription> } // I14: order preserved
pub struct MediaDescription {
    pub mid: String, pub media_type: MediaType, pub ice_ufrag: String, pub ice_pwd: String,
    pub fingerprint: CertFingerprint, pub setup: DtlsSetupRole, pub rtpmaps: Vec<RtpMap>,
}
pub enum MediaType { Audio, Video, Application }
pub enum DtlsSetupRole { ActPass, Active, Passive }
pub struct RtpMap { pub payload_type: u8, pub codec: String, pub clock_rate: u32 }
