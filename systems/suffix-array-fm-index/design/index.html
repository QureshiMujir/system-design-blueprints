Core schema & types (§11)

// ---- alphabet & text ----
type Sym = u8;                       // remapped dense symbol in [0, sigma)
type Idx = u32;                      // SA index width; u64 behind feature="wide"
struct Alphabet { map: [Sym; 256], inv: Vec<u8>, sigma: u16 }   // byte <-> dense symbol
struct SentinelText { syms: Box<[Sym]>, n: usize }              // T + '$'(=0); n = |T|+1

// ---- suffix array & LCP ----
struct SuffixArray { sa: Box<[Idx]> }                // permutation of [0, n) (I1, I2)
struct InverseSA   { isa: Box<[Idx]> }               // isa[SA[i]] = i
struct LcpArray    { lcp: Box<[Idx]> }               // lcp[0]=0; lcp[i]=lcp(SA[i-1],SA[i]) (I6)
struct Rmq         { /* sparse table or Fischer-Heun */ }    // O(1) range-min over LCP (LCE)

// ---- BWT & LF ----
struct Bwt { last: Box<[Sym]>, primary: Idx }        // last column; primary = row with SA[p]=0
struct CTable { c: [usize; 257] }                    // C[c] = #symbols < c (I10)
trait OccRank { fn occ(&self, c: Sym, i: usize) -> usize; }   // rank of c in BWT[0..i)

// ---- succinct ----
struct BitVector { words: Box<[u64]>, len: usize }
struct RankSupport { superblock: Box<[u64]>, block: Box<[u16]> }   // O(1) rank, <=0.25n bits (I8)
struct SelectSupport { samples: Box<[u32]> }                       // sampled 1-positions (I9)
struct WaveletTree { /* balanced or Huffman; bitvectors + rank per level */ }  // O(log sigma)
struct RrrVector { classes: Box<[u8]>, offsets: BitStream, samples: Box<[(u64,u64)]> } // n*H0+o(n)

// ---- FM-index, locate, extract ----
struct FmIndex<O: OccRank> { bwt: Bwt, c: CTable, occ: O, n: usize }
struct Interval { sp: usize, ep: usize }             // half-open [sp, ep) (I7)
struct SaSamples { marks: BitVector, rank: RankSupport, values: Box<[Idx]>, rate: u32 } // (I11)
struct IsaSamples { positions: Box<[Idx]>, rate: u32 }                                  // (I12)
// ---- suffix automaton ----
struct SamState { len: u32, link: i32, next: BTreeMap<Sym, u32> }   // endpos class
struct SuffixAutomaton { states: Vec<SamState>, last: u32, n: usize }  // <=2n-1 states (I13)

// ---- r-index ----
struct Run { head: Sym, len: Idx }
struct RlBwt { runs: Box<[Run]>, r: usize }          // r = #runs; sum len = n (I15)
struct RIndex { rlbwt: RlBwt, sa_at_run_bounds: Box<[(Idx, Idx)]>, toehold: Idx }

// ---- the public index & API ----
enum IndexKind { FmSampled, RIndex }
struct Index {                                        // immutable; Send + Sync (I18)
    alphabet: Alphabet, n: usize, kind: IndexKind,
    fm: Option<FmIndex<WaveletTree>>, sa_samples: Option<SaSamples>,
    isa_samples: Option<IsaSamples>, rindex: Option<RIndex>, version: u16,
}

impl Index {
    fn build(text: &[u8], params: &BuildParams) -> Result<Index, BuildError>;
    fn count(&self, pat: &[u8]) -> u64;                          // O(m * t_rank)
    fn locate(&self, pat: &[u8]) -> Vec<usize>;                  // O(m + occ*s) * t_rank
    fn extract(&self, i: usize, j: usize) -> Result<Vec<u8>, QueryError>;  // O((j-i+s)*t_rank)
    fn lrs(&self) -> Option<(usize, usize)>;                     // (pos, len) longest repeated
    fn distinct_substrings(&self) -> u64;                        // SA+LCP (cross-checked vs SAM)
    fn space_breakdown(&self) -> SpaceReport;                    // bits per structure (DP5)
    fn save(&self, w: &mut impl Write) -> io::Result<()>;        // versioned, length-prefixed
    fn load(bytes: &[u8]) -> Result<Index, IndexError>;          // bounds-checked (I17, DP7)
}

struct BuildParams { sa_sample_rate: u32, bitvector: BvKind, wavelet: WtShape,
                     repetitive_threshold: f32, build_sam: bool }
enum BvKind { Plain, Rrr }   enum WtShape { Balanced, Huffman }

// ---- error taxonomy (typed; no panic on input) ----
enum BuildError { SentinelCollision, WidthOverflow, Alphabet, Internal(&'static str) }
enum IndexError { BadMagic, Version{found:u16,expected:u16}, Truncated{field:&'static str,need:usize,have:usize},
                  CorruptPermutation, CorruptRank, Decode(&'static str) }
enum QueryError { Range{i:usize,j:usize,n:usize}, NotBuilt(&'static str) }
