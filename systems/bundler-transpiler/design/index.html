Core schema & types (§11)

/// Byte offsets into a single SourceFile. Half-open [start, end).
/// 32-bit caps a source file at 4 GiB — enforced at read time.
pub struct Span { pub start: u32, pub end: u32 }

/// Dense index into the source-file table. Copy. u32::MAX = synthetic/no-source.
pub struct SourceId(pub u32);

pub enum TokenKind {
    Ident, PrivateName, Keyword(Keyword), Num(f64), BigInt, Str,
    NoSubTemplate, TemplateHead, TemplateMiddle, TemplateTail,
    Regex, Punct(Punct),
    JsxText, JsxIdent,
    Eof,
    Invalid,                 // recovery token; never panics
}

pub struct Token {
    pub kind: TokenKind,
    pub span: Span,
    pub had_newline_before: bool,   // drives ASI in the parser
}

/// Borrows source for its lifetime; single-threaded; one Lexer per SourceFile.
pub struct Lexer<'a> { /* src: &'a str, pos: u32, regex_ctx: bool, tmpl_stack: Vec<()> */ }
/// Dense node id, assigned at parse time, stable for a Program's lifetime.
pub struct NodeId(pub u32);

pub enum SourceType { Module, Script }

pub struct Program { pub items: Vec<ModuleItem>, pub source_type: SourceType }

pub enum ModuleItem { Stmt(Stmt), Import(ImportDecl), Export(ExportDecl) }

pub enum Stmt {
    Expr(Box<Expr>), Var(VarDecl), Func(FuncDecl), Class(ClassDecl),
    Return(/* .. */), If(/* .. */), For(/* .. */), Block(Vec<Stmt>),
    // While, DoWhile, Switch, Try, Throw, Break, Continue, Labeled, Empty, Debugger ...
}

pub enum Expr {
    Ident(Ident), Lit(Lit), Bin(BinExpr), Call(CallExpr), Member(MemberExpr),
    Arrow(ArrowExpr), Object(/* .. */), Array(/* .. */), Template(/* .. */),
    Import(ImportExpr),          // dynamic import()
    Assign(/* .. */), Optional(OptChain),
    // Unary, Update, Logical, Cond, New, Seq, Tagged, Class, Func, Await, Yield, Paren ...
}

pub enum Pat { Ident(Ident), Array(/* .. */), Object(/* .. */), Rest(/* .. */),
               Assign(/* default */), Expr(Box<Expr>) /* member target */ }

pub struct Ident { pub sym: Atom, pub span: Span, pub node_id: NodeId }

pub struct ImportDecl { pub specifiers: Vec<ImportSpec>, pub source: Atom, pub span: Span }
// ImportSpec ::= Default(Ident) | Namespace(Ident) | Named { imported: Atom, local: Ident }

pub enum ExportDecl {
    Named { specifiers: Vec<ExportSpec>, source: Option<Atom> },  // export {a as b} [from "x"]
    Default(Box<Expr>),                                           // export default <expr>
    All { source: Atom, alias: Option<Atom> },                   // export * [as ns] from "x"
    Decl(Box<Stmt>),                                             // export const/function/class
}

pub trait Visit   { /* read-only walk; default impls recurse */ }
pub trait VisitMut{ /* in-place rewrite; transform passes implement this */ }
pub struct Syntax { pub typescript: bool, pub jsx: bool }   // gates TS/JSX grammar

/// Borrows source; holds the Lexer, 1-token lookahead, error sink, syntax flags.
pub struct Parser<'a> { /* lexer: Lexer<'a>, cur: Token, ahead: Option<Token>,
                           errors: Vec<ParseError>, syntax: Syntax */ }

/// Total: never panics. Returns a (possibly partial) Program + accumulated errors.
pub fn parse_program(src: &str, id: SourceId, syntax: Syntax)
    -> (Program, Vec<ParseError>);

pub struct ParseError { pub kind: ParseErrorKind, pub span: Span }

pub enum ParseErrorKind {
    UnexpectedToken, UnterminatedString, UnterminatedTemplate, InvalidRegex,
    BadImport, BadExport, ReservedWord, TrailingTsType, JsxNotEnabled,
}
pub enum Condition { Import, Require, Browser, Node, Default,
                     Development, Production, Custom(Atom) }

pub struct ResolveOptions {
    pub conditions: Vec<Condition>,     // ordered; first match wins in exports map
    pub extensions: Vec<String>,        // probe order, e.g. [".ts",".tsx",".js",".jsx",".json"]
    pub main_fields: Vec<String>,       // e.g. ["browser","module","main"]
    pub ts_config: Option<TsConfig>,    // paths/baseUrl — applied before node_modules
    pub prefer_relative: bool,
}

pub enum Specifier {
    Relative(String),                   // "./x", "../y"
    Absolute(String),                   // "/abs", "C:\\abs"
    Bare { pkg: String, sub: String },  // "react" / "lodash/fp" -> pkg + subpath
}

pub struct ResolveResult {
    pub path: PathBuf,                  // realpath'd (symlinks resolved)
    pub kind: ModuleKind,               // from extension + package "type"
    pub side_effects: SideEffects,      // package.json sideEffects -> tree-shake hint
    pub from_node_modules: bool,        // pre-bundling / external heuristics
}

pub enum ResolveError { NotFound, InvalidExports, PackageNotFound, NotExported, BadTsPaths }

/// Caches fs stat, parsed package.json, and realpath. Shared read-mostly.
pub struct Resolver { /* opts, fs_cache, pkg_cache, realpath_cache */ }
pub struct ModuleId(pub u32);

pub enum ModuleKind { Esm, Cjs, Json, Css, Asset, Builtin, External }

pub struct SymbolId(pub u32);

pub struct Symbol { pub name: Atom, pub kind: SymbolKind, pub local: bool }
// SymbolKind ::= Var | Func | Class | Import | Default | Namespace | ...

pub struct ImportRecord {
    pub specifier: Atom,
    pub resolved: Option<ModuleId>,     // None until resolution wires the edge
    pub kind: ImportKind,
}
pub enum ImportKind { Static, Dynamic, Require, ExportStar }

pub struct ExportRecord {
    pub name: Atom,                              // exported name ("default" for default)
    pub local: Option<SymbolId>,                 // Some -> local binding
    pub reexport: Option<(ModuleId, Atom)>,      // Some -> re-export from another module
}

pub struct Module {
    pub id: ModuleId,
    pub path: PathBuf,
    pub kind: ModuleKind,
    pub ast: Program,                  // frozen after build
    pub imports: Vec<ImportRecord>,
    pub exports: Vec<ExportRecord>,
    pub symbols: Vec<Symbol>,          // indexed by SymbolId
    pub has_side_effects: bool,        // top-level effects pin the module
}

pub struct ModuleGraph {
    pub modules: Vec<Module>,          // indexed by ModuleId
    pub entries: Vec<ModuleId>,
    /* edges (derived from ImportRecord.resolved), cycle set (SCCs) */
}
pub enum Target { Es5, Es2015, Es2017, Es2020, Es2022, EsNext }

/// name -> replacement Expr (e.g. process.env.NODE_ENV -> "production").
pub struct DefineMap(/* Map<Atom, Expr> */);

pub trait TransformPass {
    fn name(&self) -> &str;
    fn visit_mut(&mut self, p: &mut Program, ctx: &mut TransformContext);
}

pub enum JsxRuntime { Classic, Automatic }    // createElement vs jsx-runtime

/// Constructed per worker per module. Holds shared (&) read-only inputs +
/// per-pass mutable scratch. NEVER shared across threads.
pub struct TransformContext<'a> {
    pub target: Target,
    pub defines: &'a DefineMap,        // shared read-only
    pub syntax: Syntax,
    pub jsx_runtime: JsxRuntime,
}
/// Per-symbol dependency edges + side-effect markers, derived from the graph.
pub struct SymbolGraph { /* edges: SymbolId -> [SymbolId], effects: [SideEffect] */ }

pub enum SideEffect { Pure, Impure, Unknown }   // Unknown is treated conservatively

/// Worklist reachability from entry exports.
pub struct TreeShaker { /* worklist: Vec<SymbolId>, reached: BitSet */ }

/// Marks reachable symbols; mutates graph metadata (not AST shape) for pruning.
pub fn shake(graph: &mut ModuleGraph, entries: &[ModuleId]) -> ReachableSet;
// ReachableSet: bitset of reachable (ModuleId, SymbolId); statements/exports/imports
// not in the set are eliminated by codegen.
pub struct ChunkId(pub u32);

pub enum OutputFormat { Esm, Cjs, Iife }

pub struct Chunk {
    pub id: ChunkId,
    pub modules: Vec<ModuleId>,        // in deterministic order
    pub entry: Option<ModuleId>,       // Some -> entry chunk
    pub is_dynamic: bool,              // produced by a dynamic import()
    pub hash: u64,                     // content hash (fixed-point across xchunk refs)
    pub imports: Vec<ChunkId>,         // cross-chunk edges
}

pub struct ChunkGraph { pub chunks: Vec<Chunk> }   // indexed by ChunkId

pub fn split(graph: &ModuleGraph, reachable: &ReachableSet, fmt: OutputFormat) -> ChunkGraph;
/// Scopes, bindings, references, scope ids. Built by scope analysis (Phase 8).
pub struct ScopeTree { /* scopes, bindings, refs, parent links */ }

/// Scope-aware rename map; produces collision-free names (short names for minify).
pub struct Renamer { /* per-scope rename table, used-name reservation */ }

/// Writer + indent + a source-map builder handle from project #265.
pub struct Emitter { /* out: String, indent, map_builder: SourceMapBuilder /*#265*/ */ }

/// mangle + DCE + fold passes over an AST already lowered/transformed.
pub struct Minifier { /* mangle: bool, fold: bool, dce: bool */ }

pub struct OutputChunk {
    pub file_name: String,
    pub code: String,
    pub map: Option<SourceMap>,        // #265 codec; bundler never hand-rolls VLQ
}

pub fn emit_chunk(chunk: &Chunk, graph: &ModuleGraph, fmt: OutputFormat,
                  opts: &CodegenOptions) -> OutputChunk;
/// BLAKE3 of (source content + relevant options). Equal key => reuse artifact.
pub struct CacheKey(pub [u8; 32]);

/// In-memory + on-disk content-addressed store. Thread-safe.
pub struct Cache { /* mem: ShardedMap<CacheKey, Artifact>, dir: PathBuf */ }

pub enum Artifact { Parsed(Program), Transformed(Program), Emitted(OutputChunk) }
/// http + ws, owns graph + transform pipeline + cache + watcher.
pub struct DevServer { /* listener, graph: ModuleGraph, cache: Cache, watcher: Watcher */ }

/// Boundary computation + accepted-modules tracking.
pub struct HmrEngine { /* boundaries, accepted: Set<ModuleId> */ }

pub struct HmrBoundary { pub module: ModuleId, pub accepts: bool }

/// fs events -> invalidation signals into the graph + cache.
pub struct Watcher { /* fs watch, debounce, ModuleId mapping */ }
/// Hooks fire in order: resolve -> load -> transform -> render_chunk.
/// Each returns Option: Some short-circuits the default for that stage.
pub trait Plugin {
    fn resolve(&self, /* specifier, importer, opts */) -> Option<ResolveResult>;
    fn load(&self, /* path */) -> Option<String>;
    fn transform(&self, /* code, path, program */) -> Option<Program>;
    fn render_chunk(&self, /* chunk, code */) -> Option<OutputChunk>;
}

/// Subcommands: build serve transform analyze graph.
pub fn run(args: CliArgs) -> ExitCode;
