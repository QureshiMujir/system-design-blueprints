Core schema & types (§11)

#[repr(transparent)] pub struct PageId(pub u64);     // dense; file_offset = id * PAGE_SIZE
pub struct RowId { pub page: PageId, pub slot: u16 } // 10-byte logical row addr; slot indexes SlottedPage dir
#[repr(transparent)] pub struct SegmentId(pub u64);  // one column-segment / dict blob file region
#[repr(transparent)] pub struct ColumnId(pub u32);   // ordinal within Schema
#[repr(transparent)] pub struct RowGroupId(pub u64); // unit of columnar storage (65_536 rows)
#[repr(transparent)] pub struct TxnId(pub u64);      // monotonic; 0 = uncommitted/none
#[repr(transparent)] pub struct Lsn(pub u64);        // byte offset into WAL; monotonic; gates redo
#[repr(transparent)] pub struct Timestamp(pub u64);  // commit-order clock (monotonic, HLC-free)
pub enum DataType { Bool, I32, I64, F64, Decimal{ p: u8, s: u8 }, Str, Bytes, Date, Timestamp }
pub enum Datum { Null, Bool(bool), I32(i32), I64(i64), F64(f64), Str(Box<str>), Bytes(Box<[u8]>) }
pub struct Error { pub kind: ErrorKind, /* context */ }
pub enum ErrorKind { Io, Corruption, Checksum, OutOfSpace, Conflict, NotFound, BadEncoding }
pub const PAGE_SIZE: usize = 4096;
#[repr(C)] pub struct PageHeader { pub lsn: Lsn, pub kind: PageKind, pub free_ptr: u16,
                                   pub checksum: u32, pub _pad: [u8;_] }
pub struct Page { /* 4 KiB, 8-byte-aligned buffer; header at offset 0 */ }
pub enum PageKind { Heap, BTreeLeaf, BTreeInternal, ColumnData, Meta, Free }
pub trait Pageable { fn page_kind() -> PageKind; }

| lsn  |checksum|kind|free_ptr| reserved|  payload (slotted heap / node / col) |
| u64  | u32    | u8 | u16    |  4 B    |              4073 B                  |
+------+--------+----+--------+---------+-----------------------------------+
pub struct BufferPool { /* frame table[N], clock hand, pin_count[], dirty[], page_table: HashMap<PageId,FrameId> */ }
impl BufferPool {
    pub fn fetch(&self, id: PageId) -> Result<PageGuard<'_>>;   // pins; reads+verifies on miss
    pub fn new_page(&self, kind: PageKind) -> Result<PageGuard<'_>>;
    pub fn flush(&self, id: PageId) -> Result<()>;              // WAL-flush(page.lsn) FIRST, then write
}
pub struct PageGuard<'a> { /* &BufferPool + FrameId; Deref<Target=Page>; unpins on Drop */ }
pub struct FileManager { /* segment files; read_page/write_page/extend; one file per SegmentId range */ }
#[repr(C)] pub struct TupleHeader {
    pub txn_min: TxnId,        // creator; visible iff committed-before-snapshot
    pub txn_max: TxnId,        // deleter/superseder; 0 = live
    pub next_version: RowId,   // version chain: newest -> ... -> oldest
    pub flags: u8,             // DELETED | LOCKED | bits
    pub null_bitmap: SmallBitset, // inline; 1 bit / column
}
pub struct Tuple<'a> { pub header: &'a TupleHeader, pub payload: &'a [u8] } // borrows page bytes; decode via Schema
pub struct SlottedPage<'a> { /* slot directory grows ->, heap grows <-; free = gap */ }
pub struct RowStore { /* heap of Heap pages over BufferPool; free-space map */ }
impl RowStore {
    pub fn insert(&self, hdr: TupleHeader, payload: &[u8]) -> Result<RowId>;
    pub fn get(&self, rid: RowId) -> Result<Option<Tuple<'_>>>;
    pub fn update_as_version(&self, rid: RowId, new: &[u8], txn: TxnId) -> Result<RowId>; // links next_version
    pub fn delete_as_version(&self, rid: RowId, txn: TxnId) -> Result<()>;               // sets txn_max
}
pub struct BTree { /* root PageId in Meta page; key = primary-key bytes (memcmp-ordered) */ }
pub enum BTreeNode { Leaf{ keys: Vec<Box<[u8]>>, rids: Vec<RowId> },
                     Internal{ keys: Vec<Box<[u8]>>, children: Vec<PageId> } }
impl BTree {
    pub fn lookup(&self, key: &[u8], snap: &Snapshot) -> Option<RowId>; // then walk version chain for visibility
    pub fn range(&self, lo: &[u8], hi: &[u8], snap: &Snapshot) -> RangeIter;
    pub fn insert(&self, key: &[u8], rid: RowId) -> Result<()>;          // split on overflow
}
pub enum WalRecord {
    Begin{ txn: TxnId },
    Insert{ txn: TxnId, rid: RowId, after: Box<[u8]> },
    Update{ txn: TxnId, rid: RowId, before: Box<[u8]>, after: Box<[u8]> },
    Delete{ txn: TxnId, rid: RowId, before: Box<[u8]> },
    Commit{ txn: TxnId, ts: Timestamp },
    Abort{ txn: TxnId },
    Clr{ txn: TxnId, undo_next: Lsn },          // compensation: makes undo idempotent
    ColumnFlush{ rg: RowGroupId, seg: SegmentId }, // tuple-mover output durable
    Checkpoint(CheckpointRec),                   // ATT + DPT snapshot
}
pub struct Wal { /* append-only file + in-mem tail buf + group commit + flush(lsn) */ }
impl Wal {
    pub fn append(&self, rec: &WalRecord) -> Result<Lsn>;   // returns assigned Lsn = byte offset
    pub fn flush(&self, upto: Lsn) -> Result<()>;           // fsync; group commit batches waiters
}
pub struct Recovery { /* analysis -> redo -> undo */ }
pub struct CheckpointRec { pub att: Vec<(TxnId,Lsn)>, pub dpt: Vec<(PageId,Lsn)>, pub ts: Timestamp }
pub enum TxnState { Active, Committed, Aborted }
pub enum IsolationLevel { SnapshotIsolation, ReadCommitted }
pub struct Snapshot { pub xmin: TxnId, pub xmax: TxnId, pub active: Box<[TxnId]>, pub ts: Timestamp }
pub struct Transaction { pub id: TxnId, pub snapshot: Snapshot, pub state: TxnState }
pub struct VersionChain;  // logical: follow RowId.next_version via TupleHeader
pub struct TxnManager { /* id oracle, commit-ts oracle, active set, Vacuum */ }
impl TxnManager {
    pub fn begin(&self, iso: IsolationLevel) -> Transaction;       // snapshots active set
    pub fn commit(&self, txn: &mut Transaction) -> Result<Timestamp>; // WAL Commit + fsync, assign ts
    pub fn abort(&self, txn: &mut Transaction) -> Result<()>;      // undo via CLRs
    pub fn vacuum(&self) -> Result<usize>;                         // GC versions < oldest live snapshot
}
pub fn visible(h: &TupleHeader, snap: &Snapshot) -> bool;
visible(h,snap) = committed_before(h.txn_min, snap) && h.txn_min ∉ snap.active
                  && (h.txn_max == 0 || !committed_before(h.txn_max, snap) || h.txn_max ∈ snap.active)
pub struct RowGroup { pub id: RowGroupId, pub n_rows: u32, pub columns: Vec<ColumnSegment> } // n_rows ≤ 65_536
pub struct ColumnSegment { pub col: ColumnId, pub encoding: Encoding, pub zone: ZoneMap,
                           pub n_rows: u32, pub null_count: u32, pub data: SegmentId }
pub enum Encoding { Plain, Dictionary{ dict: SegmentId }, Rle, BitPacked{ bits: u8 },
                    FrameOfReference{ base: i64, bits: u8 } }
pub struct ZoneMap { pub min: Datum, pub max: Datum, pub null_count: u32 }
pub struct SelectionVector(pub Vec<u32>);  // surviving row indices after a predicate
pub struct ColumnBatch { /* up to 1024–2048 decoded values + SelectionVector; vectorization unit */ }
pub trait ColumnReader { fn scan(&self, sel: &SelectionVector, out: &mut ColumnBatch); }
+--------+----------+-----------+----------------+-------------------+
| magic  | enc tag  | n_rows u32| null_count u32 | encoded data ...  | crc32c (footer)
+--------+----------+-----------+----------------+-------------------+
pub struct DeltaStore { /* RowStore (recent writes, row format) + per-row validity bitmap */ }
pub struct MainStore  { /* immutable Vec<RowGroup> + per-row-group delete vector (bitmap) */ }
pub struct TupleMover { /* delta->main merge: build RowGroups, pick encodings, write segs, atomic swap, GC */ }
pub enum AccessPath { RowLookup(RowId), IndexLookup, ColumnScan(ScanSpec), DeltaScan }
pub struct Predicate { /* col op const, conjuncts; drives zone-map pruning */ }
pub struct ScanSpec { pub cols: Vec<ColumnId>, pub predicate: Option<Predicate>, pub snap: Snapshot }
pub struct HybridTable { /* RowStore+BTree (point) | DeltaStore | MainStore (scan) | Wal | TxnManager | Schema */ }
impl HybridTable {
    pub fn get(&self, key: &[u8], txn: &Transaction) -> Option<Tuple>;  // BTree -> delta/row, visibility-filtered
    pub fn scan(&self, spec: &ScanSpec) -> BatchIter;                   // merge DELTA(row)+MAIN(column) under snap
}
pub struct BatchIter; // yields ColumnBatch; concatenates main row-groups then delta rows, both snapshot-filtered
