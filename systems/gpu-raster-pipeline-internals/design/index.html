Core schema & types (§11)

// ownership: DrawCall is immutable once submitted; owned by graster-cmdproc queue.
// mutation: none post-validation. concurrency: Send+Sync, read-only after validation.
// persistence: serialized into graster-trace. stage: draw-call front end (host-side, on-chip N/A).
struct DrawCall { id: u64, pipeline_state_id: u32, primitive_count: u32, bindings: BindingSet }

// ownership: singleton per simulated device. mutation: internally synchronized queue state.
// concurrency: thread-safe issue path. persistence: not persisted, rebuilt per run.
struct CommandProcessor { queue: DrawCallQueue, validation: ValidationRules }

// ownership: owned by graster-geom per vertex-fetch batch. mutation: write-once per fetch.
// stage: vertex fetch (fixed-function), on-chip: none (streamed from VRAM).
struct Vertex { position: Vec4, attribs: Vec<VertexAttrib> }

// ownership: borrowed view into a vertex buffer. immutable. serialization: layout-versioned.
struct VertexAttrib { semantic: AttribSemantic, format: AttribFormat, offset: u32 }

// ownership: owned by a SimtCore slot; mutable per-cycle. concurrency: lockstep across Lanes.
// stage: vertex/fragment programmable stages, on-chip: register file.
struct Warp { lanes: [Lane; WARP_WIDTH], exec_mask: ExecMask, pc: u32 }

// ownership: owned by its parent Warp. mutation: per-instruction. persistence: none.
struct Lane { registers: RegisterFile, active: bool, thread_id: u64 }

// ownership: stack-owned by a Warp during divergent control flow. mutation: push/pop only.
struct ExecMask { bits: u64, reconverge_pc: u32 }

// ownership: owned by graster-simt scheduler. mutation: warp-slot occupancy per cycle.
// concurrency: schedules across resident Warps for latency hiding (P1).
struct SimtCore { resident_warps: Vec<Warp>, occupancy: Occupancy, scheduler: WarpScheduler }

// ownership: computed value, not stored mutable state. persistence: reported via PipelineStats.
struct Occupancy { register_limited: u32, shared_mem_limited: u32, scheduler_limited: u32, resolved: u32 }

// ownership: owned by graster-geom during primitive assembly. mutation: write-once per triangle.
// stage: primitive assembly/clipping (fixed-function), on-chip: register file (post-VS).
struct Primitive { vertices: [Vertex; 3], topology: Topology }

// ownership: derived from Primitive after clipping. immutable. stage: rasterizer input.
struct Triangle { v0: Vec4, v1: Vec4, v2: Vec4, winding: Winding }

// ownership: owned by graster-raster per triangle. mutation: write-once at setup.
// stage: triangle setup (fixed-function).
struct TriangleSetup { edges: [EdgeFunction; 3], bbox: BoundingBox, area: f64 }

// ownership: value type, immutable. stage: rasterizer (fixed-function).
struct EdgeFunction { a: f64, b: f64, c: f64, is_top_or_left: bool }
// ownership: owned by graster-raster; stateless across triangles besides Hi-Z. concurrency: per-tile parallel.
struct Rasterizer { hiz: HiZ }

// ownership: per-fragment value, immutable once computed. stage: rasterizer output.
struct Coverage { sample_mask: u32 }

// ownership: owned by graster-raster, persists across a render pass. mutation: mip-chain updates.
// stage: rasterizer (fixed-function), on-chip: SRAM mip cache.
struct HiZ { mip_chain: Vec<DepthMip> }

// ownership: owned by a Quad; mutation: write-once per shading pass. stage: fragment stage input.
// persistence: none, transient.
struct Fragment { screen_pos: (u32, u32), depth: f32, coverage: Coverage, is_helper: bool }

// ownership: owned by graster-frag scheduler, groups 4 Fragments. stage: fragment stage (I2).
struct Quad { fragments: [Fragment; 4] }

// ownership: computed per Quad, immutable. stage: fragment stage / texture-unit input.
struct Derivatives { du_dx: f32, du_dy: f32, dv_dx: f32, dv_dy: f32 }

// ownership: owned by graster-rop per render-target tile. mutation: read-modify-write per fragment.
// stage: ROP (fixed-function), on-chip for TBDR / VRAM-direct for IMR.
struct DepthStencil { depth_buffer: DepthMip, stencil_buffer: Vec<u8> }

// ownership: stateless gate function, no owned state besides config flags. stage: pre-fragment.
struct EarlyZ { enabled: bool, defeat_reason: Option<EarlyZDefeat> }

// ownership: owned by graster-frag; interprets graster-isa instructions per Warp. stage: fragment stage.
struct FragmentShaderStage { isa_program: IsaProgram, simt: SimtCore }

// ownership: owned by graster-tex; stateless besides cache. stage: texture units.
struct TextureUnit { cache: TextureCache }

// ownership: value type, immutable. stage: texture-unit input.
struct TexAddr { u: f32, v: f32, mip: f32 }

// ownership: pure function config, no mutable state. stage: texture-unit addressing.
struct Swizzle { order: MortonOrder }

// ownership: owned by graster-tex per SM/CU. mutation: LRU update per fetch. persistence: none.
// stage: texture units, on-chip L1, backed by L2/VRAM (graster-mem).
struct TextureCache { l1_lines: Vec<CacheLine>, hit_count: u64, miss_count: u64 }

// ownership: owned by graster-rop, stateless besides blend config. stage: ROP.
struct Rop { blend: BlendState, msaa: Msaa }

// ownership: value type, immutable per pipeline state object. stage: ROP.
struct BlendState { src_factor: BlendFactor, dst_factor: BlendFactor, op: BlendOp }
// ownership: owned by graster-rop per render target. mutation: per-sample write. stage: ROP.
struct Msaa { sample_count: u32, sample_mask: SampleMask, resolve_filter: ResolveFilter }

// ownership: value type. stage: ROP / MSAA.
struct SampleMask { bits: u32 }

// ownership: owned by graster-tbdr per tile, on-chip SRAM. mutation: accumulates until flush (I6).
struct TileBuffer { tile_id: (u32, u32), color: Vec<Rgba>, depth: Vec<f32>, flushed: bool }

// ownership: singleton per simulated device. mutation: bandwidth counters per cycle.
// stage: cross-cutting, on-chip (RF/L1/L2) vs off-chip (VRAM) explicitly tagged.
struct MemoryHierarchy { register_file: RegisterFile, l1: Cache, l2: Cache, vram: Vram }

// ownership: computed/aggregated value. persistence: reported via PipelineStats.
struct Bandwidth { bytes_transferred: u64, peak_gbps: f64, utilization: f32 }

// ownership: owned by graster-stats, append-only per stage per cycle. persistence: exported/serialized.
struct PipelineStats { per_stage: HashMap<StageId, StageCounters>, invariant_violations: Vec<InvariantEvent> }

// error taxonomy: validation errors (reject at front end), fidelity errors (invariant violation,
// I1-I12), resource errors (occupancy/texture/memory bounds), determinism errors (replay-hash
// mismatch, I11), format errors (graster-trace version skew, §27).
enum GRasterError { Validation(String), Fidelity(InvariantId), Resource(ResourceKind), Determinism, FormatVersion(u32) }
