Core schema & types (§11)

// Observer: the sink contract. Grammar enforced by a wrapper, not by callers.
trait Observer<T> {
    fn on_next(&mut self, value: T);
    fn on_error(&mut self, err: ReactiveError);
    fn on_complete(&mut self);
}

// Observable: cold producer. subscribe installs an Observer, returns a Subscription.
struct Observable<T> { on_subscribe: Rc<dyn Fn(SafeObserver<T>) -> Subscription> }
impl<T> Observable<T> {
    fn subscribe(&self, obs: impl Observer<T> + 'static) -> Subscription;
    fn lift<R>(&self, op: impl Operator<T, R> + 'static) -> Observable<R>;
    // pipe(...) -> Observable<...>            // operator composition sugar
}

// SafeObserver: grammar guard. Enforces next* (error|complete)?; drops post-terminal.
struct SafeObserver<T> { /* inner: Box<dyn Observer<T>>, state: SubState, teardown: Teardown */ }

// Operator: lift unit. Wraps a downstream observer, returns the upstream-facing observer.
trait Operator<In, Out> { fn call(&self, downstream: SafeObserver<Out>) -> SafeObserver<In>; }

// Subscription: teardown handle. Idempotent unsubscribe. Composable (add child).
struct Subscription { /* state: SubState, teardowns: Vec<Teardown>, children: Vec<Subscription> */ }
impl Subscription {
    fn unsubscribe(&self);                       // idempotent, total (I2)
    fn add(&self, child: Subscription);          // compose inner subscriptions
    fn is_closed(&self) -> bool;
}
enum SubState { Idle, Subscribed, Unsubscribed, Errored, Completed }  // FSM (I13)

// Subject family: multicast. Subject is hot (no replay); Behavior keeps last; Replay keeps N.
struct Subject<T> { /* observers: Vec<Weak<SafeObserver<T>>>, state: SubState */ }
struct BehaviorSubject<T> { /* current: T, ... */ }
struct ReplaySubject<T> { /* buffer: VecDeque<T>, cap: usize, ... */ }

enum Notification<T> { Next(T), Error(ReactiveError), Complete }  // materialize/dematerialize
enum ReactiveError { Timeout, Cancelled, Cycle(Vec<NodeId>), Overflow, User(Box<dyn Error + 'static>) }
trait Scheduler {
    fn schedule(&self, action: SchedulerAction) -> Disposable;     // now
    fn schedule_at(&self, delay: VirtualOrReal, action: SchedulerAction) -> Disposable;
    fn now(&self) -> Instant;                                      // virtual for TestScheduler
}
type SchedulerAction = Box<dyn FnOnce(&dyn Scheduler)>;            // may re-schedule (recursive op)
struct Disposable { /* teardown */ }                              // cancels a scheduled action

struct ImmediateScheduler;        // synchronous; reentrancy-guarded
struct QueueScheduler;            // trampoline: drains a FIFO work queue (bounds recursion, I7)
struct AsyncScheduler<H: AsyncHost>;  // defers to host microtask/timer
struct AnimationFrameScheduler;   // stub hook
struct TestScheduler { /* clock: VirtualClock, queue: BinaryHeap<(VTime, seq, action)> */ }
trait AsyncHost { fn post_micro(&self, f: SchedulerAction); fn post_timer(&self, ms: u64, f: SchedulerAction); }
struct Runtime { /* nodes: Slab<NodeData>, batch: BatchState, owner_thread: ThreadId */ }  // !Send/!Sync
type NodeId = u32;                                  // dense index into slab
struct NodeData { rank: Rank, state: NodeState, deps: Vec<NodeId>, observers: Vec<NodeId>, gen: u64 }
type Rank = u32;                                    // Height; Signals=0
enum NodeState { Clean, Check, Dirty, Computing, Errored }  // (§8.2)

struct Signal<T> { /* id: NodeId + value cell */ }
impl<T: Clone> Signal<T> {
    fn get(&self) -> T;            // tracks: registers a dep edge if inside Computed/Effect
    fn set(&self, v: T);           // marks observers dirty; flushes within current/next batch
    fn update(&self, f: impl FnOnce(&T) -> T);
}
struct Computed<T> { /* id: NodeId + memo cell + eq: fn(&T,&T)->bool */ }
impl<T: Clone> Computed<T> {
    fn new(f: impl Fn() -> T + 'static) -> Self;       // auto-tracks deps on each run
    fn with_eq(self, eq: impl Fn(&T,&T) -> bool) -> Self;  // equality cutoff (I11)
    fn get(&self) -> T;                                 // pulls: recompute if Dirty/Check
}
struct Effect { /* id: NodeId + run: Box<dyn FnMut()>, cleanup: Option<Teardown> */ }
impl Effect {
    fn new(f: impl FnMut() + 'static) -> Self;          // runs once, re-runs when deps change
    fn dispose(&self);
}
fn batch<R>(f: impl FnOnce() -> R) -> R;                // glitch barrier; defers flush (I3,I9)
fn untracked<R>(f: impl FnOnce() -> R) -> R;            // read without registering a dep
trait Publisher<T> { fn subscribe(&self, s: Box<dyn Subscriber<T>>); }
trait Subscriber<T> {
    fn on_subscribe(&mut self, sub: RsSubscription);   // MUST call request(n) to start
    fn on_next(&mut self, v: T);                        // only up to demand
    fn on_error(&mut self, e: ReactiveError);
    fn on_complete(&mut self);
}
struct RsSubscription { /* demand: i64 (saturating), cancelled: bool */ }
impl RsSubscription { fn request(&self, n: u64); fn cancel(&self); }   // n>0; no unbounded recursion (TCK 3.3)
trait Processor<T, R>: Subscriber<T> + Publisher<R> {}
enum Overflow { Drop, Latest, Buffer(usize), Error }    // policy when demand exhausted (I8)
struct BoundedBuffer<T> { /* ring: VecDeque<T>, cap: usize, high_water: usize */ }
