Core schema & types (§11)

pub struct Cursor<'a> { /* &'a [u8] + pos */ }                 // all reads checked (I1/I6)
impl<'a> Cursor<'a> {
    pub fn u8(&mut self) -> Result<u8, DecodeError>;
    pub fn i8(&mut self) -> Result<i8, DecodeError>;
    pub fn i32(&mut self) -> Result<i32, DecodeError>;         // sign-extended by caller (I5)
    pub fn peek(&self, n: usize) -> Result<&'a [u8], DecodeError>;
    pub fn pos(&self) -> usize;
}

pub enum DecodeError { Truncated, UnknownOpcode(u8), Reserved, BadModRM }   // → (bad), resync (I2)

pub enum Reg { Gpr{size:u8, idx:u8}, Xmm(u8), Ymm(u8), Seg(u8), Rip, Rflags }
pub enum Operand { Reg(Reg), Imm(i64), Rel(i64),
    Mem{ base:Option<Reg>, index:Option<Reg>, scale:u8, disp:i64, seg:Option<Reg>, rip_rel:bool } }
pub enum Flow { Seq, Jump(u64), CondJump{taken:u64, fall:u64, cc:Cc}, Call(u64),
                Ret, IndirectJump(Operand), IndirectCall(Operand), Int, Unknown }

pub struct Instruction<'a> {                                   // immutable, Send+Sync (I11)
    pub addr: u64, pub len: u8, pub mnemonic: Mnemonic,
    pub operands: SmallVec<[Operand;4]>, pub prefixes: Prefixes, pub flow: Flow, pub bytes: &'a [u8] }

pub trait Decoder {                                            // ISA seam (x64 / arm64)
    fn decode(&self, cur: &mut Cursor, addr: u64) -> Result<Instruction, DecodeError>;
    fn length(&self, bytes: &[u8]) -> Result<u8, DecodeError>; // fast path (I2/principle 2)
}

pub struct BasicBlock { pub start:u64, pub end:u64, pub term:Flow }
pub struct Cfg { pub blocks: Vec<BasicBlock>, pub edges: Vec<(u32,u32,Edge)>, pub functions: Vec<Function> }
pub fn sweep(bytes:&[u8], base:u64, d:&dyn Decoder) -> Vec<Instruction>;        // linear (principle 3)
pub fn recursive_descent(img:&BinaryImage, d:&dyn Decoder) -> (Vec<Instruction>, Cfg);  // from entry+symbols
