Core schema & types (§11)

pub enum EventKind {
    Syscall,
    Signal { synchronous: bool },
    SchedSwitch { from: TaskId, to: TaskId },
    RdtscTrap,
    CpuidTrap,
    PatchVsyscall,
    Exec,
    Exit { code: i32 },
    Checkpoint,
}

pub struct TraceFrame {
    pub event_id: EventId,
    pub tid:      TaskId,
    pub ticks:    Ticks,
    pub kind:     EventKind,
    pub regs:     RecordedRegs,
    pub data_ref: DataRef,
}
pub struct RecordedRegs {
    pub rax: u64, pub rbx: u64, pub rcx: u64, pub rdx: u64,
    pub rsi: u64, pub rdi: u64, pub rsp: u64, pub rbp: u64,
    pub r8:  u64, pub r9:  u64, pub r10: u64, pub r11: u64,
    pub r12: u64, pub r13: u64, pub r14: u64, pub r15: u64,
    pub rip: u64, pub rflags: u64,
    pub orig_rax: u64,
    pub fs_base: u64, pub gs_base: u64,
    pub cs: u16, pub ss: u16, pub ds: u16, pub es: u16, pub fs: u16, pub gs: u16,
}
pub struct MemRange { pub addr: u64, pub len: usize }

pub struct MemDelta { pub range: MemRange, pub bytes_ref: DataRef }

pub struct MappedSegment {
    pub range:   MemRange,
    pub perms:   MapPerms,    // r/w/x/p/s flags
    pub offset:  u64,
    pub provenance: MmapProvenance,
}

pub enum MmapProvenance { Anonymous, File { path: PathBuf, inode: u64 }, Vdso, Vsyscall }

pub struct MapPerms { pub read: bool, pub write: bool, pub exec: bool, pub shared: bool }

pub struct AddressSpace {
    pub segments: Vec<MappedSegment>,  // sorted by addr; non-overlapping
    pub brk: u64,
}
pub struct SyscallRecord {
    pub nr:          u32,
    pub args:        [u64; 6],
    pub ret:         i64,
    pub errno:       i32,
    pub mem_effects: Vec<MemDelta>,
}

pub struct SignalRecord {
    pub signo:            i32,
    pub si_code:          i32,
    pub si_info_bytes:    [u8; 128],   // raw siginfo_t bytes; platform-specific fields preserved
    pub synchronous:      bool,
    pub delivery_ticks:   Ticks,
}
pub struct Checkpoint {
    pub event_id:     EventId,
    pub ticks:        Ticks,
    pub regs_per_task: HashMap<TaskId, RecordedRegs>,
    pub dirty_pages:  Vec<(MemRange, PageRef)>,
    pub fd_table:     Vec<FdRecord>,
    pub brk:          u64,
    pub mmaps:        AddressSpace,
}

pub struct FdRecord {
    pub fd:     i32,
    pub flags:  i32,
    pub kind:   FdKind,
}

pub enum FdKind { File { path: PathBuf, offset: u64 }, Pipe { id: u64 }, Socket { kind: SocketKind }, Epoll, Other }
pub struct SchedPoint { pub tid: TaskId, pub ticks: Ticks }

pub struct SchedSwitch {
    pub event_id:    EventId,
    pub from:        TaskId,
    pub to:          TaskId,
    pub at_ticks:    Ticks,
}
pub struct DivergenceReport {
    pub event_id:  EventId,
    pub tid:       TaskId,
    pub kind:      DivergenceKind,
    pub expected:  DivergenceValue,
    pub actual:    DivergenceValue,
}

pub enum DivergenceKind { RegMismatch, TickMismatch, MemHashMismatch }

pub enum DivergenceValue {
    Regs(RecordedRegs),
    Ticks(Ticks),
    MemHash { range: MemRange, hash: u64 },
}
pub struct TraceWriter { /* append-only; owns file handles for events.log, data.log, mmap.log */ }
impl TraceWriter {
    pub fn new(trace_dir: &Path) -> Result<Self, TraceError>;
    pub fn write_frame(&mut self, frame: &TraceFrame, payload: Option<&[u8]>) -> Result<DataRef, TraceError>;
    pub fn write_checkpoint(&mut self, ckpt: &Checkpoint) -> Result<(), TraceError>;
    pub fn write_mmap_event(&mut self, event: &MmapEvent) -> Result<(), TraceError>;
    pub fn flush(&mut self) -> Result<(), TraceError>;
    pub fn finish(self) -> Result<TraceHeader, TraceError>;
}

pub struct TraceReader { /* seekable; owns read-only file handles */ }
impl TraceReader {
    pub fn open(trace_dir: &Path) -> Result<Self, TraceError>;
    pub fn next_frame(&mut self) -> Result<Option<TraceFrame>, TraceError>;
    pub fn seek_to_event(&mut self, id: EventId) -> Result<(), TraceError>;
    pub fn seek_to_ticks(&mut self, tid: TaskId, ticks: Ticks) -> Result<Option<EventId>, TraceError>;
    pub fn read_payload(&self, data_ref: DataRef, len: usize) -> Result<Vec<u8>, TraceError>;
    pub fn nearest_checkpoint_before(&self, id: EventId) -> Result<Option<Checkpoint>, TraceError>;
}
pub struct RecordSession {
    pub trace_dir: PathBuf,
    // owns: TraceWriter, ptrace supervisor state, per-task tick counters
}
impl RecordSession {
    pub fn launch(argv: &[&str], trace_dir: &Path, opts: RecordOpts) -> Result<Self, RecordError>;
    pub fn run_to_exit(self) -> Result<RecordSummary, RecordError>;
}

pub struct ReplaySession {
    pub trace_dir: PathBuf,
    // owns: TraceReader, re-exec'd tracee, DivergenceChecker
}
impl ReplaySession {
    pub fn open(trace_dir: &Path, opts: ReplayOpts) -> Result<Self, ReplayError>;
    pub fn run_to_exit(self) -> Result<ReplaySummary, ReplayError>;
    pub fn run_to_event(self, target: EventId) -> Result<ReplayState, ReplayError>;
    pub fn attach_gdb(&mut self, port: u16) -> Result<GdbSession, ReplayError>;
}
