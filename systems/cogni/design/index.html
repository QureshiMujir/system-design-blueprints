Core schema & types (§11)

pub struct NodeIdentity {
    pub key: PubKey,                 // Ed25519 (Phase 3), BLS (Phase 10)
    pub address: Address,
    pub stake: TokenAmount,
    pub joined_at: Timestamp,
    pub jurisdiction_attestation: Option<Bytes>,
    pub hardware_class: HardwareClass,   // Consumer | Datacenter | TeeAttested
}

pub struct ModelCommitment {
    pub commitment: Hash,             // merkle root over weight tensors (I1, I14)
    pub architecture_id: ArchitectureId,
    pub param_count: u64,
    pub quantisation: Quantisation,   // FP32|FP16|BF16|INT8|INT4 (I8)
    pub author: Address,
    pub license: LicenseRef,          // SPDX id, OSI-approved only (P2)
    pub registered_at: Timestamp,
    pub author_sig: Signature,
}

pub struct Receipt {
    pub job_id: JobId,
    pub operator: NodeIdentity,
    pub output: Bytes,
    pub proof: Proof,
    pub started_at: Timestamp,
    pub completed_at: Timestamp,
    pub fee_paid: TokenAmount,
    pub block_reward: TokenAmount,
    pub operator_sig: Signature,       // I13
}

pub enum ExecError { ModelNotLoaded, Timeout, ProverPanic, QuantisationDrift }
pub struct HealthStatus { pub healthy: bool, pub queue_depth: u32, pub gpu_util_pct: u8 }
pub struct Quote { pub price: TokenAmount, pub eta: Timestamp }
pub struct EncryptionSpec { pub scheme: EncryptionScheme, pub pubkey: Bytes }
pub enum SlaTier { BestEffort, Standard, Premium }
pub enum HardwareClass { Consumer, Datacenter, TeeAttested }
pub struct ArchitectureId(pub u32);
pub struct CircuitId(pub u32);
pub struct SchemaId(pub u32);
pub trait NodeOperator: Send + Sync {
    fn identity(&self) -> NodeIdentity;
    fn supported_models(&self) -> Vec<ModelCommitment>;
    fn quote(&self, job: &InferenceJob) -> Option<Quote>;
    fn execute(&self, job: &InferenceJob) -> Result<Receipt, ExecError>;
    fn health(&self) -> HealthStatus;   // MUST return within 100 ms (§4)
}

pub trait ProofSystem: Send + Sync {   // P10 swappability boundary
    fn prove(&self, circuit_id: CircuitId, witness: &Bytes) -> Proof;
    fn verify(&self, proof: &Proof, public_inputs: &Bytes) -> bool;
    fn proof_size_ceiling(&self) -> u32;
}
interface ICogniVerifier {
    // Deterministic (I3); MUST NOT revert on malformed-but-valid input,
    // returns false instead; gas independent of model size (I4/P11).
    function verify(bytes32 commitment, bytes calldata input,
        bytes calldata output, bytes calldata proof) external view returns (bool);
    function proofSystem() external pure returns (bytes32);
    function limits() external pure returns
        (uint32 maxInputBytes, uint32 maxOutputBytes, uint32 maxProofBytes);
}
