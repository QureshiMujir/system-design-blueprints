Core schema & types (§11)

struct ConfidentTuple<T> {
    v: T,                    // owned; mutation: replace-on-write (I3 additive-only)
    c: f64,                  // [0.0, 1.0]; concurrency: MVCC row-level; persistence: heap
    s: SourceId,             // uuid; FK fuzzy_source; serialization: fixed 16B
    t: TimestampTz,          // fixed 8B; default now()
    decay_ref: Option<DecayId>,   // uuid nullable; FK fuzzy_decay_def
    conflict_ix: Option<i64>,     // nullable; FK fuzzy_conflict_log (Phase 7+)
}
table fuzzy_source {
    source_id:     uuid PRIMARY KEY,      -- generated
    handle:        text UNIQUE NOT NULL,  -- e.g. "model:gpt-4o-2025-09"
    kind:          source_kind,           -- enum {model, agent, sensor, human, system}
    metadata:      jsonb,
    reliability_a: float8 DEFAULT 1.0,    -- Beta(α) prior, §8.4.3
    reliability_b: float8 DEFAULT 1.0,    -- Beta(β) prior, §8.4.3
    created_at:    timestamptz,
    last_seen_at:  timestamptz,
}
-- mutation: reliability_a/b only via feedback update rule (I10); ownership: registry-wide
table fuzzy_column_config {
    rel_id:               oid,
    attnum:                int,
    decay_ref:              Option<DecayId>,
    default_propagation:    PropagationMode,   -- enum {min, product, bayesian}, default min
    threshold:              float8,             -- IS DECAYED cutoff
    track_conflicts:        bool,               -- Phase 7 gate
    PRIMARY KEY (rel_id, attnum),
}
table fuzzy_decay_def {
    decay_id:   uuid PRIMARY KEY,
    kind:       decay_kind,   -- enum {half_life, linear, step, none}
    params:     jsonb,        -- e.g. {"days": 14}
    created_at: timestamptz,
}
table fuzzy_conflict_log {          -- append-only (P10, I6)
    conflict_ix:    bigserial PRIMARY KEY,
    rel_id:          oid,
    attnum:           int,
    row_id:           bytea,         -- serialized PK
    observation:      jsonb,         -- {v, c, s, t, decay_ref}
    written_at:       timestamptz,
    superseded_by:    Option<i64>,   -- FK self
}
-- index: (rel_id, attnum, row_id, written_at DESC)

table fuzzy_feedback {
    feedback_id:        uuid PRIMARY KEY,
    rel_id:              oid,
    attnum:               int,
    row_id:               bytea,
    corrected_to:         jsonb,
    correcting_source:    uuid,     -- FK fuzzy_source
    rationale:            Option<text>,
    applied_at:           timestamptz,
}
enum FuzzyErrorCode {
    InvalidConfidence,        -- ERRCODE_FUZZY_INVALID_CONFIDENCE (I1)
    UnregisteredSource,       -- warning-level, I2
    AmbiguousPropagation,     -- ERRCODE_FUZZY_AMBIGUOUS_PROPAGATION (I4)
    MalformedLiteral,         -- type input parse failure (T4)
}

struct DecaySweepHandle {         -- bgworker-local; not persisted
    database_oid: Oid,
    cursor:        RowId,           -- resumable checkpoint position
    cancelled:     AtomicBool,       -- checked every ≤10,000 rows (I9)
}

view fuzzy_stat_columns    { rel_id, attnum, rejected_writes, unregistered_sources, conflict_writes }
view fuzzy_stat_propagation { rel_id, join_id, mode, ambiguous_count }
view fuzzy_stat_decay_runs  { database_oid, started_at, rows_visited, out_of_range_clamped, cancelled_gracefully }
