Core schema & types (§11)

pub enum Type {
    Tyvar(Name),              // type variable, e.g. 'a
    Tyapp(Name, Vec<Type>),   // type constructor applied to args
}
pub enum Term {
    Var(Name, Type),          // FREE variable, carries its type
    Bound(u32),               // de Bruijn index: 0 = innermost Abs
    Const(Name, Type),        // declared/defined constant, instantiated monotype
    Comb(Box<Term>, Box<Term>), // application (f x)
    Abs(Type, Box<Term>),     // lambda: binder TYPE + body; NO binder name stored
}
pub struct Sequent { pub hyps: Vec<Term>, pub concl: Term }

/// The field is PRIVATE to tp-kernel. No public constructor, no From, no Deserialize.
/// Value of type Thm EXISTS ==> its Sequent is derivable in the logic. (LCF discipline)
pub struct Thm(Sequent);

impl Thm {
    pub fn sequent(&self) -> &Sequent;
    pub fn concl(&self) -> &Term;
    pub fn hyps(&self) -> &[Term];
}
pub fn refl(t: &Term) -> Thm;
pub fn trans(a: &Thm, b: &Thm) -> Result<Thm, KernelError>;
pub fn mk_comb(a: &Thm, b: &Thm) -> Result<Thm, KernelError>;
pub fn abs(v: &Term, a: &Thm) -> Result<Thm, KernelError>;
pub fn beta(t: &Term) -> Result<Thm, KernelError>;
pub fn assume(p: &Term) -> Result<Thm, KernelError>;
pub fn eq_mp(a: &Thm, b: &Thm) -> Result<Thm, KernelError>;
pub fn deduct_antisym(a: &Thm, b: &Thm) -> Thm;
pub fn inst(theta: &[(Term,Term)], a: &Thm) -> Thm;
pub fn inst_type(theta: &[(Type,Type)], a: &Thm) -> Thm;
pub fn new_basic_definition(eq: &Term) -> Result<Thm, KernelError>;
pub fn new_basic_type_definition(..) -> Result<(Thm,Thm), KernelError>;
pub enum KernelError { TypeMismatch, NotEquation, FreeVarEscape, IllTyped, BadDefinition }
pub fn type_of(t: &Term) -> Result<Type, KernelError>;
pub fn subst(theta: &[(Term,Term)], t: &Term) -> Term; // capture-avoiding (de Bruijn shift)
pub fn free_vars(t: &Term) -> Vec<(Name, Type)>;
pub fn alpha_eq(a: &Term, b: &Term) -> bool;           // == under de Bruijn (syntactic)
pub fn beta_eta_normal(t: &Term) -> Term;              // normalization (NOT trusted)
pub fn term_match(pat: &Term, t: &Term) -> Option<Subst>; // one-sided matching
pub struct Goal { pub hyps: Vec<(Name, Term)>, pub concl: Term }
pub struct GoalState { pub goals: Vec<Goal>, pub justify: Justification }
pub type Justification = Box<dyn Fn(Vec<Thm>) -> Thm>;
pub type Tactic = Box<dyn Fn(&Goal) -> Result<(Vec<Goal>, Justification), TacticError>>;
pub fn then(t1: Tactic, t2: Tactic) -> Tactic;
pub fn orelse(t1: Tactic, t2: Tactic) -> Tactic;
pub fn repeat(t: Tactic) -> Tactic;
pub fn try_tac(t: Tactic) -> Tactic;
pub fn intro() -> Tactic;
pub fn assumption() -> Tactic;
pub fn exact(th: Thm) -> Tactic;
pub fn apply(th: Thm) -> Tactic;
pub fn rewrite(eqs: &[Thm]) -> Tactic;
pub fn induct(v: &Term) -> Tactic;
pub fn simp() -> Tactic;
pub fn prove(goal: Term, tac: Tactic) -> Result<Thm, TacticError>;
pub enum TacticError { NoApplicableRule, OpenGoalsRemain, UnifyFailed, IllFormedGoal, ConclMismatch }
pub enum UnifyResult { Unifiable(Subst), NotUnifiable }
pub fn unify_fo(a: &Term, b: &Term) -> UnifyResult;
pub fn unify_pattern(a: &Term, b: &Term) -> UnifyResult;
pub struct ResolutionProver { /* clauses, ordering, age/weight queue */ }
pub fn prove_fol(goal: &Term) -> Option<Proof>;
pub struct CongruenceClosure { /* union-find, pending merges, use-lists */ }
pub struct LinearArith { /* Fourier-Motzkin / Simplex certificate generator */ }
pub enum Proof { Refl(Term), Trans(Box<Proof>,Box<Proof>), MkComb(Box<Proof>,Box<Proof>),
                 Abs(Term,Box<Proof>), Beta(Term), Assume(Term), EqMp(Box<Proof>,Box<Proof>),
                 DeductAntisym(Box<Proof>,Box<Proof>), Inst(/*..*/Box<Proof>),
                 InstType(/*..*/Box<Proof>), Definition(Name) }
pub fn run_script(path: &str) -> Result<(), CliError>;
pub fn recheck_proof(proof: &Proof) -> Result<Thm, KernelError>;
pub enum CliError { Parse(ParseError), Tactic(TacticError), Kernel(KernelError), Io }
