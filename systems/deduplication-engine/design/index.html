Core schema & types (§11)

//! dedup-core/src/fingerprint.rs — §11.1
#[derive(Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Hash)]
pub struct Fingerprint(pub [u8; 32]);          // I1: address = SHA-256(bytes)

impl Fingerprint {
    pub const ZERO: Self;
    pub fn of(bytes: &[u8]) -> Self;            // owning: copies into a new value
    pub fn hex(&self) -> String;
    pub fn prefix64(&self) -> u64;               // shard-routing key, I14
}
//! dedup-core/src/chunk.rs — §11.2
pub const CHUNK_MIN: u32 = 16 * 1024;           // I11
pub const CHUNK_AVG: u32 = 64 * 1024;
pub const CHUNK_MAX: u32 = 256 * 1024;

#[derive(Clone)]
pub struct Chunk { pub fp: Fingerprint, pub len: u32, pub bytes: bytes::Bytes }
//! dedup-core/src/chunk_ref.rs — §11.3
#[derive(Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Hash)]
pub struct ContainerId(pub u64);

#[derive(Copy, Clone)]
pub struct ChunkRef { pub fp: Fingerprint, pub container: ContainerId,
                       pub offset: u32, pub stored_len: u32 }
//! dedup-core/src/index_entry.rs — §11.4
bitflags::bitflags! {
    pub struct EntryFlags: u16 {
        const COMPRESSED = 0b0000_0001;
        const ENCRYPTED  = 0b0000_0010;
        const VERIFIED   = 0b0000_0100;
        const TOMBSTONE  = 0b0000_1000;
    }
}
#[derive(Copy, Clone)]
pub struct IndexEntry { pub fp: Fingerprint, pub container: ContainerId,
                         pub offset: u32, pub stored_len: u32,
                         pub original_len: u32, pub flags: EntryFlags }
//! dedup-core/src/manifest.rs — §11.5
#[derive(Copy, Clone, Eq, PartialEq, Hash)]
pub struct ManifestId(pub Fingerprint);          // a manifest is a chunk (§8.1)

pub struct Manifest { pub id: ManifestId, pub stream: String,
                       pub chunks: Vec<Fingerprint>, pub total_len: u64,
                       pub created_at: chrono::DateTime<chrono::Utc>,
                       pub parent: Option<ManifestId> }
//! dedup-chunker/src/lib.rs — §11.6 (Chunker trait, shared by FastCDC/Rabin)
pub struct ChunkSlice<'a> { pub offset: u64, pub bytes: &'a [u8] }

pub trait Chunker: Send {
    fn feed<'a>(&mut self, buf: &'a [u8]) -> Box<dyn Iterator<Item = ChunkSlice<'a>> + 'a>;
    fn finish(&mut self) -> Option<bytes::Bytes>;   // trailing partial chunk, §8.2
}
//! dedup-index/src/service.rs, dedup-store/src/lib.rs, dedup-manifest/src/lib.rs — §11.7
#[async_trait::async_trait]
pub trait IndexService: Send + Sync {
    async fn lookup(&self, fp: Fingerprint) -> Result<Option<IndexEntry>>;
    async fn insert(&self, e: IndexEntry) -> Result<()>;
    async fn remove(&self, fp: Fingerprint) -> Result<()>;
    async fn scan(&self, from: Fingerprint, n: u32) -> Result<Vec<IndexEntry>>;
    async fn flush(&self) -> Result<()>;
}
#[async_trait::async_trait]
pub trait StoreService: Send + Sync {
    async fn write(&self, fp: Fingerprint, bytes: bytes::Bytes)
        -> Result<(ContainerId, u32, u32)>;
    async fn read(&self, r: ChunkRef) -> Result<bytes::Bytes>;
    async fn seal_active(&self) -> Result<()>;
    async fn list_containers(&self) -> Result<Vec<ContainerId>>;
    async fn delete_container(&self, c: ContainerId) -> Result<()>;
}
#[async_trait::async_trait]
pub trait ManifestService: Send + Sync {
    async fn put(&self, m: Manifest) -> Result<ManifestId>;
    async fn get(&self, id: ManifestId) -> Result<Manifest>;
    async fn list(&self) -> Result<Vec<ManifestId>>;
    async fn delete(&self, id: ManifestId) -> Result<()>;
}
//! dedup-compress/src/lib.rs, dedup-encrypt/src/lib.rs — §11.8
pub trait Compressor: Send + Sync {
    fn compress(&self, input: &[u8]) -> Result<bytes::Bytes>;
    fn decompress(&self, input: &[u8]) -> Result<bytes::Bytes>;
}
pub struct ZstdCompressor { pub level: i32 }      // ADR-12, default level 3
pub struct Lz4Compressor;
pub struct NoopCompressor;

pub trait AeadCipher: Send + Sync {
    fn seal(&self, key: &[u8; 32], nonce12: &[u8; 12], aad: &[u8], pt: &[u8])
        -> Result<bytes::Bytes>;
    fn open(&self, key: &[u8; 32], nonce12: &[u8; 12], aad: &[u8], ct: &[u8])
        -> Result<bytes::Bytes>;
}
pub struct ChaCha20Poly1305Cipher;                // ADR-14, default
pub struct Aes256GcmCipher;                       // FIPS-mode alternative
//! dedup-gc/src/{mark,sweep,pin}.rs — §11.9
pub struct GcEpoch(pub u64);
pub struct WriterPin { pub manifest: ManifestId, pub epoch: GcEpoch }

pub trait GarbageCollector: Send + Sync {
    async fn mark(&self, live_manifests: &[ManifestId]) -> Result<GcEpoch>;
    async fn sweep(&self, epoch: GcEpoch) -> Result<u64>;   // returns chunks tombstoned
    fn register_pin(&self, p: WriterPin);
    fn release_pin(&self, m: ManifestId);
}
