Core schema & types (§11)

// cos::object
struct ObjId { num: u32, gen: u16 }            // indirect object identity

enum Object {
    Null,
    Bool(bool),
    Int(i64),
    Real(f64),
    Str(PdfString),
    Name(Name),
    Array(Vec<Object>),
    Dict(Dict),
    Stream(Stream),
    Ref(ObjId),                                  // unresolved indirect reference
}

enum PdfString { Literal(Vec<u8>), Hex(Vec<u8>) } // raw bytes; encoding context-dependent
struct Name(Vec<u8>);                             // decoded (#xx resolved), still bytes
struct Dict(IndexMap<Name, Object>);              // insertion-ordered; typed getters
struct Stream { dict: Dict, raw: Range<usize>, decoded: OnceCell<Vec<u8>> } // lazy decode
// xref
enum XrefEntry {
    Free,
    InUse   { offset: u64, gen: u16 },
    InStream{ objstm: u32, index: u32 },         // compressed in an ObjStm
}
struct XrefTable { entries: HashMap<u32, XrefEntry>, size: u32 } // num → entry
struct Trailer { root: ObjId, size: u32, prev: Option<u64>,
                 info: Option<ObjId>, id: Option<[Vec<u8>;2]>, encrypt: Option<Object> }

// doc
struct Document<'a> {
    src: &'a [u8],                                // mmap'd source
    xref: XrefTable,
    trailer: Trailer,
    cache: RwLock<HashMap<ObjId, Object>>,        // resolved-object cache
    security: Option<SecurityHandler>,
    quirks: QuirkFlags,                           // recorded leniencies applied
}
impl<'a> Document<'a> {
    fn open(src: &'a [u8]) -> Result<Self, PdfError>;
    fn resolve(&self, id: ObjId) -> Result<Object, PdfError>;
    fn get<T: FromObject>(&self, o: &Object) -> Result<T, PdfError>;
    fn page_count(&self) -> usize;
    fn page(&self, idx: usize) -> Result<Page, PdfError>;
}

struct Catalog { pages: ObjId, names: Option<ObjId>, acroform: Option<ObjId>,
                 metadata: Option<ObjId>, version: Option<Name> }
struct PageTree { root: ObjId, count: u32 }       // flattened lazily
struct Page {
    media_box: Rect, crop_box: Option<Rect>, rotate: i32,  // 0/90/180/270
    resources: Dict, contents: Vec<ObjId>, annots: Vec<ObjId>, group: Option<Dict>,
}
struct Resources { font: Dict, xobject: Dict, colorspace: Dict,
                   extgstate: Dict, pattern: Dict, shading: Dict, procset: Vec<Name> }
// graphics
struct Matrix { a: f64, b: f64, c: f64, d: f64, e: f64, f: f64 } // [a b 0; c d 0; e f 1]
impl Matrix { fn concat(&self, o:&Matrix)->Matrix; fn invert(&self)->Option<Matrix>;
              fn apply(&self, p: Point)->Point; fn det(&self)->f64; }
struct Point { x: f64, y: f64 }
struct Rect { x0:f64, y0:f64, x1:f64, y1:f64 }

struct GraphicsState {
    ctm: Matrix, clip: ClipPath,
    fill_cs: ColorSpace, stroke_cs: ColorSpace,
    fill_color: Color, stroke_color: Color,
    line_width: f64, line_cap: u8, line_join: u8, miter_limit: f64,
    dash: (Vec<f64>, f64), alpha_fill: f64, alpha_stroke: f64,
    blend: BlendMode, text: TextState,
}
struct ClipPath { mask: Option<CoverageMask>, depth: usize } // None = unclipped

// content::textstate
struct TextState {
    font: Option<FontRef>, size: f64,
    char_spacing: f64, word_spacing: f64, h_scale: f64, leading: f64, rise: f64,
    render_mode: u8, tm: Matrix, tlm: Matrix,
}
// graphics::colorspace
enum ColorSpace {
    DeviceGray, DeviceRGB, DeviceCMYK,
    CalGray(CalParams), CalRGB(CalParams), Lab(LabParams),
    ICCBased { n: u8, alternate: Box<ColorSpace>, profile: Option<IccProfile> },
    Indexed { base: Box<ColorSpace>, hival: u16, lookup: Vec<u8> },
    Separation { name: Name, alt: Box<ColorSpace>, tint: PdfFunction },
    DeviceN { names: Vec<Name>, alt: Box<ColorSpace>, tint: PdfFunction },
    Pattern(Option<Box<ColorSpace>>),
}
struct Color(SmallVec<[f32; 4]>);                  // component values
enum PdfFunction { Sampled(Box<SampledFn>), Exponential(ExpFn),
                   Stitching(StitchFn), PostScript(PsCalcFn) }
impl ColorSpace { fn to_rgba(&self, c:&Color)->[u8;4]; fn n_components(&self)->u8; }
// graphics::path
struct Path { subpaths: Vec<Subpath> }
struct Subpath { points: Vec<PathSeg>, closed: bool }
enum PathSeg { Move(Point), Line(Point), Cubic(Point,Point,Point) }

// raster
struct CoverageMask { w:u32, h:u32, data: Vec<u8> }     // 8-bit AA coverage
trait Rasterizer {
    fn fill(&mut self, path:&Path, rule:FillRule, ctm:&Matrix, clip:&ClipPath)->CoverageMask;
    fn stroke(&mut self, path:&Path, style:&StrokeStyle, ctm:&Matrix, clip:&ClipPath)->CoverageMask;
}
enum FillRule { NonZero, EvenOdd }

// filters
trait Filter { fn decode(&self, input:&[u8], parms:&DecodeParms) -> Result<Vec<u8>, PdfError>; }
struct DecodeParms { predictor:u8, colors:u8, bpc:u8, columns:u32, early_change:bool, /* … */ }
fn run_filter_chain(raw:&[u8], filters:&[Name], parms:&[DecodeParms], cap:RatioCap)
    -> Result<Vec<u8>, PdfError>;
// text::font
enum Font { Simple(SimpleFont), Type0(Type0Font), Type3(Type3Font) }
struct SimpleFont { program: FontProgram, encoding: Encoding,
                    widths: WidthTable, descriptor: FontDescriptor }
struct Type0Font { cmap: Cmap, descendant: CidFont, to_unicode: Option<ToUnicode> }
struct CidFont { program: FontProgram, cid_to_gid: CidToGid, widths: CidWidths, dw: f64 }
struct Type3Font { font_matrix: Matrix, char_procs: Dict,
                   encoding: Encoding, resources: Dict }

enum FontProgram { TrueType(TtHandle /*#385*/), Cff(CffFont), Type1(Type1Font), None }
enum Encoding { Standard, WinAnsi, MacRoman, Builtin, Differences(Box<Encoding>, Vec<(u8,Name)>) }
struct Cmap { ranges: Vec<CmapRange> }              // code bytes → CID
enum CidToGid { Identity, Map(Vec<u16>) }
struct WidthTable { first:u32, widths:Vec<f64>, missing:f64 }
trait GlyphSource { fn outline(&self, gid:u16) -> Option<Path>; fn advance(&self, gid:u16)->f64; }
// image
struct Image { width:u32, height:u32, colorspace:ColorSpace, bpc:u8,
               data:Vec<u8>, mask:ImageMaskKind, smask:Option<Box<Image>> }
enum ImageMaskKind { None, Stencil(Vec<u8>), ColorKey(Vec<(u16,u16)>) }

// security
struct SecurityHandler { revision:u8, key_len:usize, key:Vec<u8>,
                         v:u8, cf:CryptFilters, perms:u32, encrypt_metadata:bool }
impl SecurityHandler {
    fn from_encrypt(dict:&Dict, id0:&[u8], user_pw:&[u8]) -> Result<Self, PdfError>;
    fn decrypt_stream(&self, id:ObjId, data:&[u8]) -> Result<Vec<u8>, PdfError>;
    fn decrypt_string(&self, id:ObjId, data:&[u8]) -> Result<Vec<u8>, PdfError>;
}
enum PdfError {
    Lex(LexSpan),                 // malformed token
    BadXref(String),              // unparseable / inconsistent xref
    MissingObject(ObjId),         // referenced object absent
    CycleDetected(ObjId),         // reference/page-tree cycle
    RecursionLimit(&'static str), // cap kind (Type3/form/pattern/nest)
    DecompressBomb { ratio:u32 }, // ratio cap exceeded
    BadStreamLength{ id:ObjId },  // /Length reconciliation failed badly
    Encrypted,                    // needs key before decode
    BadPassword,                  // standard handler auth failure
    UnsupportedFilter(Name),      // e.g. JBIG2Decode/JPXDecode (deferred)
    UnsupportedFeature(&'static str),
    Io(std::io::ErrorKind),
}
