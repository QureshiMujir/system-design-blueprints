Core schema & types (§11)

pub trait PrimeField: Copy + Clone + Eq + ConstantTimeEq {
    const MODULUS_LIMBS: [u64; 4];
    fn from_repr(repr: [u64; 4]) -> CtOption<Self>;   // I4: canonical-form check
    fn to_repr(&self) -> [u64; 4];
    fn add(&self, rhs: &Self) -> Self;                // constant-time (P6)
    fn mul(&self, rhs: &Self) -> Self;                 // Montgomery multiplication (ADR-02)
    fn invert(&self) -> CtOption<Self>;
    fn pow_vartime(&self, exp: &[u64]) -> Self;        // exponent must be public (P6)
}
pub struct Fr([u64; 4]);   // BLS12-381 scalar field, Montgomery form — owned, Copy, no interior mutability
pub struct Fq([u64; 6]);   // BLS12-381 base field
pub struct G1Affine { x: Fq, y: Fq, infinity: Choice }
pub struct G1Projective { x: Fq, y: Fq, z: Fq }
pub enum PointError { NotOnCurve, NotInSubgroup, ZeroInfinity, InvalidLength(usize) }
impl G1Affine {
    pub fn is_on_curve(&self) -> Choice;
    pub fn is_in_correct_subgroup(&self) -> Choice;          // I5
    pub fn clear_cofactor(p: G1Projective) -> G1Projective;
    pub fn to_compressed(&self) -> [u8; 48];                  // ADR-11 canonical encoding
    pub fn from_compressed(bytes: &[u8; 48]) -> Result<Self, PointError>;
}
pub struct MillerLoopResult(Fq12);
pub struct Gt(Fq12);
pub fn multi_miller_loop(pairs: &[(G1Affine, G2Affine)]) -> MillerLoopResult;  // I12: batched
pub fn final_exponentiation(f: MillerLoopResult) -> Gt;
pub fn pairing(p: &G1Affine, q: &G2Affine) -> Gt;
pub struct Polynomial<F: PrimeField> { coeffs: Vec<F> }
pub struct EvaluationDomain<F: PrimeField> { size: usize, generator: F, size_inv: F }
impl<F: PrimeField> EvaluationDomain<F> {
    pub fn fft(&self, coeffs: &mut Vec<F>);
    pub fn ifft(&self, evals: &mut Vec<F>);
    pub fn vanishing_polynomial(&self) -> Polynomial<F>;      // Z(x), §8
}
pub struct WireId(u32);
pub enum Gate { Add(WireId, WireId), Mul(WireId, WireId), Const(Fr), PublicInput(usize), Witness(usize) }
pub struct Circuit { gates: Vec<Gate>, num_public: usize, num_witness: usize }
pub trait ConstraintSynthesizer<F: PrimeField> {
    fn synthesize(&self, cs: &mut ConstraintSystem<F>) -> Result<(), SynthesisError>;
}
pub struct LinearCombination<F: PrimeField>(Vec<(WireId, F)>);
pub struct R1CSInstance<F: PrimeField> {
    a: Vec<LinearCombination<F>>, b: Vec<LinearCombination<F>>, c: Vec<LinearCombination<F>>,
    num_constraints: usize, num_public: usize,
}
pub struct Witness<F: PrimeField>(Vec<F>);
impl<F: PrimeField> ConstraintSystem<F> {
    pub fn alloc_witness(&mut self, value: F) -> WireId;      // I11 bounds-checked
    pub fn alloc_public(&mut self, value: F) -> WireId;
    pub fn enforce(&mut self, a: LinearCombination<F>, b: LinearCombination<F>, c: LinearCombination<F>);
    pub fn is_satisfied(&self, witness: &Witness<F>) -> bool;
}
pub struct QAP<F: PrimeField> {
    a_polys: Vec<Polynomial<F>>, b_polys: Vec<Polynomial<F>>, c_polys: Vec<Polynomial<F>>,
    domain: EvaluationDomain<F>, target: Polynomial<F>,
}
pub fn r1cs_to_qap<F: PrimeField>(r1cs: &R1CSInstance<F>) -> QAP<F>;
pub fn compute_h_polynomial<F: PrimeField>(qap: &QAP<F>, witness: &Witness<F>) -> Polynomial<F>;
pub struct KzgSrs { powers_g1: Vec<G1Affine>, powers_g2: Vec<G2Affine> }
pub struct KzgCommitment(G1Affine);
pub struct KzgOpeningProof(G1Affine);
pub fn kzg_setup(max_degree: usize, tau: Fr) -> KzgSrs;                 // τ discarded after (P3)
pub fn kzg_commit(srs: &KzgSrs, poly: &Polynomial<Fr>) -> KzgCommitment;
pub fn kzg_open(srs: &KzgSrs, poly: &Polynomial<Fr>, point: Fr) -> (Fr, KzgOpeningProof);
pub fn kzg_verify(srs: &KzgSrs, c: &KzgCommitment, point: Fr, value: Fr, proof: &KzgOpeningProof) -> bool;
pub struct ProvingKey { alpha_g1: G1Affine, beta_g1: G1Affine, beta_g2: G2Affine,
    delta_g1: G1Affine, delta_g2: G2Affine, a_query: Vec<G1Affine>,
    b_g1_query: Vec<G1Affine>, b_g2_query: Vec<G2Affine>, h_query: Vec<G1Affine>, l_query: Vec<G1Affine> }
pub struct VerifyingKey { alpha_g1: G1Affine, beta_g2: G2Affine, gamma_g2: G2Affine,
    delta_g2: G2Affine, gamma_abc_g1: Vec<G1Affine>, circuit_hash: [u8; 32] }   // circuit_hash: T11
pub struct Proof { a: G1Affine, b: G2Affine, c: G1Affine }                       // I9: 3 elements
pub fn setup<C: ConstraintSynthesizer<Fr>>(circuit: &C, srs: &Phase2Output) -> (ProvingKey, VerifyingKey);
pub fn prove<C: ConstraintSynthesizer<Fr>>(pk: &ProvingKey, circuit: &C, rng: &mut impl CryptoRng) -> Result<Proof, ProveError>;
pub fn verify(vk: &VerifyingKey, public_inputs: &[Fr], proof: &Proof) -> Result<bool, VerifyError>;
pub struct Phase1Output { tau_powers_g1: Vec<G1Affine>, tau_powers_g2: Vec<G2Affine>,
    alpha_tau_g1: Vec<G1Affine>, beta_tau_g1: Vec<G1Affine>, beta_g2: G2Affine }
pub struct Phase2Output { /* circuit-specific SRS elements, opaque outside zkp-groth16::setup */ }
pub struct ContributionProof { /* pairing-based proof of correct update, I7 */ }
pub fn phase1_new(domain_size: usize) -> Phase1Output;                          // dev/test only, ADR-20
pub fn phase1_contribute(prev: &Phase1Output, rng: &mut impl CryptoRng) -> (Phase1Output, ContributionProof);
pub fn phase1_verify_contribution(prev: &Phase1Output, new: &Phase1Output, proof: &ContributionProof) -> bool;
pub fn phase2_from_circuit<C>(phase1: &Phase1Output, circuit: &C) -> Phase2Output;
pub trait SigmaProtocol {
    type Statement; type Witness; type Commitment; type Challenge; type Response;
    fn commit(&self, w: &Self::Witness, rng: &mut impl CryptoRng) -> (Self::Commitment, SigmaState);
    fn challenge(&self, transcript: &mut Transcript) -> Self::Challenge;         // I8
    fn respond(&self, state: &SigmaState, c: &Self::Challenge) -> Self::Response;
    fn verify(&self, x: &Self::Statement, commit: &Self::Commitment, c: &Self::Challenge, resp: &Self::Response) -> bool;
}
pub struct SchnorrProtocol { generator: G1Affine }
pub struct SchnorrProof { commitment: G1Affine, response: Fr }
pub struct Transcript { state: Blake2bState }                                    // ADR-10
impl Transcript {
    pub fn new(label: &'static str) -> Self;
    pub fn append_message(&mut self, label: &'static str, message: &[u8]);
    pub fn challenge_scalar(&mut self, label: &'static str) -> Fr;
}
pub struct TestVector { name: String, circuit_id: String, witness_inputs: Vec<Fr>,
    expected_proof_bytes_arkworks: Vec<u8>, expected_proof_bytes_bellman: Vec<u8> }
pub struct ConformanceReport { passed: usize, failed: Vec<String>, oracle_disagreements: Vec<String> }
pub fn run_differential_suite(vectors: &[TestVector]) -> ConformanceReport;
