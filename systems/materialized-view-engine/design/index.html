Core schema & types (§11)

type ViewId = u64
type SourceId = u32
type PlanHash = [u8; 32]

struct FieldDef {
  name: String,
  ty: LogicalType,
  nullable: bool
}

enum LogicalType { Int64, Float64, Utf8, Bool, Bytes, Timestamp }

struct RelationSchema {
  fields: Vec<FieldDef>,
  schema_version: u32
}

struct ViewDefinition {
  view_id: ViewId,
  name: String,
  plan_hash: PlanHash,
  refresh_policy: RefreshPolicy,
  schema: RelationSchema,
  created_at_epoch: CheckpointEpoch
}

trait Catalog {
  fn register_view(&mut self, def: ViewDefinition) -> Result<(), EngineError>
  fn resolve(&self, name: &str) -> Option<ViewDefinition>
  fn drop_view(&mut self, id: ViewId) -> Result<(), EngineError>
}
enum PlanNode {
  Source { source_id: SourceId, schema: RelationSchema },
  Filter { input: Box<PlanNode>, predicate: PredicateExpr },
  Project { input: Box<PlanNode>, columns: Vec<String> },
  Join { left: Box<PlanNode>, right: Box<PlanNode>, on: PredicateExpr },
  Aggregate { input: Box<PlanNode>, group_by: Vec<String>, agg: AggExpr },
  Sink { input: Box<PlanNode>, view_id: ViewId }
}

struct LogicalPlan {
  root: PlanNode,
  hash: PlanHash
}

struct IncrementalPlan {
  source_plan: PlanHash,
  arrangements: Vec<ArrangementSpec>
}

trait OptimizerRule {
  fn apply(&self, plan: &LogicalPlan) -> Option<LogicalPlan>
}
type OperatorId = u32
type LogicalTime = u64

struct ExecutionBudget {
  max_rows: usize,
  max_runtime_ns: u64
}

enum TaskResult { Progressed, Blocked, Idle, Completed }

trait SchedulerTask {
  fn poll_batch(&mut self, budget: ExecutionBudget) -> TaskResult
}

struct Watermark {
  source_id: SourceId,
  logical_time: LogicalTime
}

enum RefreshPolicy { Continuous, Interval(u64), Manual, OnCommit }

struct RefreshTask {
  view_id: ViewId,
  policy: RefreshPolicy,
  next_due: LogicalTime
}
struct Delta {
  row_key: Vec<u8>,
  diff: i64,
  ts: LogicalTime
}

struct DeltaBatch {
  operator_id: OperatorId,
  deltas: Vec<Delta>,
  watermark: Watermark
}

trait Arrangement {
  fn apply(&mut self, batch: DeltaBatch) -> DeltaBatch
  fn snapshot(&self) -> ArrangementSnapshot
}

trait StateHandle {
  fn begin_txn(&self) -> StateTxn
}

trait StateTxn {
  fn get(&self, key: &[u8]) -> Option<Vec<u8>>
  fn put(&mut self, key: &[u8], value: &[u8])
  fn commit(self) -> Result<(), EngineError>
}
type CheckpointEpoch = u64

struct ReplayFrame {
  frame_id: u64,
  wal_offset: u64,
  epoch: CheckpointEpoch,
  operator_id: OperatorId,
  payload_checksum: u128,
  payload: Vec<u8>
}

struct WalSegment {
  segment_id: u64,
  start_offset: u64,
  sealed: bool
}

struct ReplayCursor {
  wal_offset: u64,
  epoch: CheckpointEpoch,
  checksum: u128
}

struct FileRef {
  path: String,
  byte_len: u64,
  checksum: u128
}

struct CheckpointManifest {
  epoch: CheckpointEpoch,
  wal_range_start: u64,
  wal_range_end: u64,
  snapshot_files: Vec<FileRef>,
  checksum: u128
}
enum AdminCommand {
  CreateView { def: ViewDefinition },
  DropView { id: ViewId },
  PauseView { id: ViewId },
  ResumeView { id: ViewId },
  ForceCheckpoint { epoch: CheckpointEpoch },
  Rehydrate { id: ViewId }
}

enum EngineError {
  WalCorruption,
  SnapshotMismatch,
  WatermarkRegression,
  SchedulerStarvation,
  StateConflict,
  QueueOverflow,
  ProtocolViolation,
  SchemaIncompatible,
  ReplayDivergence
}
