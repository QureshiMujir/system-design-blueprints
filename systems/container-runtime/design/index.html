Core schema & types (§11)

// pkg/spec/spec.go — OCI Runtime Spec config.json (Runtime Spec v1.1.0)
type Spec struct {
    OCIVersion  string
    Process     *Process
    Root        *Root
    Hostname    string
    Mounts      []Mount
    Hooks       *Hooks
    Annotations map[string]string
    Linux       *LinuxSpec
}
type Root struct { Path string; Readonly bool }
type Process struct {
    Terminal     bool
    User         User
    Args         []string
    Env          []string
    Cwd          string
    Capabilities *LinuxCapabilities
    Rlimits      []POSIXRlimit
    NoNewPriv    bool
    OOMScoreAdj  *int
}
type Mount struct { Destination, Type, Source string; Options []string }
// pkg/spec/spec.go (continued) — Linux isolation fields (Runtime Spec §linux)
type LinuxSpec struct {
    Namespaces        []LinuxNamespace
    UIDMappings       []LinuxIDMap
    GIDMappings       []LinuxIDMap
    Resources         *LinuxResources
    CgroupsPath       string
    Seccomp           *LinuxSeccomp
    RootfsPropagation string
    MaskedPaths       []string
    ReadonlyPaths     []string
    Sysctl            map[string]string
}
type LinuxNamespace struct { Type, Path string } // Type ∈ §8.3 table
type LinuxIDMap struct { ContainerID, HostID, Size uint32 }
type LinuxCapabilities struct {
    Bounding, Effective, Inheritable, Permitted, Ambient []string
}
type LinuxResources struct {
    Memory *LinuxMemory; CPU *LinuxCPU; Pids *LinuxPids
    BlockIO *LinuxBlockIO; Devices []LinuxDevice
}
// pkg/spec/validate.go — signatures only; bodies implement §13 rules
func Load(bundlePath string) (*Spec, error)
func Validate(s *Spec) []ValidationError
func Default(rootfs string) *Spec
type ValidationError struct { Path, Rule string }
// pkg/runtime/state.go — OCI Runtime Spec §state; persisted per I1
type State struct {
    OCIVersion  string
    ID          string
    Status      string // "creating"|"created"|"running"|"stopped" (§8.4)
    Pid         int
    Bundle      string
    Annotations map[string]string
}
func (s *State) WriteAtomic(dir string) error  // tmp+fsync+rename, I1/I13
func ReadState(dir string) (*State, error)
// pkg/image/manifest.go — OCI Image Spec v1.1.0 (§8.1)
type Manifest struct {
    SchemaVersion int
    MediaType     string
    Config        Descriptor
    Layers        []Descriptor
}
type Descriptor struct { MediaType, Digest string; Size int64 }
type Index struct { SchemaVersion int; Manifests []Descriptor }

func Unpack(m Manifest, store Store, dstRootfs string) error
// pkg/registry/client.go — OCI Distribution v2 client (signatures only)
type Client struct { HTTP *http.Client; Auth AuthProvider; UserAgent string }
type AuthProvider interface {
    Token(ctx context.Context, scope string) (string, error)
}
func (c *Client) Pull(ctx context.Context, ref Reference, dst Store) (Manifest, error)
func (c *Client) FetchManifest(ctx context.Context, ref Reference) (Manifest, []byte, error)
func (c *Client) FetchBlob(ctx context.Context, ref Reference, digest string) (io.ReadCloser, error)
// pkg/store/store.go — content-addressable blob store (I7, §34)
type Store interface {
    Put(ctx context.Context, r io.Reader) (digest string, err error)
    Get(ctx context.Context, digest string) (io.ReadCloser, error)
    Has(ctx context.Context, digest string) bool
    GC(ctx context.Context, live []string) (removed int, err error)
}
// pkg/overlay/mount.go — overlayfs orchestration (§8.1, §15)
type Layout struct { Lower []string; Upper, Work, Merged string }
func Mount(l Layout, readonly bool) error
func Unmount(l Layout) error // must precede rmdir(Upper/Work), I12
// pkg/namespace/clone.go — clone3 composition (signatures only, Phase 4)
type CloneArgs struct {
    Namespaces     uint64 // CLONE_NEW* bitmask, from LinuxNamespace list
    CgroupFD       int    // CLONE_INTO_CGROUP target, I4
    UIDMap, GIDMap []LinuxIDMap
    ChildSig       int
}
type Child struct { Pid int; Sync io.ReadWriteCloser }
func Clone(args CloneArgs, payload func(io.ReadWriteCloser) error) (Child, error)
func WriteUserMap(pid int, uidMap, gidMap []LinuxIDMap) error // I9 ordering
// pkg/cgroup/manager.go — cgroup v2 subtree (Phase 6, I4, I6)
type Manager interface {
    Create(path string, controllers []string) error
    Apply(path string, res *LinuxResources) error
    Stats(path string) (Stats, error)
    Destroy(path string) error // must wait cgroup.events:populated==0, I6
}
type Stats struct { MemoryCurrent, PidsCurrent uint64; CPUUsageNanos uint64 }
// pkg/capability/apply.go and pkg/seccomp/compile.go (Phase 7, I8)
func ApplyCapabilities(c *LinuxCapabilities) error // capset(2) + prctl bounding drop
func CompileSeccomp(s *LinuxSeccomp) (Filter, error)
func LoadSeccomp(f Filter) error // must follow prctl(PR_SET_NO_NEW_PRIVS,1), I8
type Filter struct { Arch string; Program []byte }
// pkg/network/bridge.go and errs/errs.go
type NetOpts struct { Bridge string; PortMappings []PortMapping }
type PortMapping struct { HostPort, ContainerPort int; Proto string }
func Setup(pid int, opts NetOpts) (NetEndpoint, error)
func Teardown(id string) error // I10, exactly-once

type Code string
const (
    CodeNotFound Code = "not_found"; CodeInvalidSpec Code = "invalid_spec"
    CodeInvalidState Code = "invalid_state"; CodeKernel Code = "kernel"
    CodePhaseDeferred Code = "phase_deferred"
)
type Error struct { C Code; P string; W error; R string } // R: "§N.s" when CodePhaseDeferred
func (e *Error) Code() Code; func (e *Error) Path() string; func (e *Error) Wrapped() error
// pkg/runtime/machine.go — the public contract (§8.4); conformance drives this directly
type Runtime interface {
    Create(ctx context.Context, id, bundle string) error
    Start(ctx context.Context, id string) error
    Kill(ctx context.Context, id string, sig syscall.Signal) error
    Delete(ctx context.Context, id string) error
    State(ctx context.Context, id string) (State, error)
    Exec(ctx context.Context, id string, p Process) error
    List(ctx context.Context) ([]State, error)
}
