Core schema & types (§11)

// experiment.rs
pub struct Experiment {
    pub id: Uuid,
    pub version: SchemaVersion,                 // breaking-change guard
    pub title: String,
    pub description: Option<String>,
    pub tags: Vec<String>,
    pub steady_state_hypothesis: SteadyStateHypothesis,
    pub method: Vec<Action>,
    pub rollbacks: Vec<Action>,
    pub configuration: Config,
    pub blast_radius: BlastRadius,
    pub schedule: Option<Schedule>,
    pub created_at: DateTime<Utc>,
}

// hypothesis.rs
pub struct SteadyStateHypothesis { pub title: String, pub probes: Vec<Probe> }
pub struct Probe {
    pub name: String,
    pub provider: ProbeProviderRef,             // discriminated union: HTTP/TCP/exec/metric
    pub tolerance: Tolerance,
    pub frequency: Option<Duration>,            // continuous-check interval; None = once
    pub timeout: Duration,
}

pub enum Tolerance {
    Exact(Value),
    Range { min: Value, max: Value },
    Threshold { op: CmpOp, value: Value },
    Regex(String),
    Jsonpath { path: String, expect: Box<Tolerance> },
}

pub enum CmpOp { Lt, Le, Gt, Ge, Eq, Ne }
pub enum Value  { Number(f64), Str(String), Bool(bool), Null }

// action.rs
pub enum Action {
    Inject(Fault),
    Probe(Probe),
    Pause { duration: Duration },
    Rollback(FaultHandleRef),
}

// fault.rs
pub enum Fault {
    NetworkLatency  { ms: u32, jitter_ms: u32, correlation: f32, interface: Option<String> },
    NetworkLoss     { percent: f32, correlation: f32 },
    NetworkPartition{ groups: Vec<TargetGroup>, direction: Direction },
    NetworkBandwidth{ rate_kbps: u64 },
    NetworkCorrupt  { percent: f32 },
    NetworkReorder  { percent: f32, correlation: f32 },
    CpuStress       { workers: u32, load_percent: f32, duration: Option<Duration> },
    MemStress       { bytes: u64 },
    IoStress        { workers: u32, mode: IoMode },
    ForkBomb        { processes: u32 },
    ProcessKill     { selector: ProcSelector, signal: Signal },
    ProcessStop     { selector: ProcSelector },
    ClockSkew       { offset_ms: i64, drift_ppm: i32 },
    FsFault         { kind: FsFaultKind, path: PathBuf },
    HttpFault       { match_: HttpMatch, abort: Option<u16>, delay: Option<Duration>, fault_percent: f32 },
    GrpcFault       { match_: GrpcMatch, status: GrpcStatus, fault_percent: f32 },
}

pub enum FsFaultKind  { PermissionDenied, IoError, Latency(Duration), NotFound }
pub enum Direction    { From, To, Both }
pub enum IoMode       { Read, Write, ReadWrite }

// blast_radius.rs
pub struct BlastRadius {
    pub max_targets:  Option<usize>,
    pub max_percent:  Option<f32>,              // 0.0–100.0
    pub scope:        Scope,
    pub exclude:      Vec<Selector>,
}
pub enum Scope { Host, Service, Namespace, Cluster }

// handle.rs
pub struct FaultHandle {
    pub id:          Uuid,
    pub fault:       Fault,
    pub targets:     Vec<TargetId>,
    pub ttl:         Duration,
    pub applied_at:  DateTime<Utc>,
    pub revert:      RevertSpec,                // typed revert instructions
}
pub enum RevertSpec {
    TcQdiscDel      { dev: String, handle: String },
    IptablesDel     { rules: Vec<String> },
    CgroupRemove    { path: PathBuf },
    ProcessSignal   { pids: Vec<Pid>, signal: Signal },
    ClockReset,
    FuseUnmount     { mountpoint: PathBuf },
    EnvoyRuleRemove { cluster: String, route_id: Uuid },
    Noop,
}

// verdict.rs
pub enum Verdict   { Hold, Deviated, Aborted, Error(String) }
pub enum HypothesisResult { Passing(Vec<ProbeResult>), Failing(Vec<ProbeResult>) }
pub struct ProbeResult { pub probe_name: String, pub value: Value, pub passed: bool, pub latency_ms: u64 }

// state.rs
pub enum ExperimentState {
    Created, Validating, BeforeSteadyState, Running,
    Monitoring, RollingBack, AfterSteadyState, Verdict, Done, Aborted,
}

// report.rs
pub struct Report {
    pub experiment_id: Uuid,
    pub verdict:       Verdict,
    pub before:        HypothesisResult,
    pub during:        Vec<HypothesisResult>,   // one per continuous-check interval
    pub after:         HypothesisResult,
    pub timeline:      Vec<Event>,
    pub faults:        Vec<FaultHandle>,
    pub duration:      Duration,
    pub generated_at:  DateTime<Utc>,
}

pub enum Event {
    StateTransition     { from: ExperimentState, to: ExperimentState, at: DateTime<Utc> },
    ProbeResult         { probe: String, result: ProbeResult, phase: CheckPhase, at: DateTime<Utc> },
    FaultApplied        { handle: FaultHandle, at: DateTime<Utc> },
    FaultReverted       { handle_id: Uuid, ok: bool, at: DateTime<Utc> },
    SteadyStatePassed   { phase: CheckPhase, at: DateTime<Utc> },
    SteadyStateFailed   { phase: CheckPhase, probes: Vec<ProbeResult>, at: DateTime<Utc> },
    DeviationDetected   { probe: String, value: Value, at: DateTime<Utc> },
    GuardRailTripped    { condition: String, at: DateTime<Utc> },
    KillSwitchEngaged   { operator: String, reason: String, at: DateTime<Utc> },
    DeadManSwitchTriggered{ reason: DeadManReason, at: DateTime<Utc> },
    TtlExpired          { handle_id: Uuid, at: DateTime<Utc> },
    RollbackComplete    { reverted: Vec<Uuid>, at: DateTime<Utc> },
    ExperimentDone      { verdict: Verdict, at: DateTime<Utc> },
}

pub enum CheckPhase { Before, During, After }
pub enum DeadManReason { HeartbeatLost, TtlExpired }
// provider.rs
pub trait ProbeProvider: Send + Sync {
    type Config: DeserializeOwned;
    async fn probe(&self, config: &Self::Config, timeout: Duration) -> Result<Value, ProbeError>;
}

// checker.rs
pub struct SteadyStateChecker { /* probes, evaluator, event_tx */ }
impl SteadyStateChecker {
    pub async fn check_before(&self) -> HypothesisResult;
    pub async fn check_after(&self) -> HypothesisResult;
    pub fn start_continuous(&self, interval: Duration) -> ContinuousCheckHandle;
    // ContinuousCheckHandle::deviation_rx() -> broadcast::Receiver<DeviationEvent>
}

// tolerance.rs
pub struct ToleranceEvaluator;
impl ToleranceEvaluator {
    pub fn evaluate(&self, tolerance: &Tolerance, value: &Value) -> bool;
}
// injector.rs
pub trait FaultInjector: Send + Sync {
    async fn apply(&self, fault: &Fault, targets: &[TargetId]) -> Result<FaultHandle, FaultError>;
    async fn revert(&self, handle: &FaultHandle) -> Result<(), FaultError>;  // idempotent
    fn supports(&self, fault: &Fault) -> bool;
}
// FaultError variants: KernelError, PermissionDenied, TargetNotFound, AlreadyApplied, RevertFailed
// selector.rs
pub struct TargetSelector { expr: SelectorExpr }  // label key=value, logical AND/OR
// registry.rs
pub trait TargetRegistry: Send + Sync {
    async fn resolve(&self, selector: &TargetSelector) -> Result<Vec<Target>, RegistryError>;
}
// blast_radius.rs
pub struct BlastRadiusEnforcer;
impl BlastRadiusEnforcer {
    pub fn enforce(&self, targets: Vec<Target>, br: &BlastRadius) -> Result<EnforcedTargets, BlastRadiusError>;
    pub fn dry_run_plan(&self, targets: Vec<Target>, br: &BlastRadius) -> DryRunPlan;
}
pub struct DryRunPlan { pub selected: Vec<Target>, pub excluded: Vec<Target>, pub capped: bool }
// runner.rs
pub struct ExperimentRunner { /* deps: checker, injector, safety, reporter */ }
impl ExperimentRunner {
    pub async fn run(&self, experiment: Experiment) -> Report;
    pub async fn abort(&self, experiment_id: Uuid) -> Result<(), OrchestratorError>;
}
// kill_switch.rs
pub struct KillSwitch { state: Arc<AtomicBool>, /* agent_tx */ }
impl KillSwitch {
    pub async fn engage(&self, operator: &str, reason: &str) -> Result<(), SafetyError>;
    pub async fn disengage(&self, operator: &str) -> Result<(), SafetyError>;
    pub fn is_engaged(&self) -> bool;
}
// dead_man.rs (agent-side)
pub struct HeartbeatWatchdog { last_seen: Arc<Mutex<Instant>>, interval: Duration }
impl HeartbeatWatchdog {
    pub fn update(&self);
    pub async fn run(&self, revert_fn: impl Fn() -> BoxFuture<()> + Send + Sync);
    // fires revert_fn if Instant::now() - last_seen > 2 * interval
}
// directive.rs
pub struct FaultDirective {
    pub handle_id: Uuid,
    pub fault: Fault,
    pub targets: Vec<TargetId>,
    pub ttl: Duration,
    pub signature: String,          // JWT; validated before apply
}
// executor.rs
pub struct LocalFaultExecutor { /* injector dispatch */ }
impl LocalFaultExecutor {
    pub async fn apply_directive(&self, d: FaultDirective) -> Result<(), AgentError>;
    pub async fn clear_handle(&self, handle_id: Uuid) -> Result<(), AgentError>;
    pub async fn clear_all(&self) -> Vec<Result<(), AgentError>>;  // dead-man path
}
// state.rs
pub enum AgentFaultState { Applying, Active { expires_at: Instant }, Reverting, Cleared, Error(String) }
