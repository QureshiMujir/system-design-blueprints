Core schema & types (§11)

// ownership: value type, copy               mutation: immutable
// concurrency: none                          persistence: embedded in every persisted key
// serialization: fixed 16 bytes              compatibility: stable since v1
struct IndexId(u128);

// ownership: borrowed by callers              mutation: none (trait)
// concurrency: implementors must be Send+Sync persistence: n/a
// serialization: n/a                          compatibility: additive methods only, semver-gated
trait AccessMethod {
    fn lookup(&self, key: &Key) -> Result<Option<Rid>, AieError>;
    fn scan(&self, predicate: &Predicate) -> Result<ScanBatch, AieError>;
    fn insert(&self, key: Key, rid: Rid) -> Result<(), AieError>;
    fn cost_estimate(&self, predicate: &Predicate) -> CostEstimate;
}

// ownership: value type, byte buffer          mutation: immutable once constructed
// concurrency: none                            persistence: raw bytes in pages
// serialization: variable-length              compatibility: encoding versioned via layout_version
struct Key(Box<[u8]>);

// ownership: value type, copy                 mutation: immutable
// concurrency: none                            persistence: stored in leaf entries
// serialization: fixed 12 bytes (page_id+slot) compatibility: stable since v1
struct Rid { page_id: u64, slot: u32 }

// ownership: owned                            mutation: immutable once built
// concurrency: shared read-only across router candidates
// persistence: n/a (query-scoped)             serialization: n/a
enum Predicate { PointEq(Key), Range { lo: Key, hi: Key } }

// ownership: value type (f64 newtype)         mutation: immutable
// concurrency: none                            persistence: embedded in HistogramSnapshot
// serialization: 8 bytes                      compatibility: stable
struct Selectivity(f64);

// ownership: value type                       mutation: immutable
// concurrency: none                            persistence: query-scoped, not persisted
// serialization: n/a                           compatibility: internal only
struct CostEstimate { path: AccessPath, estimated_ns: u64 }

// ownership: enum, copy                       mutation: immutable
// concurrency: none                            persistence: emitted in telemetry
// serialization: tagged enum                   compatibility: additive variants only
enum AccessPath { BTree, Art, CrackedColumn, FullScan }

// ownership: owned by aie-router               mutation: internal state updated per decision
// concurrency: Sync, internally latched         persistence: none (recomputed each startup)
// serialization: n/a                            compatibility: internal
struct Router { /* holds references to candidate AccessMethods and the StatisticsCollector */ }
// ownership: owned by aie-btree                mutation: structural mutation under latch coupling (P7)
// concurrency: latch-coupled (ADR-11)          persistence: pages durable via aie-storage
// serialization: page byte layout (§8.5)       compatibility: layout_version-gated
struct BTree { root: PageId, layout_version: u16 }

// ownership: owned by containing BTree page     mutation: in-place under held latch
// concurrency: single-writer-per-latch          persistence: 16KB page (§8.5)
// serialization: byte layout per §8.5           compatibility: layout_version-gated
struct BTreeNode { level: u16, key_count: u16, prefix: PrefixCompressor }

// ownership: RAII guard                        mutation: n/a
// concurrency: read/write mode, coupled descent persistence: n/a
// serialization: n/a                            compatibility: internal
struct Latch;
struct LatchGuard<'a> { latch: &'a Latch }

// ownership: owned by BTreeNode                 mutation: rebuilt on page-level prefix change
// concurrency: read under node latch            persistence: embedded in page bytes
// serialization: shared-prefix + suffix array   compatibility: layout_version-gated
struct PrefixCompressor { shared_prefix: Box<[u8]> }

// ownership: owned by aie-art                   mutation: structural mutation under OLC (ADR-11)
// concurrency: optimistic lock coupling          persistence: snapshot-persisted (ADR-14)
// serialization: node-type-tagged (§8.6)         compatibility: layout_version-gated
struct ArtTree { root: ArtNode }

// ownership: owned by parent node/tree           mutation: grows/shrinks per P8
// concurrency: generation-counter-guarded OLC    persistence: within ArtTree snapshot
// serialization: tagged union of Node4/16/48/256 compatibility: layout_version-gated
enum ArtNode { Leaf(Rid), Inner4(Node4), Inner16(Node16), Inner48(Node48), Inner256(Node256) }
struct Node4 { keys: [u8; 4], children: [Option<Box<ArtNode>>; 4] }
struct Node16 { keys: [u8; 16], children: [Option<Box<ArtNode>>; 16] }
struct Node48 { index: [u8; 256], children: [Option<Box<ArtNode>>; 48] }
struct Node256 { children: [Option<Box<ArtNode>>; 256] }

// ownership: owned by ArtNode                   mutation: rebuilt on divergence (§8.6)
// concurrency: read under OLC validation         persistence: within node bytes
// serialization: length-prefixed byte string     compatibility: layout_version-gated
struct PathCompression { prefix: Box<[u8]> }
// ownership: owned by aie-cracker                mutation: partitions pieces in place (P1, P6)
// concurrency: copy-on-crack versioning (ADR-10) persistence: not WAL-logged (ADR-13)
// serialization: n/a (ephemeral)                 compatibility: rebuilt lazily (ADR-14)
struct Cracker { column: Column }

// ownership: owned by Cracker                    mutation: boundary insert (§8.2)
// concurrency: latched during BoundaryInsert     persistence: checkpointed, not per-crack logged
// serialization: sorted boundary array            compatibility: layout_version-gated
struct CrackerIndex { boundaries: Vec<PieceBoundary> }
struct Piece { id: u32, size_estimate: u64, refined_generation: u32 }
struct PieceBoundary { value: Key, piece_id: u32 }

// ownership: owned by aie-cracker                 mutation: run merge (§8.3)
// concurrency: latched during merge step          persistence: not persisted, rebuilt on restart
// serialization: n/a                              compatibility: internal
struct AdaptiveMerge { runs: RunIndex }
struct RunIndex { runs: Vec<(Key, Key)> }

// ownership: owned by aie-storage                 mutation: append-mostly, cracked in place
// concurrency: reader/writer via ConcurrencyToken  persistence: column segment files
// serialization: columnar byte layout             compatibility: layout_version-gated
struct Column { segment: SegmentId }

// ownership: owned by aie-scan                    mutation: none (read-only batch)
// concurrency: Send across worker pool             persistence: transient
// serialization: n/a                               compatibility: internal
struct ScanAccelerator;
// ownership: owned by aie-stats                   mutation: incremental update (§8.8, ADR-09)
// concurrency: lock-free append, latched split     persistence: HistogramSnapshot checkpoints
// serialization: bucket array                      compatibility: layout_version-gated
struct Statistics { histograms: Vec<Histogram>, sketches: Vec<Sketch> }
struct Histogram { buckets: Vec<(Key, Key, u64)> }
struct Sketch { registers: Vec<u8> }

// ownership: owned by aie-compaction               mutation: drives tier moves (§8.10)
// concurrency: throttled by backpressure (P10)     persistence: manifest-logged (I12)
// serialization: manifest JSON                     compatibility: layout_version-gated
struct CompactionJob { source_tier: Tier, target_tier: Tier }
enum Tier { Hot, Warm, Cold }

// ownership: owned by aie-concurrency               mutation: monotonic bump only (I13)
// concurrency: atomic                               persistence: none
// serialization: u64                                compatibility: internal
struct ConcurrencyToken(u64);
struct Version(u64);

// ownership: enum, owned by call sites              mutation: immutable
// concurrency: Send+Sync                            persistence: n/a
// serialization: n/a                                compatibility: additive variants only
enum AieError {
    Corruption { detail: String },
    ConcurrencyConflict,
    Backpressure,
    NotFound,
    InvalidPredicate,
    StorageIo(String),
}
