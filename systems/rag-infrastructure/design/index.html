Core schema & types (§11)

trait StorageEngine
  fn append_wal(&self, record: WalRecord) -> Result<u64, StorageError>;
  fn snapshot(&self) -> Result<SnapshotManifest, StorageError>;
  fn replay(&self, from_offset: u64) -> Result<ReplayCursor, StorageError>;
  fn compact(&self, policy: CompactionPolicy) -> Result<CompactionReport, StorageError>;

trait RetrievalPlanner
  fn plan_fanout(&self, req: &RetrievalRequest) -> Result<FanoutPlan, PlannerError>;
  fn budget_timeout(&self, plan: &FanoutPlan) -> TimeoutBudget;
  fn compile_filter(&self, expr: &FilterExpr) -> Result<CompiledFilter, PlannerError>;

trait EmbeddingProvider
  fn batch_embed(&self, chunks: &[Chunk]) -> Result<Vec<EmbeddingRecordRef>, ProviderError>;
  fn retry_policy(&self) -> RetryPolicy;
  fn normalize(&self, raw: RawEmbedding) -> Result<EmbeddingRecordRef, ProviderError>;
