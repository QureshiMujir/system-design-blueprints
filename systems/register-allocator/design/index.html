Core schema & types (§11)

// crates/ra-ir/src/vreg.rs — immutable identifiers; Copy; total order for I4/P4
#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, PartialOrd, Ord)]
pub struct VirtReg(pub u32);           // no serialization compat guarantee across allocator runs

#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, PartialOrd, Ord)]
pub struct PhysReg(pub u16);           // stable per target; RegisterFile-scoped

#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
pub struct RegClass(pub u8);

pub struct RegisterFile {              // built once/target; Arc-shared; immutable after build (I13)
    pub classes: Vec<RegClassInfo>,
    pub aliases: AdjacencyMatrix<PhysReg>,   // symmetric, transitively closed
}

pub struct RegClassInfo { pub id: RegClass, pub physical: SmallVec<[PhysReg; 16]>, pub width_bits: u16, pub name: &'static str }
// crates/ra-liveness/src/range.rs — LiveRange owned per-vreg; rebuilt on mutation (P3)
#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct ProgramPoint(pub u32);      // even=before, odd=after (ADR-09, I4)

pub struct LiveRange { pub vreg: VirtReg, pub class: RegClass, pub intervals: SmallVec<[Interval; 4]>, pub uses: SmallVec<[Use; 8]>, pub def: Def }
                                        // intervals: sorted, non-overlapping (I3)

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Interval { pub start: ProgramPoint, pub end: ProgramPoint }  // half-open [start,end)

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Use { pub point: ProgramPoint, pub constraint: Option<UseConstraint> }

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum UseConstraint { AnyOf(RegClass), Specific(PhysReg), SameAs(VirtReg) }  // I11 enforces Specific
// crates/ra-igraph/src/lib.rs — pure function of LiveRanges (§8.3); mutation only via documented API
pub enum InterferenceGraph {
    Sparse { adjacency: BTreeMap<VirtReg, BTreeSet<VirtReg>>, degrees: BTreeMap<VirtReg, u32> },  // BTree: P4/I9
    Dense  { bitsets: BTreeMap<VirtReg, FixedBitSet>, degrees: BTreeMap<VirtReg, u32> },
}
impl InterferenceGraph {
    pub fn build(liveness: &Liveness) -> Self;                 // O(V·D) sparse | O(V²/8) dense (ADR-04)
    pub fn interferes(&self, a: VirtReg, b: VirtReg) -> bool;  // I1 check dependency
    pub fn neighbors(&self, v: VirtReg) -> Box<dyn Iterator<Item = VirtReg> + '_>;
    pub fn degree(&self, v: VirtReg) -> u32;
    pub fn remove(&mut self, v: VirtReg);                       // Simplify (§8.4)
    pub fn add_edge(&mut self, a: VirtReg, b: VirtReg);         // spill-introduced vreg re-entry
}
// crates/ra-coalesce/src/lib.rs — I7 enforced inside coalesce()
pub trait Coalescer { fn coalesce(&mut self, g: &mut InterferenceGraph, moves: &[MovePair], cfg: &Cfg) -> CoalesceReport; }
pub struct MovePair { pub src: VirtReg, pub dst: VirtReg, pub at: ProgramPoint, pub weight: f32 }
pub struct CoalesceReport { pub merged: BTreeMap<VirtReg, VirtReg>, pub frozen: BTreeSet<VirtReg> }
// crates/ra-graph/src/lib.rs and crates/ra-linear/src/lib.rs — both implement Allocator (P2)
pub trait Allocator: Send + Sync { fn allocate(&self, ir: &Module, liveness: &Liveness) -> Assignment; }

pub struct GraphAllocator { pub config: GraphConfig, pub register_file: Arc<RegisterFile> }
pub struct GraphConfig { pub coalescer: CoalescerChoice, pub spill_weight: SpillWeightModel, pub iter_cap: u32 }  // I8/ADR-28
pub enum CoalescerChoice { Conservative, IteratedConservative }             // ADR-06

pub struct LinearScanAllocator { pub config: LinearConfig }
pub struct LinearConfig { pub split_intervals: bool, pub second_chance: bool, pub use_constraints_as_hints: bool }  // ADR-17/18

pub struct Assignment { pub map: BTreeMap<VirtReg, Location>, pub spills: Vec<SpillSlot>, pub rematerializations: Vec<Remat>, pub stats: AllocStats }
pub enum Location { Physical(PhysReg), Spill(SpillSlot), Constant(ConstantId) }  // Constant = rematerialised (I14)
// crates/ra-resolver/src/lib.rs — sole IR-mutation point (P9)
pub fn resolve(ir: &mut Module, assignment: &Assignment, cfg: &Cfg) -> ResolveReport;
pub struct ResolveReport { pub critical_edges_split: usize, pub spill_instructions_inserted: usize, pub reload_instructions_inserted: usize, pub move_instructions_inserted: usize }

// crates/ra-verifier/src/lib.rs — checks I1–I14
pub fn verify(ir: &Module, assignment: &Assignment, liveness: &Liveness, cc: &CallingConvention) -> Vec<VerifierError>;
