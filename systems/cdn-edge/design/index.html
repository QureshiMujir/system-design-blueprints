Core schema & types (§11)

// crates/cdn-core — request/response, ownership: caller-owned, immutable after parse
struct Request {
  method: Method, uri: CanonicalUri, version: HttpVersion,
  headers: HeaderMap, body: BodyRef, client_ip: IpAddr,
  tls_info: Option<TlsInfo>, trace_ctx: Option<W3cTraceContext>,
}
struct Response {
  status: u16, version: HttpVersion, headers: HeaderMap, body: BodyRef,
}
enum BodyRef { Empty, Fixed(Bytes), Streaming(ByteStreamHandle) }
// crates/cdn-core — cache key + entry; mutation: entry replaced atomically
// on write (never mutated in place); persistence: entry serialized via
// rkyv for L2 (ADR-06); concurrency: entries are Arc<CacheEntry>, CoW
struct CacheKey { hash: u64, debug_repr: Option<SmolStr> } // debug_repr dev-only
struct CacheEntry {
  key: CacheKey, status: u16, headers: HeaderMap, body: BodyRef,
  stored_at: Instant, freshness_lifetime: Duration,
  validators: Validators, vary_headers: SmallVec<[HeaderName; 4]>,
  tags: SmallVec<[Tag; 4]>, version: u64, tier: TierId, size_bytes: u32,
}
struct Validators { etag: Option<ETag>, last_modified: Option<HttpDate> }
enum TierId { L1, L2, Shield(NodeId) }
// crates/cdn-cache-mem — L1 store; concurrency: DashMap-sharded, per-shard
// lock; persistence: none (volatile, RTO 30s per §4)
struct L1Store {
  shards: [L1Shard; 256], eviction: WTinyLFU, total_bytes: AtomicUsize,
  max_bytes: usize,
}
struct WTinyLFU { sketch: CountMinSketch4Row, window: BoundedLru, main: BoundedLru }
struct CountMinSketch4Row { rows: [Vec<AtomicU8>; 4], reset_at: AtomicU64 }
// crates/cdn-cache-nvme — L2 store; persistence: mmap-backed file per
// shard, survives process restart; concurrency: single-writer/shard,
// multi-reader via mmap
struct L2Store { shards: [L2Shard; 64], eviction: S3Fifo, index: L2Index }
struct L2Shard { file: MmapMut, free_list: FreeListAllocator }
struct L2Index { entries: BTreeMap<u64, L2Offset> } // hash -> byte offset
// crates/cdn-ring — placement; immutable snapshot, swapped via ArcSwap
// on membership change (never mutated in place)
struct HashRing { points: BTreeMap<u64, NodeId>, virtual_per_node: u16 }
struct NodeId(u32)
// placement(key) -> NodeId : first ring point clockwise of hash(key)
// crates/cdn-origin — fetch client; concurrency: per-origin connection
// pool, bounded (ADR-` region); no persistence
struct OriginPool { origins: Vec<OriginHandle>, selection: SelectionPolicy }
struct OriginHandle { addr: SocketAddr, weight: u16, breaker: CircuitBreaker }
enum SelectionPolicy { RoundRobin, Weighted, LeastLoaded, ConsistentHash }
struct CircuitBreaker { state: BreakerState, window: SlidingWindow }
enum BreakerState { Closed, Open { since: Instant }, HalfOpen }
// crates/cdn-collapse — single-flight; concurrency: one Fetching state
// per key, waiters queued (bounded, ADR-15)
enum CoalesceState { Idle, Fetching { waiters: BoundedVec<Waker, 4096> } }
struct CollapseTable { entries: DashMap<u64, CoalesceState> }
// crates/cdn-purge — purge messages; wire format for propagation bus
struct PurgeMsg {
  kind: PurgeKind, purge_seq: u64, zone_id: ZoneId, issued_at: Instant,
}
enum PurgeKind { ByUrl(CacheKey), ByTag(Tag), Soft(CacheKey), FullFlush }
struct Tag(SmolStr)
