Core schema & types (§11)

// ipc-core — traits, wait/readiness primitives, error taxonomy
pub trait Pollable {                       // owned by object; &self; concurrency: interior-mut
    fn poll(&self, want: PollMask) -> PollMask;
    fn add_waiter(&self, w: WaiterRef);     // mutation: WaitQueue push, lock-guarded
    fn remove_waiter(&self, w: WaiterRef);
}
pub trait Readable { fn read(&self, buf: &mut [u8]) -> Result<usize, IpcError>; }
pub trait Writable { fn write(&self, buf: &[u8]) -> Result<usize, IpcError>; }
pub struct WaitQueue { waiters: Mutex<VecDeque<WaiterRef>> }   // wake_one default (P10, I9)
pub struct PollMask(u32);                   // bitflags: READABLE|WRITABLE|HUP|ERR
pub struct FdHandle(RawFd);                 // Copy; not Send across processes (only via ADR-25)
#[non_exhaustive]
pub enum IpcError { EAGAIN, EPIPE, EMSGSIZE, EPERM, EBUSY, ENOENT, EEXIST, EFAULT }
// ipc-pipe — anonymous + named pipe; ownership: kernel-object, refcounted by open ends
pub struct Pipe {
    pub buffer: PageList,            // owned; mutation: mutex-guarded append/pop
    pub readers: AtomicUsize,        // I1
    pub writers: AtomicUsize,        // I1
    pub capacity_bytes: usize,       // immutable post-create unless F_SETPIPE_SZ
    pub waiters_read: WaitQueue,
    pub waiters_write: WaitQueue,
}                                     // not persisted; not serialized; not fork-shared beyond FD dup
pub struct Fifo { pub inode_path: PathBuf, pub backing: Pipe }   // persists as a filesystem name only
// ipc-shm — segment: named, refcounted, kernel-backed; sync layered by the caller
pub struct ShmRegion {
    pub id: ShmId,
    pub size: usize,                  // immutable post-create
    pub attached_count: AtomicUsize,  // I4; concurrency: atomic across processes via shared page
    pub creator: Pid,
    pub futex_word: AtomicU32,        // caller-defined protocol; see §8.2
}
// ipc-mq — priority queue; ownership: single kernel object, many senders/one+ receivers
pub struct Mqueue {
    pub heap: BinaryHeap<MqMsg>,       // I5 comparator inverts (max-heap on priority)
    pub max_msgs: usize,               // immutable post-create
    pub max_msg_size: usize,           // I13
    pub waiters: WaitQueue,
}
pub struct MqMsg { pub priority: u32, pub data: Bytes }   // Bytes: refcounted immutable payload
// ipc-unix-sock — stream/dgram/seqpacket; fd + credential passing (I6)
pub enum SockKind { Stream, Dgram, Seqpacket }
pub enum SockState { Closed, Listening, Connecting, Connected, HalfClosed }
pub struct SocketCreds { pub pid: Pid, pub uid: Uid, pub gid: Gid }   // kernel-populated only (T10)
pub struct UnixSocket {
    pub kind: SockKind,
    pub state: SockState,
    pub recv_buf: PageList,
    pub send_peer: Option<Arc<UnixSocket>>,
    pub creds_to_pass: Option<SocketCreds>,
    pub fds_to_pass: Vec<FdHandle>,      // I6: refcount bumped before queuing
}
// ipc-sync — POSIX named semaphore / PTHREAD_PROCESS_SHARED mutex+condvar equivalents
pub struct Semaphore { pub count: AtomicIsize, pub waiters: WaitQueue }       // I11
pub struct Mutex     { pub owner: AtomicOption<Tid>, pub waiters: WaitQueue } // I12
pub struct CondVar   { pub generation: AtomicU32, pub waiters: WaitQueue }
// ipc-cap — unified capability object (§20); ownership: process-scoped CapSet
pub struct CapToken { pub kind: CapKind, pub target: ObjectId, pub rights: RightsMask }
pub struct CapSet   { pub tokens: HashSet<CapToken> }   // per-process; I7 checked before every op
pub enum CapKind { PipeAccess, ShmAccess, MqAccess, SockAccess, SyncAccess }
// ipc-notify — readiness surface (§21); one Notifier per interested process/thread
pub struct Interest  { pub handle: FdHandle, pub mask: PollMask, pub mode: TriggerMode }
pub enum TriggerMode { LevelTriggered, EdgeTriggered }    // I8 governs EdgeTriggered
pub struct ReadyEvent { pub handle: FdHandle, pub mask: PollMask }
pub struct Notifier { pub interests: HashMap<FdHandle, Interest>, pub ready: VecDeque<ReadyEvent> }
// ipc-flow — backpressure (§22); ownership: one FlowController per bounded channel
pub struct Watermark { pub high: usize, pub low: usize }               // I10
pub struct FlowController { pub bytes_buffered: AtomicUsize, pub mark: Watermark }
