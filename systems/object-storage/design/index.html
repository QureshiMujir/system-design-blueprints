Core schema & types (§11)

pub struct BucketName(String);         // owned, immutable post-parse; Send+Sync; wire: UTF-8
impl BucketName {
    pub fn parse(s: &str) -> Result<Self, BucketNameError>;  // 3-63 chars, lowercase+hyphen
    pub fn as_str(&self) -> &str;
}

pub struct ObjectKey(String);          // owned, immutable; opaque to the system (P1); <=1024B
impl ObjectKey {
    pub const MAX_LEN: usize;
    pub fn parse(s: &str) -> Result<Self, ObjectKeyError>;
    pub fn as_str(&self) -> &str;
}

pub struct VersionId(pub u128);        // Copy; monotonic per (bucket,key) (I2); 0 = "no version"
impl VersionId {
    pub const NONE: Self;
    pub fn is_none(self) -> bool;
    pub fn encode(self) -> String;     // 32 lowercase hex chars, wire-stable
}
pub enum ETag {                        // Clone, Eq, Hash; serialized as a quoted wire string
    SinglePartMd5([u8; 16]),
    Multipart { combined_md5: [u8; 16], n_parts: u16 },
    Opaque(String),                     // SSE-C / SSE-KMS / server-assigned
}
impl ETag {
    pub fn from_body(body: &[u8]) -> Self;
    pub fn from_parts(part_md5s: &[[u8; 16]]) -> Self;
    pub fn to_header_value(&self) -> String;
}
pub struct MultipartUploadId(pub u128);         // cluster-allocated; base64url wire encoding
impl MultipartUploadId { pub fn encode(self) -> String; }

pub struct PartNumber(pub u16);                 // Copy, Ord; range 1..=10_000 (I7)
impl PartNumber {
    pub const MIN: Self;
    pub const MAX: Self;
    pub fn parse(n: u16) -> Result<Self, PartNumberError>;
}

pub const MIN_PART_BYTES: u64;        // 5 MiB
pub const MAX_PART_BYTES: u64;        // 5 GiB
pub const MAX_OBJECT_BYTES: u64;      // 5 TiB
pub struct FragmentId(pub [u8; 16]);   // xxh3-128 fingerprint; storage key (P6, ADR-03)

pub struct ManifestEntry {             // owned by ObjectManifest; immutable once installed
    pub part_number: u16,              // 0 for non-multipart
    pub fragment:    FragmentId,
    pub bytes:       u64,
    pub md5:         [u8; 16],         // ETag reconstruction
    pub sha256:      [u8; 32],         // integrity verification (I4)
    pub placement:   PlacementPlan,
}

pub struct ObjectManifest {            // installed atomically in one WAL entry (I1, I3)
    pub bytes_total:  u64,
    pub durability:   Durability,
    pub entries:      Vec<ManifestEntry>,   // ordered by part_number then offset
    pub completed_at: chrono::DateTime<chrono::Utc>,
    pub etag:         ETag,
}
pub enum Durability {                  // Copy; per-bucket policy (P7)
    Replication { rf: u8 },            // 1..=7, default 3 (ADR-04)
    Erasure     { k: u8, m: u8 },      // default (6,3) (ADR-05)
    Tiered      { hot: Box<Durability>, cold: Box<Durability> },
}

pub struct StorageNodeId(pub u64);     // Copy, Hash

pub enum PlacementPlan {               // set once by the metadata service (I13); immutable
    Replicated { nodes: Vec<StorageNodeId> },
    EcShards   { data: Vec<StorageNodeId>, parity: Vec<StorageNodeId> },
}
impl PlacementPlan { pub fn replicas(&self) -> usize; }
pub struct Object {                    // one version; immutable post-install (P2)
    pub bucket:        BucketName,
    pub key:           ObjectKey,
    pub version:       VersionId,
    pub manifest:      ObjectManifest,
    pub headers:       UserHeaders,
    pub uploaded_at:   chrono::DateTime<chrono::Utc>,
    pub uploader:      AccessKeyId,
    pub legal_hold:    bool,
    pub retention:     Option<RetentionConfig>,
    pub delete_marker: bool,
}

pub struct UserHeaders {               // Default; owned copy per version
    pub content_type:     Option<String>,
    pub content_encoding:  Option<String>,
    pub content_md5:       Option<[u8; 16]>,
    pub cache_control:     Option<String>,
    pub user_meta:         std::collections::BTreeMap<String, String>,
}

pub struct BucketMeta {                // owned by the metadata service; mutated via WAL ops
    pub name:        BucketName,
    pub created_at:  chrono::DateTime<chrono::Utc>,
    pub region:      String,
    pub durability:  Durability,
    pub versioning:  VersioningState,
    pub object_lock: Option<ObjectLockConfig>,
    pub lifecycle:   Vec<LifecycleRule>,
    pub policy:      Option<BucketPolicy>,
    pub replication: Option<ReplicationConfig>,
    pub acl:         BucketAcl,
}

pub enum VersioningState { Unversioned, Enabled, Suspended }   // Copy; monotone per P9/I5
pub struct MultipartUpload {
    pub id:            MultipartUploadId,
    pub bucket:        BucketName,
    pub key:           ObjectKey,
    pub initiated_at:  chrono::DateTime<chrono::Utc>,
    pub initiator:     AccessKeyId,
    pub headers:       UserHeaders,
    pub parts:         std::collections::BTreeMap<PartNumber, PartRecord>,
    pub durability:    Durability,
    pub state:         MpuState,
}

pub struct PartRecord {
    pub number:      PartNumber,
    pub bytes:       u64,
    pub md5:         [u8; 16],
    pub fragment:    FragmentId,
    pub placement:   PlacementPlan,
    pub uploaded_at: chrono::DateTime<chrono::Utc>,
}

pub enum MpuState { InProgress, Completing, Completed, Aborting, Aborted, Expired }
pub struct AccessKeyId(pub String);    // 20-char alnum; S3-compatible

pub struct AccessKey {
    pub id:         AccessKeyId,
    pub secret:      Secret,           // zeroize-on-drop
    pub created_at:  chrono::DateTime<chrono::Utc>,
    pub disabled:    bool,
    pub policy:      Option<BucketPolicy>,
}

pub enum RetentionMode { Governance, Compliance }   // Compliance is irreversible (P10, I8)

pub struct RetentionConfig { pub mode: RetentionMode, pub retain_until: chrono::DateTime<chrono::Utc> }

pub struct ObjectLockConfig {
    pub default_mode:  RetentionMode,
    pub default_days:  Option<u32>,
    pub default_years: Option<u32>,
}
pub struct BucketPolicy { pub statements: Vec<PolicyStatement> }
pub struct PolicyStatement {
    pub sid:        Option<String>,
    pub effect:      PolicyEffect,
    pub principals:  Vec<Principal>,
    pub actions:     Vec<S3Action>,
    pub resources:   Vec<S3Resource>,
    pub conditions:  Vec<PolicyCondition>,
}
pub enum PolicyEffect { Allow, Deny }
pub enum Principal { Anonymous, AccessKey(AccessKeyId), Account(String) }

pub struct LifecycleRule { pub id: String, pub filter: LifecycleFilter, pub actions: Vec<LifecycleAction>, pub enabled: bool }
pub struct LifecycleFilter { pub prefix: Option<String>, pub object_tag: Option<(String,String)>, pub size_gt: Option<u64>, pub size_lt: Option<u64> }
pub enum LifecycleAction {
    Expire                      { days: u32 },
    ExpireNoncurrentVersion     { days: u32 },
    Transition                  { days: u32, target: StorageClass },
    TransitionNoncurrentVersion { days: u32, target: StorageClass },
    AbortIncompleteMultipart    { days: u32 },
}
pub enum StorageClass { Standard, InfrequentAccess, Glacier, DeepArchive }

pub struct ReplicationConfig { pub role: AccessKeyId, pub rules: Vec<ReplicationRule> }
pub struct ReplicationRule { pub id: String, pub priority: u32, pub filter: LifecycleFilter, pub destination: ReplicationDestination, pub delete_marker: bool, pub existing_objects: bool }
pub struct ReplicationDestination { pub bucket: String, pub region: String, pub storage_class: Option<StorageClass> }
pub trait MetadataService: Send + Sync {   // one authoritative writer (I13); async RPC
    async fn create_bucket(&self, name: &BucketName, cfg: BucketCreateRequest) -> Result<BucketMeta>;
    async fn put_object_begin(&self, b: &BucketName, k: &ObjectKey, hdrs: UserHeaders) -> Result<PutObjectTicket>;
    async fn put_object_commit(&self, t: PutObjectTicket, m: ObjectManifest) -> Result<VersionId>;
    async fn head_object(&self, b: &BucketName, k: &ObjectKey, v: Option<VersionId>) -> Result<Object>;
    async fn delete_object(&self, b: &BucketName, k: &ObjectKey, v: Option<VersionId>) -> Result<VersionId>;
    async fn list_objects_v2(&self, b: &BucketName, q: ListV2Query) -> Result<ListV2Response>;
    async fn mpu_initiate(&self, b: &BucketName, k: &ObjectKey, h: UserHeaders) -> Result<MultipartUpload>;
    async fn mpu_complete(&self, id: MultipartUploadId, parts: Vec<(PartNumber,[u8;16])>) -> Result<(ObjectManifest, VersionId)>;
    async fn put_bucket_policy(&self, b: &BucketName, p: BucketPolicy) -> Result<()>;
    async fn put_bucket_lifecycle(&self, b: &BucketName, l: Vec<LifecycleRule>) -> Result<()>;
    async fn put_bucket_versioning(&self, b: &BucketName, s: VersioningState) -> Result<()>;
}

pub trait BlobService: Send + Sync {       // content-addressed (P6); async RPC
    async fn put_fragment(&self, frag: FragmentId, bytes: bytes::Bytes, plan: PlacementPlan) -> Result<()>;
    async fn get_fragment(&self, frag: FragmentId, range: Option<(u64,u64)>) -> Result<bytes::Bytes>;
    async fn checksum(&self, frag: FragmentId) -> Result<[u8; 32]>;
    async fn repair(&self, frag: FragmentId, from: PlacementPlan, to: PlacementPlan) -> Result<()>;
}
