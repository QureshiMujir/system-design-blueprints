pub trait Workflow: Send + 'static {
    type Input:  serde::de::DeserializeOwned + Send + 'static;
    type Output: serde::Serialize + Send + 'static;
    fn name() -> &'static str;
    async fn execute(ctx: &WorkflowContext, input: Self::Input) -> Result<Self::Output, WorkflowError>;
    async fn on_signal(_ctx: &WorkflowContext, _signal_name: &str, _payload: Vec<u8>) {}
    fn on_query(_state: &Self, _query_name: &str, _payload: Vec<u8>) -> Result<Vec<u8>, QueryError>;
}

pub struct WorkflowContext { /* opaque; see §11.6 for concurrency semantics */ }
impl WorkflowContext {
    pub fn now(&self) -> SystemTime;
    pub fn random(&self) -> u64;
    pub async fn sleep(&self, d: Duration);
    pub async fn execute_activity<A: Activity>(&self, args: A::Input) -> Result<A::Output, ActivityError>;
    pub fn schedule_compensation<A: Activity>(&self, args: A::Input);
    pub async fn await_signal(&self, name: &str) -> Vec<u8>;
    pub fn get_version(&self, change_id: &str, min: u32, max: u32) -> u32;
    pub async fn side_effect<T>(&self, marker_name: &str, f: impl FnOnce() -> T) -> T;
    pub fn continue_as_new(&self, input: Vec<u8>) -> !;
}

pub trait Activity: Send + 'static {
    type Input:  serde::de::DeserializeOwned + Send + 'static;
    type Output: serde::Serialize + Send + 'static;
    fn name() -> &'static str;
    async fn execute(ctx: &ActivityContext, input: Self::Input) -> Result<Self::Output, ActivityError>;
}

pub struct ActivityContext { /* opaque */ }
impl ActivityContext {
    pub fn activity_id(&self) -> ActivityId;
    pub fn attempt(&self) -> u32;
    pub async fn heartbeat(&self, details: Vec<u8>);
    pub fn deadline(&self) -> SystemTime;
}
