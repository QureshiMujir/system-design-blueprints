Core schema & types (§11)

// render-dom
pub struct Document { pub root: NodeId, nodes: Arena<Node>, pub quirks: QuirksMode }
pub struct Node { pub data: NodeData, pub parent: Option<NodeId>, pub children: Vec<NodeId>, pub dirty: DirtyBits }
pub enum NodeData { Document, Doctype{name:Atom}, Element(Element), Text(String), Comment(String) }
pub struct Element { pub name: QualName, pub attrs: Vec<Attr>, pub style: Option<ComputedStyleId> }
pub enum QuirksMode { NoQuirks, LimitedQuirks, Quirks }
// render-css
pub struct StyleSheet { pub rules: Vec<CssRule>, pub origin: Origin }
pub enum CssRule { Style(StyleRule), Media(MediaRule), Import(ImportRule), FontFace(FontFaceRule) }
pub struct StyleRule { pub selectors: SelectorList, pub block: Vec<Declaration> }
pub struct Declaration { pub property: PropertyId, pub value: ComputedValue, pub important: bool }
pub enum Origin { UserAgent, User, Author }
// render-style
pub struct ComputedStyle { /* one resolved value per property */ }
pub enum ComputedValue { Keyword(Atom), Length(Au), Percentage(f32), Color(Rgba), Number(f32), Auto }
pub struct StyleCache { /* shares ComputedStyle across equal-input elements, I7 */ }
pub fn cascade(el: &Element, rules: &MatchedRules, parent: &ComputedStyle) -> ComputedStyle;
// render-layout
pub struct Au(pub i32);                                   // 1px = 60 Au
pub struct Rect { pub x: Au, pub y: Au, pub w: Au, pub h: Au }
pub struct BoxTree { pub root: BoxId, boxes: Arena<LayoutBox> }
pub struct LayoutBox { pub kind: BoxKind, pub style: ComputedStyleId, pub children: Vec<BoxId> }
pub enum BoxKind { BlockContainer, InlineBox, AnonymousBlock, AnonymousInline, Flex, Text(TextRun), Replaced }
pub enum FormattingContext { Block, Inline, Flex }
pub struct FragmentTree { pub root: FragmentId, frags: Arena<Fragment> }
pub struct Fragment { pub rect: Rect, pub content: FragmentContent, pub style: ComputedStyleId }
pub enum FragmentContent { Box, Line(LineBox), Text(GlyphRun), Image }
pub fn layout(box_tree: &BoxTree, viewport: Size) -> FragmentTree;
// render-paint
pub struct DisplayList { pub items: Vec<DisplayItem> }
pub enum DisplayItem { Rect{rect:Rect,color:Rgba}, Border{rect:Rect,sides:BorderSides},
    Text{run:GlyphRun,color:Rgba}, Image{rect:Rect,image:ImageId}, PushClip(Rect), PopClip }
pub struct StackingContext { pub z_index: i32, pub children: Vec<StackingContext>, pub items: Vec<usize> }
pub fn build_display_list(frags: &FragmentTree) -> DisplayList;
// render-compositor + render-raster + render-engine
pub struct LayerTree { pub root: LayerId, layers: Arena<Layer> }
pub struct Layer { pub id: LayerId, pub bounds: Rect, pub transform: Transform, pub opacity: f32,
    pub items: Vec<DisplayItem>, pub children: Vec<LayerId> }
pub fn composite(layers: &LayerTree, canvas: &mut Canvas);
pub struct Canvas { pub w: u32, pub h: u32, pub pixels: Vec<Rgba> }
pub trait RasterBackend { fn fill_rect(&mut self, r: Rect, c: Rgba);
    fn fill_path(&mut self, p: &Path, c: Rgba); fn blit_glyph(&mut self, g: &GlyphBitmap, at: Point); }
pub struct RenderEngine { /* document, stylesheets, lifecycle, scheduler, canvas */ }
pub enum DocumentLifecycle { Loading, StyleClean, LayoutClean, PrePaintClean, PaintClean, Composited }
pub struct DirtyBits(u32);   // NEEDS_STYLE | NEEDS_LAYOUT | NEEDS_PAINT | NEEDS_COMPOSITE
pub enum InvalidationKind { Style, Layout, Paint, Composite }
impl RenderEngine { pub fn render_frame(&mut self) -> &Canvas; }
