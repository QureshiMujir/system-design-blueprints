Core schema & types (§11)

struct FactId(u64);            // stable identity across modify where possible
struct WmeId(u32);             // arena index of a working-memory element
enum Value { Int(i64), Float(f64), Str(Box<str>), Bool(bool), Sym(SymId) }
struct Fact { id: FactId, type_id: TypeId, slots: Box<[Value]> }   // SoA per template possible
struct Wme  { id: WmeId, fact_id: FactId, type_id: TypeId }

trait WorkingMemory {
    fn assert_fact(&mut self, f: Fact) -> WmeId;        // returns WME; routes into alpha net via callback
    fn retract(&mut self, w: WmeId) -> Result<(), WmError>;
    fn modify(&mut self, w: WmeId, slots: SlotPatch) -> Result<WmeId, WmError>; // atomic retract+assert
    fn get(&self, w: WmeId) -> Option<&Fact>;
    fn by_type(&self, t: TypeId) -> impl Iterator<Item = WmeId>;
}
struct Var(SymId);
enum Op { Eq, Ne, Lt, Le, Gt, Ge, In, Matches }
enum Constraint { Const { slot: SlotId, op: Op, val: Value },     // intra-condition (alpha)
                  Bind  { slot: SlotId, var: Var },               // binding / join key
                  Test  { expr: Expr } }                          // predicate over bound vars
enum Condition { Pattern { type_id: TypeId, cs: Vec<Constraint>, bind: Option<Var> },
                 Not(Box<Condition>), Exists(Box<Condition>),
                 Ncc(Vec<Condition>) }                            // negated conjunction
enum Action { Assert(PatternTemplate), Retract(Var), Modify(Var, Vec<(SlotId, Value)>),
              LogicalAssert(PatternTemplate), Print(Expr), Halt }
struct Rule { name: Box<str>, salience: i32, lhs: Vec<Condition>, rhs: Vec<Action> }
struct NodeId(u32); struct AlphaMemId(u32); struct BetaMemId(u32); struct TokenId(u32);

struct AlphaNode { test: Constraint, parent: NodeId, mem: AlphaMemId }    // one const test
struct AlphaMemory { wmes: Vec<WmeId>, successors: Vec<NodeId> }          // WMEs passing tests

struct Token { id: TokenId, parent: Option<TokenId>, wme: Option<WmeId>, node: BetaMemId }
struct BetaMemory { tokens: Vec<TokenId>, children: Vec<NodeId> }

struct JoinNode { left: BetaMemId, right: AlphaMemId, tests: Vec<JoinTest>, children: Vec<NodeId> }
struct JoinTest { left_field: FieldRef, right_slot: SlotId, op: Op }      // inter-condition test
struct NegationNode { left: BetaMemId, right: AlphaMemId, tests: Vec<JoinTest> /* + match counters */ }
struct TerminalNode { rule_id: RuleId, parent: BetaMemId }

enum ActivationSource { Left(TokenId), Right(WmeId) }                     // activation provenance
struct ReteNetwork { /* arenas of nodes/memories/tokens; root OTNs by TypeId */ }
impl ReteNetwork {
    fn assert_wme(&mut self, w: WmeId, t: TypeId, ag: &mut Agenda, tms: &mut Tms);   // right-activate alpha→joins
    fn retract_wme(&mut self, w: WmeId, ag: &mut Agenda, tms: &mut Tms);             // remove tokens/activations (I3)
    fn left_activate(&mut self, node: NodeId, tok: TokenId);
    fn right_activate(&mut self, node: NodeId, w: WmeId);
}
struct RuleId(u32);
struct Activation { rule_id: RuleId, token: TokenId, bindings: Bindings,
                    salience: i32, recency: u64, seq: u64 }
trait ConflictResolutionStrategy { fn order_key(&self, a: &Activation) -> OrderKey; } // total order
struct Agenda { heap: BinaryHeap<OrderKey>, refraction: HashSet<RefractionKey>, strategy: Box<dyn ConflictResolutionStrategy> }
struct RefractionKey { rule_id: RuleId, fact_sig: FactSignature }   // identical tuple = no re-fire (I2)
enum SessionState { Idle, Matching, AgendaPopulated, Firing, Quiescent, Halted, Aborted }
struct KnowledgeBase { rules: Vec<Rule>, network: ReteNetwork, version: SchemaVersion }
struct Session<'kb> { kb: &'kb KnowledgeBase, wm: WorkingMemory, agenda: Agenda,
                      tms: Tms, state: SessionState, governor: Governor }
impl Session<'_> {
    fn assert(&mut self, f: Fact) -> WmeId;
    fn retract(&mut self, w: WmeId) -> Result<(), EngineError>;
    fn modify(&mut self, w: WmeId, p: SlotPatch) -> Result<WmeId, EngineError>;
    fn fire(&mut self) -> Result<Option<RuleId>, EngineError>;       // one activation
    fn fire_all(&mut self) -> Result<FireReport, EngineError>;       // until quiescent / fire_limit
    fn query(&self, goal: &Pattern) -> Vec<Bindings>;                // backward chaining (Phase 10)
}
struct Tms { /* fact -> set<Justification>; Justification -> set<WmeId supports> */ }
struct Governor { max_tokens: u64, max_agenda: u64, max_wm: u64, fire_limit: u64 /* + counters */ }
enum EngineError { Wm(WmError), Parse(Diagnostic), Capacity(CapKind), InvariantViolation(InvKind), Halted }
enum WmError { NoSuchWme(WmeId), TypeMismatch, SlotOutOfRange }
enum InvKind { MatchIncompleteness, RefractionBreach, RetractionInconsistency,
               DanglingToken(TokenId), NodeSharingDivergence, UnsupportedJustified }
enum CapKind { Tokens, Agenda, WorkingMemory, FireLimit }
struct Diagnostic { code: &'static str, span: Span, message: String }  // parser/semantic errors
