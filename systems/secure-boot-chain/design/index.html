Core schema & types (§11)

struct BootManifest
  version: u32
  manifest_id: Uuid
  build_epoch: u64
  rollback_index: u64
  stages: Vec<StageEntry>
  measurements: Vec<MeasurementRecord>
  signatures: Vec<SignatureBlock>
  compatibility_matrix: CompatibilityMatrix
struct StageEntry
  stage_id: u32
  artifact_digest: Digest256
  load_address: u64
  artifact_size: u64
  compression: CompressionType
  required_pcr: u8
struct VerificationContext
  trust_anchor: KeyHandle
  active_policy: PolicyHandle
  measurement_sink: MeasurementSink
  rollback_guard: RollbackGuard
struct MeasurementRecord
  timestamp_ns: u64
  pcr_index: u8
  digest: Digest256
  event_type: EventType
struct RecoveryDescriptor
  recovery_slot: SlotId
  policy_mode: RecoveryPolicy
  operator_required: bool
struct RollbackCounter
  epoch: u64
  value: u64
  crc32: u32
struct AttestationQuote
  nonce: [u8; 32]
  pcr_selection: PcrSelection
  quote_bytes: Vec<u8>
  signature: SignatureBlock
struct TelemetryEvent
  seq: u64
  span_id: Uuid
  field: TelemetryField
  severity: Severity
type Digest256 = [u8; 32]
type CompatibilityMatrix = Vec<AbiVersionRange>
struct AbiVersionRange { min_kernel_abi: u32, max_kernel_abi: u32 }
enum CompressionType { None, Zstd, Lz4 }
enum EventType { StageLoad, SignatureVerify, PcrExtend, RollbackCheck, RecoveryEnter }
enum SlotId { A, B }
enum RecoveryPolicy { Restricted, ForensicOnly }
enum Severity { Info, Warn, Critical }
