Core schema & types (§11)

#[derive(Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug)]
pub struct Sym(pub u32);

pub struct SymInterner {
    map: HashMap<Arc<str>, Sym>,
    vec: Vec<Arc<str>>,
}

impl SymInterner {
    pub fn intern(&mut self, s: &str) -> Sym;          // idempotent; O(1) amortized
    pub fn resolve(&self, sym: Sym) -> Option<&str>;   // O(1); None if out of range
    pub fn len(&self) -> usize;
}
#[derive(Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug)]
pub struct PredId(pub u32);

pub struct PredInterner {
    map: HashMap<Arc<str>, PredId>,
    vec: Vec<Arc<str>>,
}

impl PredInterner {
    pub fn intern(&mut self, name: &str) -> PredId;
    pub fn resolve(&self, id: PredId) -> Option<&str>;
}
#[derive(Clone, Eq, PartialEq, Hash, Debug)]
pub enum Term {
    Const(Sym),
    Var(u32),   // variable index; scoped to a single Rule
}
#[derive(Copy, Clone, Eq, PartialEq, Debug)]
pub enum CmpOp { Lt, Le, Eq, Ne, Ge, Gt }
#[derive(Clone, Eq, PartialEq, Hash, Debug)]
pub struct Atom {
    pub pred: PredId,
    pub args: Vec<Term>,
}

impl Atom {
    pub fn arity(&self) -> usize;
    pub fn is_ground(&self) -> bool;       // true iff all args are Const
    pub fn apply(&self, s: &Substitution) -> Option<Atom>; // returns None if var unbound
}
#[derive(Clone, Eq, PartialEq, Hash, Debug)]
pub enum Literal {
    Pos(Atom),
    Neg(Atom),
    Cmp(Term, CmpOp, Term),
}

impl Literal {
    pub fn is_positive(&self) -> bool;
    pub fn atom(&self) -> Option<&Atom>;   // None for Cmp
    pub fn pred(&self) -> Option<PredId>;
}
#[derive(Clone, Eq, PartialEq, Debug)]
pub struct Rule {
    pub head: Atom,
    pub body: Vec<Literal>,
}

impl Rule {
    pub fn head_pred(&self) -> PredId;
    pub fn positive_body_preds(&self) -> impl Iterator<Item = PredId> + '_;
    pub fn negative_body_preds(&self) -> impl Iterator<Item = PredId> + '_;
    pub fn is_fact(&self) -> bool;         // body is empty
}
#[derive(Copy, Clone, Eq, PartialEq, Debug)]
pub enum PredKind { Edb, Idb }

#[derive(Clone, Eq, PartialEq, Debug)]
pub struct Predicate {
    pub id: PredId,
    pub arity: usize,
    pub kind: PredKind,
}
pub type Tuple = Vec<Sym>;  // length == relation arity; invariant enforced at insert
#[derive(Clone, Default, Debug)]
pub struct Substitution(pub HashMap<u32, Sym>);

impl Substitution {
    pub fn bind(&mut self, var: u32, val: Sym) -> bool;  // false if conflict
    pub fn get(&self, var: u32) -> Option<Sym>;
    pub fn apply_atom(&self, atom: &Atom) -> Option<Tuple>; // None if var unbound
    pub fn merge(&self, other: &Substitution) -> Option<Substitution>; // None on conflict
}
pub struct Index {
    pub key_cols: Vec<usize>,                        // which argument positions form the key
    pub map: HashMap<Vec<Sym>, Vec<Tuple>>,          // key -> matching tuples
}

impl Index {
    pub fn new(key_cols: Vec<usize>) -> Self;
    pub fn insert(&mut self, tuple: &Tuple);
    pub fn lookup(&self, key: &[Sym]) -> &[Tuple];  // empty slice if absent
    pub fn build_from(tuples: &HashSet<Tuple>, key_cols: Vec<usize>) -> Self;
}
pub struct Relation {
    pub pred: PredId,
    pub arity: usize,
    pub tuples: HashSet<Tuple>,
    pub indexes: HashMap<Vec<usize>, Index>,  // key = sorted col-set
}

impl Relation {
    pub fn new(pred: PredId, arity: usize) -> Self;
    pub fn insert(&mut self, t: Tuple) -> bool;        // true = new; false = duplicate
    pub fn contains(&self, t: &Tuple) -> bool;
    pub fn get_or_build_index(&mut self, cols: Vec<usize>) -> &Index;
    pub fn len(&self) -> usize;
    pub fn iter(&self) -> impl Iterator<Item = &Tuple>;
}
pub struct DeltaRelation {
    pub stable: Relation,
    pub delta: Relation,
    pub next_delta: Relation,   // accumulates new tuples before swap
}

impl DeltaRelation {
    pub fn new(pred: PredId, arity: usize) -> Self;
    pub fn insert_new(&mut self, t: Tuple) -> bool;  // into next_delta if not in stable
    pub fn promote(&mut self);                        // stable |= delta; delta = next_delta; next_delta = empty
    pub fn delta_is_empty(&self) -> bool;
    pub fn stable_contains(&self, t: &Tuple) -> bool;
}
pub struct DependencyGraph {
    // nodes: PredId; edges carry EdgeKind
}

#[derive(Copy, Clone, Eq, PartialEq, Debug)]
pub enum EdgeKind { Positive, Negative, Aggregate }

impl DependencyGraph {
    pub fn new() -> Self;
    pub fn add_edge(&mut self, from: PredId, to: PredId, kind: EdgeKind);
    pub fn has_neg_cycle(&self) -> bool;    // true = not stratifiable
    pub fn stratify(&self) -> Result<Vec<Vec<PredId>>, StratifyError>;
    pub fn successors(&self, p: PredId) -> impl Iterator<Item = (PredId, EdgeKind)> + '_;
}
pub type Stratum = Vec<PredId>;   // ordered list of predicates in one stratum level
pub struct Program {
    pub predicates: HashMap<PredId, Predicate>,
    pub edb: HashMap<PredId, Relation>,
    pub rules: Vec<Rule>,
    pub strata: Vec<Stratum>,    // computed by DependencyGraph::stratify
}

impl Program {
    pub fn build(facts: Vec<Rule>, rules: Vec<Rule>) -> Result<Self, BuildError>;
    pub fn rules_for_pred(&self, p: PredId) -> impl Iterator<Item = &Rule>;
    pub fn stratum_of(&self, p: PredId) -> Option<usize>;
    pub fn idb_preds(&self) -> impl Iterator<Item = PredId> + '_;
    pub fn edb_preds(&self) -> impl Iterator<Item = PredId> + '_;
}
pub struct Query {
    pub goal: Atom,   // may contain Var terms for free variables
}

pub struct QueryResult {
    pub answers: Vec<Substitution>,   // one per matching ground tuple
}

impl Query {
    pub fn new(goal: Atom) -> Self;
    pub fn execute(&self, model: &HashMap<PredId, Relation>) -> QueryResult;
}
#[derive(Clone, Eq, PartialEq, Hash, Debug)]
pub enum Binding { Bound, Free }

#[derive(Clone, Eq, PartialEq, Hash, Debug)]
pub struct Adornment(pub Vec<Binding>);   // length == arity

#[derive(Clone, Eq, PartialEq, Hash, Debug)]
pub struct AdornedPred {
    pub pred: PredId,
    pub adornment: Adornment,
}
pub struct MagicTransformResult {
    pub transformed_program: Program,
    pub magic_seed: Vec<Rule>,   // seed facts for magic predicates derived from query
    pub applicable: bool,        // false = fell back to full eval
}

pub fn magic_transform(program: &Program, query: &Query) -> MagicTransformResult;
#[derive(Copy, Clone, Eq, PartialEq, Debug)]
pub enum AggOp { Count, Sum, Min, Max }

#[derive(Clone, Debug)]
pub struct AggRule {
    pub head_pred: PredId,
    pub group_vars: Vec<u32>,      // variables forming the group key
    pub agg_op: AggOp,
    pub agg_var: u32,              // variable being aggregated over
    pub input_pred: PredId,        // fully computed in lower stratum
}
