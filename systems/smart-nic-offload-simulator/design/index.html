Core schema & types (§11)

// snic-sim — discrete-event core
pub struct VirtualClock { now_ns: u64 }                  // monotone (I2); the ONLY time source
pub struct SimEngine { clock: VirtualClock, heap: BinaryHeap<Reverse<Event>>,
                       next_seq: u64, rng: ChaCha20Rng }  // single-writer; owns the run (ADR-004)
pub struct Event { time: u64, seq: u64, kind: EventKind } // Ord by (time, seq) (I4); owns its payload refs
pub enum EventKind {                                      // exhaustive (principle 14)
  HostDoorbell { vq: VqId },
  DmaComplete { dma: DmaId, dir: DmaDir },
  RxArrive { pkt: Box<Packet> },
  PipelineStage { pkt: PktId, stage: u16 },
  EngineDone { engine: EngineId, pkt: PktId, cost: CycleCost },
  CoreDispatch { core: CoreId, pkt: PktId },
  CoreDone { core: CoreId, pkt: PktId },
  WireTx { pkt: PktId },
  Interrupt { vq: VqId },
  TimerFired { timer: TimerId },
}
pub struct CycleCost(pub u64);                            // host cycles modeled-saved unit
pub enum DmaDir { HostToNic, NicToHost }

// snic-types — packet + headers
pub struct Packet { meta: Meta, headers: HeaderStack, payload: Bytes, len: u32, ts: u64 }
                                                          // owned by the in-flight stage; Bytes refcounted (ADR-007)
pub struct HeaderStack { items: Vec<(Header, u32 /*offset*/)>, parse_err: Option<ParseError> }
pub enum Header { Eth(Eth), Vlan(Vlan), Ipv4(Ipv4), Ipv6(Ipv6),
                  Tcp(Tcp), Udp(Udp), Vxlan(Vxlan), Geneve(Geneve), Gre(Gre), Esp(Esp) }
pub struct Meta { rx_queue: Option<QueueId>, tx_queue: Option<QueueId>, mark: u32,
                  csum_dirty: bool, flow_hash: Option<u32>, core: Option<CoreId> }
pub struct QueueId(pub u16);  pub struct CoreId(pub u16);  pub struct PktId(pub u64);
pub struct EngineId(pub u16); pub struct VqId(pub u16);    pub struct DmaId(pub u64);
pub struct TimerId(pub u64);  pub struct CounterId(pub u32);

// snic-pipeline — match-action
pub struct MatchActionTable { kind: TableKind, entries: TableEntries, caps: TableCaps }
pub enum TableKind { Exact, Lpm, Ternary }                // lookup semantics + tie-break (I8)
pub struct TableEntry { key: KeyValue, mask: Option<KeyMask>, prio: u32, actions: Vec<Action> }
pub enum Action { SetField(FieldRef, u64), Fwd(QueueId), Drop, Count(CounterId),
                  Encap(TunnelConfig), Decap, ToCore(CoreId), Mark(u32), Mirror(QueueId) }
pub struct Parser { graph: ParserGraph, max_vlan_depth: u8 }
pub struct Deparser;                                      // re-serialize dirty headers + payload

// snic-offload
pub trait OffloadEngine {                                 // engines hold only config; no per-packet alloc on hot path
  fn process(&mut self, pkt: &mut Packet, ctx: &mut EngineCtx) -> (PktDisposition, CycleCost);
}
pub enum PktDisposition { Pass, Drop(DropReason), Steer(QueueId),
                          SegmentedInto(SmallVec<[Box<Packet>; 8]>), CoalescedInto(Box<Packet>),
                          ToCore(CoreId), Mark }
pub struct RssConfig { key: [u8; 40], indirection: Vec<QueueId>, hash_tuple: HashTuple }
pub struct FlowRule { pred: Vec<Pred>, action: Action }
pub struct TunnelConfig { kind: TunnelKind, outer_src: IpAddr, outer_dst: IpAddr, vni: u32 }
pub enum TunnelKind { Vxlan, Geneve, Gre }

// snic-host
pub struct HostInterface { rx_vqs: Vec<VirtQueue>, tx_vqs: Vec<VirtQueue>, dma: DmaEngine, msix: Msix }
pub struct VirtQueue { layout: VqLayout, desc: Vec<Descriptor>, avail_idx: u16, used_idx: u16 }
pub enum VqLayout { Split, Packed }
pub struct Descriptor { addr: u64, len: u32, flags: u16, next: u16 }
pub struct DmaEngine { latency_ns: u64, bw_bps: u64, inflight: u32, cap: u32 }  // I12: inflight ≤ cap
pub struct Msix { coalesce_n: u16, coalesce_ns: u64, pending: u16 }

// snic-cores
pub struct EmbeddedCore { id: CoreId, cycle_budget: u64, occupancy: u64, queue: VecDeque<PktId> }
pub struct OffloadProgram { insns: Vec<Insn>, max_steps: u32, cycle_cost_base: u32 }  // bounded (I10)
pub enum Insn { LoadField(u8, FieldRef), LoadImm(u8, u64), Cmp(u8, u8),
                Jmp(u16 /*forward only*/), JmpIf(Cond, u16), SetField(FieldRef, u8),
                Emit(PktDisposition), AddCycles(u32), Halt }
pub struct Verifier;                                      // static: bounded, in-range, no backward jump (I10)

// snic-rdma
pub struct QueuePair { qpn: u32, send_q: VecDeque<WorkReq>, recv_q: VecDeque<WorkReq>, cq: CqId }
pub struct CompletionQueue { id: CqId, entries: VecDeque<Completion> }
pub struct MemoryRegion { lkey: u32, rkey: u32, addr: u64, len: u64 }
pub enum Verb { RdmaWrite, RdmaRead, Send, Recv }

// snic-telemetry
pub struct Counters { map: HashMap<&'static str, u64> }   // O(1) update, always on (principle 15)
pub struct LatencyHist { hdr: hdrhistogram::Histogram<u64> }
pub struct CpuSavingsModel { per_engine: HashMap<EngineId, u64>, total: u64 }  // non-negative, conserved (I11)
pub struct Throughput { pkts: u64, bytes: u64, window_ns: u64 }

// errors (thiserror; no panics on packet path — ADR-027)
pub enum SnicError { Parse(ParseError), Table(TableError), Offload(OffloadError),
                     Host(HostError), Core(CoreError), Config(ConfigError), Rdma(RdmaError) }
pub enum ParseError { Truncated, UnknownEtherType, VlanDepthExceeded, BadHeaderLen }
pub enum OffloadError { BadGso, BadChecksumInput, TunnelDepthExceeded, EspBadFrame }
pub enum CoreError { ProgramRejected(String), StepCapHit, FieldOutOfRange }
