Core schema & types (§11)

struct Token
  kind: TokenKind;
  span: Span;
  trivia: TriviaId;
// ownership: arena-allocated in lexer::arena; immutable after lexing (P4, I11)
// mutation: none post-construction
// concurrency: shared read-only across parser via index, never by pointer
// persistence: never independently serialized
// serialization: n/a
// compatibility: internal-only, not wire-visible
enum PatternNode
  Wildcard;
  Literal(LiteralId);
  Constructor(ConstructorId, Vec<PatternId>);
  Tuple(Vec<PatternId>);
  Slice(Vec<PatternId>, RestId);
  Range(LiteralId, LiteralId);
  Binding(SymbolId);
  Or(Vec<PatternId>);
  Guard(PatternId, GuardId);
// ownership: arena-owned in ast crate, referenced by PatternId index (P4)
// mutation: immutable after parse; normalization produces new nodes, never mutates in place
// concurrency: read-shared across type-resolution and normalization stages
// persistence: snapshotted post-normalization only (pattern-ir), not at this stage
// serialization: stable discriminant ordering (additive-only evolution, I7)
// compatibility: variant order is append-only across releases
struct PatternMatrix
  rows: Vec<Row>;
  columns: Vec<Column>;
// ownership: matrix-compiler owns; specialization produces new immutable matrices (I1)
// mutation: none — specialization/default-matrix construction always allocates fresh
// concurrency: immutable snapshots; parallel specialization permitted across constructor branches (ADR-08)
// persistence: not persisted directly; witness sets derived from it are logged for replay
// serialization: n/a (in-memory only)
// compatibility: row/column shape is internal, not a stability surface
enum DecisionNode
  Test(ConstructorId);
  Guard(GuardId);
  Bind(SymbolId);
  Success(ArmId);
  Failure;
// ownership: decision-tree crate owns the DAG arena (I6 acyclicity enforced at construction)
// mutation: optimizer passes rewrite by producing a new DAG, never in-place graph surgery
// concurrency: read-only DAG shared with both backends for differential codegen
// persistence: optionally snapshotted for DOT-graph export and replay correlation
// serialization: stable discriminant, additive-only
// compatibility: internal IR, not a wire-stability surface
struct Diagnostic
  code: DiagnosticCode;
  severity: Severity;
  primary_span: Span;
  secondary_spans: Vec<Span>;
// ownership: diagnostics crate owns; borrowed by callers for rendering only
// mutation: immutable once constructed
// concurrency: collected into a deterministically-ordered Vec, never an unordered set (P3, I8)
// persistence: snapshotted in fixture .snap files for regression testing
// serialization: stable DiagnosticCode enum, additive-only
// compatibility: DiagnosticCode values are a public stability surface post-v1.0 (ADR-18)
struct ReplayEvent
  timestamp: LogicalClock;
  stage: CompilerStage;
  payload_hash: u128;
// ownership: replay crate owns the event log arena
// mutation: append-only log, never mutated post-write
// concurrency: single-writer per compilation unit; checksum-covered end to end (I10)
// persistence: persisted to fixtures/replay/*.bin, versioned envelope (ADR-05, I7)
// serialization: postcard, versioned envelope, N-2 backward-readable
// compatibility: payload_hash algorithm is a locked stability surface — changing it
//   breaks replay of every previously-recorded trace
struct CompilerConfig
  max_concurrent_compilations: u32;
  branch_depth_limit: u32;
  exhaustiveness_memory_ceiling_bytes: u64;
  replay_seed: Option<u64>;
// ownership: cli crate owns, immutable for the lifetime of one compilation invocation
// mutation: none post-load
// concurrency: read-shared across all stages of one invocation
// persistence: loaded from CLI flags/config file, not itself persisted
// serialization: TOML on disk, in-memory struct otherwise
// compatibility: additive fields only, defaulted for backward config compatibility
