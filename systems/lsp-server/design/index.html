Core schema & types (§11)

pub struct Position { pub line: u32, pub character: u32 }   // wire type; unit
                                                             // depends on negotiated
                                                             // PositionEncodingKind
pub struct Range { pub start: Position, pub end: Position } // half-open [start,end)

pub struct ByteRange { pub start: u32, pub end: u32 }        // internal-only; UTF-8 bytes

pub struct PositionConverter;   // per-document; line_index + optional utf16_index
// fn to_byte_range(Range, encoding: PositionEncodingKind) -> Result<ByteRange, PosError>;
// fn to_wire_position(u32 /*byte offset*/, encoding) -> Position;
pub struct TextDocument {
    pub uri: DocumentUri, pub language_id: LanguageId,
    pub version: DocumentVersion, pub text: Arc<String>,
    pub line_index: Arc<LineIndex>, pub utf16_index: Option<Arc<Utf16Index>>,
}
pub struct DocumentUri(pub String);          // #[serde(transparent)]
pub struct LanguageId(pub &'static str);
pub struct DocumentVersion(pub i32);         // Ord; is_stale(self, current) -> bool
pub struct FileId(pub u32);                  // interned per workspace (I14)
pub enum TextDocumentChangeEvent {
    Incremental { range: Range, range_length: Option<u32>, text: String },
    Full { text: String },
}
pub struct TextEdit { pub range: Range, pub new_text: String }
pub struct AnalysisDb {
    revision: AtomicU64,                       // I1: monotonic global clock
    memo: DashMap<QueryKey, MemoEntry>,         // I2
    inputs: DashMap<FileId, InputEntry>,
    cancel: Arc<AtomicBool>,                    // I4
}
pub enum Durability { Low, Medium, High }
pub struct MemoEntry {
    result: Arc<dyn Any + Send + Sync>, deps: SmallVec<[QueryKey; 4]>,
    computed_at: u64, verified_at: u64, durability: Durability,
}
pub trait Query: 'static {
    type Key: Hash + Eq + Clone; type Value: Clone + Send + Sync;

}
pub struct GreenNode { pub kind: SyntaxKind, pub text_len: TextSize,
                        pub children: Arc<[GreenChild]> }
pub enum GreenChild { Node(Arc<GreenNode>), Token(Arc<GreenToken>) }
pub struct GreenToken { pub kind: SyntaxKind, pub text: SmolStr }
pub struct SyntaxKind(pub u16);
pub struct SyntaxNode { green: Arc<GreenNode>, parent: Option<Arc<SyntaxNode>>,
                         offset: TextSize }
pub struct AstId { pub file: FileId, pub kind: SyntaxKind, pub index: u32 } // I8
pub struct NameResolution {
    pub references: HashMap<ReferenceId, Option<DeclarationId>>,
    pub declarations: HashMap<DeclarationId, SmallVec<[ReferenceId; 4]>>,
    pub imports: SmallVec<[(ImportSyntax, Option<ModuleId>); 2]>,
    pub errors: SmallVec<[ResolutionError; 0]>,
}
pub struct ReferenceId(pub AstId); pub struct DeclarationId(pub AstId);
pub struct ModuleId(pub u32);

pub struct TypeContext {
    pub expressions: HashMap<ExprId, TypeId>, pub patterns: HashMap<PatId, TypeId>,
    pub locals: HashMap<LocalId, TypeId>,
    pub diagnostics: SmallVec<[TypeDiagnostic; 0]>,
    pub complete: bool,                         // P5 partial-information marker
}
pub struct TypeId(pub u32);
pub struct CompletionContext {
    pub file: FileId, pub position: ByteRange, pub trigger: CompletionTrigger,
    pub prefix: String, pub kind: CompletionKind, pub node: SyntaxNode,
    pub scope: ScopeId,
}
pub enum CompletionTrigger { Invoked, Character(char), Retrigger }
pub enum CompletionKind { Identifier, DotAccess, PathSegment, Type, Pattern,
                           Attribute, String, Comment }
pub struct SemanticTokens { pub result_id: Option<String>, pub data: Vec<u32> }
pub struct TokenSet { pub tokens: Vec<SemanticToken> }
pub struct SemanticToken { pub range: ByteRange, pub kind: TokenType,
                            pub modifiers: TokenModifiers }
pub enum TokenType { Namespace, Type, Class, Enum, Interface, Struct,
                      TypeParameter, Parameter, Variable, Property, EnumMember,
                      Event, Function, Method, Macro, Keyword, Modifier,
                      Comment, String, Number, Regexp, Operator, Decorator }
pub struct TokenModifiers(pub u32);            // bitmask
pub struct WorkspaceEdit {
    pub changes: Option<HashMap<DocumentUri, Vec<TextEdit>>>,
    pub document_changes: Option<Vec<DocumentChange>>,
    pub change_annotations: Option<HashMap<ChangeAnnotationId, ChangeAnnotation>>,
}
pub enum DocumentChange { Edit(TextDocumentEdit), Create(CreateFile),
                           Rename(RenameFile), Delete(DeleteFile) }
pub struct CodeAction {
    pub title: String, pub kind: Option<CodeActionKind>,
    pub diagnostics: Option<Vec<Diagnostic>>, pub is_preferred: Option<bool>,
    pub edit: Option<WorkspaceEdit>, pub command: Option<Command>,
    pub data: Option<serde_json::Value>,        // two-phase resolve, §19
}
pub struct Diagnostic {
    pub range: Range, pub severity: Option<DiagnosticSeverity>,
    pub code: Option<DiagnosticCode>, pub source: Option<String>,
    pub message: String, pub tags: Option<Vec<DiagnosticTag>>,
    pub related_information: Option<Vec<DiagnosticRelatedInformation>>,
}
pub enum DiagnosticSeverity { Error = 1, Warning = 2, Information = 3, Hint = 4 }

pub trait WorkspaceLoader {                     // §1 delegation boundary
    // fn discover_files(root: &Path) -> Vec<FileId>;   (no body — host-language-specific)
}
pub struct ServerCapabilities {
    pub position_encoding: Option<PositionEncodingKind>,
    pub text_document_sync: Option<TextDocumentSyncCapability>,
    pub completion_provider: Option<CompletionOptions>,
    pub hover_provider: Option<HoverProviderCapability>,
    pub definition_provider: Option<DefinitionCapability>,
    pub references_provider: Option<ReferencesCapability>,
    pub rename_provider: Option<RenameCapability>,
    pub code_action_provider: Option<CodeActionCapability>,
    pub semantic_tokens_provider: Option<SemanticTokensCapability>,
    pub workspace_symbol_provider: Option<WorkspaceSymbolCapability>,
    pub document_formatting_provider: Option<DocumentFormattingCapability>,
    pub experimental: Option<serde_json::Value>,
}
pub enum RequestId { Int(i64), String(String) }   // #[serde(untagged)]
