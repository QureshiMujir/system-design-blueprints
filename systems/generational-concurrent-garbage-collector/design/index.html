Core schema & types (§11)

// gc-object: header and coloring — hot path: mutator field read/write; collection time: mark/sweep/compact
struct ObjectHeader { mark: TriColor, forward: Option<ObjRef>, generation: Generation, age: u8, size_class: u16, type_desc: *const TypeDescriptor }
  // ownership: owned by the heap region containing it; mutation: mark/forward fields mutated only by collector threads under CAS (I9); concurrency: read on mutator hot path (type_desc), written only by collector; persistence: none (in-heap only); serialization: none; compatibility: header layout versioned (§27)

struct ObjRef(*mut ObjectHeader)
  // ownership: not owned, a typed raw pointer; mutation: rewritten on promotion/compaction by collector; concurrency: read on mutator hot path every dereference; hot path: yes (every load)

enum Generation { Nursery, Survivor(u8), Old }
  // ownership: copy type; mutation: set once at allocation, updated on promotion; concurrency: read on both paths; hot path: allocation + promotion

enum TriColor { White, Gray, Black }
  // ownership: copy type; mutation: collector-only transitions White->Gray->Black; concurrency: read by barrier (mutator hot path, incremental-update only), written by marker; hot path: barrier fast path reads it

struct Nursery { bump: BumpPointer, limit: *mut u8 }
  // ownership: owned by gc-alloc; mutation: bump pointer advanced by mutator via TLAB; concurrency: per-thread TLAB avoids contention; hot path: yes, the allocation fast path

struct SurvivorSpace { from: Nursery, to: Nursery, epoch: u64 }
  // ownership: owned by gc-minor; mutation: from/to swapped each minor cycle; concurrency: collector-only during STW copy; hot path: no (collection time only)
// gc-alloc / gc-major: old-gen and allocation structures
struct OldGen { regions: Vec<Region>, free_list: FreeList, occupancy_bytes: u64 }
  // ownership: owned by gc-core; mutation: free_list mutated on alloc/sweep; concurrency: concurrent sweep mutates under region-local locks; hot path: promotion allocation only

struct Space { base: *mut u8, len: usize, kind: Generation }
  // ownership: owned by heap manager; mutation: immutable after creation except occupancy counters; concurrency: read-mostly; hot path: no

struct Region { id: u32, state: RegionState, live_bytes: u32 }
  // ownership: owned by OldGen; mutation: state transitions during sweep/compact; concurrency: single-writer per region during its collection slice; hot path: no

struct MarkStack { local: Vec<ObjRef>, stealable: SegQueue<ObjRef> }
  // ownership: per-marker-thread local + shared stealable segment; mutation: push/pop by owning thread, steal by others; concurrency: lock-free work-stealing (ADR-16); hot path: collection time (marker), not mutator

struct WriteBarrier { mode: BarrierMode }
  // ownership: process-global config; mutation: immutable after startup; concurrency: read on every mutator pointer store; hot path: yes, the single most performance-critical mutator-visible structure (P2)

struct CardTable { bytes: *mut u8, card_shift: u8 }
  // ownership: owned by gc-barrier; mutation: dirtied by mutator barrier (unconditional store), cleared by refinement worker (I12); concurrency: many-writer (mutators), single-writer-per-region (refinement); hot path: yes (dirty), no (clear)

struct RememberedSet { edges: HashSet<(ObjRef, ObjRef)>, epoch: u64 }
  // ownership: owned by gc-barrier, alternative/complement to CardTable (ADR-07); mutation: inserted on barrier fire; concurrency: sharded by region to reduce contention; hot path: yes if precise-RS mode chosen
// gc-barrier / gc-mark: concurrent-marking support — hot path split noted per field
struct Satb { local_buf: Vec<ObjRef>, global_queue: SegQueue<ObjRef> }
  // ownership: per-mutator-thread local buffer, flushed to global queue; mutation: append-only until flush; concurrency: lock-free flush; hot path: yes, SATB barrier fast path (I11)

struct SnapshotBuffer(Satb)
  // ownership: alias/newtype over Satb for the initial-mark snapshot use case; hot path: yes at initial-mark boundary only

struct Allocator { nursery: Nursery, old: OldGen }
  // ownership: owned by gc-core; mutation: delegates to Nursery/OldGen; concurrency: TLAB-partitioned; hot path: yes

struct Tlab { start: *mut u8, cursor: *mut u8, end: *mut u8 }
  // ownership: owned by a single mutator thread; mutation: cursor advanced with no synchronization (single-writer); concurrency: none needed, thread-local; hot path: yes, the fastest path in the system

struct BumpPointer { cursor: AtomicPtr<u8>, limit: *mut u8 }
  // ownership: shared refill source behind Tlab; mutation: CAS-advanced on TLAB refill only (rare); concurrency: contended only at refill; hot path: refill only, not per-allocation

struct FreeList { buckets: [Vec<FreeSpan>; NUM_SIZE_CLASSES] }
  // ownership: owned by OldGen; mutation: pop on alloc, push on sweep; concurrency: per-bucket locks or lock-free stack; hot path: promotion allocation
// gc-core / gc-mark / gc-major: orchestration types
struct Collector { phase: CollectorPhase, config: CollectorConfig }
  // ownership: process-singleton; mutation: phase transitions are the state machine (§8.5); concurrency: phase field read by all threads, written under a single coordinator; hot path: no

struct MinorGc { survivor: SurvivorSpace, promotion: PromotionPolicy }
  // alias: Scavenge
  // ownership: owned by gc-minor; mutation: invoked per minor cycle; concurrency: STW during copy (I7); hot path: no (triggered, not per-allocation)

struct MajorGc { marker: ConcurrentMarker, sweep_state: SweepState }
  // ownership: owned by gc-major; mutation: cycle state advances per §8.5; concurrency: concurrent phases run alongside mutator; hot path: no

struct ConcurrentMarker { workers: Vec<MarkWorker>, gray_stack: MarkStack }
  // ownership: owned by gc-mark; mutation: workers drain gray_stack; concurrency: fully concurrent with mutator (P3); hot path: no (collector-thread-only)

struct ForwardingPointer(AtomicPtr<ObjectHeader>)
  // ownership: embedded in ObjectHeader union; mutation: CAS-published once (I9); concurrency: multiple potential writers race, one wins; hot path: read on every dereference during/after compaction (Phase 10)

struct ReadBarrier { enabled: bool }
  // ownership: process-global config, ADR-11; mutation: immutable after startup; concurrency: read on every mutator load when enabled; hot path: yes, only in concurrent-compaction mode (Phase 10)

struct Safepoint { generation: u64, requested: AtomicBool }
  // ownership: owned by gc-safepoint; mutation: coordinator sets requested, mutators clear on poll; concurrency: many-reader (mutator poll), single-writer (coordinator); hot path: yes, the poll check itself

struct RootSet { threads: Vec<ThreadRoots>, globals: Vec<ObjRef> }
  // ownership: built fresh per cycle by root scanner; mutation: write-once per snapshot; concurrency: collector-only after safepoint convergence; hot path: no

struct PromotionPolicy { tenuring_threshold: u8, humongous_size: usize }
  // ownership: config, owned by gc-minor; mutation: tunable at startup/Phase 11; concurrency: read-only at runtime; hot path: read per-object during scavenge (§8.4 step 6)

struct GcStats { pause_histogram: Histogram, promotion_rate: f64, barrier_overhead_cycles: u64 }
  // ownership: owned by gc-stats; mutation: appended per cycle; concurrency: lock-free counters; hot path: no (sampled)

struct Mutator { tlab: Tlab, satb_buf: Satb, thread_roots: ThreadRoots }
  // ownership: one per application thread; mutation: owns its own fast-path state; concurrency: no cross-thread mutation except by the collector at a safepoint; hot path: yes, entirely

enum GcError { RegionCorruption, InvalidForwardingPointer, BarrierInvariantViolation, AllocationFailure, SafepointTimeout, MarkRaceLost, PromotionFailure }
  // ownership: n/a (error enum); mutation: n/a; concurrency: propagated via Result, never panics across the mutator/collector boundary (P12)
