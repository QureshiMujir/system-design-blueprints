Core schema & types (§11)

struct Vector { dim: u32, data: Vec<f32> }              // owned; validated at construction (I1)
type VectorId = u64;                                      // user-supplied or checked-unique (ADR-16)

enum Metric { L2, Cosine, DotProduct }                     // fixed per collection (P2, I2)

enum VecDbError {
    DimensionMismatch { expected: u32, got: u32 },
    DuplicateId(VectorId),
    NotFound(VectorId),
    CorruptSnapshot { reason: &'static str },
    WalCrcMismatch { seq: u64 },
    NonFiniteVector,
    FilterParseError(String),
}
struct CollectionConfig {                                   // persisted in snapshot Header (P11)
    dim: u32,
    metric: Metric,
    index_kind: IndexKind,                                  // Flat | Hnsw | IvfPq | Lsh
    hnsw_params: Option<HnswParams>,
    ivf_params: Option<IvfParams>,
    lsh_params: Option<LshParams>,
    over_fetch_factor: u32,                                  // metadata-filter over-fetch (P9)
}

trait AnnIndex {                                             // common interface, all index crates (P12)
    fn insert(&mut self, id: VectorId, v: &Vector) -> Result<(), VecDbError>;
    fn search(&self, q: &Vector, k: usize, ef_or_nprobe: u32) -> Vec<SearchResult>;
    fn delete(&mut self, id: VectorId) -> bool;
    fn len(&self) -> usize;
}

struct SearchResult { id: VectorId, distance: f32 }
struct HnswParams { m: u16, m_max0: u16, ef_construction: u32, ef_search_default: u32 }
struct HnswNode { id: VectorId, level: u8, neighbors: Vec<Vec<VectorId>> }  // neighbors[layer] (I8)
struct HnswIndex {
    nodes: HashMap<VectorId, HnswNode>,
    vectors: HashMap<VectorId, Vector>,
    entry: Option<VectorId>,                                  // I7
    max_level: u8,
    tombstones: HashSet<VectorId>,                             // P8
}
struct PqCodebook { m_subquantizers: u16, bits_per_code: u8, centroids: Vec<Vec<Vector>> }
struct PqCode { codes: Vec<u8> }                              // one byte per subquantizer (bits_per_code<=8)

struct IvfParams { nlist: u32, nprobe_default: u32 }
struct IvfList { centroid: Vector, members: Vec<(VectorId, PqCode)> }  // I10
struct IvfIndex { lists: Vec<IvfList>, coarse: Vec<Vector>, pq: PqCodebook }
struct LshParams { num_tables: u16, num_bits: u16, bucket_width: f32 }
struct LshTable { hyperplanes: Vec<Vector>, buckets: HashMap<u64, Vec<VectorId>> }  // I11
struct LshIndex { tables: Vec<LshTable>, vectors: HashMap<VectorId, Vector> }
enum MetadataValue { Int(i64), Float(f64), Str(String), Bool(bool) }
enum MetadataPredicate {                                       // AST for §8 grammar
    Cmp { field: String, op: CmpOp, value: MetadataValue },
    In { field: String, values: Vec<MetadataValue> },
    Not(Box<MetadataPredicate>),
    And(Vec<MetadataPredicate>),
    Or(Vec<MetadataPredicate>),
}
enum CmpOp { Eq, Ne, Lt, Le, Gt, Ge }
struct SnapshotHeader { magic: [u8; 4], version: u32, dim: u32, metric: u8, index_kind: u8, count: u64 }
struct WalRecord { seq: u64, op: WalOp, id: VectorId, vector: Option<Vector>, crc32: u32 }
enum WalOp { Insert, Delete, Checkpoint }

type ShardId = u32;
struct ShardRouter { shard_count: u32, hash_seed: u64 }        // I13, ADR-11

Repository layout (§10)

Vector-Database/
  Cargo.toml                     # workspace manifest, pinned dependency versions (ADR-01)
  crates/
    vecdb-core/                  # Vector, VectorId, Metric, distance kernels, errors (§11, Phase 0)
      src/vector.rs               # Vector newtype, dimension validation
      src/metric.rs               # Metric enum, distance function dispatch
      src/distance.rs             # L2, cosine, dot-product kernels (ADR-02, ADR-20)
      src/error.rs                # VecDbError enum
    vecdb-flat/                  # brute-force exact search, the recall oracle (ADR-17)
      src/flat_index.rs
    vecdb-hnsw/                  # HNSW graph index (Phase 2, Phase 3)
      src/node.rs                 # Node, neighbor lists per layer
      src/index.rs                # HnswIndex, insert/search
      src/select_neighbors.rs     # diversity heuristic (ADR-05)
    vecdb-quant/                 # Product Quantization (Phase 4)
      src/codebook.rs             # PqCodebook, k-means training (ADR-15)
      src/codes.rs                # PqCode, encode/decode
    vecdb-ivf/                   # inverted-file coarse quantizer + IVF-PQ (Phase 4)
      src/coarse.rs                # coarse quantizer / centroids
      src/inverted_list.rs         # IvfList
      src/ivf_index.rs             # IvfIndex, IVF-PQ search (ADC)
    vecdb-lsh/                   # LSH index (Phase 5)
      src/projection.rs            # p-stable random projections (Datar et al. 2004)
      src/lsh_index.rs             # LshIndex, multi-probe (ADR-23)
    vecdb-filter/                # metadata predicate model, hybrid filtering (Phase 6)
      src/predicate.rs             # MetadataPredicate AST, EBNF grammar (§8)
      src/evaluator.rs             # pre/post filter dispatch (ADR-10)
    vecdb-storage/               # WAL, snapshot, incremental build (Phase 7)
      src/wal.rs                   # WalRecord, append/replay (I14)
      src/snapshot.rs              # Header, VectorRecord, atomic write (ADR-21)
      src/collection.rs            # Collection lifecycle state machine (§8)
    vecdb-shard/                 # consistent-hash routing, scatter-gather merge (Phase 8)
      src/router.rs                # ShardRouter (ADR-11, I13)
      src/merge.rs                 # top-k merge across shards (P10)
    vecdb-server/                 # CLI + minimal HTTP inspection/admin API (Phase 0, Phase 9)
      src/cli.rs                   # insert/search/save/load/bench subcommands
      src/http.rs                  # minimal HTTP surface (§35, ADR-18)
  conformance/                   # differential-test fixtures vs vecdb-flat (P1, P12)
    fixtures/                     # *.json fixtures referenced by each phase's N.5
  benchmarks/                    # criterion benches: recall@k vs latency (§28)
  docs/
    adr/                          # ADR-01.md .. ADR-25.md mirroring §9
