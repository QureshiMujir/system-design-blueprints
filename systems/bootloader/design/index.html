Core schema & types (§11)

trait Firmware: Send {
    fn console_init(&mut self);                                    // mut, no persistence
    fn console_print(&mut self, s: &str);                          // mut
    fn read_disk(&mut self, lba: u64, buf: &mut [u8]) -> Result<(), FwError>;   // mut, no persistence beyond call
    fn memory_map(&self) -> MemoryMap;                              // snapshot, immutable return (P7)
    fn exit_boot_services(self, map: &MemoryMap) -> Result<(), FwError>; // consumes self (I2, type-enforced)
}

enum FwError { DiskIo, MapKeyStale, Unsupported, Timeout }          // no payload beyond variant; Copy

struct MemoryMap {                                                  // owned, immutable after snapshot (P7)
    entries: [MemRange; MAX_MEM_RANGES],                            // fixed-capacity, no heap (P11)
    len: u16,
}

struct MemRange { base: u64, length: u64, class: MemClass }         // Copy, serialization: raw struct copy to HandoffState

enum MemClass { Usable, Reserved, AcpiReclaim, AcpiNvs, Bad }       // Copy, stable discriminant values (compat: append-only)

enum KernelImage { Elf(ElfImage), Multiboot2(Mb2Image), BzImage(BzImageDesc) } // owned; tag drives loader dispatch (P12)

struct ElfImage { entry: u64, phdrs: [ProgramHeader; MAX_PHDRS], phdr_count: u16 } // Copy; entry validated against phdrs (I8)

struct ProgramHeader { vaddr: u64, paddr: u64, filesz: u64, memsz: u64, flags: u32 } // Copy

struct Mb2Image { info_addr: u64, tag_count: u16 }                  // owned; tags read lazily, bounds-checked (P3)

struct BzImageDesc { protocol_version: u16, setup_sects: u8, kernel_entry: u64 } // Copy

struct BootEntry {                                                  // owned, config-derived
    title: [u8; MAX_TITLE_LEN],
    kernel_path: [u8; MAX_PATH_LEN],
    initramfs_path: Option<[u8; MAX_PATH_LEN]>,
    cmdline: [u8; MAX_CMDLINE_LEN],
    signature_path: Option<[u8; MAX_PATH_LEN]>,
}

struct HandoffState {                                               // built once, immutable, per-protocol-serializable (P12, I14)
    memory_map: MemoryMap,
    cmdline_len: u16,
    cmdline: [u8; MAX_CMDLINE_LEN],
    initramfs: Option<MemRange>,
    framebuffer: Option<FramebufferInfo>,
    acpi_rsdp: Option<u64>,
    smbios: Option<u64>,
}

struct FramebufferInfo { base: u64, width: u32, height: u32, stride: u32, format: PixelFormat } // Copy

enum PixelFormat { Rgb888, Bgr888 }                                 // Copy; matches GOP query result (Phase 3)

struct BootControlBlock {                                           // 32 bytes, on-disk, ping-pong pair (ADR-06, I5)
    magic: u32,                 // 0x1ABA1ABA
    active_slot: u8,            // 0=A 1=B
    trying_slot: u8,            // 0/1/0xFF
    successful_boots: u8,       // 0xFF = confirmed
    reserved: u8,
    rollback_counter: u32,      // monotonic (I11)
    crc32: u32,                 // over preceding bytes
    padding: [u8; 16],
}

struct BcbSlot { generation: u32, bcb: BootControlBlock }           // on-disk; higher generation + valid CRC wins on read

struct Pkcs7SignedData { signer_cert: CertRef, cert_chain: [CertRef; MAX_CHAIN_DEPTH], signature: [u8; SIG_LEN] } // parsed, transient

struct CertRef { der_offset: u32, der_len: u32 }                    // borrows into the original PKCS#7 buffer, no copy

struct PcrEvent { pcr_index: u8, event_type: u32, digest: [u8; 32], data_len: u16 } // Copy; fed to bl-measure (I7)

trait Verifier { fn verify(&self, image: &[u8], sig: &Pkcs7SignedData) -> Result<(), VerifyError>; } // stateless, no persistence

enum VerifyError { ChainNotTrusted, HashMismatch, Revoked, MalformedAsn1 } // Copy
