Core schema & types (§11)

type Revision struct { Main uint64; Sub uint64 }
func (r Revision) GreaterThan(o Revision) bool

type generation struct {
    Ver       uint64
    Created   Revision
    Tombstone Revision   // zero value = not deleted
    Revisions []Revision
}

type KeyIndex struct {
    Key         []byte
    Modified    Revision
    Generations []generation
}
func (k *KeyIndex) Put(rev Revision) (ver uint64)
func (k *KeyIndex) Get(atRev Revision) (rev Revision, created Revision, ver uint64, ok bool)
func (k *KeyIndex) Tombstone(rev Revision)
type PutRequest struct {
    Key, Value  []byte
    Lease       LeaseID
    PrevKv      bool
    IgnoreValue bool
    IgnoreLease bool
}

type RangeRequest struct {
    Key, RangeEnd     []byte
    Limit             int64
    Revision          int64     // 0 = current
    SortOrder         SortOrder
    SortTarget        SortTarget
    Serializable      bool      // true skips ReadIndex (P1)
    KeysOnly, CountOnly bool
    MinModRevision, MaxModRevision       int64
    MinCreateRevision, MaxCreateRevision int64
}

type KeyValue struct {
    Key, Value      []byte
    CreateRevision  int64
    ModRevision     int64
    Version         int64
    Lease           LeaseID
}
type WatchRequest struct {
    Key, RangeEnd  []byte
    StartRevision  int64
    ProgressNotify bool
    Filters        []EventType
    PrevKv         bool
}

type Event struct { Type EventType; Kv KeyValue; PrevKv *KeyValue }

type LeaseID uint64

type Lease struct {
    ID        LeaseID
    TTL       time.Duration
    Granted   time.Time
    Keys      [][]byte
    LastTouch time.Time
}

type Hub struct {
    // Own: one instance per server process (P3).
    // Mut: watcher set mutated under mu; event delivery is non-blocking (tryEnqueue).
    // Conc: RWMutex-guarded range index; per-watcher bounded channel outbox.
    // Pers: none — rebuilt from client re-subscription on restart.
}
func (h *Hub) Watch(rng KeyRange, start Revision, opts WatchOptions) (Iterator, error)
func (h *Hub) Notify(ev Event)

type Manager struct {
    // Own: one instance per server; follower-replicated map (P4, I5).
    // Mut: writes flow only through Raft-applied LeaseGrant/LeaseRevoke; expiry scan is
    //      leader-only and itself proposes LeaseRevoke rather than mutating locally.
    // Conc: guarded by mu; expiryQ is a min-heap keyed by deadline.
    // Pers: lease bucket + lease-key bucket (§8.4).
}
func (m *Manager) Grant(ttl time.Duration) (LeaseID, error)
func (m *Manager) Attach(id LeaseID, key []byte) error
func (m *Manager) Revoke(id LeaseID) error
type TxnRequest struct { Compare []Compare; Then []Op; Else []Op }

type Compare struct {
    Result      CompareResult   // EQUAL, GREATER, LESS, NOT_EQUAL
    Target      CompareTarget   // VERSION, CREATE, MOD, VALUE, LEASE
    Key, RangeEnd []byte
    TargetUnion CompareTargetValue
}

type Apply struct {
    // Own: one instance per node; the StateMachine the Raft library calls into.
    // Mut: rev (atomic.Uint64) is the only field mutated outside the single apply goroutine
    //      (read via atomic load for local reads); backend/keyIdx/watcher/leases mutated only
    //      from Apply().
    // Conc: single-writer discipline (I1); reads take a backend.ReadTx snapshot.
    // Pers: delegates to backend.Backend (§11.6).
}
func (a *Apply) Apply(idx, term uint64, env *ApplyEnvelope) (*ApplyResult, error)
func (a *Apply) ApplyPut(idx uint64, op *ApplyEnvelope_Put) (*ApplyResult, error)
func (a *Apply) ApplyDelete(idx uint64, op *ApplyEnvelope_Delete) (*ApplyResult, error)
func (a *Apply) ApplyTxn(idx uint64, op *ApplyEnvelope_Txn) (*ApplyResult, error)
type StateMachineAdapter struct {
    // Own: one per node; implements rank-39's StateMachine trait.
    // Mut: stateless wrapper — delegates all mutation to mvcc.Apply.
    // Conc: Apply() is called by the Raft library on its own apply goroutine; must not block.
    // Pers: none directly (delegates to mvcc.Apply -> backend.Backend).
}
func (a *StateMachineAdapter) Apply(idx, term uint64, payload []byte) ([]byte, error)

type Node struct {
    // Own: one per node; the proposer (§16).
    // Mut: pending map keyed by proposal id, guarded by mu; entries removed on Apply callback
    //      or context cancellation (I13 dedup lives one layer down in mvcc.Apply).
    // Conc: Propose() may be called concurrently by many request-handler goroutines; onApply()
    //      is called by the Raft library's own goroutine — pending map access must be synchronized.
    // Pers: none (in-memory correlation only; durable state is in Raft's log/stable store).
}
func (n *Node) Propose(ctx context.Context, env *ApplyEnvelope) (ApplyResult, error)
func (n *Node) ChangeMembership(ctx context.Context, change ConfigChange) (MembershipResult, error)

type SnapshotterAdapter struct{}
func (s *SnapshotterAdapter) Capture(meta SnapshotMeta) (io.ReadCloser, error)
func (s *SnapshotterAdapter) Restore(r io.Reader) error

type TransportAdapter struct{} // implements rank-39's Transport trait over RaftBridgeService
type Backend interface {
    BatchTx() BatchTx
    ReadTx() ReadTx
    Snapshot() Snapshot
    Defrag() error
    Close() error
    SizeInUse() int64
    OpenSnapshot(r io.Reader) error
}

type BatchTx interface {
    UnsafePut(bucket Bucket, key, value []byte)
    UnsafeDelete(bucket Bucket, key []byte)
    Commit() error   // fsync inside; single-writer (P10)
}

type ReadTx interface {
    UnsafeRange(bucket Bucket, key, end []byte, limit int64) ([][]byte, [][]byte)
}
