Core schema & types (§11)

/// Replica identifier. Stable for a replica's lifetime (P12); persisted in snapshot.
#[derive(Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug)]
pub struct ReplicaId(pub u64);

/// Monotonic per-replica counter. 1-indexed; Tick(0) means "never observed" (I6).
#[derive(Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug)]
pub struct Tick(pub u64);

/// Globally-unique identifier for one update event (I5).
#[derive(Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug)]
pub struct Dot { pub replica: ReplicaId, pub tick: Tick }

impl Dot { pub fn new(replica: ReplicaId, tick: Tick) -> Self; }
/// Contract every CRDT implements. The merge law (I1–I3) is verified per impl
/// by the proptest harness. `State` is implicit (Self); `Delta` is the
/// delta-state representation; both round-trip through encode/decode.
pub trait Crdt: Sized + Clone + Eq {
    type Delta: Clone;
    type Error: std::error::Error;

    fn merge(&mut self, other: &Self);
    fn apply_delta(&mut self, delta: &Self::Delta) -> Result<(), Self::Error>;
    fn delta_since(&self, since: &VersionVector) -> Self::Delta;
    fn encode_state(&self) -> Vec<u8>;
    fn decode_state(bytes: &[u8]) -> Result<Self, Self::Error>;
    fn encode_delta(delta: &Self::Delta) -> Vec<u8>;
    fn decode_delta(bytes: &[u8]) -> Result<Self::Delta, Self::Error>;
    /// GC history that is causally stable (observed by every replica in `stable`) — I4.
    fn gc(&mut self, stable: &VersionVector);
}
#[derive(Clone, Eq, PartialEq, Debug, Default)]
pub struct VectorClock { /* BTreeMap<ReplicaId, u64> */ }

impl VectorClock {
    pub fn tick(&mut self, replica: ReplicaId) -> Tick;            // 1-indexed (I6)
    pub fn merge(&mut self, other: &Self);                          // per-key max
    pub fn partial_cmp(&self, other: &Self) -> Option<Ordering>;    // None iff concurrent (I9)
    pub fn dominates(&self, other: &Self) -> bool;
    pub fn dominates_dot(&self, d: &Dot) -> bool;
}

#[derive(Clone, Eq, PartialEq, Debug, Default)]
pub struct VersionVector { /* BTreeMap<ReplicaId, Tick> */ }

impl VersionVector {
    pub fn observe(&mut self, dot: Dot);
    pub fn contains(&self, dot: &Dot) -> bool;
    pub fn merge(&mut self, other: &Self);
    pub fn tick_for(&self, r: &ReplicaId) -> u64;
    pub fn diff(&self, since: &Self) -> impl Iterator<Item = Dot> + '_;
}

/// Dotted Version Vector (Almeida et al. 2014): base VV + dots concurrent with base.
#[derive(Clone, Eq, PartialEq, Debug, Default)]
pub struct DottedVersionVector { /* base: VersionVector, dots: BTreeSet<Dot> */ }

impl DottedVersionVector {
    pub fn observe(&mut self, dot: Dot);   // inserts then compacts; order-independent
    fn compact(&mut self);                 // pull contiguous dots into base
}

#[derive(Clone, Eq, PartialEq, Debug)]
pub struct LamportTimestamp { /* replica: ReplicaId, counter: u64 */ }

#[derive(Clone, Eq, PartialEq, Debug)]
pub struct HybridLogicalClock { /* physical_us: u64, logical: u32, replica: ReplicaId */ }

impl HybridLogicalClock {
    pub fn tick(&mut self, now_us: u64) -> Self;
    pub fn observe(&mut self, remote: &Self, now_us: u64) -> Result<(), HlcError>; // SkewExceeded if skew > MAX_SKEW_US
}

/// Stability: has every known peer observed `dot`? (precondition for GC, I4)
pub fn is_stable(dot: &Dot, peers: &[VersionVector]) -> bool;
#[derive(Debug, thiserror::Error)]
pub enum MergeError {
    #[error("malformed wire format: {0}")]            BadWire(String),
    #[error("unknown wire version: {0}")]             UnknownVersion(u8),
    #[error("delta from unknown replica: {0:?}")]     UnknownReplica(ReplicaId),
    #[error("schema mismatch: expected {expected}, got {actual}")]
                                                      SchemaMismatch { expected: String, actual: String },
    #[error("phase deferred: {0}")]                   PhaseDeferred(&'static str),
}
pub trait Causality: Clone + Eq {
    fn observe(&mut self, dot: Dot);
    fn contains(&self, dot: &Dot) -> bool;
    fn merge(&mut self, other: &Self);
    fn dominates(&self, other: &Self) -> bool;
}

#[async_trait::async_trait]
pub trait Transport: Send + Sync {
    type Error: std::error::Error + Send + Sync + 'static;
    async fn send(&self, peer: PeerId, bytes: &[u8]) -> Result<(), Self::Error>;
    async fn recv(&self) -> Result<(PeerId, Vec<u8>), Self::Error>;
    async fn peers(&self) -> Vec<PeerId>;
}

pub trait Topology: Send + Sync {
    fn fanout(&self, n: usize) -> Vec<PeerId>;
    fn neighbours(&self, p: PeerId) -> Vec<PeerId>;
}

pub trait Storage: Send + Sync {
    fn append(&mut self, bytes: &[u8]) -> Result<(), StorageError>;
    fn snapshot(&mut self, state: &[u8]) -> Result<(), StorageError>;
    fn recover(&mut self) -> Result<RecoveryView, StorageError>;
}
