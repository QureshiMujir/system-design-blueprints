Core schema & types (§11)

/// Stable per-connection congestion-control policy vtable (mirrors tcp_congestion_ops).
/// One active impl per CcState. All hooks: no alloc, no block, no wall-clock (INV-11).
pub trait CongestionControl {
    fn name(&self) -> &'static str;                 // registry key, e.g. "cubic"
    fn init(&mut self, st: &mut CcState);           // set IW, ssthresh=∞, private state
    fn on_ack(&mut self, st: &mut CcState, rs: &RateSample, now_us: u64); // primary growth hook
    fn on_rtt_update(&mut self, st: &mut CcState, srtt_us: u64, min_rtt_us: u64); // optional
    fn on_loss(&mut self, st: &mut CcState, ev: &LossEvent);              // FastRetx/Rto/RackTlp
    fn on_ecn(&mut self, st: &mut CcState, ce_count: u64, acked: u64);    // ECN/AccECN response
    fn cwnd_event(&mut self, st: &mut CcState, ev: CwndEvent);            // init/idle/tx-start/loss-recovery
    fn ssthresh(&mut self, st: &CcState) -> u64;    // value to set on loss (mechanism reads it)
    fn pacing_rate(&self, st: &CcState) -> u64;     // bytes/sec; default cwnd*MSS/SRTT
    fn undo(&mut self, st: &mut CcState);           // spurious-loss: restore snapshot
    fn min_cwnd(&self) -> u64;                       // floor (INV-4); default provided = 2 * SMSS
}
pub enum CwndEvent { Init, TxStart, Idle, LossRecoveryEnter, LossRecoveryExit, PrrUpdate }
#[repr(C)]
pub struct CcState {
    pub cwnd: u64,           // bytes (INV-3/4)
    pub ssthresh: u64,       // bytes; u64::MAX == ∞
    pub bytes_in_flight: u64,
    pub mss: u32,            // SMSS
    pub ca_state: CaState,   // SlowStart|CongAvoid|Recovery|Rto
    pub delivered: u64,      // total bytes delivered (rate estimator)
    pub delivered_time_us: u64,
    pub app_limited_until: u64, // delivered-count watermark; 0 == not app-limited
    pub pacing_rate_bps: u64,
    pub snapshot: CwndSnapshot, // for undo (cwnd, ssthresh, ca_state)
    pub priv_off: usize,     // offset to module-private region (FFI)
    pub flags: CcFlags,      // ECN_OK, IN_RECOVERY, ...
}
pub enum CaState { SlowStart, CongAvoid, Recovery, Rto }
pub struct AckEvent  { pub acked_bytes: u64, pub cum_ack: u64, pub now_us: u64 }
pub struct SackBlock { pub start: u64, pub end: u64 }
pub struct SackEvent { pub blocks: ArrayVec<SackBlock, MAX_SACK_BLOCKS>, pub now_us: u64 }
pub struct EcnEvent  { pub ce_count: u64, pub ect_count: u64, pub now_us: u64 }
pub struct LossEvent { pub lost_bytes: u64, pub kind: LossKind, pub now_us: u64 }
pub enum   LossKind  { FastRetx, RackTlp, Rto }
pub struct RttSample { pub rtt_us: u64, pub is_retrans: bool, pub now_us: u64 }
pub struct SendDecision { pub send_bytes: u64, pub next_send_us: u64 }
pub struct CcSnapshot { pub cwnd: u64, pub ssthresh: u64, pub pacing_rate_bps: u64,
                        pub state: CaState, pub min_rtt_us: u64, pub srtt_us: u64, pub btlbw_bps: u64 }
pub struct RttEstimator { srtt_us: u64, rttvar_us: u64, rto_us: u64, has_sample: bool }
impl RttEstimator { fn update(&mut self, sample: &RttSample); fn rto(&self) -> u64; }

pub struct MinRttFilter { /* windowed min over W us */ }
impl MinRttFilter { fn observe(&mut self, rtt_us: u64, now_us: u64); fn min(&self) -> u64;
                    fn expired(&self, now_us: u64) -> bool; /* PROBE_RTT trigger */ }

pub struct PacketMeta { pub sent_us: u64, pub delivered_at_send: u64,
                        pub delivered_time_us: u64, pub size: u64, pub is_app_limited: bool }
pub struct RateSample { pub delivered: u64, pub interval_us: u64, pub is_app_limited: bool,
                        pub rtt_us: u64, pub delivery_rate_bps: u64 }
pub struct DeliveryEstimator { ring: Ring<PacketMeta>, delivered: u64, delivered_time_us: u64 }
impl DeliveryEstimator {
    fn on_send(&mut self, size: u64, now_us: u64, app_limited: bool);
    fn on_ack(&mut self, acked: u64, now_us: u64) -> RateSample; // INV-5
}
pub type CtorFn = fn() -> Box<dyn CongestionControl>;
pub struct Registry { table: BTreeMap<&'static str, CtorFn> } // ordered → deterministic listing
impl Registry {
    fn register(&mut self, name: &'static str, ctor: CtorFn) -> Result<(), CcError>;
    fn make(&self, name: &str) -> Result<Box<dyn CongestionControl>, CcError>;
    fn names(&self) -> impl Iterator<Item = &'static str>; // sorted
}
pub struct Connection { pub st: CcState, module: Box<dyn CongestionControl> }
impl Connection {
    fn swap(&mut self, reg: &Registry, name: &str) -> Result<(), CcError>; // INV-10/15
}
pub struct Pacer { tokens: u64, rate_bps: u64, last_us: u64, burst_cap: u64, quantum: u64 }
impl Pacer {
    fn set_rate(&mut self, bps: u64);
    fn on_send(&mut self, bytes: u64, now_us: u64);
    fn poll(&mut self, st: &CcState, now_us: u64) -> SendDecision; // INV-12, cwnd cap
}
pub struct Event { pub time_us: u64, pub seq: u64, pub kind: EventKind } // ordered (INV-13)
pub enum EventKind { Send{flow:FlowId,bytes:u64}, Arrive{flow:FlowId,pkt:PktId},
                     Dequeue, Ack{flow:FlowId,acked:u64}, Loss{flow:FlowId}, Tick }
pub struct Scheduler { heap: BinaryHeap<Reverse<Event>>, now_us: u64, next_seq: u64, rng: Pcg64 }
pub trait Qdisc { fn enqueue(&mut self, pkt: Pkt, now_us: u64) -> EnqResult; // Enq|Drop|Mark
                  fn dequeue(&mut self, now_us: u64) -> Option<Pkt>;
                  fn len_bytes(&self) -> u64; }
pub struct BottleneckLink { bandwidth_bps: u64, prop_delay_us: u64, qdisc: Box<dyn Qdisc> }
pub struct Dumbbell { flows: Vec<Flow>, link: BottleneckLink }
pub struct Histogram { /* integer buckets; p50/p95/p99 */ }
pub struct RunReport {
    pub seed: u64, pub build_hash: String, pub scenario: String,
    pub per_flow: Vec<FlowStats>, pub jain_fairness: Q16_16,
    pub aggregate_throughput_bps: u64, pub goodput_bps: u64,
    pub loss_rate_ppm: u64, pub queue_occupancy: QueueStats,
    pub rtt_p50_us: u64, pub rtt_p95_us: u64, pub rtt_p99_us: u64,
    pub convergence_time_us: u64,
}
pub struct FlowStats { pub flow: FlowId, pub algo: String, pub throughput_bps: u64,
                       pub rtt_p50_us: u64, pub retransmits: u64, pub rtt_us_base: u64 }
pub enum CcError {
    DuplicateName(&'static str), UnknownModule(String),
    InvalidSignal(&'static str), Overflow(&'static str),
    SwapDuringRecovery, AbiMismatch{expected:u32, got:u32},
}
