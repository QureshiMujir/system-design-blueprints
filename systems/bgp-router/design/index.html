Core schema & types (§11)

// crates/bgp-core/src/types.rs — owned by no task; Clone+immutable-after-build

pub type AsNumber = u32;                       // RFC 6793, 4-byte

#[derive(Clone, Debug, PartialEq, Eq, Hash, PartialOrd, Ord)]
pub struct Prefix { pub address: IpAddr, pub length: u8 }
// PartialOrd by (address bytes, length) — trie ordering depends on this (I3-adjacent)

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Route {
    pub prefix: Prefix,
    pub afi_safi: (Afi, Safi),
    pub attrs: Arc<PathAttributes>,             // P11: interned, shared
    pub source_peer: PeerId,
}

#[derive(Clone, Debug, PartialEq, Eq, Hash)]
pub struct PathAttributes {                     // interned key (P11)
    pub origin: Origin,
    pub as_path: AsPath,
    pub next_hop: IpAddr,
    pub med: Option<u32>,
    pub local_pref: Option<u32>,
    pub atomic_aggregate: bool,
    pub aggregator: Option<Aggregator>,
    pub communities: Vec<Community>,
    pub originator_id: Option<u32>,
    pub cluster_list: Option<Vec<u32>>,
    pub extended_communities: Vec<ExtendedCommunity>,   // §38 usage deferred
    pub large_communities: Vec<LargeCommunity>,
    pub unknown: Vec<UnknownAttribute>,         // preserved per Optional-Transitive rule
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Origin { Igp, Egp, Incomplete }
// crates/bgp-core/src/types.rs (cont.)

#[derive(Clone, Debug, PartialEq, Eq, Hash)]
pub struct AsPath { pub segments: Vec<AsPathSegment> }

#[derive(Clone, Debug, PartialEq, Eq, Hash)]
pub enum AsPathSegment {
    AsSet(Vec<AsNumber>),
    AsSequence(Vec<AsNumber>),
    AsConfedSequence(Vec<AsNumber>),            // RFC 5065
    AsConfedSet(Vec<AsNumber>),
}

impl AsPath {
    pub fn path_length(&self) -> u32;           // I3: AsSet counts as 1
    pub fn contains(&self, asn: AsNumber) -> bool;  // I1 loop check
    pub fn strip_confed_segments(&self) -> AsPath;  // §26 FM-9 eBGP egress scrub
}
// crates/bgp-fsm/src/state.rs — one instance per peer task, not Send across tasks

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum State { Idle, Connect, Active, OpenSent, OpenConfirm, Established }

pub struct PeerStateMachine {
    pub state: State,
    pub local_as: AsNumber,
    pub remote_as: AsNumber,
    pub local_router_id: Ipv4Addr,
    pub remote_router_id: Option<Ipv4Addr>,
    pub hold_time: Duration,                    // negotiated, min(local,remote)
    pub keepalive_time: Duration,                // hold_time / 3
    pub connect_retry_counter: u32,
    pub connect_retry_timer: Option<Instant>,    // monotonic clock, ADR-22
    pub hold_timer: Option<Instant>,
    pub keepalive_timer: Option<Instant>,
    pub negotiated_capabilities: Capabilities,
    pub gr_state: Option<GracefulRestartState>,  // §20, present only if capability negotiated
}

impl PeerStateMachine {
    pub fn handle_event(&mut self, event: Event) -> Vec<Action>;   // I10: total over (State,Event)
}

pub enum Event { /* 28 events per RFC 4271 §8.1, enumerated in §14 */ }
pub enum Action {
    InitiateTcp, SendOpen, SendKeepalive, SendNotification(ErrorCode, u8, Vec<u8>),
    StartConnectRetryTimer, StopConnectRetryTimer, StartHoldTimer(u16), StartKeepaliveTimer,
    SyncRib, ReleaseResources, WithdrawAllRoutes,
}
// crates/bgp-codec/src/message.rs — pure encode/decode, no I/O, no shared state

#[derive(Clone, Debug)]
pub enum Message {
    Open(OpenMessage), Update(UpdateMessage),
    Notification(NotificationMessage), Keepalive,
    RouteRefresh(RouteRefreshMessage),           // RFC 2918
}

#[derive(Clone, Debug)]
pub struct OpenMessage {
    pub version: u8,                             // = 4
    pub my_as: u16,                               // legacy field; see FourOctetAsNumber capability
    pub hold_time: u16,
    pub bgp_id: Ipv4Addr,
    pub optional_params: Vec<OptionalParameter>,
}

#[derive(Clone, Debug)]
pub struct UpdateMessage {
    pub withdrawn: Vec<Prefix>,                  // IPv4 unicast NLRI only
    pub path_attributes: PathAttributes,
    pub nlri: Vec<Prefix>,                        // IPv4 unicast NLRI only
    pub mp_reach_nlri: Option<MpReachNlri>,        // §18
    pub mp_unreach_nlri: Option<MpUnreachNlri>,    // §18, §20 GR end-of-RIB carrier
}

#[derive(Clone, Debug)]
pub struct NotificationMessage { pub error_code: ErrorCode, pub error_subcode: u8, pub data: Vec<u8> }

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ErrorCode {
    MessageHeader = 1, OpenMessage = 2, UpdateMessage = 3,
    HoldTimerExpired = 4, FsmError = 5, Cease = 6, RouteRefresh = 7,
}

pub trait Codec {
    fn decode(buf: &[u8]) -> Result<Message, CodecError>;   // I4: length-validated before parse
    fn encode(msg: &Message, out: &mut Vec<u8>) -> Result<(), CodecError>;
}
// crates/bgp-rib/src/tables.rs — Loc-RIB shared via Arc<RwLock<_>> per (Afi,Safi), short critical sections

pub struct AdjRibIn { pub peer_id: PeerId, pub afi_safi: (Afi, Safi), pub routes: HashMap<Prefix, Route> }
pub struct AdjRibOut { pub peer_id: PeerId, pub afi_safi: (Afi, Safi), pub advertised: HashMap<Prefix, Route> }

pub struct LocRib { tables: HashMap<(Afi, Safi), PrefixTable> }
pub struct PrefixTable { trie: PrefixTrie, by_prefix: HashMap<Prefix, PrefixEntry> }
pub struct PrefixEntry { pub best: Option<RouteId>, pub candidates: Vec<RouteId> }
pub struct PrefixTrie { /* Patricia trie, ADR-06; nodes hold RouteId lists */ }

impl LocRib {
    pub fn insert(&mut self, route: Route) -> RibChange;              // I5 sole FIB-facing writer
    pub fn withdraw(&mut self, peer: PeerId, prefix: &Prefix) -> RibChange;
    pub fn best_path(&self, prefix: &Prefix) -> Option<&Route>;
    pub fn select_best(&mut self, prefix: &Prefix) -> Option<RouteId>; // pure, P1
}

pub enum RibChange { Announce(Route), Withdraw(Prefix), BestPathChanged { prefix: Prefix, old: Option<RouteId>, new: Option<RouteId> } }
// crates/bgp-filter/src/ast.rs — pure AST, no side effects (I9)

pub enum FilterStmt {
    If { cond: Expr, then: Vec<FilterStmt>, else_: Vec<FilterStmt> },
    Reject, Accept,
    Assign { target: FieldPath, value: Expr },
    Call { name: String, args: Vec<Expr> },
}
pub enum Expr {
    Literal(Value), Variable(FieldPath),
    BinaryOp { op: BinOp, lhs: Box<Expr>, rhs: Box<Expr> },
    AsPathRegex(AsPathPattern), CommunityMatch(CommunitySet),
}
pub trait FilterEngine {
    fn evaluate(&self, route: &Route, filter: &Filter) -> FilterResult;  // I9: total, pure
}

// crates/bgp-fib-linux/src/netlink.rs — FIB programmer, abstracted (P6)
pub trait FibBackend {
    fn install_route(&self, prefix: &Prefix, nexthop: IpAddr, table: RouteTable) -> Result<(), FibError>;
    fn withdraw_route(&self, prefix: &Prefix, table: RouteTable) -> Result<(), FibError>;
}
