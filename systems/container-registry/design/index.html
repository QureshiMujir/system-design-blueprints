Core schema & types (§11)

// ownership: value type, Copy-able; mutation: none (immutable); persistence: TEXT column;
// serialization: "algo:hex"; compatibility: new algorithms require an enum variant + migration.
struct Digest { algorithm: DigestAlgorithm, hex: Vec<u8> }
enum DigestAlgorithm { Sha256, Sha512 }

// ownership: value type; mutation: none; persistence: not stored directly (derived);
// serialization: OCI mediaType string constants.
enum MediaType {
    DockerManifestV2Schema2, DockerManifestListV2Schema2,
    OciImageManifestV1, OciImageIndexV1, OciArtifactManifestV1,
    SigstoreBundle, InTotoAttestation, SpdxJson, CycloneDxJson, Unknown(String),
}

// ownership: value type constructed from Descriptor fields; mutation: none;
// persistence: embedded JSON in manifests.body; concurrency: read-only after Committed (I2).
struct Descriptor {
    media_type: MediaType, digest: Digest, size: u64,
    annotations: BTreeMap<String, String>, platform: Option<Platform>,
    artifact_type: Option<String>,
}
// ownership: owned by MetadataStore row; mutation: none post-commit (I2, I4);
// concurrency: single-writer-per-digest via INSERT..ON CONFLICT DO NOTHING; persistence: Postgres row.
struct Manifest {
    digest: Digest, media_type: MediaType, size: u64, body: Vec<u8>,
    config: Option<Descriptor>, layers: Vec<Descriptor>, subject: Option<Descriptor>,
    annotations: BTreeMap<String, String>, created_at: SystemTime,
}

// ownership: MetadataStore row; mutation: none (index is immutable once Committed);
// serialization: JSON body identical to pushed bytes (P1).
struct ImageIndex { digest: Digest, manifests: Vec<Descriptor>, subject: Option<Descriptor> }

// ownership: MetadataStore row; mutation: none post-commit; opaque payload to the registry.
struct ImageConfig { digest: Digest, architecture: String, os: String, diff_ids: Vec<Digest> }

// ownership: value type; mutation: none; used only as a graph-walk edge type, never persisted directly.
struct LayerRef { digest: Digest, media_type: MediaType }
// ownership: value type validated against RFC 3986 path segment rules; mutation: none.
struct RepositoryName(String);   // e.g. "myorg/myimage"

// ownership: value type; mutation: none; a Reference never mutates — pulling a Tag
// re-resolves through TagIndex each time (P5).
enum Reference { Tag(String), Digest(Digest) }

// ownership: MetadataStore row, one live row per (repository, name); mutation: atomic
// upsert only (I5); concurrency: row-level lock via UPDATE..RETURNING in one transaction.
struct TagIndex { repository_id: Uuid, name: String, manifest_digest: Digest, updated_at: SystemTime }

// ownership: handle into StorageBackend, not the bytes themselves; mutation: none;
// concurrency: Send+Sync, safe to share across handler tasks.
struct BlobHandle { digest: Digest, size: u64, storage_key: String }
// concurrency: implementors must be Send+Sync; persistence: backend-specific;
// compatibility: adding a method is a breaking trait change, gate behind a default impl.
trait StorageBackend {
    fn write_temp(&self, key: &str, offset: u64, bytes: &[u8]) -> Result<(), StorageError>;
    fn finalize(&self, temp_key: &str, final_key: &str) -> Result<(), StorageError>;
    fn read(&self, key: &str, range: Option<ByteRange>) -> Result<ByteStream, StorageError>;
    fn delete_batch(&self, keys: &[String]) -> Result<(), StorageError>;
    fn presign_get(&self, key: &str, ttl: Duration) -> Option<PresignedUrl>;
}

// concurrency: all methods take &self, internal pooling; persistence: Postgres;
// compatibility: schema migrations versioned in registry-metadata/migrations.
trait MetadataStore {
    fn upsert_tag(&self, repo: Uuid, name: &str, digest: &Digest) -> Result<(), StoreError>;
    fn insert_manifest(&self, m: &Manifest, repo: Uuid) -> Result<(), StoreError>;
    fn blob_exists(&self, digest: &Digest, repo: Uuid) -> Result<bool, StoreError>;
    fn referrers(&self, subject: &Digest, artifact_type: Option<&str>) -> Result<Vec<Manifest>, StoreError>;
}
// ownership: value type, opaque handle; mutation: none.
struct UploadSessionId(Uuid);

// ownership: MetadataStore row; mutation: offset/state transitions only (§8.9 state machine);
// concurrency: single-writer via SELECT..FOR UPDATE per uuid; persistence: durable (ADR-09).
struct UploadSession {
    id: UploadSessionId, repository_id: Uuid, current_offset: u64,
    temp_storage_key: String, expected_digest: Option<Digest>,
    state: UploadState, created_at: SystemTime, last_chunk_at: SystemTime,
}
enum UploadState { InProgress, Verifying, Finalized, Aborted }

// ownership: value type; mutation: none; used for I6 continuity checks.
struct ChunkRange { start: u64, end: u64 }
// ownership: GC-worker-owned, ephemeral (not client-visible); mutation: append during mark;
// concurrency: single GC worker per shard, no cross-worker mutation.
struct GcMark { manifest_digest: Digest, marked_at: SystemTime }

// ownership: GC-worker-owned; mutation: advances monotonically per §8.8 step; persistence: durable
// (survives GC-worker restart mid-scan).
struct GcCursor { last_repository_id: Uuid, mark_start_ts: SystemTime }

// ownership: MetadataStore row; mutation: quarantined_at set once, cleared on rescue (I3);
// persistence: durable, drives the 48h window (§4, §18).
struct QuarantineEntry { digest: Digest, quarantined_at: SystemTime, rescued: bool }
// ownership: value type parsed from Authorization header; mutation: none; compatibility:
// grammar changes are additive-only (§8.10 EBNF).
struct TokenScope { resource_type: String, resource_name: RepositoryName, actions: Vec<String> }

// ownership: value type minted by token service; mutation: none (immutable once signed).
struct Grant { scope: TokenScope, expires_at: SystemTime }

// ownership: value type decoded from JWT (RFC 7519); mutation: none; serialization: JWT claims.
struct AuthClaims { subject: String, grants: Vec<Grant>, issued_at: SystemTime, expires_at: SystemTime }

// ownership: Redis-backed shared counter, keyed (token, repository); mutation: atomic decrement
// per request; concurrency: Redis Lua script for atomicity.
struct RateLimitBucket { key: String, capacity: u32, refill_per_sec: u32, tokens_remaining: u32 }

// ownership: MetadataStore row; mutation: admin-updated only; persistence: durable.
struct Quota { repository_id: Uuid, max_storage_bytes: u64, max_pushes_per_day: u32 }
// ownership: value type wrapping cosign/sigstore bundle bytes; mutation: none; verified,
// never trusted without signature check against the configured trust root (P13).
struct SignatureEnvelope { subject_digest: Digest, bundle: Vec<u8>, verified: bool }

// ownership: value type, referrer-manifest-backed (§8.6); mutation: none.
struct AttestationRef { subject_digest: Digest, predicate_type: String, referrer_digest: Digest }

// ownership: MetadataStore row keyed by manifest digest; mutation: append-only (new scan
// creates a new row, never overwrites); persistence: durable, exposed as an annotation/referrer.
struct ScanReport { subject_digest: Digest, scanner: String, severity_counts: BTreeMap<String, u32>, scanned_at: SystemTime }

// ownership: replication-worker-owned; mutation: advances monotonically per peer;
// persistence: durable (outbox cursor, ADR-24).
struct ReplicationCursor { peer_id: Uuid, last_sequence: u64, lag: Duration }

// ownership: append-only MetadataStore row (I12); mutation: none, ever; persistence: durable,
// never deleted (retention policy is archival, not deletion).
struct AuditEntry { sequence: u64, actor: String, action: String, resource: String, at: SystemTime }

// ownership: value type; mutation: none; compatibility: every variant maps 1:1 to an OCI
// error code (§8.4) — adding a variant requires adding its OCI code mapping.
enum RegistryError {
    DigestInvalid, ManifestInvalid, ManifestBlobUnknown, ManifestUnknown,
    ManifestTooLarge, BlobUnknown, BlobUploadUnknown, RangeNotSatisfiable,
    NameUnknown, Denied, Unauthorized, TooManyRequests, Unsupported,
}
