Core schema & types (§11)

pub struct Vcpu {
    pub id: VcpuId,                       // owned by exactly one thread (I1)
    pub fd: KvmVcpuFd,                     // not Sync; move-only
    pub regs: CpuRegs,                     // mutated only from owning thread
    pub state: VcpuState,                  // see state machine, §8.1
}

pub enum VcpuState {
    Created, Running, ExitedForReason(ExitReason), Halted, Crashed,
}

pub enum ExitReason {                      // exhaustive match required (I9)
    Mmio { gpa: GpAddr, is_write: bool, data: [u8; 8], len: u8 },
    Io { port: u16, is_write: bool, data: [u8; 4], len: u8 },
    Hlt,
    Shutdown,
    Hypercall { num: u64, args: [u64; 6] },
    EptViolation { gpa: GpAddr, access: AccessKind },
    Interrupted,
    Unknown(u32),                          // catch-all, never silently dropped
}

pub struct VmConfig {                      // serde (de)serializable, ADR-15
    pub vcpus: u32,
    pub memory_size: u64,
    pub devices: Vec<DeviceConfig>,
    pub kernel: KernelImage,
    pub initramfs: Option<Vec<u8>>,
    pub cmdline: String,
}

pub trait Device: Send + Sync {            // device-thread-owned; no interior mutability leaks
    fn name(&self) -> &str;
    fn pci_id(&self) -> Option<PciId>;
    fn mmio_read(&self, offset: u64, buf: &mut [u8]);
    fn mmio_write(&self, offset: u64, buf: &[u8]);
    fn irq_lines(&self) -> &[IrqLine];
}

pub struct MemoryRegion {                  // registered once, immutable after (P3)
    pub gpa: GpAddr,
    pub hva: HostVAddr,
    pub size: u64,
    pub dirty_tracking: bool,              // I4 precondition
}
pub struct VringDescriptor {               // guest-writable, host-validated (I5, I6)
    pub addr: GpAddr,
    pub len: u32,
    pub flags: DescFlags,
    pub next: Option<u16>,
}

pub enum MemoryError {                     // ownership: returned, never panics on guest input
    Overlap { existing: GpAddr, requested: GpAddr },
    Unaligned(GpAddr),
    OutOfBounds { addr: GpAddr, len: u64 },
}

pub struct MigrationRecord {               // append-only wire format (ADR-09, P7)
    pub seq: u64,
    pub kind: MigrationRecordKind,
    pub payload: Vec<u8>,
}

pub enum MigrationRecordKind {
    MemoryPage { gpa: GpAddr },
    CpuState { vcpu_id: VcpuId },
    DeviceState { device: PciId },
    EndOfStream,
}
