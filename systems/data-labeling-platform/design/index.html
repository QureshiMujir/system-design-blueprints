Core schema & types (§11)

struct OrganizationId(Uuid)          // tenant root; RLS partition key (I5)
struct ProjectId(Uuid)
struct DatasetId(Uuid)
struct AssetId(Uuid)
struct TaskId(Uuid)
struct RevisionId(Uuid)              // monotonic per TaskId (I1)
struct LeaseToken(u64)               // fencing token, CAS-compared (I12)
struct ReviewDecisionId(Uuid)
struct WorkflowId(Uuid)
struct EventId(Uuid)                 // dedup key component (P9)
struct UserId(Uuid)

enum TaskState {                     // §8 state-transition table
  Created, Assigned, InProgress, Submitted,
  ReviewPending, Approved, Rejected, Requeue, Archived
}

enum AnnotationPayload {             // schema-validated per type (§8, I10)
  Bbox { x: f32, y: f32, w: f32, h: f32 },
  Polygon { points: Vec<Point2D> },
  Segmentation { mask_ref: ContentHash, class_map: ClassMapId },
  Keypoint { points: Vec<KeypointRef> },
  NerSpan { start: u32, end: u32, label: LabelId },
  Transcription { start_ms: u64, end_ms: u64, text: String, speaker: SpeakerId },
}

struct AnnotationRevision {          // owned by annotation-service; immutable payload
  revision_id: RevisionId,
  task_id: TaskId,
  author_id: UserId,
  payload: AnnotationPayload,
  payload_hash: Sha256,              // I9
  created_at: Timestamp,
  parent_revision: Option<RevisionId>,
}

struct LeaseRecord {                 // owned by task-service; pessimistic (I3, I12)
  task_id: TaskId,
  worker_id: UserId,
  fencing_token: LeaseToken,
  acquired_at: Timestamp,
  ttl_seconds: u32,
}

struct ConsensusRecord {             // owned by review-service (ADR-16, I11)
  task_id: TaskId,
  algorithm: ConsensusAlgorithm,     // enum { DawidSkene, MajorityVote }
  algorithm_version: SemVer,
  raw_votes: Vec<AnnotatorVote>,
  aggregated_label: LabelId,
  agreement_score: f32,              // kappa or alpha, per ADR-18
}

struct ReviewDecision {
  decision_id: ReviewDecisionId,
  task_id: TaskId,
  reviewer_id: UserId,
  outcome: ReviewOutcome,            // enum { Approved, Rejected }
  prior_hash: Sha256,                // signed chain (I8)
  signature: Ed25519Signature,
}

struct WorkflowDefinition {          // compiled DAG -> FSM (ADR-14, P1)
  workflow_id: WorkflowId,
  states: Vec<TaskState>,
  transitions: Vec<(TaskState, EventKind, TaskState)>,
}
struct EventEnvelope<T> {            // event-bus wire wrapper (I6, P9)
  event_id: EventId,
  aggregate_id: Uuid,                // Kafka partition key
  occurred_at: Timestamp,
  schema_version: SemVer,
  payload: T,
}

enum StorageError { HashMismatch, StaleFencingToken, SchemaIncompatible, QuotaExceeded }
