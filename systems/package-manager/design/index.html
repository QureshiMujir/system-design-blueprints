Core schema & types (§11)

struct Version { major: u64, minor: u64, patch: u64, pre: Vec<PreId>, build: Vec<String> }
struct VersionReq { comparators: Vec<Comparator> }
struct Comparator { op: ComparatorOp, version: Version }
enum ComparatorOp { Exact, Gt, Gte, Lt, Lte, Caret, Tilde, Wildcard }
struct PackageId { name: String, source: SourceId }
struct SourceId { url: String, kind: SourceKind }
enum SourceKind { Registry, Local, Vendored }
struct Manifest { package: PackageMeta, dependencies: BTreeMap<String, DepSpec>, dev_dependencies: BTreeMap<String, DepSpec>, features: BTreeMap<String, Vec<String>> }
struct PackageMeta { name: String, version: Version, edition: Option<String> }
struct DepSpec { req: VersionReq, features: Vec<String>, optional: bool, registry: Option<String> }
struct Dependency { id: PackageId, req: VersionReq, features: Vec<String>, optional: bool }
enum Term { Positive(PackageId, VersionSet), Negative(PackageId, VersionSet) }
struct Incompatibility { terms: Vec<Term>, cause: Cause }
enum Cause { Root, Dependency { from: PackageId, to: PackageId }, NoVersions(PackageId), Conflict(Box<Incompatibility>, Box<Incompatibility>) }
struct Assignment { term: Term, decision_level: u32, cause: Option<Cause> }
struct PartialSolution { assignments: Vec<Assignment>, decisions: BTreeMap<PackageId, Version> }
struct Lockfile { version: u32, packages: Vec<LockedPackage>, index_snapshot: String }
struct LockedPackage { name: String, version: Version, source: SourceId, checksum: Blake3Hash, dependencies: Vec<String> }
struct IndexEntry { name: String, version: Version, deps: Vec<Dependency>, checksum: Blake3Hash, yanked: bool, features: BTreeMap<String, Vec<String>> }
struct CacheKey(Blake3Hash);
struct StoreHandle { key: CacheKey, path: PathBuf }
trait Fetcher { fn fetch(&self, id: &PackageId, version: &Version) -> Result<Bytes, FetchError>; }
struct BuildUnit { package: PackageId, version: Version, inputs: Vec<StoreHandle>, script: Option<BuildScriptRef> }
enum StoreError { HashMismatch, NotFound, GcRaceDetected, IoError(String) }
enum ResolveError { NoVersions(PackageId), Conflict(Incompatibility), IndexUnavailable(String) }
enum FetchError { Timeout, HashMismatch, SignatureInvalid, MirrorExhausted }
struct SignedArtifact { bytes_hash: Blake3Hash, cert_chain: CertChain, rekor_proof: InclusionProof }
struct Attestation { subject_hash: Blake3Hash, command: String, materials: Vec<Blake3Hash>, products: Vec<Blake3Hash> }
enum SignError { MissingSignature, InvalidSignature, RekorMismatch, RevokedCertificate }
enum AuditFinding { Clean, Vulnerable { cve_id: String, severity: Severity, fixed_range: VersionReq } }
enum Severity { Low, Medium, High, Critical }
