Core schema & types (§11)

pub trait CausalClock: Sized + Clone {
    type Actor: ActorId;
    type Error;

    fn tick(&mut self, actor: Self::Actor);
    fn merge(&mut self, other: &Self);
    fn partial_cmp(&self, other: &Self) -> Option<core::cmp::Ordering>;
    fn concurrent_with(&self, other: &Self) -> bool;   // default: derived, I3
    fn serialize(&self, out: &mut Vec<u8>);
    fn deserialize(bytes: &[u8]) -> Result<Self, Self::Error>;
}
pub trait ActorId: Copy + Ord + core::hash::Hash + Sized {}
// Blanket impls: u8, u32, u64, Uuid (feature "uuid"), [u8; 16].
pub struct VectorClock<A: ActorId> { entries: alloc::collections::BTreeMap<A, u64> }

impl<A: ActorId> VectorClock<A> {
    pub fn new() -> Self;
    pub fn get(&self, actor: &A) -> u64;                // missing => 0 (I5)
    pub fn dominates(&self, other: &Self) -> bool;
    pub fn actors(&self) -> impl Iterator<Item = &A>;
}
impl<A: ActorId> CausalClock for VectorClock<A> { /* signatures per §11.1 */ }
pub struct LamportTs(pub u64);                          // PartialOrd only, not Ord (P9)
impl LamportTs {
    pub const ZERO: Self;
    pub fn tick(&mut self) -> Self;
    pub fn observe(&mut self, remote: Self) -> Self;
}

pub struct TotalOrder<A: ActorId> { pub ts: LamportTs, pub actor: A }  // Ord (I13)
pub struct Dot<A: ActorId> { pub actor: A, pub counter: u64 }
pub struct VersionVector<A: ActorId>(pub alloc::collections::BTreeMap<A, u64>);
pub struct DottedVersionVector<A: ActorId> {
    pub cc: VersionVector<A>,
    pub dots: alloc::collections::BTreeSet<Dot<A>>,      // I8
}
pub struct DvvSet<A: ActorId, V> { entries: Vec<(DottedVersionVector<A>, V)> }
impl<A: ActorId, V> DvvSet<A, V> {
    pub fn sync(&mut self, write: DottedVersionVector<A>, value: V);
    pub fn read(&self) -> &[(DottedVersionVector<A>, V)];
    pub fn compact(&mut self);                           // I8, invoked post-merge
}
pub struct Hlc { pub wall_ns: u64, pub logical: u32, pub node_id: NodeId }
pub trait PhysicalClockSource { fn now_ns(&self) -> u64; }
impl Hlc {
    pub fn tick<C: PhysicalClockSource>(&mut self, src: &C) -> Result<Self, SkewError>;
    pub fn observe<C: PhysicalClockSource>(&mut self, remote: Self, src: &C)
        -> Result<Self, SkewError>;
}
pub enum Id { Leaf(bool), Node(alloc::boxed::Box<Id>, alloc::boxed::Box<Id>) }
pub enum Event { Leaf(u32), Node(u32, alloc::boxed::Box<Event>, alloc::boxed::Box<Event>) }
pub struct IntervalTreeClock { pub id: Id, pub event: Event }
impl IntervalTreeClock {
    pub fn seed() -> Self;
    pub fn fork(&self) -> (Self, Self);
    pub fn event(&mut self);
    pub fn join(self, other: Self) -> Self;
}
pub fn atomic_write(path: &std::path::Path, bytes: &[u8]) -> std::io::Result<()>;
pub trait ClockStore {
    fn put(&self, key: &[u8], clock_bytes: &[u8]) -> Result<(), StoreError>;
    fn get(&self, key: &[u8]) -> Result<Option<Vec<u8>>, StoreError>;
}
pub struct Frame { pub version: u8, pub clock_type: u8, pub payload: Vec<u8> }
impl Frame {
    pub fn encode(&self, out: &mut Vec<u8>);
    pub fn decode(bytes: &[u8]) -> Result<Self, FrameError>;
}
pub struct HeartbeatDetector<A: ActorId> {
    last_seen: alloc::collections::BTreeMap<A, std::time::Instant>,
    timeout: std::time::Duration,
}
impl<A: ActorId> HeartbeatDetector<A> {
    pub fn observe(&mut self, actor: A);
    pub fn dead(&self) -> Vec<A>;
}
pub struct Tombstone<A: ActorId> { pub actor: A, pub evicted_counter: u64, pub epoch: u64 }
