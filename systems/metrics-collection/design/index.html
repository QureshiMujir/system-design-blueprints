Core schema & types (§11)

// tempora-model
struct LabelSet(Vec<(Interned<str>, Interned<str>)>)     // owned; immutable after build(); Send+Sync
  fn build(pairs: impl Iterator<Item=(&str,&str)>) -> LabelSet   // sorts, dedups, interns
  fn series_id(&self) -> u64                                // FNV1a of canonical bytes; pure
struct Sample { series_id: u64, timestamp_ms: i64, value: f64 } // Copy; wire-serializable (Prost)
enum MetricType { Counter, Gauge, Histogram, Summary }      // Copy; persisted in WAL Series record
struct MetricFamily { name: String, help: String, kind: MetricType, unit: Option<String> }
// tempora-client — concurrency: Counter/Gauge lock-free (AtomicU64 bit-cast); Histogram per-bucket
// atomic; Summary behind a per-quantile Mutex<TDigest> (Gjengset ch.6, applied P1/§14)
struct Counter { series_id: u64, value: AtomicU64 }              // monotonic; inc()/inc_by() only
struct Gauge   { series_id: u64, value: AtomicU64 }              // arbitrary set()/inc()/dec()
struct Histogram { buckets: Vec<(f64 /*le*/, AtomicU64)>, sum: AtomicU64, count: AtomicU64 }
struct Summary { digest: Mutex<TDigest>, sum: AtomicU64, count: AtomicU64 }  // Greenwald-Khanna/t-digest
struct MetricRegistry { families: DashMap<String, MetricFamily>, series: DashMap<u64, SeriesHandle> }
// tempora-tsdb — persistence: WAL fsynced per I2; Head is process-local RAM, not persisted directly
struct HeadBlock { shard_window: TimeWindow, series: DashMap<u64, SeriesHandle>, index: InvertedIndex }
struct SeriesHandle { labels: LabelSet, chunk_in_progress: GorillaChunkBuilder }  // single-writer
enum WalRecord { Series{id:u64, labels:LabelSet}, Sample(Sample), Tombstone{id:u64}, Checkpoint }
struct ChunkFile { header: ChunkHeader, entries: Vec<(u64 /*series_id*/, TimeRange, Bytes)> }
struct InvertedIndex { symbols: SymbolTable, postings: BTreeMap<(u32,u32), Vec<u64>> } // label→series
struct Tenant { id: TenantId, head: HeadBlock, wal: Wal, chunks: ChunkStore, retention: Duration }
// tempora-promql — no mutable shared state; Expr is an owned tree with source spans (P14)
enum Token { Ident(String), Number(f64), Str(String), Op(OpKind), LBrace, RBrace, LParen, RParen, … }
enum Expr {
  VectorSelector{ name:String, matchers:Vec<Matcher>, offset:Option<Duration> },
  MatrixSelector{ inner:Box<Expr>, range:Duration },
  BinaryOp{ op:OpKind, lhs:Box<Expr>, rhs:Box<Expr>, matching:VectorMatching },
  Call{ func:String, args:Vec<Expr> },
  Aggregate{ op:AggOp, by:Vec<String>, without:bool, inner:Box<Expr> },
  Span(Box<Expr>, SourceSpan),
}
enum QueryResult { Instant(Vec<(LabelSet,f64)>), Range(Vec<(LabelSet,Vec<(i64,f64)>)>), Scalar(f64) }
// tempora-rule / tempora-api — wire/state types
enum Rule { Recording{name:String,labels:LabelSet,expr:Expr},
            Alerting{name:String,labels:LabelSet,expr:Expr,for_:Duration,annotations:BTreeMap<String,String>} }
enum AlertState { Inactive, Pending{since:i64}, Firing{since:i64} }   // per (name,LabelSet); checkpointed (I14)
struct WriteRequest { series: Vec<(LabelSet, Vec<Sample>)> }          // Protobuf, Snappy-framed (ADR-10/11)
struct ReadRequest  { queries: Vec<(Vec<Matcher>, TimeRange)> }
