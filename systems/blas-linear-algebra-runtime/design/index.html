Core schema & types (§11)

pub trait Scalar: Copy + Send + Sync {
    type Real: Scalar;                 // f64 for c64, itself for f64
    const EPS: Self::Real; const SAFMIN: Self::Real; const ZERO: Self; const ONE: Self;
    fn mul_add(self, b: Self, c: Self) -> Self;   // FMA (single rounding on real)
    fn conj(self) -> Self; fn abs(self) -> Self::Real; fn is_complex() -> bool;
}
// impls: f32, f64, Complex<f32>=c32, Complex<f64>=c64   (#[repr(C)] interleaved)
pub struct MatView<'a, T>  { ptr:*const T, nrows:usize, ncols:usize, rs:isize, cs:isize, _p:PhantomData<&'a T> }
pub struct MatViewMut<'a,T>{ ptr:*mut T,   nrows:usize, ncols:usize, rs:isize, cs:isize, _p:PhantomData<&'a mut T> }
pub struct VecView<'a, T>  { ptr:*const T, len:usize, inc:isize, _p:PhantomData<&'a T> }
impl MatView { fn transpose(self)->Self; fn submat(self,r,c,nr,nc)->Self; fn col(self,j)->VecView; fn is_col_major(&self)->bool; }
#[repr(i32)] enum Trans { N=111, T=112, C=113 }     // match CBLAS
#[repr(i32)] enum Uplo  { Upper=121, Lower=122 }
#[repr(i32)] enum Diag  { NonUnit=131, Unit=132 }
#[repr(i32)] enum Side  { Left=141, Right=142 }
#[repr(i32)] enum Order { ColMajor=102, RowMajor=101 }
pub struct BlockSizes { mr:usize, nr:usize, mc:usize, kc:usize, nc:usize, vlen:usize, isa:Isa }
pub struct Machine { l1d:usize, l2:usize, l3:usize, line:usize, vregs:usize, fma_peak_gflops:f64, isa:Isa, cores:usize }
pub enum Isa { ScalarFallback, Avx2Fma, Avx512, Neon }
pub struct Packed<T> { buf: AlignedArena<T> /*64B*/, panel_stride:usize, n_panels:usize, kc:usize, kind:PackKind }
enum PackKind { ColPanelsOfMr, RowPanelsOfNr }   // Ap vs Bp
type MicroKernel<T> = unsafe fn(kc:usize, alpha:T, ap:*const T, bp:*const T, beta:T, c:*mut T, rs:isize, cs:isize);
struct KernelSet<T> { gemm_ukr: MicroKernel<T>, mr:usize, nr:usize /*+ packers, axpy/dot vec kernels*/ }
fn gemm<T:Scalar>(ta:Trans, tb:Trans, alpha:T, a:MatView<T>, b:MatView<T>, beta:T, c:MatViewMut<T>) -> Result<(),BlasError>;
fn syrk<T>(uplo:Uplo, trans:Trans, alpha:T::Real, a:MatView<T>, beta:T::Real, c:MatViewMut<T>)->Result<(),BlasError>;
fn trsm<T>(side:Side,uplo:Uplo,trans:Trans,diag:Diag,alpha:T,a:MatView<T>,b:MatViewMut<T>)->Result<(),BlasError>;
fn getrf<T>(a:MatViewMut<T>, ipiv:&mut [i32]) -> Result<i32 /*info*/, BlasError>;     // PA=LU
fn potrf<T>(uplo:Uplo, a:MatViewMut<T>) -> Result<i32, BlasError>;                    // A=LLᵀ
fn geqrf<T>(a:MatViewMut<T>, tau:&mut [T], work:Workspace) -> Result<(),BlasError>;   // A=QR
fn getrs<T>(trans:Trans,a:MatView<T>,ipiv:&[i32],b:MatViewMut<T>)->Result<(),BlasError>;
fn gels<T>(trans:Trans,a:MatViewMut<T>,b:MatViewMut<T>,work:Workspace)->Result<(),BlasError>; // min‖Ax−b‖
pub enum BlasError { BadParam{routine:&'static str, pos:i32}, DimMismatch, SingularAt(i32), NotSpd(i32), Workspace{need:usize,got:usize} }
pub struct KernelTrace { op:&'static str, dtype:&'static str, m:usize,n:usize,k:usize,
                         isa:Isa, mr:usize,nr:usize,mc:usize,kc:usize,nc:usize,
                         threads:usize, pack_ns:u64, compute_ns:u64, gflops:f64, accum_order_id:u64 }
