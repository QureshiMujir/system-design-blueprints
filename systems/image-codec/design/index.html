Core schema & types (§11)

// bitio.py — ownership: caller owns the backing bytes; mutation: internal cursor only;
// concurrency: not thread-safe, one reader per stream; persistence: none; serialization: n/a
struct BitReader:
    order: BitOrder            // MSB_FIRST (JPEG) | LSB_FIRST (DEFLATE) — P10
    buf: bytes
    byte_pos: int
    bit_pos: int
    limit: int                 // hard end offset — never read past (I3)

struct BitWriter:
    order: BitOrder
    out: bytearray
    bit_accum: int
    bit_count: int
// buffer.py — ownership: decoder produces, caller owns after return; mutation: write-once
// post-decode; concurrency: immutable-after-construction, safe to share read-only;
// persistence: none (in-memory); serialization: via encoder modules only
struct ImageBuffer:
    width: int
    height: int
    color: ColorType
    depth: BitDepth
    pixels: bytes               // tightly packed, row-major
    truncated: bool             // I7
    partial_error: CodecError | None

enum ColorType: GRAY | GRAY_ALPHA | RGB | RGBA | PALETTE
enum BitDepth: D1 | D2 | D4 | D8 | D16
struct PixelFormat:
    color: ColorType
    depth: BitDepth
    bytes_per_pixel: int         // derived, cached
// color.py — ownership: transform functions are pure (no shared mutable state);
// concurrency: safe, stateless; serialization: n/a; untrusted-input: none (post-decode only)
struct YCbCrPlanes:
    y: bytes
    cb: bytes
    cr: bytes
    subsample: SubsampleFactor

enum SubsampleFactor: F444 | F422 | F420
// jpeg/dct.py — ownership: pure functions over Block8x8; mutation: returns new block;
// concurrency: safe; persistence: none; untrusted-input: operates only on validated,
// dequantized coefficient blocks (post I6/I12)
struct Block8x8:
    coeffs: array[int32, 64]     // natural order after zigzag reversal
    precision: FixedPointScale   // P11 — explicit, never implicit

struct QuantTable:
    precision: int               // 0 = 8-bit, 1 = 16-bit
    values: array[uint16, 64]    // zigzag order, per §8.4
    id: int                      // 0-3, referenced by ScanComponent

struct QualityScale:
    quality: int                 // 1-100
    scale_factor: int            // derived per §8.3, monotonic (I14)
// jpeg/quant.py — ZigzagTable is a constant lookup, immutable, shared read-only
struct ZigzagTable:
    forward: array[int, 64]      // natural → zigzag position
    inverse: array[int, 64]      // zigzag position → natural
// huffman.py — shared by DEFLATE and JPEG paths; ownership: built once per table,
// immutable after construction (P5); concurrency: safe (read-only lookup);
// untrusted-input: rejects over-subscribed tables at construction (I5)
struct CanonicalCode:
    length: int                  // 1-16 (JPEG) or 1-15 (DEFLATE)
    code: int                    // canonical bit pattern
    symbol: int

struct HuffmanSpec:
    lengths: array[int, MAX_LEN]  // count of codes at each length
    values: list[int]             // canonical value order

struct HuffmanTable:
    spec: HuffmanSpec
    order: BitOrder
    lookup: dict[tuple[int,int], int]   // (length, code) → symbol
    fast_lookup: array[int16, 256]      // first-byte accelerator
    table_class: HuffmanClass            // DC | AC (JPEG only)
// jpeg/entropy.py — ownership: per-scan mutable state (DC predictors, I8);
// concurrency: one ScanComponent set per scan, not shared across threads
struct RunLengthSymbol:
    run: int                      // 0-15
    size: int                     // category, 0-11 (DC) / 0-10 (AC)
    is_eob: bool
    is_zrl: bool

struct ScanComponent:
    component_id: int
    dc_table_id: int
    ac_table_id: int
    dc_predictor: int             // reset at SOI and every restart marker (I8)
// jpeg/markers.py — untrusted-input handling: length fields checked against remaining
// bytes before segment read (P1); EXIF/ICC parsed as inert data only (P13)
enum JpegMarker: SOI | APPn | DQT | DHT | SOF0 | SOF2 | SOS | DRI | RSTn | COM | EOI

struct JfifHeader:
    version_major: int
    version_minor: int
    density_units: int
    x_density: int
    y_density: int

struct ExifSegment:
    byte_order: ExifByteOrder      // little/big-endian TIFF header
    orientation: int | None         // 1-8, applied only when requested (ADR-15)
    gps_present: bool               // flagged for PII policy (T10)
// png/chunks.py — untrusted-input handling: CRC verified before payload use (I4);
// length checked against remaining file bytes before allocation (I2)
struct PngChunk:
    length: int
    chunk_type: ChunkType
    data: bytes
    crc: uint32

enum ChunkType: IHDR | PLTE | IDAT | IEND | TRNS | GAMA | SRGB | ICCP | PHYS | TEXT | ZTXT | ITXT

struct Crc32:
    polynomial: uint32 = 0xEDB88320
    def compute(data: bytes) -> uint32   // signature only, no body
// png/filters.py, png/interlace.py — untrusted-input: row order enforced (I10);
// pass geometry validated against declared dimensions (I9)
enum FilterType: NONE | SUB | UP | AVERAGE | PAETH
struct ScanlineFilter:
    filter_type: FilterType
    bpp: int
    prior_row: bytes | None

struct Adam7Pass:
    index: int                    // 1-7
    start_col: int
    start_row: int
    col_step: int
    row_step: int
    pixel_count: int              // derived, summed check against I9
// deflate/*.py — untrusted-input: streaming size guard against declared bound (I11);
// window bounded to 32 KiB (RFC 1951)
struct Lz77Match:
    length: int                   // 3-258
    distance: int                 // 1-32768

struct DeflateBlock:
    bfinal: bool
    btype: DeflateBtype            // STORED | FIXED_HUFFMAN | DYNAMIC_HUFFMAN | RESERVED
// limits.py, metrics.py, errors.py
struct DecodeLimits:
    max_width: int = 65536
    max_height: int = 65536
    max_pixels: int = 256_000_000
    max_buffer_size: int = 1_073_741_824
    max_decode_seconds: float = 2.0
    bomb_ratio_multiple: int = 4    // I11

struct QualityReport:
    psnr_db: float
    ssim: float
    bits_per_pixel: float
    compression_ratio: float

enum CodecError:
    InvalidSignature
    UnexpectedEof
    UnsupportedFeature(feature: str)
    Malformed(reason: str)
    ChecksumMismatch
    AllocationTooLarge(requested: int, limit: int)
    Io(inner: OSError)
