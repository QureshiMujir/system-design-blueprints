Core schema & types (§11)

//! crates/curve-field/src/fe25519.rs
pub struct FieldElement([u64; 5]);                       // radix-2^51, 5 limbs, P9

impl FieldElement {
    pub fn zero() -> Self;
    pub fn one() -> Self;
    pub fn add(&self, rhs: &Self) -> Self;
    pub fn sub(&self, rhs: &Self) -> Self;
    pub fn mul(&self, rhs: &Self) -> Self;
    pub fn square(&self) -> Self;
    pub fn invert(&self) -> Self;                        // Fermat, §8.1
    pub fn ct_eq(&self, rhs: &Self) -> subtle::Choice;
    pub fn ct_select(a: &Self, b: &Self, c: subtle::Choice) -> Self;
    pub fn to_bytes(&self) -> [u8; 32];                   // forces canonical reduction, I1
    pub fn from_bytes(bytes: &[u8; 32]) -> Self;
}
//! crates/curve-scalar/src/scalar.rs
pub const L: Scalar;                                      // group order, §8.2
pub struct Scalar([u32; 8]);                               // 256-bit, < L

impl Scalar {
    pub fn from_bytes_mod_order(bytes: &[u8; 32]) -> Self;
    pub fn from_bytes_mod_order_wide(bytes: &[u8; 64]) -> Self;  // for nonce/challenge, I3
    pub fn is_canonical(&self) -> subtle::Choice;          // I6
    pub fn add(&self, rhs: &Self) -> Self;
    pub fn mul(&self, rhs: &Self) -> Self;                 // Barrett, §14
    pub fn invert(&self) -> Self;
    pub fn to_bytes(&self) -> [u8; 32];
}

pub struct ClampedScalar(Scalar);                          // X25519-only, I2

impl ClampedScalar {
    pub fn from_bits_clamped(bytes: &[u8; 32]) -> Self;
}
//! crates/curve-edwards/src/point.rs
pub struct EdwardsPoint { x: FieldElement, y: FieldElement, z: FieldElement, t: FieldElement }
pub struct CompressedEdwardsY([u8; 32]);

impl EdwardsPoint {
    pub fn identity() -> Self;
    pub fn add(&self, rhs: &Self) -> Self;                 // HWCD 2008, §15
    pub fn double(&self) -> Self;
    pub fn ct_select(a: &Self, b: &Self, c: subtle::Choice) -> Self;
    pub fn is_small_order(&self) -> bool;                   // I11
    pub fn compress(&self) -> CompressedEdwardsY;
    pub fn mul_by_8(&self) -> Self;                         // cofactor scaling, ADR-08
}

impl CompressedEdwardsY {
    pub fn decompress(&self) -> Result<EdwardsPoint, DecodeError>;  // §8.5, I1
}
//! crates/curve-montgomery/src/point.rs
pub struct MontgomeryU(FieldElement);                       // x-only, §8.3

impl MontgomeryU {
    pub fn ladder(scalar: &ClampedScalar, u: &Self) -> Self;  // RFC 7748 ladder, §8.4
    pub fn is_small_order(&self) -> bool;                     // I11
    pub fn to_bytes(&self) -> [u8; 32];
    pub fn from_bytes(bytes: &[u8; 32]) -> Self;              // masks high bit, RFC 7748 section 5
}
//! crates/ed25519/src/signing_key.rs
pub struct SigningKey { seed: [u8; 32], expansion: [u8; 64], public: VerifyingKey }
pub struct VerifyingKey(CompressedEdwardsY);
pub struct Signature { r: CompressedEdwardsY, s: Scalar }

impl SigningKey {
    pub fn from_seed(seed: &[u8; 32]) -> Self;               // SHA-512(seed) expansion
    pub fn sign(&self, message: &[u8]) -> Signature;         // RFC 8032 section 5.1.6, I3
    pub fn sign_prehashed(&self, hash: &[u8; 64], ctx: &[u8]) -> Signature;  // Ed25519ph, ADR-17
    pub fn verifying_key(&self) -> &VerifyingKey;
}

impl VerifyingKey {
    pub fn verify_strict(&self, message: &[u8], sig: &Signature) -> Result<(), VerifyError>;  // §21
    pub fn verify_lax(&self, message: &[u8], sig: &Signature) -> Result<(), VerifyError>;      // ADR-07
}

impl Signature {
    pub fn to_bytes(&self) -> [u8; 64];
    pub fn from_bytes(bytes: &[u8; 64]) -> Result<Self, DecodeError>;   // I6
}
//! crates/x25519/src/agree.rs
pub struct PublicKey(MontgomeryU);
pub struct SecretKey(ClampedScalar);
pub struct SharedSecret(FieldElement);

impl SecretKey {
    pub fn generate(rng: &mut dyn rand_core::CryptoRng) -> Self;   // caller-supplied CSPRNG only
    pub fn public_key(&self) -> PublicKey;
    pub fn diffie_hellman(&self, peer: &PublicKey) -> Result<SharedSecret, AgreeError>;  // I7
}
//! crates/ed25519/src/error.rs, crates/x25519/src/error.rs
pub enum DecodeError { NonCanonicalEncoding, NotOnCurve, NonCanonicalScalar }
pub enum VerifyError { BadSignature, SmallOrder, Decode(DecodeError) }
pub enum AgreeError { ZeroSharedSecret, SmallOrderPeerKey }
