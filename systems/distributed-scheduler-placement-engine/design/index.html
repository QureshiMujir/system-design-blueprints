Core schema & types (§11)

// types — the object model (normative). Quantities are exact integers (ADR-17, P1).
pub struct Milli(pub i64);                  // CPU in millicores; 1000m = 1 core
pub struct Bytes(pub i64);                  // memory / storage
pub enum ResourceName { Cpu, Memory, EphemeralStorage, Gpu, Extended(SmolStr) }
pub struct ResourceList(BTreeMap<ResourceName, i64>);   // exact; ordered for determinism (I10)
impl ResourceList {
    pub fn fits_in(&self, free: &ResourceList) -> bool;     // node-fit predicate core
    pub fn add(&mut self, o: &ResourceList); pub fn sub(&mut self, o: &ResourceList);
}
pub enum QoSClass { Guaranteed, Burstable, BestEffort }     // derived (I11)
// types — node and pod (normative)
pub struct ResourceVersion(pub u64);        // store mod-revision; the CAS token (I5)
pub struct Uid(pub u128);
pub struct Topology { pub zone: SmolStr, pub rack: SmolStr, pub host: SmolStr }
pub struct Node {
    pub uid: Uid, pub name: SmolStr, pub rv: ResourceVersion,
    pub capacity: ResourceList, pub allocatable: ResourceList,   // allocatable = capacity − reserved
    pub labels: BTreeMap<SmolStr, SmolStr>, pub taints: Vec<Taint>,
    pub topology: Topology, pub conditions: NodeConditions,
}
pub struct Pod {
    pub uid: Uid, pub name: SmolStr, pub rv: ResourceVersion, pub namespace: SmolStr,
    pub requests: ResourceList,                 // summed over containers; placement driver
    pub limits: ResourceList,
    pub priority: i32, pub priority_class: SmolStr,
    pub node_selector: LabelSelector, pub affinity: Affinity, pub tolerations: Vec<Toleration>,
    pub topology_spread: Vec<TopologySpreadConstraint>,
    pub gang: Option<GangRef>,                  // Some → all-or-nothing group (I6)
    pub node_name: Option<SmolStr>,             // set by Binding when Bound
    pub tenant: TenantId,                        // for DRF/quota (I9)
}
// types — policy (normative)
pub struct PriorityClass { pub name: SmolStr, pub value: i32, pub preempting: bool }
pub struct PodDisruptionBudget {                // I7
    pub selector: LabelSelector,
    pub min_available: Option<u32>, pub max_unavailable: Option<u32>,
}
pub struct ResourceQuota { pub scope: TenantId, pub hard: ResourceList, pub used: ResourceList }
// cache — snapshot + accounting (normative, P14/I4)
pub struct NodeInfo {                           // per-node aggregate the scheduler reasons over
    pub node: Arc<Node>,
    pub committed: ResourceList,                 // Σ requests of pods (incl. assumed)
    pub free: ResourceList,                      // allocatable − committed (I1 guard)
    pub pods: Vec<Arc<Pod>>, pub images: HashSet<SmolStr>,
}
pub struct Snapshot {                            // immutable per cycle (I4)
    pub nodes: Arc<[NodeInfo]>, pub generation: u64,
    pub tree: Arc<TopologyTree>,
}
pub trait Cache {
    fn snapshot(&self) -> Snapshot;              // O(1)-ish: hand out the current immutable view
    fn assume(&self, pod: &Pod, node: &str) -> Result<(), AssumeErr>;   // local delta (I2)
    fn forget(&self, pod: &Pod);                 // rollback an assume on conflict (I2)
    fn apply(&self, delta: WatchEvent);          // informer-only writer (P14); reconciles assumes (I15)
}
// queue — three bands (normative, P5/I3)
pub trait SchedulingQueue {
    fn add(&self, pod: Arc<Pod>);                          // → activeQ
    fn pop_active(&self) -> Option<Arc<Pod>>;              // priority+time order
    fn requeue_backoff(&self, pod: Arc<Pod>, attempt: u32); // exponential backoff
    fn park_unschedulable(&self, pod: Arc<Pod>, reason: UnschedReason);
    fn move_request(&self, ev: ClusterEvent);              // re-activate parked pods (P5)
}
pub enum QueueBand { Active, Backoff, Unschedulable, InFlight }   // exactly one (I3)
// framework — extension points (normative, P13)
pub struct CycleState { /* typed key→value; PreFilter→Filter handoff */ }
pub enum Status { Success, Unschedulable(Reason), Error(Err), Wait(Duration) }
pub trait FilterPlugin   { fn filter(&self, p:&Pod, n:&NodeInfo, s:&CycleState) -> Status; }      // hard (P4)
pub trait ScorePlugin    { fn score(&self,  p:&Pod, n:&NodeInfo, s:&CycleState) -> i64;          // soft (P4)
                           fn normalize(&self, scores:&mut [i64]); fn weight(&self) -> i64; }
pub trait PostFilterPlugin { fn postfilter(&self, p:&Pod, snap:&Snapshot, s:&CycleState) -> PreemptResult; }
pub trait ReservePlugin   { fn reserve(&self, p:&Pod, node:&str) -> Status; fn unreserve(&self, p:&Pod, node:&str); }
pub trait PermitPlugin    { fn permit(&self,  p:&Pod, node:&str) -> Status; }   // Wait → gang/quota gate
pub trait BindPlugin      { fn bind(&self, p:&Pod, node:&str, rv:RvSet) -> BindOutcome; }   // CAS (I5)
// fairness — DRF (normative, I9)
pub struct TenantId(pub SmolStr);
pub struct DrfState { shares: HashMap<TenantId, DominantShare> }
pub struct DominantShare(pub f64);             // max over resources of used/total (compare only)
pub trait Fairness {
    fn next_tenant(&self, pending: &[(TenantId, ResourceList)], total: &ResourceList) -> TenantId; // lowest dom share (I9)
    fn on_allocate(&mut self, t: &TenantId, r: &ResourceList);
}
// preempt (normative, I7)
pub struct PreemptResult { pub nominated_node: Option<SmolStr>, pub victims: Vec<Arc<Pod>> }
pub trait Preemptor {
    fn select_victims(&self, p:&Pod, node:&NodeInfo, pdbs:&[PodDisruptionBudget]) -> PreemptResult; // PDB-safe (I7)
}
// bind + store (normative, I5/I8/I14)
pub enum BindOutcome { Bound, Conflict, StoreUnavailable }   // Conflict → forget+requeue; Unavailable → pause (I14)
pub trait Store {
    fn list(&self, prefix:&str) -> Result<(Vec<Object>, ResourceVersion), StoreErr>;
    fn watch(&self, prefix:&str, from: ResourceVersion) -> WatchStream;
    fn cas_bind(&self, pod:&Pod, node:&str, rv_pod:ResourceVersion, rv_node:ResourceVersion)
        -> Result<BindOutcome, StoreErr>;                    // Txn compare-mod-revision (I5)
    fn lease_grant(&self, ttl: Duration) -> Result<LeaseId, StoreErr>;      // #199
    fn lease_keepalive(&self, id: LeaseId) -> Result<(), StoreErr>;
}
pub struct PhaseDeferredError { pub reference: &'static str }   // e.g. PhaseDeferred("FINAL_BLUEPRINT §24.1")
