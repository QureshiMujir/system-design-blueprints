Core schema & types (§11)

type Operation = {
  id: OperationId;                   // ULID, monotonic in append order; immutable
  tenantId: TenantId;
  serviceId: ServiceId;              // logical service that issued the op
  verb: VerbId;                      // catalogue entry, e.g. "pg.update.orders"
  args: SerializedArgs;               // canonical-JSON (ADR-05), schema-validated (I8)
  result?: SerializedResult;          // present iff the op completed
  causalPredecessors: OperationId[];  // dependency edges consumed by the saga coordinator
  sagaId?: SagaId;                    // present iff inside a saga
  startedAt: Timestamp;               // wall-clock, external-facing only (ADR-22)
  completedAt?: Timestamp;
  status: "pending" | "ok" | "failed" | "forward-only";
  hash: string;                       // H(args || result || prevHash) — I2
  prevHash: string;
};
// Ownership: written once by the interceptor/SDK, never mutated after `status` settles.
// Concurrency: append-only; reads are lock-free (MVCC via Postgres).
// Persistence: hot tier Postgres, cold tier S3 after 30 days (ADR-03/04).
// Compatibility: envelope carries `__rewindVersion`; N/N-1 readable (I7).
type Inverse = {
  id: InverseId;
  operationId: OperationId;           // back-reference, 1:1 with Operation
  strategy: "compensating" | "tombstone" | "quiesce-replay" | "forward-only";
  payload?: SerializedInversePayload;
  idempotencyKey: IdempotencyKey;      // I3
  appliedAt?: Timestamp;
  appliedBy?: ActorId;                 // operator, agent, or "auto"
  result?: "applied" | "skipped" | "conflicted" | "failed";
  hash: string;
  prevHash: string;
};
// Ownership: written by the same actor that wrote the Operation; result field mutated
//   exactly once, at apply time, guarded by the idempotency-store lookup.
// Concurrency: apply is serialized per operationId via the idempotency store (I3).
// Persistence/compatibility: identical tier and versioning rules as Operation.
type Saga = {
  id: SagaId;
  tenantId: TenantId;
  name: string;                        // e.g. "checkout.complete"
  status: "open" | "committed" | "compensating" | "compensated" | "abandoned"; // §8.1
  operations: OperationId[];           // ordered by insertion; NOT the causal order
  startedAt: Timestamp;
  closedAt?: Timestamp;
  policy: SagaPolicy;                  // retry budget, timeout, on-conflict strategy (P14)
};
// Ownership: mutated only by the saga coordinator; replicated via etcd/Raft (ADR-07)
//   from Phase 7 onward — Phase 3's skeleton keeps it single-instance.
// Concurrency: one active state-transition per Saga at a time (I11).
type Checkpoint = {
  id: CheckpointId;
  tenantId: TenantId;
  at: Timestamp;
  pgLsn: string;                       // Postgres log-sequence-number anchor
  operationCursor: OperationId;        // last operation included
  inverseCursor: InverseId;            // last inverse included
  hashAnchor: string;                  // hash-chain anchor at this checkpoint
  retentionClass: "hot" | "warm" | "cold" | "audit";
};
// Written every 60s (configurable). Purposes: dependency-graph cut points for rewinds
//   (§8.2), sandbox base images (§20 Phase 8, Chandy & Lamport 1985, §5), compliance
//   attestations (§25).
type IdempotencyKey = {
  scope: "verb" | "saga" | "tenant";
  value: string;                       // SHA-256 of canonical-args (+ tenant/saga if scoped)
  ttl: Seconds;                        // must satisfy I9
};
// Storage: a Postgres table indexed on (scope, value) with a TTL sweep job.
export interface ReversibleOperation<Args, Result> {
  readonly verb: VerbId;
  readonly version: string;            // op semver; schema changes bump the major (I7)
  forward(args: Args, ctx: OperationContext): Promise<Result>;
  buildInverse(args: Args, result: Result, ctx: OperationContext): Inverse | null;
  applyInverse(inverse: Inverse, ctx: RewindContext): Promise<InverseResult>;
  preconditions?(args: Args): VerbId[];              // feeds the causal-order DAG (P8)
  postRewindInvariants?(ctx: RewindContext): Invariant[];
}
export interface VerbCatalogue {
  register<A, R>(op: ReversibleOperation<A, R>): void;
  declareForwardOnly(verb: VerbId, reason: string): void;   // P7
  resolve(verb: VerbId): ReversibleOperation<unknown, unknown> | null;
}
export interface RewindSDK {
  http<R>(req: HttpRequest, options: WrapOptions): Promise<R>;
  publish(topic: string, msg: unknown, options: WrapOptions): Promise<void>;
  saga<R>(name: string, fn: (s: SagaHandle) => Promise<R>): Promise<R>;
  withTenant<R>(tenantId: string, fn: () => Promise<R>): Promise<R>;
}
type AgentAction = {
  agentId: AgentId;
  actionId: ActionId;
  parentAction?: ActionId;             // for nested/sub-agent actions
  policyRef: PolicyId;                 // I12 — must resolve at apply time
  operation: Operation;                // wraps the underlying reversible operation
};
type AgentPolicy = {
  id: PolicyId;
  quota: { maxActionsPerHour: number };
  allowlist: VerbPattern[];            // e.g. "^pg\\.update\\.public\\.user_.*$"
  blastRadiusCap: number;              // max rows/records a single action may touch (P10)
  coSignVerbs: VerbId[];               // verbs requiring human co-sign regardless of quota
};
