Core schema & types (§11)

export interface Agent<I extends StepInput, O extends StepOutput> {
  readonly id: AgentId;                 // e.g. "coder.node-express"
  readonly version: string;             // semver; pinned per Pipeline (I10)
  readonly inputs: ReadonlyArray<InputDecl>;
  readonly outputs: ReadonlyArray<OutputDecl>;
  readonly retry: RetryPolicy;          // ADR-15
  readonly budget: CostBudget;          // I6, I9
  readonly escalation: EscalationPolicy;
  execute(input: I, ctx: AgentContext): Promise<O>;
  summarise?(output: O): Promise<string>;
}
export interface StepInput {
  runId: RunId; stepId: StepId;
  spec: PipelineSpec;                   // redacted of secret values (I8)
  upstream: ReadonlyArray<{ stepId: StepId; artifact: ArtifactRef }>;
  budget: CostBudget;
  modelHint?: ModelHint;
}
export interface StepOutput {
  status: "ok" | "retry" | "block" | "escalate" | "failed";
  artifacts: ReadonlyArray<ArtifactRef>;       // content-addressed (I5)
  metrics: { tokensIn: number; tokensOut: number; wallMs: number; costUsd: number };
  notes: string;
  reviewer?: ReviewerVerdict;
}
export interface ReviewerVerdict {
  decision: "approve" | "block";
  severity?: "low" | "medium" | "high" | "critical";
  findings: ReadonlyArray<{
    code: string; severity: "low" | "medium" | "high" | "critical";
    location: string; message: string; suggestedFix?: string;
  }>;
  binding: true;                        // P3, I3 — never false
}
export interface PipelineDAG {
  id: PipelineDagId; version: string;
  nodes: ReadonlyArray<DagNode>; edges: ReadonlyArray<DagEdge>;
  reviewerNode: NodeId; deployNode: NodeId;
}
export interface DagNode {
  id: NodeId; agent: AgentId; config: Record<string, unknown>;
  inputs: ReadonlyArray<NodeId>; budget: CostBudget; retry: RetryPolicy;
}
export interface DagEdge { from: NodeId; to: NodeId; artifactKey: string; }
export interface PipelineSpec {
  id: SpecId; version: "1.0"; intent: string; stack: StackId;
  target: DeployTarget;
  budgets: { maxCostUsd: number; maxWallMinutes: number };
  policy: { requireReviewer: true; autoDeploy: boolean; secrets: SecretBindingRef[] };
  metadata: { createdBy: UserId; orgId: OrgId; createdAt: ISODate; sourceLabel?: string };
}
export interface Pipeline {
  id: PipelineId; specId: SpecId; dag: PipelineDAG;
  agentVersions: ReadonlyArray<{ agentId: AgentId; version: string }>;
  templateId: StackTemplateId; templateVersion: string; createdAt: ISODate;
}
export interface Run {
  id: RunId; pipelineId: PipelineId;
  status: "pending" | "planning" | "running" | "blocked" | "succeeded" | "failed" | "cancelled";
  steps: ReadonlyArray<{ nodeId: NodeId; status: StepStatus; startedAt?: ISODate;
    finishedAt?: ISODate; output?: StepOutput; attempts: number }>;
  costUsd: number; startedAt: ISODate; finishedAt?: ISODate;
  reviewerVerdict?: ReviewerVerdict; deployedUrl?: string;
}
export interface Artifact {
  id: ArtifactId;                       // sha256 of bytes (ADR-13)
  runId: RunId; producerStepId: StepId;
  kind: "source-tar" | "test-report" | "tf-plan" | "tf-state" | "dockerfile"
      | "ci-workflow" | "review-report" | "deploy-log" | "deploy-url";
  s3Key: string; sizeBytes: number; createdAt: ISODate;
  redactedFields: ReadonlyArray<string>;
}
