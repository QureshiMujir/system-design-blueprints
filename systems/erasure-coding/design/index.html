Core schema & types (§11)

// ec-core/src/gf256.rs  (§8.1, P1, I3)
pub struct Gf256(pub u8);
impl Gf256 {
    pub const ZERO: Self; pub const ONE: Self;
    pub const fn new(b: u8) -> Self;
    pub fn add(self, other: Self) -> Self;
    pub fn mul(self, other: Self) -> Self;   // table lookup, ec-gf256
    pub fn inv(self) -> Option<Self>;        // None iff self == ZERO
    pub fn div(self, other: Self) -> Option<Self>;
}
pub const FIELD_SIZE: usize = 256;
pub const IRREDUCIBLE_POLY: u16 = 0x11D;     // ADR-04
// ec-core/src/matrix.rs  (§8.2-8.6, P2, I4, I9)
pub struct Matrix { rows: usize, cols: usize, data: Vec<Gf256> }
impl Matrix {
    pub fn identity(n: usize) -> Self;
    pub fn from_rows(rows: Vec<Vec<Gf256>>) -> Self;
    pub fn at(&self, r: usize, c: usize) -> Gf256;
    pub fn multiply(&self, other: &Matrix) -> Matrix;
    pub fn invert(&self) -> Result<Matrix, MatrixError>;   // Gauss-Jordan, §8.6
    pub fn submatrix(&self, rows: &[usize], cols: &[usize]) -> Matrix;
}
pub enum MatrixError { NotSquare, Singular }
// ec-core/src/shard.rs  (§8.9, I6, I13)
pub struct ShardIndex(pub u16);
pub struct ShardHeader {
    pub magic: [u8; 4], pub version: u16, pub kind: ShardKind,
    pub index: ShardIndex, pub k: u8, pub m: u8,
    pub payload_bytes: u32, pub crc32c: u32, pub reserved: [u8; 12],
}
pub enum ShardKind { Data, Parity, LrcLocal }
pub struct Shard { pub header: ShardHeader, pub bytes: bytes::Bytes }
pub const SHARD_HEADER_SIZE: usize = 32;
pub const SHARD_PAYLOAD_MAX: usize = 64 * 1024 * 1024;
// ec-core/src/params.rs + encoding_matrix.rs  (P9, I7, ADR-01/02/06)
pub enum Flavour { Cauchy, Vandermonde, Lrc(u8, u8) }  // (locality groups, per-group)
pub struct EncodingMatrix { pub k: u8, pub m: u8, pub matrix: Matrix, pub flavour: Flavour }
impl EncodingMatrix {
    pub fn cauchy(k: u8, m: u8) -> Result<Self, MatrixError>;
    pub fn vandermonde(k: u8, m: u8) -> Result<Self, MatrixError>;
    pub fn lrc_azure(k: u8, l: u8, r: u8) -> Result<Self, MatrixError>;
    pub fn decode_matrix(&self, present: &[bool]) -> Result<Matrix, MatrixError>;
}
pub struct CodecParams { pub k: u8, pub m: u8, pub flavour: Flavour }
impl CodecParams {
    pub fn new(k: u8, m: u8) -> Result<Self, CodecError>;   // enforces I7
    pub fn n(&self) -> u16;
}
pub enum CodecError { ZeroK, ZeroM, TooManyShards }
pub const RS_6_3: CodecParams;      // (6,3) Cauchy
pub const RS_17_3: CodecParams;     // (17,3) Cauchy — Backblaze shape
pub const LRC_12_2_2: CodecParams;  // (12,4) Lrc(2,2) — Azure shape
// ec-core/src/repair_plan.rs  (P5, I10, §19)
pub struct RepairPlan { pub reads: Vec<ShardIndex>, pub ops: Vec<RepairOp>,
                        pub bandwidth: u64, pub flops: u64 }
pub enum RepairOp {
    InvertSubmatrix { rows: Vec<u8>, cols: Vec<u8> },
    MultiplyAndCombine { coeffs: Vec<Gf256>, into: ShardIndex },
    XorCombine { sources: Vec<ShardIndex>, into: ShardIndex },
}
pub trait RepairPlanner: Send + Sync {
    fn plan(&self, params: CodecParams, present: &[bool]) -> Result<RepairPlan, PlannerError>;
}
pub struct RsPlanner; pub struct LrcPlanner; pub struct HitchhikerPlanner;
pub enum PlannerError { Length, Insufficient }
// ec-core traits: Encoder / Decoder / Streaming  (P3, P6, §15-16, §23)
pub trait Encoder: Send + Sync {
    fn params(&self) -> CodecParams;
    fn encode(&self, data: &[&[u8]]) -> Result<Vec<Vec<u8>>, EncodeError>;
    fn encode_into(&self, data: &[&[u8]], parity: &mut [&mut [u8]]) -> Result<(), EncodeError>;
}
pub trait Decoder: Send + Sync {
    fn params(&self) -> CodecParams;
    fn decode(&self, present: &[Option<&[u8]>]) -> Result<Vec<Vec<u8>>, DecodeError>;
}
pub trait StreamingEncoder: Send + Sync {
    fn chunk_size(&self) -> usize;
    fn feed(&mut self, chunk: &[u8]) -> Result<EncodedChunk, EncodeError>;
    fn flush(&mut self) -> Result<(), EncodeError>;
}
pub struct EncodedChunk { pub data: Vec<Vec<u8>>, pub parity: Vec<Vec<u8>>, pub offset: u64 }
pub enum EncodeError { ShardCount, ShardLen, FlavourMismatch }
pub enum DecodeError { ShardCount, Insufficient, Matrix(MatrixError) }
