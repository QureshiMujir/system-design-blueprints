Core schema & types (§11)

type Key = bytes::Bytes;
type Value = bytes::Bytes;

struct KeyRange { start: Key /* inclusive */, end: Key /* exclusive */ }
struct RangeId(u64);   // stable across membership changes

impl KeyRange { fn contains(&self, key: &[u8]) -> bool; }
struct Hlc { physical: u64 /* ms since epoch */, logical: u32, node_id: u32 }
struct VersionedValue { value: Value, ts: Hlc, tombstone: bool }
type Hash256 = [u8; 32];
struct RangeFingerprint { range_id: RangeId, root_hash: Hash256, depth: u8,
                           leaf_count: u32, generation: u64 }
struct MerkleTree { leaves: Vec<Hash256>, internal: Vec<Hash256>, depth: u8 }

impl MerkleTree {
    fn build<I: Iterator<Item = LeafEntry>>(depth: u8, entries: I) -> Self;
    fn root(&self) -> Hash256;
    fn children_of(&self, node: usize) -> (usize, usize);
    fn node_at(&self, node: usize) -> Hash256;
    fn is_leaf_index(&self, node: usize) -> bool;
}
struct Hint { target: NodeId, range_id: RangeId, key: Key, value: VersionedValue,
              created_at: SystemTime, expires_at: SystemTime }
struct HintBatch { hints: Vec<Hint>, batch_id: u64 }
struct RepairTask { range_id: RangeId, peers: Vec<NodeId>, mode: RepairMode,
                     priority: RepairPriority }
enum RepairMode { Active, OnRead, Hinted, Manual }
enum RepairPriority { Background, Normal, High }   // Ord: Background < Normal < High
trait Resolver: Send + Sync + 'static {
    fn resolve(&self, values: &[VersionedValue]) -> ResolveOutcome;   // CAI contract (I6)
}
enum ResolveOutcome { Single(VersionedValue), Siblings(Vec<VersionedValue>), Tombstone(Hlc) }
trait AntiEntropyEngine: Send + Sync + 'static {
    async fn fingerprint(&self, range: RangeId) -> Result<RangeFingerprint, Error>;
    async fn compare(&self, peer: NodeId, fp: RangeFingerprint) -> Result<Vec<KeyRange>, Error>;
    async fn fetch(&self, peer: NodeId, sub_range: KeyRange) -> Result<Vec<(Key, VersionedValue)>, Error>;
    async fn apply(&self, updates: Vec<(Key, VersionedValue)>) -> Result<(), Error>;
}
struct Cell { id_xor: u64, hash_xor: u64, count: i32 }
struct Iblt { cells: Vec<Cell>, hash_count: u8 /* = 3 */, seed: u64 }

impl Iblt {
    fn add(&mut self, element_id: u64);
    fn sub(&self, other: &Self) -> Self;
    fn decode(self) -> Result<(Vec<u64>, Vec<u64>), DecodeError>;
}
struct HintLog { db: Arc<rocksdb::DB>, cap_bytes: u64, cap_per_target_bytes: u64,
                  ttl: Duration }

impl HintLog {
    fn enqueue(&self, hint: Hint) -> Result<(), Error>;          // enforces I4
    fn drain_for_target(&self, target: NodeId) -> Result<HintIterator<'_>, Error>;
}
