Core schema & types (§11)

// own: sidecar process, immutable after construction | mut: none | conc: Send+Sync
// pers: none (derived at startup from SA token) | ser: not wire-serialized directly
// compat: stable across versions
pub struct WorkloadId { kube_namespace: String, kube_service_account: String, pod_uid: String }

// own: mesh-ca + sidecar | mut: none (value type) | conc: Send+Sync, Copy-cheap (Arc<str> inner)
// pers: none, derived from CertBundle SAN | ser: URI string form (§8.4 grammar)
// compat: grammar-locked, additive path segments only
pub struct SpiffeId { trust_domain: String, namespace: String, service_account: String }

// own: sidecar (via CertManager) | mut: replaced wholesale on rotation, never mutated in place
// conc: wrapped in ArcSwap<Svid> for lock-free hot-path reads (P10)
// pers: in-memory only, never written to disk | ser: X.509 DER within CertBundle
// compat: not_after must be monotonically increasing across rotations (I14)
pub struct Svid { spiffe_id: SpiffeId, cert_der: Vec<u8>, not_before: SystemTime, not_after: SystemTime }

// own: sidecar | mut: swapped atomically (ArcSwap, P6) | conc: lock-free reader path
// pers: none | ser: PEM cert_pem/key_pem/chain_pem fields | compat: triple swapped together, never partially
pub struct CertBundle { svid: Svid, key_pem: Vec<u8>, chain_pem: Vec<u8> }

// own: mesh-ca (root authority) | mut: rotates on CA root rotation only (rare) | conc: Send+Sync
// pers: root key sealed per ADR-08 | ser: trust-domain name string | compat: one trust domain (P12)
pub struct TrustDomain { name: String, root_cert_der: Vec<u8> }
// own: xDS client, referenced by ConfigSnapshot | mut: replaced on LDS push, never edited in place
// conc: read-shared across accept loops via ConfigSnapshot | pers: none (control-plane sourced)
// ser: protobuf Listener message | compat: filter_chains additive; port change requires new Listener
pub struct Listener { name: String, address: SocketAddr, filter_chains: Vec<FilterChain>, transparent: bool }

// own: Listener | mut: immutable once part of an active snapshot | conc: shared via Arc
// pers: none | ser: protobuf FilterChain | compat: filter order is part of the wire contract
pub struct FilterChain { filter_chain_match: Option<FilterChainMatch>, filters: Vec<NetworkFilter> }

// own: xDS client / RDS | mut: replaced wholesale per push (I4) | conc: Arc-shared, read-only on hot path
// pers: none | ser: protobuf RouteConfiguration | compat: version-tagged (VersionInfo)
pub struct RouteConfig { name: String, version: VersionInfo, virtual_hosts: Vec<RouteMatch> }

// own: RouteConfig | mut: immutable | conc: shared | pers: none
// ser: protobuf RouteMatch (uri/method/header StringMatch variants) | compat: additive match kinds only
pub struct RouteMatch { uri: Option<StringMatch>, headers: BTreeMap<String, StringMatch>, route: Vec<WeightedCluster> }

// own: RouteMatch | mut: immutable | conc: shared | pers: none | ser: protobuf WeightedCluster
// compat: weights normalized at compile time, not on the wire (§8.6)
pub struct WeightedCluster { cluster_name: String, weight: u32 }

// own: xDS client / CDS | mut: replaced wholesale per push | conc: Arc-shared across cluster manager
// pers: none | ser: protobuf Cluster | compat: lb_policy change requires full cluster replace, not patch
pub struct ClusterConfig { name: String, lb_policy: LbPolicy, circuit_breakers: RetryBudget, outlier_detection: Option<OutlierState> }
// own: EDS / cluster manager | mut: health/weight updated per EDS push, address is immutable
// conc: per-endpoint atomics for health counters (P10) | pers: none | ser: protobuf LbEndpoint
// compat: locality/metadata fields additive
pub struct Endpoint { address: SocketAddr, health: EndpointHealth, weight: u32, locality: Option<String> }

// own: Endpoint | mut: transitions per §8.7 state table | conc: atomic enum-like (u8 CAS)
// pers: none | ser: protobuf HealthStatus enum | compat: new states must be additive, not renumbered
pub enum EndpointHealth { Healthy, Unhealthy, Ejected, RemovedPending }

// own: EDS producer | mut: replaced wholesale per cluster per push | conc: Arc-shared
// pers: none | ser: protobuf ClusterLoadAssignment | compat: cluster_name is the join key, stable
pub struct LoadAssignment { cluster_name: String, endpoints: Vec<Endpoint> }

// own: xDS server/client shared envelope | mut: immutable per message | conc: Send
// pers: none | ser: protobuf Any{type_url, value} | compat: type_url enumerates ResourceType
pub struct XdsResource { type_url: String, version: VersionInfo, resource_bytes: Vec<u8> }

// own: xDS protocol layer | mut: n/a (enum) | conc: Copy | pers: none | ser: type_url string mapping
// compat: closed set matching §8.2's five rows, extending requires a new row there first
pub enum ResourceType { Listener, RouteConfiguration, Cluster, ClusterLoadAssignment, Secret }
// own: AckState / snapshot cache | mut: monotonically increasing per resource type (I2)
// conc: compared under the snapshot-cache's per-node lock (control-plane side only)
// pers: none | ser: version_info string | compat: string, not integer — allows non-numeric schemes
pub struct VersionInfo(String);

// own: DiscoveryResponse/Request pairing | mut: single-use per exchange | conc: Send
// pers: none | ser: nonce string | compat: opaque, server-generated, echoed verbatim by client
pub struct Nonce(String);

// own: xDS client per resource type | mut: updated on every ACK/NACK (§8.2) | conc: single-writer (client task)
// pers: none | ser: not wire-serialized, client-local | compat: n/a, internal
pub struct AckState { last_acked_version: VersionInfo, last_nonce: Nonce, pending_nack: bool }

// own: sidecar (xDS client) | mut: replaced wholesale, never edited in place (P11, I4)
// conc: ArcSwap<ConfigSnapshot>, lock-free reads on every accept/route (P10)
// pers: none | ser: not wire-serialized (assembled client-side from XdsResource set)
// compat: a snapshot is internally consistent by construction — no partial-apply state exists
pub struct ConfigSnapshot { listeners: Vec<Listener>, routes: Vec<RouteConfig>, clusters: Vec<ClusterConfig>, endpoints: Vec<LoadAssignment>, version: VersionInfo }
// own: PeerAuthentication compiler output | mut: n/a (enum) | conc: Copy | pers: none
// ser: protobuf enum | compat: closed set, STRICT/PERMISSIVE/DISABLE per §8.9
pub enum MtlsMode { Strict, Permissive, Disable }

// own: mTLS handshake result, attached to connection state | mut: set once per connection, immutable after
// conc: Send (moved into RequestCtx-equivalent) | pers: none | ser: SPIFFE URI string form
// compat: absent (None) means unauthenticated — never defaulted to a false identity
pub struct PeerIdentity { spiffe_id: SpiffeId, cert_not_after: SystemTime }

// own: mesh-policy compiler | mut: recompiled wholesale on CRD change | conc: Arc-shared, read-only on hot path
// pers: none | ser: internal compiled form, not wire | compat: default effect is Deny (I6)
pub struct AuthzPolicy { principal_match: SpiffeIdPattern, route_match: RouteMatch, effect: PolicyEffect }

// own: authz_eval output, per-request | mut: single-assignment | conc: Send | pers: none
// ser: not wire | compat: n/a
pub enum PolicyDecision { Allow, Deny { reason: String } }
// own: ClusterConfig | mut: counters reset per window, ceiling immutable | conc: atomic counters (P10)
// pers: none | ser: protobuf CircuitBreakers | compat: percentage fields, not absolute counts, across cluster resizes
pub struct RetryBudget { max_retries_percent: u32, max_pending_requests: u32, max_connections: u32 }

// own: cluster_manager per connection pool | mut: transitions per §8.7 breaker table | conc: atomic CAS
// pers: none | ser: not wire (derived, local) | compat: n/a
pub enum CircuitState { Closed, Open { since: Instant }, HalfOpen }

// own: lb.rs per endpoint | mut: transitions per §8.7 health table | conc: atomic CAS
// pers: none | ser: not wire (derived, local) | compat: n/a
pub struct OutlierState { consecutive_5xx: u32, ejected_until: Option<Instant>, ejection_count: u32 }
// own: mesh-inject (generation time), mesh-proxy (self-check at startup) | mut: generated once per pod
// conc: n/a (data, not runtime state) | pers: written into iptables/eBPF program, not the mesh's own store
// ser: rule-table row form (§8.1) | compat: chain/match/action vocabulary is closed and versioned with ADR-03
pub struct InterceptRule { chain: String, direction: Direction, matcher: String, action: RedirectAction, exclusion: Option<String> }

// own: mesh-telemetry | mut: append-only, single-writer per request | conc: Send, moved to async logger
// pers: none in-process — exported to external backend (§1 delegates) | ser: structured JSON
// compat: field set is the cardinality budget (P13); adding a field requires a §26 review
pub struct TelemetryRecord { timestamp: SystemTime, cluster: String, method: String, response_code: u16, duration_ms: u32, peer_principal: Option<SpiffeId>, trace_id: Option<String> }

// own: every crate, propagated via Result | mut: n/a (enum) | conc: Send+Sync (no borrowed data)
// pers: none | ser: numeric code + string detail for xDS error_detail / audit logs
// compat: variants are additive; removing one is a breaking change per §27
pub enum MeshError {
    InterceptRuleConflict, XdsVersionRegression, XdsNack { detail: String },
    CsrIdentityMismatch, CertExpired, PolicyEvalFailed, RetryBudgetExhausted,
    OutlierCapExceeded, ResourceWarmingTimeout, AdminBindViolation,
}
