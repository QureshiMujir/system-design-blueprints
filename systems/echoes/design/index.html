Core schema & types (§11)

// Branded identifiers — nominal typing prevents cross-ID mixups
type Brand<T, B> = T & { __brand: B };
type MessageId = Brand<string, "MessageId">;   // ULID; immutable; never in audit plaintext
type VaultId    = Brand<string, "VaultId">;     // ULID; owner-scoped; FK in Postgres
type EchoId     = Brand<string, "EchoId">;      // ULID; 1:1 with subject's AIEcho
type UserId     = Brand<string, "UserId">;
type SubjectId  = UserId;                        // alias used in DeliveryRule/AIEcho context

interface EchoesCore {
  composeMessage(input: ComposeInput): Promise<Message>;
  updateMessage(messageId: MessageId, patch: MessagePatch): Promise<Message>;
  withdrawMessage(messageId: MessageId, reason: WithdrawReason): Promise<Tombstone>;
  setDelivery(messageId: MessageId, rule: DeliveryRule): Promise<DeliveryHandle>;
  verifyRecipient(messageId: MessageId, attempt: VerificationAttempt): Promise<VerificationResult>;
  open(messageId: MessageId, evidence: VerificationEvidence): Promise<OpenedMessage>;
  trainEcho(input: EchoTrainingInput): Promise<EchoTrainingJob>;
  authoriseEchoDialogue(echoId: EchoId, recipient: RecipientRef): Promise<EchoGrant>;
  openEchoDialogue(grant: EchoGrant): Promise<EchoSession>;
  setTrustee(vaultId: VaultId, trustee: TrusteeDesignation): Promise<TrusteeRecord>;
  assertDeathEvent(vaultId: VaultId, evidence: DeathEvidence): Promise<DeathReviewCase>;
}

type WithdrawReason = "sender-request" | "compliance-hold" | "duplicate" | "duress";
type EchoTrainingInput = { subjectId: UserId; sources: CorpusSourceRef[]; consent: ConsentRecord[] };

class EchoesError extends Error {
  code: EchoesErrorCode;
  retryable: boolean;
}
type EchoesErrorCode =
  | "IDENTITY_UNRESOLVED" | "SCHEMA_VERSION_UNSUPPORTED" | "QUORUM_NOT_MET"
  | "TOMBSTONED" | "RATE_LIMITED" | "RULE_DEPTH_EXCEEDED" | "DIGEST_MISMATCH";
