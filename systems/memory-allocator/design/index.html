typedef struct smalloc_stats {
    uint64_t allocated_bytes;
    uint64_t resident_bytes;
    uint64_t mapped_bytes;
    uint64_t allocations;
    uint64_t frees;
    uint64_t arenas;
    uint64_t threads;
    uint64_t large_objects;
    uint64_t fragmentation_bytes;
} smalloc_stats_t;
typedef struct sm_arena {
    sm_lock_t     lock;                         /* ADR-11 */
    sm_seglist_t  bins[SM_NUM_CLASSES];         /* Phase 15 */
    sm_buddy_t    buddy;                        /* Phase 16 */
    sm_slab_t    *slabs[SM_NUM_SLAB_CLASSES];    /* Phase 17 */
    sm_atomic_u64 bytes_allocated;
    sm_atomic_u64 bytes_resident;
    sm_atomic_u64 bytes_mapped;
    sm_atomic_u64 alloc_calls;
    sm_atomic_u64 free_calls;
    uint32_t      id;
} sm_arena_t;
typedef struct sm_tcache {
    sm_arena_t *arena;
    void       *bins[SM_NUM_TC_CLASSES];
    uint16_t    fill[SM_NUM_TC_CLASSES];
    uint16_t    limit[SM_NUM_TC_CLASSES];       /* I8 */
    sm_mpsc_t   remote_free_queue;              /* I13 */
    uint64_t    seq;
    uint64_t    foreign_frees;
} sm_tcache_t;

extern __thread sm_tcache_t sm_tcache;
typedef struct sm_block_hdr {
    uint64_t magic;      /* SM_MAGIC, randomized per process (T5 mitigation) */
    uint64_t size;
    uint32_t arena_id;
    uint16_t class_id;
    uint16_t flags;      /* SM_HDR_LARGE | SM_HDR_ALIGNED | ... */
} sm_block_hdr_t;
static const uint16_t sm_size_classes[SM_NUM_CLASSES] = {
    8, 16, 32, 48, 64, 80, 96, 112, 128,
    160, 192, 224, 256, 320, 384, 448, 512,
    640, 768, 896, 1024, 1280, 1536, 1792, 2048,
    2560, 3072, 3584, 4096
};
typedef struct sm_buddy_block {
    struct sm_buddy_block *next;
    struct sm_buddy_block *prev;
    uint8_t                order;   /* 12..21, ADR-09 */
    uint8_t                state;   /* SM_FREE | SM_USED | SM_SPLIT */
} sm_buddy_block_t;
typedef struct sm_slab_cache {
    const char    *name;
    size_t         object_size;
    size_t         align;
    sm_slab_t     *partial;
    sm_slab_t     *full;
    sm_slab_t     *empty;
    sm_lock_t      lock;
    sm_magazine_t *mag_full[SM_MAX_CPUS];      /* ADR-10 */
    sm_magazine_t *mag_partial[SM_MAX_CPUS];
} sm_slab_cache_t;
typedef enum {
    SM_OPT_HARDEN_REDZONE = 1, SM_OPT_HARDEN_CANARY = 2,
    SM_OPT_HARDEN_DFD = 3,     SM_OPT_HARDEN_QUARANTINE = 4,
    SM_OPT_HARDEN_GUARD_LARGE = 5, SM_OPT_TCACHE_LIMIT = 6,
    SM_OPT_ARENA_COUNT = 7,    SM_OPT_LARGE_THRESHOLD = 8,
    SM_OPT_MTRACE_FILE = 9,    SM_OPT_DECOMMIT_DELAY_MS = 10
} sm_opt_t;

typedef enum {
    SM_OK = 0, SM_E_INVAL = -1, SM_E_NOMEM = -2, SM_E_FAULT = -3,
    SM_E_RANGE = -4, SM_E_DOUBLE_FREE = -5, SM_E_REDZONE = -6
} sm_status_t;
