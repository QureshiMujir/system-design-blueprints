Core schema & types (§11)

enum SymbolicValue
  Concrete(u64)
  Symbolic(SymbolicExprId)
  Hybrid(SymbolicExprId, u64)
struct SymbolicExpr
  expr_id: ExprId
  opcode: ExprOpcode
  operands: SmallVec<ExprId>
  width_bits: u16
  hash: u128
struct ConstraintSet
  path_id: PathId
  assertions: Vec<ConstraintId>
  solver_epoch: u64
struct ExecutionState
  registers: RegisterBank
  memory: SymbolicMemory
  constraints: ConstraintSet
  pc: VirtualAddress
  status: ExecutionStatus
  lineage: LineageMetadata
struct SnapshotRecord
  snapshot_id: u64
  parent_snapshot: Option<u64>
  wal_offset: u64
  compressed_pages: Vec<PageRef>
  checksum: [u8; 32]
struct SchedulerLease
  worker_id: WorkerId
  path_id: PathId
  lease_expiry: Instant
trait SolverBackend
  fn assert_incremental(&mut self, query: &SolverQuery) -> Result<(), SolverError>;
  fn check_sat(&mut self, timeout: Duration) -> SolverOutcome;
  fn unsat_core(&self) -> Vec<ConstraintId>;
  fn push(&mut self);
  fn pop(&mut self);
