Core schema & types (§11)

// ---- identifiers & primitives ----
type NodeId = u32;                  // < n; tie-break order (I19)
type EdgeId = u32;                  // < m
type Weight = u32;                  // deciseconds/meters; u32::MAX = INF (I16)
const INF: Weight = u32::MAX;
struct Coord { lat_e7: i32, lon_e7: i32 }   // fixed-point deg ×1e7 (ADR-6)

// ---- CSR graph ----
struct Edge { target: NodeId, weight: Weight }
struct Csr  { head: Vec<u32> /*len n+1*/, edges: Vec<Edge> /*len m*/ }   // I3
struct Graph { fwd: Csr, rev: Csr, coords: Vec<Coord>, n: u32, m: u32 }
impl Graph {
  fn out(&self, v: NodeId) -> &[Edge];        // edges[head[v]..head[v+1]]
  fn r#in(&self, v: NodeId) -> &[Edge];       // reverse adjacency
}

// ---- priority queue (4-ary, lazy-delete) ----
struct Queue { heap: Vec<(Weight, NodeId)>, /* dist owned by search */ }   // ADR-8
fn push(&mut self, w: Weight, v: NodeId);
fn pop_min(&mut self) -> Option<(Weight, NodeId)>;   // skip stale entries

// ---- search scratch (pooled, reused per thread; I13/P13) ----
struct SearchSpace { dist: Vec<Weight>, parent: Vec<EdgeId>, touched: Vec<NodeId> }
fn reset(&mut self);                          // clear only touched entries (O(touched))

// ---- results ----
struct Path { edges: Vec<EdgeId>, nodes: Vec<NodeId>, cost: Weight, geometry: Vec<Coord> }
enum RouteError { Disconnected, NodeOutOfRange, NegativeWeight, NonFifoTtf,
                  BudgetExceeded, ParseError, BuildError, PhaseDeferred(&'static str) }

// ---- heuristic / ALT ----
trait Heuristic { fn h(&self, v: NodeId, t: NodeId) -> Weight; }   // admissible (I4)
struct Haversine<'a> { coords: &'a [Coord], inv_max_speed: u32 }
struct Alt { landmarks: Vec<NodeId>, to_l: Vec<Weight>, from_l: Vec<Weight> }  // k×n each
// ---- contraction hierarchies ----
struct ChGraph { rank: Vec<u32> /*per node*/, up: Csr, down: Csr,
                 shortcut_mid: Vec<NodeId> /*per shortcut edge; NONE for original*/ }
struct Shortcut { from: NodeId, to: NodeId, weight: Weight, middle: NodeId }
fn ch_query(&self, s: NodeId, t: NodeId) -> Result<Weight, RouteError>;    // bidir up-graph
fn unpack(&self, e: EdgeId, out: &mut Vec<EdgeId>);                         // I8, recursive

// ---- customizable route planning ----
struct Partition { levels: u8, cell_of: Vec<Vec<u32>> /*[level][node]*/ }   // I9
struct OverlayTopology { /* boundary vertices + arc shapes per cell/level */ }
struct Metric { base: Vec<Weight> /*per edge*/, overlay: Vec<Weight> /*per overlay arc*/ }
fn customize(&self, topo: &OverlayTopology, base: &[Weight]) -> Metric;     // seconds (I10)
fn crp_query(&self, m: &Metric, s: NodeId, t: NodeId) -> Result<Weight, RouteError>;

// ---- time-dependent ----
struct Ttf { points: Vec<(u32, u32)>, period: u32 }     // PWL, FIFO (I11/I12)
fn eval(&self, tau: u32) -> Weight;                     // interpolate + wrap
fn link(&self, g: &Ttf) -> Ttf;                         // composition f∘g (profile)
fn merge(&self, g: &Ttf) -> Ttf;                        // pointwise min
fn td_dijkstra(&self, s: NodeId, t: NodeId, depart: u32) -> Result<Weight, RouteError>;

// ---- batched ----
fn phast(&self, s: NodeId) -> Vec<Weight>;              // one-to-all (I20)
fn isochrone(&self, s: NodeId, budget: Weight) -> GeoJsonPolygon;
fn matrix(&self, src: &[NodeId], tgt: &[NodeId]) -> Vec<Weight>;  // |S|*|T|, bucket-CH

// ---- routing index (versioned, immutable, mmap) ----
struct IndexHeader { magic: [u8;8], version: u32, extract_sha: [u8;32],
                     profile_id: u32, techniques: u32 /*bitset*/, n: u32, m: u32 }
struct RoutingIndex { header: IndexHeader, graph: Graph, ch: Option<ChGraph>,
                      crp: Option<(Partition, OverlayTopology)>, alt: Option<Alt>,
                      ttf: Option<Vec<Ttf>> }   // I17/I18
