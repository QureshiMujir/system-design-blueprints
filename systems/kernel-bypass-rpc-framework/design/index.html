Core schema & types (§11)

// krpc-types — wire header + ids + flags
pub struct Pkthdr {                 // ≤1 cache line; fixed-offset; little-endian; carried in MsgBuffer headroom
  pub wire_ver: u8,                 // I14: mismatch ⇒ reject
  pub req_type: ReqType,            // dispatch key
  pub flags: PktFlags,              // REQ/RESP/RFR/CR/SM/ECN_ECHO/RETX
  pub msg_size: u32,                // whole-message bytes
  pub pkt_num: u16, pub num_pkts: u16, // fragment index / count (num_pkts=1 ⇒ small)
  pub req_num: u64,                 // monotone per session; matches req↔resp (I1)
  pub dest_session: SessionId, pub src_session: SessionId,
  pub pkt_seq: u64,                 // per-session sequence (I5)
}
pub struct ReqType(pub u8);                 // 0..255 → registered handler
pub struct SessionId(pub u32);              // local id at each endpoint
bitflags! { pub struct PktFlags: u8 { const REQ=0x01; const RESP=0x02; const RFR=0x04;
  const CR=0x08; const SM=0x10; const ECN_ECHO=0x20; const RETX=0x40; } }
pub struct WireVersion { pub major: u8, pub minor: u8 }   // SemVer of wire (ADR-033)

// krpc-buf — zero-copy registered buffers
pub struct MsgBuffer {              // app-owned until RPC completes (I2); transport borrows for TX/RX
  buf: *mut u8, headroom: usize,    // headroom for Pkthdr (no app-data copy)
  max_data: usize, data_size: usize,
  sg: SgList,                       // scatter-gather for multi-buffer messages
  lkey: Lkey,                       // memory-registration key (verbs-style)
}
pub struct SgList { entries: SmallVec<[(*mut u8, usize); 4]> }  // scatter-gather
pub struct Lkey(pub u32);           // local registration key
pub struct Rkey(pub u32);           // remote key (RDMA)
pub struct BufferPool { /* hugepage slab; NUMA-local; per-Rpc slice */ }      // !Sync; owned by one Rpc
                                    // alloc_msg_buffer(size) -> MsgBuffer; free on RPC completion

// krpc-transport — pluggable datapath
pub trait Transport: Send {
  fn tx_burst(&mut self, pkts: &[TxDesc]) -> usize;          // returns # accepted
  fn rx_burst(&mut self, out: &mut [RxDesc], max: usize) -> usize;
  fn resolve_endpoint(&self, uri: &str) -> Result<Endpoint, TransportError>;
  fn reg_mr(&mut self, buf: &mut MsgBuffer) -> Result<Lkey, TransportError>;
  fn mtu(&self) -> usize;
}
pub struct TxDesc { pub buf: *const u8, pub len: usize, pub lkey: Lkey, pub dst: Endpoint }
pub struct RxDesc { pub buf: *mut u8, pub len: usize, pub src: Endpoint, pub ecn: bool }
pub struct Endpoint { /* loopback id | (ip,port) | (lid/gid, qpn) */ }
// krpc-session
pub struct Session {                // owned by exactly one Rpc (I7); !Sync
  local: SessionId, remote: SessionId, peer: Endpoint,
  credits: Credits,                 // flow control (I3, I11)
  slots: [Slot; K_SESSION_REQS],    // fixed concurrent-RPC table (=8)
  next_req_num: u64,                // monotone (I1)
  next_seq: u64,                    // per-session pkt_seq (I5)
  cc: Box<dyn CongestionControl>,   // per-session CC state (I8)
  rto: RtoMap,                      // outstanding pkt → deadline (I9)
  sm_state: SmState,                // handshake (I10)
}
pub struct Credits { available: u32, granted: u32, consumed: u64, returned: u64 } // returned ≤ consumed (I11)
pub enum SlotState { Free, WaitTx, WaitResp, Retx, Done, Error }                   // reuse from Free only (I12)
pub struct Slot {                   // holds buffers (I2), continuation, RFR cursor
  state: SlotState, req_num: u64,
  req_buf: Option<MsgBuffer>, resp_buf: Option<MsgBuffer>,
  cont: Option<ContFunc>, tag: Tag,
  rfr_cursor: u16, retries: u8,     // large-path pull cursor; retransmit budget
}
pub enum SmState { None, ConnectPending, AcceptPending, Connected, Closing, Closed, Error }
pub struct SmPkt { sub: SmSub, session_pair: (SessionId, SessionId), sm_seq: u32 } // idempotent (I10)
pub enum SmSub { Connect, ConnectAck, Disconnect, DisconnectAck }

// krpc-msg
pub struct ReqHandle<'a> {          // passed to server handler; borrows req, lends resp
  pub req_msgbuf: &'a MsgBuffer,    // reassembled request (zero-copy view)
  pre_resp: MsgBuffer,              // pre-allocated small response buffer
  // dyn_resp(size) -> &mut MsgBuffer for large responses
}
pub struct Reassembler { msg_size: u32, num_pkts: u16, got: BitSet, buf: MsgBuffer } // I6
pub enum RxClass { ReqSmall, ReqFragment, RespSmall, RespFragment, Rfr, Cr, Sm, Drop } // §8.7
// krpc-rpc — the per-thread engine
pub struct Rpc {                    // !Sync; one per thread (I7); owns sessions/rings/buffers/timers/cc
  id: RpcId, transport: Box<dyn Transport>, pool: BufferPool,
  sessions: SlabMap<SessionId, Session>, wheel: TimingWheel,
  // handler registry is shared read-only via Nexus
}
impl Rpc {
  pub fn create_session(&mut self, peer: &str, rpc_id: RpcId) -> Result<SessionId, RpcError>;
  pub fn alloc_msg_buffer(&mut self, size: usize) -> MsgBuffer;     // I2: app owns until completion
  pub fn enqueue_request(&mut self, s: SessionId, t: ReqType,
                         req: MsgBuffer, resp: MsgBuffer, cont: ContFunc, tag: Tag) -> Result<(), RpcError>;
  pub fn enqueue_response(&mut self, h: ReqHandle);                 // server side
  pub fn run_event_loop(&mut self, timeout_us: u64);                // poll: rx→process→timers→continuations
}
pub type ContFunc = Box<dyn FnOnce(&mut Rpc, Result<&MsgBuffer, RpcError>, Tag)>; // continuation
pub struct Tag(pub u64);            // opaque app correlation token
pub struct RpcId(pub u16);
// krpc-nexus
pub struct Nexus {                  // one per process; registry read-only after start
  transport_factory: Box<dyn Fn() -> Box<dyn Transport>>,
  registry: HashMap<ReqType, ReqHandler>,     // immutable post-start (concurrency boundary)
  bg_large: BgLargeThreads, bg_sm: BgSmThread, // background threads via channels (ADR-008)
}
impl Nexus {
  pub fn new(uri: &str, transport: TransportKind) -> Self;
  pub fn register_req_func(&mut self, t: ReqType, h: ReqHandler);   // before threads run
  pub fn create_rpc(&self, rpc_id: RpcId) -> Rpc;                   // per-thread handle
}
pub type ReqHandler = Box<dyn Fn(&mut Rpc, ReqHandle) + Send + Sync>;

// krpc-timer
pub struct TimingWheel { /* hierarchical; O(1) insert/expire */ }   // RTO + pacing slots
pub struct Timer { deadline_us: u64, kind: TimerKind }
pub enum TimerKind { Rto { session: SessionId, slot: u8, pkt_seq: u64 }, Pace { session: SessionId } }
pub struct RtoMap { /* pkt_seq → Timer handle; cancel on ack (I9) */ }

// krpc-cc
pub trait CongestionControl: Send {
  fn on_ack(&mut self, bytes: u32, rtt_us: u32);     // → adjust rate/window (I8)
  fn on_loss(&mut self, bytes: u32);                 // multiplicative decrease (clamped)
  fn on_ecn(&mut self, marked: bool);                // DCQCN signal
  fn window_bytes(&self) -> u32;                      // current admission window (≥1 pkt, ≤ cap)
  fn rate_bps(&self) -> u64;                          // current pacing rate (∈[min,line])
}
pub struct Timely { /* RTT-gradient; rate from delay derivative */ }   // eRPC default
pub struct Dcqcn  { /* ECN-mark → rate; RoCE-style */ }
pub struct WindowCc { /* simple AIMD per session */ }

// errors (thiserror; no panic on data path — ADR-030)
pub enum RpcError { Timeout, SessionClosed, NoCredits, Oversize, WireVersion, Transport(TransportError), HandlerPanic }
pub enum TransportError { Unreachable, RegFailed, RingFull, Closed }
