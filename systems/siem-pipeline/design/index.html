Core schema & types (§8)

EventEnvelope {
  tenant_id:          TenantId;
  event_id:           EventId;        // ULID, monotonic within tenant
  ingest_ts:          Timestamp;      // set at ingest-gateway, wall-clock
  event_ts:           Timestamp;      // producer-asserted, may be skewed
  source_type:        SourceType;     // syslog | cloudtrail | windows-event | custom
  schema_version:     SchemaVersion;
  payload:            BlobRef;        // original bytes, forensic fidelity (P2)
  normalized_fields:  Option<NormalizedFields>;
  enrichment_fields:  Option<EnrichmentMap>;
  integrity_hash:     Sha256Digest;  }
RECEIVED -> VALIDATED -> APPENDED -> NORMALIZED -> ENRICHED -> DETECTED -> INDEXED -> ARCHIVED
QUERY {
  tenant:        TenantId;
  range_start:   Timestamp;
  range_end:     Timestamp;
  predicates:    PredicateTree;
  aggregations:  Vec<AggregationOp>;
  pagination:    Cursor;  }

RESULT {
  query_id:        QueryId;
  shard_results:   Vec<ShardResult>;
  next_cursor:     Option<Cursor>;
  execution_stats: ExecutionStats;    // shards touched, bytes scanned, latency  }
