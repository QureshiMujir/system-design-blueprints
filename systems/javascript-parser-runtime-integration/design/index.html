Core schema & types (§11)

pub enum CowStr<'s> { Borrowed(&'s str), Owned(String) }
pub enum Token<'s> {
    Ident(CowStr<'s>), PrivateName(CowStr<'s>),
    Keyword(Keyword), Punctuator(Punct),
    NumericLiteral{ value: f64, raw: &'s str },
    BigIntLiteral(&'s str),
    StringLiteral{ cooked: CowStr<'s>, raw: &'s str },
    TemplatePart{ cooked: Option<CowStr<'s>>, raw: &'s str, head: bool, tail: bool },
    RegExpLiteral{ pattern: &'s str, flags: &'s str },
    LineTerminatorBefore, Comment(&'s str), Eof,
}
pub enum LexGoal { Div, RegExp, TemplateTail }
pub struct Lexer<'s> { /* src,pos,goal,strict,template_brace_stack,had_line_terminator */ }
pub struct Span { pub lo: u32, pub hi: u32 }
pub enum SourceType { Script, Module }
pub struct Program { pub body: Vec<ModuleItem>, pub source_type: SourceType, pub span: Span }
pub enum Expr {
    Ident(Ident), Literal(Lit), Template(TemplateLit), TaggedTemplate(Box<TaggedTemplate>),
    Array(Box<ArrayExpr>), Object(Box<ObjectExpr>), Function(Box<FnExpr>),
    Arrow(Box<ArrowFn>), Class(Box<ClassExpr>),
    Unary(Box<UnaryExpr>), Update(Box<UpdateExpr>), Binary(Box<BinaryExpr>),
    Logical(Box<LogicalExpr>), Assign(Box<AssignExpr>),
    Member(Box<MemberExpr>), OptionalChain(Box<OptionalChainExpr>),
    Call(Box<CallExpr>), New(Box<NewExpr>), Conditional(Box<CondExpr>),
    Sequence(Vec<Expr>), Spread(Box<Expr>), Await(Box<Expr>), Yield(Box<YieldExpr>),
    MetaProperty(MetaProp), Super, This, ParenthesizedHole,
}
pub enum Stmt {
    Block(BlockStmt), Empty, Expr(ExprStmt), If(Box<IfStmt>),
    For(Box<ForStmt>), ForIn(Box<ForInStmt>), ForOf(Box<ForOfStmt>),
    While(Box<WhileStmt>), DoWhile(Box<DoWhileStmt>), Switch(Box<SwitchStmt>),
    Try(Box<TryStmt>), Return(ReturnStmt), Throw(ThrowStmt),
    Break(BreakStmt), Continue(ContinueStmt), Labeled(Box<LabeledStmt>),
    With(Box<WithStmt>), Debugger, Decl(Decl),
}
pub enum Pattern { Ident(Ident), Array(ArrayPat), Object(ObjectPat), Assign(AssignPat), Rest(RestPat) }
pub trait Visit { /* visit_expr, visit_stmt; default-walk recurses children */ }
pub struct Parser<'s> { /* lexer, cur, lookahead, ctx: ParseCtx, errors: Vec<SyntaxError> */ }
pub struct ParseCtx { pub strict: bool, pub in_function: bool, pub in_loop: bool,
    pub allow_await: bool, pub allow_yield: bool }
pub fn parse_script(src: &str) -> Result<Program, ParseError>;
pub fn parse_module(src: &str) -> Result<Program, ParseError>;
pub enum ParseErrorKind {
    UnexpectedToken, UnexpectedEof, InvalidRegExp, InvalidNumber,
    InvalidAssignmentTarget, IllegalReturn, DuplicateParam, InvalidCover,
    EarlyError(EarlyError),
}
pub struct ScopeTree { /* arena: Vec<Scope>, root: ScopeId */ }
pub enum ScopeKind { Global, Module, Function, Block, Catch, With, ClassStaticBlock }
pub struct Scope { pub kind: ScopeKind, pub parent: Option<ScopeId>, pub bindings: Vec<Binding>, pub strict: bool }
pub enum BindingKind { Var, Let, Const, Function, Class, Param, Import, CatchParam }
pub struct Binding { pub name: Atom, pub kind: BindingKind, pub tdz: bool, pub mutable: bool, pub decl_span: Span }
pub fn analyze(program: &Program) -> Result<ScopeTree, Vec<EarlyError>>;
pub struct Realm { /* global_env: EnvId, global_object hook, intrinsics hook */ }
pub enum EnvironmentRecord { Declarative(DeclEnv), Object(ObjEnv), Function(FnEnv), Global(GlobalEnv), Module(ModuleEnv) }
pub struct ScriptRecord { pub realm: RealmId, pub code: Program, pub scopes: ScopeTree }
pub enum ModuleRecord { SourceText(SourceTextModuleRecord) }
pub struct SourceTextModuleRecord {
    pub realm: RealmId, pub code: Program, pub requested_modules: Vec<Atom>,
    pub import_entries: Vec<ImportEntry>, pub local_export_entries: Vec<ExportEntry>,
    pub status: ModuleStatus, pub environment: Option<EnvId>,
}
pub enum ModuleStatus { Unlinked, Linking, Linked, Evaluating, Evaluated }
pub trait ModuleLoader { fn resolve(&self, specifier: &str, referrer: &ModuleRecord) -> Result<ModuleRecord, LoadError>; }
pub enum Completion { Normal(Value), Throw(Value), Return(Value), Break(Option<Atom>), Continue(Option<Atom>) }
pub trait Evaluator { fn evaluate(&mut self, node: &Stmt, ctx: &mut ExecutionContext) -> Completion; }
pub struct ExecutionContext { /* env: EnvId, realm: RealmId, function: Option<FnHandle> */ }
pub fn global_declaration_instantiation(script: &ScriptRecord, env: &mut EnvironmentRecord) -> Completion;
pub fn link(module: &mut ModuleRecord, loader: &dyn ModuleLoader) -> Result<(), LinkError>;
pub fn resolve_export(module: &ModuleRecord, name: &Atom) -> ResolveSet;
