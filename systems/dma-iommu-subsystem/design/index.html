Core schema & types (§11)

struct DomainId(u64)
    Own: allocated once by kernel-iommu-core, globally unique, immutable after creation
    Conc: Copy, no synchronization needed
    Pers: none (RAM-resident, rebuilt on boot from topology snapshot)
    Ser: bincode, fixed-width — stable wire field (P9)
    Compat: additive-only, never reused across a boot epoch (I12)

struct DeviceId(u32)
    Own: assigned at PCIe enumeration (kernel/pci/topology.rs), immutable
    Conc: Copy
    Pers: none
    Ser: bincode, fixed-width
    Compat: stable across topology snapshot versions (I11)

struct IovaRange
    start: u64
    end: u64
    alignment: u64
    Own: kernel/iommu/iova/allocator.rs, single-writer per arena shard
    Conc: immutable once handed out; freed ranges quarantined (I6)
    Pers: none
    Ser: bincode
    Compat: additive-only

struct DmaDomain
    id: DomainId
    page_table_root: PhysAddr
    attached_devices: Vec<DeviceId>
    fault_state: FaultState
    refcount: AtomicU64
    Own: kernel/iommu/core/domain.rs, single authoritative owner (P2)
    Conc: sharded mutex per domain (ADR-08), refcount atomic
    Pers: none, rebuilt from durable teardown markers on crash recovery (I14)
    Ser: not wire-exposed directly; summarized via telemetry
    Compat: internal type, no external compatibility contract
struct MappingDescriptor
    iova: u64
    phys: u64
    len: u64
    perms: Permissions
    Own: created by kernel/dma/api.rs, owned by the returned MappingHandle
    Conc: immutable once activated
    Pers: none
    Ser: bincode, logged to telemetry on every transition (P6)
    Compat: additive-only fields

struct MappingHandle(u64)
    Own: returned to the calling driver by dma_map, single owner until dma_unmap
    Conc: not Send across drivers without explicit transfer (P2)
    Pers: none
    Ser: opaque token, not decoded by consumers
    Compat: stable token shape across minor versions
struct FaultRecord
    seq: u64
    timestamp: u64
    device: DeviceId
    addr: u64
    reason: FaultReason
    Own: kernel/iommu/core/faults.rs, appended to append-only telemetry log (ADR-04)
    Conc: single-writer lock-free MPSC ring (ADR-26)
    Pers: append-only log, durable (I4 replay contract)
    Ser: bincode, versioned schema (P9)
    Compat: additive-only, tagged-field evolution (§25)

enum FaultReason
    TranslationMiss
    PermissionViolation
    PasidInvalid
    IrteInvalid
    Own: kernel/iommu/core/faults.rs
    Conc: Copy
    Pers: embedded in FaultRecord
    Ser: bincode tagged enum
    Compat: additive-only variants (never remove/renumber)

enum FaultState
    Clean
    Faulted
    Quarantined
    Own: kernel/iommu/core/domain.rs
    Conc: guarded by the domain's sharded mutex
    Pers: none
    Ser: internal only
    Compat: internal
struct QueueInvalidationEntry
    descriptor_type: u8
    domain: DomainId
    addr: u64
    Own: kernel/iommu/intel/qi.rs, single-producer per invalidation queue shard
    Conc: lock-free ring (ADR-26 analog for invalidation), NUMA-sharded (P10)
    Pers: none
    Ser: hardware descriptor format (VT-d spec), not bincode
    Compat: pinned to the hardware spec, no evolution

struct CompletionToken(u64)
    Own: returned by flush_iotlb, consumed by wait_completion
    Conc: Copy, polled or waited on
    Pers: none
    Ser: internal only
    Compat: internal

struct PasidHandle
    pasid: u32
    domain: DomainId
    Own: kernel/iommu/intel/pasid.rs, one domain per PASID at a time (I8)
    Conc: guarded by PASID table lock
    Pers: none
    Ser: bincode
    Compat: additive-only, Phase 9 (§21)

struct IrteEntry
    vector: u8
    dest_apic_id: u32
    validated: bool
    Own: kernel/interrupt/remapping.rs
    Conc: guarded by interrupt-remapping table lock
    Pers: none
    Ser: hardware IRTE format
    Compat: pinned to VT-d/PCIe spec
struct TopologySnapshot
    version: u64
    devices: Vec<DeviceId>
    drhd_units: Vec<DrhdUnit>
    Own: kernel/pci/topology.rs, published immutable (I11, P14)
    Conc: readers hold an Arc to a version, no locking needed
    Pers: dumped to docs/runbooks/iommu support bundles on demand
    Ser: bincode, decoded by tools-dmar-dump
    Compat: additive-only per version

struct BounceBuffer
    handle: MappingHandle
    backing_phys: u64
    len: u64
    Own: kernel/dma/bounce.rs, one owning mapping handle at a time (I10)
    Conc: ownership-checked on claim
    Pers: none
    Ser: internal only
    Compat: internal

struct QuarantineRecord
    domain: DomainId
    reason: FaultReason
    entered_at: u64
    grace_period_ends: u64
    Own: kernel/iommu/core/quarantine.rs
    Conc: guarded by domain's sharded mutex
    Pers: durable teardown marker (I14, P4)
    Ser: bincode
    Compat: additive-only
fn dma_map(domain: DomainId, pages: &[PhysAddr], perms: Permissions) -> Result<MappingHandle, DmaError>;
fn dma_unmap(handle: MappingHandle) -> Result<(), DmaError>;
fn attach_device(domain: DomainId, device: DeviceId) -> Result<(), AttachError>;
fn flush_iotlb(domain: DomainId) -> CompletionToken;
fn wait_completion(token: CompletionToken) -> Result<(), InvalidationError>;
fn pin_pages(pages: &[PhysAddr], quota: &PinQuota) -> Result<PinHandle, PinError>;
fn quarantine_domain(domain: DomainId, reason: FaultReason) -> QuarantineRecord;
