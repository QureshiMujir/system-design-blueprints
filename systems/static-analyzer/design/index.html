Core schema & types (§11)

pub struct FileId(pub u32);                      // Copy; issued by SourceMap
pub struct Span { pub file: FileId, pub lo: u32, pub hi: u32 } // half-open, byte offsets
impl Span {
    pub fn join(self, other: Span) -> Span;
    pub fn empty_at(file: FileId, at: u32) -> Span;
    pub fn is_empty(self) -> bool;
}
pub struct SourceMap { files: Vec<SourceFile> }  // owned by AnalysisDb
pub struct SourceFile {
    pub id: FileId, pub path: PathBuf, pub bytes: Arc<str>,
    pub line_starts: Vec<u32>, pub content_hash: u64,
}
pub struct Token { pub kind: TokenKind, pub span: Span,
    pub leading_trivia: TriviaRun, pub trailing_trivia: TriviaRun }
pub enum TokenKind {
    LParen, RParen, LBrace, RBrace, LBracket, RBracket,
    Comma, Semi, Colon, ColonColon, Dot, Arrow, FatArrow,
    Int(IntKind), Float, String, Char, True, False,
    Ident, Kw(Keyword),
    Plus, Minus, Star, Slash, Percent, Eq, EqEq, Ne, Lt, Le, Gt, Ge,
    And, Or, Not, Amp, Pipe, Caret, Tilde, Shl, Shr,
    Assign, PlusAssign, MinusAssign, StarAssign, SlashAssign,
    Eof, Error,                    // Error: recovery token, span = bad bytes
}
pub struct TriviaRun { pub spans: SmallVec<[Span; 2]> }
pub struct GreenNode { pub kind: SyntaxKind, pub width: u32,
    pub children: Vec<GreenElement> }               // Arc-shared, immutable
pub enum GreenElement { Node(Arc<GreenNode>), Token(GreenToken) }
pub struct GreenToken { pub kind: SyntaxKind, pub text: SmolStr }
pub struct RedNode { pub green: Arc<GreenNode>, pub parent: Option<Box<RedNode>>,
    pub index_in_parent: u32, pub offset: u32 }      // lazy, parent-aware view
impl RedNode {
    pub fn span(&self, file: FileId) -> Span;
    pub fn children(&self) -> impl Iterator<Item = RedElement> + '_;
    pub fn first_token(&self) -> Option<RedToken>;
}
pub enum SyntaxKind {                                // file/expr/pattern/type/trivia kinds
    SourceFile, Item, FnDecl, StructDecl, EnumDecl, TraitDecl, ImplBlock,
    UseDecl, ModDecl, LiteralExpr, PathExpr, BinaryExpr, UnaryExpr, CallExpr,
    IndexExpr, FieldExpr, IfExpr, MatchExpr, BlockExpr, LoopExpr, WhileExpr,
    ForExpr, BindPat, LiteralPat, TuplePat, StructPat, EnumPat, WildcardPat,
    PathType, FnType, TupleType, ArrayType, RefType, LetStmt, ExprStmt,
    Whitespace, LineComment, BlockComment, Error, Token,
}
pub trait AstNode: Sized {
    fn can_cast(kind: SyntaxKind) -> bool;
    fn cast(red: RedNode) -> Option<Self>;
    fn syntax(&self) -> &RedNode;
}
pub struct BodyId(pub u32);
pub struct ExprId(pub u32);
pub struct PatId(pub u32);
pub struct LocalId(pub u32);
pub struct Body { pub id: BodyId, pub owner: ItemId,
    pub exprs: Arena<HirExpr>, pub pats: Arena<HirPat>,
    pub locals: Arena<HirLocal>, pub source_map: HirSourceMap }
pub enum HirExpr {
    Literal(Lit), Path(Path),
    Let { pat: PatId, value: ExprId, body: ExprId },
    Call { callee: ExprId, args: Vec<ExprId> },
    Binary { op: BinOp, lhs: ExprId, rhs: ExprId },
    If { cond: ExprId, then: ExprId, else_: Option<ExprId> },
    Match { scrut: ExprId, arms: Vec<HirArm> },
    Block(Vec<HirStmt>, Option<ExprId>), Loop(ExprId),
    Break(Option<ExprId>), Return(Option<ExprId>),
}
pub enum HirStmt { Let { pat: PatId, ty: Option<TyId>, init: Option<ExprId> },
    Expr(ExprId) }
pub struct Ty<'cx>(pub &'cx TyKind<'cx>);            // interned, pointer-eq identity
pub enum TyKind<'cx> {
    Var(TyVar), Bound(BoundVar), Int, Bool, Float, String, Unit,
    Fn(&'cx [Ty<'cx>], Ty<'cx>), Adt(DefId, &'cx [Ty<'cx>]),
    Tuple(&'cx [Ty<'cx>]), Ref(Mutability, Ty<'cx>), Error,
}
pub struct TyVar(pub u32);
pub struct InferCtxt<'cx> { pub arena: &'cx TyArena, pub uf: UnionFind<TyVar, UfNode<'cx>>,
    pub level: u32, pub constraints: Vec<Constraint<'cx>> }
pub enum Constraint<'cx> {
    Eq(Ty<'cx>, Ty<'cx>, Span), Subtype(Ty<'cx>, Ty<'cx>, Span),  // reserved, off in P0
    Field(Ty<'cx>, Name, Ty<'cx>, Span), Call(Ty<'cx>, Vec<Ty<'cx>>, Ty<'cx>, Span),
}
pub enum UnifyError<'cx> { Mismatch(Ty<'cx>, Ty<'cx>), OccursCheck(TyVar, Ty<'cx>),
    ArityMismatch(usize, usize) }
pub struct BlockId(pub u32);
pub struct Cfg { pub body: BodyId, pub blocks: Arena<BasicBlock>,
    pub entry: BlockId, pub exit: BlockId }
pub struct BasicBlock { pub id: BlockId, pub stmts: Vec<CfgStmt>,
    pub term: Terminator, pub preds: SmallVec<[BlockId; 4]> }
pub enum Terminator {
    Fall(BlockId), Branch { cond: ExprId, then: BlockId, else_: BlockId },
    Switch { scrut: ExprId, arms: Vec<(MatchPat, BlockId)>, default: BlockId },
    Return(Option<ExprId>), Throw(ExprId), Unreachable,
}
pub enum CfgStmt { Assign(LocalId, ExprId), StoreField { base: LocalId, field: Name, value: ExprId },
    Call(LocalId, ExprId, Vec<ExprId>), Nop }
pub enum EdgeKind { Fall, BranchTrue, BranchFalse, SwitchArm(MatchPatId),
    SwitchDefault, Exceptional, Return }
pub struct SsaValue(pub u32);
pub struct SsaCfg { pub body: BodyId, pub blocks: Arena<SsaBlock>,
    pub values: Arena<SsaDef>, pub entry: BlockId, pub exit: BlockId }
pub struct SsaBlock { pub id: BlockId, pub phis: Vec<Phi>,
    pub stmts: Vec<SsaStmt>, pub term: SsaTerminator }
pub struct Phi { pub dest: SsaValue, pub var: LocalId,
    pub operands: Vec<(BlockId, SsaValue)> }
pub enum SsaDef {
    Param(LocalId, Ty<'static>), Const(ConstValue, Ty<'static>),
    BinOp(BinOp, SsaValue, SsaValue, Ty<'static>), Call(SsaValue, Vec<SsaValue>, Ty<'static>),
    Load(SsaValue, Field, Ty<'static>), Store(SsaValue, Field, SsaValue), Phi(LocalId),
}
pub trait Lattice: Clone + Eq {
    fn bottom() -> Self; fn top() -> Self;
    fn join(&self, other: &Self) -> Self; fn leq(&self, other: &Self) -> bool;
    fn widen(&self, other: &Self) -> Self { self.join(other) }
    fn narrow(&self, other: &Self) -> Self { self.clone() }
}
pub trait Transfer {
    type L: Lattice;
    fn transfer_stmt(&mut self, stmt: &SsaStmt, st: &Self::L) -> Self::L;
    fn transfer_term(&mut self, term: &SsaTerminator, st: &Self::L) -> Vec<(BlockId, Self::L)>;
    fn direction(&self) -> Direction;
}
pub enum Direction { Forward, Backward }
pub struct DataflowState<L: Lattice> { pub in_: HashMap<BlockId, L>,
    pub out: HashMap<BlockId, L>, pub iterations: u64 }
pub struct FixpointSolver<T: Transfer> { pub transfer: T, pub budget: u64 }  // P14
impl<T: Transfer> FixpointSolver<T> {
    pub fn solve(&mut self, cfg: &SsaCfg) -> Result<DataflowState<T::L>, FixpointError>;
}
pub enum FixpointError { BudgetExhausted(u64), NonMonotoneTransfer { block: BlockId } }
pub trait Rule: Send + Sync {
    fn id(&self) -> RuleId; fn name(&self) -> &'static str;
    fn category(&self) -> Category; fn default_severity(&self) -> Severity;
    fn run(&self, ctx: &mut Ctx<'_>);
}
pub struct RuleId(pub &'static str);
pub enum Category { Correctness, Style, Performance, Security, Complexity, Suspicious, Pedantic }
pub enum Severity { Error, Warning, Info, Hint }
pub struct Diagnostic { pub rule: RuleId, pub severity: Severity, pub message: String,
    pub primary: Label, pub secondary: Vec<Label>, pub fix: Option<Fix>, pub notes: Vec<String> }
pub struct Label { pub span: Span, pub message: Option<String> }
pub struct Fix { pub title: String, pub edits: Vec<Edit>, pub idempotent: bool }  // I10
pub struct Edit { pub span: Span, pub replacement: String }
pub struct Suppression { pub rule: RuleId, pub span: Span, pub reason: Option<String>,
    pub source: SuppressionSource }
pub enum SuppressionSource { AllowAttr, ConfigGlob, InlineComment }  // InlineComment: rejected, I9
pub struct AnalysisDb { pub sources: SourceMap, pub revision: AtomicU64,
    pub storage: QueryStorage }
pub trait Query {
    type Input: Hash + Eq + Clone; type Output: Send + Sync;
    const NAME: &'static str;
    fn execute(db: &AnalysisDb, input: Self::Input) -> Arc<Self::Output>;
}
pub struct QueryStorage { cache: DashMap<QueryKey, QueryEntry> }
pub struct QueryEntry { pub result: Arc<dyn Any + Send + Sync>, pub input_hash: u64,
    pub dep_keys: Vec<QueryKey>, pub verified_at: u64 }
