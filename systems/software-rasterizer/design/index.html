Core schema & types (§11)

// raster-math: fundamental vector/matrix types. Copy, no interior mutability, no persistence.
pub struct Vec2 { pub x: f32, pub y: f32 }                         // Copy; stack-only
pub struct Vec3 { pub x: f32, pub y: f32, pub z: f32 }             // Copy; stack-only
pub struct Vec4 { pub x: f32, pub y: f32, pub z: f32, pub w: f32 } // Copy; stack-only; homogeneous
pub struct Mat4 { pub cols: [Vec4; 4] }                            // Copy; column-major (ADR matches OpenGL)

// Viewport: immutable per-draw state, Copy, no serialization needed at runtime.
pub struct Viewport { pub x: u32, pub y: u32, pub width: u32, pub height: u32 }

// Vertex: input attribute bundle. Copy; owned by VertexBuffer; no concurrency (read-only during a frame).
pub struct Vertex { pub position: Vec3, pub normal: Vec3, pub uv: Vec2, pub color: Vec3 }

// VertexAttributes: post-shader interpolable payload. Copy; fixed-size, no heap.
pub struct VertexAttributes { pub world_position: Vec3, pub normal: Vec3, pub uv: Vec2, pub color: Vec3 }
// VertexBuffer/IndexBuffer: scene-owned, immutable during a frame, Vec-backed heap storage.
pub struct VertexBuffer { pub vertices: Vec<Vertex> }               // append-only build, frozen at draw
pub struct IndexBuffer { pub indices: Vec<u32> }                    // triangle-list, 3 per triangle

// ClipVertex: post-vertex-shader, pre-clip. Copy; w may be non-positive transiently (I2 guards its exit).
pub struct ClipVertex { pub clip_position: Vec4, pub varyings: ShaderVaryings }

// ClipPlane: one of six frustum planes in homogeneous form. Copy; compile-time constant set.
pub struct ClipPlane { pub coeffs: Vec4 }                            // ax+by+cz+dw >= 0 test

// Triangle: index triple into a VertexBuffer/ClipVertex array. Copy; no ownership of vertex data.
pub struct Triangle { pub indices: [u32; 3] }
// TriangleSetup: post-setup, per-triangle precomputed rasterization state. Copy; tile-read-only.
pub struct TriangleSetup {
    pub screen_positions: [Vec2; 3],       // subpixel-quantized (FixedPoint) at rasterization time
    pub edges: [EdgeFunction; 3],
    pub inv_w: [f32; 3],                   // 1/w per vertex, P3
    pub bbox: (Vec2, Vec2),                // screen-space bounding box
    pub two_area: f32,
}

// EdgeFunction: linear coefficients for one triangle edge (section 8 equation). Copy.
pub struct EdgeFunction { pub a: f32, pub b: f32, pub c: f32, pub is_top_left: bool }

// FixedPoint: 24.8 subpixel-quantized screen coordinate (ADR-02/03, I9). Copy; integer-backed.
pub struct FixedPoint(pub i32)                                       // .8 fractional bits
// Barycentric: per-pixel triangle weights. Copy; ephemeral, computed per pixel, never stored.
pub struct Barycentric { pub l0: f32, pub l1: f32, pub l2: f32 }

// Fragment: per-pixel shader input. Copy; ephemeral within the tile rasterize loop.
pub struct Fragment { pub screen_xy: (u32, u32), pub depth: f32, pub varyings: ShaderVaryings, pub derivatives: Derivatives }

// FragmentQuad: 2x2 lane grouping for derivative computation (P6). Copy; ephemeral.
pub struct FragmentQuad { pub lanes: [Option<Fragment>; 4] }         // None = off-triangle lane (P6 fallback)

// Derivatives: screen-space partial derivatives of a varying (section 8 LOD equation). Copy.
pub struct Derivatives { pub d_dx: Vec2, pub d_dy: Vec2 }            // per-attribute, e.g. UV
// VertexShader trait: user-supplied, called per vertex (ADR-11, trait-object dispatch).
pub trait VertexShader {
    fn shade(&self, vertex: &Vertex, uniforms: &UniformBlock) -> ClipVertex;
}

// FragmentShader trait: user-supplied, called per pixel; static contract gates early-Z (I10).
pub trait FragmentShader {
    const DISCARDS: bool;
    const WRITES_DEPTH: bool;
    fn shade(&self, fragment: &Fragment, uniforms: &UniformBlock) -> Option<Vec4>;
}

// ShaderVaryings: fixed-size interpolable payload passed vertex-shader -> fragment-shader.
pub struct ShaderVaryings { pub world_position: Vec3, pub normal: Vec3, pub uv: Vec2, pub color: Vec3 }
// Texture2D: immutable after load; shared read-only across tile threads (no interior mutability).
pub struct Texture2D { pub width: u32, pub height: u32, pub texels: Vec<[u8; 4]>, pub mips: MipChain }

// MipChain: precomputed downsample levels (Williams 1983). Immutable; owned by Texture2D.
pub struct MipChain { pub levels: Vec<Texture2D> }                   // level 0 = base; box-filtered chain

// SamplerState: per-draw sampling configuration. Copy; no mutation after pipeline setup.
pub struct SamplerState { pub filter: FilterMode, pub wrap_u: WrapMode, pub wrap_v: WrapMode }

// FilterMode / WrapMode: fixed enums, no runtime extension (ADR-14).
pub enum FilterMode { Nearest, Bilinear, Trilinear }
pub enum WrapMode { Repeat, Clamp, MirroredRepeat }

// Lod: resolved mip selection for one fragment (section 8 LOD equation). Copy; ephemeral.
pub struct Lod { pub level: u32, pub frac: f32 }
// DepthBuffer / ColorBuffer: tile-owned mutable slices during rasterization (I5); global post-resolve.
pub struct DepthBuffer { pub width: u32, pub height: u32, pub samples: Vec<f32> }
pub struct ColorBuffer { pub width: u32, pub height: u32, pub samples: Vec<[u8; 4]> }

// Framebuffer: resolve target. Mutable only during single-threaded resolve/output stages.
pub struct Framebuffer { pub color: ColorBuffer, pub depth: DepthBuffer, pub msaa_samples: u32 }

// BlendState / DepthState: fixed per-draw pipeline state (P14, ADR-16). Copy; immutable during a draw.
pub struct BlendState { pub enabled: bool, pub src_factor: BlendFactor, pub dst_factor: BlendFactor }
pub struct DepthState { pub compare: DepthCompare, pub write_enabled: bool }
pub enum BlendFactor { Zero, One, SrcAlpha, OneMinusSrcAlpha }
pub enum DepthCompare { Less, LessEqual, Equal, Always }
pub enum CullMode { None, Back, Front }
// Tile / TileBin / BinnedPrimitive: binning-stage output (section 8 tables). No cross-tile aliasing (I5).
pub struct Tile { pub id: u32, pub x: u32, pub y: u32, pub w: u32, pub h: u32 }
pub struct TileBin { pub tile_id: u32, pub primitives: Vec<BinnedPrimitive> }
pub struct BinnedPrimitive { pub triangle_id: u32, pub setup: TriangleSetup }

// RasterConfig: top-level renderer configuration. Immutable for a run (ADR-24 reproducibility).
pub struct RasterConfig { pub tile_size: u32, pub msaa_samples: u32, pub thread_count: Option<u32> }

// FrameStats: per-frame telemetry counters (section 26). Mutable accumulator, single-writer per field.
pub struct FrameStats {
    pub triangles_submitted: u64,
    pub triangles_after_cull: u64,
    pub fragments_shaded: u64,
    pub fragments_early_z_killed: u64,
    pub clip_amplification_max: u32,
    pub dropped_triangles: u64,
}

// Error types: no panics on malformed input (T1/T2 mitigation); all fallible paths return Result.
pub enum RasterError { InvalidFramebufferSize, ClipDegenerate, TextureOutOfRange, GltfParseError(String) }
