Core schema & types (§11)

enum Bits        { INT8, INT4, NF4, FP8_E4M3, FP8_E5M2 }
enum Layout       { PER_TENSOR, PER_CHANNEL, PER_GROUP }
struct QScheme    { bits: Bits, layout: Layout, symmetric: bool,
                    group_size: Option<u32> }        # I1 enforced in constructor
struct QTensor {
    data:        Bytes,               # packed, sub-byte for INT4/NF4
    scales:      Tensor<fp16|fp32>,   # shape depends on layout
    zero_points: Option<Tensor<i32>>, # None iff symmetric
    scheme:      QScheme,
    shape:       (u32, ...),          # dequantized shape
    dtype_out:   { fp16, bf16 },
}
fn QTensor::num_bytes(self) -> usize       # ceil(prod(shape) * scheme.bits / 8)
struct CalibrationStats {
    abs_max:       Tensor<fp32>,          # per-tensor or per-channel
    percentile_99: Tensor<fp32>,
    hist:          Option<Tensor<u64>>,   # per-bin counts (entropy/MSE calibration)
    outlier_mask:  Option<Tensor<bool>>,  # per-channel; LLM.int8() input (§19.1)
    n_samples:     u64,
}
fn CalibrationStats::merge(self, other: CalibrationStats) -> CalibrationStats
const GROUP_SIZE_DEFAULT: u32 = 128       # Q4_K (ADR-03)
const GROUP_SIZE_FINE:    u32 = 64
const GROUP_SIZE_COARSE:  u32 = 256
const INT4_SYMMETRIC_RANGE:  (i8, i8) = (-8, 7)
const INT4_ASYMMETRIC_RANGE: (u8, u8) = (0, 15)
const INT8_SYMMETRIC_RANGE:  (i8, i8) = (-127, 127)
const INT8_ASYMMETRIC_RANGE: (u8, u8) = (0, 255)
const NF4_GRID: [f32; 16]                  # QLoRA quantile grid, exact (I8)
struct QSchema {
    default_scheme: QScheme,
    per_layer:      Map<LayerName, QScheme>,   # ADR-24
    skip_layers:    Set<LayerName>,             # left in fp16
}
fn QSchema::scheme_for(self, name: LayerName) -> Option<QScheme>
struct BlockQ4_0 { d: f16, qs: [u8; 16] }              # 32 weights, 18 B, 4.5 bits/w
struct BlockQ4_K { d: f16, dmin: f16, scales: [u8;12], qs: [u8;128] }  # 256 w, 144 B
struct BlockQ8_0 { d: f16, qs: [i8; 32] }              # 32 weights, 34 B, 8.5 bits/w
const BLOCK_SIZE_Q4_0: u32 = 32
const BLOCK_SIZE_Q4_K: u32 = 256
const BLOCK_SIZE_Q8_0: u32 = 32
struct GgufHeader     { magic: [u8;4], version: u32, n_tensors: u64, n_kv: u64 }
struct TensorDescriptor { name: String, n_dims: u32, dims: Vec<u64>,
                          block_type: GgmlType, offset: u64 }
enum GgmlType { F32, F16, Q4_0, Q4_1, Q5_0, Q5_1, Q8_0, Q8_1,
                Q2_K, Q3_K, Q4_K, Q5_K, Q6_K, Q8_K,
                NF4, FP8_E4M3, FP8_E5M2 }        # extension values, ADR-23
trait DequantFn: Send + Sync {
    fn dequant(&self, packed: &[u8], scales: &[f32],
               zero_points: Option<&[i32]>, out: &mut [f32]);
}
struct DequantInt8Symmetric;
struct DequantInt8Asymmetric;
struct DequantInt4SymmetricPerGroup  { group_size: u32 }
struct DequantInt4AsymmetricPerGroup { group_size: u32 }
struct DequantNf4PerGroup            { group_size: u32 }
struct DequantFp8E4M3;
struct DequantFp8E5M2;
protocol Calibrator { fn collect(module: Module, batches: Iterable<Tensor>) -> Map<str, CalibrationStats> }
protocol Quantizer  { fn quantize(weight: Tensor, stats: Option<CalibrationStats>, scheme: QScheme) -> QTensor }
protocol Serializer { fn write(path: str, tensors: Map<str, QTensor>, metadata: Map<str, Any>) -> None }
protocol Eval       { fn perplexity(model: Module, corpus: str) -> f64 }
struct GptqState        { hessian: Tensor<fp64>, damp_percent: f32, block_size: u32,
                           act_order: bool, permutation: Option<Vec<u32>> }      # §16
struct AwqScaleTable     { per_layer_alpha: Map<LayerName, f32>, grid: [f32; 21] }  # §17
struct SmoothQuantScale  { per_channel_scale: Tensor<fp32>, alpha: f32,
                           folded_into: LayerName }                               # §18
struct LlmInt8Decomposition { outlier_cols: Tensor<i64>, inlier_w: QTensor,
                               outlier_w: Tensor<fp16>, threshold: f32 }          # §19
struct QatObserver       { running_min: f32, running_max: f32, momentum: f32 }    # §20
struct LsqParam          { scale: Tensor<fp32>, grad_scale_factor: f32 }          # §20
struct MixedPrecisionPlan{ schema: QSchema, size_budget_bytes: u64,
                            accuracy_budget: f32 }                                 # §21
