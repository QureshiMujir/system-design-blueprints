Core schema & types (§11)

// --- Handles ---------------------------------------------------------------
pub struct Env { /* mmap base, fd, meta cache, lock handle, geometry, flags */ }
impl Env {
    pub fn open(path: &Path, geom: Geometry, flags: EnvFlags) -> Result<Env>;
    pub fn begin_read(&self) -> Result<ReadTxn<'_>>;
    pub fn begin_write(&self) -> Result<WriteTxn<'_>>;   // blocks until write mutex
    pub fn sync(&self, force: bool) -> Result<()>;       // flush map (NOSYNC catch-up)
    pub fn stat(&self) -> Stat;                          // geometry + reader info
    pub fn copy(&self, dst: &Path, compact: bool) -> Result<()>; // online backup/compact
}
// Ownership: Env owns mmap + fd + lock. Concurrency: Send+Sync; 1 writer, N readers.
// Persistence: opening reads metas; dropping munmaps (no implicit sync unless asked).

pub struct Geometry { pub page_size: u32, pub initial: u64, pub max: u64,
                      pub growth_step: u64, pub shrink_threshold: u64 }
// Persistence: page_size+max are recorded in meta; fixed for file lifetime.

bitflags! { pub struct EnvFlags { NOSYNC; WRITEMAP; MAPASYNC; NORDAHEAD;
                                  NOMETASYNC; RDONLY; NOLOCK /*unsafe*/; } }
// --- Transactions ----------------------------------------------------------
pub struct ReadTxn<'e> { /* txid, root, free_root, reader_slot, &Env */ }
impl<'e> ReadTxn<'e> {
    pub fn get<'t>(&'t self, db: Db, key: &[u8]) -> Result<Option<&'t [u8]>>; // zero-copy
    pub fn cursor(&self, db: Db) -> Result<Cursor<'_>>;
    pub fn open_db(&self, name: Option<&str>) -> Result<Db>;
}   // Drop releases reader slot. Borrow 't ties returned &[u8] to txn lifetime.

pub struct WriteTxn<'e> { /* txid, dirty map, freed PNL, next_pgno, roots, write_guard */ }
impl<'e> WriteTxn<'e> {
    pub fn get<'t>(&'t self, db: Db, key: &[u8]) -> Result<Option<&'t [u8]>>;
    pub fn put(&mut self, db: Db, key: &[u8], val: &[u8], f: PutFlags) -> Result<()>;
    pub fn del(&mut self, db: Db, key: &[u8], val: Option<&[u8]>) -> Result<bool>;
    pub fn open_db(&mut self, name: Option<&str>, f: DbFlags) -> Result<Db>;
    pub fn cursor(&mut self, db: Db) -> Result<CursorMut<'_>>;
    pub fn commit(self) -> Result<()>;   // runs §8.7 state machine
    pub fn abort(self);                   // discard; release write mutex
}   // Mutation: COW only; consumes self on commit/abort. Single-writer enforced by lock.

bitflags! { pub struct PutFlags { NOOVERWRITE; NODUPDATA; APPEND; APPENDDUP; RESERVE; } }
bitflags! { pub struct DbFlags  { CREATE; DUPSORT; DUPFIXED; INTEGERKEY; REVERSEKEY; } }

// --- Cursor ----------------------------------------------------------------
pub struct Cursor<'t> { /* stack of (pgno, slot) from root to leaf, db, &txn */ }
impl<'t> Cursor<'t> {
    pub fn first(&mut self) -> Result<Option<(&[u8], &[u8])>>;
    pub fn last(&mut self)  -> Result<Option<(&[u8], &[u8])>>;
    pub fn seek(&mut self, key: &[u8]) -> Result<Option<(&[u8], &[u8])>>; // >= key
    pub fn next(&mut self)  -> Result<Option<(&[u8], &[u8])>>;
    pub fn prev(&mut self)  -> Result<Option<(&[u8], &[u8])>>;
    pub fn next_dup(&mut self) -> Result<Option<&[u8]>>;     // DUPSORT
    pub fn first_dup(&mut self) -> Result<Option<&[u8]>>;
}   // Concurrency: bound to its txn; invalidated by commit/abort (lifetime-enforced).
// --- On-disk types (codec; repr(C) LE) -------------------------------------
#[repr(C)] pub struct PageHeader { pgno:u64, flags:u16, lower:u16, upper:u16, ovfl:u16 }
#[repr(C)] pub struct MetaPage   { magic:u32, fmt_ver:u32, page_size:u32, flags:u32,
                                   maxpg:u64, main:DbInfo, free:DbInfo,
                                   last_pgno:u64, txid:u64, checksum:u64 }
#[repr(C)] pub struct DbInfo     { root:u64, depth:u16, branch:u64, leaf:u64,
                                   overflow:u64, entries:u64, flags:u32 }
pub struct Val<'t>(&'t [u8]);     // zero-copy slice into map, lifetime = txn
pub struct Pnl(Vec<u64>);         // page-number list, descending; freelist payload

// --- Reader table (in lock file) -------------------------------------------
#[repr(C)] pub struct ReaderSlot { pid:u32, tid:u64, txid:u64 /* 0 = free */ }
pub struct ReaderTable<'l> { slots: &'l [ReaderSlot] } // mmap'd shared region
impl<'l> ReaderTable<'l> { pub fn oldest_txid(&self) -> u64; /* min nonzero */ }

// --- Errors ----------------------------------------------------------------
pub enum Error {
  Io(io::Error), Corrupt(CorruptKind), VersionMismatch{found:u32, want:u32},
  WriteBusy, ReadersFull, MapFull /*addr space*/, KeyExist, NotFound,
  BadValSize, TxnFull, Sigbus, PhaseDeferred(&'static str),
}
pub enum CorruptKind { MetaBothInvalid, PageChecksum{pgno:u64}, FreelistInconsistent,
                       BadMagic, LastPgnoViolation }
pub type Result<T> = std::result::Result<T, Error>;
