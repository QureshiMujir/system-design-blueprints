Core schema & types (§11)

struct Token(NonZeroU64)                    // non-zero: fits kernel epoll_data/udata; 0 reserved
enum Direction { Read, Write }
struct Interest(u8)                          // bits: READABLE=0b01 WRITABLE=0b10 BOTH=0b11
struct Readiness(u8)                         // bits: READABLE WRITABLE READ_CLOSED WRITE_CLOSED
type RawHandle = RawFd (unix) | RawWindowsHandle (windows)
struct OwnedHandle { raw: RawHandle }         // Drop closes the handle
trait Poller: Send + Sync {
    fn add(handle, token: Token, interest: Interest) -> io::Result<()>
    fn modify(handle, token: Token, interest: Interest) -> io::Result<()>
    fn remove(handle) -> io::Result<()>
    fn poll(out: &mut Vec<Event>, timeout: Option<Duration>) -> io::Result<()>
    fn wake() -> io::Result<()>               // signal-safe, idempotent, cross-thread
}
struct Event { token: Token, readiness: Readiness }
struct Source {
    token: Token
    handle: RawHandle
    interest: Mutex<Interest>
    readers: Mutex<Option<Waker>>             // at most one waker per direction (I2)
    writers: Mutex<Option<Waker>>
    last_seen: Mutex<Readiness>
}
struct Reactor {
    poller: Arc<dyn Poller>
    sources: RwLock<HashMap<Token, Arc<Source>>>
    events: Mutex<Vec<Event>>                 // reused scratch space, no per-turn allocation (P10)
    next_token: AtomicU64
}
fn Reactor::new(poller: Arc<dyn Poller>) -> Reactor
fn Reactor::register(handle, interest: Interest) -> io::Result<Arc<Source>>
fn Reactor::deregister(source: &Source) -> io::Result<()>
fn Reactor::turn(timeout: Option<Duration>) -> io::Result<()>
#[repr(C)]
struct Header { state: AtomicU32, refcnt: AtomicU32, vtable: &'static Vtable }
struct Vtable {

    drop_abort: unsafe fn(*const ())
    try_take_output: unsafe fn(*const (), *mut ()) -> bool
}
struct JoinHandle<T> { raw: *const Header, _t: PhantomData<T> }
enum JoinError { Cancelled, Panic(Box<dyn Any + Send>) }
