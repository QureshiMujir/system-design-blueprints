Core schema & types (§11)

pub enum DigestAlgo { Sha256 }
pub struct Digest { pub algo: DigestAlgo, pub hex: String }
// Display: "sha256:<hex>"; hex is lowercase, 64 chars
impl Digest {
    pub fn from_bytes(algo: DigestAlgo, data: &[u8]) -> Self;
    pub fn parse(s: &str) -> Result<Self, ForgeError>;
    pub fn verify_bytes(&self, data: &[u8]) -> Result<(), ForgeError>;
}
pub struct Descriptor {
    pub media_type: MediaType, pub digest: Digest, pub size: u64,
    pub urls: Option<Vec<String>>,
    pub annotations: Option<BTreeMap<String, String>>,
    pub platform: Option<Platform>,
}
pub enum MediaType {
    OciManifestV1, OciIndexV1, OciConfigV1,
    OciLayerTarGz, OciLayerZstd, OciLayerUncompressed,
    DockerManifestV2S2, DockerManifestListV2, DockerLayerGz, DockerConfigV1,
    Other(String),
}
pub struct DigestVerifyingReader<R: Read> { /* accumulates sha2::Sha256; checks at EOF */ }
pub struct DigestVerifyingWriter<W: Write> { /* same for write path */ }
pub fn canonical_json_digest<T: Serialize>(val: &T) -> Result<Digest, ForgeError>;
pub struct ContentStore { root: PathBuf, refcounts: Mutex<HashMap<Digest, u32>> }
impl ContentStore {
    pub fn open(root: &Path) -> Result<Self, ForgeError>;
    pub fn begin_ingest(&self) -> Result<IngestSession, ForgeError>;
    pub fn get(&self, digest: &Digest) -> Result<impl Read, ForgeError>;  // streaming
    pub fn exists(&self, digest: &Digest) -> bool;
    pub fn add_ref(&self, digest: &Digest) -> Result<(), ForgeError>;
    pub fn remove_ref(&self, digest: &Digest) -> Result<(), ForgeError>;
    pub fn gc(&self, roots: &HashSet<Digest>) -> Result<GcStats, ForgeError>;
    pub fn blob_path(&self, digest: &Digest) -> PathBuf;  // blobs/sha256/<hex>
}
pub struct IngestSession { /* tmp_path, DigestVerifyingWriter<File> */ }
impl IngestSession {
    pub fn write_all(&mut self, data: &[u8]) -> Result<(), ForgeError>;
    pub fn commit(self) -> Result<Digest, ForgeError>;  // atomic rename
    pub fn abort(self);
}
pub struct GcStats { pub deleted_count: u64, pub freed_bytes: u64, pub duration_ms: u64 }
pub struct Lease { digests: Vec<Digest> }  // Drop releases refs
pub struct ImageManifest {
    pub schema_version: u32,  pub media_type: MediaType,

    pub subject: Option<Descriptor>,
    pub annotations: Option<BTreeMap<String, String>>,
}
pub struct ImageIndex {
    pub schema_version: u32, pub media_type: MediaType,
    pub manifests: Vec<Descriptor>,
    pub annotations: Option<BTreeMap<String, String>>,

    pub architecture: String, pub os: String,
    pub config: ContainerConfig, pub rootfs: RootFs,
    pub history: Vec<History>, pub created: Option<DateTime<Utc>>,
}
pub struct ContainerConfig {
    pub env: Vec<String>, pub entrypoint: Option<Vec<String>>,
    pub cmd: Option<Vec<String>>, pub working_dir: String, pub user: String,
    pub labels: BTreeMap<String, String>, pub exposed_ports: BTreeMap<String, ()>,
    pub volumes: BTreeMap<String, ()>, pub stop_signal: Option<String>,
}
pub struct RootFs { pub r#type: String, pub diff_ids: Vec<Digest> }
pub struct History {
    pub created: Option<DateTime<Utc>>, pub created_by: String,

}
pub struct Platform { pub os: String, pub architecture: String, pub variant: Option<String> }

pub fn read_oci_layout(dir: &Path) -> Result<(ImageIndex, ContentStore), ForgeError>;

pub enum Compression { None, Gzip(u32), Zstd(i32) }
pub struct LayerBuilder<W: Write> { /* sorts entries; streams to W + sha2 tee */ }

    pub fn add_entry(&mut self, entry: TarEntry) -> Result<(), ForgeError>;
    pub fn finish(self) -> Result<LayerResult, ForgeError>;
}
pub struct LayerResult { pub diff_id: Digest, pub descriptor: Descriptor }

    Rm { path: String, allow_not_found: bool },
}

impl Definition {

}
pub struct DockerfileAst { pub stages: Vec<Stage> }
pub struct Stage { pub from: FromClause, pub name: Option<String>, pub instructions: Vec<Instruction> }
pub enum Instruction {
    Run(Vec<String>), Copy(CopyArgs), Add(AddArgs),
    Env(Vec<(String,String)>), Arg(ArgDef), Workdir(String), User(String),
    Entrypoint(Vec<String>), Cmd(Vec<String>), Label(Vec<(String,String)>),
    Expose(Vec<String>), Volume(Vec<String>), Healthcheck(HealthcheckArgs),
    Onbuild(Box<Instruction>), Stopsignal(String), Shell(Vec<String>),
}
pub fn parse_dockerfile(src: &str, build_args: &HashMap<String, String>) -> Result<DockerfileAst, ForgeError>;

pub fn parse_dockerignore(src: &str) -> Result<DockerignoreFilter, ForgeError>;
pub trait Snapshotter: Send + Sync {
    fn prepare(&self, parent: Option<SnapshotId>) -> Result<WritableMount, ForgeError>;
    fn commit(&self, mount: WritableMount) -> Result<SnapshotId, ForgeError>;
    fn diff(&self, parent: SnapshotId, child: SnapshotId) -> Result<ChangeSet, ForgeError>;
    fn remove(&self, id: SnapshotId) -> Result<(), ForgeError>;
    fn get_mounts(&self, id: SnapshotId) -> Result<Vec<Mount>, ForgeError>;
}
pub struct SnapshotId(pub u64);
pub struct WritableMount { pub lower: Vec<PathBuf>, pub upper: PathBuf, pub work: PathBuf }
pub struct ChangeSet { pub entries: Vec<ChangeEntry> }
pub struct ChangeEntry { pub path: PathBuf, pub kind: ChangeKind, pub stat: Option<FileStat> }
pub enum ChangeKind { Add, Modify, Delete }
pub trait Executor: Send + Sync {
    fn exec(&self, op: &ExecOp, rootfs: &WritableMount) -> Result<ExecResult, ForgeError>;
}
pub struct ExecResult { pub exit_code: i32, pub upper_dir: PathBuf }
pub enum NetworkMode { None, Loopback, Host }
// linux.rs: LinuxExecutor implements Executor via clone(CLONE_NEWUSER|CLONE_NEWNS|CLONE_NEWPID)
// stub.rs: StubExecutor returns ForgeError::Unsupported on all platforms except Linux
pub struct CacheKey(pub [u8; 32]);  // sha256 of op bytes + input keys + platform
impl CacheKey {
    pub fn for_vertex(op: &Op, input_keys: &[CacheKey], platform: &Platform) -> Self;
    pub fn for_copy(op: &FileOp, input_key: &CacheKey, file_digests: &[Digest]) -> Self;
}
pub trait CacheBackend: Send + Sync {
    fn lookup(&self, key: &CacheKey) -> Result<Option<CacheEntry>, ForgeError>;
    fn store(&self, key: &CacheKey, entry: CacheEntry) -> Result<(), ForgeError>;
    fn prune(&self, max_age: Duration, max_bytes: u64) -> Result<PruneStats, ForgeError>;
}
pub struct CacheEntry { pub snapshot_id: SnapshotId, pub descriptor: Descriptor }
pub trait RegistryClient: Send + Sync {
    async fn pull_manifest(&self, r: &Reference) -> Result<(Bytes, MediaType, Digest), ForgeError>;
    async fn pull_blob(&self, r: &Reference, digest: &Digest) -> Result<impl AsyncRead, ForgeError>;
    async fn push_blob(&self, r: &Reference, descriptor: &Descriptor, data: impl AsyncRead) -> Result<(), ForgeError>;
    async fn put_manifest(&self, r: &Reference, manifest: &Bytes, media_type: MediaType) -> Result<Digest, ForgeError>;
    async fn head_blob(&self, r: &Reference, digest: &Digest) -> Result<bool, ForgeError>;
    async fn mount_blob(&self, src: &Reference, dst: &Reference, digest: &Digest) -> Result<bool, ForgeError>;
}
pub struct Reference {
    pub registry: String, pub repository: String,
    pub tag: Option<String>, pub digest: Option<Digest>,
}
pub struct BuildContext {
    pub definition: Definition, pub context_path: PathBuf,
    pub platform: Platform, pub cache_config: CacheConfig,
    pub source_date_epoch: u64, pub build_args: HashMap<String, String>,
}
pub struct BuildResult {
    pub image: Descriptor, pub config: ImageConfig,
    pub layers: Vec<LayerResult>, pub manifest_digest: Digest,
}
pub enum Exporter {
    OciLayout { dir: PathBuf },
    DockerArchive { path: PathBuf },
    RegistryPush { reference: Reference },
}
pub struct Solver {
    pub cas: Arc<ContentStore>, pub snapshotter: Arc<dyn Snapshotter>,
    pub cache: Arc<dyn CacheBackend>, pub executor: Arc<dyn Executor>,
    pub registry: Arc<dyn RegistryClient>,
}
impl Solver { pub async fn solve(&self, ctx: BuildContext) -> Result<BuildResult, ForgeError>; }
pub enum ForgeError {
    Parse { msg: String, span: Option<Span> },
    DigestMismatch { expected: Digest, actual: Digest },
    Unsupported { feature: String, platform: String },
    Registry { status: u16, body: String },
    Sandbox { exit_code: i32, stderr: String },
    Io { source: std::io::Error, path: Option<PathBuf> },
    CachePoison { key: CacheKey, reason: String },
    Validation { field: String, msg: String },
}
