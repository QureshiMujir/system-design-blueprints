Core schema & types (§11)

namespace mini {
enum class Dtype : uint8_t { F32, F16, BF16, Q8_0, Q4_0, Q4_K, Q5_K, Q6_K, Q8_K, I32, I8 };
struct DtypeInfo {
    const char* name;
    size_t bytes_per_element;      // 0 for block dtypes
    size_t block_size_in_elements; // 0 for non-block
    size_t bytes_per_block;        // 0 for non-block
    bool   is_quantized;
};
constexpr DtypeInfo dtype_info(Dtype dt) noexcept;
} // namespace mini
namespace mini {
class Storage {
public:
    Storage() = default;
    Storage(Arena* arena, size_t bytes, size_t alignment);
    void* data() noexcept; const void* data() const noexcept;
    size_t bytes() const noexcept;
private:
    Arena* arena_; void* data_; size_t bytes_;
};

class Tensor {
public:
    Tensor() = default;
    Tensor(Storage storage, Dtype dtype, std::span<const int64_t> shape,
           std::span<const int64_t> strides);
    Dtype dtype() const noexcept; int64_t ndim() const noexcept;
    int64_t numel() const noexcept; int64_t shape(int64_t axis) const noexcept;
    int64_t stride(int64_t axis) const noexcept; bool is_contiguous() const noexcept;
    template <typename T> T* data_as() noexcept;
    Tensor view(std::span<const int64_t> new_shape) const;
    Tensor slice(int64_t axis, int64_t start, int64_t stop) const;
    Tensor permute(std::span<const int64_t> axes) const;
private:
    Storage storage_; Dtype dtype_; std::array<int64_t, MINI_MAX_RANK> shape_;
    std::array<int64_t, MINI_MAX_RANK> strides_; int64_t ndim_;
};
} // namespace mini
namespace mini {
class Arena {
public:
    explicit Arena(size_t initial_capacity_bytes);
    ~Arena();
    void* allocate(size_t bytes, size_t alignment);
    void  reset() noexcept;                    // bulk-free, guarded by I8
    size_t bytes_in_use() const noexcept;
    size_t high_water_mark() const noexcept;
private:
    struct Block { void* base; size_t cap; size_t cursor; };
    std::vector<Block> blocks_; size_t high_water_;
};

class MemoryPool {
public:
    Arena* request_arena();
    void   release(Arena* arena);              // I8-guarded reset + return to pool
    Arena* weights_arena() noexcept;            // global, never reset
private:
    std::mutex mu_; std::vector<std::unique_ptr<Arena>> idle_; Arena weights_;
};
} // namespace mini
namespace mini {
using TokenId = int32_t;
struct Token { TokenId id; std::string_view bytes; };

class Tokenizer {
public:
    enum class Kind { Bpe, SentencePiece, ByteFallback };
    static std::unique_ptr<Tokenizer> from_gguf(const gguf::Reader& reader);
    std::vector<TokenId> encode(std::string_view text, bool add_bos = true,
                                bool add_eos = false) const;
    std::string decode(std::span<const TokenId> ids, bool skip_special = true) const;
    TokenId bos_id() const noexcept; TokenId eos_id() const noexcept;
    TokenId pad_id() const noexcept; TokenId unk_id() const noexcept;
    size_t  vocab_size() const noexcept;
};
} // namespace mini
namespace mini::gguf {
struct Header { uint32_t magic; uint32_t version; uint64_t tensor_count;
                uint64_t metadata_kv_count; };
struct TensorDescriptor { std::string name; Dtype dtype;
                          std::array<int64_t, MINI_MAX_RANK> dims; uint64_t offset; };

class Reader {
public:
    static Reader open_mmap(std::filesystem::path path);   // I9-bounds-checked
    const Header& header() const noexcept;
    const MetadataView& metadata() const noexcept;
    std::span<const TensorDescriptor> tensors() const noexcept;
    Tensor view_tensor(std::string_view name, Arena* arena) const;
};
} // namespace mini::gguf
namespace mini {
struct ModelConfig {
    int64_t vocab_size, hidden_size, intermediate_size, num_hidden_layers;
    int64_t num_attention_heads, num_kv_heads, head_dim, max_position_embeddings;
    float   rms_norm_eps, rope_theta; bool tie_word_embeddings;
};
struct Layer { Tensor attn_q, attn_k, attn_v, attn_output, attn_norm;
              Tensor ffn_gate, ffn_up, ffn_down, ffn_norm; };

class Model {
public:
    static std::unique_ptr<Model> from_gguf(const gguf::Reader&, Backend*, MemoryPool*);
    const ModelConfig& config() const noexcept;
    Tensor embed_tokens() const noexcept;
    std::span<const Layer> layers() const noexcept;
    Tensor lm_head() const noexcept;              // aliases embed_tokens if tied (I16)
};
} // namespace mini
namespace mini {
struct KvBlock { Tensor k; Tensor v; };   // [num_kv_heads, block_len, head_dim]

class KvCache {
public:
    virtual ~KvCache() = default;
    virtual void allocate(int64_t request_id, int64_t prefill_len) = 0;
    virtual void append(int64_t request_id, int64_t layer,
                        const Tensor& k_step, const Tensor& v_step) = 0;
    virtual KvBlock view(int64_t request_id, int64_t layer,
                        int64_t start, int64_t stop) const = 0;
    virtual void release(int64_t request_id) = 0;
    virtual size_t bytes_in_use() const = 0;
    virtual size_t bytes_capacity() const = 0;
    virtual size_t pages_free() const = 0;         // Paged only; Contiguous returns 0
};
class ContiguousKvCache : public KvCache { /* §18.2 */ };
class PagedKvCache      : public KvCache { /* PhaseDeferred("§18") */ };

class PrefixCache {
public:
    struct Hit { std::span<const int64_t> page_ids; int64_t matched_tokens; };
    std::optional<Hit> lookup(std::span<const TokenId> prefix) const;  // I13-verified
    void insert(std::span<const TokenId> prefix, std::span<const int64_t> page_ids);
};
} // namespace mini
namespace mini {
struct SamplingParams {
    float temperature = 1.0f; int32_t top_k = 0; float top_p = 1.0f, min_p = 0.0f;
    float repetition_penalty = 1.0f, frequency_penalty = 0.0f, presence_penalty = 0.0f;
    int32_t n_beam = 1; uint64_t seed = 0;
    std::vector<TokenId> stop_token_ids; std::vector<std::string> stop_strings;
    int32_t max_new_tokens = 256; bool ignore_eos = false;
};
struct SamplingState { uint64_t rng_state; std::vector<int32_t> token_counts; };

class Sampler {
public:
    virtual ~Sampler() = default;
    virtual TokenId sample(std::span<const float> logits, const SamplingState&) = 0;
    virtual void    update_state(SamplingState&, TokenId chosen) = 0;
};
class GreedySampler  : public Sampler { /* §19.1, I15 */ };
class NucleusSampler : public Sampler { /* §19.2 */ };
class BeamSampler     : public Sampler { /* §19.3 */ };
} // namespace mini
namespace mini {
enum class RequestStatus : uint8_t { Queued, Prefill, Decode, Finished, Failed, Cancelled };
struct GenerationStep { TokenId token; float logprob; bool is_eos; bool is_stop; };

class Request {
public:
    int64_t id() const noexcept; RequestStatus status() const noexcept;
    std::span<const TokenId> prompt_ids() const noexcept;
    std::span<const TokenId> output_ids() const noexcept;
};

struct EngineConfig { Backend* backend; int64_t decode_batch_slots = 64;
                      int64_t prefill_token_budget = 4096; };

class Engine {
public:
    static std::unique_ptr<Engine> create(EngineConfig cfg);
    int64_t submit(std::vector<TokenId> prompt, SamplingParams params);  // P12: sync reject
    RequestStatus poll(int64_t request_id) const;
    GenerationStep next_step(int64_t request_id);
    void cancel(int64_t request_id);
    void shutdown();
};

class Scheduler {
public:
    virtual ~Scheduler() = default;
    virtual void step() = 0;                       // one iteration (I14 budget-checked)
};
class ContinuousBatcher : public Scheduler { /* §18.4 */ };
} // namespace mini
namespace mini {
struct MatmulOptions { bool transpose_b = false; };
struct RopeOptions { int64_t pos_offset; float theta_base; float scaling; };
struct AttentionInputs { Tensor q, k, v; bool causal; };
struct AttentionOptions { int64_t head_dim; bool use_flash; };

class Backend {
public:
    virtual ~Backend() = default;
    virtual Tensor allocate(Dtype, std::span<const int64_t> shape, Arena*) = 0;
    virtual void copy_h2d(const void*, Tensor&, size_t) = 0;
    virtual void copy_d2h(const Tensor&, void*, size_t) = 0;
    virtual void synchronize() = 0;
    virtual void matmul(const Tensor&, const Tensor&, Tensor&, const MatmulOptions&) = 0;
    virtual void rmsnorm(const Tensor&, const Tensor&, Tensor&, float eps) = 0;
    virtual void rope(Tensor&, Tensor&, const RopeOptions&) = 0;
    virtual void softmax(const Tensor&, Tensor&, int64_t axis) = 0;
    virtual void swiglu(const Tensor&, const Tensor&, const Tensor&, Tensor&) = 0;
    virtual void attention(const AttentionInputs&, Tensor&, const AttentionOptions&) = 0;
};
class CpuBackend  : public Backend { /* §15 */ };
class CudaBackend : public Backend { /* §16-§17 */ };
} // namespace mini
namespace mini {
enum class ErrorCode { GgufBoundsViolation, OutOfMemory, UnsupportedDtype,
                       IllegalStateTransition, TokenizerTimeout, KernelUnsupported };
class MiniError : public std::runtime_error {
public:
    MiniError(ErrorCode code, std::string message);
    ErrorCode code() const noexcept;
};
} // namespace mini
