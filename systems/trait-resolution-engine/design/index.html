Core schema & types (§11)

struct TypeId(u32);          // owner: crates/intern; persisted: yes; compat: versioned
struct TraitId(u32);         // owner: crates/intern; persisted: yes; compat: versioned
struct GoalId(u64);          // owner: crates/type_ir; persisted: replay-only; compat: versioned
struct UniverseId(u32);      // owner: crates/unify;  persisted: within snapshot; compat: versioned
struct ParamId(u32);
struct InferVarId(u32);      // owner: query-local arena; NOT persisted across queries (I6)
struct PlaceholderId(u32);
struct CandidateId(u32);
struct ProofNodeId(u64);
struct ProofTreeId(u64);
struct ImplId(u32);
enum Type { Scalar(ScalarType), Tuple(Vec<TypeId>), Ref(Mutability, TypeId),
            Projection(ProjectionType), Param(ParamId), Infer(InferVarId),
            Placeholder(PlaceholderId) }
enum Mutability { Shared, Unique }
struct Goal { environment: EnvId, predicate: Predicate, depth: u16 }
struct CanonicalGoal { binders: Vec<CanonicalVar>, goal: Goal }
struct TraitRef { trait_id: TraitId, substitution: Substitution }
enum SolveResult { Proven(ProofTreeId), Ambiguous(AmbiguityReason), NoSolution, Overflow }
struct ProofNode { goal: GoalId, candidate: CandidateId, children: Vec<ProofNodeId>,
                    duration_ns: u64 }
struct DependencyEdge { source: QueryKey, target: QueryKey, edge_kind: EdgeKind }
enum EdgeKind { Reads, Invalidates, Ambiguates }
struct QueryKey(u64);   // stable hash of a CanonicalGoal, per I1
trait SolverDatabase;
fn lookup_impls(&self, trait_id: TraitId) -> &[ImplId];
fn normalize_projection(&self, projection: ProjectionType) -> TypeId;
fn intern_type(&self, ty: Type) -> TypeId;
