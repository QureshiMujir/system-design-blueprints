Core schema & types (§11)

struct ExperimentId
    uuid: [u8; 16]
struct ProbeSample
    timestamp_start: u64
    timestamp_end:   u64
    latency_cycles:  u32
    cpu_id:          u16
    cache_hit:       bool
    sequence:        u64
struct ReplayFrame
    frame_id:      u64
    experiment_id: ExperimentId
    samples:       Vec<ProbeSample>
    pmu_samples:   Vec<PmuSample>
    checksum:      [u8; 32]
enum TelemetryEvent
    Probe
    CacheMiss
    BranchMiss
    SynchronizationWait
    ReplayCheckpoint
    RuntimeFault
    ExperimentAbort
trait RuntimeBarrier
    fn wait(&self) -> BarrierResult
    fn reset(&self)
trait TimerProvider
    fn timestamp(&self) -> u64
    fn calibrate(&mut self) -> CalibrationReport
struct VictimProfile
    mode:          VictimMode      # Vulnerable | Mitigated
    workload:      WorkloadKind    # LookupTableCrypto | BranchDependent | ConstantTime
    memory_layout: LayoutPolicy
    secret_seed:   [u8; 32]
struct PmuSample
    counter_id:  u16
    value:       u64
    interval_ns: u32
    overflow:    bool
