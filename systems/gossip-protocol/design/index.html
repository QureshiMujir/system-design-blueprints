Core schema & types (§11)

// crates/gossip/src/types.rs
use std::net::SocketAddr;
use uuid::Uuid;

/// Cluster-unique node identifier (UUIDv4; avoids rejoin collisions).
#[derive(Copy, Clone, Eq, PartialEq, Hash, Debug)]
pub struct NodeId(pub Uuid);

/// Logical incarnation; bumped to overrule stale suspicion about self.
#[derive(Copy, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Debug, Default)]
pub struct Incarnation(pub u64);

/// Routable endpoint; v4 and v6 are first-class.
#[derive(Clone, Eq, PartialEq, Hash, Debug)]
pub struct Endpoint { pub addr: SocketAddr, pub bind_port: u16 }

/// Opaque application metadata, capped at `meta_max_bytes` (default 512).
#[derive(Clone, Eq, PartialEq, Debug, Default)]
pub struct NodeMeta(pub Vec<u8>);
// crates/gossip/src/view.rs
#[derive(Copy, Clone, Eq, PartialEq, Debug)]
pub enum Status { Alive, Suspect, Dead, Leaving }

#[derive(Clone, Debug)]
pub struct NodeState {
    pub id: NodeId,
    pub endpoint: Endpoint,
    pub incarnation: Incarnation,
    pub status: Status,
    pub status_since: Instant,      // local clock only
    pub meta: NodeMeta,
}

pub struct View {
    members: BTreeMap<NodeId, NodeState>,   // ordered (P11); includes Dead in reclaim window
    local_health: u8,                        // 0..max_local_health (Lifeguard)
    local: NodeState,                        // always Alive locally
}
// crates/gossip/src/types.rs
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Message {
    Probe(ProbeReq), ProbeAck(ProbeAck),
    IndirectProbe(IndirectProbeReq), IndirectProbeAck(IndirectProbeAck),
    Push(PushReq), Pull(PullReq), PushPull(PushPullReq), StateSync(StateSyncRsp),
    Leave(LeaveReq), UserEvent(UserEventReq),
    Ihave(IhaveReq), Graft(GraftReq), Prune(PruneReq),
}

// crates/gossip/src/outbox.rs
#[derive(Default)]
pub struct Outbox {
    pub udp_messages: Vec<(Endpoint, Message)>,   // best-effort UDP
    pub tcp_messages: Vec<(Endpoint, Message)>,   // full-state sync
    pub events: Vec<Event>,                        // application callbacks to dispatch
    pub persist: Vec<PersistOp>,                   // durable incarnation requests
    pub observ: Vec<ObservEvent>,                  // observability events
}
// crates/gossip/src/node.rs
pub struct Node { /* id, endpoint, view, swim, lifeguard, accrual,
                     plumtree, hyparview: Option<_>, settings, outbox */ }

impl Node {
    pub fn new(id: NodeId, settings: Settings, restored: RestoredState) -> Self;
    /// Drive with an inbound packet (pure w.r.t. time).
    pub fn step(&mut self, from: Endpoint, msg: Message) -> Result<()>;
    /// Drive with elapsed logical time.
    pub fn tick(&mut self, elapsed: Duration);
    pub fn join(&mut self, seeds: &[Endpoint]) -> Result<()>;
    pub fn leave(&mut self) -> Result<()>;
    pub fn broadcast(&mut self, event: UserEvent) -> Result<()>;
    pub fn drain_outbox(&mut self) -> Outbox;
    pub fn on_persisted(&mut self, id: PersistId);
    pub fn members(&self) -> impl Iterator<Item = &NodeState>;
    pub fn local_health(&self) -> u8;
}
// crates/gossip-membership/src/delegate.rs
pub trait Delegate: Send + 'static {
    fn on_join(&mut self, node: &NodeState);
    fn on_leave(&mut self, node: &NodeState);
    fn on_failed(&mut self, node: &NodeState);
    fn on_update(&mut self, node: &NodeState);
    fn on_user_event(&mut self, event: &UserEvent);
    fn local_state(&self, join: bool) -> Vec<u8>;
    fn merge_remote_state(&mut self, state: &[u8], join: bool);
}

// crates/gossip-transport/src/datagram.rs
pub trait Datagram: Send + Sync + 'static {
    fn send_udp(&self, to: Endpoint, payload: &[u8]) -> Result<()>;
    fn recv_udp(&self) -> Result<(Endpoint, Vec<u8>)>;
    fn open_tcp(&self, to: Endpoint) -> Result<TcpStream>;
    fn accept_tcp(&self) -> Result<(Endpoint, TcpStream)>;
}

// crates/gossip-anti-entropy/src/sync.rs
pub trait AntiEntropy: Send + 'static {
    fn root_hash(&self) -> [u8; 32];
    fn diff_level(&self, level: u8, peer_hashes: &[[u8; 32]]) -> Vec<u32>;
    fn leaves(&self, indices: &[u32]) -> Vec<Leaf>;
    fn merge_leaves(&mut self, leaves: &[Leaf]);
}
// crates/gossip/src/error.rs
use thiserror::Error;
#[derive(Debug, Error)]
pub enum GossipError {
    #[error("phase deferred: {0}")] PhaseDeferred(&'static str),
    #[error("not yet joined: call Node::join() first")] NotJoined,
    #[error("unknown peer {id:?}")] UnknownPeer { id: NodeId },
    #[error("incarnation rollback: observed {observed:?}, local {local:?}")]
    IncarnationRollback { observed: Incarnation, local: Incarnation },
    #[error("transport error: {0}")] Transport(#[from] crate::TransportError),
}
