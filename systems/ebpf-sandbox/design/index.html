Core schema & types (§11)

pub struct Instruction { opcode: u8, dst_reg: RegId, src_reg: RegId, offset: i16, imm: i32 }
// Immutable once decoded; Copy; no interior mutability; serialization = raw 8-byte LE.

pub struct RegId(u8); // 0..=10; Copy; validated range at decode time.

pub enum AbstractValue {
    Unknown,
    Scalar { smin: i64, smax: i64, umin: u64, umax: u64 },
    Pointer { base: PointerBase, off_min: i64, off_max: i64 },
    Constant(i64),
}
// Owned per VerifierState; Clone (cheap, no heap); join() total, defined for every pair.

pub enum PointerBase { Ctx, MapValue(MapId), Stack, Packet, PacketEnd, FuncArg(u8) }
// Copy; identity is structural (two Pointers to the same MapId are provenance-equal).

pub struct VerifierState { regs: [AbstractValue; 11], stack: BTreeMap<i64, AbstractValue>, pc: u32 }
// Owned per CFG node during analysis; Clone at branch/join; not persisted post-verify.

pub trait Map: Send + Sync {
    fn kind(&self) -> MapKind;
    fn key_size(&self) -> u32;
    fn value_size(&self) -> u32;
    fn lookup(&self, key: &[u8]) -> Option<&[u8]>;
    fn update(&self, key: &[u8], value: &[u8]) -> Result<(), MapError>;
    fn delete(&self, key: &[u8]) -> Result<(), MapError>;
}
// Concurrency: Send+Sync required — maps are shared across attach points and CPUs;
// mutation via interior atomics/RCU per map kind (§17.3); no persistence across restart.

pub enum MapKind { Hash, Array, RingBuf }
// Fixed at construction; drives which Map impl backs the trait object.
pub struct HelperSignature { id: HelperId, args: [ArgKind; 5], ret: RetKind }
// Immutable, 'static — one per helper, checked at verify time only (I5), never at call time.

pub enum ArgKind { Void, ScalarAny, PointerToMapKey, PointerToMapValue, PointerToStack, ConstSize(u32) }

pub enum HelperId {
    MapLookupElem, MapUpdateElem, MapDeleteElem,
    RingbufOutput, KtimeGetNs, GetSmpProcessorId,
    GetCurrentPidTgid, ProbeReadKernel, TracePrintk,
}
// 9 variants — matches P3/ADR-10's fixed helper count exactly.

pub struct LoadedProgram { insns: Vec<Instruction>, maps: Vec<MapId>, prog_type: ProgType }
// Produced by ebs-loader (§18); not yet verified; consumed only by ebs-verify.

pub struct VerifiedProgram { insns: Vec<Instruction>, maps: Vec<MapId>, prog_type: ProgType, loop_bounds: Vec<LoopBound> }
// Capability type — the ONLY input type ebs-interp / ebs-jit-x86_64 accept (I1);
// constructed exclusively by ebs-verify::verify(); immutable; Clone is deep but rare.

pub struct LoopBound { back_edge_pc: u32, counter_reg: RegId, max_iters: u32 }

pub struct JitProgram { code: Vec<u8>, entry_offset: usize }
// Executable mmap'd region (W^X enforced, §19); Send but not Sync (one JIT'd instance
// per attach point); no serialization — regenerated from VerifiedProgram on each load.

pub enum AttachPoint { FunctionProbe { symbol: String }, Tracepoint { name: String }, Seccomp { syscall_nr: i64 } }

pub enum VerifyError {
    UnprovenPointer { pc: u32, reg: u8 },
    OutOfBounds { pc: u32, reg: u8 },
    UnboundedLoop { pc: u32 },
    HelperArgMismatch { pc: u32, arg: u8, expected: ArgKind, got: &'static str },
    StackOverflow { pc: u32, depth: u32 },
    MaybeNullDeref { pc: u32, reg: u8 },
    MapSizeMismatch { pc: u32, map: MapId },
}
// Exhaustive, non_exhaustive=false — every reject path names one variant (P14).

pub enum MapError { KeyNotFound, ValueSizeMismatch, Full }
pub enum LoaderError { TooManyInstructions, MalformedElf, UnresolvedRelocation { symbol: String } }
pub enum AttachError { TypeMismatch { declared: ProgType, surface: &'static str } }

pub struct ProgramLimits { max_insns: u32, max_stack_bytes: u32, max_tailcall_depth: u8 }
// Compile-time constants surfaced as a struct for testability; values = 4096/512/32 (§4).
