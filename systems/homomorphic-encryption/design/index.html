Core schema & types (§11)

pub fn modexp(base: &BigUint, exp: &BigUint, modulus: &BigUint) -> BigUint;   // sliding-window
pub fn modinv(a: &BigUint, modulus: &BigUint) -> Option<BigUint>;             // ext-Euclid
pub fn gcd(a: &BigUint, b: &BigUint) -> BigUint;
pub fn crt(residues: &[(BigUint, BigUint)]) -> BigUint;                        // (rem, mod) pairs
pub struct MontgomeryCtx { /* modulus, r, r2, n' */ }
impl MontgomeryCtx {
    pub fn new(modulus: &BigUint) -> Self;                                     // odd modulus
    pub fn mul(&self, a: &MontForm, b: &MontForm) -> MontForm;                 // division-free
    pub fn pow(&self, a: &MontForm, e: &BigUint) -> MontForm;
}
pub struct BarrettCtx { /* modulus, mu */ }                                    // general reduction
// constant-time variants for secret-dependent paths (subtle) (I9, P7)
pub fn ct_modexp(base: &CtUint, exp: &CtUint, modulus: &CtUint) -> CtUint;
pub fn is_probable_prime(n: &BigUint, rounds: usize) -> bool;                  // Miller–Rabin (I6)
pub fn gen_safe_prime(bits: usize, rng: &mut dyn CryptoRng) -> BigUint;        // p,(p-1)/2 prime
pub fn random_coprime(n: &BigUint, rng: &mut dyn CryptoRng) -> BigUint;        // Z_n* element
pub fn jacobi(a: &BigInt, n: &BigUint) -> i8;
pub fn l_function(u: &BigUint, n: &BigUint) -> BigUint;                        // (u-1)/n (Paillier)
pub trait CryptoRng { fn fill_bytes(&mut self, out: &mut [u8]); }
pub struct Csprng;          // ChaCha20 from getrandom
pub struct DetRng;          // --insecure-deterministic (tests only, I11, P11)
pub fn uniform_below(bound: &BigUint, rng: &mut dyn CryptoRng) -> BigUint;     // rejection (no bias)
pub fn discrete_gaussian(sigma: f64, n: usize, rng: &mut dyn CryptoRng) -> Vec<i64>; // RLWE error
pub fn ternary(n: usize, rng: &mut dyn CryptoRng) -> Vec<i8>;                  // {-1,0,1}
pub struct PaillierPublicKey { pub n: BigUint, pub g: BigUint, pub n2: BigUint }
pub struct PaillierSecretKey { pub lambda: BigUint, pub mu: BigUint }          // secret (CT paths)
pub struct PaillierCiphertext(pub BigUint);                                    // in Z_{n²}
pub fn keygen(bits: usize, rng: &mut dyn CryptoRng)
    -> (PaillierPublicKey, PaillierSecretKey);                                 // (I6)
pub fn encrypt(pk: &PaillierPublicKey, m: &BigUint, rng: &mut dyn CryptoRng)
    -> PaillierCiphertext;                                                     // fresh r (I10,P4)
pub fn decrypt(sk: &PaillierSecretKey, pk: &PaillierPublicKey, c: &PaillierCiphertext)
    -> BigUint;                                                                // CT (I9)
pub fn add(pk: &PaillierPublicKey, a: &PaillierCiphertext, b: &PaillierCiphertext)
    -> PaillierCiphertext;                                                     // c1*c2 (I4)
pub fn mul_plain(pk: &PaillierPublicKey, c: &PaillierCiphertext, k: &BigUint)
    -> PaillierCiphertext;                                                     // c^k
pub fn rerandomize(pk: &PaillierPublicKey, c: &PaillierCiphertext, rng: &mut dyn CryptoRng)
    -> PaillierCiphertext;
// he-ring
pub struct Poly { pub coeffs: Vec<u64>, pub modulus: u64, pub n: usize }       // R_q residue
pub fn poly_add(a: &Poly, b: &Poly) -> Poly;
pub fn poly_mul_schoolbook(a: &Poly, b: &Poly) -> Poly;                        // negacyclic reference

// he-ntt
pub struct NttCtx { pub q: u64, pub n: usize, /* psi, psi_inv, n_inv tables */ }
impl NttCtx {
    pub fn new(q: u64, n: usize) -> Result<Self, HeError>;                     // q ≡ 1 mod 2n (I14)
    pub fn forward(&self, p: &mut [u64]);                                      // in place
    pub fn inverse(&self, p: &mut [u64]);                                      // INTT (I7)
    pub fn mul_pointwise(&self, a: &mut [u64], b: &[u64]);
}

// he-rns
pub struct RnsContext { pub moduli: Vec<u64>, /* NttCtx per prime, CRT consts */ }
pub struct RnsPoly { pub residues: Vec<Vec<u64>>, pub n: usize }               // k residue polys
impl RnsContext {
    pub fn to_rns(&self, x: &BigUint) -> Vec<u64>;
    pub fn from_rns(&self, residues: &[u64]) -> BigUint;                       // exact CRT (I8)
    pub fn base_convert(&self, p: &RnsPoly, to: &RnsContext) -> RnsPoly;       // (I8)
}
pub struct BfvParams { pub n: usize, pub q: Vec<u64>, pub t: u64,
                       pub sigma: f64, pub depth: usize, pub sec_level: u32 }  // (I12,I13)
pub struct SecretKey(pub RnsPoly);                                            // ternary s
pub struct PublicKey { pub b: RnsPoly, pub a: RnsPoly }                        // (-(as+e), a)
pub struct RelinKey { /* enc of s² */ }
pub struct BfvCiphertext { pub comps: Vec<RnsPoly>, pub level: usize,
                           pub noise_est: NoiseBudget }                        // 2 or 3 comps
pub fn keygen(p: &BfvParams, rng: &mut dyn CryptoRng)
    -> (SecretKey, PublicKey, RelinKey);
pub fn encrypt(pk: &PublicKey, pt: &Plaintext, rng: &mut dyn CryptoRng) -> BfvCiphertext; // (I10)
pub fn decrypt(sk: &SecretKey, ct: &BfvCiphertext) -> Plaintext;              // round(t/q·...) (I1)
pub fn add(a: &BfvCiphertext, b: &BfvCiphertext) -> Result<BfvCiphertext, HeError>;        // (I5)
pub fn mul(a: &BfvCiphertext, b: &BfvCiphertext, rk: &RelinKey)
    -> Result<BfvCiphertext, HeError>;                                        // tensor→relin (I3,I5)
pub fn mod_switch(ct: &BfvCiphertext) -> Result<BfvCiphertext, HeError>;      // drop a prime
// he-noise
pub struct NoiseBudget { pub bits: f64 }                                      // headroom (I2)
pub fn fresh_budget(p: &BfvParams) -> NoiseBudget;
pub fn after_add(a: &NoiseBudget, b: &NoiseBudget) -> NoiseBudget;
pub fn after_mul(a: &NoiseBudget, b: &NoiseBudget, p: &BfvParams) -> NoiseBudget;
pub fn guard(planned: &NoiseBudget) -> Result<(), HeError>;                   // abort if ≤0 (I3)

// he-params
pub fn select(depth: usize, sec_level: u32) -> Result<BfvParams, HeError>;    // (I12,I14)
pub fn check_security(p: &BfvParams) -> u32;                                   // estimated bits
pub fn validate(p: &BfvParams) -> Result<(), HeError>;                        // I13/I14 asserts

// he-encode
pub struct Plaintext { pub poly: Poly }                                       // in R_t
pub fn encode_int(m: i64, p: &BfvParams) -> Plaintext;
pub fn decode_int(pt: &Plaintext, p: &BfvParams) -> i64;
pub fn encode_batch(slots: &[i64], p: &BfvParams) -> Result<Plaintext, HeError>; // t≡1 mod 2n
pub fn decode_batch(pt: &Plaintext, p: &BfvParams) -> Vec<i64>;
pub enum HeError {
    CompositeFactor,                              // I6 (P9)
    PlaintextOutOfRange { value: i128, modulus: u64 },
    NoiseExhausted { remaining: f64 },            // I3 (P3)
    ParamMismatch,                                // I13
    InsecureParameters { got: u32, want: u32 },   // I12 (P9)
    BadNttPrime { q: u64, n: usize },             // I14
    ModSwitchAtBottom,                            // §8.12
    SeededKeygenRefused,                          // I11 (P11)
    Deserialization { msg: String },
}
