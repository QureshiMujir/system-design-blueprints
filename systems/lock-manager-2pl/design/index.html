Core schema & types (§11)

pub struct TxnId(core::num::NonZeroU64);
impl TxnId {
    pub const fn new(id: u64) -> Self;   // panics on 0
    pub const fn get(self) -> u64;
}
#[repr(u8)]
pub enum LockMode { IS = 0, IX = 1, S = 2, SIX = 3, U = 4, X = 5 }
impl LockMode { pub const fn index(self) -> usize; pub const COUNT: usize; }
pub const COMPAT: [[bool; 6]; 6];                 // see §8.1 for values
pub const fn is_compatible(held: LockMode, requested: LockMode) -> bool;
pub const UPGRADE: [[Option<LockMode>; 6]; 6];    // see §8.2 for legal upgrades
pub fn upgrade_target(from: LockMode, to: LockMode) -> Option<LockMode>;
pub enum Grant { Grant, Wait, WaitAsUpgrader }
pub struct LockResource { bytes: smallvec::SmallVec<[u8; 24]> }
impl LockResource {
    pub fn new(bytes: &[u8]) -> Result<Self, Error>;   // Error::ResourceTooLarge on overflow
    pub fn as_bytes(&self) -> &[u8];
    pub fn row(table_id: u64, row_id: &[u8]) -> Self;
    pub fn table(table_id: u64) -> Self;
}
pub struct Holder { pub txn: TxnId, pub mode: LockMode, pub acquired_at: std::time::Instant }
pub struct WaitEntry {
    pub txn: TxnId, pub mode: LockMode,
    pub enqueued_at: std::time::Instant, pub deadline: Option<std::time::Instant>,
    signal: parking_lot::Mutex<WaitResult>, condvar: parking_lot::Condvar,
}
pub enum WaitResult { Pending, Granted, Aborted { reason: AbortReason }, TimedOut }
pub enum AbortReason { DeadlockVictim, Wounded, LockTimeout, UserAbort }
pub struct LockEntry {
    pub resource: LockResource,
    pub holders: smallvec::SmallVec<[Holder; 4]>,
    pub wait_queue: std::collections::VecDeque<WaitEntry>,
    pub current_mode: Option<LockMode>,           // I5
}
pub struct LockTable {
    buckets: Vec<parking_lot::RwLock<std::collections::HashMap<LockResource, std::sync::Arc<parking_lot::Mutex<LockEntry>>>>>,
    bucket_mask: usize,
}
impl LockTable {
    pub fn new(shard_count_pow2: usize) -> Self;      // ADR-03
    fn bucket_index(&self, resource: &LockResource) -> usize;
}
pub struct WaitForGraph { pub edges: std::collections::HashMap<TxnId, smallvec::SmallVec<[TxnId; 4]>> }
impl WaitForGraph {
    pub fn add_edge(&mut self, waiter: TxnId, holder: TxnId);
    pub fn nodes(&self) -> impl Iterator<Item = &TxnId>;
}
pub trait DeadlockDetector: Send + Sync {
    fn detect(&self, graph: &WaitForGraph, cost: &dyn VictimCost) -> Vec<TxnId>;
}
pub trait VictimCost: Send + Sync { fn cost(&self, txn: TxnId) -> u64; }
pub struct PeriodicCycleDetect { period: std::time::Duration }
pub struct WoundWait;
pub struct WaitDie;
pub struct EscalationPolicy { pub row_threshold: u64, pub check_every_nth_acquire: u32 }
pub enum EscalationError { ConflictingHolders, RefusedDuringCommit }
pub enum KeyRange {
    NextKey { this_key: Vec<u8> },
    Gap { upper_bound_exclusive: Vec<u8> },
    SupremumNextKey,
}
impl KeyRange { pub fn to_resource(&self, index_id: u64) -> LockResource; }
pub struct ShardId(u64);
pub struct Probe { pub initiator: TxnId, pub current: TxnId, pub target: TxnId, pub origin_shard: ShardId }
pub struct ChandyMisraHaas { shard_id: ShardId, transport: std::sync::Arc<dyn Transport>, local_table: std::sync::Arc<LockTable> }
pub struct GraphSummary { /* per-shard wait-for edges, remote-holder-annotated */ }
pub struct Obermarck { shard_id: ShardId, transport: std::sync::Arc<dyn Transport> }
pub trait Transport: Send + Sync {
    fn send_probe(&self, to: ShardId, probe: Probe);
    fn send_summary(&self, to: ShardId, summary: GraphSummary);
    fn send_abort(&self, to: ShardId, victim: TxnId);
}
pub struct LockManager {
    table: LockTable, detector: std::sync::Arc<dyn DeadlockDetector>,
    cost: std::sync::Arc<dyn VictimCost>, detection_period: std::time::Duration,
    default_timeout: std::time::Duration, escalation: EscalationPolicy,
}
pub enum AcquireOutcome { Granted, Aborted(AbortReason) }
pub trait AbortSink: Send + Sync { fn abort(&self, txn: TxnId, reason: AbortReason); }
impl LockManager {
    pub fn acquire(&self, txn: TxnId, resource: LockResource, mode: LockMode, timeout: Option<std::time::Duration>) -> AcquireOutcome;
    pub fn release_all(&self, txn: TxnId);
    pub fn try_escalate(&self, txn: TxnId, table_id: u64) -> Result<bool, EscalationError>;
}
