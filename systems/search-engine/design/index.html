Core schema & types (§11)

pub type DocId = u32;          // 4B-document ceiling per shard
pub type TermId = u32;
pub type FieldId = u16;
pub type SegmentId = u32;
pub const NULL_DOC: DocId = u32::MAX;
pub const NULL_TERM: TermId = u32::MAX;
pub enum FieldType {
    Text { analyser: String, index_options: IndexOptions, store: bool, positions: bool, offsets: bool },
    Keyword { store: bool, doc_values: bool },
    Integer { doc_values: bool, indexed: bool },
    Float   { doc_values: bool, indexed: bool },
    Date    { doc_values: bool, indexed: bool, format: DateFormat },
    Boolean { doc_values: bool, indexed: bool },
    DenseVector { dim: u32, distance: VectorDistance },
    Geo     { doc_values: bool, indexed: bool },
}
pub enum IndexOptions { Docs, DocsFreqs, DocsFreqsPositions, DocsFreqsPositionsOffsets }
pub struct Field { pub id: FieldId, pub name: String, pub typ: FieldType }
pub struct Schema { fields: Vec<Field>, by_name: HashMap<String, FieldId> }
pub struct Document { pub external_id: String, pub fields: Vec<(FieldId, FieldValue)> }
pub enum FieldValue {
    Text(String), Keyword(String), Integer(i64), Float(f64), Date(i64),
    Boolean(bool), DenseVector(Vec<f32>), Geo(f64, f64),
}
pub const BLOCK_SIZE: usize = 128;
pub struct Posting { pub doc_id: DocId, pub term_freq: u32, pub positions: Option<Vec<u32>>,
                      pub offsets: Option<Vec<(u32, u32)>> }
pub struct PostingBlock { pub doc_count: u32, pub doc_id_codec: BlockCodec, pub tf_codec: BlockCodec,
                          pub doc_id_bytes: Vec<u8>, pub tf_bytes: Vec<u8>,
                          pub block_max_doc_id: DocId, pub block_max_score: f32, pub crc32c: u32 }
pub enum BlockCodec { Varbyte, For { bit_width: u8 }, Pfor { bit_width: u8 },
                      SimdBp128 { bit_width: u8 }, EliasFano }
pub trait PostingListReader: Send {
    fn doc_count(&self) -> u64;
    fn doc(&self) -> DocId;
    fn freq(&self) -> u32;
    fn next(&mut self) -> Option<DocId>;
    fn advance(&mut self, target: DocId) -> Option<DocId>;
    fn positions(&mut self) -> Option<&[u32]>;
    fn block_max_doc_id(&self) -> DocId;
    fn block_max_score(&self) -> f32;
    fn cost(&self) -> u64;
}
pub trait Scorer: Send { fn score(&self, doc: DocId) -> f32; fn max_score(&self) -> f32;
                          fn block_max_score(&self) -> f32; }
pub struct BM25Scorer   { /* k1, b, idf, avgdl, doc-length doc-values handle */ }
pub struct BM25FScorer  { /* per-field k1, b, weights */ }
pub struct ConstantScorer { value: f32 }
pub enum Query {
    Term { field: FieldId, term: String }, Phrase { field: FieldId, terms: Vec<String>, slop: u32 },
    Boolean { must: Vec<Query>, should: Vec<Query>, must_not: Vec<Query>, minimum_should_match: u32 },
    Fuzzy { field: FieldId, term: String, max_edits: u8, prefix_length: u8 },
    Range { field: FieldId, lower: Bound<RangeValue>, upper: Bound<RangeValue> },
    Wildcard { field: FieldId, pattern: String }, Regexp { field: FieldId, pattern: String },
    MatchAll, MatchNone, DenseVector { field: FieldId, query_vector: Vec<f32>, k: u32 },
    Boost { query: Box<Query>, factor: f32 }, ConstantScore { query: Box<Query>, score: f32 },
    DisjunctionMax { queries: Vec<Query>, tie_breaker: f32 },
}
pub enum RangeValue { Integer(i64), Float(f64), Date(i64), Keyword(String) }
pub struct SearchRequest { pub query: Query, pub top_k: u32, pub from: u32,
    pub fields_to_return: Vec<FieldId>, pub aggregations: Vec<AggregationRequest>,
    pub highlight: Option<HighlightRequest>, pub timeout_ms: u32, pub track_total_hits: bool,
    pub min_score: Option<f32> }
pub struct SearchHit { pub doc_id: DocId, pub external_id: String, pub score: f32,
    pub fields: Vec<(FieldId, FieldValue)>, pub highlights: Vec<(FieldId, Vec<String>)> }
pub struct SearchResponse { pub hits: Vec<SearchHit>, pub total: u64, pub took_ms: u32,
    pub timed_out: bool, pub aggregations: Vec<AggregationResult>, pub max_score: f32 }
pub struct SegmentMeta { pub id: SegmentId, pub generation: u64, pub doc_count: u32,
    pub deleted_count: u32, pub size_bytes: u64, pub created_at: i64, pub last_merged_at: i64,
    pub fields: Vec<FieldId>, pub min_doc_id: DocId, pub max_doc_id: DocId }
pub trait IndexWriter: Send {
    fn add_document(&mut self, doc: Document) -> Result<DocId>;
    fn delete_by_id(&mut self, external_id: &str) -> Result<()>;
    fn delete_by_query(&mut self, q: &Query) -> Result<u32>;
    fn commit(&mut self) -> Result<u64>;
    fn flush(&mut self) -> Result<SegmentMeta>;
    fn merge(&mut self, segments: &[SegmentId]) -> Result<SegmentMeta>;
}
pub trait IndexReader: Send + Sync {
    fn segments(&self) -> &[SegmentMeta]; fn doc_count(&self) -> u64; fn deleted_count(&self) -> u64;
    fn schema(&self) -> &Schema; fn snapshot(&self) -> SnapshotHandle;
}
pub enum SearchError {
    Schema(String), QueryParse { col: u32, msg: String }, Corruption(SegmentId, u64),
    Io(std::io::Error), Timeout(u32), PhaseDeferred(&'static str),
}
