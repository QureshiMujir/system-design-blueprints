Core schema & types (§11)

// formats (storage newtypes over raw bit patterns)
struct Binary16(u16);  struct Binary32(u32);  struct Binary64(u64);  struct Binary128(u128);
struct Bfloat16(u16);
struct Decimal32(u32); struct Decimal64(u64); struct Decimal128(u128);

trait Float {                      // associated constants per format
  const EXP_BITS: u32; const SIG_BITS: u32; const BIAS: i32;
  const EMIN: i32; const EMAX: i32;
  fn classify(self) -> FpClass;
  fn unpack(self) -> Unpacked;
  fn pack(u: Unpacked, ctx: &mut FpContext) -> Self;   // normalizes + rounds + sets flags
}

enum FpClass { Zero{neg:bool}, Subnormal{neg:bool}, Normal{neg:bool},
               Infinity{neg:bool}, Nan{signaling:bool, payload:u128} }

struct Unpacked { sign: bool, exp: i32, sig: u128, g: bool, r: bool, s: bool }  // G/R/S = guard/round/sticky

enum RoundingMode { TiesToEven, TiesToAway, TowardPos, TowardNeg, TowardZero }
struct Flags(u8);                  // NV|DZ|OF|UF|NX sticky bits
struct FpContext { mode: RoundingMode, flags: Flags }

// binary ops (NO bodies) — each updates ctx.flags
fn add<T:Float>(a:T, b:T, ctx:&mut FpContext) -> T;
fn mul<T:Float>(a:T, b:T, ctx:&mut FpContext) -> T;
fn div<T:Float>(a:T, b:T, ctx:&mut FpContext) -> T;
fn sqrt<T:Float>(a:T, ctx:&mut FpContext) -> T;
fn fma<T:Float>(a:T, b:T, c:T, ctx:&mut FpContext) -> T;     // single rounding
fn compare<T:Float>(a:T, b:T) -> Option<Ordering>;          // None = unordered (NaN)

// decimal
enum DecimalEncoding { Bid, Dpd }
struct DecimalUnpacked { sign: bool, coeff: u128, exp: i32 }      // coeff = integer significand
struct BigDecimal { unscaled: BigInt /*#370*/, scale: i32 }       // arbitrary precision
enum RoundPolicy { HalfEven, HalfUp, HalfDown, Ceiling, Floor, Down, Up }
struct Money { amount: BigDecimal, currency: Currency, policy: RoundPolicy }
struct Currency { code: [u8;3], minor_units: u8 }                 // e.g., USD, 2

// errors are expressed as flags + value; conversions that cannot represent set NV/NX
