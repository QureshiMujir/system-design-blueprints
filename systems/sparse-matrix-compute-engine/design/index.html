Core schema & types (§11)

pub enum SparseMat<T: Scalar, I: Index = i32> {
    Coo(Coo<T,I>), Csr(Csr<T,I>), Csc(Csc<T,I>),
    Bsr(Bsr<T,I>), Dia(Dia<T,I>), Ell(Ell<T,I>), SellCs(SellCs<T,I>),
}
pub struct Csr<T,I> { nrows:usize, ncols:usize, rowptr:Vec<I>, colind:Vec<I>, vals:Vec<T> }   // canonical
pub struct Csc<T,I> { nrows:usize, ncols:usize, colptr:Vec<I>, rowind:Vec<I>, vals:Vec<T> }   // = CSR of Aᵀ
pub trait Index: Copy + Ord { fn to_usize(self)->usize; fn from_usize(x:usize)->Self; const MAX:Self; }  // i32,i64
pub trait Scalar: la_core::Scalar { }   // reuse #349's Scalar (f32/f64)
pub struct PatternView<'a, I> { nrows:usize, ncols:usize, rowptr:&'a [I], colind:&'a [I] }  // structure only
pub struct Permutation<I> { perm:Vec<I>, iperm:Vec<I> }   // perm[new]=old; iperm[old]=new
impl Permutation { fn invert(&self)->Self; fn compose(&self,&Self)->Self; fn apply_sym(&self, &Csr)->Csr; }
pub struct EliminationTree<I> { parent: Vec<I>, post: Vec<I> /*postorder*/ }
pub struct SymbolicChol<I> { etree:EliminationTree<I>, colcount:Vec<I>, lnz:usize,
                             supernodes:Vec<Supernode<I>>, lpattern:CscPattern<I> }
pub struct Supernode<I> { first_col:I, n_cols:I, row_indices:Vec<I> /*shared structure*/ }
pub struct SymbolicLU<I> { col_etree:EliminationTree<I>, /*per-column patterns filled during numeric*/ }
pub struct CholFactor<T,I> { l: Csc<T,I>, perm: Permutation<I>, supernodes: Vec<Supernode<I>> }
pub struct LuFactor<T,I>   { l: Csc<T,I>, u: Csc<T,I>, p: Permutation<I>, q: Permutation<I>, growth:f64 }
fn spmv<T,I>(alpha:T, a:&Csr<T,I>, x:&[T], beta:T, y:&mut [T]) -> Result<(),SparseError>;
fn spgemm<T,I>(a:&Csr<T,I>, b:&Csr<T,I>) -> Result<Csr<T,I>,SparseError>;       // 2-pass
fn sptrsv<T,I>(uplo:Uplo, diag:Diag, a:&Csc<T,I>, x:&mut [T]) -> Result<(),SparseError>;
fn transpose<T,I>(a:&Csr<T,I>) -> Csc<T,I>;
fn cholesky<T,I>(a:&Csr<T,I>, order:Ordering) -> Result<CholFactor<T,I>,SparseError>;  // SPD
fn lu<T,I>(a:&Csr<T,I>, order:Ordering, pivot_tol:f64) -> Result<LuFactor<T,I>,SparseError>;
fn chol_solve<T,I>(f:&CholFactor<T,I>, b:&mut [T]);   // L Lᵀ x = b via SpTRSV ×2
enum Ordering { Natural, Amd, Rcm, NestedDissection }
fn cg<T,I, M: Preconditioner<T>>(a:&Csr<T,I>, b:&[T], x:&mut [T], m:&M, opt:IterOpts) -> IterResult;
fn gmres<T,I,M>(a:&Csr<T,I>, b:&[T], x:&mut [T], m:&M, restart:usize, opt:IterOpts) -> IterResult;
pub trait Preconditioner<T> { fn apply(&self, r:&[T], z:&mut [T]); }   // z = M⁻¹ r
pub struct IterOpts { rtol:f64, atol:f64, max_iter:usize }
pub struct IterResult { iters:usize, resid:f64, converged:bool, breakdown:Option<Breakdown> }
pub enum SparseError { BadIndex{row:usize,pos:usize}, NotCanonical, DimMismatch,
                       NotSpd, ZeroPivot(usize), FillMismatch{predicted:usize,actual:usize},
                       IndexOverflow, Breakdown(Breakdown) }
pub enum Breakdown { Stagnation, NanResidual, IcIndefinite, Lucky }
pub struct SolveTrace { method:&'static str, n:usize, nnz_a:usize, ordering:&'static str,
                        nnz_l:usize, fill_ratio:f64, n_supernodes:usize, level3_frac:f64,
                        symbolic_ns:u64, numeric_ns:u64, solve_ns:u64,
                        iters:usize, resid_history:Vec<f64> }
pub struct MatrixProfile { n:usize, nnz:usize, structurally_symmetric:bool, numerically_symmetric:bool,
                           max_row_nnz:usize, avg_row_nnz:f64, row_nnz_skew:f64, est_fill:Option<usize> }
pub struct SolvePlan { ordering:Ordering, format:Format, method:Method, precond:Option<PrecondKind>,
                       reason:&'static str /*human-readable justification*/ }
pub enum Method { CholeskyDirect, LuDirect, Cg, BiCgStab, Gmres }
