Core schema & types (§11)

struct Revision(u64);                 // monotonic logical clock value
struct Clock { current: AtomicU64, by_durability: [AtomicU64; 3] }
impl Clock { fn current(&self)->Revision; fn bump(&self, d: Durability)->Revision; }
enum Durability { Low, Medium, High }    // ordered; High changes rarest
impl Durability { fn index(self)->usize; }
struct QueryTypeId(u32);                 // one per #[query]/#[input] type
struct QueryKey { ty: QueryTypeId, bytes: KeyBytes }  // identifies a node
struct NodeId { index: u32, generation: u16 }          // interned, recyclable (INV-11)
struct Memo<V> {
  value: V, verified_at: Revision, changed_at: Revision,
  durability: Durability, deps: SmallVec<[Dependency; 4]>, state: NodeState,
}
struct Dependency { node: NodeId, changed_at_when_read: Revision }
enum NodeState { Unverified, Verifying, Verified(Revision), MaybeDirty, Computing }
struct Database { /* Arc<Inner>: clock, inputs, memo_cache, interner, ddg, exec */ }
impl Database {
  fn builder() -> DatabaseBuilder;
  fn query<Q: Query>(&self, key: Q::Key) -> Q::Value;        // demand (pull)
  fn set_input<I: Input>(&mut self, key: I::Key, value: I::Value); // mutation
  fn current_revision(&self) -> Revision;
  fn snapshot(&self) -> Snapshot;                            // read-only handle (Phase 9)
  fn sweep_gc(&mut self, target_bytes: usize);               // §8 phase
}
trait Query {
  type Key: Clone + Eq + Hash + Send + Sync;
  type Value: Clone + Eq + Send + Sync;     // Eq -> early cutoff (P4/INV-4)
  const TYPE_ID: QueryTypeId;
  const NONDETERMINISTIC: bool;             // INV-9 opt-out (disables memo)
  fn execute(db: &Database, key: &Self::Key) -> Self::Value; // the QueryFn body
}
trait Input {
  type Key: Clone + Eq + Hash + Send + Sync;
  type Value: Clone + Eq + Send + Sync;
  const DEFAULT_DURABILITY: Durability;
}
thread_local! { static STACK: RefCell<DependencyStack>; }
struct DependencyStack { frames: Vec<Frame> }
struct Frame { node: NodeId, reads: SmallVec<[Dependency; 8]>, on_stack: bool }
fn record_read(node: NodeId, changed_at: Revision); // called by every Database read
struct MemoCache<V> { shards: Box<[RwLock<HashMap<NodeId, Memo<V>>>]>, lru: LruIndex }
fn get(&self, id: NodeId) -> Option<MemoGuard<V>>;
fn insert(&self, id: NodeId, memo: Memo<V>);
fn evict_to(&self, target_bytes: usize, reachable: &ReachableSet);
enum EngineError {
  Cycle(CycleError),                 // recoverable (INV-7)
  OutOfMemo,                         // hard memory cap hit (P11)
  PersistFormat(PersistError),       // Phase 11 load mismatch
}
struct CycleError { cycle: Vec<QueryKey> }   // participating keys, in order
struct CancellationToken(Arc<AtomicBool>);
impl CancellationToken { fn cancel(&self); fn is_cancelled(&self)->bool; }
enum TraceEvent {
  RevisionBump{rev:Revision, cause:QueryKey, durability:Durability},
  GreenHit{key:QueryKey}, Recompute{key:QueryKey, reason:RecomputeReason},
  Backdate{key:QueryKey}, CutoffFired{key:QueryKey},
  CycleDetected{cycle:Vec<QueryKey>}, Evicted{key:QueryKey, bytes:usize},
}
enum RecomputeReason { FirstRun, DepChanged(NodeId), Nondeterministic, EvictedMiss }
