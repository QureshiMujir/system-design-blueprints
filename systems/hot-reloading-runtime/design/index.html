Core schema & types (§11)

// identity & versioning
pub struct ModuleId(pub String);                          // stable across versions
pub struct Version { pub major: u32, pub minor: u32, pub patch: u32 }   // SemVer (Compatibility & Versioning)
pub struct Generation(pub u64);                           // monotonic per module (I9)
pub struct ReloadId(pub u128);                            // unique per reload attempt (trace key, I7)
pub struct TypeFingerprint(pub [u8; 32]);                 // structural hash of a type (compat, I5)

pub struct Signature {                                    // ABI + type fingerprint of a version
    pub abi: AbiKind,                                     // Native | Interpreted
    pub entry_points: Vec<EntrySig>,                      // name + arg/ret fingerprints
    pub state_type: TypeFingerprint,
}
pub struct EntrySig { pub name: String, pub args: TypeFingerprint, pub ret: TypeFingerprint }

pub enum HrrError {                                       // typed; reject-not-crash (I5, ADR-015)
    Load(LoadError), Compat(CompatError), Quiesce(QuiesceError),
    Migrate(MigrateError), Verify(VerifyError), Rollback(RollbackError) }

// module abstraction (host implements — P13 stable API, ADR-022)
pub trait Module: Send + Sync {
    fn signature(&self) -> &Signature;
    fn entry(&self, name: &str) -> Option<EntryFn>;       // called THROUGH hrr-swap only (I11)
}
pub trait Reloadable: Module {
    type State: Send + 'static;                           // the live, preserved state
    fn init_state() -> Self::State;                       // first load
    fn migrate(old: ErasedState, to: &Version)            // code_change; total (I6)
        -> Result<Self::State, MigrateError>;
    fn rollback_migrate(new: Self::State)                 // inverse; exact revert (I4/I6)
        -> Result<ErasedState, MigrateError>;
}
pub type EntryFn = unsafe extern "Rust" fn(state: *mut (), args: ErasedArgs) -> ErasedRet;
// loaded code (hrr-loader, ADR-002/ADR-003)
pub enum CodeHandle { Native(NativeLib), Interpreted(InterpModule) }   // owns code lifetime (I4)
pub struct NativeLib { /* libloading::Library + resolved EntryFn table */ }
impl NativeLib {
    pub fn load(path: &Path) -> Result<Self, LoadError>;  // dlopen + resolve symbols
    pub fn resolve(&self, name: &str) -> Option<EntryFn>; // Symbol lifetime ⊆ Library (I4)
}                                                         // Drop = dlclose after grace period only (I4)

// registry (hrr-module, I9/I10)
pub struct LoadedVersion { pub gen: Generation, pub version: Version,
    pub sig: Signature, pub code: CodeHandle, pub health: Health }
pub enum Health { Healthy, Suspect, Faulted }
pub struct ModuleSlot {                                   // ≤2 live versions (I10)
    pub id: ModuleId, pub active: Generation,
    pub versions: BTreeMap<Generation, LoadedVersion>,    // bounded to {active, incoming}
    pub state: StateCell }                                // live state, migrated across reloads
impl ModuleSlot { pub fn active_version(&self) -> &LoadedVersion; }

// atomic swap (hrr-swap, ADR-004)
pub struct Indirection { /* arc-swap of immutable dispatch snapshot */ }
impl Indirection {
    pub fn dispatch(&self, module: &ModuleId, entry: &str) -> EntryFn;  // wait-free read (I1)
    pub fn cutover(&self, module: &ModuleId, to: Generation);           // ONE atomic store (I1)
}
// quiescence (hrr-safepoint, ADR-006, I3/I12)
pub trait SafePoint {
    fn begin_quiesce(&self, m: &ModuleId);                // stop admitting / arm barrier
    fn wait_quiescent(&self, m: &ModuleId, deadline: Duration)
        -> Result<Quiescent, QuiesceError>;               // count==0 or epoch grace, else timeout
}
pub struct CallGuard<'a> { /* RAII: acq-rel atomic on entry/drop; ties symbol to generation (I4) */ }

// state migration (hrr-state, ADR-007, I2/I6)
pub struct StagedState { /* migrated copy; original untouched until commit */ }
impl StagedState {
    pub fn migrate(old: &ErasedState, r: &dyn Reloadable, to: &Version)
        -> Result<Self, MigrateError>;                    // total (I6); runs at safe point
    pub fn commit(self, slot: &mut ModuleSlot);           // atomic StateCell swap
}

// compatibility (hrr-compat, ADR-008, I5)
pub fn check(old: &Signature, new: &Signature, has_migration: bool)
    -> Result<CompatLevel, CompatError>;                  // reject before swap (I5)
pub enum CompatLevel { Identical, MinorAdditive, MajorMigrated }

// coordinator (hrr-coordinator, ADR-010, I2/I10)
pub enum ReloadState { Idle, Loading, Compat, Quiescing, Swapping,
    Migrating, Verifying, Committed, RollingBack, RolledBack }
pub struct ReloadRequest { pub id: ReloadId, pub module: ModuleId, pub path_or_src: Source }
pub struct ReloadOutcome { pub id: ReloadId, pub committed: bool,
    pub from: Generation, pub to: Generation }
impl Coordinator {
    pub fn reload(&self, req: ReloadRequest) -> Result<ReloadOutcome, HrrError>;
    pub fn force_rollback(&self, m: &ModuleId) -> Result<ReloadOutcome, HrrError>;
}

// events / replay (hrr-replay, ADR-019, I7)
pub struct ReloadEvent { pub id: ReloadId, pub at: Instant,
    pub from: ReloadState, pub to: ReloadState, pub detail: EventDetail }
