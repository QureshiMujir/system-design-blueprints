Core schema & types (§11)

// crates/serdes-schema/src/kind.rs
pub enum SchemaKind — variants: Primitive(PrimitiveType), Record(RecordSchema),
    Enum(EnumSchema), Array(Box<SchemaKind>), Map(Box<SchemaKind>), Union(Vec<SchemaKind>),
    Optional(Box<SchemaKind>), Reference(String) /* PhaseDeferred("§15") resolution */;
pub enum PrimitiveType — variants: Int32, Int64, UInt32, UInt64, Float32, Float64, Bool,
    String, Bytes;
// crates/serdes-schema/src/field.rs
pub struct Field — fields: name: String; field_number: Option<u32> /* Some for proto,
    None for Avro, order-based, I4 */; schema: SchemaKind; required: bool; repeated: bool;
    default: Option<LiteralValue>; aliases: Vec<String>;
pub struct RecordSchema — fields: namespace: String; name: String; fields: Vec<Field>;
    fingerprint: SchemaFingerprint;
pub struct EnumSchema — fields: namespace: String; name: String; symbols: Vec<String>;
    default: Option<String>;
// crates/serdes-schema/src/fingerprint.rs, ids.rs
pub struct SchemaFingerprint(pub [u8; 8]);
pub struct SchemaId(pub i32);
pub enum CompatibilityMode — variants: None, Backward, Forward, Full, BackwardTransitive,
    ForwardTransitive, FullTransitive;
// crates/serdes-compat/src/report.rs, lib.rs
pub struct CompatibilityResult — fields: compatible: bool; violations: Vec<CompatibilityViolation>;
pub struct CompatibilityViolation — fields: field_path: String; reason: ViolationReason;
pub trait CompatibilityChecker — methods:
    fn check(&self, old: &SchemaKind, new: &SchemaKind, mode: CompatibilityMode) -> CompatibilityResult;
// crates/serdes-store/src/lib.rs
pub struct RegisterRequest — fields: subject: String; schema: SchemaKind; mode: CompatibilityMode;
pub struct StoredSchema — fields: id: SchemaId; subject: String; version: u32;
    schema: SchemaKind; fingerprint: SchemaFingerprint;
pub trait SchemaStore: Send + Sync — methods:
    async fn insert_schema(&self, req: RegisterRequest) -> Result<SchemaId, StoreError>;
    async fn get_schema_by_id(&self, id: SchemaId) -> Result<StoredSchema, StoreError>;
    async fn get_schema_by_fingerprint(&self, fp: SchemaFingerprint) -> Result<Option<StoredSchema>, StoreError>;
    async fn list_versions(&self, subject: &str) -> Result<Vec<u32>, StoreError>;
    async fn list_subjects(&self) -> Result<Vec<String>, StoreError>;
// crates/serdes-proto-parse/src/descriptor.rs
pub struct ProtoDescriptor — fields: package: String; messages: Vec<RecordSchema>;
    enums: Vec<EnumSchema>; imports: Vec<String>;
// crates/serdes-proto-codec/src/wire.rs
pub struct WireTag — fields: field_number: u32; wire_type: WireType;
pub enum WireType — variants: Varint, Fixed64, LengthDelimited, Fixed32;
pub trait ProtoEncoder — methods:
    fn encode(&self, value: &SchemaKind, buf: &mut Vec<u8>) -> Result<(), EncodeError>;
pub trait ProtoDecoder<'buf> — methods:
    fn decode(&self, buf: &'buf [u8]) -> Result<DecodedMessage<'buf>, DecodeError>;
pub struct DecodedMessage<'buf> — fields: fields: Vec<DecodedField<'buf>>; unknown: UnknownFieldSet;
pub struct UnknownFieldSet — fields: raw: Vec<u8> /* I3: retained byte-exact */;
