Core schema & types (§11)

#[derive(Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]
pub struct OpId { pub site: SiteId, pub seq: u64 }
pub struct Lamport(pub u64);
impl Lamport {
    pub fn tick(&mut self) -> Lamport;          // self += 1; return new value
    pub fn merge(&mut self, other: Lamport);    // self = max(self, other) + 1
}
pub struct StateVector(BTreeMap<SiteId, u64>);
impl StateVector {
    pub fn observe(&mut self, id: OpId);
    pub fn get(&self, site: SiteId) -> u64;
    pub fn merge(&mut self, other: &StateVector);
    pub fn diff<'a>(&'a self, other: &'a StateVector) -> impl Iterator<Item=(SiteId,u64,u64)> + 'a;
    pub fn dominates(&self, other: &StateVector) -> bool;
}
pub enum Content { Char(char), Deleted }

pub struct Item {
    pub id:           OpId,
    pub origin_left:  Option<OpId>,
    pub origin_right: Option<OpId>,
    pub content:      Content,
    pub lamport:      Lamport,
}
pub enum TextOp {
    Insert { id: OpId, left: Option<OpId>, right: Option<OpId>, ch: char },
    Delete { target: OpId },
}
pub fn integration_ordering(new: &Item, items: &Arena) -> ArenaKey;
// Contract: inserts `new` at deterministic position among concurrent items;
//           outcome depends only on {id, origin_left, origin_right, lamport};
//           never on arrival order (P1, P2).
pub struct Position { pub line: u32, pub col: u32 }   // UTF-8 codepoint column
pub struct Range    { pub start: Position, pub end: Position }
pub struct Document {
    pub id:           DocId,
    items:            Arena,                    // lcide-crdt arena
    index:            PositionIndex,            // visible-offset <-> OpId; updated on each apply
    pub state_vector: StateVector,
    undo_stack:       Vec<UndoEntry>,           // per-site; selective undo entries
    redo_stack:       Vec<UndoEntry>,
}
impl Document {
    pub fn apply_local(&mut self, pos: Position, ch: char) -> TextOp;
    pub fn apply_remote(&mut self, op: &TextOp) -> Result<(), CausalNotReady>;
    pub fn materialize(&self) -> String;        // pure fn of item set; deterministic (P9)
    pub fn content_hash(&self) -> u64;          // xxHash of materialized text (P10)
    pub fn undo(&mut self, site: SiteId) -> Option<TextOp>;
    pub fn redo(&mut self, site: SiteId) -> Option<TextOp>;
}
pub struct Snapshot {
    pub doc_id:              DocId,
    pub state_vector:        StateVector,
    pub materialized_items:  Vec<Item>,         // all non-GC'd items in integration order
    pub taken_at_op:         OpId,              // last op included in this snapshot
}
pub struct LWWReg<T: Clone> { value: T, ts: Lamport, site: SiteId }
impl<T: Clone> LWWReg<T> {
    pub fn set(&mut self, value: T, ts: Lamport, site: SiteId);
    pub fn get(&self) -> &T;
    pub fn merge(&mut self, other: &LWWReg<T>);
}
pub struct FileNode {
    pub id:       OpId,                          // creation op; stable identity
    pub kind:     NodeKind,                      // enum { File, Dir }
    pub name_reg: LWWReg<String>,                // concurrent rename: LWW
    pub parent:   LWWReg<OpId>,                  // concurrent move: LWW parent pointer
    pub doc:      Option<DocId>,                 // Some(_) iff kind == File
    pub deleted:  bool,                          // tombstone; set by delete op; never cleared
}
pub struct WorkspaceTree {
    nodes:      BTreeMap<OpId, FileNode>,
    path_index: PathIndex,                       // path string -> OpId; invalidated on rename/move
    root:       OpId,
}
impl WorkspaceTree {
    pub fn apply_op(&mut self, op: &FsCrdtOp) -> Result<(), FsOpError>;
    pub fn resolve_path(&self, path: &str) -> Option<OpId>;
    pub fn children(&self, parent: OpId) -> impl Iterator<Item=&FileNode>;
}
pub enum FsCrdtOp {
    Create { id: OpId, kind: NodeKind, parent: OpId, name: String, doc: Option<DocId> },
    Delete { target: OpId },
    Rename { target: OpId, name: String, ts: Lamport, site: SiteId },
    Move   { target: OpId, new_parent: OpId, ts: Lamport, site: SiteId },
}
pub enum SyncMessage {
    Hello     { client_id: ClientId, workspace: WorkspaceId, token: String },
    AuthOk    { site_id: SiteId, server_sv: StateVector },
    SvRequest { sv: StateVector },
    Delta     { ops: Vec<TextOp>, fs_ops: Vec<FsCrdtOp> },
    Awareness { state: AwarenessState },
    FsOp      { op: FsCrdtOp },
    LspMsg    { payload: Vec<u8> },              // raw LSP JSON-RPC bytes
    Ack       { up_to_seq: u64, site: SiteId },
    Bye       { reason: String },
}
pub struct OpLogEntry {
    pub seq:     u64,       // server-global monotone sequence; used for op-log file offsets
    pub doc_id:  DocId,
    pub op:      TextOp,    // or FsCrdtOp variant; distinguished by tag byte in framing
    pub recv_ts: u64,       // server wall-clock micros; diagnostic only; not causal
}
pub struct AwarenessState {
    pub client:     ClientId,
    pub site:       SiteId,
    pub doc:        DocId,
    pub cursor:     Position,
    pub selection:  Range,
    pub viewport:   Range,           // visible line range in client editor
    pub color:      u32,             // ARGB; assigned by server on join
    pub expires_at: u64,             // server monotone micros; evicted when now > expires_at
}
pub struct SessionRoom {
    pub workspace:    WorkspaceId,
    members:          BTreeMap<ClientId, Member>,
    pub order_clock:  Lamport,                   // server delivery authority; not causal
    send_queues:      BTreeMap<ClientId, BoundedSender<SyncMessage>>,
    docs:             BTreeMap<DocId, Document>,
    tree:             WorkspaceTree,
    presence:         BTreeMap<ClientId, AwarenessState>,
}

pub struct Member {
    pub client_id: ClientId,
    pub site_id:   SiteId,
    pub acl:       AccessLevel,                  // enum { Read, Write }
    pub sv:        StateVector,                  // last known peer SV for delta computation
}
