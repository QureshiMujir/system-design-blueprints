Core schema & types (§11)

pub struct Vector { dim: u32, dtype: Dtype, bytes: Box<[u8]>, layout: Layout }
pub struct VectorView<'a> { dim: u32, dtype: Dtype, bytes: &'a [u8], layout: Layout }

pub enum Dtype { F32, F16, Bf16, I8, U8, PqCode { m: u8, ks: u8 } }
pub enum Layout { Contiguous, Strided { byte_stride: u32 } }

impl<'a> VectorView<'a> {
    fn dim(&self) -> u32;
    fn dtype(&self) -> Dtype;
    fn as_f32(&self) -> Result<&'a [f32], Error>;
    fn as_i8(&self) -> Result<&'a [i8], Error>;
}
pub struct Embedding { id: EmbeddingId, vector: Vector, metadata: Metadata }
pub struct EmbeddingId(pub u64);
pub struct Metadata { source: Option<String>, chunk_id: Option<ChunkId>,
    modality: Modality, model_id: Option<ModelId>, created_at_unix_ns: Option<u64> }

pub struct Chunk { id: ChunkId, content: ChunkContent, source: ChunkSource, hash: ContentHash }
pub enum ChunkContent { Text(String), Image(ImageBytes), Audio(AudioBytes) }
pub enum Distance { L2, Ip, Cosine }

pub struct IndexParams { dim: u32, distance: Distance, family: IndexFamily,
    seed: u64, thread_pool: ThreadPool }

pub enum IndexFamily {
    Flat { dtype: Dtype },
    KdTree { leaf_size: u32 },
    BallTree { leaf_size: u32 },
    Ivf { nlist: u32, nprobe: u32 },
    IvfPq { nlist: u32, nprobe: u32, m: u8, ks: u8 },
    Hnsw { m: u32, ef_construction: u32, ef_search: u32 },
    Nsg { r: u32, l: u32, c: u32 },
    Scann { leaves: u32, avq_eta: f32 },
}

pub struct SearchParams { k: u32, ef_search: Option<u32>, nprobe: Option<u32>,
    timeout: Option<core::time::Duration> }
pub struct SearchHit { id: EmbeddingId, score: f32, distance_kind: Distance }
pub type SearchResult = Vec<SearchHit>;

pub struct GroundTruth { queries: Vec<EmbeddingId>, neighbours: Vec<Vec<EmbeddingId>>, k: u32 }
pub struct Recall { k: u32, mean: f32, min: f32, p50: f32, p95: f32, p99: f32 }
pub enum Error {
    DimMismatch { expected: u32, got: u32 },
    DtypeMismatch { expected: Dtype, got: Dtype },
    ModelNotFound(String),
    ModelChecksum { expected: String, got: String },
    UnsupportedDistance { family: &'static str, distance: Distance },
    RecallBelowThreshold { got: f32, threshold: f32 },
    Timeout,
    PhaseDeferred(&'static str),
    Io(std::io::Error),
    Other(String),
}
pub trait Quantizer: Send + Sync {
    fn quantize(&self, v: VectorView<'_>) -> Result<Vector, Error>;
    fn dequantize(&self, v: VectorView<'_>) -> Result<Vector, Error>;
    fn distance(&self, a: VectorView<'_>, b: VectorView<'_>, kind: Distance) -> Result<f32, Error>;
    fn reconstruction_error(&self, sample: &[VectorView<'_>]) -> f32;   // P11
}

pub struct PqCodebook { m: u8, ks: u8, sub_dim: u32, centroids: Box<[f32]> }
pub trait Fusion: Send + Sync {
    fn merge(&self, dense: &[SearchHit], sparse: &[SearchHit], k: u32) -> SearchResult;
}
pub trait Tokenizer: Send + Sync {
    fn encode(&self, text: &str) -> Result<Encoded, Error>;
    fn encode_batch(&self, texts: &[&str]) -> Result<Vec<Encoded>, Error>;
    fn decode(&self, ids: &[u32]) -> Result<String, Error>;
    fn vocab_size(&self) -> u32;
    fn pad_id(&self) -> u32;
    fn max_length(&self) -> Option<u32>;
}

pub struct Encoded { ids: Vec<u32>, attention_mask: Vec<u8>, offsets: Vec<(u32, u32)> }
pub trait Embedder: Send + Sync {
    fn dim(&self) -> u32;
    fn dtype(&self) -> Dtype;
    fn modality(&self) -> Modality;
    fn embed(&self, inputs: &[EmbeddingInput<'_>]) -> Result<Vec<Vector>, Error>;
}

pub enum EmbeddingInput<'a> { Text(&'a str), Image(&'a [u8]), AudioPcm(&'a [f32], u32) }
pub trait Index: Send + Sync {
    fn dim(&self) -> u32;
    fn distance(&self) -> Distance;
    fn len(&self) -> usize;
    fn search(&self, query: VectorView<'_>, params: &SearchParams) -> Result<SearchResult, Error>;
    fn search_batch(&self, queries: &[VectorView<'_>], params: &SearchParams)
        -> Result<Vec<SearchResult>, Error>;
    fn recall_at(&self, k: u32, gt: &GroundTruth) -> Recall;               // P5
    fn serialize(&self, w: &mut dyn std::io::Write) -> Result<(), Error>;
}
pub trait IndexBuilder: Send {
    type Output: Index + 'static;
    fn add(&mut self, embedding: Embedding) -> Result<(), Error>;
    fn add_batch(&mut self, batch: &[Embedding]) -> Result<(), Error>;
    fn build(self) -> Result<Self::Output, Error>;                        // consumes self (I6)
}
