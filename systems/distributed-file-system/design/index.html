Core schema & types (§11)

// dfs-core: identifiers
pub struct ChunkId(pub u64);
pub struct FileId(pub u64);
pub struct ChunkserverId(pub u64);
pub struct LeaseId(pub u128);

pub const CHUNK_SIZE_DEFAULT: u64;        // 64 MiB
pub const CHUNK_SIZE_LARGE: u64;          // 128 MiB, optional
pub const REPLICATION_FACTOR_DEFAULT: u8; // 3
pub const REPLICATION_FACTOR_MAX: u8;     // 7
pub const LEASE_DURATION_DEFAULT: Duration;  // 60s
pub const LEASE_RENEW_INTERVAL: Duration;    // 20s
pub const MAX_RECORD_SIZE: u32;              // 16 MiB
// dfs-core: namespace types — ownership: Master exclusive; concurrency: RwLock-guarded (§13)
pub struct Path(Vec<String>);          // normalised, no trailing slash
pub enum InodeKind { File, Directory, Symlink }
pub enum Durability { Replication { rf: u8 }, Erasure { k: u8, m: u8 } }  // Erasure deferred, §36

pub struct Inode {
    pub id: FileId, pub kind: InodeKind, pub mode: u32,
    pub uid: u32, pub gid: u32,
    pub mtime_ns: i64, pub atime_ns: i64, pub ctime_ns: i64,
    pub size: u64, pub replication: u8, pub chunk_size: u64,
    pub durability: Durability,
    pub chunks: Vec<ChunkId>,          // ordered by index within file
}
// dfs-core: lease + replica — persistence: logged (I7); concurrency: per-chunk mutex (§13)
pub struct Lease {
    pub id: LeaseId, pub chunk_id: ChunkId, pub primary: ChunkserverId,
    pub replicas: Vec<ChunkserverId>,
    pub issued_at: Timestamp, pub expires_at: Timestamp,
    pub epoch: u64,                   // I10: monotonic per chunk
}

pub struct ReplicaSet {
    pub chunk_id: ChunkId, pub members: Vec<ChunkserverId>,
    pub primary: Option<ChunkserverId>, pub durability: Durability,
}

pub enum ChunkState { Allocating, Open, Closed, UnderReplicated, Corrupt }

pub struct ChunkMeta {
    pub id: ChunkId, pub file: FileId, pub index: u32,
    pub size: u64, pub epoch: u64, pub durability: Durability,
    pub replicas: Vec<ChunkserverId>, pub primary: Option<ChunkserverId>,
    pub state: ChunkState,
}
// dfs-core: cluster topology & heartbeat — serialization: protobuf (ADR-12)
pub struct ClusterTopology { pub regions: BTreeMap<String, Region> }
pub struct Region { pub zones: BTreeMap<String, Zone> }
pub struct Zone { pub racks: BTreeMap<String, Rack> }
pub struct Rack { pub chunkservers: Vec<ChunkserverId> }

pub enum ChunkserverStatus { Live, Suspect, Dead, Decommissioning, Decommissioned }

pub struct Heartbeat {
    pub from: ChunkserverId, pub at: Timestamp,
    pub used_bytes: u64, pub free_bytes: u64,
    pub chunk_report: Vec<ChunkReportItem>,   // delta since last report
    pub failed_replicas: Vec<ChunkId>,
}

pub struct ChunkReportItem { pub chunk_id: ChunkId, pub epoch: u64, pub bytes: u64, pub crc: u64 }
// dfs-core: RPC payloads — compatibility: versioned (P13); no bodies, signatures only
pub struct WriteChunk { pub chunk_id: ChunkId, pub epoch: u64, pub offset: u64, pub data: Bytes, pub crc: u64 }
pub struct ReadChunk { pub chunk_id: ChunkId, pub offset: u64, pub length: u32 }
pub struct RecordAppend { pub chunk_id: ChunkId, pub epoch: u64, pub data: Bytes, pub crc: u64 }

pub trait MasterService {
    fn create(&self, path: &Path, mode: u32) -> Result<FileId>;
    fn open(&self, path: &Path) -> Result<(Inode, Vec<ChunkId>)>;
    fn list(&self, path: &Path) -> Result<Vec<(String, FileId)>>;
    fn rename(&self, src: &Path, dst: &Path) -> Result<()>;
    fn delete(&self, path: &Path, recursive: bool) -> Result<()>;
    fn allocate_chunk(&self, file: FileId) -> Result<(ChunkId, ReplicaSet, Lease)>;
    fn complete_chunk(&self, chunk: ChunkId, len: u64, crc: u64) -> Result<()>;
    fn lease(&self, chunk: ChunkId) -> Result<Lease>;
    fn renew_lease(&self, id: LeaseId) -> Result<Lease>;
    fn snapshot(&self, path: &Path, name: &str) -> Result<()>;
    fn fsck(&self, path: &Path) -> Result<FsckReport>;
}

pub trait ChunkserverService {
    fn write(&self, w: WriteChunk) -> Result<()>;
    fn read(&self, r: ReadChunk) -> Result<Bytes>;
    fn record_append(&self, a: RecordAppend) -> Result<u64>;   // returns offset
    fn replicate(&self, chunk: ChunkId, peer: ChunkserverId) -> Result<()>;
    fn merkle_root(&self, chunk: ChunkId) -> Result<[u8; 32]>;
    fn report(&self) -> Result<Vec<ChunkReportItem>>;
}

pub trait Placement {
    fn place(&self, chunk_id: ChunkId, topology: &ClusterTopology, durability: Durability)
        -> Result<Vec<ChunkserverId>>;
}
// dfs-core: quota & fsck — ownership: Master exclusive; used by Phase 8 (§20) and fsck audits (§7, §24)
pub struct Quota { pub bytes: Option<u64>, pub files: Option<u64> }

pub struct QuotaUsage {
    pub limit_bytes: Option<u64>, pub limit_files: Option<u64>,
    pub used_bytes: u64, pub used_files: u64,
}

pub struct FsckReport {
    pub under_replicated: Vec<ChunkId>,
    pub over_replicated: Vec<ChunkId>,
    pub corrupt_replicas: Vec<(ChunkId, ChunkserverId)>,
    pub misplaced: Vec<ChunkId>,       // violates I2
    pub orphan_chunks: Vec<(ChunkserverId, ChunkId)>,
}
