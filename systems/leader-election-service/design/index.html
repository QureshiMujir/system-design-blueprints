Core schema & types (§11)

// election-core
struct ElectionRecord { group_id:u64, owner:SessionId, fence_token:u64,
                        lease_expiry_mono:Instant, term:u64, revision:u64 }
   // owned by apply loop; immutable after commit; persisted in WAL; proto3; add-only fields (P12)

enum LeaseState { Pending, Active, Expiring, Revoked, Expired }   // §8 state table; Copy

struct Lease { owner:SessionId, fence_token:u64, state:LeaseState, expiry_mono:Instant }
   // mutated only on apply; single-writer; not directly serialized (derived from ElectionRecord)

struct FenceToken(u64);                  // monotonic per group (I2); Ord; serialized as fixed64
struct SessionId(Uuid);                  // client identity; Copy; proto bytes(16)
struct SessionHandle { id:SessionId, last_renew_mono:Instant, ttl_ms:u32 }
   // shared read (Arc), atomic expiry stamp; not persisted (rebuilt from active leases)

struct WatchCursor { group_id:u64, from_revision:u64 }   // per-subscriber; Copy; proto

// raft-engine
struct LogEntry { term:u64, index:u64, payload:Bytes, crc32c:u32 }
   // append-only (I5); persisted in WAL; length-prefixed proto; crc covers term|index|payload
enum RaftRole { Follower, Candidate, Leader }            // single apply thread owns transitions
struct RaftState { current_term:u64, voted_for:Option<u64>, commit_index:u64,
                   last_applied:u64, role:RaftRole }
   // current_term/voted_for durable before responding (I14); rest in-memory
struct RequestVote { term:u64, candidate_id:u64, last_log_index:u64, last_log_term:u64,
                     pre_vote:bool }   // pre_vote round does not bump term (§15.2)
struct RequestVoteResp { term:u64, vote_granted:bool }
struct AppendEntries { term:u64, leader_id:u64, prev_log_index:u64, prev_log_term:u64,
                       entries:Vec<LogEntry>, leader_commit:u64 }
struct AppendEntriesResp { term:u64, success:bool, conflict_index:u64, conflict_term:u64 }
   // conflict hints let the leader back up by term, not one index at a time (I6)
struct InstallSnapshot { term:u64, leader_id:u64, last_index:u64, last_term:u64,
                         offset:u64, data:Bytes, done:bool }   // chunked (§18)
struct ReadIndex { group_id:u64 }   // linearizable-read barrier request (§15.2, MM9)
struct MembershipConfig { voters:BTreeSet<u64>, learners:BTreeSet<u64>,
                          pending_change:Option<NodeChange> }
   // single-server change only (ADR-21,I11); persisted as a special log entry
enum NodeChange { AddLearner(u64), PromoteVoter(u64), RemoveNode(u64) }   // §16
// wal
struct Segment { id:u64, base_index:u64, path:PathBuf, sealed:bool }   // rotation unit
struct RecordHeader { len:u32, crc32c:u32, kind:u8 }                   // on-disk frame (I15)
struct SyncPolicy { group_commit_bytes:usize, max_batch:usize, fdatasync:bool } // ADR-07

// snapshot
struct SnapshotManifest { snapshot_id:u64, last_index:u64, last_term:u64,
                          chunk_count:u32, sha256:[u8;32] }   // persisted; verified on install (I12)

// transport
struct PeerLink { node_id:u64, tls:TlsConn, inflight:BoundedWindow }   // bounded (P4/P5)
enum FrameKind { Vote, VoteResp, Append, AppendResp, Snapshot, SnapshotResp }

// failure-detector
struct PhiEstimator { window:RingBuffer<Duration>, phi_threshold:f64 }  // ADR-08; suspicion=f64

// election-core watch + dedup
struct WatchEvent { group_id:u64, revision:u64, owner:Option<SessionId>, fence_token:u64,
                    kind:WatchKind }   // shared Bytes across subscribers (§19)
enum WatchKind { OwnershipChanged, LeaseExpiring, Compacted }
struct DedupEntry { request_id:u64, result_revision:u64 }   // per-session, bounded (§8)
struct BoundedWindow { max_inflight:usize, in_use:usize }   // P4/P5 flow control

// error taxonomy (returned across api boundary; stable proto codes, P12)
enum ElectionError { NotLeader{leader_hint:Option<u64>}, LeaseExpired, QuorumUnavailable,
                     TermMismatch{observed:u64}, StaleFenceToken{have:u64, need:u64},
                     CorruptWal{segment:u64}, SnapshotRejected, BackpressureExceeded,
                     InvalidRequest{reason:String} }
