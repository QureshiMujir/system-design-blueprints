Core schema & types (§11)

// crates/ssa-ir/src/module.rs — ownership: Module owns all Functions/Globals;
// mutation: only via ssa-builder cursor API (P9); concurrency: Send, not Sync
// during single-pass mutation; persistence: none (Phase 0–9, ADR-05); serialization: none.
pub struct Module {
    pub name: String,
    pub functions: SlotMap<FunctionId, Function>,
    pub globals: SlotMap<GlobalId, GlobalVar>,
    pub types: TypeTable,
}

pub struct Function {
    pub name: String,
    pub sig: FunctionType,
    pub blocks: SlotMap<BlockId, BasicBlock>,
    pub entry: BlockId,
    pub values: SlotMap<ValueId, ValueDef>,
}

pub struct BasicBlock {
    pub id: BlockId,
    pub insts: Vec<InstructionId>,
    pub terminator: InstructionId,   // I6: last slot, always present post-construction
    pub preds: SmallVec<[BlockId; 4]>,
}
// crates/ssa-ir/src/instruction.rs — mutation: via replace_all_uses_with /
// set_operand only (I13); concurrency: interior SlotMap access guarded by
// &mut Function; compatibility: Opcode is #[non_exhaustive] across minor versions.
pub struct Instruction {
    pub id: InstructionId,
    pub opcode: Opcode,
    pub operands: SmallVec<[ValueId; 3]>,
    pub result: Option<ValueId>,
    pub ty: TypeId,
}

#[non_exhaustive]
pub enum Opcode {
    Br { target: BlockId },
    CondBr { cond: ValueId, t: BlockId, f: BlockId },
    Return { value: Option<ValueId> },
    Load { ptr: ValueId, volatile: bool },
    Store { value: ValueId, ptr: ValueId, volatile: bool },
    Add(ValueId, ValueId, WrapFlags), Sub(ValueId, ValueId, WrapFlags),
    Mul(ValueId, ValueId, WrapFlags),
    ICmp { pred: ICmpPred, lhs: ValueId, rhs: ValueId },
    Phi { incoming: Vec<(BlockId, ValueId)> },   // I3
    Call { callee: CalleeRef, args: Vec<ValueId> },
}
// crates/ssa-ir/src/value.rs — ownership: Value owned by the defining Function's
// SlotMap; mutation: uses list maintained transactionally with operand edits (I13).
pub struct ValueDef {
    pub id: ValueId,
    pub ty: TypeId,
    pub kind: ValueKind,
    pub uses: SmallVec<[Use; 4]>,   // I13: reverse edge set
}

pub enum ValueKind {
    Instruction(InstructionId),
    Argument(usize),
    Constant(ConstantValue),
    Undef,
}

pub struct Use {
    pub user: InstructionId,
    pub operand_index: usize,
}
// crates/ssa-domtree/src/lib.rs — concurrency: computed once per Function,
// Send + immutable after construction; persistence: cached in AnalysisManager (I9).
pub struct DominatorTree {
    idom: HashMap<BlockId, BlockId>,
    dfs_pre: HashMap<BlockId, u32>,
    dfs_post: HashMap<BlockId, u32>,
}

impl DominatorTree {
    pub fn idom(&self, b: BlockId) -> Option<BlockId>;
    pub fn dominates(&self, a: BlockId, b: BlockId) -> bool;   // O(1), I2
    pub fn dominance_frontier(&self, cfg: &CfgView) -> HashMap<BlockId, BTreeSet<BlockId>>;
    pub fn build_cooper_harvey_kennedy(f: &Function) -> Self;   // ADR-32 primary
    pub fn build_lengauer_tarjan(f: &Function) -> Self;         // ADR-32 cross-check
}
// crates/ssa-analyses/src/manager.rs — I9: generation-counter invalidation;
// concurrency: one AnalysisManager per Function, no cross-thread sharing (P9, ADR-08).
pub trait Analysis: 'static + Send {
    type Result: 'static + Send;
    fn run(&self, f: &Function) -> Self::Result;
    fn name(&self) -> &'static str;
}

pub struct AnalysisManager {
    cached: HashMap<TypeId, (Box<dyn Any + Send>, u64)>,   // (result, generation)
    generation: u64,
}

impl AnalysisManager {
    pub fn get_or_compute<A: Analysis>(&mut self, f: &Function, a: &A) -> &A::Result;
    pub fn invalidate_all_except(&mut self, preserved: &PreservedAnalyses);   // I9
}

pub struct PreservedAnalyses(BTreeSet<TypeId>);
impl PreservedAnalyses {
    pub fn none() -> Self;
    pub fn all() -> Self;
    pub fn preserve<A: Analysis>(self) -> Self;
}
// crates/ssa-passmgr/src/pass.rs — P7: fixed versioned sequence, no runtime DSL;
// concurrency: FunctionPass invocations parallelized via rayon (ADR-08, ADR-12).
pub trait FunctionPass: 'static + Send + Sync {
    fn name(&self) -> &'static str;
    fn run(&self, f: &mut Function, am: &mut AnalysisManager) -> PreservedAnalyses;
}

pub enum OptLevel { O0, O1, O2 }

pub struct Pipeline {
    passes: Vec<Box<dyn FunctionPass>>,
    level: OptLevel,
}

impl Pipeline {
    pub fn locked_for(level: OptLevel) -> Self;   // ADR-10: table, not a string DSL
    pub fn run(&self, m: &mut Module) -> PipelineReport;
}

pub struct PipelineReport {
    pub per_pass_timing: Vec<(&'static str, Duration)>,
    pub per_pass_memory_delta: Vec<(&'static str, isize)>,
}
// crates/ssa-verifier/src/lib.rs — I1–I14; never panics, always returns a Result.
pub enum VerifierError {
    DuplicateDefinition { value: ValueId, sites: [InstructionId; 2] },      // I1
    DominanceViolation { use_site: InstructionId, def: ValueId },           // I2
    PhiPredecessorMismatch { block: BlockId, phi: InstructionId, missing: Vec<BlockId> }, // I3
    TypeMismatch { inst: InstructionId, expected: TypeId, found: TypeId },  // I4
    TerminatorMissing { block: BlockId },                                  // I6
    UnreachableUnmarked { block: BlockId },                                // I7
}

pub fn verify(m: &Module) -> Result<(), Vec<VerifierError>>;
pub fn verify_after_pass(f: &Function, pass_name: &'static str) -> Result<(), Vec<VerifierError>>;
// crates/ssa-outofssa/src/lib.rs — I12: congruence-class integrity;
// ADR-18: Boissinot et al. interference-based destruction.
pub struct CongruenceClass {
    pub representative: ValueId,
    pub members: BTreeSet<ValueId>,
}

pub struct DestructionResult {
    pub classes: Vec<CongruenceClass>,
    pub inserted_copies: u32,          // §4 bound: ≤ 1.15× theoretical minimum
    pub split_critical_edges: u32,
}

pub fn destroy_ssa(f: &mut Function, dom: &DominatorTree) -> DestructionResult;
// crates/ssa-interp/src/lib.rs — the differential-testing oracle (P10).
pub struct InterpState {
    pub regs: HashMap<ValueId, RuntimeValue>,
    pub memory: ByteAddressableHeap,
}

pub enum RuntimeValue { I32(i32), I64(i64), Ptr(u64), Undef }

pub fn interpret(m: &Module, entry_args: &[RuntimeValue]) -> InterpResult;

pub struct InterpResult {
    pub return_value: Option<RuntimeValue>,
    pub trap: Option<TrapKind>,
    pub memory_hash: u64,   // for differential comparison, not a full-heap diff
}
