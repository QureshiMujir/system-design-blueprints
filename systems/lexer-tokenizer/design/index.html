Core schema & types (§11)

#[repr(transparent)]
pub struct ByteOffset(pub u32);                       // ADR-21: u32, 4 GiB cap (I10)
impl ByteOffset {
    pub const ZERO: ByteOffset;
    pub fn as_usize(self) -> usize;
    pub fn checked_add(self, n: u32) -> Option<ByteOffset>;
}

pub struct Span { pub start: ByteOffset, pub end: ByteOffset }
impl Span {
    pub fn new(start: ByteOffset, end: ByteOffset) -> Self;   // requires start <= end
    pub fn len(self) -> u32;
    pub fn is_empty(self) -> bool;
    pub fn as_range(self) -> core::ops::Range<usize>;
    pub fn join(self, other: Span) -> Span;                   // union of two spans
}
pub trait TokenKind: Copy + Eq + core::fmt::Debug + 'static {
    fn is_trivia(self) -> bool;
    fn is_error(self) -> bool;
    fn is_eof(self) -> bool;
    fn name(self) -> &'static str;
}

pub struct Token<K: TokenKind> { pub kind: K, pub span: Span }

pub enum TriviaKind { Whitespace, LineComment, BlockComment, DocComment, Shebang, NewLine }
pub struct TriviaToken { pub kind: TriviaKind, pub span: Span }
pub struct Cursor<'src> { /* bytes: &'src [u8], pos: ByteOffset */ }
impl<'src> Cursor<'src> {
    pub fn new(bytes: &'src [u8]) -> Self;                 // consumes BOM, I10
    pub fn pos(&self) -> ByteOffset;
    pub fn is_eof(&self) -> bool;
    pub fn peek(&self) -> Option<u8>;
    pub fn peek_at(&self, n: usize) -> Option<u8>;
    pub fn bump(&mut self) -> Option<u8>;
    pub fn bump_while<F: Fn(u8) -> bool>(&mut self, f: F) -> u32;
    pub fn slice(&self, span: Span) -> &'src [u8];
}
pub struct FileId(pub u32);
pub struct SourceMap { /* file_id, bytes, has_bom, line_table: OnceLock<Vec<u32>> */ }
impl SourceMap {
    pub fn new(file_id: FileId, bytes: Vec<u8>) -> Self;
    pub fn bytes(&self) -> &[u8];
    pub fn has_bom(&self) -> bool;                          // ADR-08
    pub fn line_col(&self, offset: ByteOffset) -> (u32, u32);  // lazy line-table build, P10
}
pub trait Lexer<'src> {
    type Kind: TokenKind;
    fn next_token(&mut self) -> Option<Token<Self::Kind>>;
    fn pos(&self) -> ByteOffset;
    fn slice(&self, span: Span) -> &'src [u8];
    fn source_len(&self) -> ByteOffset;
}
pub struct ModeId(pub u16);
pub struct ModeStack { /* stack: SmallVec<[ModeId; 8]> */ }
impl ModeStack {
    pub fn new(initial: ModeId) -> Self;
    pub fn current(&self) -> ModeId;
    pub fn push(&mut self, m: ModeId);                  // bounded by §4's 65,535 depth cap
    pub fn pop(&mut self) -> Option<ModeId>;             // refuses to pop the base mode, I5
    pub fn depth(&self) -> usize;
}
pub enum IndentToken { Indent(u32), Dedent(u32), Newline }
pub struct IndentStack { /* levels: SmallVec<[u32; 16]>, pending: SmallVec<[IndentToken; 4]> */ }
impl IndentStack {
    pub fn new() -> Self;
    pub fn feed(&mut self, line_indent: u32);            // I6; emits IndentMismatch on partial dedent
    pub fn drain(&mut self) -> /* iterator of IndentToken */ ();
}
pub enum LexError {
    InvalidUtf8, UnterminatedString, UnterminatedRawString, UnterminatedBlockComment,
    UnterminatedCharLiteral, UnknownToken, InvalidEscape, InvalidNumber, IndentMismatch,
    UnexpectedDedent, InterpolationMismatch, OffsideViolation, LimitExceeded,
}
pub struct EditEnvelope { pub start: ByteOffset, pub old_end: ByteOffset, pub new_end: ByteOffset }

pub enum TokenDelta<K: TokenKind> {
    Replace { start_idx: usize, end_idx: usize, with: Vec<Token<K>> },
    Insert  { at_idx: usize, tokens: Vec<Token<K>> },
    Delete  { start_idx: usize, end_idx: usize },
}

pub struct IncrementalLexer<'src, K: TokenKind> { /* source: &'src [u8], prev: Vec<Token<K>> */ }
impl<'src, K: TokenKind> IncrementalLexer<'src, K> {
    pub fn new(source: &'src [u8], prev: Vec<Token<K>>) -> Self;
    pub fn apply_edit<L: Lexer<'src, Kind = K>>(&mut self, edit: EditEnvelope, lex: L) -> TokenDelta<K>;  // I12
}
