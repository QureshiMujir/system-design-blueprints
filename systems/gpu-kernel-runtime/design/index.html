Core schema & types (§11)

pub struct Register(pub u32);                       // SSA virtual register id

pub enum Address {                                    // §8 addressing
    Global { base: Register, offset: i32 },
    Shared { base: Register, offset: i32 },
    Const  { base: Register, offset: i32 },
    Local  { base: Register, offset: i32 },
}

pub enum Instruction {                                 // ADR-07: SSA-form IR
    Mov(Register, Register),
    Add(Register, Register, Register),
    Sub(Register, Register, Register),
    Mul(Register, Register, Register),
    Load(Register, Address),
    Store(Address, Register),
    SetPredicate(Register, CmpOp, Register, Register),
    Branch { cond: Option<Register>, target: BlockId },
    BarrierSync,
    Return,
}

pub enum CmpOp { Lt, Le, Gt, Ge, Eq, Ne }

pub struct BlockId(pub u32);

pub struct BasicBlock {                                // ADR-07
    pub id: BlockId,
    pub instrs: Vec<Instruction>,
    pub successors: Vec<BlockId>,
}

pub struct KernelMetadata {                             // I4, I5
    pub max_registers: u32,
    pub shared_mem_bytes: u32,
}

pub struct KernelParam { pub name: String, pub space: StateSpace }

pub enum StateSpace { Global, Shared, Const, Local, Register }

pub struct Kernel {
    pub name: String,
    pub params: Vec<KernelParam>,
    pub blocks: Vec<BasicBlock>,
    pub metadata: KernelMetadata,
}

pub struct KernelModule { pub kernels: Vec<Kernel> }
pub struct ThreadState { pub tid: u32, pub regs: Vec<u64>, pub pc: BlockId }

pub struct Warp {                                        // I1, I2
    pub warp_id: u32,
    pub active_mask: u32,                                 // exactly 32 bits (I2)
    pub reconverge_stack: Vec<(u32, BlockId)>,             // (mask, target) — P11
    pub threads: Vec<ThreadState>,                         // len == 32 (I1)
}

pub struct Block { pub block_id: u32, pub warps: Vec<Warp> }

pub struct DevicePtr(pub u64);

pub enum MemoryType { Global, Shared, Constant, Local, Register }

pub struct DeviceAllocation {                              // I6, P7
    pub ptr: DevicePtr,
    pub size: usize,
    pub memory_type: MemoryType,
}

pub struct OccupancyReport {                                // I15
    pub reg_limited_waves: u32,
    pub smem_limited_waves: u32,
    pub slot_limited_waves: u32,
    pub achieved_occupancy: u32,                             // min of the three
}

pub struct LaunchConfig { pub grid_dim: (u32,u32,u32), pub block_dim: (u32,u32,u32) }

pub struct StreamId(pub u32);
pub struct EventId(pub u32);
pub enum DispatchPacket {                                    // ADR-22
    KernelLaunch { kernel: String, config: LaunchConfig, stream: StreamId },
    MemcpyH2D { dst: DevicePtr, size: usize, stream: StreamId },
    MemcpyD2H { dst: DevicePtr, size: usize, stream: StreamId },
    EventRecord { event: EventId, stream: StreamId },
    EventWait { event: EventId, stream: StreamId },
}

pub enum GpuError {                                          // ADR-23: exhaustive
    StateSpaceMismatch,
    InvalidAddress,
    InvalidLaunchConfig,
    RegisterBudgetExceeded,
    SharedMemBudgetExceeded,
    ReconvergenceAmbiguous,
    MaskWidthMismatch,
    StreamOrderViolation,
    QueueCorrupt,
    DmaOverlap,
}

pub struct Metrics {                                          // §24, §28 telemetry
    pub launch_overhead_ns: u64,
    pub warp_instr_per_sec: u64,
    pub occupancy_pct: f32,
    pub coalesced_access_pct: f32,
    pub barrier_timeout_total: u64,
}
