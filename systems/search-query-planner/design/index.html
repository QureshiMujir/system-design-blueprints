Core schema & types (§11)

type DocId = u32;                 // ADR-04; 0..N-1 per segment
const NO_MORE_DOCS: DocId = u32::MAX;   // iterator exhaustion sentinel
type GlobalDocId = u64;           // (shard_id<<48) | (segment_ord<<32) | local DocId
type Generation = u64;            // monotonic searcher generation; cache key (INV-6)
type Score = f32;                 // BM25 output; finite, >= 0 (INV-7)
type FieldId = u16;               // resolved field handle (schema lookup)
type TermBytes = Box<[u8]>;       // analyzed term, UTF-8 NFC
enum Query {
    MatchAll { boost: f32 },
    Term { field: FieldId, term: TermBytes },
    Phrase { field: FieldId, terms: Vec<TermBytes>, slop: u32 },
    Boolean(BooleanQuery),
    Wildcard { field: FieldId, pattern: WildcardPattern },
    Prefix { field: FieldId, prefix: TermBytes },
    Fuzzy { field: FieldId, term: TermBytes, max_edits: u8, prefix_len: u16 },
    Range { field: FieldId, lo: Bound<TermBytes>, hi: Bound<TermBytes> },
    ConstantScore { sub: Box<Query>, boost: f32 },
    Boost { sub: Box<Query>, factor: f32 },
    Synonym { field: FieldId, terms: Vec<TermBytes> },
}
struct BooleanQuery { clauses: Vec<(Occur, Query)>, min_should_match: u32 }
enum Occur { Must, Should, MustNot, Filter }
struct Span { start: u32, end: u32 }   // byte span for diagnostics
trait RewriteRule {
    fn name(&self) -> &'static str;         // "RW-4 not-pushdown"
    fn apply(&self, q: Query, ctx: &RewriteCtx) -> Option<Query>;  // None = no-op
    fn cost_class(&self) -> CostClass;      // Monotone | Neutral (never CostIncreasing)
}
struct RewriteCtx<'a> { stats: &'a dyn StatsProvider, cfg: &'a QueryConfig }
struct RewriteEngine { rules: Vec<Box<dyn RewriteRule>>, max_iters: u8 }
enum LogicalOp {
    Scan { field: FieldId, term: TermBytes, est_card: u64 },
    And  { children: Vec<LogicalOp>, est_card: u64 },
    Or   { children: Vec<LogicalOp>, est_card: u64 },
    AndNot { include: Box<LogicalOp>, exclude: Box<LogicalOp>, est_card: u64 },
    Phrase { field: FieldId, terms: Vec<TermBytes>, slop: u32, est_card: u64 },
    ConstantScore { sub: Box<LogicalOp>, est_card: u64 },
    Filter { sub: Box<LogicalOp>, est_card: u64 },   // no score
}
enum PhysicalOp {
    TermScan { postings: PostingsHandle, scorer: ScorerKind },
    Conjunction { children: Vec<PhysicalOp>, lead_order: Vec<usize> }, // rarest-first
    Disjunction { children: Vec<PhysicalOp>, prune: PruneMode },        // None|Wand|Bmw|MaxScore
    AndNot { include: Box<PhysicalOp>, exclude: Box<PhysicalOp> },
    TwoPhase { approx: Box<PhysicalOp>, confirm: ConfirmKind },         // phrase/expensive
    ConstantScore { iter: Box<PhysicalOp>, score: Score },
    BitsetFilter { bitmap: RoaringHandle },                             // cached filter
}
enum PruneMode { None, Wand, BlockMaxWand, MaxScore }
trait DocIdSetIterator {
    fn doc_id(&self) -> DocId;                 // current; NO_MORE_DOCS if exhausted
    fn next_doc(&mut self) -> DocId;           // advance to next match (INV-2)
    fn advance(&mut self, target: DocId) -> DocId; // first match >= target (INV-2)
    fn cost(&self) -> u64;                      // est. #docs (drives ordering)
}
trait TwoPhaseIterator {
    fn approximation(&self) -> &dyn DocIdSetIterator; // superset (INV-5)
    fn matches(&mut self) -> bool;             // confirm current doc (may remove)
    fn match_cost(&self) -> f32;               // est. confirm cost
}
struct ConjunctionIterator { lead: Box<dyn DocIdSetIterator>,        // rarest (lowest cost)
                             others: Vec<Box<dyn DocIdSetIterator>>, // sorted by cost asc
                             two_phase: Vec<Box<dyn TwoPhaseIterator>> }
struct DisjunctionIterator { pq: BinaryHeap<DisiAtDoc>,              // min-heap by doc_id
                             scratch: Vec<usize> }                   // current matchers
struct AndNotIterator { include: Box<dyn DocIdSetIterator>,
                        exclude: Box<dyn DocIdSetIterator> }
trait Scorer: DocIdSetIterator {
    fn score(&mut self) -> Score;              // score of current doc
    fn max_score(&self) -> Score;              // TRUE upper bound (INV-1)
}
trait BlockMaxScorer: Scorer {
    fn block_max_score(&self) -> Score;        // upper bound within current block (INV-1)
    fn advance_shallow(&mut self, target: DocId);  // move block cursor w/o decoding postings
}
struct Bm25Scorer { postings: PostingsHandle, idf: f32, k1: f32, b: f32,
                    avg_len: f32, norm_decode: [f32; 256], boost: f32, max_score: Score }
struct WandScorer { scorers: Vec<Box<dyn Scorer>>, threshold: Score }       // Phase 9
struct BlockMaxWandScorer { scorers: Vec<Box<dyn BlockMaxScorer>>, threshold: Score } // Phase 10
struct MaxScoreScorer { essential: Vec<..>, non_essential: Vec<..>, threshold: Score } // Phase 10
struct CollectionStatistics { doc_count: u64, sum_total_term_freq: u64,
                              sum_doc_freq: u64, max_doc: DocId, field: FieldId }
struct TermStatistics { term: TermBytes, doc_freq: u64, total_term_freq: u64 }
trait StatsProvider {
    fn collection(&self, field: FieldId) -> CollectionStatistics;
    fn term(&self, field: FieldId, term: &[u8]) -> Option<TermStatistics>;
    fn avg_field_len(&self, field: FieldId) -> f32;
    fn generation(&self) -> Generation;
}
struct CostModel { io_block: f32, decode_doc: f32, score_doc: f32,
                   heap_op: f32, gallop: f32 }          // tunable weights
struct ClauseCost { est_docs: u64, est_score_calls: u64, est_io_blocks: u64, total: f32 }
struct Optimizer<'a> { stats: &'a dyn StatsProvider, model: CostModel, cfg: &'a QueryConfig }
impl Optimizer { fn optimize(&self, logical: LogicalOp) -> PhysicalOp; }      // pure (INV-10)
trait Collector { fn leaf(&self, seg: SegmentOrd) -> Box<dyn LeafCollector>;
                  fn merge(&self, leaves: Vec<LeafTopK>) -> TopK; }
trait LeafCollector {
    fn set_scorer(&mut self, s: &dyn Scorer);
    fn collect(&mut self, doc: DocId);
    fn min_competitive_score(&self) -> Score;   // heap root; feeds pruning (INV-3)
}
struct TopKCollector { k: usize, heap: BinaryHeap<Reverse<ScoreDoc>>, total: TotalHits }
struct ScoreDoc { score: Score, doc: GlobalDocId }    // Ord: score then ascending doc (INV-4)
enum TotalHits { Exact(u64), AtLeast(u64) }           // track_total_hits modes
struct SearchAfter { score: Score, doc: GlobalDocId } // cursor (Phase 12)
struct FilterCache { map: ShardedLru<FilterKey, RoaringHandle>, mem_budget: usize }
struct FilterKey { generation: Generation, segment: SegmentOrd, query_digest: u128 } // INV-6
struct QueryCache { map: ShardedLru<QueryKey, TopK>, mem_budget: usize, enabled: bool }
struct QueryKey { generation: Generation, shard: ShardId, query_digest: u128, k: u32 }
trait SegmentReader {
    fn generation(&self) -> Generation;
    fn max_doc(&self) -> DocId;
    fn live_docs(&self) -> Option<&LiveDocs>;                 // None = no deletes (INV-8)
    fn postings(&self, field: FieldId, term: &[u8], need: PostingsOptions) -> Option<PostingsHandle>;
    fn terms(&self, field: FieldId) -> Option<&dyn TermDictionary>; // FST (ADR-10)
    fn doc_values(&self, field: FieldId) -> Option<DocValues>;
    fn stats(&self) -> &dyn StatsProvider;
    fn norms(&self, field: FieldId) -> Option<&Norms>;
}
trait PostingsHandle {                 // a postings cursor factory
    fn iterator(&self) -> Box<dyn PostingsIterator>;          // docs + freqs
    fn block_max(&self) -> Option<&dyn BlockMaxMeta>;         // for BMW (Phase 10)
    fn impacts(&self) -> Option<&dyn ImpactsMeta>;            // impact-ordered, if present
}
trait PostingsIterator: DocIdSetIterator {
    fn freq(&self) -> u32;                                    // tf in current doc
    fn positions(&mut self) -> Option<&mut dyn PositionIterator>; // for phrases
}
enum PostingsOptions { DocsOnly, DocsFreqs, DocsFreqsPositions }
trait TermDictionary { fn lookup(&self, term: &[u8]) -> Option<TermInfo>;
                       fn enumerate(&self, automaton: &dyn Automaton) -> TermEnum; } // Phase 3
enum LexError  { Unterminated{span:Span}, EmptyField{span:Span}, BadEscape{span:Span} }
enum ParseError{ Unexpected{found:Token,span:Span}, Unbalanced{open:u32}, BoostNoClause{span:Span} }
enum SemError  { LeadingWildcard, UnknownField{name:String}, FuzzyClamped{got:u8} } // some are warnings
enum ResourceError { ExpansionTooLarge{bound:usize}, MemoryBudget{kind:&'static str} }
enum ExecError { DeadlineExceeded, ShardUnavailable{shard:ShardId} }   // -> partial, not fatal
