Core schema & types (§11)

// value.rs — ValType, RefType, Val, Ref

/// Every distinct Wasm value type. V128 and Ref variants gated by feature flags decoded by wrt-decoder.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub enum ValType {
    I32, I64, F32, F64,
    V128,            // Phase 12; 128-bit lane vector
    Ref(RefType),    // funcref | externref
}

/// Reference sub-classification; determines null default and table element type.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub enum RefType { FuncRef, ExternRef }

/// Runtime value. Floats stored as bit-patterns (u32/u64) so canonical NaN payloads and
/// signaling NaN encodings survive round-trips through Val without being normalized by the
/// host FPU. Arithmetic reinterprets to f32/f64, operates, then reinterprets back.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Val {
    I32(i32),
    I64(i64),
    F32(u32),   // IEEE-754 binary32 bit-pattern; NOT f32 to preserve NaN payloads
    F64(u64),   // IEEE-754 binary64 bit-pattern; NOT f64
    V128(u128), // 128-bit opaque; lane interpretation is per-opcode
    Ref(Ref),
}

/// Reference value; null carries type for typed-select and table default.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Ref {
    Null(RefType),
    Func(FuncAddr),    // index into Store::funcs
    Extern(ExternAddr),
}
// types.rs — structural types and index spaces

/// Wasm function signature. Box<[ValType]> over Vec to signal immutability post-construction.
#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct FuncType {
    pub params: Box<[ValType]>,
    pub results: Box<[ValType]>,
}

/// Memory/table size constraints; max = None means unbounded (but spec caps at 2^32-1 pages).
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Limits { pub min: u32, pub max: Option<u32> }

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Mutability { Const, Var }

/// Aggregate extern description; used in import/export type matching (Phase 4).
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ExternType {
    Func(FuncType),
    Table(TableType),
    Mem(MemType),
    Global(GlobalType),
}

/// Opaque index newtypes — zero-cost wrappers around u32; prevent cross-space confusion.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub struct TypeIdx(pub u32);
// Similarly: FuncIdx, TableIdx, MemIdx, GlobalIdx, ElemIdx, DataIdx, LocalIdx, LabelIdx
// all pub struct XIdx(pub u32); — each in its own newtype for exhaustive match safety.
// trap.rs

/// All traps that can terminate Wasm execution; surfaces as Err(Trap) at the host boundary.
/// NOT a Rust panic; must never cross into host code as unwinding.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Trap {
    Unreachable,
    OutOfBoundsMemory,
    OutOfBoundsTable,
    IntegerDivByZero,
    IntegerOverflow,
    InvalidConversionToInt,
    IndirectCallTypeMismatch,
    UndefinedElement,
    UninitializedElement,
    CallStackExhausted,
    MemoryGrowFailed,
}
// error.rs

/// Decode-time errors; carry byte-offset for diagnostics.
#[derive(Debug)]
pub enum DecodeError {
    InvalidMagic, UnsupportedVersion, UnexpectedEof { offset: usize },
    InvalidSection { id: u8 }, MalformedLeb128, InvalidUtf8,
    InvalidOpcode { byte: u8, prefix: Option<u8> }, DataCountMismatch,
    /* ... */
}

/// Validation errors; carry section/function context.
#[derive(Debug)]
pub enum ValidateError {
    TypeMismatch { expected: ValType, got: ValType }, StackUnderflow,
    UnknownFunction(FuncIdx), CtrlStackMismatch, InvalidAlignment,
    /* ... */
}

/// Link/instantiation errors.
#[derive(Debug)]
pub enum LinkError {
    MissingImport { module: String, name: String },
    TypeMismatch { import: ExternType, provided: ExternType },
    MemoryLimitExceeded, TableLimitExceeded,
}

pub enum JitError { UnsupportedOpcode(u8), CodeBufferFull, MmapFailed(i32) }
// module.rs

/// Fully decoded module IR; all indices validated to be in-bounds within the module's own
/// index spaces (section cross-reference), but NOT type-checked. Immutable post-decode.
pub struct Module {
    pub types:      Vec<FuncType>,
    pub imports:    Vec<Import>,
    pub funcs:      Vec<Func>,          // local functions only; imports prepend index space
    pub tables:     Vec<TableType>,
    pub mems:       Vec<MemType>,
    pub globals:    Vec<Global>,
    pub exports:    Vec<Export>,
    pub start:      Option<FuncIdx>,
    pub elems:      Vec<ElemSegment>,
    pub datas:      Vec<DataSegment>,
    pub data_count: Option<u32>,        // present iff bulk-memory used; cross-checked vs datas
    pub customs:    Vec<Custom>,
}

/// One local function: type index + local-variable RLE + decoded instruction sequence.
pub struct Func {
    pub ty:     TypeIdx,
    pub locals: Box<[(u32, ValType)]>, // RLE: (count, type); expand on demand, not eagerly
    pub body:   Box<[Instr]>,
}
// instr.rs (abbreviated; full set locked in canonical spine)

pub enum Instr {
    Unreachable, Nop,
    Block(BlockType), Loop(BlockType), If(BlockType), Else, End,
    Br(LabelIdx), BrIf(LabelIdx), BrTable(Box<[LabelIdx]>, LabelIdx), Return,
    Call(FuncIdx), CallIndirect(TypeIdx, TableIdx),
    Drop, Select(Option<ValType>),
    LocalGet(u32), LocalSet(u32), LocalTee(u32),
    GlobalGet(GlobalIdx), GlobalSet(GlobalIdx),
    TableGet(TableIdx), TableSet(TableIdx),
    Load(MemArg, LoadKind), Store(MemArg, StoreKind),
    MemorySize, MemoryGrow, MemoryFill, MemoryCopy,
    MemoryInit(u32), DataDrop(u32),
    TableInit(u32, TableIdx), ElemDrop(u32),
    TableCopy(TableIdx, TableIdx), TableGrow(TableIdx), TableSize(TableIdx), TableFill(TableIdx),
    RefNull(RefType), RefIsNull, RefFunc(FuncIdx),
    Const(Val),
    IUnOp(IUnOp), IBinOp(IBinOp), FUnOp(FUnOp), FBinOp(FBinOp),
    ITestOp(ITestOp), IRelOp(IRelOp), FRelOp(FRelOp), CvtOp(CvtOp),
    V128(V128Op), // Phase 12; entire variant absent when feature disabled
}

/// Memory immediate: align is log2(bytes); offset is effective-address addend.
/// u64 offset supports future memory64 extension; currently upper 32 bits must be zero.
pub struct MemArg { pub align: u32, pub offset: u64 }
// operand_stack.rs

/// Abstract value on the operand stack during validation.
/// Unknown represents a value of any type: produced after 'unreachable', consumed by any pop.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum OperandType { Known(ValType), Unknown }

pub struct OperandStack(Vec<OperandType>);

// Method signatures — no bodies; implementations in operand_stack.rs
pub fn pop_operand(stack: &mut OperandStack, frame: &CtrlFrame, expected: ValType) -> Result<OperandType, ValidateError>;
pub fn push_operand(stack: &mut OperandStack, t: OperandType);
// ctrl_stack.rs

/// One entry on the control stack; corresponds to one block/loop/if scope.
pub struct CtrlFrame {
    pub opcode:      CtrlOp,             // Block | Loop | If
    pub start_types: Box<[ValType]>,     // types expected at block entry (for multi-value)
    pub end_types:   Box<[ValType]>,     // types produced at block end / branch target
    pub height:      usize,              // operand stack height at block entry
    pub unreachable: bool,               // true after 'unreachable' or unconditional branch
}

/// Label type resolution: loop -> start_types (re-entry); block/if -> end_types.
pub fn label_types(f: &CtrlFrame) -> &[ValType];
// context.rs

/// Module-level validation context; built once, shared across all function validations.
pub struct Context {
    pub types:    Vec<FuncType>,
    pub funcs:    Vec<FuncType>,    // resolved: import funcs first, then local
    pub tables:   Vec<TableType>,
    pub mems:     Vec<MemType>,
    pub globals:  Vec<GlobalType>,
    pub elems:    Vec<RefType>,     // element segment type per segment
    pub datas:    usize,            // count; used for data.drop bounds check
    // Per-function fields (set before each body validation):
    pub locals:   Vec<ValType>,     // expanded from Func::locals RLE
    pub labels:   Vec<Box<[ValType]>>, // control stack label types
    pub return_:  Option<Box<[ValType]>>, // None at module level
    pub refs:     Vec<FuncIdx>,     // declared via declarative element segments
}
// lib.rs

/// Newtype invariant: only wrt-validator::validate() produces a ValidatedModule.
/// wrt-runtime and wrt-jit accept ONLY ValidatedModule, never raw Module.
/// This eliminates all runtime type-error checks from the hot dispatch loop.
pub struct ValidatedModule(pub(crate) Module);

// Signature only — no body:
pub fn module(vm: &ValidatedModule) -> &Module;

// ValidatedModule is Send + Sync (Module is Send + Sync, newtype inherits).
// Serialization: can cache validated IR; deserializer MUST re-run validate() —
// the serialized form carries no "pre-validated" trust.
// store.rs

/// Global mutable state of the Wasm abstract machine; single-threaded ownership.
/// All addresses (FuncAddr, MemAddr, …) are indices into these Vecs.
pub struct Store {
    pub funcs:   Vec<FuncInst>,
    pub tables:  Vec<TableInst>,
    pub mems:    Vec<MemInst>,
    pub globals: Vec<GlobalInst>,
    pub elems:   Vec<ElemInst>,
    pub datas:   Vec<DataInst>,
}
// Store is NOT Send; it holds raw mutable state + host closures (HostFunc may capture !Send).
// Lifetime: typically one per embedding thread; created empty, populated by instantiate().
// mem.rs

/// Linear memory instance. data.len() is always a multiple of PAGE (65_536).
pub struct MemInst { pub data: Vec<u8>, pub max: Option<u32> }
pub const MEM_PAGE: usize = 65_536;

// Signatures only — implementations in mem.rs:
pub fn mem_load_u8(m: &MemInst, addr: u64) -> Result<u8, Trap>;
pub fn mem_store_u8(m: &mut MemInst, addr: u64, v: u8) -> Result<(), Trap>;
// … load/store for u16/u32/u64/u128; all little-endian; addr arithmetic in u64 (no wrap).
pub fn mem_grow(m: &mut MemInst, delta: u32) -> i32; // Returns old size in pages, or -1 on failure.
// instance.rs + store.rs (combined)

pub struct TableInst  { pub elem: Vec<Ref>,     pub ty: TableType }
pub struct GlobalInst { pub value: Val,          pub mutability: Mutability }
pub struct ElemInst   { pub dropped: bool,       pub refs: Vec<Ref> }
pub struct DataInst   { pub dropped: bool,       pub data: Vec<u8> }

pub enum FuncInst {
    Wasm { ty: FuncType, module: ModuleAddr, code: FuncIdx },
    Host { ty: FuncType, host: HostFunc },
}
// HostFunc = Box<dyn Fn(&[Val]) -> Result<Vec<Val>, Trap>>; not Send by default.

pub struct ModuleInst {
    pub types:    Vec<FuncType>,
    pub func_addrs:   Vec<FuncAddr>,
    pub table_addrs:  Vec<TableAddr>,
    pub mem_addrs:    Vec<MemAddr>,
    pub global_addrs: Vec<GlobalAddr>,
    pub elem_addrs:   Vec<ElemAddr>,
    pub data_addrs:   Vec<DataAddr>,
    pub exports:      Vec<Export>,   // resolved to ExternVal (addr) at instantiation
}
// engine.rs

/// Interpreter execution frame: locals + return-arity + owning module reference.
pub struct Frame {
    pub locals: Vec<Val>,
    pub module: ModuleAddr,
    pub arity:  usize,   // number of return values expected by caller
}

/// Branch continuation target on the label stack.
pub struct Label {
    pub arity: usize,    // values to preserve across branch
    pub cont:  usize,    // instruction index of continuation (loop head / block end)
}

/// The interpreter execution engine; owns the operand/frame/label stacks.
pub struct Engine {
    pub(crate) values:  Vec<Val>,     // operand stack
    pub(crate) frames:  Vec<Frame>,   // call stack
    pub(crate) labels:  Vec<Label>,   // label stack
    pub(crate) fuel:    Option<u64>,  // None = unlimited; Some(0) = trap CallStackExhausted
}

/// Host entry point; invokes function at FuncAddr with args; returns results or Trap.
/// Single-threaded; Store must not be aliased across call.
pub fn invoke(store: &mut Store, f: FuncAddr, args: &[Val]) -> Result<Vec<Val>, Trap>;
// codebuf.rs

/// mmap-backed byte buffer. Starts PROT_READ|PROT_WRITE; sealed to PROT_READ|PROT_EXEC.
/// W^X invariant: write and execute permissions are NEVER simultaneously active on a page.
pub struct CodeBuffer { ptr: *mut u8, len: usize, capacity: usize }

// Signatures only — implementations in codebuf.rs:
pub fn codebuf_new(capacity: usize) -> Result<CodeBuffer, JitError>; // mmap(MAP_ANONYMOUS|MAP_PRIVATE)
pub fn codebuf_emit(buf: &mut CodeBuffer, bytes: &[u8]);             // append; panics if full
pub fn codebuf_seal(buf: CodeBuffer) -> Result<SealedBuffer, JitError>; // mprotect RX
pub fn codebuf_offset(buf: &CodeBuffer) -> usize;

pub struct SealedBuffer { ptr: *const u8, len: usize }
// SealedBuffer is Send (read-only after seal); raw fn pointers cast from ptr are callable.
// compiler.rs

/// Single-pass per-function compiler; no IR, no SSA, no optimizing passes.
/// Correctness target: identical results to Engine interpreter for all supported opcodes.
/// Unsupported opcodes trigger deopt (see deopt.rs).
pub struct Compiler<'a> {
    module:  &'a ValidatedModule,
    buf:     CodeBuffer,
    asm:     Assembler,
    // value-stack state tracker (types + locations: reg | rsp-offset)
}

/// Final product: function entry point + byte length of emitted code.
/// entry is callable via SysV ABI: fn(locals_ptr: *mut Val, store_ptr: *mut Store) -> u64
pub struct CompiledFunc {
    pub entry: *const u8,
    pub len:   usize,
}
// CompiledFunc is Send (immutable code page); raw ptr is valid for process lifetime of SealedBuffer.

pub fn compile(m: &ValidatedModule, f: FuncIdx) -> Result<CompiledFunc, JitError>;
// asm_x64.rs — Minimal x86-64 encoder; no external assembler library dependency.
// Signatures only — REX prefix emitted automatically for 64-bit operands.
pub struct Assembler { buf: *mut CodeBuffer }

pub fn asm_mov_rr(a: &mut Assembler, dst: Reg, src: Reg);
pub fn asm_push_r(a: &mut Assembler, r: Reg);
pub fn asm_pop_r(a: &mut Assembler, r: Reg);
pub fn asm_add_rr(a: &mut Assembler, dst: Reg, src: Reg);
pub fn asm_sub_ri32(a: &mut Assembler, dst: Reg, imm: i32);
pub fn asm_imul_rr(a: &mut Assembler, dst: Reg, src: Reg);
pub fn asm_idiv_r(a: &mut Assembler, divisor: Reg);      // uses RAX:RDX implicitly
pub fn asm_cmp_rr(a: &mut Assembler, lhs: Reg, rhs: Reg);
pub fn asm_jcc(a: &mut Assembler, cond: Cond, target: &mut PatchSite);
pub fn asm_jmp(a: &mut Assembler, target: &mut PatchSite);
pub fn asm_call_r(a: &mut Assembler, r: Reg);
pub fn asm_ret(a: &mut Assembler);
pub fn asm_patch(a: &mut Assembler, site: PatchSite);    // backpatch relative offset at site
