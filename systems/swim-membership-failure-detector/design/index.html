struct NodeId
    cluster_epoch: u64
    logical_id: [u8; 16]
struct MemberRecord
    node_id: NodeId
    address: SocketAddr
    incarnation: u64
    state: MemberState
    protocol_version: u16
    labels: LabelSet
    last_update_ts: LogicalTime
    checksum: u32
enum MemberState { Alive, Suspect, Confirmed, Left, Tombstoned }
struct GossipEnvelope
    sequence: u64
    payload: GossipPayload
    ttl: u8
    checksum: u32
struct ProbeContext
    correlation_id: u64
    target: NodeId
    started_at: Instant
    retries: u8

    member: NodeId
    expires_at: Instant
    incarnation: u64
enum RuntimeCommand { InjectPacket, TriggerProbe, ForceSync, Shutdown, ReloadConfig }

    source: SocketAddr
    destination: SocketAddr
    bytes: Bytes
    recv_ts: Instant

    sequence: u64
