Core schema & types (§11)

struct Bytes { ptr: NonNull<u8>, len: usize, cap: usize, align: usize } // 64-byte aligned, owns alloc
struct Buffer { data: Arc<Bytes>, offset: usize, len: usize }           // zero-copy view
impl Buffer { fn as_slice(&self)->&[u8]; fn slice(&self, off:usize,len:usize)->Buffer; fn typed<T:Pod>(&self)->&[T]; }
struct Bitmap { buffer: Buffer, len: usize }                            // validity, LSB0
impl Bitmap { fn get(&self,i:usize)->bool; fn count_set(&self)->usize; fn slice(&self,off,len)->Bitmap; fn and(&self,&Bitmap)->Bitmap; }
struct ArrayData { dtype: DataType, len: usize, null_count: usize, offset: usize,
                   buffers: Vec<Buffer>, child_data: Vec<ArrayData>, dict: Option<Arc<ArrayData>> }
impl ArrayData { fn try_new(...) -> Result<Self, EngineError>; fn validate(&self) -> Result<(),EngineError>; }

trait Array: Send + Sync {
  fn len(&self) -> usize;
  fn null_count(&self) -> usize;
  fn is_null(&self, i: usize) -> bool;
  fn data_type(&self) -> &DataType;
  fn slice(&self, offset: usize, len: usize) -> ArrayRef;  // zero-copy
  fn as_any(&self) -> &dyn Any;                            // downcast to concrete
}
type ArrayRef = Arc<dyn Array>;
struct PrimitiveArray<T: ArrowPrimitive> { data: ArrayData, values: Buffer, validity: Option<Bitmap> }
struct BooleanArray { data: ArrayData, values: Bitmap, validity: Option<Bitmap> }   // values packed 1 bit
struct Utf8Array<O: Offset> { data: ArrayData, offsets: Buffer /*[O]*/, values: Buffer, validity: Option<Bitmap> }
struct BinaryArray<O: Offset> { ... }
struct ListArray<O: Offset> { data: ArrayData, offsets: Buffer, child: ArrayRef, validity: Option<Bitmap> }
struct StructArray { data: ArrayData, children: Vec<ArrayRef>, validity: Option<Bitmap> }
struct DictionaryArray<K: ArrowDictKey> { keys: PrimitiveArray<K>, values: ArrayRef }
struct NullArray { len: usize }
enum DataType { Null, Boolean,
  Int8, Int16, Int32, Int64, UInt8, UInt16, UInt32, UInt64, Float32, Float64,
  Decimal128(u8 /*precision*/, i8 /*scale*/),
  Date32, Date64, Time32(TimeUnit), Time64(TimeUnit), Timestamp(TimeUnit, Option<String> /*tz*/),
  Duration(TimeUnit), Interval(IntervalUnit),
  Utf8, LargeUtf8, Binary, LargeBinary,
  List(Box<Field>), LargeList(Box<Field>), FixedSizeList(Box<Field>, i32),
  Struct(Vec<Field>), Map(Box<Field>, bool), Dictionary(Box<DataType>/*key*/, Box<DataType>/*value*/) }
struct Field { name: String, data_type: DataType, nullable: bool, metadata: BTreeMap<String,String> }
struct Schema { fields: Vec<Field>, metadata: BTreeMap<String,String> }
fn coerce(lhs: &DataType, rhs: &DataType) -> Result<DataType, EngineError>; // promotion lattice
struct ChunkedArray { dtype: DataType, chunks: Vec<ArrayRef>, len: usize, null_count: usize }
struct Series { name: String, array: ChunkedArray }
struct DataFrame { columns: Vec<Series>, schema: Arc<Schema> }   // all columns equal length (I7)
struct RecordBatch { schema: Arc<Schema>, columns: Vec<ArrayRef>, num_rows: usize } // single-chunk batch
enum Expr { Column(String), Literal(Scalar),
  Binary{left: Box<Expr>, op: BinOp, right: Box<Expr>},
  Unary{op: UnOp, expr: Box<Expr>},
  Function{name: String, args: Vec<Expr>},
  Agg{func: AggFunc, expr: Box<Expr>},
  Window{func: AggFunc, expr: Box<Expr>, partition: Vec<Expr>},
  When{branches: Vec<(Expr,Expr)>, otherwise: Box<Expr>},
  Cast{expr: Box<Expr>, dtype: DataType}, Alias(Box<Expr>, String) }

enum LogicalPlan {
  Scan{source: Source, projection: Option<Vec<usize>>, predicate: Option<Expr>, schema: Arc<Schema>},
  Projection{exprs: Vec<Expr>, input: Box<LogicalPlan>},
  Filter{predicate: Expr, input: Box<LogicalPlan>},
  Aggregate{keys: Vec<Expr>, aggs: Vec<Expr>, input: Box<LogicalPlan>},
  Join{left: Box<LogicalPlan>, right: Box<LogicalPlan>, on: Vec<(Expr,Expr)>, how: JoinType},
  Sort{keys: Vec<SortKey>, input: Box<LogicalPlan>},
  Limit{offset: usize, n: Option<usize>, input: Box<LogicalPlan>},
  Distinct{subset: Option<Vec<String>>, input: Box<LogicalPlan>},
  Union{inputs: Vec<LogicalPlan>} }
impl LogicalPlan { fn schema(&self) -> Arc<Schema>; fn children(&self) -> Vec<&LogicalPlan>; }

trait ExecNode: Send + Sync {
  fn schema(&self) -> Arc<Schema>;
  fn execute(&self, ctx: &TaskContext) -> RecordBatchStream;   // streaming, morsel-driven
  fn children(&self) -> Vec<Arc<dyn ExecNode>>;
}
type RecordBatchStream = Box<dyn Iterator<Item = Result<RecordBatch, EngineError>> + Send>;
trait OptimizerRule { fn name(&self)->&str; fn rewrite(&self, p: LogicalPlan) -> Result<LogicalPlan, EngineError>; }
struct Morsel { batch: RecordBatch, partition: usize }
struct SortKey { expr: Expr, descending: bool, nulls_first: bool }
struct ColumnStatistics { min: Option<Scalar>, max: Option<Scalar>, null_count: Option<usize>, distinct_count: Option<usize> }
struct HashTable { /* open-addressing, ahash; keys: row-encoded; payload: agg state or row ptr */ }
struct MemoryPool { budget: usize, used: AtomicUsize, spill_threshold: usize }
struct Reservation<'p> { pool: &'p MemoryPool, bytes: usize }   // RAII; releases on drop
impl MemoryPool { fn try_reserve(&self, bytes: usize) -> Result<Reservation, EngineError /*OutOfMemory*/>; }
struct ParquetFileMetaData { schema: Arc<Schema>, row_groups: Vec<RowGroupMeta>, num_rows: i64, kv: Map<String,String> }
struct RowGroupMeta { columns: Vec<ColumnChunkMeta>, num_rows: i64, total_byte_size: i64 }
struct ColumnChunkMeta { path: String, codec: Compression, encodings: Vec<Encoding>, stats: ColumnStatistics, data_page_offset: i64, dictionary_page_offset: Option<i64> }
struct PageHeader { kind: PageKind, uncompressed_size: i32, compressed_size: i32, num_values: i32, encoding: Encoding, def_level_enc: Encoding, rep_level_enc: Encoding }
#[repr(C)] struct ArrowSchema { format: *const c_char, name: *const c_char, /* … */ release: Option<extern "C" fn(*mut ArrowSchema)>, private_data: *mut c_void }
#[repr(C)] struct ArrowArray { length: i64, null_count: i64, offset: i64, n_buffers: i64, buffers: *mut *const c_void, /* … */ release: Option<extern "C" fn(*mut ArrowArray)>, private_data: *mut c_void }
enum EngineError {
  Schema(String),         // length/type/name mismatch, invalid array data (I1,I3,I6,I7,I10)
  TypeCoercion(String),   // no valid promotion (I9)
  Overflow(String),       // integer agg / offset overflow (§4)
  NotFound(String),       // column/field missing
  Io(std::io::Error),     // file/socket errors
  Parquet(String),        // malformed metadata/page/encoding (I5, Threat Model)
  Ipc(String),            // malformed Flatbuffers/message
  Csv(String),            // parse/quoting/type-infer error
  OutOfMemory{requested:usize, budget:usize}, // pool exhausted, spill failed
  Unsupported(String),    // PhaseDeferred surfaces
  Internal(String),       // invariant violated (bug)
}
