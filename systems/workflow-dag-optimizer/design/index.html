Core schema & types (§11)

NodeId(u32); ArtifactId(u32); PoolId(u32); OpId(u32); Sym(u32); RuleId(u16)
Fp([u8;32])                       // structural fingerprint / content address
Span { lo: u32, hi: u32, file: FileId }
Dag { nodes: Vec<Node>, edges: Vec<Edge>, artifacts: Vec<Artifact>,
      pools: Vec<Pool>, params: ParamMap, roots: Vec<NodeId>, sinks: Vec<NodeId> }

ValidatedDag(Dag)                  // newtype; only constructible via validate()
CanonicalDag { dag: ValidatedDag, order: Vec<NodeId>, fps: Vec<Fp> }

Node { id, kind: NodeKind, op: OpId, params: ParamMap,
       in_ports: SmallVec<[Port;2]>, out_ports: SmallVec<[Port;2]>,
       pool: Option<PoolId>, pure: bool, span: Span, fp: Fp }

NodeKind = Source | Task | Map { over: ArtifactId } | Reduce { init: Sym }
         | Barrier | Sink | Const { value: ConstVal }
Port { name: Sym, ty: Type, idx: u16, required: bool }
Edge { from: PortRef, to: PortRef, artifact: ArtifactId }
PortRef { node: NodeId, port: u16 }
Artifact { id, ty: Type, producer: PortRef, consumers: SmallVec<[PortRef;4]>,
           size_est: u64, digest: Option<Digest>, live: bool }
Type = Int | Float | Bool | Str | Bytes | Array(Box<Type>) | Record(Vec<(Sym,Type)>)
ConstVal = ... (matches Type)
Rule { id: RuleId, name: Sym, pattern: Pattern, guard: GuardFn,
       build: BuildFn, proof: ProofTag, enabled: bool }
Pattern = Op(OpId, Vec<Pattern>) | Var(VarId) | AnyPure | AnyConst
GuardFn  = fn(&Bindings, &CanonicalDag) -> bool
BuildFn  = fn(&Bindings, &mut DagBuilder) -> NodeId
Bindings = Map<VarId, NodeId>
ProofTag = Algebraic | Identity | DeadCode | Dedup | Fold | Pushdown
RuleRegistry { rules: Vec<Rule>, by_root_op: Map<OpId, Vec<RuleId>> }
Cost { cpu_secs: f64, wall_secs: f64, mem_bytes: u64, io_bytes: u64 }
trait StatsProvider {
  fn task_runtime(&self, op: OpId, params: &ParamMap, in_sizes: &[u64]) -> f64;
  fn task_mem(&self, op: OpId, params: &ParamMap, in_sizes: &[u64]) -> u64;
  fn task_io(&self, op: OpId, params: &ParamMap, in_sizes: &[u64]) -> u64;
  fn fanout(&self, node: NodeId, dag: &CanonicalDag) -> u64;
  fn selectivity(&self, op: OpId, params: &ParamMap) -> f64;
}
CostModel { stats: Arc<dyn StatsProvider>, weights: Weights }
Weights { w_cpu: f64, w_wall: f64, w_mem: f64, w_io: f64 }  // objective coeffs
fn cost_of(&self, dag: &CanonicalDag, node: NodeId) -> Cost   // memoized by fp
fn objective(&self, c: &Cost) -> f64
EClassId(u32)
ENode { op: OpId, children: SmallVec<[EClassId;2]>, fp: Fp }
EGraph { classes: Vec<EClass>, memo: Map<ENode, EClassId>, union_find: UF }
EClass { id, nodes: Vec<ENode>, best: Option<(ENode, Cost)> }  // best = extracted
Plan = Arc<CanonicalDag>            // IR3 after extraction
SearchConfig { mode: Greedy|Dp|Bnb|Saturate, node_budget: u64,
               time_budget: Duration, mem_budget: u64, seed: u64 }
SearchResult { plan: Plan, cost: Cost, suboptimal: bool, stats: SearchStats }
MemoTable { entries: DashMap<Fp, (Plan, Cost)> }   // DP memoization
Pool { id: PoolId, cpu: u32, mem: u64, io: u32, concurrency: u32 }
ResourcePool runtime state: { remaining: AtomicCapacity, running: Vec<NodeId> }
Schedule { entries: Vec<ScheduleEntry>, makespan: f64 }
ScheduleEntry { node: NodeId, pool: PoolId, start: f64, end: f64,
                cpu: u32, mem: u64 }
ScheduledDag { plan: Plan, schedule: Schedule }     // IR4
ExecPlan { version: u32, tasks: Vec<ExecTask>, schedule: Schedule,
           fp: Fp, stats_snapshot_ref: Fp }
ExecTask { id: NodeId, op: OpId, params: ParamMap, in_digests: Vec<Fp>,
           out_artifacts: Vec<ArtifactId>, pool: PoolId, fingerprint: Fp }
trait Executor {
  fn submit(&self, task: &ExecTask, token: Token) -> Result<TaskHandle, ExecErr>;
  fn poll(&self, h: TaskHandle) -> TaskState;          // Running|Done(Fp)|Failed(e)
  fn ack(&self, h: TaskHandle, digest: Fp);
}
Token(u32)                          // backpressure credit
TokenPool { budget: u32, outstanding: AtomicU32 }
enum ParseError   { Lex{span,msg}, Unexpected{span,found,expected}, Eof }
enum ValidationError { Cycle{path:Vec<NodeId>}, TypeMismatch{edge:EdgeIdx,a:Type,b:Type},
                       DuplicateProducer{artifact:ArtifactId}, UnboundPort{port:PortRef},
                       Oversubscribed{pool:PoolId}, NoSink }
enum CanonError   { NonDeterministic, FingerprintCollision{a:NodeId,b:NodeId} }
enum SearchError  { BudgetExceeded{kind}, NoEquivalent, ExtractFailed }
enum SchedError   { PrecedenceViolation{a:NodeId,b:NodeId}, CapacityViolation{pool:PoolId},
                    Infeasible }
enum PlanError    { RoundtripMismatch, VersionUnsupported{got:u32,max:u32}, Serialize }
enum CacheError   { Corrupt{digest:Fp}, KeyResultSkew{key:Fp}, Io }
enum ReplayError  { TornTail{segment}, Divergence{expected:Fp,got:Fp}, RuleSetMismatch }
enum OptError     { Parse(ParseError), Validate(ValidationError), Canon(CanonError),
                    Search(SearchError), Sched(SchedError), Plan(PlanError) }
