Core schema & types (§11)

pub struct Label { pub repo: RepoName, pub package: PackagePath, pub name: TargetName }
pub struct RepoName(Arc<str>);
pub struct PackagePath(Arc<str>);      // forward-slash separated, no leading '/'
pub struct TargetName(Arc<str>);       // [A-Za-z0-9_./+-]+
impl Label {
    pub fn parse(s: &str) -> Result<Self, LabelError>;
    pub fn canonical(&self) -> String;
}

pub struct Digest { pub function: DigestFunction, pub size: u64, pub hash: [u8; 32] }
#[repr(u8)]
pub enum DigestFunction { Blake3_256 = 1 }   // tag reserved for future widening (ADR-07)
impl Digest {
    pub fn of_bytes(bytes: &[u8]) -> Self;
    pub fn of_file(path: &Path) -> io::Result<Self>;
    pub fn to_hex(&self) -> String;
}
pub struct Artifact { pub owner: Label, pub path: ArtifactPath, pub kind: ArtifactKind }
pub enum ArtifactKind { Source, Generated, Tree, Symlink { target: ArtifactPath } }
pub struct ArtifactPath(Arc<str>);

pub struct Action {
    pub mnemonic: Arc<str>, pub command: CommandLine, pub env: BTreeMap<String, String>,
    pub inputs: ArtifactSet, pub tools: ArtifactSet, pub outputs: Vec<Artifact>,
    pub platform: PlatformId, pub timeout: Duration,
    pub execution_requirements: BTreeMap<String, String>, pub owner: Label,
}
pub struct CommandLine { pub argv: Vec<String>, pub working_dir: String }
pub struct ArtifactSet { /* sorted, de-duplicated; see §8.2 */ }

pub struct ActionKey(pub Digest);
impl ActionKey {
    pub fn derive(action: &Action, salt: Salt, cas: &dyn CasReader)
        -> Result<Self, ActionKeyError>;   // procedure in §8.2
}
pub struct Salt(pub u32);   // bumped only via ADR-04
pub struct ActionResult {
    pub exit_code: i32, pub stdout_digest: Digest, pub stderr_digest: Digest,
    pub output_files: Vec<OutputFile>, pub output_directories: Vec<OutputDirectory>,
    pub execution_metadata: ExecutionMetadata, pub result_digest: Digest,  // ADR-14
}
pub struct OutputFile { pub path: ArtifactPath, pub digest: Digest, pub is_executable: bool }
pub struct OutputDirectory { pub path: ArtifactPath, pub tree_digest: Digest }
pub struct ExecutionMetadata {
    pub worker: Arc<str>, pub queued_at: SystemTime, pub started_at: SystemTime,
    pub completed_at: SystemTime,
}

pub struct CacheKey { pub action_key: ActionKey, pub salt: Salt }
pub struct CacheEntry { pub result: ActionResult, pub stored_at: SystemTime, pub origin: CacheOrigin }
pub enum CacheOrigin { Local, Remote { url: Arc<str>, fetched_at: SystemTime } }

#[async_trait::async_trait]
pub trait ActionCache: Send + Sync {
    async fn lookup(&self, key: ActionKey) -> Result<Option<ActionResult>, CacheError>;
    async fn update(&self, key: ActionKey, result: ActionResult) -> Result<(), CacheError>;
}
pub trait CasStore: Send + Sync {
    async fn has(&self, digest: Digest) -> Result<bool, CasError>;
    async fn get(&self, digest: Digest) -> Result<Bytes, CasError>;
    async fn put(&self, bytes: Bytes) -> Result<Digest, CasError>;
    async fn find_missing(&self, digests: &[Digest]) -> Result<Vec<Digest>, CasError>;
}
pub struct Tree { pub files: Vec<FileNode>, pub directories: Vec<DirectoryNode>, pub symlinks: Vec<SymlinkNode> }
pub struct FileNode { pub name: Arc<str>, pub digest: Digest, pub is_executable: bool }
pub struct DirectoryNode { pub name: Arc<str>, pub digest: Digest }
pub struct SymlinkNode { pub name: Arc<str>, pub target: Arc<str> }
pub struct Provider { pub id: ProviderId, pub fields: BTreeMap<Arc<str>, AttrValue> }
pub struct ProviderId(pub u32);
pub struct DefaultInfo { pub files: DepSet<Artifact>, pub runfiles: Runfiles, pub executable: Option<Artifact> }

pub struct ConfiguredTarget {
    pub label: Label, pub configuration: ConfigurationId,
    pub providers: BTreeMap<ProviderId, Provider>, pub actions: Vec<ActionKey>,
}
pub struct ConfigurationId(pub Digest);   // content hash of Configuration (I12)
pub struct Configuration {
    pub platform: PlatformId, pub cpu: Arc<str>, pub os: Arc<str>,
    pub compilation_mode: CompilationMode, pub flags: BTreeMap<Arc<str>, AttrValue>,
}
pub enum CompilationMode { Fastbuild, Opt, Dbg }
pub trait Transition: Send + Sync { fn apply(&self, parent: &Configuration) -> Configuration; }

pub struct DepSet<T> { direct: Arc<[T]>, transitive: Arc<[DepSet<T>]>, order: DepSetOrder }
impl<T: Eq + Hash + Clone> DepSet<T> {
    pub fn from(items: Vec<T>, transitive: Vec<DepSet<T>>) -> Self;
    pub fn to_list(&self) -> Vec<T>;   // O(n log n) flatten + dedupe, lazy until called
}
pub enum SkyKey {
    File { path: ArtifactPath }, Package { label_root: Label }, Target { label: Label },
    ConfiguredTarget { label: Label, configuration: ConfigurationId }, Action(ActionKey),
    Artifact { artifact: Artifact, configuration: ConfigurationId }, Glob { package: Label, pattern: Arc<str> },
}
pub struct SkyNode { pub key: SkyKey, pub value: Option<SkyValue>, pub deps: SmallVec<[SkyKey; 8]>, pub state: SkyState, pub version: u64 }
pub enum SkyState { Clean, Dirty, NeedsRebuild, InProgress, Failed }
pub trait SkyFunction: Send + Sync { fn compute(&self, key: &SkyKey, env: &mut SkyEnv<'_>) -> SkyComputation; }
pub enum SkyComputation { Done(SkyValue), Restart(Vec<SkyKey>), Failed(SkyError) }

#[async_trait::async_trait]
pub trait ExecutionEngine: Send + Sync {
    async fn execute(&self, action: &Action, inputs: &MaterialisedTree) -> Result<ActionResult, ExecError>;
}
