Core schema & types (§11)

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
pub struct ByteOffset(pub u64);

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
pub struct CharOffset(pub u64);

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
pub struct Utf16Offset(pub u64);

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
pub struct GraphemeOffset(pub u64);

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
pub struct LineIdx(pub u64);

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct LineCol { pub line: LineIdx, pub col: ByteOffset }

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct ByteRange { pub start: ByteOffset, pub end: ByteOffset }
pub trait TextBuffer: Send + Sync {
    fn len_bytes(&self) -> ByteOffset;
    fn len_chars(&self) -> CharOffset;
    fn len_utf16(&self) -> Utf16Offset;
    fn line_count(&self) -> LineIdx;

    fn insert(&mut self, at: ByteOffset, s: &str) -> Result<EditId, EditError>;
    fn delete(&mut self, range: ByteRange) -> Result<EditId, EditError>;
    fn replace(&mut self, range: ByteRange, s: &str) -> Result<EditId, EditError>;

    fn slice(&self, range: ByteRange) -> Cow<'_, str>;
    fn iter_chunks(&self, range: ByteRange) -> ChunkIter<'_>;
    fn iter_chars(&self, range: ByteRange) -> CharIter<'_>;
    fn iter_graphemes(&self, range: ByteRange) -> GraphemeIter<'_>;
    fn iter_lines(&self, range: ByteRange) -> LineIter<'_>;

    fn byte_to_line_col(&self, off: ByteOffset) -> LineCol;
    fn line_col_to_byte(&self, lc: LineCol) -> ByteOffset;
    fn byte_to_utf16(&self, off: ByteOffset) -> Utf16Offset;
    fn utf16_to_byte(&self, off: Utf16Offset) -> ByteOffset;

    fn snapshot(&self) -> Box<dyn BufferSnapshot>;
    fn subscribe(&mut self, sub: Box<dyn EditObserver>) -> SubscriptionId;
    fn unsubscribe(&mut self, id: SubscriptionId);
}
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EditOp {
    Insert { at: ByteOffset, content: SmolStr },
    Delete { range: ByteRange, removed: SmolStr },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, PartialOrd, Ord)]
pub struct EditId(pub u64);

#[derive(Clone, Debug)]
pub struct EditGroup {
    pub id: EditId,
    pub ops: Vec<EditOp>,
    pub inverse: Vec<EditOp>,
    pub timestamp: Instant,
    pub coalesce_key: CoalesceKey,
}

#[derive(Clone, Debug)]
pub struct EditEvent {
    pub id: EditId,
    pub op: EditOp,
    pub before_len: ByteOffset,
    pub after_len: ByteOffset,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
pub struct CoalesceKey { pub line: LineIdx, pub direction: EditDirection }
#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
pub struct AnchorId(pub u64);

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Stickiness {
    StickyLeft,
    StickyRight,
    Surround { stick_start: Box<Stickiness>, stick_end: Box<Stickiness> },
}

#[derive(Clone, Debug)]
pub struct Anchor {
    pub id: AnchorId,
    pub offset: ByteOffset,
    pub stickiness: Stickiness,
    pub label: Option<SmolStr>,
}

pub struct AnchorStore {
    by_offset: BTreeMap<ByteOffset, SmallVec<[AnchorId; 2]>>,
    anchors: HashMap<AnchorId, Anchor>,
    next_id: u64,
}
pub trait EditObserver: Send + Sync {
    fn name(&self) -> &str;
    fn delivery(&self) -> Delivery;
    fn on_edit(&self, ev: &EditEvent) -> Result<(), ObserverError>;
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Delivery { Sync, Async { queue_depth: usize } }

#[derive(Clone, Debug)]
pub enum ObserverError { Backpressure, Refused(SmolStr), Fatal(SmolStr) }

pub struct ObserverFanout {
    syncs: Vec<Arc<dyn EditObserver>>,
    asyncs: Vec<(SubscriptionId, Sender<EditEvent>)>,
    next_id: u64,
}
#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Revision(pub u64);

pub trait BufferSnapshot: Send + Sync {
    fn revision(&self) -> Revision;
    fn len_bytes(&self) -> ByteOffset;
    fn slice(&self, range: ByteRange) -> Cow<'_, str>;
    fn iter_chunks(&self, range: ByteRange) -> ChunkIter<'_>;
    fn byte_to_line_col(&self, off: ByteOffset) -> LineCol;
}
pub struct LineIndex { root: Node, total_bytes: u64, total_lines: u64 }

enum Node {
    Internal { children: SmallVec<[(u64, u64, Box<Node>); 8]> }, // (bytes, lines, child)
    Leaf { lf_positions: SmallVec<[u32; 64]>, chunk_bytes: u32 },
}

impl LineIndex {
    pub fn line_to_byte(&self, line: LineIdx) -> ByteOffset;
    pub fn byte_to_line_col(&self, off: ByteOffset) -> LineCol;
    pub fn apply_edit(&mut self, op: &EditOp);
    pub fn validate(&self, buf: &dyn TextBuffer) -> Result<(), DriftError>;
}
#[derive(Clone, Debug, PartialEq, Eq)]
#[non_exhaustive]
pub enum EditError {
    Utf8Boundary { offset: ByteOffset },
    EmptyEdit,
    TooLarge { requested: u64, cap: u64 },
    ObserverRefused(SmolStr),
    OutOfBounds { offset: ByteOffset, len: ByteOffset },
}

#[derive(Clone, Debug, PartialEq, Eq)]
#[non_exhaustive]
pub enum PositionError {
    OutOfBounds,
    NotACodepointBoundary,
    NotAGraphemeBoundary,
}

#[derive(Clone, Debug)]
#[non_exhaustive]
pub enum IoError {
    DecodeError { at_byte: u64, encoding: SmolStr },
    PartialWrite { written: u64, expected: u64 },
    MmapSignal,
    Filesystem(SmolStr),
}

#[derive(Clone, Debug)]
pub struct DriftError { pub at_line: LineIdx, pub expected_byte: ByteOffset, pub found_byte: ByteOffset }

#[derive(Clone, Debug)]
#[non_exhaustive]
pub enum UndoError { Empty, NothingToRedo, Corrupted(SmolStr) }

#[derive(Clone, Debug)]
pub enum FanoutError { AllSubscribersFatal }
