Core schema & types (§11)

pub struct BackendId(pub NonZeroU64);           // Copy, Eq, Hash, Debug
pub struct ClusterId(pub NonZeroU64);           // Copy, Eq, Hash, Debug
pub struct RouteId(pub NonZeroU64);             // Copy, Eq, Hash, Debug
pub struct RequestId(pub NonZeroU64);           // Copy, Eq, Hash, Debug

pub struct IdAllocator { next: AtomicU64 }
impl IdAllocator {
    pub fn next_request(&self) -> RequestId;    // monotonic, fetch_add(Relaxed)
}
pub struct Backend {
    pub id: BackendId, pub cluster: ClusterId, pub addr: SocketAddr,
    pub weight: u32, pub locality: Locality,
    pub tls: Option<TlsClientConfig>,
    pub status: BackendStatus, pub stats: BackendStats,
}

pub struct BackendStatus {
    pub healthy: AtomicBool, pub draining: AtomicBool,
    pub last_health_change_micros: AtomicU64,
    pub consecutive_5xx: AtomicU32, pub ejection_until_micros: AtomicU64,
}

pub struct BackendStats {
    pub active_requests: AtomicU32, pub total_requests: AtomicU64,
    pub total_5xx: AtomicU64, pub ewma_latency_micros: AtomicU64,
    pub last_pick_micros: AtomicU64,
}

pub struct Locality { pub region: &'static str, pub zone: &'static str }  // Copy, Eq

pub struct TlsClientConfig {
    pub server_name: String, pub root_cas: rustls::RootCertStore,
    pub client_auth: Option<ClientAuth>,
}
pub struct BackendPool { cluster: ClusterId, backends: ArcSwap<Vec<Arc<Backend>>> }

impl BackendPool {
    pub fn new(cluster: ClusterId, initial: Vec<Arc<Backend>>) -> Self;
    pub fn snapshot(&self) -> Arc<Vec<Arc<Backend>>>;      // single atomic load
    pub fn replace(&self, new_backends: Vec<Arc<Backend>>); // I8 caller validates non-empty delta
    pub fn cluster(&self) -> ClusterId;
}
pub struct Pick { pub backend: Arc<Backend>, pub algorithm: &'static str }

pub struct PickContext<'a> {
    pub request_id: RequestId, pub hash_key: Option<&'a [u8]>, pub now_micros: u64,
}

pub enum PickError { NoHealthy, AllEjected, Internal(&'static str) }

pub trait Selector: Send + Sync + 'static {
    fn pick(&self, pool: &BackendPool, ctx: PickContext<'_>) -> Result<Pick, PickError>;
    fn name(&self) -> &'static str;
}
pub struct ConfigSnapshot {
    pub generation: u64,
    pub clusters: HashMap<ClusterId, Arc<ClusterConfig>>,
    pub routes: HashMap<RouteId, Arc<RouteConfig>>,
    pub listeners: Vec<Arc<ListenerConfig>>,
    pub admin: Arc<AdminConfig>, pub observe: Arc<ObserveConfig>,
}

pub struct ClusterConfig {
    pub id: ClusterId, pub name: String, pub algorithm: AlgorithmConfig,
    pub discovery: DiscoveryConfig, pub health: HealthConfig,
    pub circuit: CircuitConfig, pub upstream_tls: Option<UpstreamTlsConfig>,
    pub connection: ConnectionConfig,
}

pub struct RouteConfig {
    pub id: RouteId, pub matcher: Matcher, pub action: RouteAction,
    pub rate_limit: Option<RateLimitConfig>, pub auth: Option<AuthConfig>,
    pub canary: Option<CanaryConfig>, pub split: Option<SplitConfig>,
    pub shadow: Option<ShadowConfig>,
}

pub enum RouteAction {
    Proxy(ClusterId), Redirect { code: u16, to: String },
    Direct { code: u16, body: bytes::Bytes },
}

pub struct ListenerConfig { pub bind: SocketAddr, pub protocol: ListenerProtocol, pub tls: Option<ServerTlsConfig> }
pub enum ListenerProtocol { Tcp, Udp, Http1, Http2, Http3 }
pub enum CheckOutcome { Healthy, Unhealthy { reason: &'static str }, Inconclusive }

pub trait Checker: Send + Sync + 'static {
    async fn probe(&self, backend: Arc<Backend>) -> CheckOutcome;
    fn interval(&self) -> Duration; fn timeout(&self) -> Duration;
    fn unhealthy_threshold(&self) -> u32; fn healthy_threshold(&self) -> u32;
    fn name(&self) -> &'static str;
}
pub trait Discoverer: Send + Sync + 'static {
    fn cluster(&self) -> ClusterId; fn name(&self) -> &'static str;
    async fn watch(self: Arc<Self>) -> watch::Receiver<Vec<Arc<Backend>>>;
}
pub struct Router { snapshot: ArcSwap<RouterSnapshot> }
pub struct RouterSnapshot { pub trie: PathTrie, pub host_index: HashMap<String, RouteSet>, pub fallback: Option<RouteId> }
pub enum RouterDecision { Matched(Arc<RouteConfig>), NotFound }
impl Router { pub fn route(&self, req: &HttpRequestHead) -> RouterDecision; }
pub trait RateLimiter: Send + Sync + 'static {
    fn check(&self, key: &[u8], cost: u32) -> RateDecision;   // O(1), I14
}
pub enum RateDecision { Allow, Deny { retry_after: Duration } }

pub enum BreakerState { Closed, Open, HalfOpen }
pub trait CircuitBreaker: Send + Sync + 'static {
    fn state(&self) -> BreakerState;
    fn record_outcome(&self, ok: bool);                        // I12
}
