Core schema & types (§11)

struct Module {
    class_file: ClassFile,               // [untrusted] raw decoded structure
}
// ownership: owned by the verification call; immutable after Phase 0 parse; no persistence.

struct ClassFile {
    magic: u32,                          // [untrusted]
    version_major: u16,                  // [untrusted]
    version_minor: u16,                  // [untrusted]
    constant_pool: ConstantPool,         // [untrusted] entries
    methods: Vec<Method>,                // [untrusted] count and bodies
}
// ownership: single-owner, immutable post-parse; no cross-thread mutation; not serialized out.

struct ConstantPool {
    entries: Vec<ConstantPoolEntry>,     // [untrusted]
}

struct ConstantPoolIndex(u16);           // [untrusted] — resolved only via bounds+tag check (I8)

struct Method {
    max_stack: u16,                      // [untrusted]
    max_locals: u16,                     // [untrusted]
    code: Vec<Instruction>,              // [untrusted] decoded, boundary-checked (I3)
    exception_table: Vec<ExceptionHandler>, // [untrusted]
}
struct Instruction {
    offset: u32,
    opcode: Opcode,
    operands: SmallVec<Operand, 4>,
}
// immutable once decoded; concurrency: shared read-only across verification workers.

enum Opcode { /* exhaustive JVM-CF opcode set, section 8.2 */ }

enum TypeTag { Top, Int, Float, Long, Double, ReturnAddress, Reference, Uninitialized }

enum VerificationType {
    Top,
    Int, Float, Long, Double,
    Null,
    Object(SymbolId),
    UninitializedThis,
    Uninitialized(u32),                  // offset of the `new` that produced it
    ReturnAddress(SubroutineId),
}
// compatibility: stable tag encoding across verifier minor versions (section 27).
struct TypeLattice;                      // stateless — join(a, b) -> VerificationType (section 8.1)

struct StackMapFrame {
    offset: u32,                         // [untrusted] when checking a supplied table
    locals: Vec<VerificationType>,       // [untrusted]
    stack: Vec<VerificationType>,        // [untrusted]
}

struct Frame {
    stack: Vec<VerificationType>,
    locals: Vec<VerificationType>,
}

struct AbstractState {
    frame: Frame,
    init_marks: BitSet,
    subroutine_id: Option<SubroutineId>,
}
// mutation: transient during fixpoint; per-block copies, never aliased across blocks.
trait TransferFunction {
    fn apply(instr: &Instruction, state: &AbstractState) -> Result<AbstractState, VerifyError>;
}

struct Cfg {
    blocks: Vec<BasicBlock>,
    entry: BlockId,
}

struct BasicBlock {
    id: BlockId,
    instructions: Range<u32>,
    successors: SmallVec<(BlockId, EdgeKind), 4>,
}

enum EdgeKind { Fallthrough, Branch, ExceptionHandler, Subroutine }

struct Worklist {
    queue: VecDeque<BlockId>,            // deterministic FIFO ordering (P12)
}

struct FixpointResult {
    block_states: HashMap<BlockId, AbstractState>,
    iterations: u32,
}
struct ExceptionHandler {
    start_pc: u32, end_pc: u32, handler_pc: u32,  // [untrusted]
    catch_type: Option<ConstantPoolIndex>,         // [untrusted]
}

struct Subroutine {
    entry: BlockId,
    call_sites: Vec<BlockId>,
    return_locals: Vec<u16>,
}

enum VerifyError {
    StackUnderflow { offset: u32 },
    MergeMismatch { offset: u32, expected: VerificationType, found: VerificationType },
    BranchTargetOutOfRange { offset: u32, target: u32 },
    ConstantPoolIndexOutOfRange { offset: u32, index: u16 },
    UninitializedUse { offset: u32 },
    ExceptionTableInvalid { handler_index: u32 },
    SubroutineAmbiguous { entry: BlockId },
    StackMapMismatch { offset: u32 },
}

struct VerdictReport {
    accepted: bool,
    errors: Vec<VerifyError>,
    replay_id: ReplayId,
}
