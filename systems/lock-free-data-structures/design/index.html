Core schema & types (§11)

struct TreiberStack<T, R: Reclaim = Hazard> { head: AtomicTaggedPtr<Node<T>>, domain: R::Domain }
// push/pop: owned-by-caller T; mutation via CAS only; concurrency: MPMC; not persisted.

struct MichaelScottQueue<T, R: Reclaim = Hazard> { head: AtomicTaggedPtr<Node<T>>,
                                                    tail: AtomicTaggedPtr<Node<T>>, domain: R::Domain }
// enqueue/dequeue; sentinel-node invariant; concurrency: MPMC; not persisted.

struct BoundedMpmcQueue<T> { slots: Box<[Slot<T>]>, mask: usize,
                              producer: CachePadded<AtomicUsize>, consumer: CachePadded<AtomicUsize> }
struct Slot<T> { sequence: AtomicUsize, value: UnsafeCell<MaybeUninit<T>> }
// try_send/try_recv; fixed capacity (power-of-two); no reclamation needed (pre-allocated).

struct WorkStealingDeque<T> { buffer: AtomicPtr<Buffer<T>>, top: AtomicUsize, bottom: AtomicUsize }
// push/pop: owner-thread only; steal: any thread; concurrency: 1 owner + N stealers.

struct SpscRing<T> { ring: Box<[UnsafeCell<MaybeUninit<T>>]>,
                      producer_idx: CachePadded<AtomicUsize>, consumer_idx: CachePadded<AtomicUsize> }
// send/recv; single-producer single-consumer; no reclamation.

struct LockFreeHashMap<K, V, R: Reclaim = Hazard> { buckets: AtomicPtr<BucketArray<K,V>>, domain: R::Domain }
// get/insert/remove; split-ordered list backing; len() approximate (documented).

struct LockFreeSkipList<K: Ord, V, R: Reclaim = Hazard> { levels: [AtomicTaggedPtr<SkipNode<K,V>>; MAX_LEVEL] }
// insert/remove/find; ordered iteration; probabilistic balancing.

struct LockFreeLinkedList<T: Ord, R: Reclaim = Hazard> { head: AtomicPtr<ListNode<T>>, domain: R::Domain }
// insert/remove/contains; Harris–Maged marked-pointer deletion.

struct WaitFreeCounter { cells: Box<[CachePadded<AtomicI64>]> }
// fetch_add: wait-free via per-thread cell isolation.

struct WaitFreeQueue<T> { announce: Box<[AtomicPtr<Op<T>>]>, /* Kogan–Petrank state */ }
// feature = "wait-free"; helping-based; bounded steps per op.

trait Guard { fn protect(&self, ptr: &AtomicPtr<()>) -> ProtectedPtr; fn retire(&self, ptr: *mut ()); }
trait Reclaim { type Domain; fn pin(domain: &Self::Domain) -> Self::Guard; }

struct TaggedPtr<T> { packed: AtomicU64 }              // (ptr: 48b, tag: 16b) on x86-64/aarch64
struct Backoff { step: Cell<u32> }                      // spin()/snooze()/is_completed()
enum SendError<T> { Full(T) }
enum RecvError { Empty }
