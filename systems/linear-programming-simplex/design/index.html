Core schema & types (§11)

enum Sense { Maximize, Minimize }
enum Relation { Le, Eq, Ge }       // constraint sense relative to RHS
enum Status {
    Optimal,
    Infeasible,
    Unbounded,
    IterationLimit,
    Numerical,
}
enum VarStatus {
    Basic,
    NonbasicLower,
    NonbasicUpper,
    Free,                           // nonbasic free variable at zero
}
struct Bound { pub lower: f64, pub upper: f64 }
// tolerances.rs — all named constants; never use magic floats
const FEAS_TOL:  f64 = 1e-7;    // primal feasibility; I2
const OPT_TOL:   f64 = 1e-8;    // reduced-cost optimality; I4
const PIVOT_TOL: f64 = 1e-7;    // minimum pivot element magnitude; I1
const REFACTOR_FILL_RATIO: f64 = 3.0;  // trigger refactorization when eta fill exceeds this × original nnz
const MAX_CONDITION_EST:   f64 = 1e12; // trigger refactorization when condition estimate exceeds
enum SolverError {
    DimensionMismatch { expected: usize, got: usize },
    InvalidBound { var: VarId, lower: f64, upper: f64 },
    SingularBasis,
    NumericalFailure { context: &'static str },
    ParseError { line: usize, msg: String },
    IterationLimitReached { limit: usize },
    InternalInvariantViolation(&'static str),
}
struct SparseMatrix {
    pub nrows: usize,
    pub ncols: usize,
    values:   Vec<f64>,   // nonzero values, column-major
    row_ind:  Vec<usize>, // row index per value
    col_ptr:  Vec<usize>, // col_ptr[j]..col_ptr[j+1] spans column j; len = ncols+1
}
struct LpProblem {
    pub sense:      Sense,
    pub objective:  Vec<f64>,       // length = n_vars; c[j] for var j
    pub matrix:     SparseMatrix,   // A (m x n); m = #constraints, n = #vars
    pub rhs:        Vec<f64>,       // length = m
    pub relations:  Vec<Relation>,  // length = m
    pub bounds:     Vec<Bound>,     // length = n_vars; default Bound{0, +inf}
    pub var_names:  Vec<String>,    // optional; empty = unnamed
    pub con_names:  Vec<String>,    // optional; empty = unnamed
}
struct StandardForm {
    pub a_std:        SparseMatrix,  // [A | I_slack | I_art] in CSC, m x n_total
    pub b:            Vec<f64>,      // RHS, length m; all >= 0 after row sign flip
    pub c:            Vec<f64>,      // minimization objective, length n_total
    pub n_struct:     usize,         // structural variables (original + bound split)
    pub n_slack:      usize,         // slack/surplus variables added
    pub n_artificial: usize,         // artificial variables for Phase I
    pub var_map:      VarMap,        // back-mapping: standard index -> original VarId + type
}
struct VarMap {
    entries: Vec<VarMapEntry>,  // index = standard-form column index
}
enum VarMapEntry {
    Structural { orig: VarId, split_neg: bool },
    Slack      { con: ConstraintId },
    Artificial { con: ConstraintId },
}
struct Tableau {
    tab:   Vec<Vec<f64>>,  // tab[i][j]: row i = constraint i, row m = objective; col n = RHS
    basis: Vec<usize>,     // basis[i] = column index of basic var for row i; length m
    m:     usize,          // number of constraints
    n:     usize,          // total columns (struct + slack + artificial)
}
struct RatioTestResult {
    pub leaving:    Option<usize>,  // None = unbounded direction
    pub step:       f64,            // theta: step size along entering direction
    pub bound_flip: bool,           // entering variable hit its opposite bound (Phase 8)
}
struct Basis {
    pub basic:   Vec<VarId>,        // length m; basic[i] = variable occupying row i of basis
    pub status:  Vec<VarStatus>,    // length n_total; status per variable
    pub factor:  LuFactorization,   // current factorization of B; see lp-lu
}
struct SavedBasis {
    pub basic:  Vec<VarId>,
    pub status: Vec<VarStatus>,
    // LuFactorization NOT saved; rebuilt on restore to avoid stale factors
}
struct LuFactorization {
    pub l_eta:     Vec<EtaVector>,    // product-form L update vectors (one per pivot after initial factor)
    pub u:         SparseMatrix,      // upper-triangular U factor (CSC)
    pub row_perm:  PermutationVec,    // row permutation P from partial pivoting
    pub col_perm:  PermutationVec,    // column permutation Q for sparsity
    pub n_updates: usize,             // count of rank-1 updates since last full factorization
}
struct EtaVector {
    pub pivot_row: usize,
    pub indices:   Vec<usize>,
    pub values:    Vec<f64>,
}
struct PermutationVec { forward: Vec<usize>, inverse: Vec<usize> }
trait Pricer {
    fn select_entering(
        &self,
        reduced_costs: &[f64],
        status: &[VarStatus],
        sense: Sense,
    ) -> Option<usize>;
}
struct Sensitivity {
    pub obj_ranges:    Vec<(f64, f64)>,  // per structural var: (c_j_lo, c_j_hi) keeping basis optimal
    pub rhs_ranges:    Vec<(f64, f64)>,  // per constraint: (b_i_lo, b_i_hi) keeping basis feasible
    pub shadow_prices: Vec<f64>,         // dual values y_i = c_B B^-1 e_i; length m
    pub reduced_costs: Vec<f64>,         // \bar{c}_j for all vars; length n_total
}
struct Solution {
    pub status:         Status,
    pub certificate:    Certificate,     // P5: always present
    pub objective:      f64,             // primal objective at optimum (or best infeasible bound)
    pub x:              Vec<f64>,        // primal variable values, length = n_struct (original vars)
    pub duals_y:        Vec<f64>,        // dual multipliers, length m
    pub reduced_costs:  Vec<f64>,        // \bar{c}_j, length = n_struct
    pub iterations:     usize,
    pub solve_time_ms:  f64,
}
enum Certificate {
    KktOptimal,                   // KKT conditions verified within tolerances
    PrimalRay(Vec<f64>),          // direction d s.t. c^T d < 0 and A d <= 0; proves unbounded
    DualRay(Vec<f64>),            // y s.t. y^T b < 0, y^T A >= 0; proves infeasibility (Farkas)
}
