Low-level design as the blueprint states it: type and trait signatures, the normative repository tree and the file inventory. Equations and state machines live on the Architecture tab (§8).

Core schema & types (§11)

RunId = NewType("RunId", str)                    // owned: Scheduler; immutable; persisted (Journal key); serializable (str); stable across versions
StepId = NewType("StepId", str)                   // owned: Agent Loop Step; immutable; persisted; serializable; stable
TenantId = NewType("TenantId", str)               // owned: caller; immutable; persisted everywhere (I11); serializable; stable
AgentId = NewType("AgentId", str)                 // owned: AgentSpec author; immutable; persisted; serializable; stable

class RunSpec(Protocol):                          // owned: caller; immutable after submit; not persisted verbatim (spec_hash is); trust: caller-provided, validated
    agent_id: AgentId
    tenant_id: TenantId
    max_steps: int
    max_wall_clock_s: int
    token_budget: "TokenBudget"

class RunState(Protocol):                         // owned: Scheduler; mutated only via journaled transitions (I1); persisted (derived from Journal); trust: internal
    run_id: RunId
    phase: Literal["PLANNING","ACTING","OBSERVING","DONE","ABORTED"]  // §8.3
    step_count: int

class JournalEventType(Enum):                     // owned: journal package; immutable enum; persisted as tag; serializable; append-only evolution (§27)
    RUN_STARTED = "RUN_STARTED"
    STEP_INTENT = "STEP_INTENT"
    STEP_RESULT = "STEP_RESULT"
    TOOL_CALL_INTENT = "TOOL_CALL_INTENT"
    TOOL_CALL_RESULT = "TOOL_CALL_RESULT"
    // full catalogue: §8.1

class JournalEntry(Protocol):                     // owned: Journal; write-once (I12); persisted; serializable (CloudEvents-shaped); trust: internal-only, never model-influenced content verbatim without provenance tag
    run_id: RunId
    seq: int
    event_type: JournalEventType
    payload_hash: str
    committed_at: float                            // monotonic + wall-clock pair (ADR-34)

class Journal(Protocol):                          // owned: worker process; concurrency: single-writer-per-run, multi-reader; persistence: itself; trust: internal
    async def append(self, entry: JournalEntry) -> None: ...
    async def read(self, run_id: RunId) -> list[JournalEntry]: ...

class Checkpoint(Protocol):                        // owned: Journal; immutable snapshot; persisted; serializable; compat: versioned (§27)
    run_id: RunId
    up_to_seq: int
    state_hash: str
class AgentSpec(Protocol):                        // owned: agent author; immutable; persisted as spec_hash; trust: author-provided, versioned (§27)
    agent_id: AgentId
    tools: list["ToolSpec"]
    policy_rules: list["PolicyRule"]
    memory_config: dict

class AgentRuntime(Protocol):                      // owned: worker process; concurrency: one instance drives many concurrent Run tasks; not persisted (stateless driver); trust: internal
    async def start(self, spec: RunSpec) -> RunId: ...
    async def resume(self, run_id: RunId) -> None: ...

class Step(Protocol):                              // owned: Agent Loop Step; immutable once journaled; persisted; trust: internal; determinism-sensitive (I2)
    step_id: StepId
    kind: Literal["model_call","tool_call","handoff"]

class StepResult(Protocol):                        // owned: Agent Loop Step; immutable once journaled; persisted; determinism-sensitive (I2, I13)
    step_id: StepId
    output_hash: str
    tokens: int
    cost_usd: float

class ToolSpec(Protocol):                          // owned: tool author; immutable; persisted (registry snapshot); versioned (§27); trust boundary definition itself
    name: str
    args_schema: dict                              // JSON Schema
    result_schema: dict
    permission_class: "PermissionClass"
    idempotent: bool
    timeout_ms: int
    retry_policy: Literal["none","fixed","exponential"]

class PermissionClass(Enum):                       // owned: framework; immutable enum; persisted as tag; trust: security-load-bearing (P3)
    READ_ONLY = "read_only"
    WRITE_LOCAL = "write_local"
    NETWORK = "network"
    DESTRUCTIVE = "destructive"

class ToolCall(Protocol):                          // owned: Agent Loop Step; immutable; persisted (TOOL_CALL_INTENT); trust: MODEL-ORIGINATED, untrusted until validated (I3)
    step_id: StepId
    tool_name: str
    args: dict
    idempotency_key: str

class ToolResult(Protocol):                        // owned: Tool dispatch; immutable; persisted (TOOL_CALL_RESULT); trust: tool-originated, untrusted content if tool touches external data (§8.7)
    is_error: bool
    result: dict | None
    error_kind: str | None                          // §8.4 failure taxonomy

class Middleware(Protocol):                         // owned: framework/extension author; concurrency: invoked in fixed total order (I8); not persisted; trust: security-critical
    async def handle(self, call: ToolCall, next: "Middleware") -> ToolResult: ...
class PolicyRule(Protocol):                         // owned: agent author/operator; immutable per version; persisted (spec); trust: security-load-bearing
    rule_id: str
    condition: str                                   // evaluated against ToolCall + provenance
    effect: Literal["allow","deny"]

class PolicyDecision(Protocol):                      // owned: Policy Engine; immutable; persisted (POLICY_DECISION); determinism-sensitive (I2)
    rule_id: str
    decision: Literal["allow","deny"]
    reason: str

class WorkingContext(Protocol):                      // owned: Agent Loop Step; mutated append-only within a run; not persisted directly (rebuilt from Journal on resume); trust: mixed (contains untrusted tool output, §8.7)
    messages: list[dict]
    token_count: int

class MemoryStore(Protocol):                         // owned: memory package; concurrency: safe for concurrent reads, single-writer per tenant partition; persisted (backend-specific); trust: mixed
    async def write(self, tenant: TenantId, item: dict) -> None: ...
    async def query(self, q: "RetrievalQuery") -> list["RetrievedItem"]: ...

class RetrievalQuery(Protocol):                      // owned: caller; immutable; not persisted; trust: internal
    tenant_id: TenantId
    text: str
    top_k: int
    allow_stale: bool

class RetrievedItem(Protocol):                       // owned: MemoryStore; immutable; not persisted beyond source index; trust: UNTRUSTED CONTENT (§8.7); provenance-tagged (I10)
    content: str
    score: float
    stale: bool
    source: str

class EpisodicRecord(Protocol):                      // owned: episodic package; immutable once written; persisted; trust: internal (derived from prior trusted+untrusted mix, re-tagged on write)
    tenant_id: TenantId
    summary: str
    covers_steps: tuple[int, int]
class TokenBudget(Protocol):                          // owned: RunSpec/TenantConfig; immutable per run; persisted; trust: internal
    max_tokens: int
    max_cost_usd: float

class BudgetLedger(Protocol):                         // owned: Budget Governor; concurrency: atomic per-scope decrement; persisted (BUDGET_CHECK entries); trust: internal
    async def check_and_reserve(self, scope: str, amount: int) -> "PolicyDecision": ...

class ProviderClient(Protocol):                       // owned: provider package; concurrency: safe for concurrent calls; not persisted (stateless client); trust: EXTERNAL BOUNDARY
    async def complete(self, req: "ModelRequest") -> "ModelResponse": ...
    async def stream(self, req: "ModelRequest") -> AsyncIterator["StreamChunk"]: ...

class ProviderRoute(Protocol):                        // owned: router; immutable per request; not persisted; determinism: fixed decision table (§8.9)
    primary: str
    fallbacks: list[str]

class ModelRequest(Protocol):                         // owned: Agent Loop Step; immutable; persisted as input_hash only (never verbatim, may contain untrusted content); trust: MIXED — contains untrusted retrieved/tool content (§8.7)
    messages: list[dict]
    tools: list[dict]
    temperature: float
    seed: int | None

class ModelResponse(Protocol):                        // owned: ProviderClient; immutable; persisted as output_hash; trust: UNTRUSTED (model output is untrusted planner output, P3); determinism-sensitive (I13)
    text: str
    tool_calls: list[ToolCall]
    tokens_in: int
    tokens_out: int

class StreamChunk(Protocol):                          // owned: ProviderClient; immutable; not persisted individually (aggregated); trust: untrusted
    delta: str
    is_final: bool

class RecordedExchange(Protocol):                      // owned: recorder package; immutable once written; persisted (replay corpus); determinism: the replay ground truth (I13)
    request_hash: str
    response: ModelResponse
class StructuredOutputSpec(Protocol):                  // owned: agent author; immutable; persisted (spec); trust: internal
    schema: dict
    max_repair_attempts: int

class ParseAttempt(Protocol):                          // owned: Structured-Output Parser; immutable; persisted (for debugging); determinism-sensitive
    attempt: int
    error: str | None

class AgentMessage(Protocol):                           // owned: Message Bus; immutable; persisted (SUBAGENT_* entries); trust: mixed (payload may be untrusted); envelope: CloudEvents-shaped
    from_agent: AgentId
    to_agent: AgentId
    kind: Literal["dispatch","result","handoff"]
    payload: dict

class MessageBus(Protocol):                             // owned: bus package; concurrency: at-least-once delivery (I14); persisted (dead-letter queue); trust: internal transport, payload trust inherited from sender
    async def publish(self, msg: AgentMessage) -> None: ...
    async def subscribe(self, agent_id: AgentId) -> AsyncIterator[AgentMessage]: ...

class SupervisorPlan(Protocol):                          // owned: supervisor agent; immutable per dispatch; persisted (spec); trust: internal
    pattern: Literal["supervisor_worker","chain","fan_out_join","handoff"]
    max_fan_out: int

class HandoffSpec(Protocol):                             // owned: supervisor/agent author; immutable; persisted; trust: security-load-bearing (authority gate, §8.5)
    granted_permissions: list[PermissionClass]
    max_retries: int
    timeout_s: int
class Span(Protocol):                                    // owned: Trace Collector; immutable once closed; persisted (exported); trust: internal, may embed untrusted attribute values (redacted, ADR-28)
    span_id: str
    parent_span_id: str | None                            // None only for root (I7)
    name: str

class TraceContext(Protocol):                             // owned: propagated per-call; immutable; not persisted itself (carried); serialization: W3C traceparent/tracestate
    trace_id: str
    span_id: str

class EvalCase(Protocol):                                  // owned: eval package; immutable; persisted (versioned corpus); trust: internal
    case_id: str
    input: dict
    expected: dict

class EvalReport(Protocol):                                 // owned: eval harness; immutable; persisted (release gate artifact); trust: internal
    pass_rate: float
    regressions: list[str]

class Quota(Protocol):                                       // owned: tenancy package; mutated atomically by Budget Governor; persisted; trust: internal
    tenant_id: TenantId
    remaining_usd: float

// Error taxonomy (all inherit AgentCoreError; each maps to a §8.4/§8.11 failure kind):
// ArgError, Timeout, PermissionDenied, ExecutionError, ResultSchemaViolation,
// ParseFailure, BudgetExceeded, StepCeilingExceeded, WallClockExceeded,
// JournalIntegrityError, JournalImmutabilityError, ReplayMiss, SubagentFailure
