Core schema & types (§11)

pub enum SinkType {
    SqlQuery,     // SQL execution: raw query string + optional params
    Exec,         // OS command execution: argv[0..n]
    FileOpen,     // filesystem open: path + flags
    HttpRequest,  // outbound HTTP: method + url + headers
    Deserialize,  // deserialization: format + raw bytes
    LdapQuery,    // LDAP filter string + base DN
    XmlParse,     // XML parse: raw bytes (XXE surface)
    NoSqlQuery,   // NoSQL doc query: BSON/JSON doc bytes
    __NonExhaustive,  // allows adding sinks without major version bump
}

pub enum SinkArgs<'a> {
    SqlQuery  { query: &'a str, params: &'a [SqlParam<'a>] },
    Exec      { argv: &'a [&'a str], env: &'a [(&'a str, &'a str)] },
    FileOpen  { path: &'a str, flags: u32 },
    HttpRequest { method: &'a str, url: &'a str, headers: &'a [(&'a str, &'a str)] },
    Deserialize { format: DeserFormat, data: &'a [u8] },
    LdapQuery { filter: &'a str, base_dn: &'a str },
    XmlParse  { data: &'a [u8] },
    NoSqlQuery{ doc: &'a [u8] },
}
pub enum DeserFormat { Java, DotNet, Python, Json, MsgPack, Cbor, Unknown }
pub enum SqlParam<'a> { Str(&'a str), Int(i64), Float(f64), Bytes(&'a [u8]), Null }
pub struct HookEvent<'a> {
    pub sink:       SinkType,
    pub args:       SinkArgs<'a>,
    pub context_id: ContextId,       // opaque u64; correlates to RequestContext
    pub phase:      HookPhase,
    pub timestamp:  u64,             // nanoseconds since UNIX epoch (monotonic at hook)
}

pub enum HookPhase { Entry, Exit }
pub type ContextId = u64;
pub struct RequestContext {
    pub id:         ContextId,
    pub http:       Option<HttpInfo>,   // None for non-HTTP entry points
    pub taint_map:  TaintMap,
    pub call_stack: CallStack,
    pub started:    u64,               // ns since UNIX epoch at request entry
    pub correlation_id: CorrelationId, // propagated from incoming X-Correlation-ID or generated
}
pub type CorrelationId = Arc<str>;   // shared ref; cheap clone for telemetry
pub struct HttpInfo {
    pub method:      HttpMethod,          // GET, POST, …
    pub path:        Arc<str>,            // raw request path
    pub query:       Option<Arc<str>>,    // raw query string
    pub headers:     Arc<[(Arc<str>, Arc<str>)]>, // name-value pairs; immutable after capture
    pub body:        Option<Bytes>,       // buffered body; None if streaming / too large
    pub remote_addr: Option<IpAddr>,
    pub tls:         bool,
}
pub enum HttpMethod { Get, Post, Put, Delete, Patch, Head, Options, Other(Arc<str>) }
pub enum TaintSource {
    HttpParam,    // URL query parameter value
    HttpHeader,   // request header value (excludes Host, Content-Type by default policy)
    HttpBody,     // POST/PUT body, URL-decoded or raw
    HttpPath,     // URL path segments
    Cookie,       // Cookie header parsed value
}

pub struct TaintTag {
    pub source:     TaintSource,
    pub byte_range: Range<usize>,  // byte offset within the original captured value
    pub param_name: Option<Arc<str>>, // e.g. "id", "q"; None for body blobs
}
pub struct TaintedValue {
    pub value:  Arc<[u8]>,             // the tainted byte sequence (possibly derived via concat etc.)
    pub tags:   SmallVec<[TaintTag; 2]>, // origin tags; multiple if from concat of two tainted inputs
    pub derived: bool,                 // true if produced by propagation op, false if direct source
}
pub struct TaintMap {
    entries: HashMap<TaintKey, TaintedValue>,  // keyed by value identity
    budget:  CostBudget,
}

pub enum TaintKey {
    StrPtr(usize),    // raw pointer address of tracked &str
    HashKey(u64),     // xxhash of bytes when address is unreliable (e.g., format! result)
}
pub struct CostBudget { remaining: u32 }   // initialized to MAX_PROPAGATION_OPS per request
pub const MAX_PROPAGATION_OPS: u32 = 8192;
pub struct Detection {
    pub rule_id:    RuleId,
    pub sink:       SinkType,
    pub severity:   Severity,
    pub confidence: Confidence,
    pub evidence:   Evidence,
    pub taint_path: TaintPath,
    pub ts:         u64,      // ns timestamp at detection
}

pub struct Evidence {
    pub raw_excerpt: Arc<str>,    // ≤256 bytes of the suspicious sink arg (sanitized for logs)
    pub matched_pattern: Option<Arc<str>>, // signature pattern id or None for semantic
    pub analyzer_signals: Vec<(Arc<str>, f32)>, // (signal_name, weight)
}

pub struct TaintPath {
    pub source: TaintSource,
    pub param_name: Option<Arc<str>>,
    pub ops: Vec<PropagationOp>,  // ordered list of propagation steps from source to sink
}

pub enum PropagationOp { Concat, Substring, Format, Decode, Unknown }
pub enum Action { Allow, Block, Monitor }
pub enum Severity { Critical, High, Medium, Low, Informational }
pub enum Confidence { High, Medium, Low }
pub type RuleId = Arc<str>;   // UUID string; e.g. "sqli-001"
pub struct Rule {
    pub id:          RuleId,
    pub sink:        SinkType,
    pub kind:        RuleKind,
    pub action:      Action,    // action when this rule fires
    pub severity:    Severity,
    pub enabled:     bool,
    pub description: Arc<str>,
}

pub enum RuleKind {
    Signature { pattern_id: Arc<str> },   // named pattern in rasp-detect
    Semantic  { analyzer_id: Arc<str> },  // heuristic/scoring analyzer
}

pub struct Policy {
    pub mode:       PolicyMode,
    pub rules:      Vec<Rule>,          // ordered; first match wins
    pub waivers:    Vec<Waiver>,
    pub fail_mode:  FailMode,
    pub version:    u32,                // monotonically increasing; checked on reload
}

pub enum PolicyMode  { Monitor, Block }
pub enum FailMode    { Open, Closed }    // Open = allow on RASP internal error; Closed = block

pub struct Waiver {
    pub rule_id:    RuleId,
    pub path_glob:  Option<Arc<str>>,   // HTTP path glob; None = all paths
    pub expires_at: Option<u64>,        // ns UNIX timestamp; None = permanent
    pub reason:     Arc<str>,
}
pub enum RaspError {
    Blocked { rule_id: RuleId, detection: Box<Detection> }, // sink blocked; app must handle
    InternalError { msg: Arc<str>, source: Option<Arc<dyn std::error::Error + Send + Sync>> },
    ContextNotFound { context_id: ContextId },    // hook fired outside tracked request
    BudgetExhausted { remaining: u32 },            // taint budget cap hit
    PolicyLoadError { path: Arc<str>, reason: Arc<str> },
    SinkNotRegistered { sink: SinkType },
}

pub type RaspResult<T> = Result<T, RaspError>;
pub struct AttackEvent {
    pub event_id:      Uuid,
    pub ts:            u64,
    pub detection:     Detection,
    pub action_taken:  Action,
    pub request:       Option<Arc<HttpInfo>>,
    pub context_id:    ContextId,
    pub correlation_id: CorrelationId,
    pub agent_version: Arc<str>,
    pub host:          Arc<str>,
}
pub trait RuntimeAdapter: Send + Sync + 'static {
    fn runtime_name(&self)    -> &'static str;
    fn runtime_version(&self) -> &'static str;
    fn supported_sinks(&self) -> &'static [SinkType];
    fn on_init(&self, registry: &mut SinkRegistry) -> RaspResult<()>;
    fn on_shutdown(&self)     -> RaspResult<()>;
}
pub struct SinkRegistry {
    handlers: HashMap<SinkType, Box<dyn SinkHandler>>,
}

pub trait SinkHandler: Send + Sync {
    fn on_entry(&self, event: &HookEvent<'_>) -> RaspResult<Action>;
    fn on_exit (&self, event: &HookEvent<'_>, action_taken: Action);
}
pub struct ContextGuard {
    _priv: (),   // non-constructable outside rasp-context
}
impl ContextGuard {
    pub fn enter(ctx: RequestContext) -> Self;  // installs ctx into thread/task-local
    pub fn current() -> Option<&'static RequestContext>; // borrow current; None if no active request
    pub fn current_mut() -> Option<RefMut<'static, RequestContext>>;
}
impl Drop for ContextGuard {
    fn drop(&mut self); // removes context from local store; zeroes taint_map
}
pub struct InlineGuard {
    registry: Arc<SinkRegistry>,
    policy:   Arc<ArcSwap<Policy>>,
    telemetry: TelemetrySender,
}
impl InlineGuard {
    pub fn on_sink_entry<'a>(&self, event: &HookEvent<'a>) -> Action;
    // returns Action to caller (trampoline); Block => caller aborts sink op
}
pub struct TelemetrySender {
    tx: tokio::sync::mpsc::Sender<AttackEvent>,
}
impl TelemetrySender {
    pub fn send(&self, event: AttackEvent);  // non-blocking; drops on full channel
    pub fn flush_sync(&self, timeout: Duration) -> bool; // blocking drain; used at shutdown
}
pub struct PolicyEngine {
    policy: ArcSwap<Policy>,
}
impl PolicyEngine {
    pub fn evaluate(&self, detection: &Detection, http_path: Option<&str>) -> Action;
    pub fn reload(&self, new_policy: Policy) -> RaspResult<()>;
    pub fn current_version(&self) -> u32;
}
