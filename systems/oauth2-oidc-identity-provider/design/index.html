Core schema & types (§11)

pub struct ClientId(pub String);            pub struct Sub(pub String);     // subject id
pub struct Kid(pub String);                 pub struct Jti(pub String);     // JWS thumbprint kid / token id
pub struct Scope(pub String);               pub struct ScopeSet(/* sorted set */);
pub struct Cnf { pub jkt: Option<String>, pub x5t_s256: Option<String> }   // confirmation (DPoP/mTLS)

pub enum GrantType { AuthorizationCode, RefreshToken, ClientCredentials,
                     DeviceCode, TokenExchange /* sketch */ }              // implicit/ROPC NOT present (P9)
pub enum ResponseType { Code, CodeIdToken /* hybrid sketch */ }            // token/id_token implicit NOT present
pub enum TokenType { AccessJwt, AccessOpaque, IdToken, Refresh, Code }
pub enum SubjectType { Public, Pairwise }
pub enum ClientAuthMethod { SecretBasic, SecretPost, PrivateKeyJwt, TlsClientAuth, None /* public+PKCE */ }
pub enum CodeChallengeMethod { S256, Plain /* gated */ }

pub enum OAuthError {                        // RFC 6749 sec. 5.2 + OIDC
    InvalidRequest, InvalidClient, InvalidGrant, UnauthorizedClient,
    UnsupportedGrantType, UnsupportedResponseType, InvalidScope, AccessDenied,
    ServerError, TemporarilyUnavailable,
    // OIDC / interaction
    LoginRequired, ConsentRequired, InteractionRequired, AccountSelectionRequired,
    // device / dpop
    AuthorizationPending, SlowDown, ExpiredToken, InvalidDpopProof, UseDpopNonce,
}
pub struct ErrorResponse { pub error: OAuthError, pub description: Option<String>, pub state: Option<String> }

pub trait Clock { fn now(&self) -> UnixTime; }
pub const SKEW_SECS: i64 = 60;               // ±skew (I15,P15)

pub fn rs256_sign(key: &RsaPrivate, msg: &[u8]) -> Vec<u8>;
pub fn rs256_verify(key: &RsaPublic, msg: &[u8], sig: &[u8]) -> bool;
pub fn es256_sign(key: &P256Private, msg: &[u8]) -> Vec<u8>;        // ECDSA P-256
pub fn ed25519_sign(key: &EdPrivate, msg: &[u8]) -> Vec<u8>;       // EdDSA (#90)
pub fn sha256(b: &[u8]) -> [u8;32];                                 // (#88) at_hash/c_hash/PKCE/thumbprint
pub fn hkdf(ikm: &[u8], salt: &[u8], info: &[u8], n: usize) -> Vec<u8>;
pub fn random_token(bytes: usize) -> String;                        // base64url(CSPRNG) ≥256-bit (P1)
pub fn ct_eq(a: &[u8], b: &[u8]) -> bool;                           // constant-time (I10)
pub fn pkce_s256(verifier: &str) -> String;                         // BASE64URL(SHA256(verifier)) (I4)
pub fn verify_password(hash: &str, pw: &[u8]) -> bool;              // Argon2/bcrypt (#91)
pub enum Alg { RS256, ES256, EdDSA }                                // NO None (I5,P5)
pub struct JwsHeader { pub alg: Alg, pub kid: Kid, pub typ: Option<String> }
pub struct ClaimSet { /* iss, sub, aud, exp, iat, nbf, jti, nonce, custom map */ }

pub fn sign_jwt(claims: &ClaimSet, key: &SigningKey) -> Result<String, JoseError>;   // alg from key
pub fn verify_jwt(jwt: &str, jwks: &Jwks) -> Result<ClaimSet, JoseError>;            // alg pinned to resolved kid (I5,I7)
pub fn validate_claims(c: &ClaimSet, exp_iss: &str, exp_aud: &str, clock: &dyn Clock)
    -> Result<(), JoseError>;                                       // iss/aud/exp/iat/nonce, ±skew (I6,I15)
pub fn at_hash(access_token: &str, alg: Alg) -> String;            // left-half hash (I6)
pub fn c_hash(code: &str, alg: Alg) -> String;
pub fn jwk_thumbprint(jwk: &Jwk) -> Kid;                            // RFC 7638 (kid)
pub enum JoseError { AlgNone, AlgMismatch, KidUnknown, BadSignature, Expired, BadClaim(String), Malformed }
pub struct SigningKey { pub kid: Kid, pub alg: Alg, /* private */ }
pub enum KeyState { Introduced, Active, Retiring, Retired }
pub struct KeyEntry { pub key: SigningKey, pub state: KeyState, pub activated_at: Option<UnixTime> }

pub trait KeyManager {
    fn active_signer(&self) -> &SigningKey;                         // signs new tokens
    fn jwks(&self) -> Jwks;                                         // public keys: Introduced+Active+Retiring
    fn resolve(&self, kid: &Kid) -> Option<&SigningKey>;            // for verify (I7)
    fn rotate(&mut self, alg: Alg) -> Result<Kid, KeyError>;        // introduce new active (I8)
    fn retire_expired(&mut self, max_ttl: Duration, now: UnixTime); // only after overlap (I8,P6)
}
pub enum KeyError { GenFailed, NoActiveKey, RetireTooEarly }
pub trait ClientStore { fn get(&self, id:&ClientId)->Option<Client>; fn put(&self,c:Client); fn delete(&self,id:&ClientId); }
pub trait CodeStore {
    fn put(&self, code: AuthCode) -> Result<(), StoreError>;
    fn consume(&self, code: &str) -> Result<AuthCode, StoreError>;  // ATOMIC single-use (I1)
}
pub trait TokenStore {
    fn put_refresh(&self, rt: RefreshToken) -> Result<(), StoreError>;
    fn rotate_refresh(&self, presented: &str) -> Result<RefreshOutcome, StoreError>; // (I9)
    fn revoke_family(&self, family: &FamilyId);                      // reuse detection (I9,P7)
    fn put_opaque_access(&self, at: OpaqueAccess);   fn lookup_opaque(&self, t:&str)->Option<OpaqueAccess>;
}
pub enum RefreshOutcome { Rotated{new_rt: RefreshToken, access: AccessGrant}, ReuseDetected{family: FamilyId} }
pub trait SessionStore { /* get/put/delete IdP session by sid */ }
pub trait ConsentStore { fn granted(&self, sub:&Sub, c:&ClientId)->Option<ScopeSet>; fn grant(&self, sub:&Sub, c:&ClientId, s:ScopeSet); }
pub trait ReplayCache { fn check_and_insert(&self, key:&str, ttl: Duration) -> bool; } // false = replay (I17,I23)
pub enum StoreError { NotFound, AlreadyConsumed, Conflict, Backend(String) }
pub struct Client {
    pub id: ClientId, pub secret_hash: Option<String>, pub confidential: bool,
    pub redirect_uris: Vec<String>,                                 // exact-match set (I3)
    pub grant_types: Vec<GrantType>, pub response_types: Vec<ResponseType>,
    pub scopes: ScopeSet, pub auth_method: ClientAuthMethod,
    pub jwks: Option<Jwks>, pub jwks_uri: Option<String>,           // for private_key_jwt
    pub subject_type: SubjectType, pub sector_identifier: Option<String>,
    pub post_logout_redirect_uris: Vec<String>, pub backchannel_logout_uri: Option<String>,
}
pub fn redirect_matches(client: &Client, uri: &str) -> bool;        // FULL-STRING equality (I3,P2)
pub fn authenticate_client(req: &TokenReq, store: &dyn ClientStore, cache: &dyn ReplayCache)
    -> Result<ClientId, OAuthError>;                                // dispatch 4 methods (I13)
pub fn register_dynamic(meta: ClientMetadata) -> Result<Client, OAuthError>;   // RFC 7591
pub struct AuthCode {                                               // opaque, single-use (I1,I2)
    pub code: String, pub client_id: ClientId, pub redirect_uri: String,
    pub code_challenge: String, pub challenge_method: CodeChallengeMethod,
    pub scope: ScopeSet, pub nonce: Option<String>, pub sub: Sub,
    pub auth_time: UnixTime, pub expires_at: UnixTime,              // ≤60 s (§4.1)
}
pub struct AccessGrant { pub token: String, pub kind: TokenType, pub scope: ScopeSet,
                         pub expires_in: u64, pub cnf: Option<Cnf> }
pub struct RefreshToken { pub token: String, pub family: FamilyId, pub sub: Sub,
                          pub client_id: ClientId, pub scope: ScopeSet,
                          pub abs_expires_at: UnixTime, pub idle_expires_at: UnixTime }
pub struct FamilyId(pub String);

pub fn issue_code(/* ... */) -> AuthCode;                           // bound, single-use (I2)
pub fn issue_access(sub:&Sub, client:&Client, scope:&ScopeSet, cnf:Option<Cnf>, km:&dyn KeyManager)
    -> AccessGrant;                                                 // JWT 9068 or opaque
pub fn issue_id_token(sub:&Sub, client:&Client, nonce:Option<&str>, at:&str, code:Option<&str>,
                      auth_time:UnixTime, km:&dyn KeyManager) -> String;   // JWS (I6)
pub fn issue_refresh(sub:&Sub, client:&Client, scope:&ScopeSet) -> RefreshToken;   // family head (I9)
pub fn scope_subset(requested:&ScopeSet, registered:&ScopeSet) -> Result<ScopeSet, OAuthError>; // (I14)
// /authorize
pub struct AuthorizeReq { pub client_id: ClientId, pub redirect_uri: String, pub response_type: ResponseType,
    pub scope: ScopeSet, pub state: Option<String>, pub nonce: Option<String>,
    pub code_challenge: String, pub code_challenge_method: CodeChallengeMethod,
    pub prompt: Option<Prompt>, pub max_age: Option<u64> }
pub enum AuthorizeOutcome { Redirect{ uri: String /*?code&state&iss (I22)*/ },
                            RedirectError{ uri:String, err: OAuthError },
                            NeedLogin, NeedConsent }
pub fn authorize(req: AuthorizeReq, sess: &dyn SessionStore, cons: &dyn ConsentStore) -> AuthorizeOutcome;

// /token
pub struct TokenReq { pub grant_type: GrantType, /* code, redirect_uri, code_verifier, refresh_token,
                        client auth fields, device_code, scope, audience ... */ }
pub struct TokenResp { pub access_token: String, pub token_type: String /*Bearer/DPoP*/,
    pub expires_in: u64, pub refresh_token: Option<String>, pub id_token: Option<String>, pub scope: ScopeSet }
pub fn token(req: TokenReq, ctx: &Endpoints) -> Result<TokenResp, ErrorResponse>;   // dispatch (I2,I4,I9,I13)
pub fn introspect(req: IntrospectReq, caller: ClientId) -> IntrospectResp;          // RFC 7662 (I16)
pub fn revoke(req: RevokeReq, caller: ClientId) -> Result<(), ErrorResponse>;       // RFC 7009 (I9)
pub struct DpopProof { pub htm: String, pub htu: String, pub jti: Jti, pub iat: UnixTime, pub jwk: Jwk }
pub fn verify_dpop(proof: &str, method:&str, url:&str, cache:&dyn ReplayCache)
    -> Result<Cnf /*jkt*/, OAuthError>;                            // htu/htm/jti/sig (I17,P14)
pub fn bind_mtls(cert_thumbprint: &str) -> Cnf;                    // cnf.x5t#S256 (I18)
pub fn check_mtls_binding(cnf: &Cnf, presented: &str) -> bool;    // (I18)
pub fn iss_response_param(issuer: &str) -> (&'static str, String); // ("iss", issuer) (I22,RFC9207)
pub fn build_id_token_claims(sub:&Sub, client:&Client, nonce:Option<&str>,
    at_hash:Option<String>, c_hash:Option<String>, auth_time:UnixTime, acr:Option<String>,
    amr:Vec<String>, clock:&dyn Clock) -> ClaimSet;               // (I6)
pub fn scope_to_claims(scope:&ScopeSet, user:&UserRecord) -> ClaimsMap;  // openid/profile/email/...
