Core schema & types (§11)

#[derive(Clone, Copy, PartialEq, Eq, Hash)]
pub struct Var(pub u32);                            // 0-based; DIMACS v → Var(v-1); idx < nvars; POD Send+Sync

#[derive(Clone, Copy, PartialEq, Eq, Hash)]
pub struct Lit(pub u32);                            // bits = var<<1 | sign (0=pos,1=neg); POD
impl Lit {                                          // signatures only (No-Code Rule §4)
    pub fn new(v: Var, neg: bool) -> Lit;           // (v.0<<1)|neg
    pub fn var(self) -> Var;                        // self.0>>1
    pub fn sign(self) -> bool;                      // self.0&1
}
impl std::ops::Not for Lit { type Output = Lit; }   // not = self.0 ^ 1

#[derive(Clone, Copy, PartialEq, Eq)] #[repr(u8)]
pub enum LBool { False = 0, True = 1, Undef = 2 }   // tri-value u8; ^sign flips T/F, leaves Undef
impl LBool { pub fn from_bool(b: bool) -> LBool; pub fn xor_sign(self, s: bool) -> LBool; }
pub struct ClauseHeader {     // 8 bytes, precedes the inline [Lit] in the arena
    size:   u32,              // literal count; bounds the trailing [Lit]
    lbd:    u32,              // glue; 0 for original clauses
    learnt: bool,             // reduceDB only touches learnt
    deleted:bool,             // tombstone; reclaimed at compaction
    used:   u8,               // reduceDB recency (decremented each reduce)
}
pub struct Clause { header: ClauseHeader, lits: [Lit] }   // DST; lits[0..2] = watched (I1)

#[derive(Clone, Copy, PartialEq, Eq, Hash)]
pub struct ClauseRef(pub u32);                            // u32 word-offset, NOT a pointer; survives compaction via remap

pub struct ClauseArena { words: Vec<u32>, wasted: usize } // bump alloc; alloc→ref, clause(cref), free=tombstone, compact()→remap
pub struct VarMap<T>(Vec<T>);                         // dense, len nvars; debug-bounds, release get_unchecked after one proof
impl<T> std::ops::Index<Var> for VarMap<T> { type Output = T; }
pub struct LitMap<T>(Vec<T>);                         // dense, len 2*nvars (watch/occurrence lists)
pub struct Cnf { pub nvars: u32, pub clauses: Vec<ClauseRef>, pub arena: ClauseArena }  // owns arena; Send
pub enum SolveResult { Sat(Model), Unsat, Unknown }   // #[non_exhaustive]; Unknown = budget/interrupt
pub struct Model(VarMap<bool>);                       // total over [0,nvars); reconstructed via EliminatedClauseStack on SAT
impl Model { pub fn value(&self, v: Var) -> bool; pub fn lit_true(&self, l: Lit) -> bool; }
pub struct DimacsParser { /* line/col cursor, var-count check, dedup scratch */ }  // streaming; drops tautologies, dedups, validates p cnf V C
pub fn parse_dimacs<R: Read>(r: R, out: &mut Cnf) -> Result<(), DimacsError>;
pub fn write_dimacs<W: Write>(cnf: &Cnf, w: W) -> std::io::Result<()>;
pub enum DimacsError { BadHeader, VarOutOfRange, MissingZeroTerminator, IoError, NonNumeric }  // #[non_exhaustive], carries line no.
pub struct Solver {                  // owns the arena (moved from Cnf) + per-var state; !Sync, &mut self only
    arena:   ClauseArena,            // all clause bytes
    trail:   Trail,                  // assignment stack + level limits
    watches: LitMap<Vec<Watcher>>,   // 2*nvars buckets; watched-by lists
    assigns: VarMap<LBool>,          // current truth value per var
    level:   VarMap<u32>,            // decision level a var was assigned at
    reason:  VarMap<Reason>,         // antecedent clause / Decision / Unit
    vsids:   Vsids,                  // branching heuristic
    phase:   PhaseSaving,            // saved polarity per var
    restart: RestartPolicy,          // Luby + Glucose-LBD EMA + blocking
    db:      ClauseDb,               // learnt-clause database (reduceDB)
    qhead:   usize,                  // propagation cursor into trail.lits
    conflicts: u64, decisions: u64, propagations: u64,  // statistics
}
impl Solver {
    pub fn new(cnf: Cnf) -> Solver;
    pub fn solve(&mut self) -> SolveResult;
    pub fn solve_under(&mut self, assumptions: &[Lit]) -> SolveResult;
    pub fn add_clause(&mut self, lits: &[Lit]);
    fn propagate(&mut self) -> Option<ClauseRef>;        // BCP; Some=conflict
    fn analyze(&mut self, conflict: ClauseRef) -> (Vec<Lit>, u32);  // (learnt, bj level)
    fn enqueue(&mut self, lit: Lit, reason: Reason);
    fn backjump(&mut self, level: u32);
    fn pick_branch(&mut self) -> Option<Lit>;
}
pub struct Watcher { pub cref: ClauseRef, pub blocker: Lit }   // 8 B; blocker-True ⇒ skip arena load (re-validate vs assigns)
pub struct Trail { lits: Vec<Lit>, lim: Vec<usize> }           // lim[d] = level-d start; backjump truncates to lim[L]
pub enum Reason { Decision, Clause(ClauseRef), Unit }          // #[non_exhaustive]; antecedent in the implication graph
pub struct Vsids { activity: VarMap<f64>, inc: f64, decay: f64, heap: BinaryHeap<Var> }  // bump by inc, lazy inc*=1/decay, rescale at 1e100 (order-preserving), lazy heap of unassigned
impl Vsids { fn bump(&mut self, v: Var); fn decay_inc(&mut self); fn rescale(&mut self); }
pub struct PhaseSaving { saved: VarMap<bool> }                 // pick_branch reuses last polarity
pub struct RestartPolicy {            // Glucose: fire fast>K·slow (K~1.25); block when trail>R·avg (R~1.4); Luby fallback
    fast_lbd: f64, slow_lbd: f64,     // EMAs of recent learnt LBD
    trail_ema: f64,                   // blocking restarts
    luby_idx: u64, luby_unit: u64,    // fallback sequence
}
pub struct ClauseDb { learnts: Vec<ClauseRef>, reduce_limit: u64 }  // keep LBD-2 forever; sort (lbd,activity), drop worst half, skip reasons; geometric reduce_limit
pub enum ProofStep { Add(Vec<Lit>), Delete(Vec<Lit>) }   // #[non_exhaustive]; UNSAT at Add(empty); append-only stream
pub trait ProofTracer { fn add(&mut self, lits: &[Lit]); fn delete(&mut self, lits: &[Lit]); }   // zero-cost no-op when off
pub fn check_drat(cnf: &Cnf, proof: &[ProofStep]) -> Result<(), ProofError>;   // backward RUP, else RAT on first-lit pivot; trims; Ok iff ⊥ reachable
pub fn check_lrat(cnf: &Cnf, proof: &LratProof) -> Result<(), ProofError>;     // id-hinted, linear, no search
pub enum ProofError { RupFailed, RatFailed, MissingEmptyClause, BadClauseId, Malformed } // #[non_exhaustive]
