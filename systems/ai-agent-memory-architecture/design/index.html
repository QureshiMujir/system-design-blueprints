Core schema & types (§11)

Ulid layout (128 bits):
  [ 48-bit ms timestamp | 80-bit random ]
  → monotonic within 1 ms; collision prob < 2^-80 per ms per node
enum MemoryKind {
    Working,     // transient, session-scoped; evicted to long-term on session end
    Episodic,    // specific events/observations with timestamp provenance
    Semantic,    // distilled facts, summaries; no single-event provenance
    Procedural,  // skills, how-to patterns; typically low decay rate
}
struct MemoryRecord {
    id:            MemoryId,          // immutable; set at creation
    kind:          MemoryKind,        // immutable after creation
    content:       String,            // UTF-8; max 64 KiB enforced at ingest
    embedding_ref: EmbeddingId,       // must reference valid embedding; invariant enforced
    salience:      Salience,          // mutable; updated by decay + access boost
    created_at:    Timestamp,         // monotonic; immutable after creation
    last_access:   Timestamp,         // mutable; updated on every recall hit
    access_count:  u32,               // mutable; saturates at u32::MAX
    provenance:    Provenance,        // immutable after creation
    links:         Vec<MemoryId>,     // mutable; appended by consolidation; no duplicates
    ttl:           Option<Duration>,  // mutable; None = no expiry; 0 = delete immediately
    namespace:     NsId,              // immutable; set at ingest; isolation boundary
}
struct Salience(f32);   // range [0.0, 1.0]; clamped on construction
impl Salience {
    fn new(v: f32) -> Self;                  // clamp to [0,1]
    fn decay(&self, half_life: Duration, elapsed: Duration) -> Self;  // exp decay
    fn boost(&self, delta: f32) -> Self;     // clamp after add
}
struct Embedding {
    dim:    u16,            // fixed per namespace/provider; mismatch = MemError::DimMismatch
    data:   Box<[f32]>,     // length == dim; L2-normalized on storage
    model:  Arc<str>,       // embedding model id; used to detect provider change
}
struct EmbeddingRef {
    id:     EmbeddingId,
    dim:    u16,
    model:  Arc<str>,
}
struct Provenance {
    source:       String,              // URI, agent-id, tool-id, or "user"
    session:      SessionId,           // originating session
    derived_from: Vec<MemoryId>,       // parent records (consolidation lineage)
    ingested_at:  Timestamp,           // wall-clock time of ingest
}
struct Query {
    text:            Option<String>,          // raw text; tokenized for BM25
    embedding:       Option<Embedding>,       // pre-computed query embedding; if None, embed text
    k:               usize,                   // top-K results requested; max 1000
    kind_filter:     Option<Vec<MemoryKind>>, // None = all kinds
    recency_bias:    f32,                     // [0,1]; weight on recency index in RRF
    importance_bias: f32,                     // [0,1]; weight on importance index in RRF
    filters:         MetaFilter,              // arbitrary metadata predicate
    as_of:           Option<Timestamp>,       // time-travel: exclude created_at > as_of
    namespace:       NsId,                    // required; no cross-namespace queries
}

struct MetaFilter {
    tags:       Option<Vec<String>>,          // AND-all; tag set membership
    kind:       Option<MemoryKind>,           // redundant with kind_filter; kept for API ergonomics
    min_salience: Option<Salience>,           // pre-filter; applied before ANN
    custom:     Option<Box<dyn Fn(&MemoryRecord) -> bool + Send + Sync>>, // runtime predicate
}
struct RetrievalResult {
    record:     Arc<MemoryRecord>,
    score:      f32,                  // final fused score after RRF + MMR; range [0,1]
    components: ScoreComponents,
    rank:       usize,                // 1-based final rank
}

struct ScoreComponents {
    vec:        f32,   // ANN cosine similarity contribution (normalized)
    bm25:       f32,   // BM25 score contribution (normalized)
    recency:    f32,   // recency index score contribution
    importance: f32,   // importance (salience) contribution
    rrf_score:  f32,   // raw RRF aggregate before MMR
}
struct ContextBudget {
    max_tokens: u32,    // hard ceiling; context assembly MUST NOT exceed; error if overflow
    reserved:   u32,    // tokens reserved for system prompt + agent template; deducted first
}

struct ContextWindow {
    budget:   ContextBudget,
    entries:  Vec<ContextEntry>,    // ordered by priority; assembled into prompt
    used:     u32,                  // running token count; invariant: used <= budget.max_tokens - reserved
}

struct ContextEntry {
    record:   Arc<MemoryRecord>,
    tokens:   u32,                  // pre-computed token count for this record's content
    priority: f32,                  // derived from score + kind weight; higher = kept on overflow
}
struct ConsolidationJob {
    id:          Ulid,
    kind:        ConsolidationKind,
    namespace:   NsId,
    targets:     Vec<MemoryId>,      // records to consolidate; non-empty
    created_at:  Timestamp,
    priority:    u8,                 // 0=high (user-triggered), 255=low (background sweep)
    status:      JobStatus,
}

enum ConsolidationKind {
    Summarize,   // cluster → summarize → new Semantic record
    Reflect,     // higher-order pattern extraction → new Semantic/Procedural record
    Merge,       // near-dup dedup → canonical record; tombstone losers
    LinkUpdate,  // recompute MemoryRecord.links for target set
}

enum JobStatus {
    Queued,
    Running { started_at: Timestamp, worker_id: u32 },
    Done    { finished_at: Timestamp, produced: Vec<MemoryId> },
    Failed  { reason: String, retries: u8 },
}
trait LlmIface: Send + Sync {
    async fn summarize(&self, texts: &[&str], hint: &str) -> Result<String, MemError>;
    async fn extract_facts(&self, text: &str) -> Result<Vec<String>, MemError>;
    async fn embed(&self, texts: &[&str]) -> Result<Vec<Embedding>, MemError>;
    async fn rerank(&self, query: &str, docs: &[&str]) -> Result<Vec<f32>, MemError>;
}
trait StorageBackend: Send + Sync {
    fn put(&self, record: &MemoryRecord) -> Result<(), MemError>;
    fn get(&self, id: MemoryId) -> Result<Option<Arc<MemoryRecord>>, MemError>;
    fn delete(&self, id: MemoryId, reason: EvictionReason) -> Result<(), MemError>;
    fn scan(&self, ns: &NsId, cursor: Option<MemoryId>, limit: usize)
        -> Result<Vec<Arc<MemoryRecord>>, MemError>;
    fn flush(&self) -> Result<(), MemError>;   // fsync WAL + store
}

trait IndexBackend: Send + Sync {
    fn insert(&self, id: MemoryId, embedding: &Embedding) -> Result<(), MemError>;
    fn remove(&self, id: MemoryId) -> Result<(), MemError>;
    fn search(&self, query: &Embedding, k: usize, ef: usize)
        -> Result<Vec<(MemoryId, f32)>, MemError>;
    fn rebuild(&self, records: &[(MemoryId, Embedding)]) -> Result<(), MemError>;
}
enum MemError {
    // Storage
    WalCorrupted     { path: PathBuf, offset: u64, detail: String },
    StoreFull        { namespace: NsId, bytes_free: u64 },
    RecordNotFound   { id: MemoryId },
    EmbeddingMissing { id: EmbeddingId },
    // Index
    DimMismatch      { expected: u16, got: u16 },
    IndexCorrupted   { kind: String, detail: String },
    // Ingest / Retrieval
    InvalidQuery     { reason: String },
    DeduplicateConflict { id: MemoryId },
    BudgetExceeded   { used: u32, max: u32 },
    // LLM interface
    LlmUnavailable   { provider: String, cause: String },
    LlmTimeout       { provider: String, elapsed_ms: u64 },
    RerankLenMismatch { expected: usize, got: usize },
    // System
    Io               { source: std::io::Error },
    InvalidSalience  { value: f32 },
    NamespaceNotFound { ns: NsId },
    NamespaceQuotaExceeded { ns: NsId, quota: u64, used: u64 },
    Serialization    { detail: String },
    InternalInvariantViolated { detail: String },
}
struct Timestamp(i64);   // microseconds since Unix epoch; monotonic within process via Instant
struct SessionHandle {
    id:        SessionId,
    namespace: NsId,
    opened_at: Timestamp,
    config:    SessionConfig,       // decay rates, budget overrides, llm iface binding
}

struct SessionConfig {
    budget:         ContextBudget,
    working_cap:    usize,          // max Working records in context at once (default 50)
    decay_override: Option<HashMap<MemoryKind, Duration>>,  // half-life per kind
    llm:            Arc<dyn LlmIface>,
}
enum EvictionReason {
    TtlExpired    { ttl: Duration },
    DecayEviction { final_salience: Salience },
    UserForget    { session: SessionId },
    Compaction    { superseded_by: MemoryId },  // merge/dedup: loser record
    QuotaEviction { namespace: NsId },
}
