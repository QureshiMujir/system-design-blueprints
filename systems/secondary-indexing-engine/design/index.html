Core schema & types (§11)

type SequenceId = u64;
// Ownership: copy type. Mutation: monotonic allocator only (Mutation Runtime).
// Concurrency: allocated via atomic fetch-add. Persistence: every WAL frame (§8.2).
// Serialization: fixed-width protobuf fixed64. Compatibility: never renumbered (I4, P9).

struct SnapshotHandle;
epoch: u64;
sequence_upper_bound: u64;
// Ownership: reference-counted, immutable once constructed (Arc-equivalent).
// Mutation: none post-construction. Concurrency: refcount via atomic; safe to clone across threads.
// Persistence: transient, never written to disk. Serialization: n/a (in-process only).
// Compatibility: internal type, no wire contract.

struct SegmentMetadata;
segment_id: u128;
level: u32;
key_min: bytes;
key_max: bytes;
checksum: u64;
created_at_ns: u64;
// Ownership: owned by Manifest Runtime; borrowed immutably by Segment Manager and Compaction.
// Mutation: immutable after manifest publication (I3). Concurrency: read via Arc snapshot.
// Persistence: manifest record (§8.4 footer mirror). Serialization: binary protobuf (ADR-26).
// Compatibility: new optional fields only, never removed (P9).

struct MutationBatch;
sequence_start: u64;
records: Vec<MutationRecord>;
// Ownership: owned by Mutation Runtime for the duration of one WAL append call.
// Mutation: append-only construction, then immutable. Concurrency: single-writer per shard.
// Persistence: WAL frame sequence (§8.2, §8.3). Serialization: protobuf, one frame per record.
// Compatibility: field additions only.

trait IndexIterator;
fn seek(&mut self, key: &[u8]);
fn next(&mut self);
fn valid(&self) -> bool;
fn key(&self) -> &[u8];
fn value(&self) -> &[u8];
// Ownership: implementor owns underlying cursor state (memtable/segment/merge).
// Mutation: internal cursor position mutates on seek/next; no other side effects (P1).
// Concurrency: not Send across threads without an explicit snapshot pin (I10).
// Persistence: n/a (in-process traversal only). Serialization: n/a.
// Compatibility: trait is internal API; additive methods only via default-method extension.
struct WALRecord;
lsn: u64;
payload_length: u32;
crc32c: u32;
// Ownership: owned by WAL Manager; read-only view exposed to replay scanner.
// Mutation: immutable once written (P5, I3-adjacent for WAL). Concurrency: single-writer append.
// Persistence: primary durability unit (§8.3). Serialization: fixed header, protobuf payload.
// Compatibility: header layout is versioned (ADR-23); never changes without a major bump.

struct CompactionPlan;
source_segments: Vec<SegmentId>;
target_level: u32;
priority: u32;
// Ownership: owned by Compaction Scheduler for the lifetime of one job.
// Mutation: immutable once dispatched to a worker. Concurrency: one plan per worker slot.
// Persistence: transient; recorded in manifest only after successful rewrite (§17).
// Compatibility: internal type, no wire contract.

struct ReplicationFrame;
term: u64;
index: u64;
payload: bytes;
// Ownership: owned by Replication Runtime; transferred to peer via RPC (borrowed for send).
// Mutation: immutable once constructed. Concurrency: one append-entries pipeline per follower.
// Persistence: mirrors the WAL frame it ships (§8.7). Serialization: protobuf over gRPC (ADR-06).
// Compatibility: term/index are wire-stable; payload envelope is versioned (P9).

struct BloomFilterBlock;
bit_count: u32;
hash_count: u8;
bits: bytes;
// Ownership: owned by the Segment reader that built it; immutable after segment publication.
// Mutation: none (P5). Concurrency: read-only, shared via mmap. Persistence: segment footer (§8.4).
// Serialization: raw bit-vector, no protobuf (hot-path read). Compatibility: versioned by
// segment-format version field in the footer.

struct ManifestRecord;
generation: u64;
op: ManifestOp;              // ManifestOp = AddSegment | RemoveSegment | Checkpoint
segment_metadata: Option<SegmentMetadata>;
checksum: u32;
// Ownership: owned by Manifest Runtime; append-only transaction log entry.
// Mutation: immutable once committed (I2, I13). Concurrency: single-writer, serialized commits.
// Persistence: the manifest log itself (§17). Serialization: binary protobuf (ADR-26).
// Compatibility: `op` enum is append-only; unknown ops must be forward-skippable (P9).
