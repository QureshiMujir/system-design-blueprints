Core schema & types (§11)

@dataclass(frozen=True)
class ProcessGroup:
    name: str                        # "dp" | "tp" | "pp" | "world" | "<custom>"
    ranks: tuple[int, ...]           # global ranks, sorted
    backend: str                     # "nccl" | "gloo" | "shim"
    size: int                        # property
    local_rank: int                  # property; -1 if caller not a member
    def is_member(self) -> bool: ...
@dataclass(frozen=True)
class DeviceLocation:
    host: str
    numa_node: int
    gpu_index: int

@dataclass(frozen=True)
class LinkInfo:
    bandwidth_gbps: float
    latency_us: float
    kind: str                        # "nvlink" | "pcie" | "ib" | "tcp"

@dataclass
class Topology:
    world_size: int
    locations: dict[int, DeviceLocation]
    links: dict[tuple[int, int], LinkInfo]
    def fastest_link(self, src: int, dst: int) -> LinkInfo: ...
    def same_host(self, src: int, dst: int) -> bool: ...
    def hosts(self) -> set[str]: ...
@dataclass(frozen=True)
class Mesh:
    dp_size: int
    tp_size: int
    pp_size: int
    rank_grid: np.ndarray             # shape (dp_size, tp_size, pp_size), dtype=int
    world_size: int                   # property = dp_size * tp_size * pp_size
    def world_rank(self, dp: int, tp: int, pp: int) -> int: ...
    def coords(self, world_rank: int) -> tuple[int, int, int]: ...
    def dp_peers(self, world_rank: int) -> list[int]: ...
    def tp_peers(self, world_rank: int) -> list[int]: ...
    def pp_peers(self, world_rank: int) -> list[int]: ...
class ShardKind(Enum):
    REPLICATE = "replicate"          # every rank holds a full copy
    SHARD_DIM_0 = "shard_dim_0"      # split along dim 0 (column-parallel)
    SHARD_DIM_1 = "shard_dim_1"      # split along dim 1 (row-parallel)
    SHARD_FLAT = "shard_flat"        # split along a flattened view (ZeRO-3)

@dataclass(frozen=True)
class ShardSpec:
    kind: ShardKind
    group: ProcessGroup
    full_shape: tuple[int, ...]

@dataclass
class PartitionedTensor:
    local: torch.Tensor              # the on-device shard
    spec: ShardSpec
    full_shape: tuple[int, ...]       # property = spec.full_shape
    def all_gather_full(self) -> torch.Tensor: ...   # materializes full tensor; expensive
@dataclass
class PipelineStage:
    stage_id: int                     # pp_rank
    num_stages: int                   # pp_size
    module: nn.Module
    is_first: bool
    is_last: bool
    input_shape: tuple[int, ...]
    output_shape: tuple[int, ...]
    prev_rank: int | None
    next_rank: int | None
    activation_buffers: list[torch.Tensor]
    gradient_buffers: list[torch.Tensor]

class StepKind(Enum):
    FORWARD = "forward"; BACKWARD = "backward"
    SEND_ACTIVATION = "send_activation"; RECV_ACTIVATION = "recv_activation"
    SEND_GRADIENT = "send_gradient"; RECV_GRADIENT = "recv_gradient"
    OPTIMIZER_STEP = "optimizer_step"

@dataclass(frozen=True)
class ScheduleStep:
    kind: StepKind
    mb_id: int | None                 # None for OPTIMIZER_STEP
    peer: int | None                  # peer world rank for send/recv
class CollectiveKind(Enum):
    ALL_REDUCE = "all_reduce"; ALL_GATHER = "all_gather"
    REDUCE_SCATTER = "reduce_scatter"; BROADCAST = "broadcast"
    SEND = "send"; RECV = "recv"; BARRIER = "barrier"

@dataclass
class CollectiveOp:
    kind: CollectiveKind
    group: ProcessGroup
    tensors: list[torch.Tensor]
    op_id: int                        # monotonic per group
    timeout_ms: int = 30_000
    bucket_id: int | None = None

async def issue(op: CollectiveOp) -> None: ...   # backpressure + retry + trace, §22
@dataclass
class ParallelStrategy:
    dp_size: int = 1
    tp_size: int = 1
    pp_size: int = 1
    zero_stage: int = 0                # 0, 1, 2, 3
    micro_batch_size: int = 1
    grad_accum_steps: int = 1
    activation_checkpoint: str = "none"  # "none" | "block" | "selective"
    mixed_precision: str = "bf16"        # "fp32" | "fp16" | "bf16"
    sequence_parallel: bool = False
    offload_optimizer: bool = False      # ZeRO-Infinity, deferred (§39)
    offload_parameters: bool = False     # ZeRO-Infinity, deferred (§39)

class Trainer(ABC):
    def __init__(self, model: nn.Module, optimizer: Optimizer,
                 train_loader: DataLoader, strategy: ParallelStrategy,
                 root_seed: int = 0) -> None: ...
    def step(self) -> dict[str, float]: ...
    def fit(self, num_steps: int) -> None: ...
    def save_checkpoint(self, path: str) -> None: ...
    def load_checkpoint(self, path: str) -> int: ...
class ParallelEngine(ABC):
    def wrap(self, module: nn.Module) -> nn.Module: ...
    def post_backward(self, param: nn.Parameter) -> None: ...
    def optimizer_step(self) -> None: ...
    def zero_grad(self) -> None: ...
    def state_dict(self) -> dict: ...
    def load_state_dict(self, sd: dict) -> None: ...
