Core schema & types (§11)

pub struct Image<T> { data: Vec<T>, w: usize, h: usize, stride: usize }   // row-major, explicit stride
pub type Gray8 = u8; pub type Rgb8 = [u8;3]; pub type F32 = f32;
impl<T:Copy> Image<T> { fn at(&self,x,y)->T; fn at_mut(&mut self,x,y)->&mut T;
    fn roi(&self, r:Rect)->ImageView<'_,T>;  fn dims(&self)->(usize,usize); }
pub struct ImageView<'a,T>{ /* borrows parent; never copies (principle 12) */ }
pub struct Rect { x:i32, y:i32, w:u32, h:u32 }  // page frame (I1)
impl Rect { fn contains(&self,o:&Rect)->bool; fn union(&self,o:&Rect)->Rect; fn intersect(&self,o:&Rect)->Option<Rect>; }
pub struct Point{ x:f32, y:f32 }  pub struct Polygon(Vec<Point>);  pub struct LineSeg{ a:Point, b:Point }
pub struct Page { dims:(usize,usize), dpi:f32, blocks:Vec<Block>, reading_order:Vec<usize> } // I2
pub struct Block { bbox:Rect, kind:BlockKind, lines:Vec<Line> }
pub enum BlockKind { Text, Image, Rule, Table }
pub struct Line { bbox:Rect, baseline:LineSeg, x_height:f32, words:Vec<Word> }
pub struct Word { bbox:Rect, text:String, conf:Confidence, glyphs:Vec<Glyph> }
pub struct Glyph{ bbox:Rect, ch:char, conf:Confidence }
pub struct Confidence(f32); // ∈[0,1] (I10)
pub struct RawImage(Image<Gray8>);
pub struct BinaryImage{ img:Image<Gray8> /*∈{0,255}*/, skew:Option<f32> }
pub struct AnalyzedPage{ bin:BinaryImage, blocks:Vec<Block>, order:Vec<usize>, t_deskew:Affine } // I15
pub struct SegmentedLines{ page:AnalyzedPage, lines:Vec<LineImage> }
pub struct LineImage{ bbox:Rect, norm:Image<F32> /*fixed height H*/, src_transform:Affine }
pub struct RecognizedPage(Page);
pub trait Recognizer {
    fn recognize_line(&self, line:&LineImage) -> RecOutput;   // engine-agnostic
    fn alphabet(&self) -> &Alphabet;
}
pub enum RecOutput { Posteriors(Array2<f32> /*T×(|A|+1)*/), Glyphs(Vec<(char,f32,Rect)>) }
pub struct Alphabet { chars:Vec<char>, blank:usize }          // CTC blank index (I5)
pub struct ClassicalRecognizer{ clf:Svm /*or Knn, #372*/, feat:FeatureSpec }
pub struct NeuralRecognizer{ cnn:CnnStack, rnn:BiLstm, head:Linear, alphabet:Alphabet } // #112/#349
pub struct CtcLoss { blank:usize }
impl CtcLoss { fn forward_backward(&self, logp:&Array2<f32>, label:&[usize]) -> (f32 /*loss*/, Array2<f32> /*grad*/); } // I4, log-space
pub fn greedy_decode(logp:&Array2<f32>, blank:usize) -> Vec<usize>;          // collapse+strip (I5)
pub struct BeamEntry { prefix:Vec<usize>, p_blank:f32, p_nonblank:f32, lm_state:LmState }
pub fn prefix_beam(logp:&Array2<f32>, lex:&Dawg, lm:&NGramLm, cfg:BeamCfg) -> Vec<(Vec<usize>,f32)>;
pub struct Dawg { /* compact directed acyclic word graph */ }
impl Dawg { fn contains(&self,w:&str)->bool; fn walk(&self, prefix:&str)->Option<Node>;
            fn within_edit_distance(&self, w:&str, k:u32)->Vec<(String,u32)>; }   // Levenshtein cutoff
pub struct NGramLm { n:u8, /* counts + backoff weights, Kneser-Ney */ }
impl NGramLm { fn logprob(&self, hist:&[Sym], next:Sym)->f32; fn perplexity(&self, text:&[Sym])->f32; }
pub struct BeamCfg { beam:usize, lm_weight:f32 /*α*/, word_bonus:f32 /*β*/, use_lexicon:bool }
pub struct Integral { sum:Vec<u64>, sqsum:Vec<u64>, w:usize, h:usize }     // O(1) window stats
impl Integral { fn window_mean_std(&self, r:Rect)->(f32,f32); }
pub enum BinMethod { Otsu, Sauvola{k:f32,window:u32,r:f32}, Niblack{k:f32,window:u32}, Wolf{..} }
pub struct CcLabels { labels:Image<u32>, n:u32, boxes:Vec<Rect>, areas:Vec<u32> } // from #389 union-find
pub struct ReadingOrder { order:Vec<usize> }   // total order over blocks (I2)
pub struct Config { engine:Engine, bin:BinMethod, deskew:bool, beam:BeamCfg, langs:Vec<Lang>, n_jobs:usize }
pub enum Engine { Classical, Neural }
pub enum OcrError { BadParam{what:&'static str}, DecodeFailed, DimMismatch{expected:usize,got:usize},
    ImageTooLarge{px:u64}, LargeSkew{deg:f32}, IncompatibleVersion{got:u32,want:u32},
    NotFitted, ModelAlphabetMismatch }
pub struct PageTrace { stage_ms:BTreeMap<&'static str,f32>, n_components:u32, skew_deg:f32,
    n_blocks:u32, n_lines:u32, mean_line_conf:f32, beam_width:usize, unk_count:u32, seed:u64 }
