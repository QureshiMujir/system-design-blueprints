Core schema & types (§11)

TxnId: str (uuid4)                          # immutable, globally unique, serializable
Timestamp: int (nanoseconds since epoch)     # totally ordered, monotonic per I8
TrueTimeInterval: (earliest: Timestamp, latest: Timestamp)  # latest-earliest <= 2*epsilon
Epsilon: int (ms)                            # config value, monitored by ClockMonitor
Hlc: (physical: Timestamp, logical: int)     # comparison/fallback clock, §36
TimestampOracle: assigns Timestamp from TrueTimeInterval; enforces I8 epoch fencing
Transaction: (txn_id: TxnId, coordinator_epoch: int, state: TxnState,
              read_set: ReadSet, write_set: WriteSet,
              start_ts: Timestamp, commit_ts: Timestamp | None)
              # owner: coordinator; mutable only via FSM transition; persisted in TwoPcLog
TxnState: enum {PENDING, PREPARED, COMMITTED, ABORTED, IN_DOUBT}  # I13-guarded transitions
ReadSet / WriteSet: frozenset[(ShardId, key)]  # immutable once transaction enters PREPARED
Participant: (shard_id: ShardId, group: ConsensusGroup, store: MvccStore)
              # owner: shard; concurrency: single writer per key via LockManager
ShardId: str                                  # stable identifier, static per ADR-13
Coordinator: (coordinator_id: str, epoch: int, log: TwoPcLog)
              # epoch fenced per I8; concurrency: one active epoch enforced by fencing token
PrepareRequest: (txn_id: TxnId, shard_id: ShardId, writes: WriteSet, coordinator_epoch: int)
Vote: enum {COMMIT, ABORT}                    # durable before returned, per I9
CommitDecision: (txn_id: TxnId, commit_ts: Timestamp, participants: frozenset[ShardId])
              # append-only once written, per I12; serialized via protobuf, versioned per ADR-16
TwoPcLog: append-only checksummed record store  # persistence: fsync before ack, I2/I6/I12
ConsensusGroup: (members: tuple[Replica, ...], leader: Replica | None, log_index: int)
              # compatibility: consensus log format versioned independently, §27
Replica: (node_id: str, role: enum{LEADER, FOLLOWER})
MvccStore: keyed version chains; concurrency: append-only per key, GC per retention window
Version: (key: bytes, commit_ts: Timestamp, value: bytes, tombstone: bool)  # immutable
SnapshotRead: (read_ts: Timestamp, keys: frozenset[bytes])  # no lock acquisition, ADR-11
Lock: (key: bytes, owner: TxnId, mode: LockMode, acquired_at: Timestamp)
LockMode: enum {SHARED, EXCLUSIVE}
WoundWait: comparator(older: Transaction, younger: Transaction) -> bool  # I10
DeadlockResolver: uses WoundWait; no separate detection round-trip, ADR-10
InDoubt: (txn_id: TxnId, last_known_state: TxnState, since: Timestamp)  # resolved via RecoveryManager
Recovery: replays TwoPcLog; concurrency: single recovery pass per coordinator restart, I6
ClockMonitor: samples TT.now() width; trips FailSafe on epsilon violation, I11
// Error taxonomy (all inherit TxnCoordError)
TxnAbortedError, PrepareTimeoutError, QuorumLostError, EpsilonViolationError,
InDoubtUnresolvedError, DeadlockDetectedError, StaleSnapshotReadError,
IllegalStateTransitionError, LogChecksumError, ProtocolVersionMismatchError
