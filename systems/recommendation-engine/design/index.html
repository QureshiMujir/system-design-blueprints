Core schema & types (§11)

type UserId(int)                      # validated >= -1; -1 = anonymous sentinel; immutable
type ItemId(int)                      # validated >= -1; -1 = "empty slot" sentinel; immutable
  # ownership: value type, copied freely; concurrency: immutable, no locking needed
  # serialization: plain int64 on the wire; compatibility: widening int32->int64 only
enum InteractionKind = view | click | play | complete | like | share | purchase | skip | negative

record InteractionEvent:                # frozen, one per raw event
  user: UserId
  item: ItemId
  ts: datetime                          # invariant I1: UTC, tz-aware
  kind: InteractionKind
  weight: float                         # invariant: >= 0
  position: Optional[int]               # invariant: >= 0 if set; slot at serve time
  trace_id: Optional[str]               # links to impression log (§21.4)
  model_version: Optional[str]
  # ownership: immutable value; persistence: parquet row (§21.1); no in-place mutation
record InteractionMatrix:               # frozen; §8.1 layout
  indptr: int64[n_users + 1]
  indices: int32[nnz]
  data: float32[nnz]
  n_users: int
  n_items: int
  # invariants (I3): indptr[0] == 0; indptr[-1] == len(indices); max(indices) < n_items
derived  row(u: UserId) -> (int32[], float32[])       # O(deg(u)), read-only slice
derived  transpose() -> InteractionMatrix              # O(nnz), counting-sort rebuild
derived  nnz -> int ; density -> float
  # ownership: immutable after construction; concurrency: read-shared across threads, no mutation
  # persistence: not serialized directly; rebuilt from parquet on every training run
record LatentFactors:                   # frozen; output of every MF trainer (§14-§16)
  U: float32[n_users, rank]
  V: float32[n_items, rank]
  bu: Optional[float32[n_users]]        # explicit-ALS bias term only (§14)
  bi: Optional[float32[n_items]]
  global_mean: float
derived  score(u: UserId, i: ItemId) -> float          # U[u]·V[i] + bu[u] + bi[i] + global_mean
derived  score_batch(u: UserId, items: ItemId[]) -> float32[]   # O(K*rank)
  # invariant: U.shape[1] == V.shape[1] == rank; dtype float32 throughout
  # ownership: immutable; part of ModelArtifact (§11.11); mmap-loadable at serve time
record AlsConfig:                        # frozen
  rank: int = 64                          # 0 < rank <= 4096
  reg: float = 0.01                       # invariant I5: reg > 0
  iters: int = 15                         # > 0
  alpha: float = 40.0                     # > 0; confidence scale (§8.2)
  seed: int = 42
  nthreads: int = 0                       # 0 = all cores
  # ownership: value type, part of TrainMeta.hyperparameters for provenance
interface AlsSolver:
  train(matrix: InteractionMatrix, matrix_t: InteractionMatrix, config: AlsConfig)
      -> LatentFactors | raises AlsError
  objective(factors: LatentFactors, matrix: InteractionMatrix, config: AlsConfig) -> float64

enum AlsError = ShapeMismatch(users, items, users_t, items_t)
              | NonPositiveDefinite(row: int)
              | Overflow(iter: int)

impl CholeskyAlsSolver : AlsSolver        # Phase 3 default, explicit variant in Phase 2
impl ConjugateGradientAlsSolver(cg_steps: int) : AlsSolver   # deferred, §35 row 2
record Candidate:                        # frozen; one per (item, source) from candidate gen
  item: ItemId
  score: float                            # raw retriever score; NOT cross-source comparable (P7)
  source: als | popularity | content | neighborhood | coldstart
  rank_within_source: int = 0
  # invariant I9: deduplicated by item, first-source kept, before reaching the ranker
record RankFeatures:                     # the maximal feature union across phases 8/20
  u_factor: float32[rank]
  i_factor: float32[rank]
  dot_uv: float
  u_age_days: float
  i_age_days: float
  i_pop_score: float                     # log(impressions) over trailing 7d
  hour_of_day_sincos: (float, float)
  dow_sincos: (float, float)
  source_onehot: float32[n_sources]
  rank_within_source: int
  retriever_score: float
derived  to_flat(schema: FeatureSchema) -> float32[]   # fixed field order (P12 explainability)
interface Ranker:
  score(X: float32[n_candidates, n_features]) -> float32[n_candidates]      # calibrated (I10)
  predict_multi(X) -> float32[n_candidates, n_tasks]                        # multi-task variant

interface Reranker:
  rerank(candidates: Candidate[], scores: float32[], context: RerankContext, k: int = 10)
      -> Candidate[]
  # Reranker is a pure function of its inputs; no persisted state beyond construction config
record TrainMeta:                        # frozen; every artifact carries exactly one
  model_version: str                     # semver + git sha
  trained_at: datetime
  data_version: str                      # parquet snapshot hash
  train_window: (datetime, datetime)
  eval_window: (datetime, datetime)
  hyperparameters: map[str, any]
  git_sha: str
  artifact_hash: str                     # sha256 of the tarball (I12)

record ModelArtifact:                    # frozen; the deployable unit
  factors: LatentFactors
  ranker: Ranker
  calibration: Optional[IsotonicCalibration]
  train_meta: TrainMeta
  # disk layout: tarball, one file per field, per §21.2; loaded via mmap at serve boot
record EvalReport:                        # frozen; one shape for every eval variant
  variant: offline_timesplit | replay_uniform | replay_ips | online_ab
  rmse: Optional[(float, float, float)]           # (mean, lo95, hi95); explicit-path only
  ndcg_at_k: map[int, (float, float, float)]
  recall_at_k: map[int, (float, float, float)]
  precision_at_k: map[int, (float, float, float)]
  map_at_k: map[int, (float, float, float)]
  coverage: float                                  # fraction of catalog ever recommended
  diversity_intra_list: (float, float, float)
  n_users_evaluated: int
  eval_window: (datetime, datetime)
  # every metric bootstrap-resampled (1000 resamples) per P4; variance across users is nontrivial
record ExperimentAssignment:              # frozen; one per (user_id, experiment_id)
  user_id: UserId
  experiment_id: str
  model_version: str
  assigned_at: datetime
  bucket: int                             # 0-99, from deterministic hash (I11, ADR-23)
