Core schema & types (§11)

struct ArtifactRecord — fields:
  artifact_id:       ArtifactId;      // owner: metadata-catalog; immutable after insert
  tenant_id:         TenantId;        // owner: metadata-catalog; immutable
  content_hash:      Blake3Hash;      // owner: ingest-router; immutable (I1)
  modality:          Modality;        // immutable
  object_uri:        ObjectUri;       // owner: object-storage-manager; immutable
  schema_version:    SchemaVersion;   // immutable; governs additive-evolution checks (P7)
  created_at:        MonotonicTime;   // immutable
  lineage_root:      ArtifactId;      // immutable; self-referential for root artifacts
  retention_policy:  RetentionPolicy; // mutable ONLY via append-only policy-change record
// concurrency: optimistic version check (compare-and-swap on schema_version + updated_at)
// persistence: WAL-before-ACK (I5) in metadata-catalog PostgreSQL instance
// serialization: protobuf, additive-only (ADR-06); compatible across N and N+1 minor versions
struct EmbeddingRecord — fields:
  embedding_id:      EmbeddingId;     // owner: embedding-runtime; immutable
  artifact_id:       ArtifactId;      // immutable foreign key into ArtifactRecord
  model_version:     ModelVersion;    // immutable; append-only reference set (I2)
  vector_dimensions: u32;             // immutable per model_version
  checksum:          Blake3Hash;      // immutable; verifies embedding integrity post-transfer
  created_at:        MonotonicTime;   // immutable
// concurrency: single-writer (embedding-runtime); no update path, only insert
// persistence: WAL-before-ACK; snapshot-replayable (ADR-19)
// serialization: protobuf; vector payload transferred as a separate Arrow IPC buffer (§5 Arrow ref)
struct SchedulerTask — fields:
  task_id:           TaskId;          // owner: scheduler-service; immutable
  workflow_id:       WorkflowId;      // immutable
  priority:          Priority;        // mutable via explicit re-prioritization event, append-only log
  retry_count:       u32;             // mutable, monotonic increase only
  lease_owner:       Option<WorkerId>;// mutable under lease protocol (I9 epoch-gated)
  timeout_deadline:  MonotonicTime;   // mutable on lease renewal only
// concurrency: lease-based single-owner mutation; contested writes rejected by epoch check (I9)
// persistence: append-only state log (I3); no in-place row mutation, only new transition rows
// serialization: internal-only; never crosses a service boundary as wire format
struct VectorPartition — fields:
  partition_id:      PartitionId;     // owner: vector-index-service; immutable
  tenant_id:         TenantId;        // immutable
  replica_set:       NodeId[3];       // mutable only via replication-coordinator reassignment (I4)
  compaction_epoch:  u64;             // monotonic increase only (I12 gate)
  vector_count:      u64;             // mutable; updated per successful insert/compaction
// concurrency: quorum write (2-of-3); read-repair reconciles divergent replicas
// persistence: snapshot to object storage per compaction_epoch (ADR-19)
// serialization: internal binary layout + Arrow IPC for bulk snapshot export
trait EmbeddingProvider — methods:
  fn enqueue(request: EmbeddingRequest) -> Result<TicketId, EnqueueError>;
  fn batch_execute(batch: EmbeddingBatch) -> Result<BatchResult, ExecutionError>;
  fn persist(result: BatchResult) -> Result<(), PersistError>;
// ownership: implemented by embedding-runtime; no other service implements this trait
// concurrency: batch_execute is the only method touching GPU state; single-writer per GPU device
trait VectorIndex — methods:
  fn insert(partition: PartitionId, vectors: VectorBatch) -> Result<(), IndexError>;
  fn search(partition: PartitionId, query: QueryVector, k: u32) -> Result<Vec<Hit>, SearchError>;
  fn compact(partition: PartitionId, epoch: u64) -> Result<CompactionReport, CompactionError>;
// ownership: implemented by vector-index-service only
// concurrency: insert/search concurrent-safe (RwLock-per-partition); compact exclusive per partition
