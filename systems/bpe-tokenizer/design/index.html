Core schema & types (§11)

#[repr(transparent)]
pub struct TokenId(pub u32);                    // Copy, Eq, Ord, Hash
impl TokenId {
    pub const BYTE_RANGE: Range<u32> = 0..256;   // I1
    pub const fn from_byte(b: u8) -> Self;
    pub fn is_byte(self) -> bool;
    pub fn as_u32(self) -> u32;
}
pub const BYTE_ALPHABET_SIZE: u32 = 256;
pub const DEFAULT_TARGET_VOCAB: u32 = 32_000;
pub const MAX_TARGET_VOCAB: u32 = 1_000_000;
pub struct Vocab {
    bytes: Vec<Vec<u8>>,                         // id -> bytes
    by_bytes: HashMap<Vec<u8>, TokenId>,          // bytes -> id
    specials: BTreeMap<TokenId, SpecialToken>,
}
impl Vocab {
    pub fn new_byte_alphabet() -> Self;           // sole path that inserts ids < 256 (I1)
    pub fn len(&self) -> usize;
    pub fn bytes_of(&self, id: TokenId) -> Option<&[u8]>;
    pub fn id_of(&self, bytes: &[u8]) -> Option<TokenId>;
    pub fn push_merged(&mut self, left: TokenId, right: TokenId) -> TokenId;
    pub fn add_special(&mut self, tok: SpecialToken) -> TokenId;
    pub fn insert_at(&mut self, id: TokenId, bytes: Vec<u8>) -> Result<()>;
}
pub struct MergeTable {
    pairs: HashMap<(TokenId, TokenId), MergeEntry>,
    ordered: Vec<Merge>,                          // by rank
}
pub struct MergeEntry { pub merged_id: TokenId, pub rank: u32 }
pub struct Merge { pub left: TokenId, pub right: TokenId, pub merged_id: TokenId, pub rank: u32 }
impl MergeTable {
    pub fn new() -> Self;
    pub fn lookup(&self, left: TokenId, right: TokenId) -> Option<MergeEntry>;
    pub fn push(&mut self, left: TokenId, right: TokenId, merged_id: TokenId) -> u32; // rank; panics on dup (I13)
    pub fn len(&self) -> usize;
}
pub struct SpecialToken { pub id: TokenId, pub bytes: Vec<u8>, pub kind: SpecialKind }
pub enum SpecialKind {
    Regular,    // always matched (e.g. <|endoftext|>)
    Optional,   // matched only if caller's AllowedSpecial includes it
}
pub trait Pretokenizer: Send + Sync {
    fn pre_tokenize<'a>(&self, input: &'a str) -> Box<dyn Iterator<Item=(usize,usize)> + 'a>;
    fn kind(&self) -> PretokKind;
}
pub enum PretokKind { Gpt2, Llama3, Whitespace, Uax29, Custom }
pub struct BpeEncoder {
    vocab: Arc<Vocab>, merges: Arc<MergeTable>,
    pretok: Arc<dyn Pretokenizer>, specials: Arc<SpecialTrie>,
}
impl BpeEncoder {
    pub fn new(vocab: Arc<Vocab>, merges: Arc<MergeTable>,
               pretok: Arc<dyn Pretokenizer>, specials: Arc<SpecialTrie>) -> Self;
    pub fn encode(&self, input: &str) -> Vec<TokenId>;
    pub fn encode_into(&self, input: &str, out: &mut Vec<TokenId>);
    pub fn encode_with_dropout(&self, input: &str, p: f64, seed: u64) -> Vec<TokenId>;
    pub fn encode_ordinary(&self, input: &str) -> Vec<TokenId>;
    pub fn encode_with_allowed(&self, input: &str, allowed: &AllowedSpecial) -> Vec<TokenId>;
}
pub struct EncoderScratch { node_arena: Vec<Node>, queue: PairQueue, byte_buf: Vec<u8> }
pub struct TrainerConfig {
    pub target_vocab: u32, pub min_pair_count: u32, pub max_pair_per_iter: u32,
    pub pretok: PretokKind, pub specials: Vec<Vec<u8>>,
    pub corpus: CorpusInput, pub n_threads: usize,
}
pub enum CorpusInput { Inline(Vec<String>), File(PathBuf), Stream(Box<dyn BufRead+Send>), Sharded(Vec<PathBuf>) }
pub struct BpeTrainer { cfg: TrainerConfig }
impl BpeTrainer {
    pub fn new(cfg: TrainerConfig) -> Self;
    pub fn train(self, progress: impl Fn(TrainerProgress)) -> Result<TrainedTokenizer>;
}
pub struct TrainedTokenizer { pub vocab: Vocab, pub merges: MergeTable, pub pretok: PretokKind, pub specials: Vec<SpecialToken> }
pub struct SpecialTrie { nodes: Vec<Node> }        // Node { children: [u16;256], accept: Option<TokenId> }
impl SpecialTrie {
    pub fn from_specials(specials: &[SpecialToken]) -> Self;
    pub fn scan<'a>(&'a self, bytes: &'a [u8]) -> impl Iterator<Item=SpecialSpan> + 'a;
}
pub struct SpecialSpan { pub start: usize, pub end: usize, pub token_id: TokenId }
pub enum Error {
    MalformedLine { line: u32 }, UnknownToken, UnknownMerged,
    DuplicateMergePair, DisallowedSpecial { bytes: Vec<u8> }, Io(std::io::Error),
    Utf8(std::str::Utf8Error), Json(serde_json::Error),
}
