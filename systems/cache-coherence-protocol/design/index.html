Core schema & types (§11)

// coherence.core -- primitive identifiers
CoreId = NewType("CoreId", int)                    // ownership: value type; immutable
Address = NewType("Address", int)                   // ownership: value type; immutable
LineId = NewType("LineId", int)                       // Address >> log2(line_size); immutable

@dataclass(frozen=True)
class CacheLine:                                        // mutation: replaced, never mutated in place
    line_id: LineId
    state: "LineState"
    data: bytes                                            // serialization: raw bytes, versioned (§27)
    dirty: bool
    last_writer: CoreId | None
class LineState(Enum):                                   // concurrency: pure value, no shared mutation
    MODIFIED = "M"; OWNED = "O"; EXCLUSIVE = "E"
    SHARED = "S"; INVALID = "I"; FORWARD = "F"
    IS_D = "IS_D"; IM_D = "IM_D"; SM_A = "SM_A"           // transient states (§8)
    MI_A = "MI_A"; OI_A = "OI_A"; SI_A = "SI_A"

class CacheSet:                                            // ownership: owned by one Cache
    ways: list[CacheLine]                                     // mutation: controller-only
    def victim(self, policy: "ReplacementPolicy") -> CacheLine: ...

class ReplacementPolicy(Protocol):                            // stateless strategy object
    def choose_victim(self, ways: list[CacheLine]) -> CacheLine: ...

class Cache:                                                    // ownership: exactly one per CoreId
    core_id: CoreId
    sets: dict[int, CacheSet]
    def lookup(self, addr: Address) -> CacheLine | None: ...
class ProcessorRequest(Enum):                                     // driver-emitted, immutable
    LOAD = "Load"; STORE = "Store"; EVICT = "Evict"

@dataclass(frozen=True)
class BusTransaction:                                                // serialization: JSON-lines log record (ADR-19)
    txn_id: int; source: CoreId; line_id: LineId
    kind: "BusTxnKind"; cycle: int

class BusTxnKind(Enum):
    BUS_RD = "BusRd"; BUS_RDX = "BusRdX"
    BUS_UPGR = "BusUpgr"; BUS_WB = "BusWB"
    C2C_TRANSFER = "C2CTransfer"

@dataclass(frozen=True)
class Snoop:                                                          // observed by every other CacheController
    txn: BusTransaction; observer: CoreId
class Interconnect(Protocol):                                          // trait; two impls, mutually exclusive (ADR-04)
    def submit(self, txn: BusTransaction) -> None: ...
    def arbitrate(self) -> BusTransaction | None: ...

class SnoopingBus(Interconnect):                                          // concurrency: single arbitration slot per tick
    arbitration_queue: list[BusTransaction]

class Directory(Interconnect):                                              // ownership: one DirEntry per LineId
    entries: dict[LineId, "DirEntry"]

@dataclass
class DirEntry:                                                                // mutation: directory-controller-only
    line_id: LineId; state: "DirState"; sharers: "SharerSet"; owner: CoreId | None

SharerSet = NewType("SharerSet", frozenset[CoreId])                            // persistence: bitmap-serializable

class DirState(Enum):
    UNCACHED = "Uncached"; SHARED = "Shared"; MODIFIED = "Modified"
    BUSY_INV = "Busy-Inv"; BUSY_FWD = "Busy-Fwd"; BUSY_OWNXFER = "Busy-OwnXfer"
class ProtocolTrait(Protocol):                                                    // compatibility: versioned per §27
    name: str
    def transition(self, state: LineState, event: "Transition") -> tuple[LineState, list[BusTxnKind]]: ...

@dataclass(frozen=True)
class Transition:                                                                    // hashable, used as table key
    state: LineState; event: str                                                        // event e.g. "PrRd", "BusRdX"

TransitionTable = dict[Transition, tuple[LineState, list[BusTxnKind]]]                    // I8: completeness-checked at load

class Memory:                                                                                // ownership: single authoritative store
    backing: dict[Address, bytes]
    def read(self, addr: Address) -> bytes: ...
    def write(self, addr: Address, data: bytes) -> None: ...
class ConsistencyModel(Protocol):                                                              // ADR-09
    def admit(self, req: ProcessorRequest, core: CoreId) -> bool: ...

class MemoryOrder(Enum):
    SC = "SequentialConsistency"; TSO = "x86-TSO"

class StoreBuffer:                                                                                 // ownership: one per core under TSO
    core_id: CoreId; pending: list[tuple[Address, bytes]]                                              // FIFO, mutation: append/drain only

@dataclass(frozen=True)
class LitmusTest:                                                                                        // immutable fixture (§8, §28)
    name: str; initial: dict[Address, bytes]; programs: dict[CoreId, list[ProcessorRequest]]
    forbidden_under_sc: list[dict[str, int]]
class InvariantChecker:                                                                                    // I1-I13; concurrency: read-only observer
    def on_transition(self, core: CoreId, line: LineId, old: LineState, new: LineState) -> None: ...
    def on_read(self, core: CoreId, addr: Address, value: bytes) -> None: ...
    violations: dict[str, int]                                                                                // keyed by invariant ID (I1..I13)

@dataclass(frozen=True)
class TraceEvent:                                                                                              // serialization: versioned trace format (§27)
    cycle: int; core: CoreId; req: ProcessorRequest; addr: Address

class Statistics:                                                                                                // ownership: one per simulation run
    txn_counts: dict[BusTxnKind, int]
    miss_class: dict[str, int]                                                                                     // cold/capacity/conflict/coherence
    false_sharing_incidents: int

class CoherenceError(Exception): ...                                                                                 // error taxonomy root
class UndefinedTransitionError(CoherenceError): ...                                                                    // I8
class InvariantViolationError(CoherenceError): ...                                                                       // I1-I7, I12
class DirectoryDesyncError(CoherenceError): ...                                                                            // I5
class DeterminismError(CoherenceError): ...                                                                                  // I10
class MalformedTraceError(CoherenceError): ...                                                                                // driver input
