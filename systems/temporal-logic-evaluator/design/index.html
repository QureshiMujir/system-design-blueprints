Core schema & types (§11)

/// Atomic-proposition id, interned by tl-syntax (a dense u32, 0..n_aps).
/// Identity only: the string spelling lives in the parser's symbol table.
pub type Ap = u32;

/// The Formula AST: one enum spanning LTL (future), past-LTL, and CTL.
/// Owns its children via Box (tree, never a DAG; structurally shared only via Clone).
/// Immutable after construction. Derives Clone, PartialEq, Eq, Hash, Debug.
/// LTL = boolean + future ops; PastLtl adds Y/O/H/S; CTL = fused-quantifier ops.
pub enum Formula {
    True, False, Atom(Ap),
    Not(Box<Formula>), And(Box<Formula>, Box<Formula>), Or(Box<Formula>, Box<Formula>),
    // future-time LTL:
    X(Box<Formula>),                              // next
    U(Box<Formula>, Box<Formula>),                // until
    R(Box<Formula>, Box<Formula>),                // release (dual of U)
    F(Box<Formula>),                              // eventually  (= True U f)
    G(Box<Formula>),                              // globally    (= False R f)
    // past-time LTL:
    Y(Box<Formula>),                              // previous (yesterday)
    O(Box<Formula>),                              // once
    H(Box<Formula>),                              // historically
    S(Box<Formula>, Box<Formula>),                // since
    // CTL (path quantifier fused with one temporal op; arg is a STATE formula):
    Ex(Box<Formula>), Ax(Box<Formula>),           // EX / AX
    Eg(Box<Formula>), Ag(Box<Formula>),           // EG / AG
    Ef(Box<Formula>), Af(Box<Formula>),           // EF / AF
    Eu(Box<Formula>, Box<Formula>),               // E[a U b]
    Au(Box<Formula>, Box<Formula>),               // A[a U b]
}

/// Negation normal form: negation pushed to atoms only, fixed dual operator set.
/// Pure; returns a fresh tree (input untouched). Idempotent: nnf(nnf(f)) == nnf(f).
/// GPVW (## 18) + the labeling algorithm (## 17) ASSUME their input is in NNF.
pub fn nnf(f: &Formula) -> Formula;

/// Subformula closure: f and all its subformulas (and the duals GPVW needs),
/// deduplicated, in a deterministic order. Drives GPVW node sets + CTL labeling.
/// Size |closure(f)| is linear in the size of f (each node contributes O(1) entries).
pub fn closure(f: &Formula) -> Vec<Formula>;

/// Well-formedness classification of a parsed formula.
/// Ltl = boolean+future only; Ctl = every temporal op carries a path quantifier
/// and every quantified arg is itself a state formula; PastLtl = future+past;
/// Ltlf = an Ltl formula intended for finite-trace semantics.
pub enum Kind { Ltl, Ctl, PastLtl, Ltlf }
/// A single observed state: which atomic propositions hold here.
/// Bitset indexed by Ap; bit i set <=> AP i is true. Fixed-width per run.
pub struct State { pub aps: Bitset }              // Bitset = u64 words; |Ap| up to a few thousand

/// A trace = a finite prefix of States, optionally a lasso (infinite word).
/// loop_start = Some(i) => the suffix prefix[i..] repeats forever (omega-word);
/// loop_start = None     => a genuinely finite word (LTLf / monitoring prefix).
/// Immutable after construction.
pub struct Trace { pub prefix: Vec<State>, pub loop_start: Option<usize> }

/// An explicit-state Kripke structure: the model for CTL + automata-theoretic LTL.
/// states[i].aps = labeling of state i; init = initial state indices;
/// edges[i] = successors of state i (the transition relation).
/// Immutable after construction; validated for a TOTAL transition relation.
pub struct KripkeStructure {
    pub states: Vec<State>,
    pub init: Vec<usize>,
    pub edges: Vec<Vec<usize>>,                   // successors; edges[i] ⊆ 0..states.len()
}
/// A boolean condition over APs guarding an automaton edge.
/// Stored as a small DNF / bitset (mask = APs that matter, care = required polarity);
/// a State satisfies a Guard iff (state.aps & mask) == (care & mask). Immutable.
pub struct Guard(/* mask+care bitset, or small DNF of literals */);

/// Generalized Buchi automaton (GPVW output, ## 18): a SET of accepting sets.
/// nstates indices 0..nstates; edges[s] = labeled transitions; init = start states;
/// accept = one Vec<usize> per generalized acceptance condition (one per U-subformula).
/// Acceptance: a run is accepting iff it visits EACH accept[k] infinitely often.
/// Immutable after construction.
pub struct Ngba {
    pub nstates: usize,
    pub init: Vec<usize>,
    pub edges: Vec<Vec<(Guard, usize)>>,
    pub accept: Vec<Vec<usize>>,                  // accept.len() = number of acceptance sets
}

/// Buchi automaton (degeneralized, ## 19): ONE accepting set.
/// Acceptance: a run is accepting iff it visits accept infinitely often.
/// Immutable after construction.
pub struct Nba {
    pub nstates: usize,
    pub init: Vec<usize>,
    pub edges: Vec<Vec<(Guard, usize)>>,
    pub accept: Vec<usize>,
}
/// GPVW on-the-fly tableau: LTL Formula (in NNF) -> generalized Buchi automaton.
/// Nodes = obligation sets over closure(f), split now/next; one accept set per Until.
/// Built lazily; deterministic node identity -> stable nstates.
pub fn ltl_to_ngba(f: &Formula) -> Ngba;

/// Degeneralization (counter construction): Ngba with k accept sets ->
/// Nba on (state x counter in 0..k). |Nba.nstates| ≤ k * |Ngba.nstates|.
pub fn degeneralize(g: &Ngba) -> Nba;

/// Synchronous product Nba x KripkeStructure -> Nba over (auto-state x kripke-state).
/// An edge exists iff the automaton Guard holds in the Kripke state's labeling and
/// the Kripke edge is present. Accept = product states whose automaton component accepts.
pub fn product(a: &Nba, k: &KripkeStructure) -> Nba;

/// Emptiness via NESTED DFS: Some(lasso) = a reachable accepting cycle exists
/// (an accepting run / counterexample); None = language is empty.
/// The DFS run is the only MUTABLE artifact here (two color arrays + two stacks).
pub fn emptiness(a: &Nba) -> Option<Lasso>;

/// A counterexample run: stem leads into the cycle; cycle repeats forever.
/// Indices refer to the automaton `a` passed to emptiness (product states for a model check).
pub struct Lasso { pub stem: Vec<usize>, pub cycle: Vec<usize> }
/// Explicit-state CTL model checker. Borrows the model read-only for its lifetime.
/// Construct once per KripkeStructure; reuse across many formulas.
pub struct CtlChecker<'a> { /* &'a KripkeStructure */ }

impl CtlChecker<'_> {
    /// Per-state truth labeling: result[i] = does state i satisfy f.
    /// EX = preimage; E[U] = least fixpoint; EG = greatest fixpoint via SCC quotient;
    /// AX/AG/AF/EF/A[U] derived by duality. Bottom-up over subformulas. Read-only.
    pub fn sat(&self, f: &Formula) -> Vec<bool>;

    /// holds iff EVERY init state is labeled by sat(f).
    pub fn holds(&self, f: &Formula) -> bool;
}

/// Finite-trace LTL (LTLf) by formula progression over a finite Trace.
/// Folds progress() across prefix; applies the end-of-trace finite semantics
/// (weak/strong next) to yield a DEFINITIVE bool. Pure.
pub fn eval_ltlf(f: &Formula, t: &Trace) -> bool;

/// Automata-theoretic LTL model check over a KripkeStructure.
/// Ok(())   = K satisfies (A f)  [product(Nba(nnf(Not f)), K) is empty];
/// Err(cex) = a counterexample lasso (an infinite run of K violating f).
/// Internally: nnf -> ltl_to_ngba(Not f) -> degeneralize -> product -> emptiness.
pub fn check_ltl(f: &Formula, k: &KripkeStructure) -> Result<(), Lasso>;
/// 2-valued definitive verdict (a settled monitor outcome).
pub enum Verdict { True, False }

/// RV-LTL / LTL3 verdict over a finite observed prefix.
/// LTL3 uses {True, False, Unknown}; RV-LTL refines Unknown into
/// PresumablyTrue / PresumablyFalse via a finite-semantics tie-break.
pub enum RvVerdict { True, False, PresumablyTrue, PresumablyFalse, Unknown }

/// A streaming monitor: the only MUTABLE core type.
/// State = the current progressed Formula (the "residual") OR a monitor-automaton
/// state. Owns its evolving state; single-stream owner (Send, not shared).
pub struct Monitor { /* current progressed Formula or monitor-automaton state */ }

impl Monitor {
    /// Consume one event (State); advance internal state; return the current verdict.
    /// A residual of True/False => definitive (True/False); else Presumably*/Unknown.
    pub fn step(&mut self, s: &State) -> RvVerdict;

    /// The LTL "derivative": one progression step of f under observation s.
    /// progress(X f, s)=f; progress(f U g, s)=progress(g,s) ∨ (progress(f,s) ∧ (f U g));
    /// progress(G f, s)=progress(f,s) ∧ G f; progress(Atom p, s)=True/False by s. Pure.
    pub fn progress(f: &Formula, s: &State) -> Formula;
}
tl-formula:  Ap, Formula, nnf, closure, Kind          (immutable AST; Send+Sync)
   |  &Formula
   +-> tl-automata: Guard, Ngba, Nba, Lasso,
   |                ltl_to_ngba -> degeneralize -> product -> emptiness   (immutable graphs)
   +-> tl-eval:     CtlChecker(&KripkeStructure).sat/holds,
   |                eval_ltlf(&Trace), check_ltl -> Result<(), Lasso>     (read-only)
   +-> tl-monitor:  Verdict, RvVerdict, Monitor.step(&mut), progress      (Monitor mutates)
tl-trace:    State(Bitset), Trace(prefix+loop_start), KripkeStructure     (immutable; Send+Sync)
