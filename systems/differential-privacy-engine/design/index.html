#[derive(Clone, Copy, PartialEq, PartialOrd, Debug, Serialize, Deserialize)]
pub struct Epsilon(f64);

impl Epsilon {
    pub fn new(v: f64) -> Result<Self, DpError>;   // rejects NaN, Inf, <= 0
    pub fn get(self) -> f64;
    pub fn add(self, rhs: Epsilon) -> Epsilon;      // basic composition: eps_total = sum
    pub fn scale(self, k: f64) -> Epsilon;          // advanced composition helper
}
#[derive(Clone, Copy, PartialEq, PartialOrd, Debug, Serialize, Deserialize)]
pub struct Delta(f64);

impl Delta {
    pub fn new(v: f64) -> Result<Self, DpError>;   // rejects NaN, v < 0, v >= 1
    pub fn zero() -> Self;                          // pure DP: delta = 0
    pub fn get(self) -> f64;
    pub fn add(self, rhs: Delta) -> Result<Delta, DpError>; // sum; error if >= 1
}
#[derive(Clone, Copy, Debug, Serialize, Deserialize)]
pub struct PrivacyParams {
    pub epsilon: Epsilon,
    pub delta:   Delta,
}

impl PrivacyParams {
    pub fn pure_dp(epsilon: Epsilon) -> Self;                        // delta = 0
    pub fn approx_dp(epsilon: Epsilon, delta: Delta) -> Self;
    pub fn dominates(self, other: PrivacyParams) -> bool;            // self <= other component-wise
    pub fn basic_compose(a: PrivacyParams, b: PrivacyParams) -> PrivacyParams;
}
#[derive(Clone, Copy, Debug, Serialize, Deserialize)]
pub struct Sensitivity {
    pub l1: f64,   // L1 global sensitivity; >= 0
    pub l2: f64,   // L2 global sensitivity; >= 0; l2 <= l1 always
}

impl Sensitivity {
    pub fn new(l1: f64, l2: f64) -> Result<Self, DpError>;  // rejects NaN, Inf, < 0, l2 > l1
    pub fn scale(self, factor: f64) -> Sensitivity;          // transform stability: multiply both
    pub fn combine_l1(a: Self, b: Self) -> Self;             // sequential composition of transforms
}
#[derive(Clone, Copy, Debug, Serialize, Deserialize)]
pub struct Bounds {
    pub lower: f64,
    pub upper: f64,
}

impl Bounds {
    pub fn new(lower: f64, upper: f64) -> Result<Self, DpError>; // rejects NaN, lower >= upper
    pub fn clamp(self, v: f64) -> f64;                           // f64::clamp(v, lower, upper)
    pub fn range(self) -> f64;                                   // upper - lower; >= 0
    pub fn sensitivity_l1(self) -> f64;                          // = range(); sum sensitivity
}
#[derive(Debug, Serialize, Deserialize)]
pub struct DpResult<T> {
    pub value:  T,
    pub params: PrivacyParams,
    pub sensitivity: Sensitivity,
}

impl<T> DpResult<T> {
    pub fn new(value: T, params: PrivacyParams, sensitivity: Sensitivity) -> Self;
    pub fn map<U, F: FnOnce(T) -> U>(self, f: F) -> DpResult<U>; // post-processing; params unchanged (I7)
}
#[derive(Debug, thiserror::Error)]
pub enum DpError {
    #[error("invalid epsilon: {0}")]            InvalidEpsilon(f64),
    #[error("invalid delta: {0}")]              InvalidDelta(f64),
    #[error("invalid sensitivity: l1={l1}, l2={l2}")] InvalidSensitivity { l1: f64, l2: f64 },
    #[error("invalid bounds: lower={lower}, upper={upper}")] InvalidBounds { lower: f64, upper: f64 },
    #[error("budget exhausted: requested={requested:?}, remaining={remaining:?}")]
        BudgetExhausted { requested: PrivacyParams, remaining: PrivacyParams },
    #[error("unbounded sensitivity: clamp bounds required")]  UnboundedSensitivity,
    #[error("delta required for Gaussian mechanism")]         DeltaRequired,
    #[error("invalid parameter: {0}")]          InvalidParameter(String),
    #[error("sampler error: {0}")]              SamplerError(String),
    #[error("accountant overflow")]             AccountantOverflow,
}
pub fn count_sensitivity() -> Sensitivity;               // always {l1:1, l2:1}
pub fn sum_sensitivity(bounds: Bounds) -> Sensitivity;   // l1 = l2 = bounds.range()
pub fn mean_sensitivity(bounds: Bounds, n: usize) -> Sensitivity; // l1 = range/n
pub fn transform_stability(s: Sensitivity, factor: f64) -> Result<Sensitivity, DpError>;

pub trait CsprngSource: Send {
    fn fill_bytes(&mut self, dest: &mut [u8]);
    fn next_u64(&mut self) -> u64;
}
pub trait Mechanism<T>: Send + Sync {
    fn privatize(&self, exact: T, sens: Sensitivity, rng: &mut impl CsprngSource)
        -> Result<DpResult<T>, DpError>;
    fn privacy_params(&self) -> PrivacyParams;
}
pub struct LaplaceMechanism { pub epsilon: Epsilon }

impl LaplaceMechanism {
    pub fn new(epsilon: Epsilon) -> Self;
    pub fn scale(&self, sens: Sensitivity) -> f64; // = sens.l1 / epsilon.get()
}
impl Mechanism<f64> for LaplaceMechanism { /* privatize signature only */ }
pub struct GaussianMechanism { pub epsilon: Epsilon, pub delta: Delta }

impl GaussianMechanism {
    pub fn new(epsilon: Epsilon, delta: Delta) -> Result<Self, DpError>; // error if delta=0
    pub fn sigma(&self, sens: Sensitivity) -> f64; // analytic Gaussian calibration (Balle-Wang)
}
impl Mechanism<f64> for GaussianMechanism { /* privatize signature only */ }
pub struct ExponentialMechanism<C> {

}

pub struct RdpAccountant {
    pub orders: Vec<f64>,   // alpha values; typically 1.5..512 log-spaced
    pub rdp:    Vec<f64>,   // cumulative Renyi loss at each order; same length as orders
}

impl RdpAccountant {
    pub fn new(orders: Vec<f64>) -> Result<Self, DpError>;
    pub fn add_mechanism(&mut self, mech_rdp: &[f64]) -> Result<(), DpError>; // adds per-order; I9
    pub fn to_privacy_params(&self, delta: Delta) -> Result<PrivacyParams, DpError>;
    pub fn gaussian_rdp(sigma: f64, l2: f64, order: f64) -> f64; // = l2^2 * order / (2 * sigma^2)
    pub fn subsampled_gaussian_rdp(sigma: f64, l2: f64, q: f64, order: f64) -> f64; // amplification
}
pub struct ZcdpAccountant { pub rho: f64 }

impl ZcdpAccountant {
    pub fn new() -> Self;
    pub fn add(&mut self, rho_increment: f64) -> Result<(), DpError>; // rho is additive; I9
    pub fn to_privacy_params(&self, delta: Delta) -> PrivacyParams;   // rho -> (eps,delta)
    pub fn gaussian_rho(sigma: f64, l2: f64) -> f64;                  // = l2^2 / (2*sigma^2)
}
pub struct MomentsAccountant {

}

pub struct Budget {
    pub total:   PrivacyParams,
    spent:       PrivacyLoss,         // private; only mutated via `spend`
    log:         Vec<BudgetEntry>,    // append-only audit log
}

pub struct BudgetEntry {
    pub timestamp:  std::time::SystemTime,
    pub cost:       PrivacyParams,
    pub mechanism:  String,
    pub remaining:  PrivacyParams,
}

pub struct PrivacyLoss {
    pub accountant: AccountantState, // RDP curve or (eps,delta) depending on mode
}

impl Budget {
    pub fn new(total: PrivacyParams) -> Self;
    pub fn check(&self, cost: PrivacyParams) -> Result<(), DpError>;  // read-only; I3
    pub fn spend(&mut self, cost: PrivacyParams, label: &str) -> Result<(), DpError>; // atomic; I3
    pub fn remaining(&self) -> PrivacyParams;
    pub fn audit_log(&self) -> &[BudgetEntry];
}
pub trait Query<D, T>: Send {
    fn sensitivity(&self, bounds: Bounds) -> Result<Sensitivity, DpError>;
    fn exact(&self, data: &D) -> Result<T, DpError>;   // no randomness; must be clamped
}

pub struct RandomizedResponse { pub p: f64 }   // p = prob of truthful response
