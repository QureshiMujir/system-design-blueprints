Core schema & types (§11)

type CapabilityId struct {
    Namespace  string
    Name       string
    Version    SemVer            // ADR-17
    Digest     [32]byte          // sha256 of manifest, content address
}
// ownership: registry-owned once PUBLISHED; immutable after publish (P10, I8)
// mutation: none post-publish; concurrency: read-mostly, safe for concurrent resolve
// persistence: registry index row; serialization: protobuf (ADR-02); compat: stable across all phases
type ArtifactManifest struct {
    CapabilityId        CapabilityId
    BaseModelIdentity    string
    ArchitectureFingerprint BaseModelFingerprint
    QuantizationScheme    QuantScheme        // Phase 8, §20
    TargetModules        TargetModuleSet
    Lora                 LoraSpec
    TokenizerHash        [32]byte
    TrainingProvenance    ProvenanceRecord
    EvalScores           map[string]float64
    CompositionHint      []CapabilityId      // advisory order, P3
}
// ownership: packager-authored, registry-validated; mutation: immutable post-sign (I1)
// concurrency: read-only after signing; persistence: embedded in artifact + index row
// serialization: protobuf; compatibility: additive-only field evolution (§27)
type BaseModelFingerprint struct {
    ArchitectureFamily string        // e.g. "llama", "mistral"
    HiddenSize         uint32
    NumLayers          uint32
    VocabSize          uint32
    QuantizationScheme QuantScheme
    CheckpointDigest   *[32]byte     // nil unless training provenance pins one, P7
}
// ownership: derived at packaging time from base-model metadata; mutation: immutable
// concurrency: value type, safe to share; persistence: manifest field + resolver index
// serialization: protobuf; compatibility: new fields must be optional (§27)
type TargetModuleSet struct {
    ModuleNames []string    // e.g. ["q_proj","k_proj","v_proj","o_proj"]
}
// ownership: packager-declared; mutation: immutable; concurrency: value type
// persistence: manifest field; serialization: protobuf; compatibility: subset-checked at resolve (I2)
type LoraSpec struct {
    Rank      uint32   // r
    Alpha     float32  // α
    Dtype     DType    // e.g. BF16, FP16, NF4 (Phase 8)
}
// ownership: packager-declared, cross-checked against tensor shapes (I5); mutation: immutable
// concurrency: value type; persistence: manifest field; serialization: protobuf
type TensorIndexEntry struct {
    Name    string
    Dtype   DType
    Shape   []uint32
    Offset  uint64
    Length  uint64
    Sha256  [32]byte
}
// ownership: packager-computed; mutation: immutable post-sign; concurrency: read-only array
// persistence: artifact tensor-index block (§8.1); serialization: protobuf; compat: append-only per artifact
type SignatureEnvelope struct {
    Algorithm     string      // e.g. "ed25519-sigstore"
    Signature     []byte
    CertChain     []byte      // Fulcio short-lived cert chain
    TransparencyLogEntry uint64  // Rekor log index
}
// ownership: signer-produced; mutation: immutable; concurrency: read-only
// persistence: artifact trailer (§8.1); serialization: protobuf; compat: algorithm field allows rotation
type CompatibilityVerdict struct {
    Result    VerdictResult   // ACCEPT | WARN_ACCEPT | REJECT
    Reasons   []string
    EvalScore float64
}
// ownership: resolver-produced, ephemeral; mutation: immutable once returned
// concurrency: value type; persistence: not persisted, logged only; serialization: protobuf over RPC
type ResolvedPlan struct {
    PlanHash        [32]byte
    CapabilityDigests []([32]byte)   // ordered, per I7
    CompositionPolicy CompositionPolicy
    BaseFingerprint  BaseModelFingerprint
}
// ownership: serving runtime-produced, cacheable (P9, I4); mutation: immutable
// concurrency: safe for concurrent read once cached; persistence: plan cache (in-memory + optional Redis)
// serialization: protobuf; compatibility: hash function version pinned per major release
type AdapterHandle struct {
    CapabilityDigest [32]byte
    MappedTensors    []TensorIndexEntry
    PoolSlot         uint32
    RefCount         uint32
}
// ownership: adapter loader-produced, serving-runtime-owned; mutation: RefCount mutates under lock (I10)
// concurrency: refcounted, protected by pool manager mutex; persistence: process-local only
type CompositionPolicy struct {
    Mode  CompositionMode   // STACK | MERGE | ROUTE
    Order []CapabilityId    // required for STACK/MERGE, ignored for ROUTE (I7)
    MergeParams *MergeParams // TIES/DARE params when Mode == MERGE
}
// ownership: request-declared; mutation: immutable per request; concurrency: value type
// persistence: embedded in ResolvedPlan; serialization: protobuf
type ServingRequest struct {
    Capabilities CompositionPolicy
    Prompt       []byte
    BaseModelRef string
}
// ownership: client-declared; mutation: immutable; concurrency: one per inference call
// persistence: not persisted beyond request trace; serialization: protobuf/gRPC
type EvalReport struct {
    CapabilityDigest [32]byte
    BenchmarkScores  map[string]float64
    ConfidenceInterval map[string][2]float64
    PassedThreshold  bool
}
// ownership: eval harness-produced; mutation: immutable once recorded; concurrency: append-only history
// persistence: registry index; serialization: protobuf; compat: threshold policy versioned separately
type DistillationJob struct {
    SourceCapability CapabilityId
    TargetFingerprint BaseModelFingerprint
    Status           DistillationStatus
    ResultCapability *CapabilityId   // set on completion, P10/I12
}
// ownership: distillation pipeline-owned; mutation: Status transitions forward-only
// concurrency: single-writer per job; persistence: registry index; serialization: protobuf
type FederatedRound struct {
    RoundId          uint64
    Participants     []string
    PrivacyBudget    *DPBudget       // nil only if SecureAggProof set (I11)
    SecureAggProof   *[]byte
    AggregatedDelta  *CapabilityId
}
// ownership: federated bridge-owned; mutation: append-only round history
// persistence: registry index + audit log; serialization: protobuf over mTLS gRPC (ADR-24)
enum SynapseError {
    ErrVerificationFailed
    ErrShapeMismatch
    ErrModuleNotSubset
    ErrTokenizerMismatch
    ErrEvalGateFailed
    ErrQuantizationIncompatible
    ErrPoolExhausted
    ErrProvenanceMissing
    ErrPrivacyBudgetMissing
    ErrCompositionOrderViolation
}
// ownership: shared error taxonomy across crates/services; mutation: append-only enum growth
// serialization: protobuf enum; compatibility: numeric tags stable across minor versions (§27)
