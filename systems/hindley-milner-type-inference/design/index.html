Core schema & types (§11)

struct TyVar(u32)                         // interned, monotonically allocated (I7)
struct Level(u32)                         // Rémy-style rank for O(1) generalization (ADR-06)

enum Type {
    Var(TyVar),
    Con(TyCon, Vec<TypeId>),              // e.g. Con(Int, []), Con(List, [elem])
    Arrow(TypeId, TypeId),
}
// ownership: interned + arena-owned by hm-types; immutable once created (P9)
// mutation: none post-construction; union-find rebinding lives in hm-unify, not here
// concurrency: single-writer per inference session; sessions do not share arenas
// persistence: not persisted across process restarts (ADR-14); serializable for hm-trace
struct TyCon { name: Symbol, arity: u8 }  // Int, Bool, String, List, Ref, ->
struct TypeScheme { vars: SmallVec<TyVar>, qualifiers: Vec<Predicate>, body: TypeId }
// ownership: owned by TypeEnv entries and the TypeScheme store
// serialization: stable Display form is the REPL's canonical pretty-print (I12)
struct Subst { forest: UnionFind<TyVar, TypeId> }   // hm-unify's canonical store
struct UnionFind<K, V> { parents: Vec<K>, ranks: Vec<u8>, bindings: Vec<Option<V>> }
// mutation: parents/bindings mutate in place during bind(); path compression on find()
// concurrency: not thread-shared; one Subst per inference session (I7)
// invariants: acyclic post-occurs-check (I2), idempotent under canonicalize (I6, I9)
enum Constraint {
    Equality(TypeId, TypeId, NodeId),          // provenance for diagnostics (P7)
    Instance(TypeId, SchemeId, NodeId),
    RowLacks(RowVar, Symbol, NodeId),          // Phase 9 extension
    Predicate(ClassId, TypeId, NodeId),        // Phase 10 extension
}
struct TypeEnv { bindings: PersistentMap<Symbol, TypeScheme>, level: Level }
struct Assumption { var: Symbol, scheme: TypeScheme, node: NodeId }
// ownership: persistent (structurally shared) map; cheap snapshot at let-scope entry
fn generalize(env: &TypeEnv, ty: TypeId) -> TypeScheme;   // hm-infer
fn instantiate(scheme: &TypeScheme) -> TypeId;             // hm-infer
struct Unifier { subst: Subst }
impl Unifier { fn unify(&mut self, a: TypeId, b: TypeId, at: NodeId) -> Result<(), InferError>; }
fn occurs_check(subst: &Subst, v: TyVar, ty: TypeId) -> bool;  // hm-unify
struct InferState { env: TypeEnv, unifier: Unifier, fresh: TyVarAllocator, trace: TraceSink }
enum AstExpr { Var(Symbol), Lit(Literal), Lambda(Symbol, ExprId), App(ExprId, ExprId),
               Let(Symbol, ExprId, ExprId), LetRec(Symbol, ExprId, ExprId) }
enum Literal { Int(i64), Bool(bool), Str(String) }
struct Row { fields: BTreeMap<Symbol, TypeId>, tail: Option<RowVar> }   // Phase 9
struct RowVar(u32)
struct Qualified { predicates: Vec<Predicate>, ty: TypeId }             // Phase 10
struct Predicate { class: ClassId, ty: TypeId }
enum InferError {
    Clash { a: TypeId, b: TypeId, at: NodeId },
    Occurs { var: TyVar, ty: TypeId, at: NodeId },
    Unbound { name: Symbol, at: NodeId },
    RowMismatch { field: Symbol, at: NodeId },
    UnresolvedPredicate { pred: Predicate, at: NodeId },
    Divergent { at: NodeId, budget: u32 },
}
struct PrincipalType { scheme: TypeScheme, verified: bool }   // Phase 7 output wrapper
