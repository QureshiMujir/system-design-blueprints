Core schema & types (§11)

struct JobId(u64)                                  Copy, Hash — job-scoped, driver-assigned
struct StageId { job: JobId, idx: u32 }             Copy, Hash — stable per Dag build
struct TaskId  { stage: StageId, partition: u32, attempt: u32 }   Copy, Hash — universal task-attempt address
struct HostId(u32); struct RackId(u32)              Copy, Hash — scheduler-local
struct MapOutput { task: TaskId, partition_sizes: Vec<u64>, partition_record_counts: Vec<u64>,
                    locations: Vec<HostId>, checksum: [u8;32] }     owned by driver post-ACK; immutable
struct ShuffleReadRequest { stage: StageId, partition: u32, map_attempts: Vec<TaskId> }
                                                                     constructed per-reduce-task; immutable
enum TaskKind { Map { input_split: InputSplit, num_output_partitions: u32 },
                Reduce { source_stage: StageId, partition: u32 } }
trait Task: Send + 'static {
  fn id(&self) -> TaskId
  fn kind(&self) -> &TaskKind
  fn locality_hint(&self) -> &LocalityHint
  fn resource_request(&self) -> &ResourceRequest
  fn run(self: Box<Self>, ctx: TaskContext) -> Result<TaskResult, TaskError>   // MUST be
}                                                                              // deterministic (P13)
struct TaskResult { task: TaskId, map_output: Option<MapOutput>, records_written: u64,
                     bytes_written: u64, wall_time: Duration }
trait Partitioner: Send + Sync + 'static { fn num_partitions(&self) -> u32
                                            fn partition_for(&self, key: &[u8]) -> u32 }
struct HashPartitioner  { num_partitions: u32 }                       xxh3_64(key) % num_partitions
struct RangePartitioner  { num_partitions: u32, boundaries: Vec<Vec<u8>> }  ADR-09
struct PartitionAndSecondarySort<P: Partitioner> { partitioner: P,
    secondary_comparator: Arc<dyn Fn(&[u8],&[u8]) -> Ordering + Send + Sync> }   §21.2

trait Combiner: Send + Sync + 'static {
  fn combine(&self, key: &[u8], values: &mut Vec<Vec<u8>>)     // MUST be assoc.+commut. (I8)
}
trait Shuffler: Send + Sync + 'static {
  fn write_map_output<'a>(&'a self, map: TaskId, data: BoxStream<'a,(u32,Vec<u8>)>)
      -> BoxFuture<'a, Result<MapOutput, ShuffleError>>
  fn read_partition<'a>(&'a self, request: ShuffleReadRequest)
      -> BoxStream<'a, Result<Vec<u8>, ShuffleError>>
  fn cleanup_stage<'a>(&'a self, stage: StageId) -> BoxFuture<'a, Result<(), ShuffleError>>
}
struct Dag<N> { nodes: Vec<N>, edges: Vec<Vec<usize>>, reverse_edges: Vec<Vec<usize>> }  // topo-ordered, I10
struct Stage { id: StageId, kind: StageKind, num_partitions: u32,
               partitioner: Arc<dyn Partitioner>, combiner: Option<Arc<dyn Combiner>>,
               dependencies: Vec<StageId> }
enum StageKind { Map { input_splits: Vec<InputSplit> }, Reduce { source_stage: StageId } }
struct Job { id: JobId, stages: Dag<Stage>, final_outputs: Vec<StageId> }
trait Scheduler: Send + Sync {
  fn next_task(&mut self, free_host: HostId, free_resources: ResourceUnits,
               candidates: &[TaskId]) -> Option<TaskId>
  fn notify(&mut self, event: SchedulerEvent)
}
enum SchedulerEvent { TaskCompleted { task: TaskId, wall_time: Duration, output: MapOutput },
                       TaskFailed { task: TaskId, reason: FailureReason },
                       SpeculativeLaunch { task: TaskId, new_attempt: u32 } }
struct ResourceUnits { cpu_milli: u32, memory_mb: u32 }
struct ResourceRequest { cpu_milli: u32, memory_mb: u32 }
struct LocalityHint { hosts: Vec<HostId>, racks: Vec<RackId> }
struct OutputLineage { alive_map_outputs: HashMap<StageId, HashSet<TaskId>>,
                        host_to_outputs: HashMap<HostId, HashSet<TaskId>> }
