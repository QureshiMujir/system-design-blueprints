Core schema & types (§11)

// crates/ratelimiter-core/src/lib.rs — public surface
pub trait Limiter: Send + Sync {
    fn check(&self, key: &Key, cost: u32) -> Future<Output = Result<Decision>>;   // hot path, non-blocking contract (P3)
    fn state(&self, key: &Key) -> Future<Output = Result<AlgorithmState>>;        // operator inspection, not hot path
    fn clear(&self, key: &Key) -> Future<Output = Result<()>>;                    // operator/CLI only
    fn quota(&self) -> &Quota;                                                     // immutable after construction
}
// Ownership: implementors own their state (Mutex/DashMap internal); Send+Sync required
// for middleware sharing across worker threads. No implementor may block in `check`.
// decision.rs — the contract handed to every caller
pub enum Decision {
    Permit { remaining: u32, reset: Duration },              // I4-adjacent: reset always >= 0
    Reject { retry_after: RetryAfter, reset: Duration },      // I4: retry_after always >= 0
}
pub struct RetryAfter(pub Duration);                          // newtype, Copy-free (Duration is Copy)
impl RetryAfter {
    pub fn as_seconds_ceiled(&self) -> u64;                    // used by §8.9 Retry-After header
}
// Serialization: Decision is never persisted; it is computed fresh per request (P3).
// key.rs — typed identity, wire-stable (P9, I9)
pub struct Key { pub tenant: String, pub scope: String, pub value: Vec<u8> }
impl Key {
    pub fn redacted(&self) -> String;   // INFO-level logging form (P10); never logs `value` bytes
    pub fn wire(&self) -> Vec<u8>;      // tenant\0scope\0value (ADR-24); stable across encoders
}
// Mutation: immutable after construction. Hash/Eq derive on all three fields — tenants
// with the same scope+value never collide (I9). Persistence: the wire form is the Redis
// key and the cluster shard-routing input.
// quota.rs — the policy, immutable per Limiter instance
pub enum Algorithm { TokenBucket, LeakyBucket, FixedWindow, SlidingLog, SlidingCounter, Gcra }
pub struct Quota { pub algorithm: Algorithm, pub burst: u32, pub rate: u32, pub period: Duration }
impl Quota {
    pub fn per_second(burst: u32, rate: u32) -> Self;   // convenience constructor, defaults to Gcra (ADR-10)
}
// Concurrency: Quota is Copy; read-only after Limiter construction (P1 — no runtime
// algorithm switching). Compatibility: adding an Algorithm variant is additive (non_exhaustive).
// clock.rs — the monotonic contract (P2, I3, §8.8)
pub trait ClockSource: Send + Sync {
    fn now_ns(&self) -> u64;   // monotonic non-decreasing within one process; panics-in-debug on regression (I3)
}
pub struct TokioClock;         // production impl, wraps tokio::time::Instant
pub struct SimClock { /* fields elided — deterministic clock for conformance/bench */ }
// error.rs
pub enum Error {
    Backend(String),                                    // §24 FM-3
    InvalidKey(String),                                  // §8.11
    InvalidQuota(&'static str),                          // §8.11
    CapabilityNotSupported(&'static str),                // stub backend (§14)
    ClockSkew { now_ns: u64, earlier_ns: u64 },           // I3 violation surfaced, never silently swallowed
}
pub type Result<T> = std::result::Result<T, Error>;
// AlgorithmState — the operator-visible per-key view, one variant per §8.1-8.6
pub enum AlgorithmState {
    TokenBucket { tokens: f64, last_refill_ns: u64 },
    LeakyBucket { level: f64, last_drip_ns: u64 },
    FixedWindow { count: u64, window_start_ns: u64 },
    SlidingLog { events: Vec<u64> },                              // capped, I5
    SlidingCounter { current: u64, previous: u64, window_start_ns: u64 },
    Gcra { tat_ns: u64 },
}
// non_exhaustive: a new algorithm adds a variant without breaking downstream matches.
// crates/ratelimiter-backend/src/lib.rs — the narrow abstraction (P7)
pub trait Backend: Send + Sync + 'static {
    fn decide(&self, key: &Key, cost: u32, quota: &Quota) -> Future<Output = Result<Decision>>;
    fn state(&self, key: &Key, quota: &Quota) -> Future<Output = Result<AlgorithmState>>;
    fn clear(&self, key: &Key) -> Future<Output = Result<()>>;
    fn capability(&self, alg: Algorithm) -> bool;   // stub backend returns false for all
}
// middleware-facing traits (crates/ratelimiter-tower/src/lib.rs and siblings)
pub trait KeyExtractor<R>: Clone + Send + Sync + 'static {
    fn extract(&self, req: &R) -> Key;
}
pub trait FailureModeDispatcher {
    fn on_backend_error(&self, err: &Error) -> Decision;   // P4: the only place FailureMode is consulted
}
pub trait HeaderEmitter {
    fn emit(&self, decision: &Decision, into: &mut HeaderMap);   // §8.9
}
pub enum FailureMode { FailOpen, FailClosed }   // no Default impl — construction requires explicit choice (I6)
