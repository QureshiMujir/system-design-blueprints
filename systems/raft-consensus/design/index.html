Core schema & types (§11)

pub struct NodeId(pub NonZeroU64);          // own: Copy; conc: Send+Sync; ser: proto uint64
pub struct Term(pub u64);                    // own: Copy, Ord; pers: StableState; ser: proto uint64
pub struct Index(pub u64);                   // own: Copy, Ord; pers: log key; ser: proto uint64
pub struct LogId { pub term: Term, pub index: Index }  // own: Copy; travels as a pair, never compared on index alone (I3)
pub enum EntryPayload { Command(Vec<u8>), Noop, ConfigChange(ConfigChange) }
                        // mut: append-only once written; ser: proto oneof
pub struct LogEntry { pub id: LogId, pub payload: EntryPayload }
                        // pers: redb log_entries table (§8.2); own: RaftLog owns by value
pub struct StableState { pub current_term: Term, pub voted_for: Option<NodeId> }
    // pers: single fsync per change (ADR-06, I7); conc: mutated only inside step()/tick()
pub enum Configuration {
    Stable { voters: BTreeSet<NodeId>, learners: BTreeSet<NodeId> },
    Joint  { old_voters: BTreeSet<NodeId>, new_voters: BTreeSet<NodeId>, learners: BTreeSet<NodeId> },
}   // pers: embedded in every snapshot (I10, P13); ser: proto ConfigurationProto; own: RaftNode
pub enum Message {
    AppendEntries(AppendEntriesReq), AppendEntriesReply(AppendEntriesReply),
    RequestVote(RequestVoteReq), RequestVoteReply(RequestVoteReply),
    PreVote(PreVoteReq), PreVoteReply(PreVoteReply),
    InstallSnapshot(InstallSnapshotReq), InstallSnapshotReply(InstallSnapshotReply),
    TimeoutNow(TimeoutNowReq),
    ReadIndexHeartbeat(ReadIndexHeartbeatReq), ReadIndexHeartbeatReply(ReadIndexHeartbeatReply),
}   // ser: one proto message per variant (§8.1); conc: Send, moved through Outbox/Transport
pub struct Outbox {
    pub messages: Vec<(NodeId, Message)>,      // drained by Transport
    pub persist: Vec<PersistOp>,                // drained by caller, acked via on_persisted (I7)
    pub apply: Vec<LogEntry>,                   // drained by StateMachine::apply (I5)
    pub snapshot: Vec<SnapshotOp>,               // drained by Snapshotter (§18 Phase 6)
    pub events: Vec<Event>,                      // drained by observability layer (P5)
}   // own: RaftNode; mut: append-only within one step()/tick(), drained+cleared by caller
pub struct RaftNode {
    id: NodeId, role: Role, log: RaftLog, state: StableState, config: Configuration,
    pending_config: Option<PendingConfigChange>, election: ElectionState,
    leader: LeaderState, read_index: ReadIndexState, snapshot: SnapshotState,
    settings: Settings, outbox: Outbox,
}
impl RaftNode {
    pub fn new(id: NodeId, settings: Settings, restored: RestoredState) -> Self;
    pub fn step(&mut self, from: NodeId, msg: Message) -> Result<()>;      // P1, P3
    pub fn tick(&mut self, elapsed: Duration);                             // P1
    pub fn propose(&mut self, command: Vec<u8>) -> Result<ProposalId>;     // leader-only
    pub fn change_membership(&mut self, change: ConfigChangeRequest) -> Result<ProposalId>; // I9
    pub fn drain_outbox(&mut self) -> Outbox;                              // P3
    pub fn on_persisted(&mut self, id: PersistId);                         // I7
    pub fn on_snapshot(&mut self, ev: SnapshotDoneEvent);                  // I10
    pub fn read_index(&mut self, ctx: ReadCtx) -> Result<ReadIndexHandle>; // I13
    pub fn is_leader(&self) -> bool;
}   // own: exclusive &mut per call, single-threaded per node (P4); conc: Send, not Sync
pub trait StateMachine: Send + 'static {
    type ApplyError: std::error::Error + Send + Sync + 'static;
    type Snapshot: Send + 'static;
    fn apply(&mut self, idx: Index, term: Term, bytes: &[u8]) -> Result<Vec<u8>, Self::ApplyError>;
    fn snapshot(&mut self, last_applied: LogId) -> Result<Self::Snapshot, Self::ApplyError>;
    fn restore(&mut self, snap: Self::Snapshot) -> Result<(), Self::ApplyError>;
}   // ownership: implementor-owned; mut: &mut self on every call; conc: driven single-threaded
pub trait LogStorage: Send + Sync + 'static {
    fn first_index(&self) -> Result<Index>;
    fn last_index(&self) -> Result<Index>;
    fn term(&self, idx: Index) -> Result<Term>;
    fn entries(&self, lo: Index, hi: Index, max_bytes: usize) -> Result<Vec<LogEntry>>;
    fn append(&self, entries: &[LogEntry]) -> Result<()>;       // fsync inside (I7)
    fn truncate_suffix(&self, from: Index) -> Result<()>;
    fn compact_prefix(&self, up_to: Index) -> Result<()>;        // I8
}
pub trait StableStore: Send + Sync + 'static {
    fn save_state(&self, st: &StableState) -> Result<()>;        // fsync inside (I7)
    fn load_state(&self) -> Result<StableState>;
}
pub trait Snapshotter: Send + Sync + 'static {
    type Reader: std::io::Read + Send; type Writer: std::io::Write + Send;
    fn open_writer(&self, meta: SnapshotMeta) -> Result<Self::Writer>;
    fn finalize(&self, meta: SnapshotMeta) -> Result<()>;
    fn open_reader(&self, meta: SnapshotMeta) -> Result<Self::Reader>;
    fn list(&self) -> Result<Vec<SnapshotMeta>>;
    fn delete(&self, meta: SnapshotMeta) -> Result<()>;
}   // conc: Send+Sync — may be called from a background flush task outside step()/tick()
#[async_trait::async_trait]
pub trait Transport: Send + Sync + 'static {
    async fn send(&self, to: NodeId, msg: Message) -> Result<()>;
    async fn recv(&mut self) -> Result<(NodeId, Message)>;
    async fn shutdown(&self);
}   // own: one per node; conc: async, tokio-driven (ADR-02); ser: prost/tonic (ADR-15)
pub enum ProgressState { Probe { last_sent: Option<Index> }, Replicate { inflight: VecDeque<InflightEntry> },
                          Snapshot { pending: SnapshotMeta } }
pub struct Progress { pub next_index: Index, pub match_index: Index, pub state: ProgressState,
                       pub last_active: Instant, pub recent_active: bool, pub paused: bool }
    // own: LeaderState owns one Progress per peer; mut: leader-role only; pers: none (rebuilt on election)
pub struct ConfigChangeRequest { pub adds_voters: BTreeSet<NodeId>, pub removes_voters: BTreeSet<NodeId>,
                                  pub adds_learners: BTreeSet<NodeId>, pub removes_learners: BTreeSet<NodeId>,
                                  pub promote_learners: BTreeSet<NodeId> }
pub struct PendingConfigChange { pub joint_index: Index, pub final_config: BTreeSet<NodeId>,
                                  pub final_learners: BTreeSet<NodeId>, pub final_index: Option<Index> }
    // pers: joint_index/final_index are log positions, recoverable from log replay (I9)
pub struct ReadIndexState { pub pending: VecDeque<PendingRead> }
pub struct PendingRead { pub ctx: ReadCtx, pub committed_at_request: Index, pub heartbeat_round: HeartbeatRoundId }
    // own: leader-only, volatile; mut: cleared on step-down (a stepped-down leader fails all pending reads)
pub struct Simulator { seed: u64, rng: StdRng, clock: SimulatedClock, nodes: BTreeMap<NodeId, SimNode>,
                        network: Network, faults: VecDeque<Fault>, invariants: InvariantChecker, history: History }
    // own: single-threaded owner of every node's RaftNode (P4 exploited for scheduling); conc: none — this is the point (I11)
pub trait Nemesis: Send { fn arm(&mut self, t: SimulatedTime); fn fire(&mut self) -> Result<()>; fn heal(&mut self) -> Result<()>; }
    // own: one per scenario; conc: driven from the harness's single control loop, not from node processes
pub struct ConformanceFixture { pub name: String, pub setup: Setup, pub steps: Vec<StepCase>, pub expect: Vec<ExpectCase> }
    // ser: JSON on disk (conformance/fixtures/); own: loaded fresh per run, no shared mutable fixture state
