Core schema & types (§11)

// ctrl-core: numeric foundation
pub trait Scalar: Copy + /* arithmetic + */ PartialOrd {
    const ZERO: Self; const ONE: Self;
    fn is_finite(self) -> bool;
    fn sqrt(self) -> Self; fn abs(self) -> Self; /* exp, sin, atan2 ... */
}
// f64, f32, and Fixed<Q> implement Scalar.

pub struct Vector<S: Scalar, const N: usize> { /* [S; N] */ }
pub struct Matrix<S: Scalar, const R: usize, const C: usize> { /* [[S; C]; R] */ }
// Ownership: value types, Copy when small; no heap. Mutation: in-place ops return new or mutate &mut.
// Concurrency: Send+Sync (plain data). Serialization: serde derive (host feature). Compat: shape in type => no version drift.

pub struct Saturation<S: Scalar> { pub min: S, pub max: S }
pub struct RateLimit<S: Scalar> { pub max_rate: S /* per unit time */, last: S }
// Mutation: RateLimit holds last output (interior state). Reset on bumpless transfer.

pub enum CoreError { NonFinite, Singular, NotPsd, DimensionMismatch, OutOfRange }
// ctrl-sim: plant + integration
pub trait Plant<S: Scalar, const N: usize, const M: usize, const P: usize> {
    fn derivative(&self, x: &Vector<S,N>, u: &Vector<S,M>) -> Vector<S,N>; // ẋ=f(x,u)
    fn output(&self, x: &Vector<S,N>, u: &Vector<S,M>) -> Vector<S,P>;     // y=h(x,u)
}
pub struct StateSpacePlant<S, const N: usize, const M: usize, const P: usize> {
    pub a: Matrix<S,N,N>, pub b: Matrix<S,N,M>, pub c: Matrix<S,P,N>, pub d: Matrix<S,P,M>,
}
pub struct TransferFunction<S: Scalar> { pub num: /* coeffs */, pub den: /* coeffs */, pub delay: S }
impl TransferFunction { pub fn to_state_space(&self) -> StateSpacePlant<..> /* controllable canonical */ }

pub trait Integrator<S: Scalar> {
    fn step<P: Plant<..>>(&self, plant: &P, x: &Vector<S,N>, u: &Vector<S,M>, dt: S) -> Vector<S,N>;
}
pub struct Euler; pub struct Rk4;   // both fixed-step
// Ownership: plant owned by harness; integrators stateless (ZST). Serialization: plant params serde.
// Compat: StateSpacePlant matrices serialized by shape+values; version field in file header.

pub struct SimHarness<S, P> { /* plant, integrator, dt, trace buffers */ }
// ctrl-signal: discretization + filters
pub enum DiscretizeMethod { ForwardEuler, BackwardEuler, Tustin { prewarp: Option<f64> }, Zoh }
pub trait Discretizer<S: Scalar> {
    fn discretize_ss(&self, ss: &StateSpacePlant<..>, dt: S) -> StateSpacePlant<..>;       // ZOH
    fn discretize_tf(&self, tf: &TransferFunction<S>, dt: S) -> /* discrete coeffs */;
}
pub struct DerivativeFilter<S: Scalar> { pub n: S /* filter coeff */, state: S } // 1st-order LP
// Mutation: filters carry state; reset on bumpless. Concurrency: per-loop instance, not shared.
// ctrl-rt: real-time loop
pub trait Clock { fn now(&self) -> Instant; }                 // monotonic
pub trait Scheduler { fn wait_until(&mut self, deadline: Instant) -> WakeInfo; }
pub struct RealTimeLoop<C: Clock, S: Scheduler> {
    pub dt: Duration, clock: C, sched: S, telemetry: Telemetry, safe: SafeState,
}
impl RealTimeLoop {
    pub fn run<F: FnMut(TickCtx) -> Result<(), LoopError>>(&mut self, body: F) -> !; // ticks at fixed dt
}
pub struct Telemetry { /* alloc-free counters + histograms: tick_compute_us, wake_jitter, misses */ }
pub enum SafeState { HoldLast, RampTo(/* val */), Handoff, Zero }
pub enum LoopError { DeadlineMiss, NonFinite, Faulted }
// Ownership: loop owns clock/sched/telemetry. Concurrency: single-threaded; telemetry snapshot read off-loop.
// ctrl-core control interface (shared)
pub trait Controller<S: Scalar, const NR: usize, const NM: usize, const NU: usize> {
    fn update(&mut self, reference: &Vector<S,NR>, meas: &Vector<S,NM>, dt: S) -> Vector<S,NU>;
    fn reset(&mut self);                              // clear internal state
    fn preload(&mut self, current_u: &Vector<S,NU>);  // bumpless transfer
}
// ctrl-pid
pub enum PidForm { Parallel, Standard, Series }
pub struct PidConfig<S: Scalar> {
    pub form: PidForm,
    pub kp: S, pub ki: S, pub kd: S,        // or kp,ti,td for Standard
    pub setpoint_weight_b: S, pub setpoint_weight_c: S,
    pub deriv_filter_n: S,
    pub anti_windup: AntiWindup<S>,
    pub sat: Saturation<S>, pub rate: RateLimit<S>,
}
pub enum AntiWindup<S> { None, Clamp, BackCalc { tt: S } }
pub struct PidController<S: Scalar> { cfg: PidConfig<S>, integral: S, deriv: DerivativeFilter<S>, last_u: S, y_prev: S }
impl Controller for PidController { /* scalar ref/meas/u (NR=NM=NU=1) */ }
// Mutation: integral/deriv/last_u mutate per tick. Serialization: PidConfig serde (TOML). 
// Compat: config has `version`; unknown fields rejected; gains semver-stable field names.
// ctrl-ss / ctrl-lqr / ctrl-kf
pub fn controllability<..>(ss: &StateSpacePlant<..>) -> (Matrix<..>, usize /*rank*/);
pub fn observability<..>(ss: &StateSpacePlant<..>) -> (Matrix<..>, usize);
pub fn place_poles<..>(ss: &StateSpacePlant<..>, desired: &[Complex<S>]) -> Result<Matrix<S,1,N>, CoreError>; // Ackermann

pub struct LqrWeights<S, const N: usize, const M: usize> { pub q: Matrix<S,N,N>, pub r: Matrix<S,M,M> }
pub struct Lqr<S, const N: usize, const M: usize> { pub k: Matrix<S,M,N>, pub p: Matrix<S,N,N> }
impl Lqr { pub fn solve_continuous(ss:&.., w:&LqrWeights<..>) -> Result<Self,CoreError>;
           pub fn solve_discrete(ss:&.., w:&LqrWeights<..>) -> Result<Self,CoreError>; }
pub struct StateFeedback<S, const N: usize, const M: usize> { pub k: Matrix<S,M,N> } // u=-Kx̂
impl Controller for StateFeedback { /* ... */ }

pub struct Luenberger<S, const N: usize, const P: usize> { pub l: Matrix<S,N,P>, x_hat: Vector<S,N> }
impl Luenberger { pub fn design(ss:&.., desired:&[Complex<S>]) -> Result<Self,CoreError>; // dual placement
                  pub fn update(&mut self, u:&.., y:&.., dt:S) -> &Vector<S,N>; }

pub struct KalmanFilter<S, const N: usize, const M: usize, const P: usize> {
    a: Matrix<S,N,N>, b: Matrix<S,N,M>, c: Matrix<S,P,N>,
    q: Matrix<S,N,N>, r: Matrix<S,P,P>, x_hat: Vector<S,N>, p: Matrix<S,N,N>,
}
impl KalmanFilter { pub fn predict(&mut self, u:&Vector<S,M>); 
                    pub fn update(&mut self, z:&Vector<S,P>) -> &Vector<S,N>;
                    pub fn nis(&self) -> S; }  // innovation diagnostic
// Concurrency: estimator owned by loop, single-threaded. Serialization: matrices+covariance serde.
// ctrl-mpc
pub struct MpcProblem<S, const N: usize, const M: usize> {
    pub a: Matrix<S,N,N>, pub b: Matrix<S,N,M>,
    pub q: Matrix<S,N,N>, pub r: Matrix<S,M,M>, pub p_f: Matrix<S,N,N>,
    pub horizon: usize,
    pub u_min: Vector<S,M>, pub u_max: Vector<S,M>, pub du_max: Vector<S,M>,
    /* optional output constraints */
}
pub trait QpSolver<S: Scalar> {
    fn solve(&mut self, h:&Matrix<..>, f:&Vector<..>, a_c:&Matrix<..>, b_c:&Vector<..>,
             warm: Option<&Vector<..>>) -> Result<QpSolution<..>, QpError>;
}
pub enum QpError { Infeasible, MaxIters, NonFinite }
pub struct QpSolution<S, const D: usize> { pub u: Vector<S,D>, pub iters: u32, pub status: QpStatus }
pub struct MpcController<S, Q: QpSolver<S>> { problem: MpcProblem<..>, solver: Q, last_u: Vector<..>, /* condensed H,F cached */ }
impl Controller for MpcController { /* solve receding-horizon, apply u0 */ }
// Mutation: caches condensed matrices (constant if model fixed); warm-start state mutates.
// Compat: MpcProblem serialized with version; horizon bounded by const generic max.
// ctrl-track
pub struct Path<S: Scalar> { /* waypoints, arclength, curvature */ }
pub struct VehicleState<S: Scalar> { pub x: S, pub y: S, pub heading: S, pub v: S }
pub trait PathFollower<S: Scalar> {
    fn steer(&mut self, state: &VehicleState<S>, path: &Path<S>, dt: S) -> S; // returns steering angle
}
pub struct PurePursuit<S: Scalar> { pub wheelbase: S, pub ld0: S, pub kv: S }
pub struct Stanley<S: Scalar> { pub k: S, pub k_soft: S }
impl PathFollower for PurePursuit { /* ... */ }
impl PathFollower for Stanley { /* ... */ }
// Ownership: path borrowed each call. Serialization: gains serde. Compat: gain field names stable.
