Core schema & types (§11)

#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]
pub struct Pid(pub i32);

#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]
pub struct Tid(pub i32);          // kernel-level tid (gettid), never a pthread handle

pub struct Inferior {
    pid: Pid,
    leader_tid: Tid,
    threads: BTreeMap<Tid, ThreadState>,
    modules: ModuleSet,
    state: InferiorState,          // Running | Stopped | Exited(i32)
}
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]
pub struct BreakpointId(pub u64);

pub struct Breakpoint {
    pub id: BreakpointId,
    pub location: BreakLocation,       // Address | SourceLine | Symbol | Pending
    pub kind: BreakpointKind,          // Software | Hardware { slot: u8 }
    pub state: BreakpointState,        // Pending | Installed | Disabled | Deleted
    pub hits: u64,
    pub condition: Option<Expression>, // pure expr only — see Deferred Features Register
    pub installed_byte: Option<u8>,
    pub installed_addr: Option<u64>,
    pub thread_filter: Option<Tid>,
}

#[derive(Debug, Clone, Copy)]
pub enum AccessKind { Read, Write, ReadWrite }

pub struct Watchpoint {
    pub id: WatchpointId,
    pub address: u64,
    pub size: u8,               // 1,2,4,8 bytes
    pub access: AccessKind,
    pub slot: u8,                // DR0-DR3 or BRP/WRP index (Invariant I5)
    pub thread_filter: Option<Tid>,
}
pub struct Frame {
    pub id: FrameId,
    pub pc: u64,
    pub cfa: u64,                       // Canonical Frame Address
    pub sp: u64,
    pub return_address: Option<u64>,
    pub kind: FrameKind,                // Real | Inline { depth: u16, parent: FrameId }
    pub function: Option<String>,
    pub source: Option<SourceLocation>,
    pub locals: Vec<LocalVariable>,     // lazily filled (Principle P6)
}

pub struct Module {
    pub path: PathBuf,
    pub load_address: u64,
    pub aslr_slide: u64,
    pub elf: Arc<ElfImage>,
    pub dwarf: Option<Arc<DwarfData>>,
    pub debuglink: Option<PathBuf>,
    pub buildid: Option<[u8; 20]>,
}
pub enum Expression {
    Variable(String), Number(i64), Float(f64), String(String),
    BinaryOp { op: BinOp, lhs: Box<Expression>, rhs: Box<Expression> },
    Field { base: Box<Expression>, name: String },
    Index { base: Box<Expression>, index: Box<Expression> },
    Deref(Box<Expression>), AddressOf(Box<Expression>),
    Cast { type_name: String, expr: Box<Expression> },
    // Call{..} intentionally absent — see Deferred Features Register
}

pub struct Session {
    inferior: Option<Inferior>,
    breakpoints: BTreeMap<BreakpointId, Breakpoint>,
    watchpoints: BTreeMap<WatchpointId, Watchpoint>,
    follow_fork_mode: FollowForkMode,   // Parent | Child | Both
    follow_exec_mode: FollowExecMode,   // Reuse | Detach
    scheduler_locking: SchedulerLockMode, // Off | Step | On
}

pub enum InferiorEvent {
    Stopped { tid: Tid, reason: StopReason, pc: u64 },
    Exited { pid: Pid, status: i32 },
    Forked { parent: Pid, child: Pid },
    Execed { pid: Pid, path: PathBuf },
    ThreadStarted { pid: Pid, tid: Tid },
    ThreadExited { pid: Pid, tid: Tid },
}
pub trait InferiorControl: Send + Sync {
    fn launch(&mut self, argv: &[OsString]) -> Result<Pid>;
    fn attach(&mut self, pid: Pid) -> Result<()>;
    fn cont(&mut self, tid: Tid, signal: Option<i32>) -> Result<()>;
    fn singlestep(&mut self, tid: Tid) -> Result<()>;
    fn get_regs(&self, tid: Tid) -> Result<Registers>;
    fn read_memory(&self, addr: u64, len: usize) -> Result<Vec<u8>>;
    fn install_swbp(&mut self, addr: u64) -> Result<u8>;
    fn install_hwbp(&mut self, addr: u64, kind: HwBpKind) -> Result<u8>;
    fn install_watchpoint(&mut self, addr: u64, size: u8, access: AccessKind) -> Result<u8>;
}

pub trait SymbolicationSource: Send + Sync {
    fn module_at(&self, pc: u64) -> Option<&Module>;
    fn line_for_pc(&self, pc: u64) -> Result<Option<LineRow>>;
    fn locals_for_pc(&self, pc: u64) -> Result<Vec<LocalVariable>>;
}

pub trait Unwinder: Send + Sync {
    fn unwind(&self, regs: &Registers, src: &dyn SymbolicationSource) -> Result<Vec<Frame>>;
    fn unwind_async_signal_safe(&self, regs: &Registers, out: &mut [Frame]) -> Result<usize>;
}

pub trait ProtocolServer: Send + Sync {
    fn name(&self) -> &str;
}

pub trait PrettyPrinter: Send + Sync {
    fn matches(&self, ty: &Type, lang: SourceLanguage) -> bool;
    fn render(&self, value: &Value, ctx: &RenderCtx<'_>) -> Result<String>;
}
