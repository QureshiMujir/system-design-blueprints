Core schema & types (§11)

// chronodb-core/src/sample.rs
type Timestamp = i64;                    // unix millis, ADR-18
type Value     = f64;
struct Sample { ts: Timestamp, value: Value }   // Copy, no allocation

// chronodb-core/src/series.rs
struct LabelSet(Vec<(String, String)>);  // sorted, canonical hash; owned
type SeriesId = u64;                     // dense, allocator-assigned
struct SeriesRef { id: SeriesId, labels: LabelSet }  // Send, immutable after creation

// chronodb-core/src/error.rs
enum ChronoError {
    CardinalityExceeded { shard: u32, limit: u64 },   // I11
    OutOfOrderRejected { ts: Timestamp, window_start: Timestamp },  // I12
    ChunkDecodeError(ChunkError),
    WalCorrupt { segment: u64, offset: u64 },          // I4, FM-1
    BlockChecksumMismatch { block_id: BlockId },        // I10, FM-11
}

// chronodb-codec/src/chunk.rs
enum ChunkError { LengthMismatch, UnknownEncoding(u8), BitstreamUnderflow }
struct ChunkMeta { min_ts: Timestamp, max_ts: Timestamp, num_samples: u32, encoding: u8 }
struct Chunk { meta: ChunkMeta, bytes: Vec<u8> }        // owned buffer, immutable after seal

// chronodb-wal/src/record.rs
enum WalRecord {
    SeriesDef(SeriesRef),
    Samples { series: SeriesId, samples: Vec<Sample> },
    Tombstone { series: SeriesId, from: Timestamp, to: Timestamp },  // P14
    Checkpoint { head_snapshot_id: u64 },
}
// chronodb-index/src/postings.rs
struct Postings(Vec<SeriesId>);          // sorted, delta-varint encoded on disk (ADR-08)
struct LabelIndex {
    // label_name -> label_value -> Postings   (conceptual; concrete map type internal)
}

// chronodb-storage/src/manifest.rs
type BlockId = ulid::Ulid;                // sortable unique id
struct BlockMeta {
    id: BlockId, min_ts: Timestamp, max_ts: Timestamp,
    series_count: u64, sample_count: u64,
    checksum: u32, encoding_version: u16,   // P11
}

// chronodb-storage/src/head.rs
enum HeadState { Empty, Active, Sealing, Persisted }  // §8.4 (subset shown)
struct Head { state: HeadState, min_ts: Timestamp, max_ts: Timestamp }  // owns per-series chunk buffers

// chronodb-query/src/selector.rs
enum Matcher { Eq(String,String), Neq(String,String), Regex(String,String) }
struct Selector { matchers: Vec<Matcher> }

// chronodb-query/src/plan.rs
struct QueryPlan { selector: Selector, range: (Timestamp, Timestamp), aggregation: Option<AggKind> }
enum AggKind { Rate, Quantile(f64), HistogramBucket(f64) }

// chronodb-rollup/src/tiers.rs
struct RollupTier { name: &'static str, bucket_width_ms: i64, retention_ms: i64 }  // ADR-23
