Core schema & types (§11)

// -- time and events --
type SimTime = u64;                          // simulated nanoseconds; Copy, Ord

struct Event {                               // owned by EventQueue once scheduled; immutable after push
    id: u128,
    at: SimTime,
    kind: EventKind,
    target: ComponentId,
}

enum EventKind {                             // exhaustive match required at every dispatch site (P2)
    RadioArrival(Message),
    TimerFired(TimerId),
    MobilityTick,
    SchedulerTick(CellId),
    FaultInjected(FaultKind),
}

struct EventQueue {                          // single owner: Simulator; min-heap by SimTime (I4)
    // push(Event), pop() -> Option<Event>
}

struct Simulator {                           // top-level owner of clock + queue + all components
    clock: SimTime,
    queue: EventQueue,
    seed: u64,                               // ADR-17; determines all PRNG draws
}
// -- identity --
type UeId = u64;
type CellId = u32;
struct Imsi(String);                         // LTE permanent identity (opaque token, ADR-14)
struct Supi(String);                          // 5G permanent identity
struct Guti { mme_id: u32, m_tmsi: u32 }       // temporary identity, unique per I8
struct FiveGGuti { amf_id: u32, tmsi: u32 }

struct Ue {                                   // single owner: Simulator's UE table
    id: UeId,
    imsi: Imsi,
    guti: Option<Guti>,
    stack: ProtocolStack,
    rrc_state: RrcState,
    emm_state: EmmState,
    security: Option<SecurityContext>,
    bearers: Vec<Bearer>,
}
// -- protocol states --
enum RrcState { Idle, Connecting, Connected, Inactive }   // I1
enum EmmState { Deregistered, Registering, Authenticating, Registered }
enum MmState  { Deregistered, Registering, Authenticated, Registered }  // 5G equivalent
// -- protocol layers --
struct ProtocolStack {                         // single owner: Ue or ENodeB/GNb
    nas: NasLayer,
    rrc: RrcLayer,
    pdcp: PdcpLayer,
    rlc: RlcLayer,
    mac: MacLayer,
    phy: PhyAbstraction,
}

struct NasLayer  { /* EMM/MM transition fn, msg encode/decode */ }
struct RrcLayer  { /* RRC transition fn, msg encode/decode */ }
struct PdcpLayer { tx_sn: u32, rx_reorder_window: ReorderBuffer }   // I6
struct RlcLayer  { tx_sn: u32, retransmit_queue: BoundedQueue<RlcPdu> }
struct MacLayer  { harq_state: HarqState, sched_handle: SchedulerHandle }
struct PhyAbstraction { sinr_fn: fn(CellId, UeId) -> Sinr }         // ADR-06
// -- messages --
enum Message { Nas(NasMessage), Rrc(RrcMessage), GtpU(Bytes), Sbi(SbiRequest) }  // ADR-18

enum NasMessage {                              // TS 24.301 / TS 24.501 message set
    AttachRequest, AuthenticationRequest, AuthenticationResponse,
    SecurityModeCommand, SecurityModeComplete, AttachAccept, AttachComplete,
    TauRequest, TauAccept,
}

enum RrcMessage {                              // TS 36.331 / TS 38.331 message set
    ConnectionRequest, ConnectionSetup, ConnectionSetupComplete,
    Reconfiguration, ReconfigurationComplete, Release, Paging,
    MeasurementReport,
}
// -- bearers / QoS --
enum Qci { Qci1, Qci5, Qci8, Qci9 }            // §8.13
enum FiveQi { FiveQi1, FiveQi9 }
struct Bearer { id: u32, qci: Qci, ebi: u8 }
struct PduSession { id: u32, fiveqi: FiveQi }
// -- security --
struct SecurityContext { kasme: KeyToken, ksi: u8 }   // symbolic key material, ADR-10
struct KeyHierarchy { k: KeyToken, ck_ik: (KeyToken, KeyToken), kasme: KeyToken, kenb: KeyToken }
struct KeyToken(u128);                          // opaque, never real cryptographic bytes
// -- core network --
struct CoreNetwork { mme: Mme, sgw: Sgw, pgw: Pgw, hss: Hss }   // LTE (single owner: Simulator)
struct FiveGCore   { amf: Amf, smf: Smf, upf: Upf, udm: Udm }   // 5G, Phase 11

struct Mme { ue_contexts: HashMap<UeId, EmmState> }
struct Amf { ue_contexts: HashMap<UeId, MmState> }
struct Sgw { sessions: HashMap<u32, Bearer> }
struct Upf { sessions: HashMap<u32, PduSession> }
struct Hss { subscribers: HashMap<Imsi, KeyHierarchy> }
struct Udm { subscribers: HashMap<Supi, KeyHierarchy> }
// -- base station / mobility / metrics / errors --
struct ENodeB { id: CellId, stack: ProtocolStack, scheduler: SchedulerHandle }
struct GNb    { id: CellId, stack: ProtocolStack, scheduler: SchedulerHandle }

struct HandoverContext { ue: UeId, source: CellId, target: CellId, state: HandoverState }  // I5
enum HandoverState { Preparing, Forwarding, Completing, RolledBack }

struct MobilityModel { waypoints: Vec<(SimTime, Position)> }
struct ChannelModel  { path_loss_fn: fn(Position, CellId) -> f64 }
struct Sinr(f64);                               // dB, Copy

struct Metrics { latencies: HistogramSet, signaling_counts: CounterSet }  // ADR-15

enum SimError {                                 // error taxonomy
    MalformedMessage { layer: &'static str },
    InvariantViolation { id: &'static str },
    QueueOverflow { queue: &'static str },
    ScenarioParseError { reason: String },
    ReplayDivergence { at: SimTime },
}
