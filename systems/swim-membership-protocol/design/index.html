Core schema & types (§11)

// core — membership state (normative)
type NodeId string                              // stable identity (ADR-06)
type NodeState uint8                            // Alive | Suspect | Dead | Left
type Incarnation uint64                         // logical version (P3/I1)

type Member struct {
    Name        NodeId
    Addr        netip.AddrPort
    State       NodeState
    Incarnation Incarnation
    Meta        []byte                           // bounded app metadata
    stateTime   time.Time                        // local-only; never gossiped
}

type MembershipList struct {
    mu        sync.RWMutex                        // all mutations under lock (I13)
    members   map[NodeId]*Member
    ring      []NodeId                            // shuffled probe order (I8)
    ringPos   int
    suspicion map[NodeId]*suspicionTimer          // active timers (I9)
    self      NodeId
    selfInc   Incarnation
}
func (l *MembershipList) Merge(u Update) MergeResult   // conflict resolution (I2, §8.3)
func (l *MembershipList) NextProbeTarget() (NodeId, bool) // round-robin (I8)
func (l *MembershipList) Alive() []Member
func (l *MembershipList) RefuteSelf(min Incarnation)   // bump selfInc > min, emit Alive (P5/I1)
// transport — pluggable real/simulated (normative, enables sim I14)
type Packet struct{ From netip.AddrPort; Buf []byte; Timestamp time.Time }
type Transport interface {
    SendUDP(to netip.AddrPort, b []byte) error      // ≤ MTU (I6)
    RecvUDP() <-chan Packet
    DialTCP(to netip.AddrPort) (net.Conn, error)    // push-pull (I10)
    AcceptTCP() <-chan net.Conn
    LocalAddr() netip.AddrPort
}
// codec — wire (normative, I6)
type MsgType uint8 // ping ack pingReq nack alive suspect dead leave pushPull
type Message struct {
    Type      MsgType
    Seq       uint32
    Source    NodeId
    Target    NodeId                               // ping/ping-req
    Payload   any                                  // type-specific
    Piggyback []Update                             // packed gossip (freshest-first)
}
type Update struct{ Node NodeId; Inc Incarnation; State NodeState; Addr netip.AddrPort; Meta []byte }
func Encode(m Message, mtu int) ([]byte, []Update) // returns leftover piggyback that didn't fit
func Decode(b []byte) (Message, error)             // fuzzed (ADR-21)
// failuredetector + indirect (normative)
type FailureDetector interface {
    Tick()                                          // one protocol period (I8)
    HandleAck(seq uint32, from NodeId)
}
type IndirectProbe struct{ Relays int }             // k (ADR-09)
func (i *IndirectProbe) Probe(target NodeId) (acked bool, nacks int) // Lifeguard NACK count
// dissemination — infection-style buffer (normative, I7)
type GossipBuffer interface {
    Queue(u Update)                                 // add/refresh an update
    Pack(maxBytes int) []Update                     // freshest-first, ≤ MTU budget
    // each Update retransmitted ≤ λ·log(N+1) then retired
}
// suspicion FSM (normative, I3/I9)
type Suspicion interface {
    Suspect(n NodeId, inc Incarnation)              // start timer, gossip Suspect
    Refute(n NodeId, inc Incarnation) bool          // cancel if higher inc (I9)
    OnTimeout(n NodeId)                             // → Dead, gossip
    Timeout(n int /*cluster size*/) time.Duration   // ∝ log(n)·T′ · LHM (P10/I11)
}
// lifeguard (normative, P10/I11)
type Lifeguard struct {
    lhm int32                                       // local health multiplier, clamped [0, max]
}
func (lg *Lifeguard) ProbeInterval(base time.Duration) time.Duration // dilated by lhm
func (lg *Lifeguard) OnMissedAckToSelf()  // raise lhm
func (lg *Lifeguard) OnSuccessfulRefuteOrAck() // lower lhm

// node — public API (normative)
type Config struct{ Name NodeId; Bind netip.AddrPort; Seeds []netip.AddrPort; PeriodMs, K, SuspicionMult, Lambda int; Seed int64 }
type Delegate interface { NotifyJoin(Member); NotifyLeave(Member); NotifyUpdate(Member) } // callbacks (P14)
type Node interface {
    Join(seeds []netip.AddrPort) (joined int, err error)
    Leave(timeout time.Duration) error              // graceful (P13/I12)
    Members() []Member
    LocalIncarnation() Incarnation
}
type PhaseDeferredError struct{ Reference string }  // PhaseDeferred("<Phase P | Deferred Features Register>")
