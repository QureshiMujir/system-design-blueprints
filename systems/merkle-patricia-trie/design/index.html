Core schema & types (§11)

pub struct Hash([u8; 32]);                    // Copy, content-addressed identity (P6)

pub struct Nibbles { data: Vec<u8> }           // ADR-17: one nibble per byte

pub enum NodeRef {                             // I2 embedding decision
    Embedded(Vec<u8>),                          // canonical RLP < 32 bytes
    Hash(Hash),                                  // canonical RLP >= 32 bytes
}

pub enum TrieNode {                             // §8 node types
    Empty,
    Branch { children: [Option<NodeRef>; 16], value: Option<Vec<u8>> },
    Extension { path: Nibbles, child: NodeRef },
    Leaf { path: Nibbles, value: Vec<u8> },
}

pub struct Proof { nodes: Vec<Vec<u8>> }        // ADR-15: raw ordered RLP node list

pub struct SecureKey(Hash);                     // I9: keccak256(raw_key), ADR-16

pub struct Snapshot { journal_checkpoint: u64, root: Hash }  // I6

pub struct JournalEntry {                        // I5: append-only, monotonic
    seq: u64,
    op: JournalOp,
    prev_root: Hash,
    new_root: Hash,
}

pub enum JournalOp { Insert, Delete, Commit, Rollback }

pub struct WalRecord { entry: JournalEntry, crc32: u32 }  // T7/FM-4

pub struct RefCountEntry { hash: Hash, count: u32 }        // I13, P12
pub enum TrieError {                             // ADR-14: typed, exhaustive
    NonCanonicalRlp,
    HashMismatch,
    DepthExceeded,
    MalformedProof,
    OrphanReference,
    StorageCorrupt,
    SnapshotNotFound,
}

pub trait NodeStore {                            // ADR-04: backend-agnostic
    fn get(&self, hash: &Hash) -> Result<Option<Vec<u8>>, TrieError>;
    fn put(&mut self, hash: Hash, node: Vec<u8>) -> Result<(), TrieError>;
    fn delete(&mut self, hash: &Hash) -> Result<(), TrieError>;
}

pub trait ProofVerifier {                         // ADR-24: stateless verifier
    fn verify(root: &Hash, proof: &Proof, key: &[u8], value: Option<&[u8]>) -> bool;
}

pub struct DiffReport {                           // §21 differential harness
    fixture_id: String,
    expected_root: Hash,
    actual_root: Hash,
    diverged_at_depth: Option<u32>,
}

pub struct Metrics {                              // §24 telemetry surface
    root_mismatch_total: u64,
    rlp_decode_reject_total: u64,
    proof_verify_fail_total: u64,
    gc_dangling_ref_total: u64,
    cache_stale_hit_total: u64,
}
