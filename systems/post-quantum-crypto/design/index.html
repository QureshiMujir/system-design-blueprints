// params.rs
pub struct KemParams {
    pub k: usize,       // module rank: 2 (512), 3 (768), 4 (1024)
    pub eta1: usize,    // CBD noise parameter for s, e
    pub eta2: usize,    // CBD noise parameter for e1, e2
    pub du: usize,      // compress bits for u
    pub dv: usize,      // compress bits for v
    pub q: i32,         // 3329
}

pub struct SignParams {
    pub k: usize,       // rows in A: 4 (44), 6 (65), 8 (87)
    pub l: usize,       // cols in A / length of s1
    pub eta: usize,     // bound on s1, s2 coefficients
    pub tau: usize,     // number of ±1s in challenge c
    pub beta: usize,    // ||cs1||/||cs2|| rejection bound
    pub gamma1: i32,    // masking y coefficient bound
    pub gamma2: i32,    // low-bits rounding divisor
    pub omega: usize,   // max total hint bits
    pub q: i32,         // 8380417
}

// zq.rs
#[derive(Copy, Clone, Debug, PartialEq, Eq)]
pub struct Zq(pub i32);   // canonical range [0, q); stored as i32

impl Zq {
    pub fn new(v: i32, q: i32) -> Self;               // reduce v into [0, q) via barrett_reduce
    pub fn add(self, rhs: Self, q: i32) -> Self;      // (a + b) mod q, no branch
    pub fn sub(self, rhs: Self, q: i32) -> Self;      // (a - b + q) mod q, no branch
    pub fn mul(self, rhs: Self, q: i32) -> Self;      // montgomery_reduce(a * b)
    pub fn montgomery_reduce(a: i64, q: i32) -> i32;  // contract: output in (-q, q)
    pub fn barrett_reduce(a: i32, q: i32) -> i32;     // contract: output in [0, q)
}

// error.rs
#[derive(Debug)]
pub enum PqcError {
    DecapsulationFailure,  // FO mismatch (implicit rejection triggered)
    VerificationFailure,   // signature norm / hint / c~ mismatch
    EncodingError,         // byte length / bit-pack violation
    MalformedInput,        // parse failure on key / ciphertext / signature
    NormBoundExceeded,     // ||z|| or hint count out of spec
    SamplerAbortLimit,     // signing rejected too many times (> MAX_SIGN_ITER)
}
// poly.rs
pub struct Poly {
    pub coeffs: [i32; 256],  // coefficients in Z_q; domain tracked externally via Domain tag
}

impl Poly {
    pub fn zero() -> Self;
    pub fn add(&self, rhs: &Self, q: i32) -> Self;
    pub fn sub(&self, rhs: &Self, q: i32) -> Self;
    pub fn negate(&self, q: i32) -> Self;
    pub fn reduce_all(&mut self, q: i32);                          // I8: all coeffs → [0, q)
    pub fn schoolbook_mul(&self, rhs: &Self, q: i32) -> Self;     // reference; I5 oracle
}
