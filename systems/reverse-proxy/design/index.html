Core schema & types (§11)

pub struct Request {
    pub id: RequestId, pub received_at: Instant, pub client_addr: SocketAddr,
    pub tls: Option<TlsContext>, pub method: Method, pub uri: Uri, pub version: Version,
    pub headers: HeaderMap, pub body: Body, pub route: Option<RouteId>,
    pub principal: Option<Principal>, pub trace: TraceContext,
}
pub struct Response {
    pub status: StatusCode, pub headers: HeaderMap, pub body: Body,
    pub trailers: Option<HeaderMap>, pub upstream: Option<UpstreamId>,
    pub cache_status: CacheStatus,
}
pub enum Body { Empty, Bytes(Bytes), Stream(BoxStream<'static, Result<Bytes, BodyError>>) }
pub struct Route {
    pub id: RouteId, pub priority: i32, pub matcher: Matcher, pub upstream: UpstreamRef,
    pub middlewares: Vec<MiddlewareRef>, pub timeout: Duration, pub retry: RetryPolicy,
}
pub struct Matcher {
    pub host: Option<HostMatch>, pub path: Option<PathMatch>, pub method: Option<MethodSet>,
    pub headers: Vec<HeaderMatch>, pub jwt_claims: Vec<JwtClaimMatch>, pub query: Vec<QueryMatch>,
}
pub enum PathMatch { Exact(String), Prefix(String), Regex(CompiledRegex) }
pub enum Transform {
    HeaderSet { name: HeaderName, value: HeaderValue, scope: Scope },
    HeaderRemove { name: HeaderName, scope: Scope },
    HeaderRename { from: HeaderName, to: HeaderName, scope: Scope },
    PathRewrite { from: PathPattern, to: PathTemplate },
    PathStripPrefix { prefix: String }, PathAddPrefix { prefix: String },
    BodyRewrite { rule: BodyRewriteRule, max_bytes: u64 },
    Redirect { status: StatusCode, location: LocationTemplate },
    RequestInject { header: HeaderName, value: ValueSource },
    ResponseInject { fragment: ResponseFragment, position: InjectPosition },
}
pub enum Scope { Request, Response }
pub enum InjectPosition { BeforeBodyTag, AfterBodyTag, BeforeHeadTag, AfterHeadTag }
pub struct CacheKey {
    pub method: Method, pub url: String,
    pub vary: SmallVec<[(HeaderName, HeaderValue); 4]>, pub surrogate_key: Option<SurrogateKey>,
}
pub struct CacheEntry {
    pub key: CacheKey, pub status: StatusCode, pub headers: HeaderMap, pub body: Bytes,
    pub stored_at: SystemTime, pub fresh_until: SystemTime, pub stale_until: SystemTime,
    pub etag: Option<HeaderValue>, pub last_modified: Option<HeaderValue>,
}
pub enum CacheStatus {
    Hit, Miss, Bypass, Stale, StaleWhileRevalidate, Revalidated, StoredOnReturn, NotCacheable,
}
pub struct UpstreamPool {
    pub id: UpstreamId, pub endpoints: Vec<Endpoint>, pub policy: LoadBalancingPolicy,
    pub health: HealthChecker, pub circuit: CircuitBreaker, pub conn_pool: ConnectionPool,
    pub tls: Option<UpstreamTlsConfig>,
}
pub enum LoadBalancingPolicy {
    RoundRobin, LeastConn, PowerOfTwoChoices, ConsistentHash { key: HashKeySource }, Random,
}
pub enum Principal {
    Anonymous, Basic { user: String },
    Jwt { sub: String, claims: serde_json::Value, kid: Option<String> },
    OAuth { sub: String, scopes: SmallVec<[String; 8]>, expires_at: SystemTime },
    Mtls { subject_dn: String, issuer_dn: String, fingerprint: [u8; 32] },
}
pub struct Config {
    pub version: ConfigVersion, pub listeners: Vec<ListenerConfig>,
    pub upstreams: HashMap<UpstreamId, UpstreamPoolConfig>, pub routes: Vec<Route>,
    pub middlewares: HashMap<MiddlewareId, Middleware>, pub schemas: HashMap<SchemaId, JsonSchema>,
    pub tls: TlsConfig, pub cache: CacheConfig, pub waf: WafConfig, pub auth: AuthConfig,
    pub edge: EdgeConfig, pub observe: ObserveConfig, pub control: ControlPlaneConfig,
}
pub enum RevpError {
    InvalidRequest(String),        // -> 400
    Auth(String),                  // -> 401 / 403 per config
    WafBlock { rule: String },     // -> 403 / 406 per config
    NoRoute,                       // -> 404
    NoHealthyUpstream,             // -> 503
    UpstreamTimeout,               // -> 504
    Upstream(String),              // -> 502
    Config(String),                // never surfaces to client
    Internal(String),              // -> 500
}
#[async_trait] pub trait Backend: Send + Sync + 'static {
    async fn serve(&self, req: Request) -> Result<Response, RevpError>;
    fn id(&self) -> BackendId; fn kind(&self) -> BackendKind;
}
#[async_trait] pub trait Middleware: Send + Sync + 'static {
    async fn on_request(&self, req: &mut Request) -> Result<NextAction, RevpError>;
    async fn on_response(&self, req: &Request, resp: &mut Response) -> Result<(), RevpError>;
    fn name(&self) -> &str;
}
pub enum NextAction { Continue, ShortCircuit(Response) }
pub trait CertResolver: Send + Sync + 'static {
    fn resolve(&self, sni: &str) -> Option<Arc<CertifiedKey>>; fn list(&self) -> Vec<String>;
}
#[async_trait] pub trait CacheStore: Send + Sync + 'static {
    async fn get(&self, key: &CacheKey) -> Option<Arc<CacheEntry>>;
    async fn put(&self, entry: Arc<CacheEntry>) -> Result<(), CacheError>;
    async fn invalidate(&self, key: &CacheKey) -> Result<(), CacheError>;
    async fn invalidate_by_tag(&self, tag: &SurrogateKey) -> Result<u64, CacheError>;
}
#[async_trait] pub trait Authenticator: Send + Sync + 'static {
    async fn authenticate(&self, req: &Request) -> Result<Principal, AuthError>;
    fn scheme(&self) -> AuthScheme;
}
pub trait WafRule: Send + Sync + 'static {
    fn evaluate(&self, req: &Request) -> WafVerdict; fn id(&self) -> WafRuleId;
}
pub enum WafVerdict { Pass, Match { score: u32, reason: String } }
#[async_trait] pub trait EdgeRuntime: Send + Sync + 'static {
    async fn run(&self, ctx: &mut EdgeContext) -> Result<EdgeAction, EdgeError>;
    fn budget(&self) -> &EdgeBudget;
}
#[async_trait] pub trait ConfigSource: Send + Sync + 'static {
    async fn load(&self) -> Result<Config, ConfigError>;
    async fn watch(&self) -> BoxStream<'static, Result<Config, ConfigError>>;
}
