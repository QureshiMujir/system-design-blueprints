Core schema & types (§11)

// parquet_core::schema
pub enum PhysicalType { Boolean, Int32, Int64, Int96, Float, Double, ByteArray, FixedLenByteArray(i32) }
pub enum LogicalType { String, Decimal{precision:i32,scale:i32}, Date, Time{unit:TimeUnit,utc:bool},
    Timestamp{unit:TimeUnit,utc:bool}, List, Map, Enum, Uuid, Json, Bson, Unknown }
pub enum Encoding { Plain, PlainDictionary, Rle, BitPacked, DeltaBinaryPacked,
    DeltaLengthByteArray, DeltaByteArray, RleDictionary, ByteStreamSplit }
pub enum CompressionCodec { Uncompressed, Snappy, Gzip, Lzo, Brotli, Lz4, Zstd, Lz4Raw }
// parquet_core::schema
pub struct ColumnDescriptor {
    pub path: Vec<String>,        // dot-separated path from root
    pub physical_type: PhysicalType,
    pub logical_type: Option<LogicalType>,
    pub max_rep: u8,              // precomputed from schema tree
    pub max_def: u8,              // precomputed from schema tree
    pub field_id: Option<i32>,
}

pub enum SchemaNode {
    Primitive { descriptor: ColumnDescriptor },
    Group { name: String, repetition: Repetition, children: Vec<SchemaNode> },
}
pub enum Repetition { Required, Optional, Repeated }
// parquet_core::footer
pub struct FileMetaData {
    pub version: i32,
    pub schema_root: SchemaNode,
    pub num_rows: i64,
    pub row_groups: Vec<RowGroupDescriptor>,
    pub key_value_metadata: Vec<(String, String)>,
    pub created_by: Option<String>,
}

pub struct RowGroupDescriptor {
    pub ordinal: usize,
    pub num_rows: i64,
    pub total_byte_size: i64,
    pub file_offset: i64,
    pub columns: Vec<ColumnChunkDescriptor>,
}

pub struct ColumnChunkDescriptor {
    pub path: Vec<String>,
    pub codec: CompressionCodec,
    pub encodings: Vec<Encoding>,
    pub num_values: i64,
    pub total_compressed_size: i64,
    pub total_uncompressed_size: i64,
    pub data_page_offset: i64,
    pub dict_page_offset: Option<i64>,
    pub statistics: Option<ColumnStatistics>,
    pub column_index_offset: Option<i64>,
    pub column_index_length: Option<i32>,
    pub offset_index_offset: Option<i64>,
    pub offset_index_length: Option<i32>,
    pub bloom_filter_offset: Option<i64>,
    pub bloom_filter_length: Option<i32>,
}
// parquet_core::page
pub enum PageType { DataPageV1, DataPageV2, DictionaryPage, IndexPage }

pub struct DataPageHeaderV1 {
    pub num_values: i32,
    pub encoding: Encoding,
    pub definition_level_encoding: Encoding,
    pub repetition_level_encoding: Encoding,
    pub statistics: Option<PageStatistics>,
}

pub struct DataPageHeaderV2 {
    pub num_values: i32,
    pub num_nulls: i32,
    pub num_rows: i32,
    pub encoding: Encoding,
    pub definition_levels_byte_length: i32,
    pub repetition_levels_byte_length: i32,
    pub is_compressed: bool,
    pub statistics: Option<PageStatistics>,
    pub crc: Option<i32>,
}

pub struct DictionaryPageHeader {
    pub num_values: i32,
    pub encoding: Encoding,   // PLAIN or PLAIN_DICTIONARY
    pub is_sorted: bool,
}

pub struct RawPage<'a> {
    pub header: PageHeader,
    pub compressed_data: &'a [u8],      // borrowed from I/O buffer
    pub uncompressed_size: usize,
}
// parquet_core::dremel
pub struct LevelArrays {
    pub rep: Vec<u8>,    // repetition levels; len = num_values
    pub def: Vec<u8>,    // definition levels; len = num_values
}

pub struct DecodedColumn<T> {
    pub levels: LevelArrays,
    pub values: Vec<T>,          // len ≤ num_values (non-null only)
    pub null_count: usize,
}
// parquet_codec
pub struct ParquetReader<R: Read + Seek> {
    inner: R,
    metadata: Arc<FileMetaData>,
    config: ReaderConfig,
}

impl<R: Read + Seek> ParquetReader<R> {
    pub fn open(reader: R, config: ReaderConfig) -> Result<Self, Error>;
    pub fn metadata(&self) -> &FileMetaData;
    pub fn row_group(&mut self, idx: usize) -> Result<RowGroupReader<'_, R>, Error>;
}

pub struct RowGroupReader<'a, R> {
    fn column(&mut self, path: &[&str]) -> Result<ColumnChunkReader<'_, R>, Error>;
    fn with_predicate(self, pred: PredicatePushdown) -> Self;
}

pub struct ColumnChunkReader<'a, R> {
    fn next_page(&mut self) -> Result<Option<DecodedPage<'_>>, Error>;
    fn collect<T: ParquetNative>(self) -> Result<DecodedColumn<T>, Error>;
}

pub struct ParquetWriter<W: Write> {
    inner: W,
    schema: Arc<SchemaNode>,
    config: WriterConfig,
}

impl<W: Write> ParquetWriter<W> {
    pub fn new(writer: W, schema: SchemaNode, config: WriterConfig) -> Result<Self, Error>;
    pub fn start_row_group(&mut self) -> Result<RowGroupWriter<'_, W>, Error>;
    pub fn close(self) -> Result<FileMetaData, Error>;  // writes footer; returns metadata
}

pub struct RowGroupWriter<'a, W> {
    fn column(&mut self, path: &[&str]) -> Result<ColumnWriter<'_, W>, Error>;
    fn close(self) -> Result<RowGroupDescriptor, Error>;
}
// parquet_codec::predicate
pub enum PredicatePushdown {
    Eq(Value), Lt(Value), Le(Value), Gt(Value), Ge(Value),
    IsNull, IsNotNull, In(Vec<Value>), And(Box<Self>, Box<Self>), Or(Box<Self>, Box<Self>),
}

// parquet_core::statistics
pub struct ColumnStatistics {
    pub null_count: Option<i64>,
    pub distinct_count: Option<i64>,
    pub min_value: Option<Vec<u8>>,    // raw bytes; physical type dependent
    pub max_value: Option<Vec<u8>>,
    pub is_min_value_exact: bool,
    pub is_max_value_exact: bool,
}
// parquet_core::error
pub enum Error {
    NotParquet,
    FooterTooLarge { len: u64, file_size: u64 },
    ThriftDecode { field_id: Option<i16>, offset: u64, source: Box<dyn std::error::Error> },
    OffsetOutOfBounds { offset: i64, file_size: u64 },
    PageSizeMismatch { expected: i64, actual: i64, column_path: String },
    DictIndexOutOfRange { index: u32, dict_size: u32 },
    InvalidLevel { level: u8, max: u8, column_path: String },
    Corruption { page_offset: u64, column_path: String },
    DecompressionBomb { claimed: u64, limit: u64 },
    SchemaMismatch { column: String, expected: PhysicalType, found: PhysicalType },
    MalformedPage { reason: &'static str, offset: u64 },
    UnsupportedEncoding(Encoding),
    UnsupportedCodec(CompressionCodec),
    Io(std::io::Error),
}
// orc_codec
pub struct OrcReader<R: Read + Seek> { inner: R, footer: OrcFileFooter, config: OrcReaderConfig }
pub struct OrcFileFooter { pub header_length: u64, pub content_length: u64,
    pub stripes: Vec<StripeInformation>, pub types: Vec<OrcType>, pub num_rows: u64 }
pub struct StripeInformation { pub offset: u64, pub index_length: u64,
    pub data_length: u64, pub footer_length: u64, pub num_rows: u64 }
pub enum OrcType { Boolean, TinyInt, SmallInt, Int, BigInt, Float, Double,
    String, Binary, Timestamp, List(Box<OrcType>), Map(Box<OrcType>,Box<OrcType>),
    Struct(Vec<(String,OrcType)>), Union(Vec<OrcType>), Decimal{precision:u32,scale:u32},
    Date, Varchar(u32), Char(u32) }
pub enum OrcStreamKind { Present, Data, Length, DictionaryData, RowIndex, BloomFilterUtf8 }
